$ErrorActionPreference = 'Stop'

if (-not (Test-Path $csc)) { throw "csc.exe not found at $csc" }

New-Item -ItemType Directory -Force bin | Out-Null

$opts = @(
    '/nologo', '/target:exe', '/platform:x64', '/optimize+',
    '/out:bin\OrchardObliterator.exe',
    '/resource:orchard.sig,orchard.sig',
    '/r:System.Management.dll', '/r:System.Drawing.dll'
)
if (Test-Path app.ico) { $opts += '/win32icon:app.ico' }

& $csc @opts (Get-ChildItem src\*.cs | ForEach-Object FullName)
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
Get-Item bin\OrchardObliterator.exe | Select-Object Name, Length, LastWriteTime

                if (line.Length == 0 || line[0] == '#') continue;
                var parts = line.Split(new[] { ' ' }, 3);
                string rest = parts.Length > 2 ? parts[2] : "";
                switch (parts[0])
                {
                    case "file": KnownFiles[parts[1]] = rest; break;
                    case "token": NameTokens.Add(parts[1]); break;
                    case "asset": Assets[parts[1]] = rest; break;
                    case "hook": Footprint.Add(parts[1]); break;
                    case "stealth": StealthHooks[parts[1]] = rest; break;
                    case "combat": CombatHooks[parts[1]] = rest; break;
                    case "watermark": Watermarks.Add(Unescape(line.Substring(10))); break;
                    case "loader": LoaderApis.Add(parts[1]); break;
                    case "log": LogPatterns.Add(line.Substring(4)); break;
                    case "host": TrustedHosts.Add(parts[1]); break;
                    case "mem":
                        var bits = line.Split(new[] { ' ' }, 4);
                        string text = Unescape(bits[3]);
                        Memory.Add(new MemorySignature
                        {
                            Kind = bits[1],
                            Weight = int.Parse(bits[2]),
                            Text = text,
                            Bytes = Encoding.UTF8.GetBytes(text)
                        });
                        break;
                }
            }
        }

        static string Unescape(string s)
        {
            var sb = new StringBuilder();
            for (int i = 0; i < s.Length; i++)
            {
                if (s[i] == '\\' && i + 5 < s.Length && s[i + 1] == 'u')
                {
                    sb.Append((char)Convert.ToInt32(s.Substring(i + 2, 4), 16));
                    i += 5;
                }
                else sb.Append(s[i]);
            }
            return sb.ToString();
        }
    }

                        if (!string.IsNullOrEmpty(gp.GameDir)) Watch(gp.GameDir);
                        if (QuickCheck(gp, cache, probe)) { pending.Remove(gp.Pid); continue; }
                    }
                    if ((DateTime.Now - pending[gp.Pid]).TotalSeconds < 20) continue;
                    pending.Remove(gp.Pid);

                    Hunter.Inspect(gp);
                    if (gp.Memory.Level == Level.Confirmed)
                    {
                        Report.Evidence(gp.Memory);
                        Terminate(gp, "Orchard classes found in memory");
                    }
                    else
                    {
                        Report.Good(string.Format("pid {0} memory clean ({1:N0} MB read)", gp.Pid, gp.Scanned >> 20));
                    }
                }
            }
            foreach (var w in watchers) w.Dispose();

        static bool QuickCheck(GameProcess gp, Dictionary<string, Evidence> cache, Func<string, Evidence> probe)
        {
            var jars = new List<string>(gp.ArgJars);
            if (!string.IsNullOrEmpty(gp.GameDir) && Directory.Exists(Path.Combine(gp.GameDir, "mods")))
                jars.AddRange(Locations.Walk(Path.Combine(gp.GameDir, "mods"), 4));

            var bad = new List<string>();
            foreach (var j in jars.Distinct(StringComparer.OrdinalIgnoreCase))
            {
                if (!File.Exists(j)) continue;
                cache.Remove(j);
                if (probe(j).Level == Level.Confirmed) bad.Add(j);
            }
            if (bad.Count == 0) return false;
            Terminate(gp, "launching with " + bad[0]);
            foreach (var b in bad) Quarantine.Lock(b);
            return true;
        }

        static void Terminate(GameProcess gp, string why)
        {
            if (Hunter.Kill(gp.Pid)) Report.Bad(string.Format("{0:HH:mm:ss} killed {1} pid {2}: {3}", DateTime.Now, gp.Name, gp.Pid, why));
            else Report.Warn(string.Format("could not kill pid {0} ({1})", gp.Pid, why));
        }

        {

                w.Created += touch;
                w.Changed += touch;
                w.Renamed += (s, e) => queue[e.FullPath] = DateTime.Now;
                w.EnableRaisingEvents = true;
                watchers.Add(w);
            }
            catch
            {
            }
        }
    }

    static class Handles
    {
        const int SystemExtendedHandleInformation = 64;
        const uint LengthMismatch = 0xC0000004;
        const uint PROCESS_DUP_HANDLE = 0x0040;
        const uint DUPLICATE_SAME_ACCESS = 2;
        const int FILE_TYPE_DISK = 1;
        const int Entry = 40;

        [DllImport("ntdll.dll")]
        static extern uint NtQuerySystemInformation(int cls, IntPtr buffer, int length, out int returned);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool DuplicateHandle(IntPtr sourceProcess, IntPtr source, IntPtr targetProcess, out IntPtr target, uint access, bool inherit, uint options);

        [DllImport("kernel32.dll")]
        static extern IntPtr GetCurrentProcess();

        [DllImport("kernel32.dll")]
        static extern int GetFileType(IntPtr file);

        [DllImport("kernel32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
        static extern int GetFinalPathNameByHandle(IntPtr file, StringBuilder path, int size, int flags);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool GetFileSizeEx(IntPtr file, out long size);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool SetFilePointerEx(IntPtr file, long distance, out long position, int method);

        [DllImport("kernel32.dll", SetLastError = true)]
        static extern bool ReadFile(IntPtr file, byte[] buffer, int count, out int read, ref NativeOverlapped overlapped);

        public static List<OpenFile> Archives(int pid, long maxSize)
        {
            var result = new List<OpenFile>();
            IntPtr proc = Native.OpenProcess(PROCESS_DUP_HANDLE, false, pid);
            if (proc == IntPtr.Zero) return result;

            IntPtr table = IntPtr.Zero;
            try
            {
                int size = 1 << 22, needed;
                while (true)
                {
                    table = Marshal.AllocHGlobal(size);
                    uint status = NtQuerySystemInformation(SystemExtendedHandleInformation, table, size, out needed);
                    if (status == 0) break;
                    Marshal.FreeHGlobal(table);
                    table = IntPtr.Zero;
                    if (status != LengthMismatch || size > 1 << 28) return result;
                    size = Math.Max(size * 2, needed + 65536);
                }

                long count = Marshal.ReadIntPtr(table).ToInt64();
                var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
                for (long i = 0; i < count; i++)
                {
                    IntPtr row = table + 16 + (int)(i * Entry);
                    if (Marshal.ReadIntPtr(row, 8).ToInt64() != pid) continue;
                    uint access = (uint)Marshal.ReadInt32(row, 24);
                    if ((access & 1) == 0) continue;

                    IntPtr dup;
                    if (!DuplicateHandle(proc, Marshal.ReadIntPtr(row, 16), GetCurrentProcess(), out dup, 0, false, DUPLICATE_SAME_ACCESS)) continue;
                    try
                    {
                        if (GetFileType(dup) != FILE_TYPE_DISK) continue;
                        var sb = new StringBuilder(1024);
                        if (GetFinalPathNameByHandle(dup, sb, sb.Capacity, 0) <= 0) continue;
                        string path = sb.ToString();
                        if (path.StartsWith(@"\\?\")) path = path.Substring(4);
                        if (!seen.Add(path)) continue;

                        bool gone = path.Contains(@"\$Extend\$Deleted\") || !File.Exists(path);
                        if (!gone) continue;

                        long length;
                        if (!GetFileSizeEx(dup, out length) || length < 64 || length > maxSize) continue;
                        var data = ReadAll(dup, length);
                        if (data == null || data[0] != 'P' || data[1] != 'K') continue;
                        result.Add(new OpenFile { Path = path, Gone = true, Data = data });
                    }
                    finally
                    {
                        Native.CloseHandle(dup);
                    }
                }
            }
            catch
            {
            }
            finally
            {
                if (table != IntPtr.Zero) Marshal.FreeHGlobal(table);
                Native.CloseHandle(proc);
            }
            return result;
        }

        static byte[] ReadAll(IntPtr file, long length)
        {
            long before;
            bool restore = SetFilePointerEx(file, 0, out before, 1);
            try
            {
                var data = new byte[length];
                long off = 0;
                var chunk = new byte[1 << 20];
                while (off < length)
                {
                    var ov = new NativeOverlapped { OffsetLow = (int)(off & 0xFFFFFFFF), OffsetHigh = (int)(off >> 32) };
                    int want = (int)Math.Min(chunk.Length, length - off), got;
                    if (!ReadFile(file, chunk, want, out got, ref ov) || got <= 0) return null;
                    Buffer.BlockCopy(chunk, 0, data, (int)off, got);
                    off += got;
                }
                return data;
            }
            finally
            {
                long ignored;
                if (restore) SetFilePointerEx(file, before, out ignored, 0);
            }
        }
    }

        static void ParseCommandLine(GameProcess gp)
        {
            var args = Split(gp.CommandLine);

                else if (a.StartsWith("-Dfabric.addMods=", StringComparison.OrdinalIgnoreCase))
                {

                    foreach (var p in a.Substring(17).Split(';'))
                        if (p.Length > 0) gp.ArgJars.Add(p.TrimStart('@'));
                }
                else if (a.StartsWith("-javaagent:", StringComparison.OrdinalIgnoreCase))
                {
                    string jar = a.Substring(11);
                    int eq = jar.IndexOf('=');
                    if (eq > 0) jar = jar.Substring(0, eq);
                    gp.Flags.Add("java agent attached: " + jar);
                    gp.ArgJars.Add(jar);

                else if (a.StartsWith("-agentpath:", StringComparison.OrdinalIgnoreCase) || a.StartsWith("-agentlib:", StringComparison.OrdinalIgnoreCase))

                }
                else if (a == "-cp" || a == "-classpath" || a == "--class-path")
                {
                    foreach (var p in next.Split(';'))
                        if (p.EndsWith(".jar", StringComparison.OrdinalIgnoreCase)) gp.ArgJars.Add(p);
                }
                else if (a.StartsWith("-Dfabric.debug.", StringComparison.OrdinalIgnoreCase) || a.StartsWith("-Dmixin.", StringComparison.OrdinalIgnoreCase))

        static List<string> Split(string cmd)
        {
            var list = new List<string>();
            var sb = new StringBuilder();
            bool quoted = false;
            foreach (char c in cmd ?? "")
            {
                if (c == '"') { quoted = !quoted; continue; }
                if (c == ' ' && !quoted)
                {
                    if (sb.Length > 0) { list.Add(sb.ToString()); sb.Clear(); }
                    continue;
                }
                sb.Append(c);
            }
            if (sb.Length > 0) list.Add(sb.ToString());
            return list;

        public static void Inspect(GameProcess gp)
        {
            gp.Memory = new Evidence(string.Format("{0} pid {1} (live memory)", gp.Name, gp.Pid));
            Modules(gp);

            IntPtr h = Native.OpenProcess(Native.PROCESS_QUERY_INFORMATION | Native.PROCESS_VM_READ, false, gp.Pid);
            if (h == IntPtr.Zero)
            {
                gp.Denied = true;
                return;
            }

            var sigs = Fingerprints.Memory;
            var counts = new int[sigs.Count];
            var asciiJar = Encoding.ASCII.GetBytes(".jar");
            var asciiJarUp = Encoding.ASCII.GetBytes(".JAR");
            var wideJar = Encoding.Unicode.GetBytes(".jar");
            var buf = new byte[Chunk + Overlap];
            var window = new byte[Window * 2];
            var parsedAt = new HashSet<long>();
            var mixins = new Dictionary<string, ClassFile>(StringComparer.Ordinal);
            var sw = Stopwatch.StartNew();

            try
            {
                long addr = 0;
                Native.MemoryInfo mi;
                var miSize = (IntPtr)System.Runtime.InteropServices.Marshal.SizeOf(typeof(Native.MemoryInfo));
                while (Native.VirtualQueryEx(h, (IntPtr)addr, out mi, miSize) != IntPtr.Zero)
                {
                    long start = mi.BaseAddress.ToInt64();
                    long size = mi.RegionSize.ToInt64();
                    if (size <= 0) break;

                    bool readable = mi.State == Native.MEM_COMMIT && (mi.Protect & (Native.PAGE_NOACCESS | Native.PAGE_GUARD)) == 0 && mi.Protect != 0;
                    if (readable)
                    {
                        bool exec = (mi.Protect & 0xF0) != 0;
                        if (exec && mi.Type == 0x20000) CheckPe(h, start, gp);

                        for (long off = 0; off < size; off += Chunk)
                        {
                            int want = (int)Math.Min(Chunk + Overlap, size - off);
                            IntPtr read;
                            if (!Native.ReadProcessMemory(h, (IntPtr)(start + off), buf, (IntPtr)want, out read)) continue;
                            int n = read.ToInt32();
                            if (n <= 0) continue;
                            gp.Scanned += Math.Min(n, Chunk);

                            for (int s = 0; s < sigs.Count; s++)
                                counts[s] += Count(buf, n, sigs[s].Bytes);

                            if (mixins.Count < MaxMixins)
                                CarveMixins(h, buf, n, start + off, window, parsedAt, mixins);

                            if (gp.CarvedJars.Count < MaxCarved)
                            {
                                CarveAscii(buf, n, asciiJar, gp.CarvedJars);
                                CarveAscii(buf, n, asciiJarUp, gp.CarvedJars);
                                CarveWide(buf, n, wideJar, gp.CarvedJars);
                            }
                        }
                        if (sw.ElapsedMilliseconds > 250)
                        {
                            Report.Progress(string.Format("  pid {0}: {1:N0} MB scanned", gp.Pid, gp.Scanned >> 20));
                            sw.Restart();
                        }
                    }

                    long nextAddr = start + size;
                    if (nextAddr <= addr) break;
                    addr = nextAddr;
                }
            }
            finally
            {
                Native.CloseHandle(h);
                Report.ClearProgress();
            }

            gp.MixinClasses = mixins.Count;
            MixinFootprint(gp, mixins.Values);

            foreach (var open in Handles.Archives(gp.Pid, 256L * 1024 * 1024))
            {
                var ev = JarScanner.Inspect(open.Data, open.Path, Path.GetExtension(open.Path), 0);
                if (ev.Level != Level.Confirmed) continue;
                gp.DeletedOpen.Add(open.Path);
                gp.Memory.Absorb(ev, "deleted but still open " + open.Path + ": ");
            }

            for (int s = 0; s < sigs.Count; s++)
            {
                if (counts[s] == 0) continue;
                gp.Memory.Add(sigs[s].Kind, sigs[s].Weight, string.Format("\"{0}\" resident in memory x{1:N0}", sigs[s].Text, counts[s]));
            }
            foreach (var u in gp.UnbackedImages)
                gp.Memory.Add("injection", 30, u);
        }

        static void CarveMixins(IntPtr h, byte[] buf, int n, long baseAddr, byte[] window, HashSet<long> parsedAt, Dictionary<string, ClassFile> sink)
        {
            int i = 0, last = n - MixinTag.Length;
            while (i <= last && sink.Count < MaxMixins)
            {
                i = Array.IndexOf(buf, MixinTag[0], i, last - i + 1);
                if (i < 0) return;
                int k = 1;
                while (k < MixinTag.Length && buf[i + k] == MixinTag[k]) k++;
                if (k == MixinTag.Length)
                {
                    var cf = ClassAround(h, buf, n, i, baseAddr, window, parsedAt);
                    if (cf != null && cf.IsMixin && cf.Name != null && !sink.ContainsKey(cf.Name)) sink[cf.Name] = cf;
                    i += MixinTag.Length;
                    continue;
                }
                i++;
            }
        }

        static ClassFile ClassAround(IntPtr h, byte[] buf, int n, int hit, long baseAddr, byte[] window, HashSet<long> parsedAt)
        {
            int s = Magic(buf, hit, Math.Max(0, hit - Window));
            if (s >= 0)
            {
                if (!parsedAt.Add(baseAddr + s)) return null;
                var cf = ClassFile.Parse(buf, s);
                if (cf != null && cf.Complete && s + cf.Length <= n) return cf;
                return Remote(h, baseAddr + s, window);
            }
            if (hit >= Window) return null;

            long from = Math.Max(0, baseAddr + hit - Window);
            int rel = (int)(baseAddr + hit - from);
            if (ReadAt(h, from, window, Window) < rel) return null;
            s = Magic(window, rel, 0);
            if (s < 0 || !parsedAt.Add(from + s)) return null;
            return Remote(h, from + s, window);
        }

        static ClassFile Remote(IntPtr h, long addr, byte[] window)
        {
            int got = ReadAt(h, addr, window, window.Length);
            var cf = got > 10 ? ClassFile.Parse(window, 0) : null;
            return cf != null && cf.Complete && cf.Length <= got ? cf : null;
        }

        static int ReadAt(IntPtr h, long addr, byte[] into, int count)
        {
            IntPtr read;
            Native.ReadProcessMemory(h, (IntPtr)addr, into, (IntPtr)count, out read);
            return read.ToInt32();
        }

        static int Magic(byte[] b, int from, int floor)
        {
            for (int s = from - 8; s >= floor; s--)
            {
                if (b[s] != 0xCA || b[s + 1] != 0xFE || b[s + 2] != 0xBA || b[s + 3] != 0xBE) continue;
                int major = (b[s + 6] << 8) | b[s + 7];
                if (major >= 45 && major <= 90) return s;
            }
            return -1;
        }

        static void MixinFootprint(GameProcess gp, IEnumerable<ClassFile> mixins)
        {
            var groups = new Dictionary<string, HashSet<string>>(StringComparer.Ordinal);
            var members = new Dictionary<string, int>(StringComparer.Ordinal);
            foreach (var cf in mixins)
            {
                var parts = cf.Name.Split('/');
                for (int depth = 1; depth < parts.Length; depth++)
                {
                    if (depth == 1 && (Roots.Contains(parts[0]) || parts[0].Length < 4)) continue;
                    string pkg = string.Join("/", parts, 0, depth);
                    HashSet<string> set;
                    if (!groups.TryGetValue(pkg, out set))
                    {
                        groups[pkg] = set = new HashSet<string>(StringComparer.Ordinal);
                        members[pkg] = 0;
                    }
                    members[pkg]++;
                    foreach (var f in cf.Footprint()) set.Add(f);
                }
            }

            string best = null;
            int bestShared = 0;
            foreach (var g in groups)
            {
                int shared = g.Value.Count(f => Fingerprints.Footprint.Contains(f));
                if (shared > bestShared || (shared == bestShared && shared > 0 && g.Key.Length > best.Length))
                {
                    best = g.Key;

                    ev.Add("lang", 40, "Orchard keybind table (key.nsc.*) in " + item.Name);
            }
        }

        internal static void Footprint(Evidence ev, HashSet<string> footprint, int mixins)
        {
            if (footprint.Count == 0) return;
            var reference = Fingerprints.Footprint;
            int shared = footprint.Count(f => reference.Contains(f));
            double ratio = reference.Count == 0 ? 0 : (double)shared / reference.Count;

            int weight = ratio >= 0.70 ? 100 : ratio >= 0.50 ? 60 : 0;
            if (weight > 0)
                ev.Add("footprint", weight, string.Format("mixin footprint matches Orchard {0:P0} ({1}/{2} hooks, {3} mixins)", ratio, shared, reference.Count, mixins));

            foreach (var kv in Fingerprints.StealthHooks)
                if (footprint.Contains(kv.Key)) ev.Add("stealth", 15, kv.Value);
            foreach (var kv in Fingerprints.CombatHooks)
                if (footprint.Contains(kv.Key)) ev.Add("combat", 8, kv.Value);
        }

        static void IntegrityTable(Evidence ev, string name, byte[] table, Dictionary<string, bool> classHashes, int classCount)
        {
            int records = table.Length / 68, matched = 0;
            for (int i = 0; i < records; i++)
            {
                if (!IsHex(table, i * 68, 64)) return;
                if (classHashes.ContainsKey(Encoding.ASCII.GetString(table, i * 68, 64))) matched++;
            }
            if (matched >= 10)
                ev.Add("integrity", 70, string.Format("self-integrity table {0}: {1} of {2} classes pinned by SHA-256", name, matched, classCount));
            else
                ev.Add("integrity", 35, string.Format("integrity table layout in {0} ({1} records)", name, records));
        }

        public static IEnumerable<string> AssetKeys(string name, byte[] bytes, bool decode)
        {
            yield return "sha:" + Sha256(bytes);
            if (!decode) yield break;
            string lower = name.ToLowerInvariant();
            if (lower.EndsWith(".png"))
            {
                var px = PixelHash(bytes);
                if (px != null) yield return "px:" + px;
            }
            else if (lower.EndsWith(".fsh") || lower.EndsWith(".vsh") || lower.EndsWith(".glsl") || lower.EndsWith(".json"))
            {
                yield return "txt:" + NormalizedHash(bytes);
            }
        }

        static string NormalizedHash(byte[] bytes)
        {
            string text = Encoding.UTF8.GetString(bytes);
            text = Comments.Replace(text, "");
            text = HexToken.Replace(text, "#");
            text = Spaces.Replace(text, "");
            return Sha256(Encoding.UTF8.GetBytes(text));
        }

        static string PixelHash(byte[] png)
        {
            try
            {
                using (var ms = new MemoryStream(png))
                using (var src = new Bitmap(ms))
                using (var bmp = new Bitmap(src.Width, src.Height, PixelFormat.Format32bppArgb))
                {
                    using (var g = Graphics.FromImage(bmp))
                    {
                        g.CompositingMode = System.Drawing.Drawing2D.CompositingMode.SourceCopy;
                        g.DrawImage(src, 0, 0, src.Width, src.Height);
                    }
                    var rect = new Rectangle(0, 0, bmp.Width, bmp.Height);
                    var lk = bmp.LockBits(rect, ImageLockMode.ReadOnly, PixelFormat.Format32bppArgb);
                    try
                    {
                        int row = bmp.Width * 4;
                        var pixels = new byte[row * bmp.Height + 8];
                        BitConverter.GetBytes(bmp.Width).CopyTo(pixels, 0);
                        BitConverter.GetBytes(bmp.Height).CopyTo(pixels, 4);
                        for (int y = 0; y < bmp.Height; y++)
                            Marshal.Copy(lk.Scan0 + y * lk.Stride, pixels, 8 + y * row, row);
                        return Sha256(pixels);
                    }
                    finally
                    {
                        bmp.UnlockBits(lk);
                    }
                }
            }
            catch
            {
                return null;
            }
        }

        static bool IsRandomPath(string name)

        static bool IsHex(byte[] b, int start, int len)
        {
            if (start + len > b.Length) return false;
            for (int i = start; i < start + len; i++)
            {
                byte c = b[i];
                if (!((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f'))) return false;
            }
            return true;
        }

        static double Entropy(byte[] b)
        {
            var counts = new int[256];
            foreach (var x in b) counts[x]++;
            double e = 0;
            foreach (var c in counts)
            {
                if (c == 0) continue;
                double p = (double)c / b.Length;
                e -= p * Math.Log(p, 2);
            }
            return e;
        }

        public static int IndexOf(byte[] hay, byte[] needle, int start)
        {
            int last = hay.Length - needle.Length;
            byte first = needle[0];
            for (int i = start; i <= last; i++)
            {
                i = Array.IndexOf(hay, first, i, last - i + 1);
                if (i < 0) return -1;
                int k = 1;
                while (k < needle.Length && hay[i + k] == needle[k]) k++;
                if (k == needle.Length) return i;
            }
            return -1;
        }

        public static string Sha256(byte[] data)
        {
            using (var sha = SHA256.Create())
            {
                var h = sha.ComputeHash(data);
                var sb = new StringBuilder(64);
                foreach (var x in h) sb.Append(x.ToString("x2"));
                return sb.ToString();
            }
        }

        public static void DumpFingerprint(string path, TextWriter w)
        {
            var data = ReadShared(path);
            var zip = Zip.Open(data);
            if (zip == null)
            {
                w.WriteLine("not an archive");
                return;
            }
            w.WriteLine("file " + Sha256(data));
            var footprint = new SortedSet<string>(StringComparer.Ordinal);
            foreach (var item in zip.Items)
            {
                if (item.IsDirectory) continue;
                var bytes = zip.Read(item);
                if (bytes == null) continue;
                if (item.Name.EndsWith(".class"))
                {
                    var cf = ClassFile.Parse(bytes);
                    if (cf != null && cf.IsMixin) foreach (var f in cf.Footprint()) footprint.Add(f);
                    continue;
                }
                if (item.Name.StartsWith("META-INF/") || item.Name.EndsWith(".mod.json") || item.Name.EndsWith(".mixins.json")) continue;
                if (item.Name.EndsWith("ofl.txt") || item.Name.EndsWith("/lang/en_us.json") || item.Name.EndsWith("sounds.json")) continue;
                if (bytes.Length < 512) continue;
                string tail = item.Name.Substring(item.Name.LastIndexOf('/') + 1);
                foreach (var key in AssetKeys(item.Name, bytes, true))
                    w.WriteLine("asset " + key + " " + Path.GetExtension(tail).TrimStart('.') + ":" + tail.Substring(0, Math.Min(9, tail.Length)));
            }
            foreach (var f in footprint) w.WriteLine("hook " + f);
        }
    }

        public static IEnumerable<string> Walk(string root, int maxDepth)
        {
            var stack = new Stack<KeyValuePair<string, int>>();
            stack.Push(new KeyValuePair<string, int>(root, 0));
            while (stack.Count > 0)
            {
                var cur = stack.Pop();
                string dir = cur.Key;
                string probe = dir.ToLowerInvariant() + "\\";
                if (SkipDirs.Any(s => probe.Contains(s)) && cur.Value > 0) continue;

                string[] files = null, subs = null;
                try { files = Directory.GetFiles(dir); } catch { }
                try { subs = Directory.GetDirectories(dir); } catch { }

                if (files != null)
                    foreach (var f in files) yield return f;

                if (subs == null || cur.Value >= maxDepth) continue;
                foreach (var s in subs)
                {
                    try
                    {
                        var attr = File.GetAttributes(s);
                        if ((attr & FileAttributes.ReparsePoint) != 0)
                        {
                            ReparsePoints.Add(s);
                            continue;
                        }
                    }
                    catch
                    {
                        continue;
                    }
                    stack.Push(new KeyValuePair<string, int>(s, cur.Value + 1));
                }
            }
        }

        public static bool IsArchive(string path, long minSize, long maxSize)
        {
            try
            {
                using (var fs = new FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite | FileShare.Delete))
                {
                    if (fs.Length < minSize || fs.Length > maxSize) return false;
                    var head = new byte[4];
                    int h = fs.Read(head, 0, 4);
                    if (h == 4 && head[0] == 'P' && head[1] == 'K' && head[2] == 3 && head[3] == 4) return true;

                    int tailLen = (int)Math.Min(fs.Length, 1024);
                    var tail = new byte[tailLen];
                    fs.Seek(-tailLen, SeekOrigin.End);
                    int t = 0;
                    while (t < tailLen)
                    {
                        int got = fs.Read(tail, t, tailLen - t);
                        if (got <= 0) break;
                        t += got;
                    }
                    return Zip.LooksLikeZip(head, h, tail, t);
                }
            }
            catch
            {
                return false;
            }
        }

                if (!noMemory)
                {
                    Hunter.Inspect(gp);
                    if (gp.Denied) Report.Warn("could not open process memory (game runs elevated? rerun as admin)");
                    else Report.Dim(string.Format("{0:N0} MB of memory read, {1} mixin classes recovered, {2} jar paths carved", gp.Scanned >> 20, gp.MixinClasses, gp.CarvedJars.Count));
                    foreach (var d in gp.DeletedOpen) Report.Bad(string.Format("pid {0} still holds deleted jar {1}", gp.Pid, d));
                    if (verbose) foreach (var m in gp.ForeignModules) Report.Dim("non-JDK native module: " + m);
                    if (gp.Memory.Level == Level.Confirmed || (verbose && gp.Memory.Hits.Count > 0)) Report.Evidence(gp.Memory);
                }

                var candidates = new List<string>(gp.ArgJars);
                candidates.AddRange(gp.CarvedJars);
                if (!string.IsNullOrEmpty(gp.GameDir) && Directory.Exists(Path.Combine(gp.GameDir, "mods")))
                    candidates.AddRange(Locations.Walk(Path.Combine(gp.GameDir, "mods"), 4));

                foreach (var c in candidates.Distinct(StringComparer.OrdinalIgnoreCase))
                {
                    if (File.Exists(c))
                    {
                        var ev = Probe(c);
                        if (ev != null && ev.Level == Level.Confirmed)
                        {
                            gp.JarsInUse.Add(c);
                            Report.Bad(string.Format("jar in use by pid {0}: {1}", gp.Pid, c));
                        }
                    }
                    else if (LooksLikeModPath(c) && (verbose || c.ToLowerInvariant().Contains("orchard") || (gp.Memory != null && gp.Memory.Level == Level.Confirmed)))
                    {
                        missingLoaded.Add(c);
                        Report.Warn(string.Format("pid {0} has {1} in memory but the file is gone", gp.Pid, c));
                    }
                }
            }

            if (only)
            {
                Report.Section("Selected paths");
                ScanExtra();
                return;
            }

            Report.Section("Game folders");
            var spent = new Dictionary<string, Stopwatch>();
            Func<string, Stopwatch> clock = k =>
            {
                Stopwatch w;
                if (!spent.TryGetValue(k, out w)) spent[k] = w = new Stopwatch();
                return w;
            };
            foreach (var dir in gameDirs)
            {
                var whole = Stopwatch.StartNew();
                Report.Info(dir);
                foreach (var sub in new[] { "mods", ".fabric", "versions", "libraries", "config", "shaderpacks", "resourcepacks" })
                {
                    string d = Path.Combine(dir, sub);
                    if (!Directory.Exists(d)) continue;
                    clock(sub).Start();
                    SweepFolder(d, 12, sub == "mods" || sub == "config");
                    clock(sub).Stop();
                }
                clock("root").Start();
                foreach (var f in SafeFiles(dir)) { Traces.AlternateStreams(f); ProbeIfArchive(f); }
                clock("root").Stop();
                clock("logs").Start();
                Traces.Logs(dir);
                clock("logs").Stop();
                clock("launch configs").Start();
                Traces.LaunchConfigs(dir);
                clock("launch configs").Stop();
                if (verbose && whole.Elapsed.TotalSeconds > 5) Report.Dim(string.Format("{0:0.0}s in this folder", whole.Elapsed.TotalSeconds));
            }
            if (verbose)
                Report.Dim("time by step: " + string.Join(", ", spent.OrderByDescending(kv => kv.Value.Elapsed).Select(kv => string.Format("{0} {1:0.0}s", kv.Key, kv.Value.Elapsed.TotalSeconds))));
            if (verbose)
                foreach (var r in Locations.ReparsePoints.Distinct()) Report.Warn("folder is a junction/symlink (could point at a USB stick): " + r);

            Report.Section("Loose files");
            foreach (var dir in Locations.LooseDirs())
            {
                Report.Progress("  " + dir);
                SweepFolder(dir, dir.StartsWith(Path.GetTempPath().TrimEnd('\\'), StringComparison.OrdinalIgnoreCase) ? 5 : 3, false);
            }
            ScanExtra();
            if (deep)
            {
                foreach (var drive in Locations.FixedDrives())
                {
                    Report.Info("deep sweep " + drive);
                    SweepFolder(drive, 40, false);
                }
            }
            Report.ClearProgress();

            Report.Section("Traces left behind");
            Traces.RecycleBin();
            Traces.RecentItems();
            if (admin)
            {
                Traces.Prefetch();
                Traces.UsnJournal(days);
            }
            foreach (var f in Traces.ExtraFiles.Distinct(StringComparer.OrdinalIgnoreCase).ToList())
            {
                if (f.IndexOf(':', 2) > 0) ProbeStream(f);
                else Probe(f);
            }
            if (!Traces.Found.Any(t => t.Tier > Traces.Noise))
                Report.Good("no Orchard traces in logs, recycle bin, recent items" + (admin ? ", prefetch or USN journal" : ""));
            foreach (var t in Traces.Found.OrderByDescending(t => t.Tier))
            {
                if (t.Tier == Traces.Proof) Report.Bad(t.What);
                else if (t.Tier == Traces.Lead) Report.Warn(t.What + "  (name only, not proof)");
                else if (verbose) Report.Dim(t.What);
                else continue;
                Report.Dim(t.Where);
            }
        }

        static void ScanExtra()
        {
            foreach (var p in extraPaths)
            {
                if (File.Exists(p)) { Traces.AlternateStreams(p); Probe(p); }
                else SweepFolder(p, 30, true);
            }
            foreach (var f in Traces.ExtraFiles.Distinct(StringComparer.OrdinalIgnoreCase).ToList())
                if (f.IndexOf(':', 2) > 0) ProbeStream(f);
        }

        static bool LooksLikeModPath(string p)
        {
            string l = p.ToLowerInvariant();
            return l.Contains("\\mods\\") || l.Contains("orchard") || l.Contains("\\temp\\") || l.Contains("\\downloads\\") || l.Contains("\\desktop\\");
        }

        static IEnumerable<string> SafeFiles(string dir)
        {
            try { return Directory.GetFiles(dir); }
            catch { return new string[0]; }
        }

        static void SweepFolder(string root, int depth, bool streams)
        {
            var files = new List<string>();
            foreach (var f in Locations.Walk(root, depth))
            {
                if (files.Count % 200 == 0) Report.Progress("  " + f);
                if (streams) Traces.AlternateStreams(f);
                if (!probed.ContainsKey(f)) files.Add(f);
            }
            ProbeMany(files);
        }

        static void ProbeMany(List<string> files)
        {
            var results = new Evidence[files.Count];
            var options = new ParallelOptions { MaxDegreeOfParallelism = Math.Max(2, Math.Min(8, Environment.ProcessorCount)) };
            Parallel.For(0, files.Count, options, i =>
            {
                if (!Locations.IsArchive(files[i], 300, 400L * 1024 * 1024)) return;
                var sw = Stopwatch.StartNew();
                var ev = JarScanner.InspectFile(files[i]);
                if (verbose && sw.ElapsedMilliseconds > 1500)
                    ev.Notes.Add(string.Format("slow: {0:N0} ms", sw.ElapsedMilliseconds));
                results[i] = ev;
            });
            for (int i = 0; i < files.Count; i++)
            {
                if (results[i] == null || probed.ContainsKey(files[i])) continue;
                probed[files[i]] = results[i];
                Show(results[i]);
            }
        }

        static void ProbeIfArchive(string f)
        {
            if (probed.ContainsKey(f)) return;
            if (!Locations.IsArchive(f, 300, 400L * 1024 * 1024)) return;
            Probe(f);
        }

        static Evidence Probe(string path)
        {
            Evidence ev;
            if (probed.TryGetValue(path, out ev)) return ev;
            var sw = Stopwatch.StartNew();
            ev = JarScanner.InspectFile(path);
            probed[path] = ev;
            if (verbose && sw.ElapsedMilliseconds > 1500)
                ev.Notes.Add(string.Format("slow: {0:N0} ms", sw.ElapsedMilliseconds));
            Show(ev);
            return ev;
        }

        static void ProbeStream(string stream)
        {
            if (probed.ContainsKey(stream)) return;
            byte[] data = null;
            try { data = Traces.ReadStream(stream); } catch { }
            if (data == null) return;
            var ev = JarScanner.Inspect(data, stream, ".stream", 0);
            probed[stream] = ev;
            Show(ev);
        }

        static void Show(Evidence ev)
        {
            if (ev.Level != Level.Clean || verbose)
            {
                Report.ClearProgress();
                Report.Evidence(ev);
                if (ev.Level != Level.Clean)
                {
                    string zone = Traces.ZoneIdentifier(ev.Source);
                    if (zone != null) Report.Dim("downloaded from: " + (zone.Length > 220 ? zone.Substring(0, 220) + "..." : zone));
                }
            }
        }

        static int Summary()
        {
            var confirmed = probed.Values.Where(e => e.Level == Level.Confirmed).ToList();
            var liveHits = live.Where(g => g.Memory != null && g.Memory.Level == Level.Confirmed).ToList();
            var proof = Traces.Found.Where(t => t.Tier == Traces.Proof).ToList();
            var leads = Traces.Found.Where(t => t.Tier == Traces.Lead).ToList();

            Report.Section("Verdict");
            foreach (var g in liveHits) Report.Bad(string.Format("Orchard is running right now in {0} pid {1} (score {2})", g.Name, g.Pid, g.Memory.Score));
            foreach (var e in confirmed) Report.Bad("Orchard jar: " + e.Source);
            if (liveHits.Count > 0 && confirmed.Count == 0)
                Report.Bad("the running cheat has no jar on disk: it was loaded and then deleted, renamed or pulled from removable media");
            foreach (var m in missingLoaded) Report.Warn("loaded but missing: " + m);
            if (proof.Count > 0) Report.Bad(proof.Count + " log/crash entr" + (proof.Count == 1 ? "y proves" : "ies prove") + " Orchard ran on this PC");
            if (leads.Count > 0) Report.Warn(leads.Count + " name-only lead(s) above; filenames are not proof");

            Report.Info(string.Format("{0} archives checked, {1} java process(es) inspected", probed.Count, live.Count));
            Report.Info("");
            if (liveHits.Count > 0 || confirmed.Count > 0)
            {
                Report.Bad("RESULT: ORCHARD DETECTED");
                return 2;
            }
            if (proof.Count > 0)
            {
                Report.Bad("RESULT: ORCHARD WAS USED (no jar left on disk)");
                return 2;
            }
            Report.Good("RESULT: NO ORCHARD");
            return 0;
        }

        static void Obliterate()
        {
            Report.Section("Blocking");
            foreach (var g in live)
            {
                bool usesJar = g.JarsInUse.Count > 0;
                bool hot = g.Memory != null && g.Memory.Level == Level.Confirmed;
                if (!hot && !usesJar) continue;
                if (Hunter.Kill(g.Pid)) Report.Bad(string.Format("killed {0} pid {1}", g.Name, g.Pid));
                else Report.Warn(string.Format("could not kill pid {0}", g.Pid));
            }
            foreach (var e in probed.Values.Where(x => x.Level == Level.Confirmed).ToList())
                Quarantine.Lock(e.Source);
        }
    }

using System;
using System.IO;

namespace OrchardObliterator
{
    static class Quarantine
    {
        public static string Folder
        {
            get
            {
                string root = Environment.GetFolderPath(Environment.SpecialFolder.CommonApplicationData);
                string dir = Path.Combine(root, "OrchardObliterator", "quarantine");
                try
                {
                    Directory.CreateDirectory(dir);
                    return dir;
                }
                catch
                {
                    dir = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "OrchardObliterator", "quarantine");
                    Directory.CreateDirectory(dir);
                    return dir;
                }
            }
        }

        public static bool Lock(string path)
        {
            if (!File.Exists(path))
            {
                Report.Warn("cannot quarantine " + path + " (not a plain file)");
                return false;
            }

            string dir = Folder;
            string sha = "";
            try { sha = JarScanner.Sha256(JarScanner.ReadShared(path)); } catch { }
            string target = Path.Combine(dir, string.Format("{0:yyyyMMdd_HHmmss}_{1}.quarantined", DateTime.Now, Path.GetFileName(path)));

            for (int attempt = 0; attempt < 10; attempt++)
            {
                try
                {
                    File.Move(path, target);
                    File.SetAttributes(target, FileAttributes.ReadOnly);
                    File.AppendAllText(Path.Combine(dir, "manifest.txt"),
                        string.Format("{0:yyyy-MM-dd HH:mm:ss}\t{1}\t{2}\t{3}{4}", DateTime.Now, sha, path, target, Environment.NewLine));
                    Report.Bad("quarantined " + path);
                    Report.Dim("moved to " + target);
                    return true;
                }
                catch (IOException)
                {
                    System.Threading.Thread.Sleep(300);
                }
                catch (UnauthorizedAccessException)
                {
                    break;
                }
            }
            Report.Warn("could not move " + path + " (still locked or no permission)");
            return false;
        }
    }
}

using System;
using System.IO;
using System.Text;

namespace OrchardObliterator
{
    static class Report
    {
        static readonly StringBuilder log = new StringBuilder();
        static readonly object gate = new object();
        public static bool Quiet;

        static readonly System.Diagnostics.Stopwatch phase = System.Diagnostics.Stopwatch.StartNew();
        static string current;

        public static void Section(string title)
        {
            if (current != null && phase.Elapsed.TotalSeconds >= 0.5)
                Line(string.Format("     ({0} took {1:0.0}s)", current.ToLowerInvariant(), phase.Elapsed.TotalSeconds), ConsoleColor.DarkGray);
            current = title;
            phase.Restart();
            Line("", ConsoleColor.Gray);
            Line("== " + title + " " + new string('=', Math.Max(3, 66 - title.Length)), ConsoleColor.DarkCyan);
        }

        public static void Info(string text) { Line("   " + text, ConsoleColor.Gray); }
        public static void Dim(string text) { Line("     " + text, ConsoleColor.DarkGray); }
        public static void Good(string text) { Line(" + " + text, ConsoleColor.Green); }
        public static void Warn(string text) { Line(" ? " + text, ConsoleColor.Yellow); }
        public static void Bad(string text) { Line(" ! " + text, ConsoleColor.Red); }

        public static void Evidence(Evidence ev)
        {
            var level = ev.Level;
            string tag = level == Level.Confirmed ? "ORCHARD" : "clean";
            string head = string.Format("[{0} {1}] {2}", tag, ev.Score, ev.Source);
            if (level == Level.Confirmed) Bad(head);
            else Info(head);

            foreach (var n in ev.Notes) Dim(n);
            foreach (var h in ev.Hits) Dim(string.Format("{0,-9} +{1,-3} {2}", h.Kind, h.Weight, h.Detail));
        }

        public static void Progress(string text)
        {
            if (Quiet || Console.IsOutputRedirected) return;
            lock (gate)
            {
                int width = 78;
                try { width = Math.Max(20, Console.WindowWidth - 1); } catch { }
                if (text.Length > width) text = "..." + text.Substring(text.Length - width + 3);
                Console.ForegroundColor = ConsoleColor.DarkGray;
                Console.Write("\r" + text.PadRight(width) + "\r");
                Console.ResetColor();
            }
        }

        public static void ClearProgress()
        {
            Progress("");
        }

        static void Line(string text, ConsoleColor color)
        {
            lock (gate)
            {
                log.AppendLine(text);
                if (Quiet) return;
                Console.ForegroundColor = color;
                Console.WriteLine(text);
                Console.ResetColor();
            }
        }

        public static string Save(string path)
        {
            if (string.IsNullOrEmpty(path))
            {
                string name = string.Format("OrchardObliterator_{0}_{1:yyyyMMdd_HHmmss}.txt", Environment.MachineName, DateTime.Now);
                path = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, name);
            }
            try
            {
                File.WriteAllText(path, log.ToString(), Encoding.UTF8);
            }
            catch
            {
                path = Path.Combine(Path.GetTempPath(), Path.GetFileName(path));
                File.WriteAllText(path, log.ToString(), Encoding.UTF8);
            }
            return path;
        }
    }
}

using System;
using System.Collections.Generic;
using System.IO;
using System.IO.Compression;
using System.Linq;
using System.Runtime.InteropServices;
using System.Security.Principal;
using System.Text;
using System.Text.RegularExpressions;
using System.Threading.Tasks;
using Microsoft.Win32.SafeHandles;

namespace OrchardObliterator
{
    sealed class Trace
    {
        public string Where;
        public string What;
        public int Tier;
    }

    static class Traces
    {
        public static readonly List<Trace> Found = new List<Trace>();
        public static readonly List<string> ExtraFiles = new List<string>();

        static readonly Regex ModLine = new Regex(@"^\s*[-|\\]+\s*([a-z0-9_.\-]+)\s+(\S+)", RegexOptions.Compiled);
        static readonly Regex Hashy = new Regex(@"^q?[0-9a-f]{16,}$", RegexOptions.Compiled);

        public const int Noise = 0, Lead = 1, Proof = 2;

        static Regex logPattern;

        static void Note(string where, string what, int tier)
        {
            lock (Found)
            {
                if (Found.Any(t => t.Where == where && t.What == what)) return;
                Found.Add(new Trace { Where = where, What = what, Tier = tier });
            }
        }

        public static bool IsAdmin()
        {
            try { return new WindowsPrincipal(WindowsIdentity.GetCurrent()).IsInRole(WindowsBuiltInRole.Administrator); }
            catch { return false; }
        }

        public static void Logs(string gameDir)
        {
            var files = new List<string>();
            foreach (var sub in new[] { "logs", "crash-reports" })
            {
                string d = Path.Combine(gameDir, sub);
                if (!Directory.Exists(d)) continue;
                try { files.AddRange(Directory.GetFiles(d)); } catch { }
            }
            files.RemoveAll(f =>
            {
                string lower = f.ToLowerInvariant();
                return !(lower.EndsWith(".log") || lower.EndsWith(".log.gz") || lower.EndsWith(".txt"));
            });
            Parallel.ForEach(files, new ParallelOptions { MaxDegreeOfParallelism = 8 }, f =>
            {
                string text = ReadLog(f);
                if (text != null) ScanLogText(f, text);
            });
        }

        static string ReadLog(string path)
        {
            try
            {
                using (var fs = new FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite | FileShare.Delete))
                {
                    Stream s = fs;
                    if (path.EndsWith(".gz", StringComparison.OrdinalIgnoreCase)) s = new GZipStream(fs, CompressionMode.Decompress);
                    using (var r = new StreamReader(s, Encoding.UTF8))
                    {
                        var sb = new StringBuilder();
                        var buf = new char[65536];
                        int n;
                        while (sb.Length < 32 * 1024 * 1024 && (n = r.Read(buf, 0, buf.Length)) > 0)
                            sb.Append(buf, 0, n);
                        return sb.ToString();
                    }
                }
            }
            catch
            {
                return null;
            }
        }

        static void ScanLogText(string file, string text)
        {
            if (logPattern == null)
                logPattern = new Regex(string.Join("|", Fingerprints.LogPatterns.Select(Regex.Escape)), RegexOptions.IgnoreCase | RegexOptions.Compiled);
            bool patterns = Fingerprints.LogPatterns.Count > 0 && logPattern.IsMatch(text);
            if (!patterns && text.IndexOf(" mods:", StringComparison.Ordinal) < 0) return;
            var lines = text.Split('\n');
            bool inModList = false;
            for (int i = 0; i < lines.Length; i++)
            {
                string line = lines[i].TrimEnd('\r');
                if (line.Contains("Loading ") && line.Contains(" mods:")) { inModList = true; continue; }
                if (inModList)
                {
                    var m = ModLine.Match(line);
                    if (!m.Success) { inModList = false; }
                    else
                    {
                        string id = m.Groups[1].Value;
                        if (id == "orchard")
                            Note(file, string.Format("line {0}: mod list contains \"{1} {2}\"", i + 1, id, m.Groups[2].Value), Proof);
                        else if (id.Contains("orchard"))
                            Note(file, string.Format("line {0}: mod list contains \"{1} {2}\"", i + 1, id, m.Groups[2].Value), Lead);
                        else if (Hashy.IsMatch(id))
                            Note(file, string.Format("line {0}: mod with a hashed id \"{1}\"", i + 1, id), Noise);
                        continue;
                    }
                }
                if (patterns && logPattern.IsMatch(line))
                {
                    string snip = line.Trim();
                    if (snip.Length > 160) snip = snip.Substring(0, 160) + "...";
                    Note(file, string.Format("line {0}: {1}", i + 1, snip), Proof);
                }
            }
        }

        public static void LaunchConfigs(string gameDir)
        {
            var files = new List<string>();
            foreach (var name in new[] { "launcher_profiles.json", "launcher_profiles_microsoft_store.json", "instance.cfg", "mmc-pack.json", "profile.json", "minecraftinstance.json" })
            {
                string a = Path.Combine(gameDir, name);
                string b = Path.Combine(Path.GetDirectoryName(gameDir) ?? gameDir, name);
                if (File.Exists(a)) files.Add(a);
                if (File.Exists(b) && b != a) files.Add(b);
            }
            string versions = Path.Combine(gameDir, "versions");
            if (Directory.Exists(versions))
            {
                try { files.AddRange(Directory.GetFiles(versions, "*.json", SearchOption.AllDirectories)); } catch { }
            }

            foreach (var f in files)
            {
                string text;
                try { text = File.ReadAllText(f); } catch { continue; }
                foreach (Match m in Regex.Matches(text, @"(-javaagent:[^""\s,]+|-agentpath:[^""\s,]+|-Dfabric\.addMods=[^""\s,]+)"))
                    Note(f, "launch argument " + m.Value, Noise);
                foreach (Match m in Regex.Matches(text, @"""url""\s*:\s*""(https?://[^""/]+)"))
                {
                    string host = m.Groups[1].Value.ToLowerInvariant();
                    if (!Fingerprints.TrustedHosts.Any(h => host.EndsWith(h)))
                        Note(f, "library pulled from unusual host " + host, Noise);
                }
            }
        }

        public static void RecycleBin()
        {
            string sid = WindowsIdentity.GetCurrent().User.Value;
            foreach (var root in Locations.FixedDrives())
            {
                string bin = Path.Combine(root, "$Recycle.Bin", sid);
                if (!Directory.Exists(bin)) continue;
                string[] infos;
                try { infos = Directory.GetFiles(bin, "$I*"); } catch { continue; }
                foreach (var info in infos)
                {
                    string original = ParseRecycleInfo(info);
                    if (original == null) continue;
                    string data = Path.Combine(bin, "$R" + Path.GetFileName(info).Substring(2));
                    bool jarish = original.EndsWith(".jar", StringComparison.OrdinalIgnoreCase) || original.ToLowerInvariant().Contains("orchard");
                    if (File.Exists(data) && Locations.IsArchive(data, 512, 400L * 1024 * 1024))
                    {
                        ExtraFiles.Add(data);
                        if (jarish) Note(data, "deleted archive waiting in the Recycle Bin, was " + original, Noise);
                    }
                    else if (jarish)
                    {
                        Note(info, "Recycle Bin record for " + original, original.ToLowerInvariant().Contains("orchard") ? Lead : Noise);
                    }
                }
            }
        }

        static string ParseRecycleInfo(string path)
        {
            try
            {
                var b = File.ReadAllBytes(path);
                long version = BitConverter.ToInt64(b, 0);
                if (version == 2)
                {
                    int chars = BitConverter.ToInt32(b, 24);
                    return Encoding.Unicode.GetString(b, 28, Math.Max(0, (chars - 1) * 2));
                }
                if (version == 1) return Encoding.Unicode.GetString(b, 24, 520).TrimEnd('\0');
            }
            catch
            {
            }
            return null;
        }

        public static void RecentItems()
        {
            string recent = Environment.GetFolderPath(Environment.SpecialFolder.Recent);
            string[] links;
            try { links = Directory.GetFiles(recent, "*.lnk"); } catch { return; }
            var rx = new Regex(@"[A-Za-z]:\\[^\0""<>|*?]{2,240}?\.(jar|zip)", RegexOptions.IgnoreCase);
            foreach (var lnk in links)
            {
                string name = Path.GetFileName(lnk).ToLowerInvariant();
                byte[] b;
                try { b = File.ReadAllBytes(lnk); } catch { continue; }
                var targets = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
                foreach (Match m in rx.Matches(Encoding.GetEncoding(1252).GetString(b))) targets.Add(m.Value);
                foreach (Match m in rx.Matches(Encoding.Unicode.GetString(b))) targets.Add(m.Value);
                foreach (Match m in rx.Matches(Encoding.Unicode.GetString(b, 1, b.Length - 1))) targets.Add(m.Value);

                foreach (var t in targets)
                {
                    bool orchard = t.ToLowerInvariant().Contains("orchard") || name.Contains("orchard");
                    if (File.Exists(t))
                    {
                        if (t.EndsWith(".jar", StringComparison.OrdinalIgnoreCase)) ExtraFiles.Add(t);
                        if (orchard) Note(lnk, "recently opened " + t, Lead);
                    }

        public static void Prefetch()
        {

        static byte[] Decompress(string pf)
        {
            byte[] b;
            try { b = File.ReadAllBytes(pf); } catch { return null; }
            if (b.Length < 8) return null;
            if (b[0] == 'M' && b[1] == 'A' && b[2] == 'M')
            {
                int size = BitConverter.ToInt32(b, 4);
                uint ws, fs;
                if (Native.RtlGetCompressionWorkSpaceSize(4, out ws, out fs) != 0) return null;
                var work = new byte[ws];
                var input = new byte[b.Length - 8];
                Buffer.BlockCopy(b, 8, input, 0, input.Length);
                var output = new byte[size];
                int final;
                if (Native.RtlDecompressBufferEx(4, output, size, input, input.Length, out final, work) != 0) return null;
                return output;
            }
            return b;
        }

        public static void UsnJournal(int days)
        {
            DateTime since = DateTime.UtcNow.AddDays(-days);

        static void ReadJournal(SafeFileHandle h, string root, DateTime since)
        {
            var q = new byte[80];
            int got;
            if (!Native.DeviceIoControl(h, Native.FSCTL_QUERY_USN_JOURNAL, null, 0, q, q.Length, out got, IntPtr.Zero)) return;
            long journalId = BitConverter.ToInt64(q, 0);
            long next = BitConverter.ToInt64(q, 8);
            long end = BitConverter.ToInt64(q, 16);

            var input = new byte[40];
            var output = new byte[1 << 20];
            var pendingRename = new Dictionary<string, string>();
            int scanned = 0;

            while (next < end)
            {
                BitConverter.GetBytes(next).CopyTo(input, 0);
                BitConverter.GetBytes(0x3300u).CopyTo(input, 8);
                BitConverter.GetBytes(0).CopyTo(input, 12);
                BitConverter.GetBytes(0L).CopyTo(input, 16);
                BitConverter.GetBytes(0L).CopyTo(input, 24);
                BitConverter.GetBytes(journalId).CopyTo(input, 32);
                if (!Native.DeviceIoControl(h, Native.FSCTL_READ_USN_JOURNAL, input, input.Length, output, output.Length, out got, IntPtr.Zero)) break;
                if (got <= 8) break;
                long after = BitConverter.ToInt64(output, 0);

                int p = 8;
                while (p + 60 <= got)
                {
                    int len = BitConverter.ToInt32(output, p);
                    if (len <= 0 || p + len > got) break;
                    int major = BitConverter.ToUInt16(output, p + 4);
                    int refLen = major >= 3 ? 16 : 8;
                    int at = p + 8 + refLen * 2;
                    long stamp = BitConverter.ToInt64(output, at + 8);
                    uint reason = BitConverter.ToUInt32(output, at + 16);
                    int nameLen = BitConverter.ToUInt16(output, at + 32);
                    int nameOff = BitConverter.ToUInt16(output, at + 34);
                    string fileRef = BitConverter.ToString(output, p + 8, refLen);
                    string name = Encoding.Unicode.GetString(output, p + nameOff, nameLen);
                    scanned++;

                    DateTime when = DateTime.FromFileTimeUtc(stamp);
                    if (when >= since) Classify(root, fileRef, name, reason, when.ToLocalTime(), pendingRename);
                    p += len;
                }
                if (after <= next) break;
                next = after;
            }
            Report.Dim(string.Format("usn journal {0}: {1:N0} records read", root, scanned));
        }

        static void Classify(string root, string fileRef, string name, uint reason, DateTime when, Dictionary<string, string> pendingRename)
        {
            string lower = name.ToLowerInvariant();
            string where = "USN " + root;
            if ((reason & 0x1000) != 0) pendingRename[fileRef] = name;
            if ((reason & 0x2000) != 0)
            {
                string old;
                if (pendingRename.TryGetValue(fileRef, out old))
                {
                    pendingRename.Remove(fileRef);
                    bool wasJar = old.EndsWith(".jar", StringComparison.OrdinalIgnoreCase);
                    bool isJar = lower.EndsWith(".jar");
                    if (wasJar != isJar || old.ToLowerInvariant().Contains("orchard") || lower.Contains("orchard"))
                        Note(where, string.Format("{0:yyyy-MM-dd HH:mm:ss} renamed {1} -> {2}", when, old, name), old.ToLowerInvariant().Contains("orchard") || lower.Contains("orchard") ? Lead : Noise);
                }
            }
            if (lower.Contains("orchard") && (lower.EndsWith(".jar") || lower.EndsWith(".zip")))
            {
                string what = (reason & 0x200) != 0 ? "deleted" : (reason & 0x100) != 0 ? "created" : null;
                if (what != null) Note(where, string.Format("{0:yyyy-MM-dd HH:mm:ss} {1} {2}", when, what, name), Lead);
            }
        }
    }

        public static bool LooksLikeZip(byte[] head, int headLen, byte[] tail, int tailLen)
        {
            if (headLen >= 4 && head[0] == 'P' && head[1] == 'K' && head[2] == 3 && head[3] == 4)
                return true;
            for (int i = tailLen - 22; i >= 0; i--)
                if (tail[i] == 'P' && tail[i + 1] == 'K' && tail[i + 2] == 5 && tail[i + 3] == 6)
                    return true;
            return false;
        }

        public static Zip Open(byte[] data)
        {
            try { return Parse(data); }
            catch { return null; }
        }

        static Zip Parse(byte[] data)
        {
            int eocd = -1;
            int floor = Math.Max(0, data.Length - 22 - 65535);
            for (int i = data.Length - 22; i >= floor; i--)
            {
                if (data[i] == 'P' && data[i + 1] == 'K' && data[i + 2] == 5 && data[i + 3] == 6)
                {
                    eocd = i;
                    break;
                }
            }
            if (eocd < 0) return null;

            long count = U16(data, eocd + 10);
            long dirSize = U32(data, eocd + 12);
            long dirOffset = U32(data, eocd + 16);
            long anchor = eocd;

            if (dirOffset == 0xFFFFFFFF || count == 0xFFFF || dirSize == 0xFFFFFFFF)
            {
                int loc = eocd - 20;
                if (loc < 0 || U32(data, loc) != 0x07064b50) return null;
                int z = -1;
                for (long probe = loc - 56; probe >= 0 && probe >= loc - 56 - 4096; probe--)
                {
                    if (U32(data, (int)probe) == 0x06064b50) { z = (int)probe; break; }
                }
                if (z < 0) return null;
                count = (long)U64(data, z + 32);
                dirSize = (long)U64(data, z + 40);
                dirOffset = (long)U64(data, z + 48);
                anchor = z;
            }

            long dirStart = anchor - dirSize;
            if (dirStart < 0) return null;
            var zip = new Zip(data, dirStart - dirOffset);

            int p = (int)dirStart;
            for (long n = 0; n < count && p + 46 <= data.Length; n++)
            {
                if (U32(data, p) != 0x02014b50) break;
                int flags = U16(data, p + 8);
                int nameLen = U16(data, p + 28);
                int extraLen = U16(data, p + 30);
                int commentLen = U16(data, p + 32);
                if (p + 46 + nameLen > data.Length) break;

                var enc = (flags & 0x800) != 0 ? Encoding.UTF8 : Encoding.GetEncoding(437);
                var item = new ZipItem
                {
                    Flags = flags,
                    Method = U16(data, p + 10),
                    Packed = U32(data, p + 20),
                    Size = U32(data, p + 24),
                    Offset = U32(data, p + 42),
                    Name = enc.GetString(data, p + 46, nameLen).Replace('\\', '/')
                };
                zip.Items.Add(item);
                if (!zip.byName.ContainsKey(item.Name)) zip.byName[item.Name] = item;
                p += 46 + nameLen + extraLen + commentLen;
            }
            return zip.Items.Count == 0 ? null : zip;
        }

        {
            ZipItem item;
            return byName.TryGetValue(name, out item) ? item : null;

        public byte[] Read(ZipItem item)
        {
            if ((item.Flags & 1) != 0) return null;
            if (item.Packed == 0xFFFFFFFF || item.Size > MaxInflate) return null;
            long local = item.Offset + Shift;
            if (local < 0 || local + 30 > data.Length) return null;
            if (U32(data, (int)local) != 0x04034b50) return null;

            long start = local + 30 + U16(data, (int)local + 26) + U16(data, (int)local + 28);
            long packed = item.Packed;
            if (start + packed > data.Length) packed = data.Length - start;
            if (packed < 0) return null;

            try
            {
                if (item.Method == 0)
                {
                    var raw = new byte[packed];
                    Buffer.BlockCopy(data, (int)start, raw, 0, (int)packed);
                    return raw;
                }
                if (item.Method != 8) return null;

                using (var src = new MemoryStream(data, (int)start, (int)packed, false))
                using (var inflate = new DeflateStream(src, CompressionMode.Decompress))
                using (var dst = new MemoryStream(item.Size > 0 ? (int)item.Size : 4096))
                {
                    var buf = new byte[16384];
                    int got;
                    while ((got = inflate.Read(buf, 0, buf.Length)) > 0)
                    {
                        dst.Write(buf, 0, got);
                        if (dst.Length > MaxInflate) return null;
                    }
                    return dst.ToArray();
                }
            }
            catch
            {
                return null;
            }
        }

        public byte[] Peek(ZipItem item, int count)
        {
            if ((item.Flags & 1) != 0) return null;
            long local = item.Offset + Shift;
            if (local < 0 || local + 30 > data.Length || U32(data, (int)local) != 0x04034b50) return null;
            long start = local + 30 + U16(data, (int)local + 26) + U16(data, (int)local + 28);
            int avail = (int)Math.Min(item.Packed, data.Length - start);
            if (avail <= 0) return null;
            try
            {
                if (item.Method == 0)
                {
                    var raw = new byte[Math.Min(count, avail)];
                    Buffer.BlockCopy(data, (int)start, raw, 0, raw.Length);
                    return raw;
                }
                if (item.Method != 8) return null;
                using (var inflate = new DeflateStream(new MemoryStream(data, (int)start, avail, false), CompressionMode.Decompress))
                {
                    var buf = new byte[count];
                    int got = 0, n;
                    while (got < count && (n = inflate.Read(buf, got, count - got)) > 0) got += n;
                    if (got < count) Array.Resize(ref buf, got);
                    return buf;
                }
            }
            catch
            {
                return null;
            }
        }
