# florida-zygisk — Florida frida-server module

[![Build Status](https://img.shields.io/github/actions/workflow/status/thelok1s/florida-zygisk/build.yml?branch=main&label=Build)](actions/workflows/build.yml)
[![Latest Release](https://img.shields.io/github/v/release/thelok1s/florida-zygisk)](releases/latest)
[![Downloads](https://img.shields.io/github/downloads/thelok1s/florida-zygisk/total)](releases)

A Magisk / KSU / APatch module that automatically starts **[Florida](https://github.com/Ylarod/Florida)** —
an anti-detection build of frida-server — on boot.

Built on top of the **[magisk-frida](https://github.com/ViRb3/magisk-frida)** module template by ViRb3.

---

## About project

| Layer | Source |
|-------|--------|
| Module installer / service scripts | [ViRb3/magisk-frida](https://github.com/ViRb3/magisk-frida) template |
| frida-server binary (patched) | Built from [Ylarod/Florida](https://github.com/Ylarod/Florida) patches applied to [frida/frida](https://github.com/frida/frida) source |

Florida applies source-level patches to frida before compilation, making the server harder to detect
via common heuristics (port, process name, memory maps, etc.).

## How updates work

A GitHub Actions workflow runs **every week** and checks for a new frida release. When one is
detected it **builds Florida from source** and publishes a new module ZIP here.

> **Why build from source?**  
> Florida's upstream CI has been unreliable. By cloning their `patches/` directory and applying
> them ourselves we are independent of their release cadence.

## Requirements

* Architectures: `arm` · `arm64` · `x86` · `x86_64`
* Magisk ≥ 24 · KernelSU · APatch

## Installation

1. Download the latest `florida-zygisk-*.zip` from the [Releases](releases) page.
2. Flash via Magisk Manager / KSU / APatch.
3. Reboot. The Florida frida-server starts automatically at late-boot.

## Status & port

The module description shown in the module manager updates at runtime:

| State | Description field |
|-------|-------------------|
| Running | `Running on port <PORT> (PID=<PID>) - tap Action to toggle` |
| Stopped | `Stopped - tap Action to restart` |
| Error | `ERROR: <reason> - check log` |

You can also read the port from the log:

```bash
adb shell grep "frida-server started" /data/local/tmp/florida-zygisk.log | tail -1
# or read the description directly
adb shell grep "^description=" /data/adb/modules/florida-zygisk/module.prop
```

## Toggle (KSU Action button)

In KernelSU, the module manager shows an **Action** button next to florida-zygisk.
Tapping it toggles the server:

- **Running → tap →** server stops, description updates to `Stopped`
- **Stopped → tap →** server starts on a new random port, description updates with port & PID

```sh
}
```

```sh
    sed -i "s|^description=.*|description=$1|" "$MODDIR/module.prop"
}

# ── Detect architecture ───────────────────────────────────────────────────────
ABI=$(getprop ro.product.cpu.abi)
case "$ABI" in
    arm64-v8a) ARCH=arm64  ;;
    x86_64)    ARCH=x86_64 ;;
    x86*)      ARCH=x86    ;;
    *)         ARCH=arm    ;;
esac

PORT=$(( ( RANDOM % 64511 ) + 1024 ))

# ── Pre-flight ────────────────────────────────────────────────────────────────
if [ ! -f "$FRIDA_BIN" ]; then
    log "ERROR: binary not found for arch=$ARCH at $FRIDA_BIN"

    exit 1

sleep 3

nohup "$FRIDA_BIN" --listen "0.0.0.0:$PORT" >> "$LOG" 2>&1 &
SERVER_PID=$!

sleep 2
if kill -0 "$SERVER_PID" 2>/dev/null; then
    log "frida-server started (PID=$SERVER_PID, port=$PORT)"
    update_desc "Running on port $PORT (PID=$SERVER_PID) - tap Action to toggle"
else
    log "ERROR: frida-server exited immediately – check $LOG"
```

`build.py`:

```py
#!/usr/bin/env python3

import gzip
import json
import logging
import os
import re
import shutil
import zipfile
from pathlib import Path

# ── Paths ─────────────────────────────────────────────────────────────────────

PATH_BASE = Path(__file__).parent.resolve()
PATH_BASE_MODULE: Path = PATH_BASE / "base"
PATH_BUILD: Path = PATH_BASE / "build"
PATH_BUILD_TMP: Path = PATH_BUILD / "tmp"
PATH_DOWNLOADS: Path = PATH_BASE / "downloads"

# ── Logging ───────────────────────────────────────────────────────────────────

logging.basicConfig(
    level=logging.INFO,
    format="%(levelname)s  %(message)s",
)
log = logging.getLogger(__name__)

# ── Helpers ───────────────────────────────────────────────────────────────────

# Architecture name mapping:
#   Florida / frida build arch  →  frida release filename arch
ARCH_MAP = {
    "android-arm": "arm",
    "android-arm64": "arm64",
    "android-x86": "x86",
    "android-x86_64": "x86_64",
}


def generate_version_code(tag: str) -> int:
    """Convert a tag like '17.9.6' or '17.9.6-1' to an integer version code."""
    parts = re.split(r"[-.]", tag)
    return int("".join(f"{int(p):02d}" for p in parts))


def write_module_prop(
    mod_dir: Path, frida_version: str, release_tag: str, github_repo: str
) -> None:
    content = (
        f"id=florida-zygisk\n"
        f"name=florida-zygisk (Florida)\n"
        f"version={release_tag}\n"
        f"versionCode={generate_version_code(release_tag)}\n"
        f"author=lok1s | Credits: Ylarod (Florida), ViRb3 (magisk-frida)\n"
        f"updateJson=https://github.com/{github_repo}/releases/latest/download/updater.json\n"
        f"description=Florida frida-server {frida_version} on boot - tap Action to toggle\n"
    )
    (mod_dir / "module.prop").write_text(content, encoding="utf-8", newline="\n")
    log.info("Wrote module.prop")


def extract_gz(gz_path: Path, dest_path: Path) -> None:
    dest_path.parent.mkdir(parents=True, exist_ok=True)
    log.info(f"Extracting {gz_path.name}  →  {dest_path.name}")
    with gzip.open(gz_path, "rb") as fin:
        dest_path.write_bytes(fin.read())
    dest_path.chmod(0o755)


def package_zip(mod_dir: Path, out_zip: Path) -> None:
    log.info(f"Creating ZIP: {out_zip.name}")
    with zipfile.ZipFile(out_zip, "w", compression=zipfile.ZIP_DEFLATED) as zf:
        for root, _, files in os.walk(mod_dir):
            for fname in files:
                if fname in ("placeholder", ".gitkeep"):
                    continue
                full = Path(root) / fname
                arcname = full.relative_to(mod_dir)
                zf.write(full, arcname=arcname)
    log.info(f"ZIP size: {out_zip.stat().st_size / 1_048_576:.1f} MB")


def write_updater_json(
    build_dir: Path, release_tag: str, frida_version: str, github_repo: str
) -> None:
    data = {
        "version": release_tag,
        "versionCode": generate_version_code(release_tag),
        "zipUrl": (
            f"https://github.com/{github_repo}/releases/download/"
            f"{release_tag}/florida-zygisk-{release_tag}.zip"
        ),
        "changelog": (f"https://github.com/{github_repo}/releases/tag/{release_tag}"),
    }
    out = build_dir / "updater.json"
    out.write_text(json.dumps(data, indent=2), encoding="utf-8", newline="\n")
    log.info("Wrote updater.json")


# ── Main ──────────────────────────────────────────────────────────────────────


def main() -> None:
    frida_version = os.environ.get("FRIDA_VERSION", "0.0.0")
    release_tag = os.environ.get("RELEASE_TAG", frida_version)
    github_repo = os.environ.get("GITHUB_REPO", "unknown/florida-zygisk")

    log.info(f"Building module  frida={frida_version}  tag={release_tag}")

    # Clean / prepare build tree
    if PATH_BUILD_TMP.exists():
        shutil.rmtree(PATH_BUILD_TMP)
    PATH_BUILD.mkdir(parents=True, exist_ok=True)

    # Copy module template
    shutil.copytree(PATH_BASE_MODULE, PATH_BUILD_TMP)

    # Write module.prop (overwrites the placeholder in the template)
    write_module_prop(PATH_BUILD_TMP, frida_version, release_tag, github_repo)

    # Extract florida-server binaries into files/
    files_dir = PATH_BUILD_TMP / "files"
    files_dir.mkdir(exist_ok=True)

    for build_arch, release_arch in ARCH_MAP.items():
        gz = PATH_DOWNLOADS / f"frida-server-{build_arch}.gz"
        if not gz.exists():
            raise FileNotFoundError(
                f"Missing artifact: {gz}\nAvailable: {list(PATH_DOWNLOADS.iterdir())}"
            )
        dest = files_dir / f"frida-server-{release_arch}"
        extract_gz(gz, dest)

    # Create the Magisk module ZIP
    out_zip = PATH_BUILD / f"florida-zygisk-{release_tag}.zip"
    package_zip(PATH_BUILD_TMP, out_zip)
    shutil.rmtree(PATH_BUILD_TMP)

    # Write updater.json for Magisk's auto-update mechanism
    write_updater_json(PATH_BUILD, release_tag, frida_version, github_repo)

    log.info("Done!")


if __name__ == "__main__":
    main()
```

```py
import lief
import sys
import random
import subprocess

def log_color(msg):
    print(f"\033[1;31;40m{msg}\033[0m")

if __name__ == "__main__":
    input_file = sys.argv[1]
    random_charset = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"
    log_color(f"[*] Patch frida-agent: {input_file}")
    binary = lief.parse(input_file)

    if not binary:
        log_color(f"[*] Not elf, exit")
        exit()

    random_name = "".join(random.sample(random_charset, 5))
    log_color(f"[*] Patch `frida` to `{random_name}`")

    for symbol in binary.symbols:
        if symbol.name == "frida_agent_main":
            symbol.name = "main"

        if "frida" in symbol.name:
            symbol.name = symbol.name.replace("frida", random_name)

        if "FRIDA" in symbol.name:
            symbol.name = symbol.name.replace("FRIDA", random_name)

    all_patch_string = ["FridaScriptEngine", "GLib-GIO", "GDBusProxy", "GumScript"]  # string signature patches
    for section in binary.sections:
        if section.name != ".rodata":
            continue
        for patch_str in all_patch_string:
            addr_all = section.search_all(patch_str)  # patch in-memory strings
            for addr in addr_all:
                patch = [ord(n) for n in list(patch_str)[::-1]]
                log_color(f"[*] Patching section name={section.name} offset={hex(section.file_offset + addr)} orig:{patch_str} new:{''.join(list(patch_str)[::-1])}")
                binary.patch_address(section.file_offset + addr, patch)

    binary.write(input_file)

    # thread_gum_js_loop
    random_name = "".join(random.sample(random_charset, 11))
    log_color(f"[*] Patch `gum-js-loop` to `{random_name}`")
    subprocess.run(["sed", "-b", "-i", f"s/gum-js-loop/{random_name}/g", input_file])

    # thread_gmain
    random_name = "".join(random.sample(random_charset, 5))
    log_color(f"[*] Patch `gmain` to `{random_name}`")
    subprocess.run(["sed", "-b", "-i", f"s/gmain/{random_name}/g", input_file])

    # thread_gdbus
    random_name = "".join(random.sample(random_charset, 5))
    log_color(f"[*] Patch `gdbus` to `{random_name}`")
    subprocess.run(["sed", "-b", "-i", f"s/gdbus/{random_name}/g", input_file])

    log_color(f"[*] Patch Finish")
```

`scripts/florida_carry.py`:

```py
#!/usr/bin/env python3
"""
florida_carry.py — local carry-fixes applied on top of Florida's patches.

Why this exists
---------------
This module builds Florida (anti-detection frida-server) from source by cloning
frida + applying the patches from https://github.com/Ylarod/Florida . Florida's
patches frequently bit-rot against new frida releases, and the *previous* workflow
masked that with `git am --reject || git am --skip`, which silently dropped hunks
and shipped broken/inconsistent binaries (see issue #1: "Failed to spawn: undefined
symbol: main").

This script applies two idempotent, content-anchored repairs so the build stays
correct regardless of whether Florida's patches applied cleanly. Each repair is a
no-op when the intended change is already present (e.g. once Florida fixes upstream).

  1. rpc_obfuscation  — keep Florida's `getRpcStr()` "frida:rpc" string obfuscation
                        self-consistent (definition present whenever it is used).
                        This is what bit-rotted against frida 17.16.x and broke CI.
  2. anti_anti_frida  — make the embed-time `anti-anti-frida.py` invocation robust
                        (env-var path + fail-hard). Without it the agent keeps the
                        `frida_agent_main` symbol while the server looks up `main`,
                        producing "Failed to spawn: undefined symbol: main" (#1).

Run from the frida *source* root (the directory that contains subprojects/frida-core),
e.g.  `cd frida && python3 ../scripts/florida_carry.py`.
"""

import os
import re
import sys
from pathlib import Path


def _frida_root() -> Path:
    # subprojects/frida-core should exist under the frida source root.
    root = Path.cwd()
    if not (root / "subprojects" / "frida-core").is_dir():
        sys.exit(
            f"florida_carry: must run from the frida source root (no "
            f"subprojects/frida-core under {root})"
        )
    return root


# ──────────────────────────────────────────────────────────────────────────────
# 1. rpc.vala — Florida "string_frida_rpc" obfuscation
# ──────────────────────────────────────────────────────────────────────────────

# Mirrors Florida's patches/frida-core/0001-Florida-string_frida_rpc.patch.
# "Wm5KcFpHRTZjbkJq" base64-decodes twice to the literal "frida:rpc".
_GETRPC_METHOD = (
    "\t\tpublic string getRpcStr(bool quote){\n"
    "\t\t\tstring result = (string) GLib.Base64.decode("
    "(string) GLib.Base64.decode(\"Wm5KcFpHRTZjbkJq\"));\n"
    "\t\t\tif(quote){\n"
    "\t\t\t\treturn \"\\\"\" + result + \"\\\"\";\n"
    "\t\t\t}else{\n"
    "\t\t\t\treturn result;\n"
    "\t\t\t}\n"
    "\t\t}\n"
)

# Stable anchors that survive frida reshuffles (constructor body + literal strings).
_RPC_CALL_SITES = [
    ('.add_string_value ("frida:rpc")', '.add_string_value (getRpcStr(false))'),
    ('if (json.index_of ("\\"frida:rpc\\"") == -1)',
     'if (json.index_of (getRpcStr(true)) == -1)'),
    ('if (type == null || type != "frida:rpc")',
     'if (type == null || type != getRpcStr(false))'),
]


def rpc_obfuscation(root: Path) -> None:
    """Ensure Florida's getRpcStr() obfuscation is present AND self-consistent.

    Handles every outcome of the (now strict) patch step:
      * Florida patch fully applied   -> no-op.
      * Florida patch wholly skipped  -> adds definition + call sites.
      * Florida patch partially there -> repairs whichever half is missing.
    """
    rpc = root / "subprojects" / "frida-core" / "lib" / "base" / "rpc.vala"
    if not rpc.exists():
        sys.exit(f"florida_carry: rpc.vala not found at {rpc}")
    src = rpc.read_text()
    orig = src

    # (a) Definition: insert once, right after the RpcClient constructor.
    if "public string getRpcStr" not in src:
        anchor = "\t\t\tObject (peer: peer);\n\t\t}\n"
        if anchor not in src:
            sys.exit(
                "florida_carry: cannot find rpc.vala constructor anchor "
                "('Object (peer: peer);') — frida restructured things; "
                "manual fix needed."
            )
        src = src.replace(anchor, anchor + "\n" + _GETRPC_METHOD, 1)

    # (b) Call sites: replace literal "frida:rpc" usages (idempotent).
    for old, new in _RPC_CALL_SITES:
        src = src.replace(old, new)

    if src != orig:
        rpc.write_text(src)
        print("florida_carry: rpc.vala — ensured getRpcStr obfuscation present")
    else:
        print("florida_carry: rpc.vala — already consistent")


# ──────────────────────────────────────────────────────────────────────────────
# 2. embed-agent.py — robust anti-anti-frida.py invocation
# ──────────────────────────────────────────────────────────────────────────────

# The robust block we want in place of Florida's fragile one. Indented to sit at
# the same level as Florida's insertion (12 spaces, sibling of the agent if/else).
_CARRY_BLOCK = '''            # --- florida-zygisk carry: robust anti-anti-frida invocation ---
            import os as _carry_os, sys as _carry_sys
            _carry_script = _carry_os.environ.get("FLORIDA_ANTI_ANTI_FRIDA")
            if not _carry_script:
                _carry_script = _carry_os.path.join(
                    _carry_os.path.dirname(_carry_os.path.realpath(_carry_sys.argv[0])),
                    "anti-anti-frida.py")
            _carry_rc = _carry_os.system(
                "python3 " + _carry_script + " " + str(priv_dir / f"frida-agent-{flavor}.so"))
            if _carry_rc != 0:
                raise SystemExit(
                    "florida-zygisk: anti-anti-frida.py failed (rc=%d); agent symbol "
                    "rename did not happen -> would cause: Failed to spawn, undefined symbol main"
                    % _carry_rc)
            print("anti-anti-frida finished (florida-zygisk carry)")
            # --- end carry ---
'''

# Florida's 0010-exec-anti-anti-frida.py.patch adds exactly this (fragile) block.
_FLORIDA_BLOCK_RE = re.compile(
    r'            import os\n'
    r'            custom_script=str\(output_dir\).*?'
    r'print\("anti-anti-frida error\. Code:?", return_code\)\n'
    r'[ \t]*\n',
    re.S,
)

_CARRY_MARKER = "# --- florida-zygisk carry: robust anti-anti-frida invocation ---"


def anti_anti_frida(root: Path) -> None:
    """Make embed-agent.py run anti-anti-frida.py reliably and fail hard on error."""
    eap = root / "subprojects" / "frida-core" / "src" / "embed-agent.py"
    if not eap.exists():
        sys.exit(f"florida_carry: embed-agent.py not found at {eap}")
    src = eap.read_text()
    orig = src

    # Already carried (idempotent refresh).
    if _CARRY_MARKER in src:
        src = re.sub(
            re.escape(_CARRY_MARKER) + r".*?# --- end carry ---\n",
            _CARRY_BLOCK,
            src,
            count=1,
            flags=re.S,
        )
    elif _FLORIDA_BLOCK_RE.search(src):
        # Florida's 0010 applied — replace its fragile block with ours.
        src = _FLORIDA_BLOCK_RE.sub(_CARRY_BLOCK, src, count=1)
    elif "anti-anti-frida" in src:
        # 0010 applied but in an unrecognised shape — don't guess, surface it.
        print("florida_carry: embed-agent.py has an unrecognised anti-anti-frida "
              "block — leaving untouched (review manually)")
    else:
        # 0010 didn't apply at all — insert our block at the 12-space level,
        # right before the Linux/Android branch's `embedded_assets += [...]`.
        # Target that specific occurrence by anchoring on the unique
        # `frida-agent-{flavor}.so` line that precedes it.
        marker = 'embedded_agent = priv_dir / f"frida-agent-{flavor}.so"'
        midx = src.find(marker)
        eidx = src.find("embedded_assets += [embedded_agent]", midx) if midx != -1 else -1
        if midx != -1 and eidx != -1:
            # Walk back to the start of the `embedded_assets` line.
            lstart = src.rfind("\n", 0, eidx) + 1
            src = src[:lstart] + _CARRY_BLOCK + src[lstart:]
        else:
            print("florida_carry: embed-agent.py — no insertion anchor for "
                  "anti-anti-frida; skipping (post-build verify will catch a "
                  "missing rename)")

    if src != orig:
        eap.write_text(src)
        print("florida_carry: embed-agent.py — installed robust anti-anti-frida hook")
    else:
        print("florida_carry: embed-agent.py — already robust")


def ensure_anti_anti_frida(root: Path) -> None:
    """Carry the combined effect of Florida patches 0003 + 0004 + 0005 + 0007.

    Those four patches all build up `src/anti-anti-frida.py` (0003 creates it,
    0004/0005/0007 extend it) and 0003 switches the agent entrypoint symbol from
    `frida_agent_main` to `main`. Patch 0003 is atomic, so when ONE of its other
    hunks (host-session.vala) bit-rots, the WHOLE patch is rejected and the script
    is never created — yet patch 0010 (the embed hook) still applies and tries to
    run it. We repair that by (a) dropping in our copy of the final script and
    (b) applying the entrypoint string swap idempotently.
    """
    src_dir = root / "subprojects" / "frida-core" / "src"
    local_script = Path(__file__).resolve().parent / "anti-anti-frida.py"

    # (a) Ensure the rename script exists (final cumulative Florida version).
    target_script = src_dir / "anti-anti-frida.py"
    if not target_script.exists():
        if not local_script.exists():
            sys.exit(f"florida_carry: cannot find local {local_script} to copy")
        target_script.write_text(local_script.read_text())
        print("florida_carry: created src/anti-anti-frida.py (patches 0003+0004+0005+0007)")
    else:
        print("florida_carry: src/anti-anti-frida.py already present")

    # (b) Switch the agent entrypoint symbol frida_agent_main -> main in the host
    # sessions + agent container, so the server looks up the renamed symbol.
    # Idempotent: matches Florida patch 0003 exactly (every quoted literal).
    targets = [
        "agent-container.vala",
        "linux/linux-host-session.vala",
        "darwin/darwin-host-session.vala",
        "freebsd/freebsd-host-session.vala",
        "qnx/qnx-host-session.vala",
        "windows/windows-host-session.vala",
    ]
    patched = 0
    for rel in targets:
        p = src_dir / rel
        if not p.exists():
            continue
        s = p.read_text()
        ns = s.replace('"frida_agent_main"', '"main"')
        if ns != s:
            p.write_text(ns)
            patched += 1
            print(f"florida_carry: entrypoint frida_agent_main -> main in {rel}")
    if patched == 0:
        print("florida_carry: entrypoint already 'main' everywhere (or files absent)")


def main() -> None:
    root = _frida_root()
    rpc_obfuscation(root)
    ensure_anti_anti_frida(root)   # patch 0003+0004+0005+0007: rename script + entrypoint
    anti_anti_frida(root)          # patch 0010: robust embed hook that runs the script
    _assert_consistent(root)
    print("florida_carry: done")


def _assert_consistent(root: Path) -> None:
    """Fail fast (before the long compile) if a carry-repaired file is still
    self-inconsistent. This is the cheap, early gate; the workflow's post-build
    'Verify Florida patches took effect' step is the final binary-level gate."""
    rpc = root / "subprojects" / "frida-core" / "lib" / "base" / "rpc.vala"
    if rpc.exists():
        src = rpc.read_text()
        if "getRpcStr(" in src and "public string getRpcStr" not in src:
            sys.exit("florida_carry: FATAL — getRpcStr used but not defined in rpc.vala")
    eap = root / "subprojects" / "frida-core" / "src" / "embed-agent.py"
    if eap.exists() and "custom_script=str(output_dir)" in eap.read_text():
        sys.exit("florida_carry: FATAL — fragile anti-anti-frida path still in embed-agent.py")
    aaf = root / "subprojects" / "frida-core" / "src" / "anti-anti-frida.py"
    if not aaf.exists():
        sys.exit("florida_carry: FATAL — anti-anti-frida.py missing; agent symbol rename can't run")
    print("florida_carry: consistency check OK")


if __name__ == "__main__":
    main()
```
