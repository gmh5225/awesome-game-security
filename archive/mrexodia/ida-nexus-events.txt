# IDA Nexus Event Viewer

A standalone Textual demo for the structured `/idb_events` stream. It discovers
all locally published Nexus databases, attaches to one selected instance,
and renders its IDB hook events live with execution provenance.

## Run

Prerequisites: Python 3.11+, `uv`, and at least one IDA database with the Code
Mode plugin running.

```console
uv sync
uv run ida-nexus-events
```

The left table refreshes every two seconds. Select a ready database with Enter
or a mouse click. The right pane streams events until another database is
selected or the application exits.

Bindings:

- `r`: refresh database discovery immediately
- `c`: clear the current event pane
- `q`: quit

## Display

Each event row shows local time, revision, event name, provenance, and all
callback-specific fields. Event families use stable colors for renames,
functions, segments, types, patches, and comments.

`execute_python()` events show the caller-supplied operation label, with the
operation ID retained alongside it. Events emitted by the GUI plugin outside
an execution are labeled `IDA GUI`; this includes direct UI actions and
background work in the GUI process. A literal `unattributed` fallback remains
for older Nexus instances that do not provide that label.

Unavailable discovery records remain visible as `BLOCKED` rows with their
registry state preserved, but only `READY` rows can be attached.

```

from __future__ import annotations

import reprlib
import threading
from datetime import datetime
from pathlib import Path
from typing import Any, ClassVar

from ida_nexus import (
    DatabaseChangeEvent,
    DatabaseHandle,
    DiscoveredDatabase,
    InstanceState,
    discover_databases,
)
from rich.text import Text
from textual.app import App, ComposeResult
from textual.containers import Horizontal, Vertical
from textual.widgets import DataTable, Footer, Header, Label, RichLog, Static

_EVENT_METADATA = {
    "event_name",
    "operation_id",
    "operation_label",
    "origin_id",
    "revision",
    "timestamp",
}


class EventConnection:
    """Own one handle and blocking event subscription on a background thread."""

    def __init__(
        self,
        app: EventViewerApp,
        generation: int,
        database: DiscoveredDatabase,
    ) -> None:
        self._app = app
        self.generation = generation
        self.database = database
        self._stop = threading.Event()
        self._lock = threading.Lock()
        self._handle: DatabaseHandle | None = None
        self._subscription: Any = None
        self._thread = threading.Thread(
            target=self._run,
            name=f"idb-events-{database.instance.record_id}",
            daemon=True,
        )

    def start(self) -> None:
        self._thread.start()

    def close(self) -> None:
        self._stop.set()
        with self._lock:
            subscription = self._subscription
            handle = self._handle
        if subscription is not None:
            subscription.close()
        if handle is not None:
            handle.close()

    def _dispatch(self, callback: Any, *args: Any) -> None:
        try:
            self._app.call_from_thread(callback, self.generation, *args)
        except RuntimeError:
            # The app may finish while a socket close wakes this thread.
            pass

    def _run(self) -> None:
        handle: DatabaseHandle | None = None
        subscription: Any = None
        try:
            handle = DatabaseHandle.attach(self.database.instance, keepalive=20.0)
            with self._lock:
                self._handle = handle
            if self._stop.is_set():
                return

            subscription = handle.subscribe_idb_events()
            with self._lock:
                self._subscription = subscription
            if self._stop.is_set():
                return

            self._dispatch(self._app._connection_ready, self.database)
            for event in subscription:
                if self._stop.is_set():
                    break
                self._dispatch(self._app._receive_event, event)
        except Exception as exc:  # noqa: BLE001 - report thread boundary failures
            if not self._stop.is_set():
                message = str(exc) or type(exc).__name__
                self._dispatch(self._app._connection_failed, message)
        finally:
            if subscription is not None:
                subscription.close()
            if handle is not None:
                handle.close()
            with self._lock:
                self._subscription = None
                self._handle = None


class EventViewerApp(App[None]):
    """Discover Nexus databases and stream one selected event feed."""

    TITLE = "IDA Nexus Events"
    SUB_TITLE = "live database provenance"
    CSS = """
    Screen {
        layout: vertical;
        background: $surface;
    }

    #workspace {
        height: 1fr;
        layout: horizontal;
    }

    #database-pane {
        width: 42;
        min-width: 32;
        border-right: tall $primary;
        background: $panel;
    }

    #event-pane {
        width: 1fr;
    }

    .pane-title {
        height: 3;
        padding: 1 2 0 2;
        text-style: bold;
        color: $text-accent;
    }

    #database-hint {
        height: 2;
        padding: 0 2;
        color: $text-muted;
    }

    #databases {
        height: 1fr;
        margin: 0 1 1 1;
    }

    #events {
        height: 1fr;
        padding: 0 1;
        scrollbar-gutter: stable;
    }

    #status {
        height: 3;
        padding: 1 2;
        background: $boost;
        color: $text-muted;
    }
    """

    BINDINGS: ClassVar = [
        ("r", "refresh", "Refresh"),
        ("c", "clear_events", "Clear events"),
        ("q", "quit", "Quit"),
    ]

    def __init__(self) -> None:
        super().__init__()
        self._closed = False
        self._discovering = threading.Event()
        self._rows: dict[str, DiscoveredDatabase] = {}
        self._row_order: list[str] = []
        self._connection: EventConnection | None = None
        self._generation = 0
        self._active_record_id: str | None = None
        self._event_count = 0

    def compose(self) -> ComposeResult:
        yield Header(show_clock=True)
        with Horizontal(id="workspace"):
            with Vertical(id="database-pane"):
                yield Label("DATABASES", classes="pane-title")
                yield Static("Enter or click to attach", id="database-hint")
                yield DataTable(id="databases", cursor_type="row", zebra_stripes=True)
            with Vertical(id="event-pane"):
                yield Label("LIVE EVENTS", classes="pane-title")
                yield RichLog(id="events", wrap=True, highlight=False, markup=False)
                yield Static("Discovering Nexus instances…", id="status")
        yield Footer()

    def on_mount(self) -> None:
        table = self.query_one("#databases", DataTable)
        table.add_columns("STATE", "TYPE", "DATABASE")
        self.refresh_databases()
        self.set_interval(2.0, self.refresh_databases)

    def on_unmount(self) -> None:
        self._closed = True
        if self._connection is not None:
            self._connection.close()

    def action_refresh(self) -> None:
        self.refresh_databases()

    def action_clear_events(self) -> None:
        self.query_one("#events", RichLog).clear()
        self._event_count = 0
        self._update_status()

    def refresh_databases(self) -> None:
        if self._closed or self._discovering.is_set():
            return
        self._discovering.set()
        threading.Thread(
            target=self._discover,
            name="ida-nexus-discovery",
            daemon=True,
        ).start()

    def _discover(self) -> None:
        try:
            databases = discover_databases(timeout=0.5)
            if not self._closed:
                self.call_from_thread(self._apply_databases, databases)
        except Exception as exc:  # noqa: BLE001 - report discovery failures
            if not self._closed:
                message = str(exc) or type(exc).__name__
                self.call_from_thread(self._show_discovery_error, message)
        finally:
            self._discovering.clear()

    def _show_discovery_error(self, message: str) -> None:
        self.query_one("#status", Static).update(
            Text.assemble(("Discovery failed: ", "bold red"), message)
        )

    def _apply_databases(self, databases: list[DiscoveredDatabase]) -> None:
        table = self.query_one("#databases", DataTable)
        previous_record_id: str | None = None
        if 0 <= table.cursor_row < len(self._row_order):
            previous_record_id = self._row_order[table.cursor_row]

        table.clear()
        self._rows = {item.instance.record_id: item for item in databases}
        self._row_order = []
        ready_count = 0
        for item in databases:
            instance = item.instance
            record_id = instance.record_id
            self._row_order.append(record_id)
            is_ready = item.state is InstanceState.READY
            ready_count += int(is_ready)
            state = (
                Text("● READY", style="bold green")
                if is_ready
                else Text("● BLOCKED", style="bold yellow")
            )
            database_path = instance.idb_path or instance.exe_path
            database_name = Path(database_path).name if database_path else "<unsaved>"
            table.add_row(
                state,
                Text(instance.backend.upper(), style="cyan"),
                Text(database_name),
                key=record_id,
            )

        if previous_record_id in self._row_order:
            table.move_cursor(row=self._row_order.index(previous_record_id))
        elif self._active_record_id in self._row_order:
            table.move_cursor(row=self._row_order.index(self._active_record_id))

        self.query_one("#database-hint", Static).update(
            f"{ready_count} ready / {len(databases)} discovered"
        )
        if not databases and self._connection is None:
            self.query_one("#status", Static).update(
                "No live Nexus databases. Open IDA with the plugin enabled."
            )

    def on_data_table_row_selected(self, message: DataTable.RowSelected) -> None:
        if message.data_table.id != "databases":
            return
        record_id = str(message.row_key.value)
        database = self._rows.get(record_id)
        if database is None:
            return
        if database.state is not InstanceState.READY:
            detail = database.detail or "instance is not attachable"
            self.query_one("#status", Static).update(
                Text.assemble(("Unavailable: ", "bold yellow"), detail)
            )
            return
        self._attach(database)

    def _attach(self, database: DiscoveredDatabase) -> None:
        if self._connection is not None:
            self._connection.close()
        self._generation += 1
        self._active_record_id = database.instance.record_id
        self._event_count = 0
        self.query_one("#events", RichLog).clear()
        path = database.instance.idb_path or database.instance.exe_path or "<unsaved>"
        self.query_one("#status", Static).update(
            Text.assemble(("Connecting to ", "bold cyan"), path, " …")
        )
        connection = EventConnection(self, self._generation, database)
        self._connection = connection
        connection.start()

    def _connection_ready(
        self,
        generation: int,
        database: DiscoveredDatabase,
    ) -> None:
        if not self._is_current(generation):
            return
        path = database.instance.idb_path or database.instance.exe_path or "<unsaved>"
        self.query_one("#status", Static).update(
            Text.assemble(("LIVE", "bold white on green"), "  ", path)
        )

    def _connection_failed(self, generation: int, message: str) -> None:
        if not self._is_current(generation):
            return
        self.query_one("#status", Static).update(
            Text.assemble(("Disconnected: ", "bold red"), message)
        )

    def _receive_event(
        self,
        generation: int,
        event: DatabaseChangeEvent,
    ) -> None:
        if not self._is_current(generation):
            return
        self._event_count += 1
        self.query_one("#events", RichLog).write(_render_event(event))
        self._update_status()

    def _update_status(self) -> None:
        if self._connection is None or self._active_record_id is None:
            return
        database = self._rows.get(self._active_record_id)
        if database is None:
            path = "attached database"
        else:
            instance = database.instance
            path = instance.idb_path or instance.exe_path or "<unsaved>"
        self.query_one("#status", Static).update(
            Text.assemble(
                ("LIVE", "bold white on green"),
                "  ",
                path,
                (f"  ·  {self._event_count} events", "bold cyan"),
            )
        )

    def _is_current(self, generation: int) -> bool:
        return (
            self._connection is not None
            and self._connection.generation == generation
            and not self._closed
        )


def _render_event(event: DatabaseChangeEvent) -> Text:
    timestamp_ns = event.get("timestamp", 0)
    try:
        timestamp = datetime.fromtimestamp(timestamp_ns / 1_000_000_000).astimezone()
        time_text = timestamp.strftime("%H:%M:%S.%f")[:-3]
    except (OSError, OverflowError, ValueError):
        time_text = "--:--:--.---"

    event_name = str(event.get("event_name", "unknown"))
    actor = event.get("operation_label") or event.get("operation_id") or "unattributed"
    actor_style = (
        "bold yellow" if actor in {"IDA GUI", "unattributed"} else "bold green"
    )
    line = Text.assemble(
        (time_text, "dim cyan"),
        "  ",
        (f"r{event.get('revision', '?'):<5}", "cyan"),
        (f"{event_name:<28}", _event_style(event_name)),
        (f"{actor}", actor_style),
    )

    operation_id = event.get("operation_id")
    if operation_id and operation_id != actor:
        line.append(f" [{operation_id}]", style="dim green")

    for name, value in event.items():
        if name in _EVENT_METADATA:
            continue
        line.append(f"  {name}=", style="dim")
        line.append(_short_repr(value), style="white")
    return line


def _event_style(event_name: str) -> str:
    if "renamed" in event_name or "name" in event_name:
        return "bold magenta"
    if "func" in event_name or "frame" in event_name:
        return "bold green"
    if "segm" in event_name:
        return "bold blue"
    if "type" in event_name or event_name.startswith(("ti_", "lt_", "udm_")):
        return "bold bright_magenta"
    if "patched" in event_name or "data" in event_name:
        return "bold red"
    if "cmt" in event_name or "comment" in event_name:
        return "bold cyan"
    return "bold white"


def _short_repr(value: Any) -> str:
    formatter = reprlib.Repr()
    formatter.maxstring = 72
    formatter.maxother = 72
    formatter.maxlist = 4
    formatter.maxdict = 4
    return formatter.repr(value)


def main() -> None:
    EventViewerApp().run()


if __name__ == "__main__":
    main()
