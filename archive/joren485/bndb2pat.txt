`README.md`:

```md
# bndb2pat
Binary Ninja plugin to generate IDA Pro FLIRT .pat pattern files.

This Binary Ninja plugin can be used to generate .pat pattern files which are used in generating [IDA Pro FLIRT](https://docs.hex-rays.com/core/flirt) signatures.

This plugin is based on the following projects:
* [idb2pat.py](https://github.com/mandiant/flare-ida/blob/master/python/flare/idb2pat.py): A IDA Python script by Mandiant FLARE to generate .pat files from IDBs.
* [idb2pat.cpp](https://github.com/KasperskyLab/hrtng/blob/master/src/idb2pat.cpp): The IDA Pro plugin [hrtng](github.com/KasperskyLab/hrtng/) by Kaspersky contains functionality to also generate .pat files from IDBs.
* [Signature Kit](https://github.com/Vector35/sigkit/tree/master): The previous signature library technology used in Binary Ninja worked similarly to FLIRT signatures.

### Limitations
This project is mostly implemented as a toy example to learn how to write Binary Ninja plugins. It has not been extensively tested and likely contains problems.

There are ofcourse many differences between how IDA Pro and Binary Ninja parse a binary.

One of the biggest differences is that IDA Pro API exposes assembly instructions on a byte-level. This allows IDA Pro scripts that specifically target the bytes of an operand in an instruction. Binary Ninja does not do this.
This allows IDA Pro to keep the static bytes of instructions referencing pointers in byte masks. As Binary Ninja does not expose this information in their API, we mask the whole instruction, including any static bytes.  

For example:
```
E8 FF FF FF FF  CALL some_function
E8 .. .. .. .. # byte mask in IDA Pro
.. .. .. .. .. # byte mask in this plugin
```

### Example
We use a `third_steps.elf` demo binary to show this works (at least as a toy example). 

```commandline
$ file third_steps.elf
third_steps.elf: ELF 64-bit LSB executable, x86-64, version 1 (SYSV), dynamically linked, interpreter /lib64/ld-linux-x86-64.so.2, for GNU/Linux 3.2.0, not stripped
```
We use this plugin to generate a .pat file and FLAIR tools to turn the .pat file into an actual .sig FLIRT signature file. 

```commandline
$ sigmake third_steps.pat third_steps.sig
```

Once we load a stripped version of `third_steps.sig` into IDA Pro, we can see that our signatures work: 

| Before third_steps.sig is loaded | After third_steps.sig is loaded |
|----------------------------------|---------------------------------|
| ![Before](docs/before.png)       | ![After](docs/after.png)        |

from pathlib import Path

from binaryninja import BinaryView, PluginCommand, get_save_filename_input, log

from .bndb2pat.task import GeneratePatternFileTask


def get_output_path() -> Path | None:
    """Get an output path."""
    path = get_save_filename_input("Select FLIRT .pat output file", "pat", "output.pat")
    if path is None:
        log.log_error("Error during file selection", logger="bndb2pat")
        return None
    return Path(path)


def generate_pat_file(bv: BinaryView) -> None:
    """Start task to generate FLIRT .pat file."""
    path = get_output_path()
    if path is None:
        return

    task = GeneratePatternFileTask(bv, path)
    task.start()


PluginCommand.register("bndb2pat", "Generate IDA Pro FLIRT .pat pattern files", generate_pat_file)

def crc16(data: bytes | list[int | None], crc: int = 0xFFFF) -> int:
    """
    Compute CRC16 checksum.

    This implementation is based on https://github.com/mandiant/flare-ida/blob/master/python/flare/idb2pat.py#L66-L101
    """
    for b in data:
        if b is None:
            continue
        crc = (crc >> 8) ^ CRC16_TABLE[(crc ^ b) & 0xFF]
    crc = (~crc) & 0xFFFF
    crc = (crc << 8) | ((crc >> 8) & 0xFF)
    return crc & 0xFFFF

```py
from binaryninja import (
    Function,
    LowLevelILConstPtr,
    LowLevelILExternPtr,
    LowLevelILInstruction,
    log,
)

from .pattern import FLIRTPattern

VARIABLE_OPERATION_CLASSES = (
    LowLevelILExternPtr,
    LowLevelILConstPtr,
)

FUNCTION_NAME_PREFX_IGNORELIST = ("sub_", "j_", "unknown_libname_", "nullsub_")

FUNCTION_LENGTH_MINIMUM = 10


def is_relevant_symbol(symbol: str) -> bool:
    """Check whether a symbol is relevant to refer to."""
    return not symbol.startswith(FUNCTION_NAME_PREFX_IGNORELIST)


def visit_instruction(inst: LowLevelILInstruction) -> int | None:
    """Visit an LLIL instruction, and return its constant if Binary Ninja determined the constant to be a pointer."""
    if isinstance(inst, VARIABLE_OPERATION_CLASSES):
        return inst.constant
    return None


def get_variable_instruction_pointer(func: Function, addr: int) -> int | None:
    """Get a pointer from the instruction on a given address, if there is one."""
    for inst in func.get_low_level_ils_at(addr):
        for pointer in inst.traverse(visit_instruction):
            if pointer is not None:
                return pointer
    return None


def generate_pattern(func: Function) -> str | None:
    """
    Generate a FLIRT pattern for a given function.

    This method loops over each instruction in the first address range of a function and builds a pattern from the
    bytes. We use the LLIL to get the benifits of the Binary Ninja API, but stay as close to the dissassembly as
    possible.

    If a function is chuncked throughout a binary, we will only consider the first chunk.
    We only consider the first address range of a function, because IDA also seems to do this.
    """
    if func.is_thunk or not is_relevant_symbol(func.name):
        log.log_info(f"Skipping generating function {func.name}", logger="bndb2pat")

    func_start, func_end = func.address_ranges[0].start, func.address_ranges[0].end
    if func_end - func_start < FUNCTION_LENGTH_MINIMUM:
        log.log_info(f"Skipping small function: {func.name}", logger="bndb2pat")

    pattern = FLIRTPattern()
    addr_inst = func_start
    while addr_inst < func_end:
        inst_length = func.view.get_instruction_length(addr_inst)
        if inst_length == 0:
            log.log_error(f"Encountered invalid instruction @0x{addr_inst:x}", logger="bndb2pat")
            return None

        pointer = get_variable_instruction_pointer(func, addr_inst)
        # Only consider pointers that are outside the function range
        if pointer is not None and (pointer < func_start or func_end <= pointer):
            pattern.mask.extend([pattern.MASKED_BYTE_MARKER] * inst_length)

            # Add a named reference (e.g. a callee)
            symbol = func.view.get_symbol_at(pointer)
            if symbol is not None and is_relevant_symbol(symbol.name):
                pattern.named_refs[addr_inst - func_start] = symbol.name

        else:
            data = func.view.read(addr_inst, inst_length)
            if len(data) != inst_length:
                log.log_error(f"Expected: {inst_length} bytes but got {len(data)} bytes", logger="bndb2pat")
                return None

            pattern.mask.extend(data)

        # Add a public reference (e.g. the name of the function we are analyzing)
        symbol = func.view.get_symbol_at(addr_inst)
        if symbol is not None and is_relevant_symbol(symbol.name):
            pattern.public_refs[addr_inst - func_start] = symbol.name

        addr_inst += inst_length

    # Pad the mask with masked bytes, to make sure the mask is at least BYTES_PATTERN_MIN_LENGTH bytes long
    if len(pattern.mask) < pattern.BYTES_PATTERN_MIN_LENGTH:
        pattern.mask.extend([pattern.MASKED_BYTE_MARKER] * (pattern.BYTES_PATTERN_MIN_LENGTH - len(pattern.mask)))

    if not pattern.is_valid():
        log.log_error("Skipping invalid pattern", logger="bndb2pat")
        return None

    return pattern.format()

`bndb2pat/pattern.py`:

```py
from dataclasses import dataclass, field
from typing import ClassVar

from .crc16 import crc16


@dataclass
class FLIRTPattern:
    """
    Dataclass representing a pattern in a FLIRT .pat file.

    This class consists of three elements:
        * A byte mask that contains all static bytes of a function with the variable bytes masked out.
        * A address-symbol mapping of public names used in the function (e.g. the name of the function)
        * A address-symbol mapping of public names used in the function  (e.g. callees of the function)

    This class is used to build the pattern and to format it.
    """

    MASKED_BYTE_MARKER: ClassVar[None] = None
    WILDCARD_FORMAT: ClassVar[str] = ".."

    BYTES_PATTERN_MIN_LENGTH: ClassVar[int] = 32

    mask: list[int | None] = field(default_factory=list)
    public_refs: dict[int, str] = field(default_factory=dict)
    named_refs: dict[int, str] = field(default_factory=dict)

    def is_valid(self) -> bool:
        """Check if the mask contains elements and contains non-None elements."""
        return len(self.mask) > 0 and any(e is not None for e in self.mask)

    def _format_masked_bytes(self, data: list[int | None]) -> str:
        """Format a list of masked bytes as a string."""
        result = ""
        for b in data:
            if b is None:
                result += self.WILDCARD_FORMAT
            else:
                result += f"{b:02X}"
        return result

    def format(self) -> str:
        """
        Format a FLIRT pattern to a line in a .pat file.

        This consits of the following elements:
            * Byte pattern prefix: 32 bytes of the mask, with variable bytes denoted as ".."
            * The length of a CRC16 checksum
            * The CRC16 checksum of all bytes after the byte pattern prefix up to the first variable byte
            * The length of the function
            * The remaining mask not used in the byte pattern and the CRC16 checksum (i.e. tail data)
        """
        # Byte pattern prefix
        pattern = self._format_masked_bytes(self.mask[: self.BYTES_PATTERN_MIN_LENGTH])

        # CRC Data
        try:
            offset_end_crc_data = self.mask.index(self.MASKED_BYTE_MARKER, self.BYTES_PATTERN_MIN_LENGTH)
        except ValueError:
            offset_end_crc_data = len(self.mask)

        # CRC16 data length is capped at 255 bytes since the CRC16 data length in the pattern line is encoded as a
        # single hex byte (max 0xFF).
        offset_end_crc_data = min(offset_end_crc_data, self.BYTES_PATTERN_MIN_LENGTH + 255)

        crc_data = self.mask[self.BYTES_PATTERN_MIN_LENGTH : offset_end_crc_data]
        crc_checksum = crc16(crc_data)

        pattern += f" {len(crc_data):02X}"
        pattern += f" {crc_checksum:04X}"

        # Function length
        pattern += f" {len(self.mask):04X}"

        # References
        for offset, symbol in self.public_refs.items():
            pattern += f" :{offset:04X} {symbol}"

        for offset, symbol in self.named_refs.items():
            pattern += f" ^{offset:04X} {symbol}"

        # tail data
        offset_start_tail_data = self.BYTES_PATTERN_MIN_LENGTH + len(crc_data)
        pattern += f" {self._format_masked_bytes(self.mask[offset_start_tail_data:])}"

        return pattern.strip()

`bndb2pat/task.py`:

```py
from pathlib import Path

from binaryninja import BackgroundTaskThread, BinaryView, log

from .generate import generate_pattern


class GeneratePatternFileTask(BackgroundTaskThread):
    """Background task for generating FLIRT .pat pattern files."""

    def __init__(self, bv: BinaryView, path: Path) -> None:
        """Set the BinaryView and Path for the background task."""
        super().__init__("Generating FLIRT .pat file", can_cancel=True)
        self.bv = bv
        self.path = path

    def run(self) -> None:
        """Generate a FLIRT .pat file for each function in the binary."""
        patterns = []
        functions = list(self.bv.functions)
        for func_i, func in enumerate(functions):
            if self.cancelled:
                return

            self.progress = f"Generating pattern for function '{func.name}' ({func_i + 1}/{len(functions)})"
            log.log_debug(f"Generating pattern for function '{func.name}'", logger="bndb2pat")
            pattern = generate_pattern(func)
            if pattern is None:
                continue

            log.log_debug(f"{func.name}: '{pattern}'", logger="bndb2pat")
            patterns.append(pattern)

        with self.path.open("w") as h_file:
            for pattern in patterns:
                h_file.write(f"{pattern}\n")
            h_file.write("---\n")
