# Vulcan AntiCheat - Built from Scratch

A custom-built Minecraft anti-cheat plugin created entirely from scratch. This is not a fork or copy of any existing Vulcan AntiCheat implementation - it's a completely original build with core detection features.

## Features

### Core Detection Algorithms
- **Fly Detection** - Detects illegal flying and excessive air time
- **Speed Detection** - Identifies unnatural movement speeds
- **KillAura Detection** - Detects 360-degree attacking patterns
- **Reach Detection** - Checks for extended attack distances
- **AutoClicker Detection** - Identifies unnaturally fast clicking
- **NoFall Detection** - Detects fall damage bypass

### Advanced Systems
- **Player Data Tracking** - Comprehensive player movement and combat analysis
- **Configurable Detection** - Adjustable thresholds for all checks
- **Smart Violation System** - Tracks violations with automatic reset
- **Performance Optimized** - Efficient algorithms with minimal server impact

### Alert & Punishment System
- **Real-time Alerts** - Notify staff when cheating is detected
- **Multi-channel Alerts** - Console, staff, and broadcast options
- **Graduated Punishments** - Warnings → Kicks → Bans
- **Setback System** - Teleports cheating players back
- **Bypass System** - Permission-based check bypassing

### Command System
- `/vulcan help` - Shows command help
- `/vulcan reload` - Reloads configuration
- `/vulcan alerts` - Toggle cheat alerts
- `/vulcan bypass` - Toggle bypass mode
- `/vulcan violations <player>` - Check player violations
- `/vulcan reset <player>` - Reset player violations
- `/vulcan info` - Shows plugin information

## Installation

### Requirements
- **Minecraft Server**: Paper/Spigot 1.21.4+
- **Java**: Version 21+
- **Build Tool**: Maven 3.9+

### Building from Source
```bash
# Clone the repository
git clone https://github.com/MayankPatil3/vulcan-anticheat-built.git
cd vulcan-anticheat-built

# Build the plugin
mvn clean package

# The compiled JAR will be in target/vulcan-anticheat-built-1.0.0.jar
```

### Server Installation
1. Build the plugin or download the latest release
2. Place the JAR file in your server's `plugins/` directory
3. Restart your server
4. Configure the plugin in `plugins/VulcanAntiCheatBuilt/config.yml`

  max-vertical-speed: 1.2

package com.mayankpatil.anticheat;

import org.bukkit.entity.Player;
import org.bukkit.Bukkit;
import org.bukkit.ChatColor;
import org.bukkit.configuration.file.FileConfiguration;

import java.util.HashSet;
import java.util.Set;
import java.util.UUID;

/**
 * AlertManager - Handles sending alerts for detected violations
 */
public class AlertManager {
    
    private final VulcanAntiCheat plugin;
    private final Set<UUID> alertReceivers = new HashSet<>();
    private final Set<UUID> bypassingPlayers = new HashSet<>();
    
    // Alert configuration
    private String alertPrefix = "&8[&cVulcan&8]";
    private boolean alertsEnabled = true;
    private boolean broadcastAlerts = false;
    private boolean consoleAlerts = true;
    private String alertFormat = "{player} &7failed &c{check}&7: &f{details} &8(&c{vl}&8)";
    
    public AlertManager(VulcanAntiCheat plugin) {
        this.plugin = plugin;
        loadConfig();
    }
    
    private void loadConfig() {
        FileConfiguration config = plugin.getPluginConfig();
        
        alertPrefix = config.getString("alerts.prefix", "&8[&cVulcan&8]");
        alertsEnabled = config.getBoolean("alerts.enabled", true);
        broadcastAlerts = config.getBoolean("alerts.broadcast", false);
        consoleAlerts = config.getBoolean("alerts.console", true);
        alertFormat = config.getString("alerts.format", "{player} &7failed &c{check}&7: &f{details} &8(&c{vl}&8)");
        
        // Load staff with alert permissions
        for (String uuidStr : config.getStringList("alerts.receivers")) {
            try {
                alertReceivers.add(UUID.fromString(uuidStr));
            } catch (IllegalArgumentException e) {
                plugin.getLogger().warning("Invalid UUID in alert receivers: " + uuidStr);
            }
        }
    }
    
    public void sendAlert(Player player, String check, String details, int violations) {
        if (!alertsEnabled) return;
        
        String formattedAlert = formatAlert(player, check, details, violations);
        
        // Send to console if enabled
        if (consoleAlerts) {
            plugin.getLogger().warning(ChatColor.stripColor(formattedAlert));
        }
        
        // Send to staff with alert permissions
        for (Player onlinePlayer : Bukkit.getOnlinePlayers()) {
            if (shouldReceiveAlerts(onlinePlayer)) {
                onlinePlayer.sendMessage(formattedAlert);
            }
        }
        
        // Broadcast if enabled
        if (broadcastAlerts) {
            Bukkit.broadcastMessage(formattedAlert);
        }
    }
    
    private String formatAlert(Player player, String check, String details, int violations) {
        String formatted = alertFormat
            .replace("{player}", player.getName())
            .replace("{check}", check)
            .replace("{details}", details)
            .replace("{vl}", String.valueOf(violations));
        
        return ChatColor.translateAlternateColorCodes('&', alertPrefix + " " + formatted);
    }
    
    private boolean shouldReceiveAlerts(Player player) {
        // Check if player has alert permission
        if (player.hasPermission("vulcan.alerts")) {
            return true;
        }
        
        // Check if player is in receivers list
        if (alertReceivers.contains(player.getUniqueId())) {
            return true;
        }
        
        // Check if player is OP (optional)
        if (player.isOp() && plugin.getPluginConfig().getBoolean("alerts.ops-receive", true)) {
            return true;
        }
        
        return false;
    }
    
    public void addAlertReceiver(Player player) {
        alertReceivers.add(player.getUniqueId());
        saveReceivers();
    }
    
    public void removeAlertReceiver(Player player) {
        alertReceivers.remove(player.getUniqueId());
        saveReceivers();
    }
    
    public void toggleAlerts(Player player) {
        if (alertReceivers.contains(player.getUniqueId())) {
            removeAlertReceiver(player);
            player.sendMessage(ChatColor.RED + "Alerts disabled.");
        } else {
            addAlertReceiver(player);
            player.sendMessage(ChatColor.GREEN + "Alerts enabled.");
        }
    }
    
    private void saveReceivers() {
        // Convert UUIDs to strings
        java.util.List<String> uuidStrings = new java.util.ArrayList<>();
        for (UUID uuid : alertReceivers) {
            uuidStrings.add(uuid.toString());
        }
        
        plugin.getPluginConfig().set("alerts.receivers", uuidStrings);
        plugin.saveConfig();
    }
    
    public void toggleBypass(Player player) {
        if (bypassingPlayers.contains(player.getUniqueId())) {
            bypassingPlayers.remove(player.getUniqueId());
            player.sendMessage(ChatColor.RED + "Bypass disabled.");
        } else {
            bypassingPlayers.add(player.getUniqueId());
            player.sendMessage(ChatColor.GREEN + "Bypass enabled.");
        }
    }
    
    public boolean isBypassing(Player player) {
        return bypassingPlayers.contains(player.getUniqueId()) || 
               player.hasPermission("vulcan.bypass.all");
    }
    
    // Getters
    public boolean isAlertsEnabled() { return alertsEnabled; }
    public boolean isBroadcastAlerts() { return broadcastAlerts; }
    public boolean isConsoleAlerts() { return consoleAlerts; }
    public Set<UUID> getAlertReceivers() { return new HashSet<>(alertReceivers); }
    
    // Setters
    public void setAlertsEnabled(boolean enabled) { 
        alertsEnabled = enabled; 
        plugin.getPluginConfig().set("alerts.enabled", enabled);
        plugin.saveConfig();
    }
    
    public void setBroadcastAlerts(boolean broadcast) { 
        broadcastAlerts = broadcast; 
        plugin.getPluginConfig().set("alerts.broadcast", broadcast);
        plugin.saveConfig();
    }
    
    public void setConsoleAlerts(boolean console) { 
        consoleAlerts = console; 
        plugin.getPluginConfig().set("alerts.console", console);
        plugin.saveConfig();
    }
}

package com.mayankpatil.anticheat;

import org.bukkit.entity.Player;
import org.bukkit.event.player.PlayerMoveEvent;
import org.bukkit.event.entity.EntityDamageByEntityEvent;
import org.bukkit.Location;
import org.bukkit.util.Vector;

import java.util.HashSet;
import java.util.Set;

/**
 * DetectionManager - Implements core anti-cheat detection algorithms
 */
public class DetectionManager {
    
    private final VulcanAntiCheat plugin;
    private final Set<String> enabledChecks = new HashSet<>();
    
    // Configuration thresholds
    private double maxHorizontalSpeed = 5.0; // blocks per second
    private double maxVerticalSpeed = 1.2; // blocks per second
    private int maxAirTicks = 20; // ticks (1 second)
    private int maxAttacksPerSecond = 15;
    private double maxReachDistance = 4.5; // blocks
    private double maxAngleChange = 90.0; // degrees per second
    private int violationThreshold = 5;
    
    public DetectionManager(VulcanAntiCheat plugin) {
        this.plugin = plugin;
        loadConfig();
    }
    
    private void loadConfig() {
        // Load configuration values
        maxHorizontalSpeed = plugin.getPluginConfig().getDouble("detection.max-horizontal-speed", 5.0);
        maxVerticalSpeed = plugin.getPluginConfig().getDouble("detection.max-vertical-speed", 1.2);
        maxAirTicks = plugin.getPluginConfig().getInt("detection.max-air-ticks", 20);
        maxAttacksPerSecond = plugin.getPluginConfig().getInt("detection.max-attacks-per-second", 15);
        maxReachDistance = plugin.getPluginConfig().getDouble("detection.max-reach-distance", 4.5);
        maxAngleChange = plugin.getPluginConfig().getDouble("detection.max-angle-change", 90.0);
        violationThreshold = plugin.getPluginConfig().getInt("detection.violation-threshold", 5);
        
        // Load enabled checks
        for (String check : plugin.getPluginConfig().getStringList("detection.enabled-checks")) {
            enabledChecks.add(check.toLowerCase());
        }
        
        plugin.getLogger().info("Loaded " + enabledChecks.size() + " detection checks");
    }
    
    public void runChecks(Player player, PlayerData data) {
        // Run all enabled checks periodically
        if (enabledChecks.contains("fly")) {
            checkFly(player, data);
        }
        
        if (enabledChecks.contains("speed")) {
            checkSpeed(player, data);
        }
        
        // Reset violations if needed
        if (System.currentTimeMillis() - data.getLastViolationReset() > 60000) { // 1 minute
            data.resetViolations();
        }
    }
    
    public void checkMovement(Player player, PlayerData data, PlayerMoveEvent event) {
        Location from = event.getFrom();
        Location to = event.getTo();
        
        if (to == null) return;
        
        // Fly detection
        if (enabledChecks.contains("fly")) {
            checkFly(player, data);
        }
        
        // Speed detection
        if (enabledChecks.contains("speed")) {
            checkSpeed(player, data);
        }
        
        // NoFall detection (simplified)
        if (enabledChecks.contains("nofall")) {
            checkNoFall(player, data, event);
        }
    }
    
    public void checkCombat(Player attacker, PlayerData data, EntityDamageByEntityEvent event) {
        if (!(event.getEntity() instanceof Player)) return;
        
        Player victim = (Player) event.getEntity();
        
        // KillAura detection
        if (enabledChecks.contains("killaura")) {
            checkKillAura(attacker, data, victim);
        }
        
        // Reach detection
        if (enabledChecks.contains("reach")) {
            checkReach(attacker, data, victim);
        }
        
        // AutoClicker detection
        if (enabledChecks.contains("autoclick")) {
            checkAutoClicker(attacker, data);
        }
    }
    
    private void checkFly(Player player, PlayerData data) {
        if (player.hasPermission("vulcan.bypass.fly")) return;
        
        // Check if player is flying without permission
        boolean isFlying = player.isFlying();
        boolean hasFlyPermission = player.getAllowFlight();
        
        if (isFlying && !hasFlyPermission) {
            data.addViolation("fly");
            data.setFlyingIllegally(true);
            
            if (data.getViolations("fly") >= violationThreshold) {
                flag(player, "Fly", "Flying without permission", data.getViolations("fly"));
            }
        } else {
            data.setFlyingIllegally(false);
        }
        
        // Advanced fly detection based on air time
        if (data.getAirTicks() > maxAirTicks && !player.isFlying() && !player.isGliding()) {
            data.addViolation("fly");
            
            if (data.getViolations("fly") >= violationThreshold) {
                flag(player, "Fly", "Excessive air time (" + data.getAirTicks() + " ticks)", data.getViolations("fly"));
            }
        }
    }
    
    private void checkSpeed(Player player, PlayerData data) {
        if (player.hasPermission("vulcan.bypass.speed")) return;
        
        double speedX = Math.abs(data.getLastSpeedX());
        double speedZ = Math.abs(data.getLastSpeedZ());
        double horizontalSpeed = Math.sqrt(speedX * speedX + speedZ * speedZ);
        
        // Check horizontal speed
        if (horizontalSpeed > maxHorizontalSpeed) {
            data.addViolation("speed");
            data.setSpeedHacking(true);
            
            if (data.getViolations("speed") >= violationThreshold) {
                flag(player, "Speed", "Horizontal speed: " + String.format("%.2f", horizontalSpeed) + " b/s", data.getViolations("speed"));
            }
        } else {
            data.setSpeedHacking(false);
        }
    }
    
    private void checkNoFall(Player player, PlayerData data, PlayerMoveEvent event) {
        if (player.hasPermission("vulcan.bypass.nofall")) return;
        
        Location from = event.getFrom();
        Location to = event.getTo();
        
        if (to == null) return;
        
        // Simple NoFall detection based on fall distance
        double fallDistance = player.getFallDistance();
        double verticalChange = to.getY() - from.getY();
        
        // If player is falling but vertical change is positive (going up)
        if (fallDistance > 3.0 && verticalChange > 0) {
            data.addViolation("nofall");
            flag(player, "NoFall", "Fall distance: " + String.format("%.1f", fallDistance), data.getViolations("nofall"));
        }
    }
    
    private void checkKillAura(Player attacker, PlayerData data, Player victim) {
        if (attacker.hasPermission("vulcan.bypass.killaura")) return;
        
        // Check attack angle consistency
        double currentAngle = data.getLastAttackAngle();
        
        // Simple KillAura detection based on attack pattern
        if (data.getHitsInRow() > 3) {
            // If hitting multiple players in quick succession (simplified check)
            data.addViolation("killaura");
            data.setHasKillAura(true);
            
            if (data.getViolations("killaura") >= violationThreshold) {
                flag(attacker, "KillAura", "Multiple rapid attacks: " + data.getHitsInRow(), data.getViolations("killaura"));
            }
        } else {
            data.setHasKillAura(false);
        }
    }
    
    private void checkReach(Player attacker, PlayerData data, Player victim) {
        if (attacker.hasPermission("vulcan.bypass.reach")) return;
        
        Location attackerLoc = attacker.getEyeLocation();
        Location victimLoc = victim.getEyeLocation();
        
        double distance = attackerLoc.distance(victimLoc);
        
        if (distance > maxReachDistance) {
            data.addViolation("reach");
            data.setHasReachHack(true);
            
            if (data.getViolations("reach") >= violationThreshold) {
                flag(attacker, "Reach", "Distance: " + String.format("%.2f", distance) + " blocks", data.getViolations("reach"));
            }
        } else {
            data.setHasReachHack(false);
        }
    }
    
    private void checkAutoClicker(Player attacker, PlayerData data) {
        if (attacker.hasPermission("vulcan.bypass.autoclick")) return;
        
        int attacksPerSecond = data.getAttacksPerSecond();
        
        if (attacksPerSecond > maxAttacksPerSecond) {
            data.addViolation("autoclick");
            data.setHasAutoclicker(true);
            
            if (data.getViolations("autoclick") >= violationThreshold) {
                flag(attacker, "AutoClicker", "CPS: " + attacksPerSecond, data.getViolations("autoclick"));
            }
        } else {
            data.setHasAutoclicker(false);
        }
    }
    
    private void flag(Player player, String check, String details, int violations) {
        // Send alert
        plugin.getAlertManager().sendAlert(player, check, details, violations);
        
        // Apply punishment if needed
        if (violations >= violationThreshold * 2) { // Double threshold for punishment
            plugin.getPunishmentManager().applyPunishment(player, check, violations);
        }
    }
    
    // Getters for configuration
    public double getMaxHorizontalSpeed() { return maxHorizontalSpeed; }
    public double getMaxVerticalSpeed() { return maxVerticalSpeed; }
    public int getMaxAirTicks() { return maxAirTicks; }
    public int getMaxAttacksPerSecond() { return maxAttacksPerSecond; }
    public double getMaxReachDistance() { return maxReachDistance; }
    public double getMaxAngleChange() { return maxAngleChange; }
    public int getViolationThreshold() { return violationThreshold; }
    public Set<String> getEnabledChecks() { return new HashSet<>(enabledChecks); }
}

package com.mayankpatil.anticheat;

import org.bukkit.entity.Player;
import org.bukkit.event.player.PlayerMoveEvent;
import org.bukkit.event.entity.EntityDamageByEntityEvent;
import org.bukkit.Location;

import java.util.ArrayList;
import java.util.List;

/**
 * PlayerData - Tracks player information for detection purposes
 */
public class PlayerData {
    
    private final Player player;
    private final UUID uuid;
    
    // Movement tracking
    private List<Location> recentLocations = new ArrayList<>();
    private double lastSpeedX = 0;
    private double lastSpeedZ = 0;
    private long lastMoveTime = 0;
    private int airTicks = 0;
    private int groundTicks = 0;
    
    // Combat tracking
    private long lastAttackTime = 0;
    private int attacksPerSecond = 0;
    private double lastAttackAngle = 0;
    private int hitsInRow = 0;
    
    // Violation tracking
    private int flyViolations = 0;
    private int speedViolations = 0;
    private int killAuraViolations = 0;
    private int reachViolations = 0;
    private int autoclickViolations = 0;
    
    // Detection flags
    private boolean isFlyingIllegally = false;
    private boolean isSpeedHacking = false;
    private boolean hasKillAura = false;
    private boolean hasReachHack = false;
    private boolean hasAutoclicker = false;
    
    // Timestamps
    private long lastViolationReset = System.currentTimeMillis();
    private long joinTime = System.currentTimeMillis();
    
    public PlayerData(Player player) {
        this.player = player;
        this.uuid = player.getUniqueId();
        this.lastMoveTime = System.currentTimeMillis();
    }
    
    public void updateMovement(PlayerMoveEvent event) {
        Location from = event.getFrom();
        Location to = event.getTo();
        
        if (to == null) return;
        
        // Calculate speed
        double deltaX = to.getX() - from.getX();
        double deltaZ = to.getZ() - from.getZ();
        double speedX = deltaX * 20; // Convert to blocks per second
        double speedZ = deltaZ * 20;
        
        // Store recent locations (max 20)
        recentLocations.add(to.clone());
        if (recentLocations.size() > 20) {
            recentLocations.remove(0);
        }
        
        // Update tracking data
        lastSpeedX = speedX;
        lastSpeedZ = speedZ;
        lastMoveTime = System.currentTimeMillis();
        
        // Update air/ground ticks
        if (player.isOnGround()) {
            groundTicks++;
            airTicks = 0;
        } else {
            airTicks++;
            groundTicks = 0;
        }
    }
    
    public void updateCombat(EntityDamageByEntityEvent event) {
        long currentTime = System.currentTimeMillis();
        
        // Calculate attacks per second
        if (currentTime - lastAttackTime < 1000) {
            attacksPerSecond++;
        } else {
            attacksPerSecond = 1;
        }
        
        lastAttackTime = currentTime;
        
        // Calculate attack angle if player is looking at entity
        if (event.getEntity() instanceof Player) {
            Player target = (Player) event.getEntity();
            Location eyeLocation = player.getEyeLocation();
            Location targetLocation = target.getEyeLocation();
            
            double dx = targetLocation.getX() - eyeLocation.getX();
            double dy = targetLocation.getY() - eyeLocation.getY();
            double dz = targetLocation.getZ() - eyeLocation.getZ();
            
            double distance = Math.sqrt(dx * dx + dy * dy + dz * dz);
            double angle = Math.toDegrees(Math.acos(dy / distance));
            lastAttackAngle = angle;
        }
        
        // Track hits in a row
        hitsInRow++;
        if (currentTime - lastAttackTime > 2000) { // Reset if 2 seconds without attack
            hitsInRow = 0;
        }
    }

        List<String> completions = new ArrayList<>();
        
        if (args.length == 1) {
            // Main command completions
            List<String> commands = Arrays.asList("help", "reload", "alerts", "bypass", "violations", "reset", "info", "debug");
            for (String cmd : commands) {
                if (cmd.startsWith(args[0].toLowerCase())) {
                    completions.add(cmd);
                }
            }
        } else if (args.length == 2) {
            // Player name completions for violations and reset
            if (args[0].equalsIgnoreCase("violations") || args[0].equalsIgnoreCase("reset")) {
                for (Player player : Bukkit.getOnlinePlayers()) {
                    if (player.getName().toLowerCase().startsWith(args[1].toLowerCase())) {
                        completions.add(player.getName());
                    }
                }
            }
        }
        
        return completions;
    }
}
