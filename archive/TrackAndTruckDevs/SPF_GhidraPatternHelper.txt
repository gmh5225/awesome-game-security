    private ProgramSelection getLiveSelection() {
        CodeViewerService cvs = state.getTool().getService(CodeViewerService.class);
        return cvs != null ? cvs.getCurrentSelection() : null;
    }

    private Address getLiveCursorAddress() {
        CodeViewerService cvs = state.getTool().getService(CodeViewerService.class);
        if (cvs == null || cvs.getCurrentLocation() == null) {
            return null;
        }
        return cvs.getCurrentLocation().getAddress();
    }

    private String formatSearchResultRow(int index, Address addr, Long baseOffset) throws Exception {
        Listing listing = currentProgram.getListing();
        Instruction instr = listing.getInstructionAt(addr);
        Function function = currentProgram.getFunctionManager().getFunctionContaining(addr);

        StringBuilder bytesText = new StringBuilder();
        String instrText = "No instruction";

        if (instr != null) {
            int length = instr.getLength();
            byte[] bytes = new byte[length];
            currentProgram.getMemory().getBytes(addr, bytes);

            for (int i = 0; i < bytes.length; ++i) {
                bytesText.append(String.format("%02x", bytes[i] & 0xff));
                if (i < bytes.length - 1) {
                    bytesText.append(" ");
                }
            }
            instrText = instr.toString();
        }

        String funcText = "no function";
        String funcAddr = "";
        if (function != null) {
            funcText = function.getName();
            funcAddr = function.getEntryPoint().toString();;
        }

        String addrText = htmlEscape(addr.toString());

        // Per-match distance from the search base (cursor-limited searches only).
        String offsetText = "";
        String offsetCell = "";
        if (baseOffset != null) {
            long delta = addr.getOffset() - baseOffset;
            offsetText = String.format("%s0x%02x (%+d)", delta >= 0 ? "+" : "-", Math.abs(delta), delta);
            offsetCell = "  <td style='padding-right: 15px; color: #666;'>%s</td>\n";
        }

        return String.format(
            "<tr>" +
            "  <td style='padding-right: 10px; color: gray;'>[%d]</td>" +
            "  <td style='padding-right: 15px; font-weight: bold;'><a href='%s'>%s</a></td>" +
            offsetCell +
            "  <td style='padding-right: 15px;'><span style='color: gray;'>%s</span> &nbsp; %s</td>" +
            "  <td style='color: gray;'>[<span style='font-weight: bold;'><a href='%s'>%s</a></span>]</td>" +
            "</tr>",
            index,
            addrText,
            addrText,
            offsetText,
            htmlEscape(bytesText.toString()),
            htmlEscape(instrText),
            htmlEscape(funcAddr),
            htmlEscape(funcText)
        );
    }

    private String htmlEscape(String text) {
        if (text == null) {
            return "";
        }
        return text
            .replace("&", "&amp;")
            .replace("<", "&lt;")
            .replace(">", "&gt;")
            .replace("\"", "&quot;")
            .replace("'", "&#39;");
    }

    private void applyGeneratorSearchStatus(SearchSummary summary,
        JLabel genStatusLabel,
        JButton genMoreInfoBtn) {
        int count = summary.matches.size();
        if (count == 1 && !summary.truncated) {
            genStatusLabel.setText("Success: signature is unique.");
            genStatusLabel.setForeground(new Color(21, 128, 61));
            genMoreInfoBtn.setVisible(false);
            Container parent = genMoreInfoBtn.getParent();
            if (parent != null) {
                parent.revalidate();
                parent.repaint();
            }
            return;
        }

        if (count == 0) {
            genStatusLabel.setText("Warning: signature was not found.");
            genStatusLabel.setForeground(new Color(185, 28, 28));
            genMoreInfoBtn.setVisible(false);
            Container parent = genMoreInfoBtn.getParent();
            if (parent != null) {
                parent.revalidate();
                parent.repaint();
            }
            return;
        }

        if (summary.truncated) {
            genStatusLabel.setText("Warning: more than 20 matches found.");
        } else {
            genStatusLabel.setText("Warning: signature is not unique, found " + count + " matches.");
        }
        genStatusLabel.setForeground(new Color(180, 83, 9));
        genMoreInfoBtn.setVisible(true);
        genMoreInfoBtn.setText("More info");
        genMoreInfoBtn.setToolTipText("Open Pattern Finder with this signature");
        Container parent = genMoreInfoBtn.getParent();
        if (parent != null) {
            parent.revalidate();
            parent.repaint();
        }
    }

    private SearchSummary performSearch(java.util.List<ByteMatcher> matchers,
        boolean all,
        boolean selection,
        boolean limit,
        String limitBytes,
        boolean backward,
        int maxResults) throws Exception {
        java.util.List<Long> results = new ArrayList<>();
        Memory memory = currentProgram.getMemory();
        AddressSet searchScope = new AddressSet();

        if (all) {
            for (MemoryBlock block : memory.getBlocks()) {
                if (block.isExecute()) {
                    searchScope.add(block.getStart(), block.getEnd());
                }
            }
        } else if (selection) {
            // Read the current selection at the moment of the click.
            ProgramSelection curSel = getLiveSelection();
            if (curSel != null && !curSel.isEmpty()) {
                searchScope.add(curSel);
            } else {
                throw new Exception("No active memory selection");
            }
        } else if (limit) {
            Address start = getLiveCursorAddress();
            if (start == null) throw new Exception("No cursor location");
            int len = Integer.parseInt(limitBytes);
            if (backward) {
                // Backward search: the window ends at the cursor.
                searchScope.add(start.subtract(len - 1), start);
            } else {
                searchScope.add(start, start.add(len - 1));
            }
        }

        int patternMinLen = 0;
        for (ByteMatcher m : matchers) patternMinLen += m.minCount;

        for (AddressRange range : searchScope.getAddressRanges()) {
            Address start = range.getMinAddress();
            Address end = range.getMaxAddress();
            long totalBytes = end.getOffset() - start.getOffset() + 1;

            if (totalBytes < patternMinLen) continue;

            byte[] buffer = new byte[(int) totalBytes];
            memory.getBytes(start, buffer);

            if (backward) {
                // Start points walk from the cursor backwards while each candidate is
                // still matched forwards (PatternFinder::FindBackward logic). Results
                // come ordered nearest-to-cursor first.
                for (int i = (int) totalBytes - 1; i >= 0; --i) {
                    int[] matchLen = new int[1];
                    if (PatternEngine.matchInternal(buffer, i, matchers, 0, matchLen)) {
                        results.add(start.getOffset() + i);
                        if (results.size() >= maxResults) {
                            return new SearchSummary(results, true);
                        }
                    }
                }
            } else {
                for (int i = 0; i <= totalBytes - patternMinLen; ++i) {
                    int[] matchLen = new int[1];
                    if (PatternEngine.matchInternal(buffer, i, matchers, 0, matchLen)) {
                        results.add(start.getOffset() + i);
                        if (results.size() >= maxResults) {
                            return new SearchSummary(results, true);
                        }
                    }
                }
            }
        }
        return new SearchSummary(results, false);
    }

    private java.util.List<Long> performSearch(java.util.List<ByteMatcher> matchers,
        boolean all,
        boolean selection,
        boolean limit,
        String limitBytes,
        boolean backward) throws Exception {
        return performSearch(matchers, all, selection, limit, limitBytes, backward, Integer.MAX_VALUE).matches;
    }

    private String generateSPFPattern(ProgramSelection selection) throws Exception {
        Listing listing = currentProgram.getListing();
        Memory memory = currentProgram.getMemory();
        StringBuilder sigBuilder = new StringBuilder();

        Address start = selection.getMinAddress();
        Address end = selection.getMaxAddress();

        Address curr = start;
        while (curr != null && curr.compareTo(end) <= 0) {
            Instruction instr = listing.getInstructionAt(curr);
            if (instr == null) {
                curr = curr.add(1);
                continue;
            }

            int length = instr.getLength();
            byte[] bytes = new byte[length];
            memory.getBytes(curr, bytes);

            // Attempt to match templates against the instruction bytes
            String bestTemplate = null;
            int templateLen = 0;

            // Check for REX prefix (0x40-0x4F)
            boolean hasREX = bytes.length > 1
                && (bytes[0] & 0xFF) >= 0x40
                && (bytes[0] & 0xFF) <= 0x4F;
            int rexByte = hasREX ? (bytes[0] & 0xFF) : -1;

            for (Map.Entry<String, String> entry : SPFTemplates.TEMPLATES.entrySet()) {
                String templName = entry.getKey();
                String rawPattern = entry.getValue();
                java.util.List<ByteMatcher> compiled = PatternEngine.compileSignature(rawPattern);

                // Try matching from byte 0 (template may contain explicit REX or bare opcode)
                int[] matchLen = new int[1];
                if (PatternEngine.matchInternal(bytes, 0, compiled, 0, matchLen)) {
                    if (matchLen[0] == bytes.length) {
                        if (matchLen[0] >= templateLen) {
                            bestTemplate = templName;
                            templateLen = matchLen[0];
                        }
                    }
                }

                // If REX prefix present, also try matching from byte 1 (skip REX)
                if (hasREX) {
                    int[] rexMatchLen = new int[1];
                    if (PatternEngine.matchInternal(bytes, 1, compiled, 0, rexMatchLen)) {
                        if (rexMatchLen[0] == bytes.length) {
                            int rexMatchedBytes = rexMatchLen[0] - 1;
                            if (rexMatchedBytes > templateLen) {
                                bestTemplate = String.format("%02X", rexByte) + " " + templName;
                                templateLen = rexMatchLen[0];
                            }
                        }
                    }
                }
            }

            if (bestTemplate != null) {
                sigBuilder.append(bestTemplate);
                // Mask the rest of the bytes if the instruction is longer than the template
                for (int i = templateLen; i < bytes.length; ++i) {
                    sigBuilder.append(" ?");
                }
            } else {
                // If no template matches, we fallback to exact bytes (except RIP-relative offsets)
                // In Ghidra, we check if there are reference operands to mask
                boolean[] mask = new boolean[length];
                int firstByte = bytes[0] & 0xFF;
                boolean isBranchRel32 = (firstByte == 0xE8 || firstByte == 0xE9) ||
                    (bytes.length > 1 && firstByte == 0x0F &&
                     (bytes[1] & 0xFF) >= 0x80 && (bytes[1] & 0xFF) <= 0x8F);
                boolean isBranchRel8 = firstByte == 0xEB ||
                    (firstByte >= 0x70 && firstByte <= 0x7F);
                boolean foundMemRef = false;

                for (int op = 0; op < instr.getNumOperands(); ++op) {
                    if (instr.getOperandRefType(op) != null) {
                        if (isBranchRel32) {
                            if (length > 4) {
                                for (int i = length - 4; i < length; ++i) mask[i] = true;
                            }
                        } else if (isBranchRel8) {
                            if (length > 1) {
                                for (int i = length - 1; i < length; ++i) mask[i] = true;
                            }
                        } else {
                            foundMemRef = true;
                        }
                    } else if (instr.getScalar(op) != null) {
                        ghidra.program.model.scalar.Scalar s = instr.getScalar(op);
                        int byteCount = Math.max(1, (s.bitLength() + 7) / 8);
                        int startOffset = Math.max(0, length - byteCount);
                        if (startOffset >= 1) {
                            for (int i = startOffset; i < length; ++i) mask[i] = true;
                        }
                    }
                }

                if (foundMemRef) {
                    int[] dispInfo = PatternEngine.getDisplacementInfo(bytes);
                    if (dispInfo != null) {
                        for (int i = dispInfo[0]; i < dispInfo[0] + dispInfo[1]; ++i) {
                            if (i < length) mask[i] = true;
                        }
                    }
                }

                for (int i = 0; i < bytes.length; ++i) {
                    if (mask[i]) {
                        sigBuilder.append("?");
                    } else {
                        sigBuilder.append(String.format("%02X", bytes[i]));
                    }
                    if (i < bytes.length - 1) sigBuilder.append(" ");
                }
            }

            sigBuilder.append(" ");
            curr = curr.add(length);
        }

        return sigBuilder.toString().trim();
    }

import java.util.ArrayList;
import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Pattern engine for GhidraPatternHelper: compiles signature strings into byte
 * matcher trees and matches them over raw byte arrays. Mirrors the SPF framework's
 * PatternFinder.cpp semantics (EXACT / WILDCARD / RANGE / LIST / SIB_CONDITIONAL /
 * GROUP / ALTERNATION).
 *
 * Template definitions live in {@link SPFTemplates}; this file holds pure
 * parsing/matching logic shared by the Ghidra script UI.
 */
final class PatternEngine {

    private PatternEngine() {
    }

    private static String normalizeSignature(String signature) {
        StringBuilder result = new StringBuilder();
        boolean insideBrackets = false;
        for (int i = 0; i < signature.length(); i++) {
            char c = signature.charAt(i);
            if (c == '[') {
                insideBrackets = true;
                result.append(c);
            } else if (c == ']') {
                insideBrackets = false;
                result.append(c);
            } else if (c == ' ' || c == '\t') {
                if (!insideBrackets) {
                    result.append(' ');
                }
            } else {
                result.append(c);
            }
        }
        return result.toString().replaceAll("\\s+", " ");
    }

    private static java.util.List<Integer> parseNibble(String str) {
        java.util.List<Integer> result = new ArrayList<>();
        if (str.isEmpty()) return result;
        // Wildcard
        if (str.equals("?")) {
            for (int i = 0; i < 16; ++i) result.add(i);
            return result;
        }
        // Bracketed expression – could be a range (e.g., [0-3]) or a list (e.g., [b|9])
        if (str.startsWith("[") && str.endsWith("]")) {
            String inner = str.substring(1, str.length() - 1);
            // List case – split by '|'
            if (inner.contains("|")) {
                String[] parts = inner.split("\\|");
                for (String part : parts) {
                    if (part.isEmpty()) continue;
                    // Each part should be a single hex digit
                    int val = Integer.parseInt(part, 16);
                    result.add(val);
                }
                return result;
            }
            // Range case – exactly three characters like "0-3"
            if (inner.length() == 3 && inner.charAt(1) == '-') {
                char startChar = Character.toLowerCase(inner.charAt(0));
                char endChar = Character.toLowerCase(inner.charAt(2));
                int startVal = (startChar >= '0' && startChar <= '9') ? (startChar - '0') : (startChar - 'a' + 10);
                int endVal = (endChar >= '0' && endChar <= '9') ? (endChar - '0') : (endChar - 'a' + 10);
                if (startVal <= endVal) {
                    for (int i = startVal; i <= endVal; ++i) {
                        result.add(i);
                    }
                }
                return result;
            }
            // If it doesn't match list or range, fall through to treat as exact hex later
        }
        // Single hex digit (exact value)
        char c = Character.toLowerCase(str.charAt(0));
        if ((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f')) {
            int val = (c >= '0' && c <= '9') ? (c - '0') : (c - 'a' + 10);
            result.add(val);
        }
        return result;
    }

    private static boolean trySplitNibbles(String part, String[] splitResult) {
        if (part.isEmpty()) return false;
        int idx = 0;
        String highStr;
        if (part.charAt(idx) == '[') {
            int close = part.indexOf(']', idx);
            if (close == -1) return false;
            highStr = part.substring(idx, close + 1);
            idx = close + 1;
        } else {
            highStr = part.substring(idx, idx + 1);
            idx += 1;
        }
        String lowStr;
        if (idx < part.length()) {
            if (part.charAt(idx) == '[') {
                int close = part.indexOf(']', idx);
                if (close == -1) return false;
                lowStr = part.substring(idx, close + 1);
            } else {
                lowStr = part.substring(idx, idx + 1);
            }
        } else {
            return false;
        }
        splitResult[0] = highStr;
        splitResult[1] = lowStr;
        return true;
    }

    /**
     * Splits an alternation group body on its top-level '|' separators.
     * Depth-aware: '|' inside nested "[...]" (byte OR-lists like "[b|9]")
     * or nested "{...}" groups is not a separator.
     */
    private static java.util.List<String> splitTopLevelAlternatives(String body) {
        java.util.List<String> branches = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        int depth = 0;
        for (int k = 0; k < body.length(); k++) {
            char c = body.charAt(k);
            if (c == '[' || c == '{') {
                depth++;
            } else if (c == ']' || c == '}') {
                depth--;
            }
            if (c == '|' && depth == 0) {
                branches.add(current.toString());
                current.setLength(0);
            } else {
                current.append(c);
            }
        }
        branches.add(current.toString());
        return branches;
    }

    static java.util.List<ByteMatcher> compileSignature(String signature) {
        java.util.List<ByteMatcher> matchers = new ArrayList<>();
        String templated = SPFTemplates.replaceTemplates(signature);
        String normalized = normalizeSignature(templated);

        Pattern rangeRegex = Pattern.compile("\\[([0-9A-Fa-f]{1,2})-([0-9A-Fa-f]{1,2})\\]");
        Pattern countRegex = Pattern.compile("\\[([0-9]+)-([0-9]+)\\?\\]");

        // Tokenize, treating brace-wrapped groups "{...}" (and their trailing "?") as single tokens.
        java.util.List<String> tokens = new ArrayList<>();
        int i = 0;
        while (i < normalized.length()) {
            char c = normalized.charAt(i);
            if (c == ' ') {
                i++;
                continue;
            }
            if (c == '{') {
                int depth = 0;
                int j = i;
                for (; j < normalized.length(); j++) {
                    char gc = normalized.charAt(j);
                    if (gc == '{') depth++;
                    else if (gc == '}') {
                        depth--;
                        if (depth == 0) break;
                    }
                }
                int end = (j < normalized.length() && j + 1 < normalized.length()
                    && normalized.charAt(j + 1) == '?') ? j + 2 : j + 1;
                tokens.add(normalized.substring(i, end));
                i = end;
            } else {
                int j = normalized.indexOf(' ', i);
                if (j == -1) j = normalized.length();
                tokens.add(normalized.substring(i, j));
                i = j;
            }
        }

        for (String part : tokens) {
            ByteMatcher bm = null;
            if (part.isEmpty()) {
                continue;
            }

            // Brace group { ... } or alternation {A|B}; trailing '?' makes it optional.
            if (part.charAt(0) == '{') {
                boolean groupOptional = part.charAt(part.length() - 1) == '?';
                int innerLen = part.length() - (groupOptional ? 3 : 2); // strip '{', '}' and optional '?'
                String inner = part.substring(1, 1 + innerLen);
                java.util.List<String> branches = splitTopLevelAlternatives(inner);

                // A stray separator (e.g. "{A|}") is a typo -- fail loudly instead of silently
                // narrowing the pattern to the surviving branch.
                boolean hasEmptyBranch = false;
                for (String branch : branches) {
                    if (branch.trim().isEmpty()) hasEmptyBranch = true;
                }
                if (hasEmptyBranch && branches.size() > 1) {
                    System.out.println("GhidraPatternHelper: empty alternation branch in '{" + inner + "}'");
                    matchers.clear();
                    return matchers;
                }

                bm = new ByteMatcher();
                bm.optional = groupOptional;
                bm.minCount = 0;
                bm.maxCount = 0;
                if (branches.size() <= 1) {
                    // Single sequence -- plain atomic group.
                    bm.type = ByteMatcher.Type.GROUP;
                    bm.group = compileSignature(branches.get(0));
                    for (ByteMatcher sub : bm.group) {
                        bm.minCount += sub.minCount;
                        bm.maxCount += sub.maxCount;
                    }
                } else {
                    // Alternation -- exactly one of the listed sequences must match.
                    bm.type = ByteMatcher.Type.ALTERNATION;
                    for (String branch : branches) {
                        java.util.List<ByteMatcher> compiled = compileSignature(branch);
                        int branchMin = 0;
                        int branchMax = 0;
                        for (ByteMatcher sub : compiled) {
                            branchMin += sub.minCount;
                            branchMax += sub.maxCount;
                        }
                        if (bm.alternatives.isEmpty() || branchMin < bm.minCount) bm.minCount = branchMin;
                        if (bm.alternatives.isEmpty() || branchMax > bm.maxCount) bm.maxCount = branchMax;
                        bm.alternatives.add(compiled);
                    }
                }
                if (groupOptional) bm.minCount = 0;
                matchers.add(bm);
                continue;
            }

            // SIB conditional token — determines SIB presence from preceding ModRM byte
            if (part.equals("[SIB?]")) {
                bm = new ByteMatcher();
                bm.type = ByteMatcher.Type.SIB_CONDITIONAL;
                bm.minCount = 0;
                bm.maxCount = 1;
                matchers.add(bm);
                continue;
            }

            // Check for count pattern [min-max?] FIRST (e.g. [0-2?] = 0 or 2 wildcard bytes).
            // Must happen before trailing ? stripping, since the ? is inside the brackets.
            Matcher mCount = countRegex.matcher(part);
            if (mCount.matches()) {
                bm = new ByteMatcher();
                bm.type = ByteMatcher.Type.WILDCARD;
                bm.minCount = Integer.parseInt(mCount.group(1));
                bm.maxCount = Integer.parseInt(mCount.group(2));
                matchers.add(bm);
                continue;
            }

            boolean isOptional = false;
            if (part.endsWith("?") && part.length() > 1 && !part.equals("??")) {
                isOptional = true;
                part = part.substring(0, part.length() - 1);
            }

            if (part.equals("?") || part.equals("??")) {
                bm = new ByteMatcher();
                bm.type = ByteMatcher.Type.WILDCARD;
                bm.optional = isOptional;
                matchers.add(bm);
            } else if (part.startsWith("[") && part.endsWith("]") && part.contains("|")) {
                bm = new ByteMatcher();
                bm.type = ByteMatcher.Type.LIST;
                bm.optional = isOptional;
                String listContent = part.substring(1, part.length() - 1);
                String[] listItems = listContent.split("\\|");
                for (String item : listItems) {
                    if (item.isEmpty()) continue;
                    if (item.contains("-")) {
                        String[] bounds = item.split("-");
                        int startVal = Integer.parseInt(bounds[0], 16);
                        int endVal = Integer.parseInt(bounds[1], 16);
                        for (int val = startVal; val <= endVal; ++val) {
                            bm.values.add(val);
                        }
                    } else {
                        bm.values.add(Integer.parseInt(item, 16));
                    }
                }
                matchers.add(bm);
            } else if (rangeRegex.matcher(part).matches()) {
                Matcher mRange = rangeRegex.matcher(part);
                mRange.matches();
                bm = new ByteMatcher();
                bm.type = ByteMatcher.Type.RANGE;
                bm.min = Integer.parseInt(mRange.group(1), 16);
                bm.max = Integer.parseInt(mRange.group(2), 16);
                bm.optional = isOptional;
                matchers.add(bm);
            } else {
                // Fallback handling for exact bytes or nibble split
                String[] split = new String[2];
                if (trySplitNibbles(part, split)) {
                    java.util.List<Integer> highVals = parseNibble(split[0]);
                    java.util.List<Integer> lowVals = parseNibble(split[1]);
                    if (!highVals.isEmpty() && !lowVals.isEmpty()) {
                        bm = new ByteMatcher();
                        bm.optional = isOptional;
                        if (highVals.size() == 1 && lowVals.size() == 1) {
                            bm.type = ByteMatcher.Type.EXACT;
                            bm.min = highVals.get(0) * 16 + lowVals.get(0);
                        } else {
                            bm.type = ByteMatcher.Type.LIST;
                            for (int h : highVals) {
                                for (int l : lowVals) {
                                    bm.values.add(h * 16 + l);
                                }
                            }
                        }
                        matchers.add(bm);
                        continue;
                    }
                }
                // Fallback
                bm = new ByteMatcher();
                bm.type = ByteMatcher.Type.EXACT;
                bm.min = Integer.parseInt(part, 16);
                bm.optional = isOptional;
                matchers.add(bm);
            }
        }
        return matchers;
    }

    // === SEARCH ALGORITHM ===

    static boolean matchInternal(byte[] data, int dataIdx, java.util.List<ByteMatcher> matchers, int matcherIdx, int[] matchLen) {
        if (matcherIdx == matchers.size()) {
            matchLen[0] = dataIdx;
            return true;
        }

        ByteMatcher m = matchers.get(matcherIdx);

        // GROUP — optional instruction. Try matching the whole group first (greedy),
        // then fall back to skipping it entirely if it is optional.
        if (m.type == ByteMatcher.Type.GROUP) {
            int[] subMatchLen = new int[1];
            if (matchInternal(data, dataIdx, m.group, 0, subMatchLen)) {
                if (m.optional) {
                    // Option 1a: consume the group and continue.
                    if (matchInternal(data, subMatchLen[0], matchers, matcherIdx + 1, matchLen)) return true;
                    // Option 1b: the group matched but the tail failed — skip the group.
                    if (matchInternal(data, dataIdx, matchers, matcherIdx + 1, matchLen)) return true;
                    return false;
                }
                return matchInternal(data, subMatchLen[0], matchers, matcherIdx + 1, matchLen);
            }
            // Group did not match in-place.
            if (m.optional) return matchInternal(data, dataIdx, matchers, matcherIdx + 1, matchLen);
            return false;
        }

        // ALTERNATION -- exactly one of the listed sub-sequences. Branches are tried
        // left to right; a matching branch continues with the remaining matchers so
        // a failed tail can still backtrack into the next branch.
        if (m.type == ByteMatcher.Type.ALTERNATION) {
            for (java.util.List<ByteMatcher> alt : m.alternatives) {
                int[] subMatchLen = new int[1];
                if (!matchInternal(data, dataIdx, alt, 0, subMatchLen)) continue;
                if (matchInternal(data, subMatchLen[0], matchers, matcherIdx + 1, matchLen)) return true;
            }
            // No branch produced a full tail match -- skip the whole group if optional.
            if (m.optional) return matchInternal(data, dataIdx, matchers, matcherIdx + 1, matchLen);
            return false;
        }

        // SIB conditional — check preceding ModRM byte
        if (m.type == ByteMatcher.Type.SIB_CONDITIONAL) {
            if (dataIdx > 0) {
                int modrm = data[dataIdx - 1] & 0xFF;
                boolean hasSib = (modrm & 0x07) == 0x04;
                if (hasSib) {
                    if (dataIdx < data.length) {
                        if (matchInternal(data, dataIdx + 1, matchers, matcherIdx + 1, matchLen))
                            return true;
                    }
                    return false;
                } else {
                    return matchInternal(data, dataIdx, matchers, matcherIdx + 1, matchLen);
                }
            }
            // Fallback: no previous byte — try greedy (1 byte) then 0
            if (dataIdx < data.length && matchInternal(data, dataIdx + 1, matchers, matcherIdx + 1, matchLen))
                return true;
            return matchInternal(data, dataIdx, matchers, matcherIdx + 1, matchLen);
        }

        if (m.optional) {
            // Option 1: Match 1 byte and proceed (Greedy match)
            if (dataIdx < data.length && m.matches(data[dataIdx])) {
                if (matchInternal(data, dataIdx + 1, matchers, matcherIdx + 1, matchLen)) {
                    return true;
                }
            }
            // Option 2: Skip this optional matcher (consume 0 bytes)
            if (matchInternal(data, dataIdx, matchers, matcherIdx + 1, matchLen)) {
                return true;
            }
            return false;
        }

        if (dataIdx + m.minCount > data.length) return false;

        if (m.minCount == m.maxCount) {
            for (int i = 0; i < m.minCount; ++i) {
                if (!m.matches(data[dataIdx + i])) return false;
            }
            return matchInternal(data, dataIdx + m.minCount, matchers, matcherIdx + 1, matchLen);
        }

        for (int count = m.maxCount; count >= m.minCount; --count) {
            if (dataIdx + count > data.length) continue;
            if (matchInternal(data, dataIdx + count, matchers, matcherIdx + 1, matchLen)) {
                return true;
            }
        }
        return false;
    }

    // === MODRM DISPLACEMENT ANALYZER ===

    static int[] getDisplacementInfo(byte[] bytes) {
        if (bytes == null || bytes.length < 2) return null;

        int idx = 0;

        // Skip legacy prefixes (66, F0, F2, F3, segment overrides, address size)
        while (idx < bytes.length) {
            int b = bytes[idx] & 0xFF;
            if (b == 0x66 || b == 0xF0 || b == 0xF2 || b == 0xF3 ||
                b == 0x2E || b == 0x36 || b == 0x3E || b == 0x26 ||
                b == 0x64 || b == 0x65 || b == 0x67) {
                idx++;
            } else {
                break;
            }
        }

        // Skip REX prefix
        if (idx < bytes.length && (bytes[idx] & 0xFF) >= 0x40 && (bytes[idx] & 0xFF) <= 0x4F) {
            idx++;
        }

        // Skip opcode bytes
        if (idx >= bytes.length) return null;

        // Handle 0F prefix for 2/3-byte opcodes
        if ((bytes[idx] & 0xFF) == 0x0F) {
            idx++;
            if (idx >= bytes.length) return null;
            // 3-byte opcodes: 0F 38 xx or 0F 3A xx
            int b = bytes[idx] & 0xFF;
            if (b == 0x38 || b == 0x3A) {
                idx++;
                if (idx >= bytes.length) return null;
            }
        }

        // Consume opcode byte
        idx++;
        if (idx >= bytes.length) return null;

        // Parse ModRM
        int modrm = bytes[idx] & 0xFF;
        int mod = (modrm >> 6) & 3;
        int rm = modrm & 7;

        if (mod == 3) return null; // register target, no displacement

        int dispOffset = idx + 1; // bytes after ModRM

        // Handle SIB (ModRM R/M = 100)
        if (rm == 4) {
            if (dispOffset >= bytes.length) return null;
            int sibBase = bytes[dispOffset] & 7;
            dispOffset++; // consume SIB byte

            if (mod == 0) {
                // Mod=00 + SIB base=101 → disp32 follows
                if (sibBase == 5) {
                    if (dispOffset + 4 > bytes.length) return null;
                    return new int[] { dispOffset, 4 };
                }
                // Mod=00 + SIB base != 101 → no displacement
                return null;
            }
        } else if (mod == 0 && rm == 5) {
            // Mod=00 + R/M=101 → RIP-relative disp32
            if (dispOffset + 4 > bytes.length) return null;
            return new int[] { dispOffset, 4 };
        }

        // Mod=01 → disp8, Mod=10 → disp32
        if (mod == 1) {
            if (dispOffset + 1 > bytes.length) return null;
            return new int[] { dispOffset, 1 };
        }
        if (mod == 2) {
            if (dispOffset + 4 > bytes.length) return null;
            return new int[] { dispOffset, 4 };
        }

        return null;
    }

    // === TEMPLATE-AWARE MASKING (for Raw Pattern Generator) ===

    private static boolean matchWithConsumption(byte[] data, int dataIdx,
        java.util.List<ByteMatcher> matchers, int matcherIdx,
        boolean[] mask, java.util.List<Integer> maskTrail, int[] matchLen) {
        if (matcherIdx == matchers.size()) {
            matchLen[0] = dataIdx;
            return true;
        }

        ByteMatcher m = matchers.get(matcherIdx);

        // GROUP -- optional instruction. Try matching the whole group first (greedy),
        // then fall back to skipping it entirely if it is optional.
        if (m.type == ByteMatcher.Type.GROUP) {
            int[] subMatchLen = new int[1];
            if (matchWithConsumption(data, dataIdx, m.group, 0, mask, maskTrail, subMatchLen)) {
                int trailMark = maskTrail.size();
                if (m.optional) {
                    // Option 1a: consume the group and continue.
                    if (matchWithConsumption(data, subMatchLen[0], matchers, matcherIdx + 1, mask, maskTrail, matchLen)) return true;
                    rollbackMask(mask, maskTrail, trailMark);
                    // Option 1b: the group matched but the tail failed -- skip the group.
                    if (matchWithConsumption(data, dataIdx, matchers, matcherIdx + 1, mask, maskTrail, matchLen)) return true;
                    return false;
                }
                return matchWithConsumption(data, subMatchLen[0], matchers, matcherIdx + 1, mask, maskTrail, matchLen);
            }
            // Group did not match in-place.
            if (m.optional) return matchWithConsumption(data, dataIdx, matchers, matcherIdx + 1, mask, maskTrail, matchLen);
            return false;
        }

        // ALTERNATION -- exactly one branch; failed tails backtrack into the next branch.
        if (m.type == ByteMatcher.Type.ALTERNATION) {
            for (java.util.List<ByteMatcher> alt : m.alternatives) {
                int[] subMatchLen = new int[1];
                if (!matchWithConsumption(data, dataIdx, alt, 0, mask, maskTrail, subMatchLen)) continue;
                int trailMark = maskTrail.size();
                if (matchWithConsumption(data, subMatchLen[0], matchers, matcherIdx + 1, mask, maskTrail, matchLen)) return true;
                rollbackMask(mask, maskTrail, trailMark);
            }
            // No branch produced a full tail match -- skip the whole group if optional.
            if (m.optional) return matchWithConsumption(data, dataIdx, matchers, matcherIdx + 1, mask, maskTrail, matchLen);
            return false;
        }

        // SIB conditional -- check preceding ModRM byte
        if (m.type == ByteMatcher.Type.SIB_CONDITIONAL) {
            boolean hasSib = false;
            if (dataIdx > 0) hasSib = (data[dataIdx - 1] & 0x07) == 0x04;
            if (hasSib) {
                if (dataIdx >= data.length) return false;
                return matchWithConsumption(data, dataIdx + 1, matchers, matcherIdx + 1, mask, maskTrail, matchLen);
            }
            // Fallback: no previous byte -- try greedy (1 byte) then 0
            if (dataIdx == 0 && dataIdx < data.length
                && matchWithConsumption(data, dataIdx + 1, matchers, matcherIdx + 1, mask, maskTrail, matchLen)) return true;
            return matchWithConsumption(data, dataIdx, matchers, matcherIdx + 1, mask, maskTrail, matchLen);
        }

        if (m.optional) {
            // Option 1: Match 1 byte and proceed (Greedy match)
            if (dataIdx < data.length && m.matches(data[dataIdx])) {
                int trailMark = maskTrail.size();
                if (m.type == ByteMatcher.Type.WILDCARD) markMask(mask, maskTrail, dataIdx, 1);
                if (matchWithConsumption(data, dataIdx + 1, matchers, matcherIdx + 1, mask, maskTrail, matchLen)) return true;
                rollbackMask(mask, maskTrail, trailMark);
            }
            // Option 2: Skip this optional matcher (consume 0 bytes)
            return matchWithConsumption(data, dataIdx, matchers, matcherIdx + 1, mask, maskTrail, matchLen);
        }

        if (dataIdx + m.minCount > data.length) return false;

        if (m.minCount == m.maxCount) {
            int count = m.minCount;
            for (int i = 0; i < count; ++i) {
                if (!m.matches(data[dataIdx + i])) return false;
            }
            int trailMark = maskTrail.size();
            if (count > 0 && m.type == ByteMatcher.Type.WILDCARD) markMask(mask, maskTrail, dataIdx, count);
            if (matchWithConsumption(data, dataIdx + count, matchers, matcherIdx + 1, mask, maskTrail, matchLen)) return true;
            rollbackMask(mask, maskTrail, trailMark);
            return false;
        }

        for (int count = m.maxCount; count >= m.minCount; --count) {
            if (dataIdx + count > data.length) continue;
            int trailMark = maskTrail.size();
            if (m.type == ByteMatcher.Type.WILDCARD) markMask(mask, maskTrail, dataIdx, count);
            if (matchWithConsumption(data, dataIdx + count, matchers, matcherIdx + 1, mask, maskTrail, matchLen)) return true;
            rollbackMask(mask, maskTrail, trailMark);
        }
        return false;
    }

    private static void markMask(boolean[] mask, java.util.List<Integer> maskTrail, int start, int count) {
        for (int i = 0; i < count; ++i) {
            mask[start + i] = true;
            maskTrail.add(start + i);
        }
    }

    private static void rollbackMask(boolean[] mask, java.util.List<Integer> maskTrail, int trailMark) {
        while (maskTrail.size() > trailMark) {
            mask[maskTrail.remove(maskTrail.size() - 1)] = false;
        }
    }

    static boolean[] buildMaskFromTemplate(byte[] bytes, java.util.List<ByteMatcher> matchers) {
        boolean[] mask = new boolean[bytes.length];
        java.util.List<Integer> maskTrail = new ArrayList<>();
        int[] matchLen = new int[1];

        if (!matchWithConsumption(bytes, 0, matchers, 0, mask, maskTrail, matchLen)) return null;
        if (matchLen[0] != bytes.length) return null;

        for (int i = matchLen[0]; i < bytes.length; ++i) mask[i] = true;
        return mask;
    }
}

/**
 * Single-byte matcher node used by the pattern engine. Deliberately top-level
 * (not nested in {@link PatternEngine}) so the Ghidra script UI can reference it
 * directly as {@code ByteMatcher} / {@code ByteMatcher.Type}.
 */
class ByteMatcher {
    enum Type { EXACT, WILDCARD, RANGE, LIST, SIB_CONDITIONAL, GROUP, ALTERNATION }
    Type type;
    int min;
    int max;
    int minCount = 1;
    int maxCount = 1;
    boolean optional = false;
    java.util.List<Integer> values = new ArrayList<>();
    java.util.List<ByteMatcher> group = new ArrayList<>();
    java.util.List<java.util.List<ByteMatcher>> alternatives = new ArrayList<>();

    boolean matches(byte b) {
        int val = b & 0xFF;
        if (type == Type.WILDCARD) {
            return true;
        }
        if (type == Type.EXACT) {
            return val == min;
        }
        if (type == Type.RANGE) {
            return val >= min && val <= max;
        }
        if (type == Type.LIST) {
            return values.contains(val);
        }
        return false;
    }
}

 * SPF pattern templates: bracketed template name -> raw byte pattern.
 * Shared with GhidraPatternHelper via {@link PatternEngine} (SPFTemplates.replaceTemplates()).
 * Edit/rebuild is picked up automatically: Ghidra compiles all .java files of this

final class SPFTemplates {

    static final Map<String, String> TEMPLATES = new LinkedHashMap<>();

    private SPFTemplates() {

    static String replaceTemplates(String signature) {
        java.util.List<Map.Entry<String, String>> sorted = new ArrayList<>(TEMPLATES.entrySet());
        sorted.sort((a, b) -> Integer.compare(b.getKey().length(), a.getKey().length()));
        for (Map.Entry<String, String> entry : sorted) {
            int idx;
            final String name = entry.getKey();
            final String replacement = entry.getValue();
            while ((idx = signature.indexOf(name)) != -1) {
                int afterName = idx + name.length();
                // "[<template>]?" -> whole instruction is optional: wrap in an optional group.
                if (afterName < signature.length() && signature.charAt(afterName) == '?') {
                    signature = signature.substring(0, idx)
                        + "{" + replacement + "}?"
                        + signature.substring(afterName + 1);
                } else {
                    signature = signature.substring(0, idx)
                        + replacement
                        + signature.substring(idx + name.length());
                }
            }
        }
        return signature;
    }

    static {
  // =========================================================================
  // ===							MOV INSTRUCTIONS                         ===
  // =========================================================================
  //
  // ModRM byte (Mod /Reg /R/M) — upper 2 bits = Mod addressing mode:
  //   Mod=00 [00-3f] — [r64]                 (no displacement)
  //   Mod=01 [40-7f] — [r64+disp8]           (8-bit displacement)
  //   Mod=10 [80-bf] — [r64+disp32]          (32-bit displacement)
  //   Mod=11 [c0-ff] — register-register
  //   R/M=100 (0x4)  — SIB byte follows      (index*scale + base)
  //   R/M=101 (0x5)  — RIP-relative (Mod=00) or disp32/64 (Mod=00)
  //
  // SIB byte (Scale:Index:Base):
  //   Scale  — [7:6] = 00(x1) 01(x2) 10(x4) 11(x8)
  //   Index  — [5:3] = register or 100=RSP (no index)
  //   Base   — [2:0] = register or 101=RBP (no base, disp follows)
  //
  // REX prefix: byte 0x40-0x4F, bit W (bit 3) = 1 → 64-bit operand

  // --- MOV: Register ← Immediate ---
        TEMPLATES.put("[MOV r8, imm8]",           "[b0-b7] ?");
        TEMPLATES.put("[MOV r32, imm32]",         "4[0-f]? [b8-bf] ? ? ? ?");
        TEMPLATES.put("[MOV r64, imm64]",         "4[8-f] [b8-bf] ? ? ? ? ? ? ? ?");
        TEMPLATES.put("[MOV r64, imm32]",         "4[8-f] c7 [c0-ff] ? ? ? ?");

  // --- MOV: Register ← Memory (RIP-Relative) ---
        TEMPLATES.put("[MOV r8, [rip+off32]]",    "8a 05 ? ? ? ?");

        TEMPLATES.put("[MOV r8, [r64+off32]]",    "8a [80-bf] [SIB?] ? ? ? ?");

## Overview

`SPF Ghidra Pattern Helper` is a Ghidra script for creating and testing byte patterns.

The script was created to make pattern work faster, cleaner, and less manual:

- generate patterns directly from selected instructions
- search with flexible byte syntax
- quickly move between generated output and search results
- avoid copying patterns by hand between tabs and tools

> [!TIP]
> For best results, select **stable instruction sequences** (3+ instructions) rather than tiny or highly repeated fragments. A single-instruction pattern often matches too many locations.

> [!TIP]
> If you generate a pattern that matches hundreds of results, try adding more instructions to the selection. The auto-verify feature will warn you when a match count seems too high.

---

## SPF Framework Integration

This script was created for [SPF Framework](https://github.com/TrackAndTruckDevs/SPF-Framework) to make plugin development faster and less manual.

For **SPF-Framework**, it supports compact template-style signatures like:

```text
[MOV [r64+off8], r32] [XOR r32, r32] [CALL rel32] [MOVSS [r64+off8], xmm] [LEA r64, [r64+off8]] ...
```

This makes it easier to work with the simplified signature style used by SPF-based plugins.

> [!NOTE]
> Template signatures are converted to raw patterns internally. If no template matches an instruction, the generator falls back to automatic displacement masking — the output still works with SPF-Framework's `PatternFinder`.

## Installation

### Script Directory

The script ships as a `GhidraPatternHelper/` folder with several `.java` files that work together. Keep all files in this one folder: Ghidra compiles them as a single bundle, and the folder name is their Java package name.

1. Copy the whole `GhidraPatternHelper` folder into any of your Ghidra script directories.
2. Open Ghidra.
3. Go to `Window -> Script Manager`.
4. Open `Script Directories` (Manage Script Directories).
5. Add exactly this folder — `<script directory>/GhidraPatternHelper`.
6. Refresh the script list.
7. Run `GhidraPatternHelper`.

### One-Time Run from Script Manager

If the script is already in a configured Ghidra script directory:

1. Open `Window -> Script Manager`.
2. Find `GhidraPatternHelper`.
3. Run it directly from the list.

### Output Fields Reference

| Field | Description | When to Use |
|---|---|---|
| **SPF Pattern** | Compact template names like `[CMP r64, [r64+off32]]` separated by spaces. Each instruction is replaced by a named template if one matches. | Readable signatures for SPF-Framework `PatternFinder`. Ideal for use in SPF-Framework.. |
| **Extended Signature** | Template raw patterns (same ranges as `PatternTemplates.hpp`) joined together. Shows the actual masked pattern like `4[8-f] 3b [80-bf] [SIB?] ? ? ? ?`. REX prefix is emitted explicitly when needed. | When you want to see (or tweak) the exact range-based pattern before using it. Fully compatible with SPF-Framework. |
| **Raw Pattern (Masked Hex)** | Specific hex bytes with `?` for masked values. Example: `48 3B 98 ? ? ? ? 0F 83 ? ? ? ?`. Uses the instruction's actual REX byte. | Quick copy-paste for most workflows. Balances readability with precision. |
| **Raw Bytes (Pure Hex)** | Exact hex bytes with no masking. Example: `48 3B 98 80 00 00 00 0F 83 0A 02 00 00`. | Debugging, diffing, or when you need the literal instruction bytes. |
| **Comment** | Formatted block with addresses and disassembly for documentation. | Pasting into code as a reference comment above a signature. |

> [!WARNING]
> An **Extended Signature** may include a variable REX prefix (`4[8-f]`) while the **Raw Pattern** uses the literal REX byte from the selection. Both are valid — the extended form is broader, the raw form is more specific to the original instruction.

## Search Syntax Reference

The finder supports a richer byte pattern syntax than the standard Ghidra pattern search.

| Syntax | Description | Examples |
|---|---|---|
| `48 8B CF` | **Exact bytes** — match only these exact byte values in order. | `48 8B CF` matches only `48 8B CF`. |
| `?` or `??` | **Wildcard** — match any single byte. | `?` = one byte, `??` = same meaning. |
| `[1-5?]` | **Length wildcard** — match 1 to 5 bytes of any value. | `[1-5?]` can cover 1, 2, 3, 4, or 5 unknown bytes. |
| `45?` | **Optional exact byte** — byte `45` may be present or omitted. | `40 45? 50` matches `40 50` and `40 45 50`. |
| `[40-4C]` | **Byte range** — match any byte between 0x40 and 0x4C inclusive. | `[80\|BF]` matches 80 or BF. `[80\|BF\|C0]` matches 80, BF, or C0. `[80\|BF\|C0-C5]` matches 80, BF, or any byte from C0 to C5. |
| `[4-7]8` | **Nibble range (high)** — high nibble varies, low nibble fixed to 8. | `[4-7]8` matches 48, 58, 68, 78. |
| `[4-7][8-B]` | **Nibble range (both)** — both nibbles vary. | `[4-7][8-B]` matches 48..4B, 58..5B, 68..6B, 78..7B. |
| `[SIB?]` | **Optional SIB byte** — 0 or 1 byte after ModRM when R/M=100. | Used internally by templates to cover both SIB and no-SIB forms. |
| `[LEA r64, [r64+off8]]?` | **Optional template instruction** — trailing `?` right after a bracketed template name: the whole instruction is either present or absent (internally wrapped into `{...}?`). | `[LEA r64, [r64+off8]]? 48 8B CF` matches with and without that `LEA`. |
| `{[MOV r64, [r64+off8]]\|[MOV r64, [r64+off32]]}` | **Alternation** — exactly one of the listed sub-sequences must match; branches may span multiple tokens and nest. | Matches either instruction form. |
| `{A\|B}?` | **Optional alternation** — trailing `?` after `}` makes the whole alternation optional: it is taken or skipped entirely. Without `\|` inside it is an optional plain group. | `{40 45\|40 46}? 50` matches `40 45 50`, `40 46 50` and `50` |

> [!IMPORTANT]
> The `[0-1?]` syntax means "zero or one byte", not "a byte in range 0x00-0x01". This is a length wildcard, not a value range.

## How to Use

### Generate a Pattern

1. Open a program in Ghidra.
2. Select one or more instructions in the Listing view.
3. Open `SPF Pattern Helper`.
4. Go to the `Pattern Generator` tab.
5. Click `Generate from Selection`.

> [!TIP]
> Select 3-5 instructions for a good balance between uniqueness and stability across game updates.

### Search a Pattern

1. Go to the `Pattern Finder` tab.
2. Paste or type the pattern.
3. Choose the search scope.
4. Click `Find Pattern`.

> [!TIP]
> Use **"Search in current block"** for faster results when you know which section contains the code. Use **"All executable sections"** for exhaustive searches.

### Optional Auto-Verify

1. Enable `Automatically verify signature`.
2. Generate a pattern from the selection.
3. The script will automatically search the program.
4. The result line under the button will show the outcome.
5. If needed, use `More info` to jump to `Pattern Finder`.
