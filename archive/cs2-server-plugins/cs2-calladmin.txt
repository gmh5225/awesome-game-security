```cs
using System;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using Sharp.Shared;

namespace CallAdmin.Core;

/// <summary>
/// CallAdmin — in-game player report system.
///
/// Architecture mirrors yappershq/poop's multi-project layout:
///   CallAdmin.Shared              — public API (events + ICallAdminShared)
///   CallAdmin.Database.Shared     — IDatabaseProvider surface (shared with sibling DBs)
///   CallAdmin.Database            — LiteDB / MySQL / Postgres via SqlSugar
///   CallAdmin.Core (this plugin)  — commands, lifecycle, business logic
///   CallAdmin.Discord             — webhook notifier (consumes ICallAdminShared events)
///
/// Lifecycle:
///   Init       — modules' Init (register chat callbacks, etc.)
///   PostInit   — SharedInterfaceModule registers ICallAdminShared on SharpModuleManager
///   OAM        — ReportManager attaches IDatabaseProvider, resolves AdminManager / Targeting /
///                MenuManager / Localizer (all optional but recommended)
/// </summary>
public sealed class CallAdminPlugin : IModSharpModule
{
    public string DisplayName   => "CallAdmin";
    public string DisplayAuthor => "yappershq";

    private readonly ServiceProvider          _serviceProvider;
    private readonly ILogger<CallAdminPlugin> _logger;
    private readonly InterfaceBridge          _bridge;

    public CallAdminPlugin(
        ISharedSystem  sharedSystem,
        string         dllPath,
        string         sharpPath,
        Version        version,
        IConfiguration configuration,
        bool           hotReload)
    {
        var loggerFactory = sharedSystem.GetLoggerFactory();
        _logger = loggerFactory.CreateLogger<CallAdminPlugin>();

        var bridge = new InterfaceBridge(dllPath,
                                         sharpPath,
                                         version,
                                         sharedSystem,
                                         this,
                                         hotReload,
                                         sharedSystem.GetModSharp().HasCommandLine("-debug"));

        var services = new ServiceCollection();
        services.AddSingleton(bridge);
        services.AddSingleton(loggerFactory);
        services.AddSingleton(sharedSystem);
        services.AddLogging();
        services.AddModuleDi();

        _bridge          = bridge;
        _serviceProvider = services.BuildServiceProvider();
    }

    public bool Init()
    {
        foreach (var service in _serviceProvider.GetServices<IModule>())
        {
            try
            {
                if (service.Init())
                {
                    if (_bridge.Debug)
                        _logger.LogInformation("{Service} initialized", service.GetType().FullName);
                    continue;
                }
                _logger.LogError("Failed to init {Service}!", service.GetType().FullName);
            }
            catch (Exception e) { _logger.LogError(e, "Init error in {Service}", service.GetType().FullName); }
            return false;
        }
        return true;
    }

    public void PostInit()
    {
        foreach (var service in _serviceProvider.GetServices<IModule>())
        {
            try { service.OnPostInit(_serviceProvider); }
            catch (Exception e) { _logger.LogError(e, "PostInit error in {Service}", service.GetType().FullName); }
        }
    }

    public void OnAllModulesLoaded()
    {
        _bridge.ResolveOptionalModules();
        _bridge.RegisterAdminPermissions();

        foreach (var service in _serviceProvider.GetServices<IModule>())
        {
            try { service.OnAllModulesLoaded(_serviceProvider); }
            catch (Exception e) { _logger.LogError(e, "OAM error in {Service}", service.GetType().FullName); }
        }

        _logger.LogInformation("[CallAdmin] Loaded (Localizer={Loc}, Menu={Menu}, Admin={Adm}, Targeting={Tgt})",
            _bridge.LocalizerManager is not null,
            _bridge.MenuManager      is not null,
            _bridge.AdminManager     is not null,
            _bridge.TargetingManager is not null);
    }

    public void Shutdown()
    {
        foreach (var service in _serviceProvider.GetServices<IModule>())
        {
            try { service.Shutdown(); }
            catch (Exception e) { _logger.LogError(e, "Shutdown error in {Service}", service.GetType().FullName); }
        }
        _serviceProvider.Dispose();
    }
}
```

```cs
using System;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;
using System.Text.Json.Serialization;
using Microsoft.Extensions.Logging;
using CallAdmin.Shared.Models;

namespace CallAdmin.Core.Configuration;

internal sealed class CallAdminConfig
{
    [JsonPropertyName("reasons")]
    public List<ReasonEntry> Reasons { get; set; } =
    [
        new() { Key = "hack",     LocaleKey = "calladmin.reason.hack" },
        new() { Key = "wallhack", LocaleKey = "calladmin.reason.wallhack" },
        new() { Key = "aimbot",   LocaleKey = "calladmin.reason.aimbot" },
        new() { Key = "toxic",    LocaleKey = "calladmin.reason.toxic" },
        new() { Key = "spam",     LocaleKey = "calladmin.reason.spam" },
        new() { Key = "griefing", LocaleKey = "calladmin.reason.griefing" },
        new() { Key = "other",    LocaleKey = "calladmin.reason.other" },
    ];

    /// <summary>Minutes a report may sit Pending (no admin claim) before it auto-closes as Cancelled. 0 disables auto-close.</summary>
    [JsonPropertyName("autoCloseMinutes")]     public int   AutoCloseMinutes     { get; set; } = 60;

    [JsonPropertyName("cooldownSeconds")]      public int   CooldownSeconds      { get; set; } = 300;
    [JsonPropertyName("cancelWindowSeconds")]  public int   CancelWindowSeconds  { get; set; } = 300;
    [JsonPropertyName("duplicateWindowSeconds")] public int DuplicateWindowSeconds { get; set; } = 600;
    [JsonPropertyName("notifyAdminsInChat")]   public bool  NotifyAdminsInChat   { get; set; } = true;
    [JsonPropertyName("allowReportAdmins")]    public bool  AllowReportAdmins    { get; set; } = false;
    [JsonPropertyName("allowReportSpectators")] public bool AllowReportSpectators { get; set; } = false;

    /// <summary>When true, only players granted <c>@calladmin/report</c> may submit reports. Off by default — everyone can report.</summary>
    [JsonPropertyName("requireReportPermission")] public bool RequireReportPermission { get; set; } = false;

    private static readonly JsonSerializerOptions JsonOpts = new()
    {
        PropertyNameCaseInsensitive = true,
        ReadCommentHandling         = JsonCommentHandling.Skip,
        AllowTrailingCommas         = true,
        WriteIndented               = true,
    };

    public IReadOnlyList<ReportReason> GetReasonModels()
    {
        var list = new List<ReportReason>(Reasons.Count);
        foreach (var r in Reasons)
            list.Add(new ReportReason { Key = r.Key, LocaleKey = r.LocaleKey });
        return list;
    }

    public static CallAdminConfig Load(string sharpPath, ILogger logger)
    {
        var path = Path.Combine(sharpPath, "configs", "calladmin.json");
        try
        {
            if (!File.Exists(path))
            {
                var def = new CallAdminConfig();
                Directory.CreateDirectory(Path.GetDirectoryName(path)!);
                File.WriteAllText(path, JsonSerializer.Serialize(def, JsonOpts));
                logger.LogInformation("[CallAdmin] Wrote default config to {Path}", path);
                return def;
            }

            var cfg = JsonSerializer.Deserialize<CallAdminConfig>(File.ReadAllText(path), JsonOpts);
            if (cfg is null)
            {
                logger.LogError("[CallAdmin] calladmin.json deserialized to null, using defaults");
                return new CallAdminConfig();
            }
            logger.LogInformation("[CallAdmin] Loaded config from {Path} ({Reasons} reasons)", path, cfg.Reasons.Count);
            return cfg;
        }
        catch (Exception e)
        {
            logger.LogError(e, "[CallAdmin] Failed to load calladmin.json, using defaults");
            return new CallAdminConfig();
        }
    }
}

internal sealed class ReasonEntry

    [JsonPropertyName("key")]       public string Key       { get; set; } = string.Empty;
    [JsonPropertyName("localeKey")] public string LocaleKey { get; set; } = string.Empty;
}
```

```cs
namespace CallAdmin.Core.Database;

[DbTable("calladmin_reports")]
internal sealed class ReportEntity
{
    [DbColumn(IsPrimaryKey = true, IsIdentity = true)]
    public int Id { get; set; }

    [DbColumn(Length = 64)]   public string  ReporterSteamId { get; set; } = string.Empty;
    [DbColumn(Length = 64)]   public string  ReporterName    { get; set; } = string.Empty;
    [DbColumn(Length = 64)]   public string  TargetSteamId   { get; set; } = string.Empty;
    [DbColumn(Length = 64)]   public string  TargetName      { get; set; } = string.Empty;
    [DbColumn(Length = 32)]   public string  ReasonKey       { get; set; } = string.Empty;
    [DbColumn(Length = 64)]   public string  MapName         { get; set; } = string.Empty;
    [DbColumn(Length = 64)]   public string  ServerName      { get; set; } = string.Empty;

    /// <summary>0=Pending 1=Claimed 2=Handled 3=Dismissed 4=Cancelled.</summary>
    [DbColumn] public int Status { get; set; }

    [DbColumn(IsNullable = true, Length = 64)] public string?  HandlerSteamId    { get; set; }
    [DbColumn(IsNullable = true, Length = 64)] public string?  HandlerName       { get; set; }
    [DbColumn(IsNullable = true)]              public DateTime? HandledAt        { get; set; }
    [DbColumn(IsNullable = true, Length = 128)] public string? ExternalMessageId { get; set; }

    /// <summary>Resolution outcome (ban/mute/gag/kick/warn/dismissed) — set by the website when a mod
    /// resolves a report. The module owns the column so CodeFirst creates it natively (the web can't
    /// rely on its own best-effort ALTER when it lacks DDL on this game-owned DB).</summary>
    [DbColumn(IsNullable = true, Length = 32)] public string? Outcome { get; set; }

    /// <summary>The Status the Discord webhook embed currently reflects. A periodic sweep compares it to
    /// Status to detect resolutions done EXTERNALLY (the website writes the DB directly, bypassing
    /// TransitionAsync) and edits the embed once — multi-server-safe via an atomic CAS on this column.</summary>
    [DbColumn] public int DiscordSyncedStatus { get; set; }

    [DbColumn] public DateTime CreatedAt { get; set; } = DateTime.UtcNow;
}
```

```cs
using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Microsoft.Extensions.Logging;
using CallAdmin.Database.Shared;
using CallAdmin.Shared.Models;

namespace CallAdmin.Core.Database;

/// <summary>
/// Wraps the shared <see cref="IDatabaseProvider"/> with report-specific queries. The DB plugin
/// publishes <see cref="IDatabaseProvider"/> in PostInit; we resolve it in OnAllModulesLoaded.
/// </summary>
internal sealed class ReportRepository
{
    private readonly InterfaceBridge          _bridge;
    private readonly ILogger<ReportRepository> _logger;
    private IDatabaseProvider?                _db;

    public ReportRepository(InterfaceBridge bridge, ILogger<ReportRepository> logger)
    {
        _bridge = bridge;
        _logger = logger;
    }

    public void Attach(IDatabaseProvider provider)
    {
        _db = provider;
        try
        {
            _db.InitTables(typeof(ReportEntity));
            _db.CreateIndex<ReportEntity>(["TargetSteamId", "Status"]);
            _db.CreateIndex<ReportEntity>(["ReporterSteamId", "CreatedAt"]);
            _logger.LogInformation("[CallAdmin] calladmin_reports table ready");
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] Failed to init calladmin_reports table");
            _db = null;
        }
    }

    public async Task<int> InsertAsync(ReportEntity entity)
    {
        if (_db is null) return 0;
        try
        {
            // Use the identity-returning insert so entity.Id reflects the DB-generated
            // PK. The plain InsertAsync (SqlSugar ExecuteCommandAsync) returns row count
            // and never backfills Id, leaving every report as #0 downstream.
            var id = await _db.InsertReturnIdentityAsync(entity);
            entity.Id = id;
            return id;
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] Insert failed");
            return 0;
        }
    }

    public async Task<ReportEntity?> GetByIdAsync(int id)
    {
        if (_db is null) return null;
        try
        {
            return await _db.Queryable<ReportEntity>().Where(x => x.Id == id).FirstAsync();
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] GetById({Id}) failed", id);
            return null;
        }
    }

    public async Task<List<ReportEntity>> GetPendingAsync()
    {
        if (_db is null) return [];
        try
        {
            return await _db.Queryable<ReportEntity>()
                .Where(x => x.Status == (int) ReportStatus.Pending || x.Status == (int) ReportStatus.Claimed)
                .OrderByDescending(x => x.CreatedAt)
                .ToListAsync();
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] GetPending failed");
            return [];
        }
    }

    /// <summary>Pending reports created before <paramref name="cutoff"/>. Claimed reports are excluded — a claim means an admin already responded.</summary>
    public async Task<List<ReportEntity>> GetExpiredPendingAsync(DateTime cutoff)
    {
        if (_db is null) return [];
        try
        {
            return await _db.Queryable<ReportEntity>()
                .Where(x => x.Status == (int) ReportStatus.Pending && x.CreatedAt < cutoff)
                .ToListAsync();
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] GetExpiredPending failed");
            return [];
        }
    }

    /// <summary>
    /// Atomic compare-and-swap: flips Pending → Cancelled only if the row is still Pending.
    /// Returns true if THIS caller made the change. With many servers sharing one DB, exactly one
    /// wins each row; the losers get false and must not re-publish the cancel event.
    /// </summary>
    public async Task<bool> TryTimeoutAsync(int id, DateTime handledAt)
    {
        if (_db is null) return false;
        try
        {
            var affected = await _db.UpdateSetWhereAsync<ReportEntity>(
                x => new ReportEntity { Status = (int) ReportStatus.Cancelled, HandledAt = handledAt },
                x => x.Id == id && x.Status == (int) ReportStatus.Pending);
            return affected > 0;
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] TryTimeout({Id}) failed", id);
            return false;
        }
    }

    /// <summary>Recently-resolved reports that HAVE a Discord message — the sweep filters these to the
    /// ones whose embed hasn't caught up (resolved externally on the website). Bounded to a recent
    /// window (<paramref name="cutoff"/>) so old reports are never touched.</summary>
    public async Task<List<ReportEntity>> GetRecentlyResolvedAsync(DateTime cutoff)
    {
        if (_db is null) return [];
        try
        {
            return await _db.Queryable<ReportEntity>()
                .Where(x => (x.Status == (int) ReportStatus.Handled || x.Status == (int) ReportStatus.Dismissed)
                            && x.ExternalMessageId != null
                            && x.HandledAt != null && x.HandledAt >= cutoff)
                .ToListAsync();
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] GetRecentlyResolved failed");
            return [];
        }
    }

    /// <summary>
    /// Atomic compare-and-swap: stamps DiscordSyncedStatus = <paramref name="status"/> only if it
    /// still differs. Returns true if THIS caller won the claim. With 10 servers sharing one DB,
    /// exactly one wins each externally-resolved report; the losers get false and must not re-edit.
    /// </summary>
    public async Task<bool> TryClaimDiscordSyncAsync(int id, int status)
    {
        if (_db is null) return false;
        try
        {
            var affected = await _db.UpdateSetWhereAsync<ReportEntity>(
                x => new ReportEntity { DiscordSyncedStatus = status },
                x => x.Id == id && x.DiscordSyncedStatus != status);
            return affected > 0;
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] TryClaimDiscordSync({Id}) failed", id);
            return false;
        }
    }

    public async Task<ReportEntity?> FindDuplicateAsync(string reporterSteamId, string targetSteamId, DateTime since)
    {
        if (_db is null) return null;
        try
        {
            return await _db.Queryable<ReportEntity>()
                .Where(x => x.ReporterSteamId == reporterSteamId
                            && x.TargetSteamId == targetSteamId
                            && x.CreatedAt >= since
                            && x.Status == (int) ReportStatus.Pending)
                .FirstAsync();
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] FindDuplicate failed");
            return null;
        }
    }

    public async Task<DateTime?> LastReportTimeForAsync(string reporterSteamId)
    {
        if (_db is null) return null;
        try
        {
            var row = await _db.Queryable<ReportEntity>()
                .Where(x => x.ReporterSteamId == reporterSteamId)
                .OrderByDescending(x => x.CreatedAt)
                .FirstAsync();
            return row?.CreatedAt;
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] LastReportTimeFor failed");
            return null;
        }
    }

    public async Task<bool> UpdateAsync(ReportEntity entity)
    {
        if (_db is null) return false;
        try
        {
            return await _db.UpdateAsync(entity) > 0;
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin] Update failed for {Id}", entity.Id);
            return false;
        }
    }
}
```

```cs
    internal IMenuManager?       MenuManager        { get; private set; }

    public InterfaceBridge(
        string         dllPath,
        string         sharpPath,
        Version        version,
        ISharedSystem  sharedSystem,
        IModSharpModule module,
        bool           hotReload,
        bool           debug)
    {
        DllPath   = dllPath;
        SharpPath = sharpPath;
        Version   = version;
        HotReload = hotReload;
        Debug     = debug;
        Module    = module;

        SharedSystem       = sharedSystem;
        SharpModuleManager = sharedSystem.GetSharpModuleManager();

        ConVarManager  = sharedSystem.GetConVarManager();
        EventManager   = sharedSystem.GetEventManager();
        ClientManager  = sharedSystem.GetClientManager();
        EntityManager  = sharedSystem.GetEntityManager();
        HookManager    = sharedSystem.GetHookManager();
        ModSharp       = sharedSystem.GetModSharp();
        LoggerFactory  = sharedSystem.GetLoggerFactory();
    }

    internal void ResolveOptionalModules()
    {
        LocalizerManager   ??= SharpModuleManager.GetOptionalSharpModuleInterface<ILocalizerManager>(ILocalizerManager.Identity)?.Instance;
        MenuManager        ??= SharpModuleManager.GetOptionalSharpModuleInterface<IMenuManager>(IMenuManager.Identity)?.Instance;
        AdminManager       ??= SharpModuleManager.GetOptionalSharpModuleInterface<IAdminManager>(IAdminManager.Identity)?.Instance;
        TargetingManager   ??= SharpModuleManager.GetOptionalSharpModuleInterface<ITargetingManager>(ITargetingManager.Identity)?.Instance;
    }

    // CallAdmin's permission strings only become grantable once they exist in the
    // AdminManager permission index. Without this, even a root ('*') admin resolves
    // to false for @calladmin/handle, because '*' expands over registered permissions
    // only — an unregistered permission is granted to nobody.
    internal void RegisterAdminPermissions()
    {
        if (AdminManager is not { } am)
            return;

        am.MountAdminManifest("CallAdmin", static () => new AdminTableManifest(
            new Dictionary<string, HashSet<string>>(StringComparer.OrdinalIgnoreCase)
            {
                ["calladmin"] = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
                {
                    "@calladmin/report",
                    "@calladmin/handle",
                    "@calladmin/report_admin",
                },
            },
            [],
            []));
    }
}
```

```cs
using MessagePipe;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using CallAdmin.Core.Configuration;
using CallAdmin.Core.Database;
using CallAdmin.Core.Modules;
using CallAdmin.Shared;
using CallAdmin.Shared.Events;

namespace CallAdmin.Core;

internal static class ModuleDependencyInjection
{
    internal static IServiceCollection AddModuleDi(this IServiceCollection services)
    {
        // Restrict handler/filter auto-registration to CallAdmin's own assembly. MessagePipe's
        // default scans the entire AppDomain (TypeCollector.CollectFromCurrentDomain), which makes
        // it read custom attributes on OTHER plugins' SqlSugar-decorated entity types. Those resolve
        // through our load context, which ships no SqlSugar.dll — once a mysql backend pulls SqlSugar
        // into the process this throws FileLoadException and CallAdmin.Core fails to load entirely.
        services.AddMessagePipe(options =>
            options.SetAutoRegistrationSearchAssemblies(typeof(ModuleDependencyInjection).Assembly));

        services.AddSingleton(sp =>
            CallAdminConfig.Load(sp.GetRequiredService<InterfaceBridge>().SharpPath,
                                 sp.GetRequiredService<Microsoft.Extensions.Logging.ILoggerFactory>().CreateLogger<CallAdminConfig>()));

        services.AddSingleton<ReportRepository>();
        services.AddSingleton<ReportManager>();
        services.AddSingleton<SharedInterfaceModule>();
```

```cs
                        return;
                    }
                    admin.Print(HudPrintChannel.Chat, $" [CallAdmin] {pending.Count} pending report(s):");

                        var status = r.Status == ReportStatus.Claimed ? $" claimed by {r.HandlerName}" : "";
                        admin.Print(HudPrintChannel.Chat,
                            $" #{r.Id} [{age}s ago] {r.ReporterName} -> {r.TargetName} ({r.ReasonKey}){status}");

    {

        var admin = ResolveAdmin(adminSlot);
        if (admin is null) return;

            {

                    var latest  = pending.OrderByDescending(r => r.CreatedAt).FirstOrDefault();

                        {
                            if (ResolveAdmin(adminSlot) is { } a)
                                a.Print(HudPrintChannel.Chat, " [CallAdmin] No pending report to act on.");
                        });
                        return;
                    }
                    targetId = latest.Id;
                }

                var r = await _reports.TransitionAsync(targetId, to, actorSid, actorName);
                _bridge.ModSharp.InvokeFrameAction(() =>
                {
                    if (ResolveAdmin(adminSlot) is not { } a) return;
                    if (r is null)
                        a.Print(HudPrintChannel.Chat, $" [CallAdmin] Report #{targetId} not found.");
                    else
                        a.Print(HudPrintChannel.Chat, $" [CallAdmin] Report #{r.Id} {verb}.");
                });
            }
            catch (Exception e) { _logger.LogError(e, "[CallAdmin] Panel transition {Status} failed", to); }
        });
    }

    private IGameClient? ResolveAdmin(int adminSlot)
    {
        var client = _bridge.ClientManager.GetGameClient((PlayerSlot) (byte) adminSlot);
        return client is { IsValid: true } ? client : null;
```

```cs
    private ECommandAction OnReport(IGameClient client, StringCommand command)
    {
        if (client.IsFakeClient) return ECommandAction.Skipped;
        if (!CheckReportPerm(client)) return ECommandAction.Handled;

        if (command.ArgCount < 2)
        {
            // Usage hint — proper menu wiring lives in TargetMenuModule (TODO v0.2).
            client.Print(HudPrintChannel.Chat,
                " [CallAdmin] Usage: !report <name|steamid|@me|@t|@ct> <reason>");
            client.Print(HudPrintChannel.Chat,
                " [CallAdmin] Reasons: " + string.Join(", ", _config.Reasons.Select(r => r.Key)));
            return ECommandAction.Handled;
        }

        var targetExpr = command.GetArg(1);
        var reasonKey  = command.GetArg(2);

        var tm = _bridge.TargetingManager;
        if (tm is null)
        {
            client.Print(HudPrintChannel.Chat, " [CallAdmin] TargetingManager unavailable.");
            return ECommandAction.Handled;
        }

        var targets = tm.GetByTarget(client, targetExpr).ToList();
        if (targets.Count == 0)
        {
            client.Print(HudPrintChannel.Chat, $" [CallAdmin] No player matches '{targetExpr}'.");
            return ECommandAction.Handled;
        }
        if (targets.Count > 1)
        {
            client.Print(HudPrintChannel.Chat, $" [CallAdmin] '{targetExpr}' matches multiple players — be more specific.");
            return ECommandAction.Handled;
        }

        var target = targets[0];

        // Surface specific rejection reasons before the generic "filtered" fallback in SubmitAsync
        if (target.SteamId == client.SteamId)
        {
            client.Print(HudPrintChannel.Chat, " [CallAdmin] You can't report yourself.");
            return ECommandAction.Handled;
        }
        if (target.IsFakeClient || target.IsHltv)
        {
            client.Print(HudPrintChannel.Chat, " [CallAdmin] You can't report bots or HLTV.");
            return ECommandAction.Handled;
        }
        if (!_config.Reasons.Any(r => string.Equals(r.Key, reasonKey, StringComparison.OrdinalIgnoreCase)))
        {
            client.Print(HudPrintChannel.Chat,
                $" [CallAdmin] Unknown reason '{reasonKey}'. Pick: " + string.Join(", ", _config.Reasons.Select(r => r.Key)));
            return ECommandAction.Handled;
        }

        HandleSubmitAsync(client, target, reasonKey);
        return ECommandAction.Handled;
    }

    private void HandleSubmitAsync(IGameClient reporter, IGameClient target, string reasonKey)
    {
        var snap = _reports.PrepareSubmit(reporter, target, reasonKey);
        if (snap is null)
        {
            reporter.Print(HudPrintChannel.Chat, " [CallAdmin] Report rejected (cooldown, invalid reason, or filtered).");
            return;
        }
        _ = Task.Run(async () =>
        {
            try
            {
                var result = await _reports.SubmitAsync(snap);
                _bridge.ModSharp.InvokeFrameAction(() =>
                {
                    if (!reporter.IsValid) return;
                    if (result is null)
                    {
                        reporter.Print(HudPrintChannel.Chat, " [CallAdmin] Report rejected (cooldown or duplicate).");
                        return;
                    }
                    reporter.Print(HudPrintChannel.Chat,
                        $" [CallAdmin] ✓ Report #{result.Id} submitted against {result.TargetName}. Admins notified.");
                    NotifyAdmins(result);
                });
            }
            catch (Exception e)
            {
                _logger.LogError(e, "[CallAdmin] HandleSubmit failed");
            }
        });
    }

    private void NotifyAdmins(Report report)
    {
        if (!_config.NotifyAdminsInChat) return;
        var am = _bridge.AdminManager;
        if (am is null) return;

        foreach (var c in _bridge.ClientManager.GetGameClients(inGame: true))
        {
            if (c.IsFakeClient || c.IsHltv) continue;
            if (am.GetAdmin(c.SteamId) is not { } admin) continue;
            if (!admin.HasPermission(PermHandle)) continue;

            c.Print(HudPrintChannel.Chat,
                $" [CallAdmin] 📢 #{report.Id} {report.ReporterName} -> {report.TargetName} ({report.ReasonKey}). !claim {report.Id}");
        }
    }

    private ECommandAction OnReportCancel(IGameClient client, StringCommand command)
    {
        if (client.IsFakeClient) return ECommandAction.Skipped;
        var reporterSid = client.SteamId;
        _ = Task.Run(async () =>
        {
            try
            {
                var pending = await _reports.GetPendingAsync();
                var own = pending.FirstOrDefault(r => r.ReporterSteamId == reporterSid
                                                      && r.Status == ReportStatus.Pending);
                if (own is null)
                {
                    _bridge.ModSharp.InvokeFrameAction(() =>
                    {
                        if (!client.IsValid) return;
                        client.Print(HudPrintChannel.Chat, " [CallAdmin] No pending report to cancel.");
                    });
                    return;
                }
                if ((DateTime.UtcNow - own.CreatedAt).TotalSeconds > _config.CancelWindowSeconds)
                {
                    _bridge.ModSharp.InvokeFrameAction(() =>
                    {
                        if (!client.IsValid) return;
                        client.Print(HudPrintChannel.Chat, " [CallAdmin] Cancel window has expired.");
                    });
                    return;
                }
                var updated = await _reports.TransitionAsync(own.Id, ReportStatus.Cancelled, null, null);
                _bridge.ModSharp.InvokeFrameAction(() =>
                {
                    if (!client.IsValid) return;
                    if (updated is not null)
                        client.Print(HudPrintChannel.Chat, $" [CallAdmin] Report #{updated.Id} cancelled.");
                });
            }
            catch (Exception e)
            {
                _logger.LogError(e, "[CallAdmin] Cancel failed");
            }
        });

    private ECommandAction HandleTransition(IGameClient client, StringCommand command, ReportStatus to, string verb)
    {
        if (!RequireHandlePerm(client)) return ECommandAction.Handled;
        if (command.ArgCount < 1)
        {
            client.Print(HudPrintChannel.Chat, " [CallAdmin] Usage: <command> <id>");
            return ECommandAction.Handled;
        }
        if (!int.TryParse(command.GetArg(1), out var id))
        {
            client.Print(HudPrintChannel.Chat, $" [CallAdmin] '{command.GetArg(1)}' is not a number.");
            return ECommandAction.Handled;
        }
        var adminSid  = client.SteamId.ToString();
        var adminName = client.Name ?? "?";
        _ = Task.Run(async () =>
        {
            try
            {
                var r = await _reports.TransitionAsync(id, to, adminSid, adminName);
                _bridge.ModSharp.InvokeFrameAction(() =>
                {
                    if (!client.IsValid) return;
                    if (r is null)
                    {
                        client.Print(HudPrintChannel.Chat, $" [CallAdmin] Report #{id} not found.");
                        return;
                    }
                    client.Print(HudPrintChannel.Chat, $" [CallAdmin] Report #{r.Id} {verb}.");
                });
            }
            catch (Exception e)
            {
                _logger.LogError(e, "[CallAdmin] Transition {Status} failed for #{Id}", to, id);
            }
        });
        return ECommandAction.Handled;
    }

    private bool RequireHandlePerm(IGameClient client)
    {
        var am = _bridge.AdminManager;
        if (am?.GetAdmin(client.SteamId) is not { } admin || !admin.HasPermission(PermHandle))

            return false;
        }
        return true;

    {
        // Gate is off by default — anyone can submit. Servers that want a gate set
        // requireReportPermission=true and grant @calladmin/report to allowed players.
        if (!_config.RequireReportPermission) return true;

        var am = _bridge.AdminManager;
        if (am?.GetAdmin(client.SteamId) is not { } admin || !admin.HasPermission(PermReport))
        {
            client.Print(HudPrintChannel.Chat, " [CallAdmin] You don't have permission to use !report on this server.");
            return false;
        }
        return true;
    }
```

```cs
/// <summary>
/// Core business logic for reports — gating (cooldown, dedupe, perms), persistence, lifecycle
/// transitions, and event publication. Stateless apart from the in-memory caches (cooldowns,
/// pending cache). All write paths go through the repository so persistence is the source of
/// truth across map changes.
/// </summary>
internal sealed class ReportManager : IModule
{
    private readonly InterfaceBridge                    _bridge;
    private readonly ILogger<ReportManager>             _logger;
    private readonly CallAdminConfig                    _config;
    private readonly ReportRepository                   _repo;
    private readonly IPublisher<ReportSubmittedEvent>   _submitted;
    private readonly IPublisher<ReportClaimedEvent>     _claimed;
    private readonly IPublisher<ReportHandledEvent>     _handled;
    private readonly IPublisher<ReportDismissedEvent>   _dismissed;
    private readonly IPublisher<ReportCancelledEvent>   _cancelled;

    private readonly List<ITargetFilter> _filters = new();

    // Auto-close (timeout) sweep. _sweeping guards against overlapping runs on the threadpool.
    private Timer? _sweepTimer;
    private int    _sweeping;

    // External-resolution → Discord sweep (website resolves the DB directly, bypassing TransitionAsync).
    private Timer? _syncTimer;
    private int    _syncing;
    private const int ExternalSyncWindowMinutes = 30;

    public ReportManager(
        InterfaceBridge                  bridge,
        ILogger<ReportManager>           logger,
        CallAdminConfig                  config,
        ReportRepository                 repo,
        IPublisher<ReportSubmittedEvent> submitted,
        IPublisher<ReportClaimedEvent>   claimed,
        IPublisher<ReportHandledEvent>   handled,
        IPublisher<ReportDismissedEvent> dismissed,
        IPublisher<ReportCancelledEvent> cancelled)
    {
        _bridge    = bridge;
        _logger    = logger;
        _config    = config;
        _repo      = repo;
        _submitted = submitted;
        _claimed   = claimed;
        _handled   = handled;
        _dismissed = dismissed;
        _cancelled = cancelled;
    }

    public bool Init() => true;

    public void OnAllModulesLoaded(ServiceProvider provider)
    {
        // Resolve DB provider once both modules are live.
        var dbIface = _bridge.SharpModuleManager
            .GetOptionalSharpModuleInterface<IDatabaseProvider>(IDatabaseProvider.Identity);
        if (dbIface?.Instance is { } db)
            _repo.Attach(db);
        else
            _logger.LogError("[CallAdmin] IDatabaseProvider not available — CallAdmin.Database not loaded?");

        // Auto-close pending reports nobody picked up. Every server runs this, but the cancel is an
        // atomic compare-and-swap (TryTimeoutAsync), so when many servers share one DB exactly one
        // wins each row and only that one publishes the cancel event — no duplicate Discord edits.
        if (_config.AutoCloseMinutes > 0)
        {
            _sweepTimer = new Timer(_ => _ = SweepTimeoutsAsync(), null,
                TimeSpan.FromSeconds(30), TimeSpan.FromSeconds(60));
            _logger.LogInformation("[CallAdmin] Auto-close enabled: pending reports cancelled after {Min} min", _config.AutoCloseMinutes);
        }

        // Sync externally-resolved reports: the website resolves by writing the DB directly (bypassing
        // TransitionAsync), so its resolutions never edit the Discord embed. This sweep catches them.
        // Runs on every server; the per-row CAS claim means exactly one server edits each report.
        _syncTimer = new Timer(_ => _ = SyncExternalResolutionsAsync(), null,
            TimeSpan.FromSeconds(45), TimeSpan.FromSeconds(60));
    }

    private async Task SyncExternalResolutionsAsync()
    {
        if (Interlocked.Exchange(ref _syncing, 1) == 1) return;
        try
        {
            var cutoff = DateTime.UtcNow.AddMinutes(-ExternalSyncWindowMinutes);
            foreach (var row in await _repo.GetRecentlyResolvedAsync(cutoff))
            {
                if (row.DiscordSyncedStatus == row.Status) continue;              // embed already reflects it
                if (!await _repo.TryClaimDiscordSyncAsync(row.Id, row.Status)) continue; // another server won it

                var model  = ToModel(row);
                var status = (ReportStatus) row.Status;
                _bridge.ModSharp.InvokeFrameAction(() =>
                {
                    try
                    {
                        if (status == ReportStatus.Handled)        _handled.Publish(new ReportHandledEvent(model));
                        else if (status == ReportStatus.Dismissed) _dismissed.Publish(new ReportDismissedEvent(model));
                    }
                    catch (Exception e) { _logger.LogError(e, "[CallAdmin] External-sync publish failed for #{Id}", model.Id); }
                });
                _logger.LogInformation("[CallAdmin] Report #{Id} resolved externally ({Status}) — Discord embed synced", row.Id, status);
            }
        }
        catch (Exception e) { _logger.LogError(e, "[CallAdmin] External-resolution sync failed"); }
        finally { Interlocked.Exchange(ref _syncing, 0); }
    }

    private async Task SweepTimeoutsAsync()
    {
        // Skip if a previous sweep is still running (slow DB / many rows).
        if (Interlocked.Exchange(ref _sweeping, 1) == 1) return;
        try
        {
            var cutoff  = DateTime.UtcNow.AddMinutes(-_config.AutoCloseMinutes);
            var expired = await _repo.GetExpiredPendingAsync(cutoff);
            foreach (var row in expired)
            {
                var now = DateTime.UtcNow;
                if (!await _repo.TryTimeoutAsync(row.Id, now))
                    continue; // another server claimed this one

                row.Status    = (int) ReportStatus.Cancelled;
                row.HandledAt = now;
                var model = ToModel(row);

                _bridge.ModSharp.InvokeFrameAction(() =>
                {
                    try { _cancelled.Publish(new ReportCancelledEvent(model)); }
                    catch (Exception e) { _logger.LogError(e, "[CallAdmin] Timeout cancel publish failed for #{Id}", model.Id); }
                });

                _logger.LogInformation("[CallAdmin] Report #{Id} auto-closed (no response in {Min} min)", row.Id, _config.AutoCloseMinutes);
            }
        }
        catch (Exception e) { _logger.LogError(e, "[CallAdmin] Auto-close sweep failed"); }
        finally { Interlocked.Exchange(ref _sweeping, 0); }
    }

    public void Shutdown()
    {
        _sweepTimer?.Dispose();
        _sweepTimer = null;
        _syncTimer?.Dispose();
        _syncTimer = null;
    }

    public void RegisterTargetFilter(ITargetFilter filter)
    {
        _filters.Add(filter);
    }

    public bool CanTarget(IGameClient reporter, IGameClient candidate)
    {
        if (candidate.IsFakeClient || candidate.IsHltv)        return false;
        if (candidate.SteamId == reporter.SteamId)             return false;
        if (!_config.AllowReportAdmins && IsAdmin(candidate))  return false;

        // Spectator gate (CT=3, T=2, Spec=1, None=0)
        if (!_config.AllowReportSpectators)
        {
            var ctrl = candidate.GetPlayerController();
            if (ctrl is not null && (int) ctrl.Team < 2)
                return false;
        }

        foreach (var f in _filters)
            if (!f.CanReport(reporter, candidate))
                return false;
        return true;
    }

    /// <summary>
    /// Must be called on the main thread. Validates target eligibility and snapshots all
    /// engine-side data. Returns null if the report should be rejected before any DB work.
    /// </summary>
    public ReportSnapshot? PrepareSubmit(IGameClient reporter, IGameClient target, string reasonKey)
    {
        if (!CanTarget(reporter, target))
        {
            _logger.LogDebug("[CallAdmin] Submit blocked by filters: {Reporter} -> {Target}", reporter.SteamId, target.SteamId);
            return null;
        }

        var reason = _config.Reasons.FirstOrDefault(r => string.Equals(r.Key, reasonKey, StringComparison.OrdinalIgnoreCase));
        if (reason is null)
        {
            _logger.LogDebug("[CallAdmin] Unknown reason key {Reason}", reasonKey);
            return null;
        }

        return new ReportSnapshot(
            ReporterSteamId: reporter.SteamId.ToString(),
            ReporterName:    reporter.Name ?? "?",
            TargetSteamId:   target.SteamId.ToString(),
            TargetName:      target.Name ?? "?",
            ReasonKey:       reason.Key,
            MapName:         _bridge.ModSharp.GetMapName() ?? "unknown",
            ServerName:      _bridge.ConVarManager.FindConVar("hostname")?.GetString() ?? string.Empty
        );
    }

    /// <summary>
    /// Safe to call off the main thread. All engine-side data must already be snapshotted via
    /// <see cref="PrepareSubmit"/>.
    /// </summary>
    public async Task<Report?> SubmitAsync(ReportSnapshot snap)
    {
        // Cooldown
        var last = await _repo.LastReportTimeForAsync(snap.ReporterSteamId);
        if (last is { } lt && (DateTime.UtcNow - lt).TotalSeconds < _config.CooldownSeconds)
        {
            _logger.LogDebug("[CallAdmin] {Reporter} on cooldown ({Remaining}s)", snap.ReporterSteamId,
                _config.CooldownSeconds - (int)(DateTime.UtcNow - lt).TotalSeconds);
            return null;
        }

        // Duplicate within window — same reporter+target => skip (no DB write)
        var since = DateTime.UtcNow.AddSeconds(-_config.DuplicateWindowSeconds);
        var dup = await _repo.FindDuplicateAsync(snap.ReporterSteamId, snap.TargetSteamId, since);
        if (dup is not null)
        {
            _logger.LogDebug("[CallAdmin] Duplicate report skipped ({Reporter} -> {Target}, id #{Id})",
                snap.ReporterSteamId, snap.TargetSteamId, dup.Id);
            return ToModel(dup);
        }

        var entity = new ReportEntity
        {
            ReporterSteamId = snap.ReporterSteamId,
            ReporterName    = snap.ReporterName,
            TargetSteamId   = snap.TargetSteamId,
            TargetName      = snap.TargetName,
            ReasonKey       = snap.ReasonKey,
            MapName         = snap.MapName,
            ServerName      = snap.ServerName,
            Status          = (int) ReportStatus.Pending,
            CreatedAt       = DateTime.UtcNow,
        };

        await _repo.InsertAsync(entity);
        var model = ToModel(entity);
        _logger.LogInformation("[CallAdmin] Report #{Id} submitted by {Reporter} against {Target} ({Reason})",
            entity.Id, entity.ReporterName, entity.TargetName, entity.ReasonKey);

        _bridge.ModSharp.InvokeFrameAction(() =>
        {
            try { _submitted.Publish(new ReportSubmittedEvent(model)); }
            catch (Exception e) { _logger.LogError(e, "[CallAdmin] ReportSubmittedEvent subscriber failed"); }
        });

        return model;
    }

    public async Task<Report?> TransitionAsync(int reportId, ReportStatus next, string? actorSid, string? actorName)
    {
        var row = await _repo.GetByIdAsync(reportId);
        if (row is null) return null;
        if (row.Status == (int) next) return ToModel(row);
        if (row.Status == (int) ReportStatus.Handled || row.Status == (int) ReportStatus.Dismissed
            || row.Status == (int) ReportStatus.Cancelled)
            return ToModel(row); // already terminal

        row.Status = (int) next;
        if (actorSid is not null && next != ReportStatus.Cancelled)
        {
            row.HandlerSteamId = actorSid;
            row.HandlerName    = actorName;
            row.HandledAt      = DateTime.UtcNow;
        }
        if (next == ReportStatus.Cancelled)
            row.HandledAt = DateTime.UtcNow;

        // Our own resolution publishes its event below → Discord edits now. Pre-stamp the sync marker
        // so the external-resolution sweep doesn't treat this as an un-synced (website) resolution.
        if (next is ReportStatus.Handled or ReportStatus.Dismissed)
            row.DiscordSyncedStatus = (int) next;

        await _repo.UpdateAsync(row);
        var model = ToModel(row);

        _bridge.ModSharp.InvokeFrameAction(() =>
        {
            try
            {
                switch (next)
                {
                    case ReportStatus.Claimed:    _claimed.Publish(new ReportClaimedEvent(model));     break;
                    case ReportStatus.Handled:    _handled.Publish(new ReportHandledEvent(model));     break;
                    case ReportStatus.Dismissed:  _dismissed.Publish(new ReportDismissedEvent(model)); break;
                    case ReportStatus.Cancelled:  _cancelled.Publish(new ReportCancelledEvent(model)); break;
                }
            }
            catch (Exception e) { _logger.LogError(e, "[CallAdmin] {Next} publisher failed for #{Id}", next, reportId); }
        });

        return model;
    }

    public async Task<Report?> GetAsync(int id)
    {
        var row = await _repo.GetByIdAsync(id);
        return row is null ? null : ToModel(row);
    }

    public async Task<IReadOnlyList<Report>> GetPendingAsync()
    {
        var rows = await _repo.GetPendingAsync();
        var list = new List<Report>(rows.Count);
        foreach (var row in rows)
            list.Add(ToModel(row));
        return list;
    }

    public async Task SetExternalMessageIdAsync(int id, string messageId)
    {
        var row = await _repo.GetByIdAsync(id);
        if (row is null) return;
        row.ExternalMessageId = messageId;
        await _repo.UpdateAsync(row);
    }

    private bool IsAdmin(IGameClient client)
    {
        var am = _bridge.AdminManager;
        return am?.GetAdmin(client.SteamId) is not null;
    }

    private static Report ToModel(ReportEntity e) => new Report
    {
        Id                = e.Id,
        ReporterSteamId   = ulong.TryParse(e.ReporterSteamId, out var rs) ? rs : 0,
        ReporterName      = e.ReporterName,
        TargetSteamId     = ulong.TryParse(e.TargetSteamId, out var ts) ? ts : 0,
        TargetName        = e.TargetName,
        ReasonKey         = e.ReasonKey,
        ReasonLabel       = e.ReasonKey, // localizer applied at render time, not here
        MapName           = e.MapName,
        ServerName        = e.ServerName,
        CreatedAt         = e.CreatedAt,
        Status            = (ReportStatus) e.Status,
        HandlerSteamId    = ulong.TryParse(e.HandlerSteamId ?? "", out var hs) && hs != 0 ? hs : null,
        HandlerName       = e.HandlerName,
        HandledAt         = e.HandledAt,
        ExternalMessageId = e.ExternalMessageId,
        Outcome           = e.Outcome,
    };
}
```

```cs
    private readonly ReportManager                  _reports;

    private readonly ISubscriber<ReportSubmittedEvent>  _submittedSub;
    private readonly ISubscriber<ReportClaimedEvent>    _claimedSub;
    private readonly ISubscriber<ReportHandledEvent>    _handledSub;
    private readonly ISubscriber<ReportDismissedEvent>  _dismissedSub;
    private readonly ISubscriber<ReportCancelledEvent>  _cancelledSub;

    private IDisposable? _eventSubscriptions;

    public event Action<Report>? OnReportSubmitted;
    public event Action<Report>? OnReportClaimed;
    public event Action<Report>? OnReportHandled;
    public event Action<Report>? OnReportDismissed;
    public event Action<Report>? OnReportCancelled;

    public SharedInterfaceModule(InterfaceBridge bridge, ILogger<SharedInterfaceModule> logger,
                                 ReportManager reports, Configuration.CallAdminConfig config,
                                 ISubscriber<ReportSubmittedEvent> submittedSub,
                                 ISubscriber<ReportClaimedEvent>   claimedSub,
                                 ISubscriber<ReportHandledEvent>   handledSub,
                                 ISubscriber<ReportDismissedEvent> dismissedSub,
                                 ISubscriber<ReportCancelledEvent> cancelledSub)
    {
        _bridge        = bridge;
        _logger        = logger;
        _reports       = reports;
        _config        = config;
        _submittedSub  = submittedSub;
        _claimedSub    = claimedSub;
        _handledSub    = handledSub;
        _dismissedSub  = dismissedSub;
        _cancelledSub  = cancelledSub;
    }

    public bool Init()
    {
        var bag = DisposableBag.CreateBuilder();
        _submittedSub.Subscribe(e => Raise(OnReportSubmitted, e.Report, nameof(OnReportSubmitted))).AddTo(bag);
        _claimedSub  .Subscribe(e => Raise(OnReportClaimed,   e.Report, nameof(OnReportClaimed))).AddTo(bag);
        _handledSub  .Subscribe(e => Raise(OnReportHandled,   e.Report, nameof(OnReportHandled))).AddTo(bag);
        _dismissedSub.Subscribe(e => Raise(OnReportDismissed, e.Report, nameof(OnReportDismissed))).AddTo(bag);
        _cancelledSub.Subscribe(e => Raise(OnReportCancelled, e.Report, nameof(OnReportCancelled))).AddTo(bag);
        _eventSubscriptions = bag.Build();
        return true;
    }

    public void Shutdown()
    {
        _eventSubscriptions?.Dispose();
        _eventSubscriptions = null;
    }

    private void Raise(Action<Report>? handlers, Report report, string name)
    {
        if (handlers is null) return;
        try { handlers.Invoke(report); }
        catch (Exception e) { _logger.LogError(e, "[CallAdmin] {Event} subscriber threw", name); }
    }

    public void OnPostInit(ServiceProvider provider)
    {
        _bridge.SharpModuleManager.RegisterSharpModuleInterface<ICallAdminShared>(
            _bridge.Module, ICallAdminShared.Identity, this);
        _logger.LogInformation("[CallAdmin] Registered ICallAdminShared ({Id})", ICallAdminShared.Identity);
    }

    // ===== ICallAdminShared =====

    public Task<Report?> SubmitReportAsync(IGameClient reporter, IGameClient target, string reasonKey)
    {
        var snap = _reports.PrepareSubmit(reporter, target, reasonKey);
        if (snap is null) return Task.FromResult<Report?>(null);
        return Task.Run(() => _reports.SubmitAsync(snap));
    }

    public Task<Report?> ClaimReportAsync(int reportId, IGameClient admin)
        => Task.Run(() => _reports.TransitionAsync(reportId, ReportStatus.Claimed, admin.SteamId.ToString(), admin.Name ?? "?"));

    public Task<Report?> MarkHandledAsync(int reportId, IGameClient admin)
        => Task.Run(() => _reports.TransitionAsync(reportId, ReportStatus.Handled, admin.SteamId.ToString(), admin.Name ?? "?"));

    public Task<Report?> DismissReportAsync(int reportId, IGameClient admin)
        => Task.Run(() => _reports.TransitionAsync(reportId, ReportStatus.Dismissed, admin.SteamId.ToString(), admin.Name ?? "?"));

    public Task<Report?> CancelReportAsync(int reportId, IGameClient reporter)
        => Task.Run(() => _reports.TransitionAsync(reportId, ReportStatus.Cancelled, null, null));

    public Task<Report?> GetReportAsync(int reportId) => _reports.GetAsync(reportId);

    public Task<IReadOnlyList<Report>> GetPendingReportsAsync() => _reports.GetPendingAsync();

    public IReadOnlyList<ReportReason> GetReasons() => _config.GetReasonModels();

    public void RegisterTargetFilter(ITargetFilter filter) => _reports.RegisterTargetFilter(filter);

    public Task SetExternalMessageIdAsync(int reportId, string messageId)
        => _reports.SetExternalMessageIdAsync(reportId, messageId);
}
```

```cs
                Connection = ConnectionType.Shared,

            services.AddSingleton<IDatabaseProvider, LiteDbDatabaseProvider>();
        }
        else
        {
            _logger.LogInformation("[CallAdmin.Database] Using SqlSugar ({DbType})", dbType);

            services.AddSingleton<ISqlSugarClient>(_ =>
                new SqlSugarScope(SugarExtensions.BuildConnectionConfig(configuration, _bridge.SharpPath)));

            File.WriteAllText(configPath, DefaultConfig);

            .AddJsonFile(configPath, optional: false, reloadOnChange: false)
            .Build();
    }
```

```cs
using System;
using System.Collections.Generic;
using System.Linq;
using System.Linq.Expressions;
using System.Reflection;
using System.Threading.Tasks;
using LiteDB;
using CallAdmin.Database.Shared;

namespace CallAdmin.Database.Provider;

internal sealed class LiteDbDatabaseProvider(LiteDatabase db) : IDatabaseProvider
{
    private static string GetCollectionName(Type type)
    {
        var attr = type.GetCustomAttribute<DbTableAttribute>();
        return attr?.TableName ?? type.Name;
    }

    private ILiteCollection<T> GetCollection<T>() where T : class, new()
    {
        return db.GetCollection<T>(GetCollectionName(typeof(T)));
    }

    public void InitTables(params Type[] entityTypes)
    {
        // LiteDB auto-creates collections on first use.
        // Ensure they exist by touching them.
        foreach (var type in entityTypes)
        {
            var name = GetCollectionName(type);
            db.GetCollection(name);
        }
    }

    public string GetTableName<T>() where T : class, new()
    {
        return GetCollectionName(typeof(T));
    }

    public void CreateIndex<T>(string[] columnNames) where T : class, new()
    {
        var col = GetCollection<T>();
        foreach (var columnName in columnNames)
        {
            col.EnsureIndex(columnName);
        }
    }

    public IDatabaseQueryable<T> Queryable<T>() where T : class, new()
    {
        return new LiteDbQueryable<T>(db, GetCollection<T>().Query());
    }

    public Task<int> InsertAsync<T>(T entity) where T : class, new()
    {
        return Task.Run(() =>
        {
            var col = GetCollection<T>();
            col.Insert(entity);
            return 1;
        });
    }

    public Task<int> InsertRangeAsync<T>(List<T> entities) where T : class, new()
    {
        return Task.Run(() =>
        {
            var col = GetCollection<T>();
            return col.InsertBulk(entities);
        });
    }

    public Task<int> InsertReturnIdentityAsync<T>(T entity) where T : class, new()
    {
        return Task.Run(() =>
        {
            var col = GetCollection<T>();
            var bsonId = col.Insert(entity);
            return bsonId.AsInt32;
        });
    }

    public Task<int> UpdateAsync<T>(T entity) where T : class, new()
    {
        return Task.Run(() =>
        {
            var col = GetCollection<T>();
            return col.Update(entity) ? 1 : 0;
        });
    }

    public Task<int> UpdateRangeAsync<T>(List<T> entities) where T : class, new()
    {
        return Task.Run(() =>
        {
            var col = GetCollection<T>();
            return col.Update(entities);
        });
    }

    public Task<int> UpdateColumnsAsync<T>(T entity, Expression<Func<T, object>> columns)
        where T : class, new()
    {
        return Task.Run(() =>
        {
            var col = GetCollection<T>();

            // Get the Id from entity to find the existing document
            var idProp = typeof(T).GetProperty("Id");
            var id = idProp?.GetValue(entity);

            if (id == null || (id is int intId && intId == 0))
            {
                return 0;
            }

            var existing = col.FindById(new BsonValue(id));
            if (existing == null) return 0;

            // Extract column names from the expression and copy only those
            var columnNames = ExtractMemberNames(columns);
            foreach (var name in columnNames)
            {
                var prop = typeof(T).GetProperty(name);
                if (prop != null)
                {
                    prop.SetValue(existing, prop.GetValue(entity));
                }
            }

            return col.Update(existing) ? 1 : 0;
        });
    }

    public Task<int> UpsertAsync<T>(T entity, Expression<Func<T, object>> matchColumns)
        where T : class, new()
    {
        return Task.Run(() =>
        {
            var col = GetCollection<T>();
            var (propName, propValue) = ExtractMatchColumn(entity, matchColumns);

            var existing = col.FindOne(Query.EQ(propName, new BsonValue(propValue)));
            if (existing != null)
            {
                // Copy the Id from existing to entity so Update targets the right document
                var idProp = typeof(T).GetProperty("Id");
                if (idProp != null)
                {
                    idProp.SetValue(entity, idProp.GetValue(existing));
                }

                col.Update(entity);
            }
            else
            {
                col.Insert(entity);
            }

            return 1;
        });
    }

    public Task<int> DeleteAsync<T>(Expression<Func<T, bool>> predicate) where T : class, new()
    {
        return Task.Run(() =>
        {
            var col = GetCollection<T>();
            return col.DeleteMany(predicate);
        });
    }

    public Task<int> UpdateSetWhereAsync<T>(Expression<Func<T, T>> setColumns, Expression<Func<T, bool>> where)
        where T : class, new()
    {
        return Task.Run(() =>
        {
            // LiteDB is a single-process embedded file, so each server has its own DB — there is no
            // cross-server race here (that only happens on a shared SQL backend, handled atomically
            // by DatabaseProvider). Apply the SET expression to every matching row.
            var col       = GetCollection<T>();
            var setMembers = ExtractInitMembers(setColumns);
            var apply      = setColumns.Compile();

            var matches = col.Find(where).ToList();
            var n = 0;
            foreach (var existing in matches)
            {
                var template = apply(existing);
                foreach (var name in setMembers)
                {
                    var prop = typeof(T).GetProperty(name);
                    if (prop is not null && prop.CanWrite)
                        prop.SetValue(existing, prop.GetValue(template));
                }
                if (col.Update(existing)) n++;
            }
            return n;
        });
    }

    public Task<IDatabaseTransaction> BeginTransactionAsync()
    {
        return Task.Run<IDatabaseTransaction>(() => new LiteDbTransaction(db));
    }

    #region Helpers

    private static (string PropertyName, object Value) ExtractMatchColumn<T>(
        T entity,
        Expression<Func<T, object>> matchColumns
    )
    {
        var body = matchColumns.Body;
        if (body is UnaryExpression unary && unary.NodeType == ExpressionType.Convert)
            body = unary.Operand;

        if (body is MemberExpression member)
        {
            var propName = member.Member.Name;
            var propValue = matchColumns.Compile()(entity);
            return (propName, propValue);
        }

        throw new ArgumentException("matchColumns must be a simple property expression");
    }

    private static List<string> ExtractMemberNames<T>(Expression<Func<T, object>> expression)
    {
        var body = expression.Body;
        if (body is UnaryExpression unary && unary.NodeType == ExpressionType.Convert)
            body = unary.Operand;

        if (body is NewExpression newExpr)
        {
            return newExpr.Members?.Select(m => m.Name).ToList() ?? [];
        }

        if (body is MemberExpression member)
        {
            return [member.Member.Name];
        }

        throw new ArgumentException("Cannot extract member names from expression");
    }

    // Member names assigned in a `x => new T { A = ..., B = ... }` initializer.
    private static List<string> ExtractInitMembers<T>(Expression<Func<T, T>> expression)
    {
        if (expression.Body is MemberInitExpression init)
            return init.Bindings.Select(b => b.Member.Name).ToList();
        if (expression.Body is NewExpression newExpr)
            return newExpr.Members?.Select(m => m.Name).ToList() ?? [];
        throw new ArgumentException("setColumns must be a member-init expression: x => new T { ... }");
    }

    #endregion
}
```

```cs
using System;
using System.Collections.Generic;
using System.Linq;
using System.Linq.Expressions;
using System.Reflection;
using System.Threading.Tasks;
using LiteDB;
using CallAdmin.Database.Shared;

namespace CallAdmin.Database.Provider;

/// <summary>
/// In-memory join implementation for LiteDB.
/// Executes the T1 query first, then fetches matching T2 records and joins in memory.
/// Suitable for small result sets (e.g., leaderboards with Take/Limit).
/// </summary>
internal sealed class LiteDbJoinQueryable<T1, T2>(
    LiteDatabase db,
    ILiteQueryable<T1> query,
    Expression<Func<T1, T2, bool>> joinExpression
) : IDatabaseJoinQueryable<T1, T2>
    where T1 : class, new()
    where T2 : class, new()
{
    private Expression<Func<T1, T2, bool>>? _whereExpr;
    private Expression<Func<T1, T2, object>>? _orderByDescExpr;
    private int? _take;

    public IDatabaseJoinQueryable<T1, T2> Where(Expression<Func<T1, T2, bool>> predicate)
    {
        _whereExpr = predicate;
        return this;
    }

    public IDatabaseJoinQueryable<T1, T2> OrderByDescending(
        Expression<Func<T1, T2, object>> keySelector
    )
    {
        _orderByDescExpr = keySelector;
        return this;
    }

    public IDatabaseJoinQueryable<T1, T2> Take(int count)
    {
        _take = count;
        return this;
    }

    public Task<List<TResult>> SelectToListAsync<TResult>(
        Expression<Func<T1, T2, TResult>> selector
    )
        where TResult : class, new()
    {
        return Task.Run(() =>
        {
            // 1. Execute T1 query
            var list1 = query.ToList();

            // 2. Fetch all T2 records
            var collectionName = GetCollectionName(typeof(T2));
            var col2 = db.GetCollection<T2>(collectionName);
            var list2 = col2.FindAll().ToList();

            // 3. Join in memory
            var joinFunc = joinExpression.Compile();
            IEnumerable<(T1 t1, T2 t2)> joined =
                from t1 in list1
                from t2 in list2
                where joinFunc(t1, t2)
                select (t1, t2);

            // 4. Apply Where
            if (_whereExpr != null)
            {
                var whereFunc = _whereExpr.Compile();
                joined = joined.Where(pair => whereFunc(pair.t1, pair.t2));
            }

            // 5. Apply OrderByDescending
            if (_orderByDescExpr != null)
            {
                var orderFunc = _orderByDescExpr.Compile();
                joined = joined.OrderByDescending(pair => orderFunc(pair.t1, pair.t2));
            }

            // 6. Apply Take
            if (_take.HasValue)
            {
                joined = joined.Take(_take.Value);
            }

            // 7. Select
            var selectFunc = selector.Compile();
            return joined.Select(pair => selectFunc(pair.t1, pair.t2)).ToList();
        });
    }

    private static string GetCollectionName(Type type)
    {
        var attr = type.GetCustomAttribute<DbTableAttribute>();
        return attr?.TableName ?? type.Name;
    }
}
```

```cs
using System;
using System.Threading.Tasks;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.Logging;
using CallAdmin.Discord.Configuration;
using CallAdmin.Discord.Services;
using CallAdmin.Shared;
using CallAdmin.Shared.Models;
using Sharp.Shared;

namespace CallAdmin.Discord;

/// <summary>
/// Discord notifier for CallAdmin. Subscribes to ICallAdminShared lifecycle events and pushes
/// (POST) a webhook embed on submit, then edits (PATCH) that same embed in place as the report
/// transitions claimed → handled / dismissed / cancelled. Pure consumer — does not reference
/// CallAdmin.Core. Event handlers fire on the publishing thread, so all network work is
/// dispatched onto Task.Run to avoid blocking the game loop.
/// </summary>
public sealed class CallAdminDiscordPlugin : IModSharpModule
{
    public string DisplayName   => "CallAdmin.Discord";
    public string DisplayAuthor => "yappershq";

    private readonly ILogger<CallAdminDiscordPlugin> _logger;
    private readonly DiscordConfig                    _config;
    private readonly WebhookClient                    _webhook;
    private readonly ISharedSystem                    _shared;
    private ICallAdminShared?                         _callAdmin;

    // Cached delegates so we can unhook on Shutdown
    private Action<Report>? _onSubmitted;
    private Action<Report>? _onClaimed;
    private Action<Report>? _onHandled;
    private Action<Report>? _onDismissed;
    private Action<Report>? _onCancelled;

    public CallAdminDiscordPlugin(
        ISharedSystem  sharedSystem,
        string         dllPath,
        string         sharpPath,
        Version        version,
        IConfiguration configuration,
        bool           hotReload)
    {
        var loggerFactory = sharedSystem.GetLoggerFactory();
        _logger  = loggerFactory.CreateLogger<CallAdminDiscordPlugin>();
        _shared  = sharedSystem;
        _config  = DiscordConfig.Load(sharpPath, loggerFactory.CreateLogger<DiscordConfig>());
        _webhook = new WebhookClient(loggerFactory.CreateLogger<WebhookClient>());
    }

    public bool Init() => true;
    public void PostInit() { }

    public void OnAllModulesLoaded()
    {
        if (string.IsNullOrWhiteSpace(_config.WebhookUrl))
        {
            _logger.LogWarning("[CallAdmin.Discord] webhookUrl is empty in calladmin.discord.json — nothing will be posted");
            return;
        }

        _callAdmin = _shared.GetSharpModuleManager()
            .GetOptionalSharpModuleInterface<ICallAdminShared>(ICallAdminShared.Identity)?.Instance;
        if (_callAdmin is null)
        {
            _logger.LogWarning("[CallAdmin.Discord] ICallAdminShared not available — CallAdmin.Core not loaded?");
            return;
        }

        _onSubmitted = r => _ = OnSubmittedAsync(r);
        _onClaimed   = r => _ = OnLifecycleAsync(r);
        _onHandled   = r => _ = OnLifecycleAsync(r);
        _onDismissed = r => _ = OnLifecycleAsync(r);
        _onCancelled = r => _ = OnLifecycleAsync(r);

        _callAdmin.OnReportSubmitted += _onSubmitted;
        _callAdmin.OnReportClaimed   += _onClaimed;
        _callAdmin.OnReportHandled   += _onHandled;
        _callAdmin.OnReportDismissed += _onDismissed;
        _callAdmin.OnReportCancelled += _onCancelled;

        _logger.LogInformation("[CallAdmin.Discord] Subscribed to CallAdmin events; webhook ready");
    }

    public void Shutdown()
    {
        if (_callAdmin is null) return;
        if (_onSubmitted is not null) _callAdmin.OnReportSubmitted -= _onSubmitted;
        if (_onClaimed   is not null) _callAdmin.OnReportClaimed   -= _onClaimed;
        if (_onHandled   is not null) _callAdmin.OnReportHandled   -= _onHandled;
        if (_onDismissed is not null) _callAdmin.OnReportDismissed -= _onDismissed;
        if (_onCancelled is not null) _callAdmin.OnReportCancelled -= _onCancelled;
    }

    private async Task OnSubmittedAsync(Report report)
    {
        try
        {
            var payload = EmbedBuilder.BuildPayload(report, _config, initialPost: true);
            var messageId = await _webhook.SendAsync(_config.WebhookUrl, payload);
            if (messageId is null) return;
            // Persist the Discord message id so subsequent lifecycle events can PATCH instead of repost.
            if (_callAdmin is not null)
                await _callAdmin.SetExternalMessageIdAsync(report.Id, messageId);
        }
        catch (Exception e) { _logger.LogError(e, "[CallAdmin.Discord] OnSubmitted failed for report #{Id}", report.Id); }
    }

    private async Task OnLifecycleAsync(Report report)
    {
        try
        {
            // If we never stored a message id (first POST failed, or report was never posted), fall back to a new POST.
            var payload = EmbedBuilder.BuildPayload(report, _config, initialPost: string.IsNullOrEmpty(report.ExternalMessageId));
            if (string.IsNullOrEmpty(report.ExternalMessageId))
            {
                var messageId = await _webhook.SendAsync(_config.WebhookUrl, payload);
                if (messageId is not null && _callAdmin is not null)
                    await _callAdmin.SetExternalMessageIdAsync(report.Id, messageId);
            }
            else
            {
                await _webhook.EditAsync(_config.WebhookUrl, report.ExternalMessageId, payload);
            }
        }
        catch (Exception e) { _logger.LogError(e, "[CallAdmin.Discord] Lifecycle update failed for report #{Id}", report.Id); }
    }
}
```

```cs
        var path = Path.Combine(sharpPath, "configs", "calladmin.discord.json");
        try
        {
            if (!File.Exists(path))
            {
                var def = new DiscordConfig();
                Directory.CreateDirectory(Path.GetDirectoryName(path)!);
                File.WriteAllText(path, JsonSerializer.Serialize(def, JsonOpts));
                logger.LogInformation("[CallAdmin.Discord] Wrote default config to {Path}", path);
                return def;
            }
            var cfg = JsonSerializer.Deserialize<DiscordConfig>(File.ReadAllText(path), JsonOpts);
            return cfg ?? new DiscordConfig();
        }
        catch (Exception e)
        {
            logger.LogError(e, "[CallAdmin.Discord] Failed to load discord config");
            return new DiscordConfig();
        }
    }
```

`CallAdmin.Discord/Services/EmbedBuilder.cs`:

```cs
using System;
using System.Collections.Generic;
using System.Globalization;
using CallAdmin.Discord.Configuration;
using CallAdmin.Shared.Models;

namespace CallAdmin.Discord.Services;

/// <summary>
/// Builds Discord webhook payload dictionaries (serialized to JSON by the client).
/// One embed per report, status-coloured. Stable field layout so PATCH edits look correct.
/// </summary>
internal static class EmbedBuilder
{
    // Discord embed colors (decimal). Pending=red so it stands out, terminal states fade.
    private const int ColorPending   = 15548997; // red
    private const int ColorClaimed   = 16776960; // yellow
    private const int ColorHandled   = 3066993;  // green
    private const int ColorDismissed = 9807270;  // grey
    private const int ColorCancelled = 4934475;  // dark grey

    public static object BuildPayload(Report report, DiscordConfig config, bool initialPost)
    {
        var embed = BuildEmbed(report, config);
        var dict  = new Dictionary<string, object?> { ["embeds"] = new[] { embed } };

        // Mention the role only on the initial post; edits suppress repeats and keep mention silent.
        if (initialPost && report.Status == ReportStatus.Pending && !string.IsNullOrWhiteSpace(config.MentionRoleId))
        {
            dict["content"] = $"<@&{config.MentionRoleId}>";
            dict["allowed_mentions"] = new Dictionary<string, object?>
            {
                ["roles"] = new[] { config.MentionRoleId },
            };
        }
        else if (!initialPost)
        {
            // Strip content on edit so we don't re-ping
            dict["content"] = "";
            dict["allowed_mentions"] = new Dictionary<string, object?>
            {
                ["parse"] = Array.Empty<string>(),
            };
        }

        return dict;
    }

    private static object BuildEmbed(Report r, DiscordConfig config)
    {
        var fields = new List<object>
        {
            Field("Reporter", PlayerCell(r.ReporterName, r.ReporterSteamId, config), inline: true),
            Field("Target",   PlayerCell(r.TargetName,   r.TargetSteamId,   config), inline: true),
            Field("Reason",   $"`{Escape(r.ReasonKey)}`",                          inline: true),
            Field("Map",      Escape(r.MapName),                                   inline: true),
            Field("Status",   StatusLabel(r.Status),                               inline: true),
        };

        if (!string.IsNullOrWhiteSpace(r.ServerName))
            fields.Add(Field("Server", Escape(r.ServerName), inline: true));

        if (r.HandlerName is { Length: > 0 })
            fields.Add(Field("Handled by", r.HandlerSteamId is { } hsid ? PlayerCell(r.HandlerName, hsid, config) : Escape(r.HandlerName), inline: false));

        if (r.Outcome is { Length: > 0 })
            fields.Add(Field("Outcome", OutcomeLabel(r.Outcome), inline: true));

        // Admin commands for this report — what to type next, based on current stage. steam:// links
        // aren't clickable in embeds, so admins need the in-game commands spelled out per stage.
        var commands = CommandsFor(r);
        if (commands is not null)
            fields.Add(Field("How to handle (in-game chat)", commands, inline: false));

        // Optional connect link. steam:// is dead text in embeds — link a https connect-redirect.
        if (!string.IsNullOrWhiteSpace(config.ConnectIp) && config.ConnectPort > 0
            && !string.IsNullOrWhiteSpace(config.ConnectUrlBase))
        {
            var url = $"{config.ConnectUrlBase.TrimEnd('/')}/{config.ConnectIp}:{config.ConnectPort}";
            fields.Add(Field("Connect", $"[{config.ConnectIp}:{config.ConnectPort}]({url})", inline: false));
        }

        var embed = new Dictionary<string, object?>
        {
            ["title"]       = string.IsNullOrWhiteSpace(config.EmbedTitle) ? "CallAdmin Report" : config.EmbedTitle,
            ["description"] = $"Report **#{r.Id}** — {StatusLabel(r.Status)}",
            ["color"]       = StatusColor(r.Status),
            ["timestamp"]   = r.CreatedAt.ToString("o", CultureInfo.InvariantCulture),
            ["fields"]      = fields,
            ["footer"]      = new Dictionary<string, object?>
            {
                ["text"] = string.IsNullOrWhiteSpace(config.EmbedFooter) ? $"CallAdmin · #{r.Id}" : $"{config.EmbedFooter} · #{r.Id}",
            },
        };

        return embed;
    }

    // In-game admin commands to advance the report from its current stage. Null for terminal
    // states (nothing left to do). Fenced as a code block so the columns line up in Discord.
    private static string? CommandsFor(Report r) => r.Status switch
    {
        ReportStatus.Pending => $"```\n1) Claim it   →  !claim {r.Id}\n2) Resolve    →  !handled {r.Id}\n   or dismiss →  !dismiss {r.Id}\n```",
        ReportStatus.Claimed => $"```\nResolve   →  !handled {r.Id}\nDismiss   →  !dismiss {r.Id}\n```",
        _                    => null,
    };

    private static int StatusColor(ReportStatus s) => s switch
    {
        ReportStatus.Pending   => ColorPending,
        ReportStatus.Claimed   => ColorClaimed,
        ReportStatus.Handled   => ColorHandled,
        ReportStatus.Dismissed => ColorDismissed,
        ReportStatus.Cancelled => ColorCancelled,
        _                      => ColorPending,
    };

    private static string StatusLabel(ReportStatus s) => s switch
    {
        ReportStatus.Pending   => "🔴 Pending",
        ReportStatus.Claimed   => "🟡 Claimed",
        ReportStatus.Handled   => "🟢 Handled",
        ReportStatus.Dismissed => "⚪ Dismissed",
        ReportStatus.Cancelled => "⚫ Cancelled",
        _                      => s.ToString(),
    };

    private static string OutcomeLabel(string outcome) => outcome.ToLowerInvariant() switch
    {
        "ban"       => "🔨 Ban",
        "mute"      => "🔇 Mute",
        "gag"       => "🤐 Gag",
        "kick"      => "👢 Kick",
        "warn"      => "⚠️ Warning",
        "dismissed" => "⚪ Dismissed (no action)",
        _           => Escape(outcome),
    };

    private static Dictionary<string, object?> Field(string name, string value, bool inline) => new()
    {
        ["name"]   = name,
        ["value"]  = string.IsNullOrEmpty(value) ? "-" : value,
        ["inline"] = inline,
    };

    // Player cell: name links to the web lookup (moderavimas) for that SteamID when configured,
    // with the raw SteamID kept in a code block for copy-paste. URL base comes from config.
    private static string PlayerCell(string name, ulong steamId, DiscordConfig config)
    {
        var esc = Escape(name);
        if (!string.IsNullOrWhiteSpace(config.LookupUrlBase))
        {
            var url = $"{config.LookupUrlBase.TrimEnd('/')}?tab=player_lookup&steamId={steamId}";
            return $"[{esc}]({url})\n`{steamId}`";
        }
        return $"{esc}\n`{steamId}`";
    }

    // Escape Discord markdown / mention chars so player names can't break the embed
    private static string Escape(string s)
    {
        if (string.IsNullOrEmpty(s)) return string.Empty;
        // Only need to neuter the formatting set. backticks block code formatting.
        return s
            .Replace("\\", "\\\\")
            .Replace("`", "\\`")
            .Replace("*", "\\*")
            .Replace("_", "\\_")
            .Replace("~", "\\~")
            .Replace("|", "\\|")
            .Replace("@everyone", "@​everyone")
            .Replace("@here",     "@​here");
    }
}
```

```cs
using System;
using System.Net.Http;
using System.Text;
using System.Text.Json;
using System.Threading.Tasks;
using Microsoft.Extensions.Logging;

namespace CallAdmin.Discord.Services;

internal sealed class WebhookClient
{
    private static readonly HttpClient Http = new();

    private readonly ILogger _logger;

    public WebhookClient(ILogger<WebhookClient> logger) => _logger = logger;

    public async Task<string?> SendAsync(string url, object payload)
    {
        if (string.IsNullOrWhiteSpace(url)) return null;

        try
        {
            // wait=true so the response carries the message id (needed to edit-in-place later).
            var endpoint = url.Contains("?wait=") ? url : (url + (url.Contains('?') ? "&" : "?") + "wait=true");

            var json    = JsonSerializer.Serialize(payload);
            using var req = new HttpRequestMessage(HttpMethod.Post, endpoint)
            {
                Content = new StringContent(json, Encoding.UTF8, "application/json"),
            };
            using var resp = await Http.SendAsync(req);
            var body = await resp.Content.ReadAsStringAsync();

            if (!resp.IsSuccessStatusCode)
            {
                _logger.LogWarning("[CallAdmin.Discord] Webhook POST {Status}: {Body}", resp.StatusCode, body);
                return null;
            }
            using var doc = JsonDocument.Parse(body);
            return doc.RootElement.TryGetProperty("id", out var idEl) ? idEl.GetString() : null;
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin.Discord] Webhook POST failed");
            return null;
        }
    }

    public async Task EditAsync(string url, string messageId, object payload)
    {
        if (string.IsNullOrWhiteSpace(url) || string.IsNullOrWhiteSpace(messageId)) return;
        try
        {
            var trimmed  = url.Split('?')[0];
            var endpoint = $"{trimmed}/messages/{messageId}";
            var json     = JsonSerializer.Serialize(payload);
            using var req = new HttpRequestMessage(HttpMethod.Patch, endpoint)
            {
                Content = new StringContent(json, Encoding.UTF8, "application/json"),
            };
            using var resp = await Http.SendAsync(req);
            if (!resp.IsSuccessStatusCode)
                _logger.LogWarning("[CallAdmin.Discord] Webhook PATCH {Status}", resp.StatusCode);
        }
        catch (Exception e)
        {
            _logger.LogError(e, "[CallAdmin.Discord] Webhook PATCH failed");
        }
    }
}
```

```cs
using System;

namespace CallAdmin.Shared.Models;

/// <summary>Public read-only view of a report. Extensions consume this via events / API.</summary>
public sealed class Report
{
    public int          Id              { get; init; }
    public ulong        ReporterSteamId { get; init; }
    public string       ReporterName    { get; init; } = string.Empty;
    public ulong        TargetSteamId   { get; init; }
    public string       TargetName      { get; init; } = string.Empty;
    public string       ReasonKey       { get; init; } = string.Empty;
    public string       ReasonLabel     { get; init; } = string.Empty;
    public string       MapName         { get; init; } = string.Empty;
    public string       ServerName      { get; init; } = string.Empty;
    public DateTime     CreatedAt       { get; init; }

    public ReportStatus Status          { get; init; }
    public ulong?       HandlerSteamId  { get; init; }
```

<div align="center">
  <h1>📣 <strong>CallAdmin</strong></h1>
  <p>In-game player report system for CS2 ModSharp servers — players <code>!report</code> cheaters, admins claim and resolve them.</p>

---

CallAdmin lets players report rule-breakers from chat and gives admins a claim/handle/dismiss workflow, with persistence (LiteDB / MySQL / PostgreSQL) and an optional Discord webhook notifier that posts an embed and edits it in place as the report progresses. It is a from-scratch ModSharp rewrite inspired by [Impact123/CallAdmin](https://github.com/Impact123/CallAdmin) (SourceMod) and [1Mack/CS2-CallAdmin](https://github.com/1Mack/CS2-CallAdmin) (CounterStrikeSharp).

The plugin ships as several projects so you can run only what you need:

| Project | Role |
|---------|------|
| `CallAdmin.Core` | Main plugin — chat commands, report lifecycle, business logic |
| `CallAdmin.Shared` | Public API (`ICallAdminShared`) + report models, consumed by other plugins |
| `CallAdmin.Database` | Persistence backend — LiteDB / MySQL / PostgreSQL, switchable in config |
| `CallAdmin.Database.Shared` | Database abstraction contract |
| `CallAdmin.Discord` | Optional Discord webhook notifier (subscribes to CallAdmin events) |

## 🚀 Install

Copy the build output into your ModSharp install (`<sharp>` = your `sharp` directory):

| From | To |
|------|----|
| `.build/modules/CallAdmin.Core/` | `<sharp>/modules/CallAdmin.Core/` |
| `.build/modules/CallAdmin.Database/` | `<sharp>/modules/CallAdmin.Database/` |
| `.build/modules/CallAdmin.Discord/` | `<sharp>/modules/CallAdmin.Discord/` (optional) |
| `.build/shared/CallAdmin.Shared/` | `<sharp>/shared/CallAdmin.Shared/` |
| `.build/shared/CallAdmin.Database.Shared/` | `<sharp>/shared/CallAdmin.Database.Shared/` |
| `.build/locales/calladmin.json` | `<sharp>/locales/calladmin.json` |
| `.assets/configs.example/*` | `<sharp>/configs/` (rename per file, see below) |

`CallAdmin.Core` writes a default `configs/calladmin.json` on first run if none exists. Restart the server (or change map) to load. CallAdmin uses ModSharp's CommandCenter, AdminManager and LocalizerManager, which ship with the framework. `CallAdmin.Discord` is optional — install it only if you want Discord notifications.

## ⌨️ Commands

All commands are chat-triggered (also usable from console).

| Command | Aliases | Description | Permission |
|---------|---------|-------------|------------|
| `!report <target> <reason>` | `!calladmin` | Submit a report. Target via ModSharp targeting (`name`, `#name`, `@me`, `@t`, `@ct`, SteamID64). | _none by default_ |
| `!report_cancel` | — | Cancel your most recent pending report (within the cancel window). | reporter only |
| `!calls` | `!reports` | List pending reports. | `@calladmin/handle` |
| `!claim <id>` | — | Claim a pending report. | `@calladmin/handle` |
| `!handled <id>` | — | Mark a report handled (terminal). | `@calladmin/handle` |
| `!dismiss <id>` | — | Dismiss a report as a false alarm (terminal). | `@calladmin/handle` |

Admin permissions are registered via `MountAdminManifest` under the `CallAdmin` table:

| Permission | Grants |
|------------|--------|
| `@calladmin/report` | Submit reports — only enforced when `requireReportPermission` is `true` |
| `@calladmin/handle` | View pending, claim, handle, dismiss |
| `@calladmin/report_admin` | Report players who otherwise have admin-target immunity |

## ⚙️ Configuration

### `configs/calladmin.json`

Auto-generated on first run. Report reasons, cooldowns and anti-abuse gates:

| Setting | Default | Meaning |
|---------|---------|---------|
| `reasons` | hack, wallhack, aimbot, toxic, spam, griefing, other | Report reason list (`key` + `localeKey` label pairs) |
| `autoCloseMinutes` | `60` | Minutes a pending report may sit before auto-closing as cancelled. `0` disables |
| `cooldownSeconds` | `300` | Per-reporter cooldown between reports |
| `cancelWindowSeconds` | `300` | How long after submitting a reporter may `!report_cancel` |
| `duplicateWindowSeconds` | `600` | Window in which a repeat report of the same target is treated as a duplicate |
| `notifyAdminsInChat` | `true` | Announce new reports to online admins in chat |
| `allowReportAdmins` | `false` | Allow reporting players who hold admin |
| `allowReportSpectators` | `false` | Allow reporting spectators |
| `requireReportPermission` | `false` | When `true`, only players with `@calladmin/report` may report |

### `configs/calladmin.database.jsonc`

Selects the persistence backend. `Type` is `litedb` (zero-config single file at `sharp/data/calladmin.db`), `mysql`, or `postgresql`. For MySQL/PostgreSQL, fill `Host` / `Port` / `Database` / `User` / `Password`.

### `configs/calladmin.discord.json` (optional)

Used by `CallAdmin.Discord`. Set `webhookUrl` to enable; leave it empty to disable posting. Other keys cover the embed title/footer, an optional role mention, and the connect/lookup links shown in the embed.

## 🔧 How it works

A player runs `!report <target> <reason>`; Core resolves the target via ModSharp targeting, applies the cooldown / duplicate / immunity checks, then persists a `Report` through the database backend and fires the lifecycle events. Admins with `@calladmin/handle` see pending reports (`!calls`) and move each through `claim → handled` or `dismiss`. After persistence each transition raises a `Report` event on `ICallAdminShared`; `CallAdmin.Discord` subscribes to those events to POST a webhook embed on submit and PATCH it in place as the report is claimed, handled, dismissed or cancelled. If [AdminPanel](https://github.com/yappershq) is installed, Core also surfaces the list/claim/handle/dismiss actions inside the in-game admin menu (resolved with a null-guard, so it stays standalone-safe).

## 🧩 Public API

Other plugins consume `ICallAdminShared` (resolve in `OnAllSharpModulesLoaded`) to subscribe to the report lifecycle or drive reports programmatically:

```csharp
var callAdmin = sharpModuleManager
    .GetOptionalSharpModuleInterface<ICallAdminShared>(ICallAdminShared.Identity)?.Instance;

callAdmin.OnReportSubmitted += report => { /* react */ };
// also: OnReportClaimed, OnReportHandled, OnReportDismissed, OnReportCancelled
```

The interface also exposes `SubmitReportAsync` / `ClaimReportAsync` / `MarkHandledAsync` / `DismissReportAsync` / `CancelReportAsync`, lookups (`GetReportAsync`, `GetPendingReportsAsync`, `GetReasons`), `RegisterTargetFilter` (hide players from the target list), and `SetExternalMessageIdAsync` (store an external message id alongside a report). Handlers run synchronously on the publishing thread — offload network/DB work with `_ = Task.Run(...)`.
