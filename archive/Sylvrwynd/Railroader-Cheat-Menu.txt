`build.ps1`:

```ps1
param(
    [string]$RailroaderManagedPath = $env:RAILROADER_MANAGED
)

$ErrorActionPreference = 'Stop'

if ([string]::IsNullOrWhiteSpace($RailroaderManagedPath)) {
    $candidates = @(
        'D:\SteamLibrary\steamapps\common\Railroader\Railroader_Data\Managed',
        'C:\Program Files (x86)\Steam\steamapps\common\Railroader\Railroader_Data\Managed'
    )
    $RailroaderManagedPath = $candidates | Where-Object { Test-Path -LiteralPath $_ } | Select-Object -First 1
}

if ([string]::IsNullOrWhiteSpace($RailroaderManagedPath) -or -not (Test-Path -LiteralPath $RailroaderManagedPath)) {
    throw 'RailroaderManagedPath must point to Railroader_Data\Managed.'
}

$contract = Join-Path $RailroaderManagedPath 'Railroader.Modding.dll'
$game = Join-Path $RailroaderManagedPath 'Railroader.dll'
if (-not (Test-Path -LiteralPath $contract) -or -not (Test-Path -LiteralPath $game)) {
    throw "The managed directory does not contain Railroader.Modding.dll and Railroader.dll: $RailroaderManagedPath"
}

$root = Split-Path -Parent $MyInvocation.MyCommand.Path
$package = Join-Path $root 'dist\RailroaderCheatMenu'
$output = Join-Path $root 'dist\build\'
if (Test-Path -LiteralPath $package) {
    Remove-Item -LiteralPath $package -Recurse -Force
}
New-Item -ItemType Directory -Path $package | Out-Null

dotnet build (Join-Path $root 'src\RailroaderCheatMenu\RailroaderCheatMenu.csproj') `
    --configuration Release `
    -p:RailroaderManagedPath=$RailroaderManagedPath `
    -p:OutputPath="$output"
if ($LASTEXITCODE -ne 0) {
    throw "dotnet build failed with exit code $LASTEXITCODE"
}

Copy-Item -LiteralPath (Join-Path $root 'mod\Info.json') -Destination $package
Copy-Item -LiteralPath (Join-Path $output 'RailroaderCheatMenu.dll') -Destination $package

Write-Output "Built asset pack: $package"
```

```cs
using System;
using System.Globalization;
using Railroader.Modding;
using UnityEngine;

namespace CAG.Sylvrwynd.RailroaderCheatMenu
{
    internal sealed class CheatMenuBehaviour : MonoBehaviour
    {
        private static CheatMenuBehaviour instance;
        private static bool worldLoaded;

        private IModLogger log;
        private bool open;
        private Rect window = new Rect(30f, 30f, 540f, 500f);
        private Vector2 scrollPosition;
        private int tab;
        private string cashAmount = "10000";
        private string materialAmount = "1000000";
        private string fuelAmount = "1000000";
        private string reputationAmount = "100";
        private string impactDamagePercent = "100";
        private string overspeedDamagePercent = "100";
        private string wearMultiplierPercent = "100";
        private string oilUsePercent = "100";
        private string fuelConsumptionPercent = "100";
        private string highSpeedThresholdMph = "50";
        private int trackClass = 5;
        private string status = Localization.Text("ready");
        private bool resizingWindow;
        private Vector2 resizeStartMouse;
        private Vector2 resizeStartSize;

        private const float MinWindowWidth = 480f;
        private const float MinWindowHeight = 320f;
        private const float ResizeGripSize = 18f;

        private GUIStyle windowStyle;
        private GUIStyle buttonStyle;
        private GUIStyle tabStyle;
        private GUIStyle labelStyle;
        private GUIStyle textFieldStyle;
        private GUIStyle closeButtonStyle;
        private GUIStyle resizeGripStyle;
        private Texture2D windowBackground;
        private Texture2D controlBackground;
        private Texture2D controlHoverBackground;
        private Texture2D controlActiveBackground;

        internal void Initialize(IModLogger logger)
        {
            instance = this;
            log = logger;
        }

        internal static void SetWorldLoaded(bool loaded)
        {
            worldLoaded = loaded;
            if (!loaded && instance != null)
            {
                instance.open = false;
            }
        }

        private void Update()
        {
            if (Input.GetKeyDown(KeyCode.F10))
            {
                open = !open;
            }
        }

        private void OnGUI()
        {
            if (!open || !worldLoaded)
            {
                return;
            }

            EnsureStyles();
            var previousColor = GUI.color;
            GUI.color = Color.white;
            window = GUI.Window(GetInstanceID(), window, DrawWindow, Localization.Text("menu_title"), windowStyle);
            GUI.color = previousColor;
        }

        private void DrawWindow(int id)
        {
            GUILayout.BeginVertical();
            scrollPosition = GUILayout.BeginScrollView(scrollPosition);
            GUILayout.BeginHorizontal();
            GUILayout.Label(Localization.Text("target"), labelStyle);
            GUILayout.FlexibleSpace();
            if (GUILayout.Button(
                Localization.Text("language") + ": " + Localization.LanguageName,
                buttonStyle,
                GUILayout.Width(145f),
                GUILayout.Height(24f)))
            {
                Localization.Toggle();
                status = Localization.Text("ready");
            }
            GUILayout.EndHorizontal();
            GUILayout.Label(GameActions.HostStatus(), labelStyle);
            GUILayout.Label(status, labelStyle);
            GUILayout.Space(5f);

            GUILayout.BeginHorizontal();
            if (GUILayout.Toggle(tab == 0, Localization.Text("company"), tabStyle, GUILayout.Height(25f))) tab = 0;
            if (GUILayout.Toggle(tab == 1, Localization.Text("equipment"), tabStyle, GUILayout.Height(25f))) tab = 1;
            if (GUILayout.Toggle(tab == 2, Localization.Text("track_mow"), tabStyle, GUILayout.Height(25f))) tab = 2;
            if (GUILayout.Toggle(tab == 3, Localization.Text("safety"), tabStyle, GUILayout.Height(25f))) tab = 3;
            if (GUILayout.Toggle(tab == 4, Localization.Text("qol"), tabStyle, GUILayout.Height(25f))) tab = 4;
            GUILayout.EndHorizontal();
            GUILayout.Space(5f);

            if (tab == 0) DrawCompanyTab();
            else if (tab == 1) DrawEquipmentTab();
            else if (tab == 2) DrawTrackTab();
            else if (tab == 3) DrawSafetyTab();
            else DrawQolTab();

            GUILayout.Space(7f);
            if (ActionButton(Localization.Text("close"), 25f)) open = false;
            GUILayout.EndScrollView();
            GUILayout.EndVertical();

            if (GUI.Button(new Rect(window.width - 28f, 2f, 24f, 20f), "X", closeButtonStyle))
            {
                open = false;
            }

            DrawResizeGrip();
            GUI.DragWindow(new Rect(0f, 0f, Mathf.Max(0f, window.width - 32f), 24f));
        }

        private void DrawCompanyTab()
        {
            GUILayout.Label(Localization.Text("cash"), labelStyle);
            cashAmount = GUILayout.TextField(cashAmount, textFieldStyle, GUILayout.Height(23f));
            GUILayout.BeginHorizontal();
            if (ActionButton(Localization.Text("add"))) Run(() => GameActions.ChangeCash(ParseInt(cashAmount)));
            if (ActionButton(Localization.Text("remove"))) Run(() => GameActions.ChangeCash(-ParseInt(cashAmount)));
            if (ActionButton(Localization.Text("set_exact"))) Run(() => GameActions.SetCash(ParseInt(cashAmount)));
            GUILayout.EndHorizontal();
            GUILayout.Label(Localization.Format("current_balance", GameActions.CurrentBalance()), labelStyle);

            GUILayout.Space(8f);
            GUILayout.Label(Localization.Text("company_reputation"), labelStyle);
            reputationAmount = GUILayout.TextField(reputationAmount, textFieldStyle, GUILayout.Height(23f));
            if (ActionButton(Localization.Text("action_set")))
            {
                Run(() => GameActions.SetReputationPercent(ParseFloat(reputationAmount)));
            }
            GUILayout.Label(Localization.Format("current_reputation", GameActions.CurrentReputation()), labelStyle);
        }

        private void DrawEquipmentTab()
        {
            GUILayout.Label(Localization.Text("equipment_repair"), labelStyle);
            DrawActionPair(
                "repair_selected", "action_repair", GameActions.RepairSelected,
                "repair_all", "action_repair_all", GameActions.RepairAll);
            GUILayout.Label(Localization.Text("repair_description"), labelStyle);

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("engine_fueling"), labelStyle);
            DrawActionPair(
                "refuel_selected", "action_refuel", GameActions.RefuelSelected,
                "refuel_all", "action_refuel_all", GameActions.RefuelAll);
            GUILayout.Label(Localization.Text("fuel_description"), labelStyle);

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("cargo_loading"), labelStyle);
            DrawActionPair(
                "fill_manifested_selected", "action_fill", GameActions.FillManifestedSelected,
                "fill_manifested_all", "action_fill", GameActions.FillManifestedAll);
            GUILayout.Label(Localization.Text("manifested_cargo_description"), labelStyle);

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("cargo_emptying"), labelStyle);
            DrawActionPair(
                "empty_selected_cargo", "action_empty", GameActions.EmptySelectedCargo,
                "empty_all_cargo", "action_empty", GameActions.EmptyAllCargo);
            GUILayout.Label(Localization.Text("empty_cargo_description"), labelStyle);

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("oiling"), labelStyle);
            DrawActionPair(
                "oil_selected", "action_oil", GameActions.OilSelected,
                "oil_all", "action_oil_all", GameActions.OilAll);
            GUILayout.Label(Localization.Text("oil_description"), labelStyle);
        }

        private void DrawTrackTab()
        {
            GUILayout.Label(Localization.Text("track_condition"), labelStyle);
            GUILayout.BeginHorizontal();
            for (var value = 1; value <= 5; value++)
            {
                if (GUILayout.Toggle(trackClass == value, "C" + value, tabStyle, GUILayout.Height(24f))) trackClass = value;
            }
            GUILayout.EndHorizontal();
            DrawActionPair(
                "repair_track", "action_repair", GameActions.RepairAllTrack,
                "set_track_class", "action_set", () => GameActions.SetAllTrackConditionClass(trackClass),
                null,
                Localization.Format("set_track_class", trackClass));

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("mow_yard_materials"), labelStyle);
            DrawAmountActions(
                ref materialAmount,
                "action_fill",
                () => GameActions.RestockMowMaterials(ParseFloat(materialAmount)),
                "action_empty",
                GameActions.ClearMowMaterials);
            GUILayout.Label(Localization.Text("materials"), labelStyle);

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("yard_fuel_supplies"), labelStyle);
            DrawAmountActions(
                ref fuelAmount,
                "action_fill",
                () => GameActions.RestockFuelSupplies(ParseFloat(fuelAmount)),
                "action_empty",
                GameActions.ClearFuelSupplies);
            GUILayout.Label(Localization.Text("fuel_materials"), labelStyle);
        }

        private void DrawSafetyTab()
        {
            GUILayout.Label(Localization.Text("derailment_tools"), labelStyle);
            if (ActionButton(Localization.Text("action_rerail"), 23f)) Run(GameActions.RerailAll);
            DrawStatePair(
                Localization.Text("derailment_toggle"),
                GameActions.ToggleDerailmentProtection,
                RuntimePatches.PreventDerailment,
                Localization.Format("speed_damage_toggle", GameActions.CurrentHighSpeedDamageThreshold()),
                GameActions.ToggleHighSpeedDamageProtection,
                RuntimePatches.ProtectHighSpeedDamage);
            GUILayout.Label(
                Localization.Format("physics_description", GameActions.CurrentHighSpeedDamageThreshold()),
                labelStyle);

            GUILayout.BeginHorizontal();
            GUILayout.Label(Localization.Text("speed_damage_threshold"), labelStyle, GUILayout.Width(185f));
            highSpeedThresholdMph = GUILayout.TextField(highSpeedThresholdMph, textFieldStyle, GUILayout.Width(75f), GUILayout.Height(23f));
            var enteredThreshold = highSpeedThresholdMph;
            if (ActionButton(Localization.Text("apply")))
            {
                Run(() => GameActions.SetHighSpeedDamageThreshold(ParseFloat(enteredThreshold)));
            }
            GUILayout.EndHorizontal();
            GUILayout.Label(
                Localization.Format("current_speed_threshold", GameActions.CurrentHighSpeedDamageThreshold()),
                labelStyle);

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("wear_tools"), labelStyle);
            DrawStatePair(
                Localization.Text("equipment_wear_toggle"),
                GameActions.ToggleEquipmentWear,
                GameActions.EquipmentWearStatus() == Localization.Text("enabled"),
                Localization.Text("oil_feature_toggle"),
                GameActions.ToggleHotboxesAndOiling,
                GameActions.HotboxesAndOilingStatus() == Localization.Text("enabled"));
            GUILayout.Label(Localization.Text("wear_description"), labelStyle);

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("damage_tuning"), labelStyle);
            DrawMultiplierControl(
                ref impactDamagePercent,
                "impact_damage_multiplier",
                GameActions.SetImpactDamagePercent,
                GameActions.CurrentImpactDamagePercent);
            DrawMultiplierControl(
                ref overspeedDamagePercent,
                "overspeed_damage_multiplier",
                GameActions.SetOverspeedDamagePercent,
                GameActions.CurrentOverspeedDamagePercent);
            DrawMultiplierControl(
                ref wearMultiplierPercent,
                "wear_multiplier",
                GameActions.SetEquipmentWearPercent,
                GameActions.CurrentEquipmentWearPercent);
            DrawMultiplierControl(
                ref oilUsePercent,
                "oil_use_multiplier",
                GameActions.SetOilUsePercent,
                GameActions.CurrentOilUsePercent);
            DrawMultiplierControl(
                ref fuelConsumptionPercent,
                "fuel_consumption_multiplier",
                GameActions.SetFuelConsumptionPercent,
                GameActions.CurrentFuelConsumptionPercent);
            GUILayout.Label(Localization.Text("multiplier_description"), labelStyle);
        }

        private void DrawQolTab()
        {
            GUILayout.Label(Localization.Text("player_speed"), labelStyle);
            DrawSpeedButtons(
                new[] { 1f, 2f, 5f },
                multiplier => Run(() => GameActions.SetPlayerSpeedMultiplier(multiplier)));
            GUILayout.Label(Localization.Text("player_speed_description"), labelStyle);

            GUILayout.Space(7f);
            GUILayout.Label(Localization.Text("game_speed"), labelStyle);
            DrawSpeedButtons(
                new[] { 0.5f, 1f, 2f, 5f },
                multiplier => Run(() => GameActions.SetGameSpeed(multiplier)));
            GUILayout.Label(Localization.Text("game_speed_description"), labelStyle);

        }

        private void DrawSpeedButtons(float[] multipliers, Action<float> action)
        {
            GUILayout.BeginHorizontal();
            foreach (var multiplier in multipliers)
            {
                var label = multiplier.ToString("0.##", CultureInfo.InvariantCulture) + "x";
                if (ActionButton(label)) action(multiplier);
            }
            GUILayout.EndHorizontal();
        }

        private void DrawAmountActions(
            ref string amount,
            string primaryKey,
            Func<string> primaryAction,
            string secondaryKey,
            Func<string> secondaryAction)
        {
            GUILayout.BeginHorizontal();
            amount = GUILayout.TextField(amount, textFieldStyle, GUILayout.Width(140f), GUILayout.Height(23f));
            if (ActionButton(Localization.Text(primaryKey))) Run(primaryAction);
            if (ActionButton(Localization.Text(secondaryKey))) Run(secondaryAction);
            GUILayout.EndHorizontal();
        }

        private void DrawMultiplierControl(
            ref string value,
            string labelKey,
            Func<float, string> action,
            Func<string> currentValue)
        {
            GUILayout.BeginHorizontal();
            GUILayout.Label(Localization.Text(labelKey), labelStyle, GUILayout.Width(185f));
            value = GUILayout.TextField(value, textFieldStyle, GUILayout.Width(75f), GUILayout.Height(23f));
            var enteredValue = value;
            if (ActionButton(Localization.Text("apply"))) Run(() => action(ParseFloat(enteredValue)));
            GUILayout.EndHorizontal();
            GUILayout.Label(Localization.Format("current_multiplier", currentValue()), labelStyle);
        }

        private void DrawResizeGrip()
        {
            var gripRect = new Rect(
                window.width - ResizeGripSize,
                window.height - ResizeGripSize,
                ResizeGripSize,
                ResizeGripSize);
            GUI.Box(gripRect, string.Empty, resizeGripStyle);

            var currentEvent = Event.current;
            if (currentEvent.type == EventType.MouseDown
                && currentEvent.button == 0
                && gripRect.Contains(currentEvent.mousePosition))
            {
                resizingWindow = true;
                resizeStartMouse = currentEvent.mousePosition;
                resizeStartSize = new Vector2(window.width, window.height);
                currentEvent.Use();
            }
            else if (resizingWindow && currentEvent.type == EventType.MouseDrag && currentEvent.button == 0)
            {
                var delta = currentEvent.mousePosition - resizeStartMouse;
                window.width = Mathf.Max(MinWindowWidth, resizeStartSize.x + delta.x);
                window.height = Mathf.Max(MinWindowHeight, resizeStartSize.y + delta.y);
                currentEvent.Use();
            }
            else if (resizingWindow && currentEvent.type == EventType.MouseUp && currentEvent.button == 0)
            {
                resizingWindow = false;
                currentEvent.Use();
            }
        }

        private void DrawActionPair(
            string leftLabelKey,
            string leftButtonKey,
            Func<string> leftAction,
            string rightLabelKey,
            string rightButtonKey,
            Func<string> rightAction,
            string leftSuffix = null,
            string rightSuffix = null)
        {
            GUILayout.BeginHorizontal();
            DrawCompactAction(
                Localization.Text(leftLabelKey) + FormatSuffix(leftSuffix),
                Localization.Text(leftButtonKey),
                leftAction);
            DrawCompactAction(
                Localization.Text(rightLabelKey) + FormatSuffix(rightSuffix),
                Localization.Text(rightButtonKey),
                rightAction);
            GUILayout.EndHorizontal();
        }

        private void DrawStatePair(
            string leftLabel,
            Func<string> leftAction,
            bool leftEnabled,
            string rightLabel,
            Func<string> rightAction,
            bool rightEnabled)
        {
            GUILayout.BeginHorizontal();
            DrawCompactAction(leftLabel, leftEnabled ? Localization.Text("state_enabled") : Localization.Text("state_disabled"), leftAction);
            DrawCompactAction(rightLabel, rightEnabled ? Localization.Text("state_enabled") : Localization.Text("state_disabled"), rightAction);
            GUILayout.EndHorizontal();
        }

        private void DrawCompactAction(string label, string buttonText, Func<string> action)
        {
            GUILayout.BeginVertical(GUILayout.ExpandWidth(true));
            GUILayout.Label(label, labelStyle);
            if (ActionButton(buttonText, 23f)) Run(action);
            GUILayout.EndVertical();
        }

        private static string FormatSuffix(string suffix)
        {
            return string.IsNullOrWhiteSpace(suffix) ? string.Empty : " (" + suffix + ")";
        }

        private bool ActionButton(string text, float height = 25f)
        {
            return GUILayout.Button(text, buttonStyle, GUILayout.Height(height));
        }

        private void EnsureStyles()
        {
            if (windowStyle != null) return;

            // Match Railroader's stock visual language while keeping this
            // IMGUI window opaque even when the global skin is translucent.
            windowBackground = MakePanelTexture(
                new Color(0.105f, 0.095f, 0.080f, 1f),
                new Color(0.55f, 0.46f, 0.31f, 1f),
                2);
            controlBackground = MakePanelTexture(
                new Color(0.13f, 0.12f, 0.105f, 1f),
                new Color(0.46f, 0.38f, 0.25f, 1f),
                2);
            controlHoverBackground = MakePanelTexture(
                new Color(0.19f, 0.16f, 0.12f, 1f),
                new Color(0.64f, 0.53f, 0.34f, 1f),
                2);
            controlActiveBackground = MakePanelTexture(
                new Color(0.25f, 0.20f, 0.14f, 1f),
                new Color(0.75f, 0.63f, 0.40f, 1f),
                2);

            windowStyle = new GUIStyle(GUI.skin.window);
            windowStyle.normal.background = windowBackground;
            windowStyle.onNormal.background = windowBackground;
            windowStyle.border = new RectOffset(4, 4, 4, 4);
            windowStyle.normal.textColor = new Color(0.78f, 0.73f, 0.64f, 1f);
            windowStyle.onNormal.textColor = windowStyle.normal.textColor;
            windowStyle.fontSize = 13;
            windowStyle.padding = new RectOffset(10, 10, 28, 9);

            buttonStyle = new GUIStyle(GUI.skin.button);
            ApplyControlStyle(buttonStyle);

            tabStyle = new GUIStyle(GUI.skin.button);
            ApplyControlStyle(tabStyle);

            labelStyle = new GUIStyle(GUI.skin.label);
            labelStyle.normal.textColor = new Color(0.77f, 0.72f, 0.63f, 1f);
            labelStyle.wordWrap = true;
            labelStyle.fontSize = 12;

            textFieldStyle = new GUIStyle(GUI.skin.textField);
            textFieldStyle.normal.background = controlBackground;
            textFieldStyle.focused.background = controlActiveBackground;
            textFieldStyle.normal.textColor = new Color(0.86f, 0.82f, 0.74f, 1f);
            textFieldStyle.focused.textColor = textFieldStyle.normal.textColor;
            textFieldStyle.border = new RectOffset(4, 4, 4, 4);
            textFieldStyle.fontSize = 13;
            textFieldStyle.padding = new RectOffset(6, 6, 3, 3);

            closeButtonStyle = new GUIStyle(buttonStyle);
            closeButtonStyle.fontSize = 13;
            closeButtonStyle.padding = new RectOffset(0, 0, 0, 0);

        private static Texture2D MakePanelTexture(Color fill, Color border, int borderWidth)
        {
            const int size = 9;
            var texture = new Texture2D(size, size, TextureFormat.RGBA32, false);
            texture.wrapMode = TextureWrapMode.Clamp;
            for (var y = 0; y < size; y++)
            {
                for (var x = 0; x < size; x++)
                {
                    var isBorder = x < borderWidth || y < borderWidth
                        || x >= size - borderWidth || y >= size - borderWidth;
                    texture.SetPixel(x, y, isBorder ? border : fill);
                }
            }
            texture.Apply();
            texture.hideFlags = HideFlags.HideAndDontSave;
            return texture;

            }

            {
                var cause = ex.InnerException ?? ex;
                status = Localization.Format("error", cause.Message);
                log?.Error(ex, status);
            }
        }

        }

        {
            float parsed;
            return float.TryParse(value, out parsed) ? parsed : 0f;
        }
```

```cs
        private static string SetMowMaterials(float quantity, bool clearing)
        {
            RequireHost();
            if (quantity < 0f) quantity = 0f;
            var loads = MowLoadIds.Select(CarLoadLibrary.GetLoad).Where(load => load != null).ToArray();

            {
                if (industry == null || !LooksLikeMowYard(industry)) continue;
                var storage = industry.Storage;
                if (storage == null) continue;

                var prefixes = new HashSet<string>(StringComparer.Ordinal);
                foreach (var component in industry.Components ?? new List<IndustryComponent>())
                {

                if (prefixes.Count == 0) prefixes.Add(string.Empty);

                foreach (var prefix in prefixes)
                {
                    foreach (var load in loads)
                    {
                        storage.SetStorage(load, quantity, prefix);
                        writes++;
                    }
                }
                yards++;
            }

            return yards == 0
                ? Localization.Text("no_mow_yards")
                : Localization.Format(clearing ? "mow_cleared" : "restocked", writes, yards);
        }

        private static string SetFuelSupplies(float quantity, bool clearing)
        {
            RequireHost();

            var fuelLoads = FuelLoadIds.Select(CarLoadLibrary.GetLoad).Where(load => load != null).ToArray();
            var repairLoad = CarLoadLibrary.GetLoad(RepairPartsLoadId);

            var ops = OpsController.Shared;
            if (ops == null || ops.AllIndustries == null) return Localization.Text("operations_not_ready");

            var yards = 0;
            var writes = 0;
            foreach (var industry in ops.AllIndustries)
            {
                if (industry == null) continue;
                var isFuelYard = LooksLikeFuelYard(industry);
                var isRepairDepot = LooksLikeRepairDepot(industry);
                if (!isFuelYard && !isRepairDepot) continue;
                var storage = industry.Storage;
                if (storage == null) continue;

                var prefixes = new HashSet<string>(StringComparer.Ordinal);
                foreach (var component in industry.Components ?? new List<IndustryComponent>())
                {
                    var prefix = GetStoragePrefix(component);
                    if (prefix != null) prefixes.Add(prefix);
                }
                if (prefixes.Count == 0) prefixes.Add(string.Empty);

                foreach (var prefix in prefixes)
                {
                    if (isFuelYard)
                    {
                        foreach (var load in fuelLoads)
                        {
                            storage.SetStorage(load, quantity, prefix);
                            writes++;
                        }
                    }
                    if (isRepairDepot && repairLoad != null)
                    {
                        storage.SetStorage(repairLoad, quantity, prefix);
                        writes++;
                    }
                }
                yards++;
            }

            return yards == 0
                ? Localization.Text("no_fuel_yards")
                : Localization.Format(clearing ? "fuel_cleared" : "fuel_restocked", writes, yards);

        private static bool TryResolveFuelCar(Car selected, out Car fuelCar)
        {
            if (selected is DieselLocomotive)
            {
                fuelCar = selected;
                return true;
            }

            if (selected is SteamLocomotive steam)
            {
                fuelCar = InvokeSteamFuelCar(steam);
                return fuelCar != null;
            }

            foreach (var car in TrainController.Shared?.Cars ?? new List<Car>())
            {
                if (!(car is SteamLocomotive parentSteam)) continue;
                if (InvokeSteamFuelCar(parentSteam) != selected) continue;
                fuelCar = selected;
                return true;

            fuelCar = null;
            return false;

            if (car?.Definition?.LoadSlots == null) return 0;

            var filled = 0;
            for (var slotIndex = 0; slotIndex < car.Definition.LoadSlots.Count; slotIndex++)
            {
                var slot = car.Definition.LoadSlots[slotIndex];
                var loadId = slot.RequiredLoadIdentifier;
                if (string.IsNullOrWhiteSpace(loadId)) continue;

                var maximum = LoadSlotExtensions.CalculateMaximumCapacity(slot, loadId, car);
                var info = new CarLoadInfo(loadId, maximum, null);
                CarOpsExtensions.SetLoadInfo(car, slotIndex, info);
                filled++;
            }
            return filled;
        }

        private static int FillManifestedCar(Car car)
        {
            if (car?.Definition?.LoadSlots == null) return 0;

            var filled = 0;
            for (var slotIndex = 0; slotIndex < car.Definition.LoadSlots.Count; slotIndex++)
            {
                var info = CarOpsExtensions.GetLoadInfo(car, slotIndex);
                if (!info.HasValue || string.IsNullOrWhiteSpace(info.Value.LoadId)) continue;

                var load = CarLoadLibrary.GetLoad(info.Value.LoadId);
                if (load == null) continue;

                var slot = car.Definition.LoadSlots[slotIndex];
                if (!LoadSlotExtensions.LoadRequirementsMatch(slot, load)) continue;

                var maximum = LoadSlotExtensions.CalculateMaximumCapacity(slot, load, car);
                var updated = new CarLoadInfo(info.Value.LoadId, maximum, info.Value.OriginId);
                CarOpsExtensions.SetLoadInfo(car, slotIndex, updated);
                filled++;
            }
            return filled;
        }

        private static int EmptyCargoCar(Car car)
        {
            if (car?.Definition?.LoadSlots == null) return 0;

            var emptied = 0;
            for (var slotIndex = 0; slotIndex < car.Definition.LoadSlots.Count; slotIndex++)
            {
                var info = CarOpsExtensions.GetLoadInfo(car, slotIndex);
                if (!info.HasValue) continue;

                CarOpsExtensions.SetLoadInfo(car, slotIndex, new Nullable<CarLoadInfo>());
                emptied++;
            }
            return emptied;
        }

        private static bool LooksLikeMowYard(Industry industry)
```

```cs
using Railroader.Modding;
using UnityEngine;

namespace CAG.Sylvrwynd.RailroaderCheatMenu
{
    public sealed class ModEntryPoint : IModEntryPoint, IWorldLifecycle
    {
        private static GameObject menuObject;
        private IModLogger log;

        public void OnModLoad(IModContext context)
        {
            log = context.Log;
            log.Info("Railroader Cheat / Debug Menu loaded from " + context.ModDirectory);
            RuntimePatches.Install(log);

            if (menuObject == null)
            {
                menuObject = new GameObject("CAG.Sylvrwynd.RailroaderCheatMenu");
                Object.DontDestroyOnLoad(menuObject);
                menuObject.AddComponent<CheatMenuBehaviour>().Initialize(log);
            }
        }

        public void OnWorldLoad()
        {
            CheatMenuBehaviour.SetWorldLoaded(true);
            log?.Info("Railroader Cheat / Debug Menu world hooks ready.");
        }

        public void OnWorldUnload()
        {
            GameActions.ResetSessionOverrides();
            CheatMenuBehaviour.SetWorldLoaded(false);
            log?.Info("Railroader Cheat / Debug Menu world hooks cleared.");
        }

        public void OnModUnload()
        {
            GameActions.ResetSessionOverrides();
            CheatMenuBehaviour.SetWorldLoaded(false);
            RuntimePatches.Uninstall();
            if (menuObject != null)
            {
                Object.Destroy(menuObject);
                menuObject = null;
            }
            log?.Info("Railroader Cheat / Debug Menu unloaded.");
        }
    }
}
```

```cs
using System;
using System.Linq;
using System.Reflection;
using Model;
using Railroader.Modding;

namespace CAG.Sylvrwynd.RailroaderCheatMenu
{
    internal static class RuntimePatches
    {
        private const string HarmonyId = "cag-sylvrwynd.railroader-cheat-menu";

        private static object harmony;
        private static MethodInfo unpatchAll;

        internal static bool IsInstalled { get; private set; }
        internal static string FailureReason { get; private set; }
        internal static bool PreventDerailment { get; private set; }
        internal static bool ProtectHighSpeedDamage { get; private set; }
        internal static float ImpactDamageMultiplier { get; private set; } = 1f;
        internal static float OverspeedDamageMultiplier { get; private set; } = 1f;
        internal static float FuelConsumptionMultiplier { get; private set; } = 1f;
        internal const float DefaultHighSpeedDamageThresholdMph = 50f;
        internal static float HighSpeedDamageThresholdMph { get; private set; } = DefaultHighSpeedDamageThresholdMph;

        [ThreadStatic]
        private static int collisionDamageDepth;

        internal static void Install(IModLogger log)
        {
            try
            {
                var harmonyType = FindType("HarmonyLib.Harmony");
                var harmonyMethodType = FindType("HarmonyLib.HarmonyMethod");
                if (harmonyType == null || harmonyMethodType == null)
                {
                    throw new MissingMethodException("Harmony runtime types are unavailable.");
                }

                harmony = Activator.CreateInstance(harmonyType, new object[] { HarmonyId });
                var patchMethod = harmonyType.GetMethods(BindingFlags.Public | BindingFlags.Instance)
                    .Where(method => method.Name == "Patch")
                    .Where(method => method.GetParameters().Length == 5)
                    .FirstOrDefault();
                unpatchAll = harmonyType.GetMethods(BindingFlags.Public | BindingFlags.Instance)
                    .Where(method => method.Name == "UnpatchAll")
                    .Where(method => method.GetParameters().Length == 1)
                    .FirstOrDefault();
                if (patchMethod == null || unpatchAll == null)
                {
                    throw new MissingMethodException("Harmony patch methods are unavailable.");
                }

                Patch(patchMethod, harmonyMethodType, "ApplyDerailmentDelta", nameof(ApplyDerailmentDeltaPrefix));
                Patch(patchMethod, harmonyMethodType, "ApplyConditionDelta", nameof(ApplyConditionDeltaPrefix));
                Patch(patchMethod, harmonyMethodType, "SetCondition", nameof(SetConditionPrefix));
                Patch(
                    patchMethod,
                    harmonyMethodType,
                    typeof(TrainController),
                    "IntegrationSetCarsDidCollide",
                    nameof(IntegrationSetCarsDidCollidePrefix),
                    nameof(IntegrationSetCarsDidCollidePostfix));
                Patch(
                    patchMethod,
                    harmonyMethodType,
                    typeof(Model.Physics.TrainMath),
                    "DamageForSpeed",
                    null,
                    nameof(DamageForSpeedPostfix));
                Patch(
                    patchMethod,
                    harmonyMethodType,
                    typeof(Model.SteamLocomotive),
                    "PeriodicUpdateConsumeFuel",
                    nameof(ScaleSteamFuelConsumptionPrefix));
                Patch(
                    patchMethod,
                    harmonyMethodType,
                    typeof(Model.Physics.DieselPrimeMover),
                    "get_FuelConsumptionRate",
                    null,
                    nameof(ScaleDieselFuelConsumptionPostfix));
                IsInstalled = true;
                FailureReason = null;
                log?.Info("Runtime protection hooks installed.");
            }
            catch (Exception ex)
            {
                try
                {
                    unpatchAll?.Invoke(harmony, new object[] { HarmonyId });
                }
                catch
                {
                    // Keep the mod load failure from becoming a game load failure.
                }
                IsInstalled = false;
                FailureReason = ex.ToString();
                harmony = null;
                unpatchAll = null;
                log?.Error(ex, "Runtime protection hooks could not be installed.");
            }
        }

        internal static void Uninstall()
        {
            if (harmony == null) return;
            unpatchAll?.Invoke(harmony, new object[] { HarmonyId });
            harmony = null;
            unpatchAll = null;
            IsInstalled = false;
            PreventDerailment = false;
            ProtectHighSpeedDamage = false;
            ImpactDamageMultiplier = 1f;
            OverspeedDamageMultiplier = 1f;
            FuelConsumptionMultiplier = 1f;
            HighSpeedDamageThresholdMph = DefaultHighSpeedDamageThresholdMph;
            collisionDamageDepth = 0;
        }

        internal static void ToggleDerailmentProtection()
        {
            PreventDerailment = !PreventDerailment;
        }

        internal static void ToggleHighSpeedDamageProtection()
        {
            ProtectHighSpeedDamage = !ProtectHighSpeedDamage;
        }

        internal static void SetImpactDamageMultiplier(float multiplier)
        {
            ImpactDamageMultiplier = ClampMultiplier(multiplier);
        }

        internal static void SetOverspeedDamageMultiplier(float multiplier)
        {
            OverspeedDamageMultiplier = ClampMultiplier(multiplier);
        }

        internal static void SetFuelConsumptionMultiplier(float multiplier)
        {
            FuelConsumptionMultiplier = ClampMultiplier(multiplier);
        }

        internal static void SetHighSpeedDamageThreshold(float mph)
        {
            HighSpeedDamageThresholdMph = Math.Max(1f, Math.Min(200f, mph));
        }

        private static void Patch(
            MethodInfo patchMethod,
            Type harmonyMethodType,
            string targetName,
            string prefixName,
            string postfixName = null)
        {
            Patch(patchMethod, harmonyMethodType, typeof(Car), targetName, prefixName, postfixName);
        }

        private static void Patch(
            MethodInfo patchMethod,
            Type harmonyMethodType,
            Type targetType,
            string targetName,
            string prefixName,
            string postfixName = null)
        {
            var targetMethods = targetType.GetMethods(
                BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Instance | BindingFlags.Static);
            var target = targetMethods
                .Where(method => method.Name == targetName)
                .FirstOrDefault();
            var prefix = string.IsNullOrWhiteSpace(prefixName)
                ? null
                : typeof(RuntimePatches).GetMethod(
                    prefixName,
                    BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static);
            var postfix = string.IsNullOrWhiteSpace(postfixName)
                ? null
                : typeof(RuntimePatches).GetMethod(
                    postfixName,
                    BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static);
            if (target == null || (!string.IsNullOrWhiteSpace(prefixName) && prefix == null) || postfixName != null && postfix == null)
            {
                throw new MissingMethodException(targetName + " / " + prefixName);
            }

            var harmonyPrefix = prefix == null
                ? null
                : Activator.CreateInstance(harmonyMethodType, new object[] { prefix });
            var harmonyPostfix = postfix == null
                ? null
                : Activator.CreateInstance(harmonyMethodType, new object[] { postfix });
            patchMethod.Invoke(harmony, new object[] { target, harmonyPrefix, harmonyPostfix, null, null });
        }

        private static Type FindType(string fullName)
        {
            var qualified = Type.GetType(fullName + ", 0Harmony", false);
            if (qualified != null) return qualified;
            return AppDomain.CurrentDomain.GetAssemblies()
                .Select(assembly => assembly.GetType(fullName, false))
                .FirstOrDefault(type => type != null);
        }

        private static bool ApplyDerailmentDeltaPrefix(float delta)
        {
            return !PreventDerailment || delta <= 0f;
        }

        private static bool ApplyConditionDeltaPrefix(Car __instance, ref float delta)
        {
            if (delta < 0f && collisionDamageDepth > 0)
            {
                delta *= ImpactDamageMultiplier;
            }

            return !ProtectHighSpeedDamage || delta >= 0f || !IsAboveHighSpeed(__instance);
        }

        private static bool SetConditionPrefix(Car __instance, float condition)
        {
            return !ProtectHighSpeedDamage || condition >= __instance.Condition || !IsAboveHighSpeed(__instance);
        }

        private static bool IsAboveHighSpeed(Car car)
        {
            return car != null && car.VelocityMphAbs >= HighSpeedDamageThresholdMph;
        }

        private static void IntegrationSetCarsDidCollidePrefix()
        {
            if (ImpactDamageMultiplier != 1f)
            {
                collisionDamageDepth++;
            }
        }

        private static void IntegrationSetCarsDidCollidePostfix()
        {
            if (collisionDamageDepth > 0)
            {
                collisionDamageDepth--;
            }
        }

        private static void DamageForSpeedPostfix(ref float __result)
        {
            __result *= OverspeedDamageMultiplier;
        }

        private static void ScaleSteamFuelConsumptionPrefix(ref float dt)
        {
            dt *= FuelConsumptionMultiplier;
        }

        private static void ScaleDieselFuelConsumptionPostfix(ref float __result)
        {
            __result *= FuelConsumptionMultiplier;
        }

        private static float ClampMultiplier(float multiplier)
        {
            return Math.Max(0f, Math.Min(5f, multiplier));
        }
    }
}
```
