        private static readonly int[] MajorFunctionOffsets64 =
        {
            0x70,  // IRP_MJ_CREATE (0)
            0x78,  // IRP_MJ_CREATE_NAMED_PIPE (1)
            0x80,  // IRP_MJ_CLOSE (2)
            0x88,  // IRP_MJ_READ (3)
            0x90,  // IRP_MJ_WRITE (4)
            0xE0,  // IRP_MJ_DEVICE_CONTROL (14)
            0xE8,  // IRP_MJ_INTERNAL_DEVICE_CONTROL (15)
        };

        private static readonly int[] MajorFunctionOffsets32 =
        {
            0x38,  // IRP_MJ_CREATE (0)
            0x3C,  // IRP_MJ_CREATE_NAMED_PIPE (1)
            0x40,  // IRP_MJ_CLOSE (2)
            0x44,  // IRP_MJ_READ (3)
            0x48,  // IRP_MJ_WRITE (4)
            0x70,  // IRP_MJ_DEVICE_CONTROL (14)
            0x74,  // IRP_MJ_INTERNAL_DEVICE_CONTROL (15)

        private static readonly int[] MajorFunctionIndices = { 0, 1, 2, 3, 4, 14, 15 };

        public static DispatchAnalysisResult Analyze(byte[] data, PeParser.PeHeaders headers,
            Dictionary<uint, string> iatMap, HashSet<string> dangerousImportNames)
        {
            var result = new DispatchAnalysisResult();
            try
            {
                int entryOffset = PeParser.RvaToOffset(headers.EntryPointRva, headers.Sections);
                if (entryOffset < 0 || entryOffset >= data.Length) return result;

                if (headers.IsPe32Plus)
                    ExtractDispatchTable64(data, entryOffset, headers, result);
                else
                    ExtractDispatchTable32(data, entryOffset, headers, result);

                if (result.DeviceControlHandlerRva != 0)
                {
                    int handlerOff = PeParser.RvaToOffset(result.DeviceControlHandlerRva, headers.Sections);
                    if (handlerOff > 0 && handlerOff < data.Length)
                    {
                        result.ExtractedIoctlCodes = ExtractIoctlCodesFromHandler(data, handlerOff, headers);
                        result.BufferValidation = CheckBufferValidation(data, handlerOff, headers, iatMap);
                        result.HasPreviousModeCheck = CheckPreviousModeCheck(data, handlerOff, headers, iatMap);
                    }
                }

                if (result.CreateHandlerRva != 0)
                {
                    int createOff = PeParser.RvaToOffset(result.CreateHandlerRva, headers.Sections);
                    if (createOff > 0 && createOff < data.Length)
                        result.CreateAccessCheck = AnalyzeCreateHandler(data, createOff, headers, iatMap);
                }

                if (result.HandlerRvas.Count >= 2)
                {
                    var distinctHandlers = new HashSet<uint>(result.HandlerRvas.Values);
                    result.AllSameHandler = distinctHandlers.Count == 1;
                }
            }
            catch { }
            return result;
        }

        public static Dictionary<string, Reachability> AnalyzeReachability(byte[] data,
            PeParser.PeHeaders headers, uint handlerRva,
            Dictionary<uint, string> iatMap, HashSet<string> dangerousNames)
        {
            var reachability = new Dictionary<string, Reachability>();
            foreach (var name in dangerousNames)
                reachability[name] = Reachability.Unknown;

            try
            {
                var dangerousIat = new Dictionary<uint, string>();
                foreach (var kv in iatMap)
                    if (dangerousNames.Contains(kv.Value))
                        dangerousIat[kv.Key] = kv.Value;

                if (dangerousIat.Count == 0) return reachability;

                // Phase 1: Bottom-up xref — scan binary for FF 15/FF 25 referencing dangerous IAT entries
                var importSites = new Dictionary<string, List<int>>(StringComparer.OrdinalIgnoreCase);
                foreach (var name in dangerousNames)
                    importSites[name] = new List<int>();

                for (int i = 0; i < data.Length - 6; i++)
                {
                    if (data[i] != 0xFF || (data[i + 1] != 0x15 && data[i + 1] != 0x25)) continue;

                    uint iatRva;
                    if (headers.IsPe32Plus)
                    {
                        int disp = BitConverter.ToInt32(data, i + 2);
                        uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                        if (instrRva == 0) continue;
                        iatRva = (uint)(instrRva + 6 + disp);
                    }
                    else
                    {
                        uint absAddr = BitConverter.ToUInt32(data, i + 2);
                        if (absAddr <= headers.ImageBase) continue;
                        iatRva = (uint)(absAddr - headers.ImageBase);
                    }

                    if (dangerousIat.TryGetValue(iatRva, out string name))
                        importSites[name].Add(i);
                }

                // Phase 2: Build reachable function map from handler
                int handlerOff = PeParser.RvaToOffset(handlerRva, headers.Sections);
                if (handlerOff < 0) return reachability;

                var level0 = CollectE8CallTargets(data, handlerOff, 8192, headers);

                var level1 = new HashSet<uint>();
                foreach (uint rva in level0)
                {
                    int off = PeParser.RvaToOffset(rva, headers.Sections);
                    if (off < 0 || off >= data.Length) continue;
                    foreach (uint sub in CollectE8CallTargets(data, off, 4096, headers))
                        level1.Add(sub);
                }

                var level2 = new HashSet<uint>();
                foreach (uint rva in level1)
                {
                    int off = PeParser.RvaToOffset(rva, headers.Sections);
                    if (off < 0 || off >= data.Length) continue;
                    foreach (uint sub in CollectE8CallTargets(data, off, 2048, headers))
                        level2.Add(sub);
                }

                var funcLevel = new Dictionary<int, int>();
                funcLevel[handlerOff] = 0;
                foreach (uint rva in level0)
                {
                    int off = PeParser.RvaToOffset(rva, headers.Sections);
                    if (off >= 0 && !funcLevel.ContainsKey(off)) funcLevel[off] = 1;
                }
                foreach (uint rva in level1)
                {
                    int off = PeParser.RvaToOffset(rva, headers.Sections);
                    if (off >= 0 && !funcLevel.ContainsKey(off)) funcLevel[off] = 2;
                }
                foreach (uint rva in level2)
                {
                    int off = PeParser.RvaToOffset(rva, headers.Sections);
                    if (off >= 0 && !funcLevel.ContainsKey(off)) funcLevel[off] = 3;
                }

                var sortedFuncOffsets = funcLevel.Keys.OrderBy(k => k).ToList();

                // Phase 3: Classify each import by checking call site containment
                foreach (var name in dangerousNames)
                {
                    var sites = importSites[name];
                    if (sites.Count == 0)
                    {
                        reachability[name] = Reachability.NotReachable;
                        continue;
                    }

                    Reachability best = Reachability.NotReachable;
                    foreach (int siteOff in sites)
                    {
                        int idx = sortedFuncOffsets.BinarySearch(siteOff);
                        if (idx < 0) idx = ~idx - 1;
                        if (idx < 0) continue;

                        int funcOff = sortedFuncOffsets[idx];
                        if (siteOff - funcOff > 16384) continue;

                        int level = funcLevel[funcOff];
                        if (level == 0)
                        {
                            best = Reachability.Direct;
                            break;
                        }
                        if (best != Reachability.Direct)
                            best = Reachability.Indirect;
                    }
                    reachability[name] = best;
                }
            }
            catch { }
            return reachability;
        }

        private static HashSet<uint> CollectE8CallTargets(byte[] data, int startOffset,
            int maxScan, PeParser.PeHeaders headers)
        {
            var targets = new HashSet<uint>();
            int end = Math.Min(startOffset + maxScan, data.Length - 5);
            for (int i = startOffset; i < end; i++)
            {
                if (data[i] == 0xE8)
                {
                    int disp = BitConverter.ToInt32(data, i + 1);
                    uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                    if (instrRva > 0)
                    {
                        uint target = (uint)(instrRva + 5 + disp);
                        if (target > 0x1000 && target < 0x10000000)
                            targets.Add(target);
                    }
                    i += 4;
                }
                else if (data[i] == 0xE9)
                {
                    int disp = BitConverter.ToInt32(data, i + 1);
                    uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                    if (instrRva > 0)
                    {
                        uint target = (uint)(instrRva + 5 + disp);
                        if (target > 0x1000 && target < 0x10000000)
                            targets.Add(target);
                    }
                    break;
                }
            }
            return targets;
        }

        public static TaintStatus AnalyzeTaint(byte[] data, int callSiteFileOffset,
            PeParser.PeHeaders headers)
        {
            try
            {
                int searchStart = Math.Max(0, callSiteFileOffset - 48);
                int searchEnd = callSiteFileOffset;

                for (int i = searchEnd - 7; i >= searchStart; i--)
                {
                    if (i + 7 > data.Length) continue;

                    // MOV RCX, imm64 (48 B9 XX..): constant -> NOT_TAINTED
                    if (data[i] == 0x48 && data[i + 1] == 0xB9)
                        return TaintStatus.NotTainted;

                    // MOV ECX, imm32 (B9 XX XX XX XX): constant -> NOT_TAINTED
                    if (data[i] == 0xB9 && i + 5 <= data.Length)
                    {
                        uint val = BitConverter.ToUInt32(data, i + 1);
                        if (val > 0x10000) return TaintStatus.NotTainted;
                    }

                    // LEA RCX, [RIP+disp32] (48 8D 0D XX XX XX XX): -> NOT_TAINTED
                    if (data[i] == 0x48 && data[i + 1] == 0x8D && data[i + 2] == 0x0D)
                        return TaintStatus.NotTainted;

                    // MOV RCX, [reg+disp8/32] patterns where reg could be buffer pointer
                    // 48 8B 4B XX or 48 8B 4D XX = MOV RCX, [RBX+disp8] / [RBP+disp8]
                    if (data[i] == 0x48 && data[i + 1] == 0x8B)
                    {
                        byte modrm = data[i + 2];
                        int reg = (modrm >> 3) & 7;
                        int mod = (modrm >> 6) & 3;
                        if (reg == 1 && (mod == 1 || mod == 2)) // dst=RCX, indirect addressing
                            return TaintStatus.LikelyTainted;
                    }

                    // MOV RCX, [RSP+disp] = loading from stack (common for input buffer)
                    // 48 8B 8C 24 XX XX XX XX (with SIB byte for RSP base)
                    if (i + 8 <= data.Length && data[i] == 0x48 && data[i + 1] == 0x8B &&
                        data[i + 2] == 0x8C && data[i + 3] == 0x24)
                        return TaintStatus.LikelyTainted;
                }
            }
            catch { }
            return TaintStatus.Unknown;
        }

        private static void ExtractDispatchTable64(byte[] data, int entryOffset,
            PeParser.PeHeaders headers, DispatchAnalysisResult result)
        {
            ScanDispatchPatterns64(data, entryOffset, headers, result);

            if (result.HandlerRvas.Count == 0)
            {
                var callTargets = new List<int>();
                int callScanEnd = Math.Min(entryOffset + 8192, data.Length - 5);
                for (int i = entryOffset; i < callScanEnd && callTargets.Count < 4; i++)
                {
                    if (data[i] == 0xE8 && i + 5 <= data.Length)
                    {
                        int disp = BitConverter.ToInt32(data, i + 1);
                        uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                        if (instrRva > 0)
                        {
                            uint targetRva = (uint)(instrRva + 5 + disp);
                            int targetOff = PeParser.RvaToOffset(targetRva, headers.Sections);
                            if (targetOff > 0 && targetOff < data.Length)
                                callTargets.Add(targetOff);
                        }
                        i += 4;
                        continue;
                    }
                    if (data[i] == 0xC3) break;
                }

                foreach (int targetOff in callTargets)
                {
                    ScanDispatchPatterns64(data, targetOff, headers, result);
                    if (result.HandlerRvas.Count > 0) break;
                }
            }
        }

        private static void ScanDispatchPatterns64(byte[] data, int startOffset,
            PeParser.PeHeaders headers, DispatchAnalysisResult result)
        {
            var regLeaTarget = new Dictionary<int, uint>();
            int scanEnd = Math.Min(startOffset + 8192, data.Length - 7);

            for (int i = startOffset; i < scanEnd; i++)
            {
                if (i + 7 <= data.Length && (data[i] == 0x48 || data[i] == 0x4C) && data[i + 1] == 0x8D)
                {
                    byte modrm = data[i + 2];
                    int mod = (modrm >> 6) & 3;
                    int rm = modrm & 7;
                    int regField = (modrm >> 3) & 7;

                    if (mod == 0 && rm == 5)
                    {
                        int disp = BitConverter.ToInt32(data, i + 3);
                        uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                        if (instrRva > 0)
                        {
                            uint targetRva = (uint)(instrRva + 7 + disp);
                            int regIndex = regField + (data[i] == 0x4C ? 8 : 0);
                            regLeaTarget[regIndex] = targetRva;
                        }
                        i += 6;
                        continue;
                    }
                }

                if (i + 7 <= data.Length &&
                    (data[i] == 0x48 || data[i] == 0x49 || data[i] == 0x4C) &&
                    data[i + 1] == 0x89)
                {
                    byte modrm = data[i + 2];
                    int mod = (modrm >> 6) & 3;
                    int rm = modrm & 7;
                    int regField = (modrm >> 3) & 7;

                    if (mod == 2 && rm != 4)
                    {
                        int disp32 = BitConverter.ToInt32(data, i + 3);

                        for (int j = 0; j < MajorFunctionOffsets64.Length; j++)
                        {
                            if (disp32 == MajorFunctionOffsets64[j])
                            {
                                int srcReg = regField + (data[i] == 0x4C ? 8 : 0);
                                if (regLeaTarget.ContainsKey(srcReg))
                                {
                                    int mjIndex = MajorFunctionIndices[j];
                                    result.HandlerRvas[mjIndex] = regLeaTarget[srcReg];

                                    if (mjIndex == 14)
                                        result.DeviceControlHandlerRva = regLeaTarget[srcReg];
                                    else if (mjIndex == 0)
                                        result.CreateHandlerRva = regLeaTarget[srcReg];
                                }
                                break;
                            }
                        }
                        i += 6;
                        continue;
                    }
                }

                if (data[i] == 0xC3 && result.HandlerRvas.Count > 0)
                    break;
            }
        }

        private static void ScanDispatchPatterns32(byte[] data, int startOffset,
            PeParser.PeHeaders headers, DispatchAnalysisResult result)
        {
            var regValues = new Dictionary<int, uint>();
            ulong imageBase = headers.ImageBase;
            int scanEnd = Math.Min(startOffset + 8192, data.Length - 6);

            for (int i = startOffset; i < scanEnd; i++)
            {
                // MOV reg, imm32: B8+rd imm32
                if (data[i] >= 0xB8 && data[i] <= 0xBF && i + 5 <= data.Length)
                {
                    int reg = data[i] - 0xB8;
                    uint imm = BitConverter.ToUInt32(data, i + 1);
                    if (imm > imageBase && imm < imageBase + 0x10000000)
                        regValues[reg] = imm;
                    i += 4;
                    continue;
                }

                // LEA reg, [disp32]: 8D ModRM disp32 (mod=0, rm=5)
                if (data[i] == 0x8D && i + 6 <= data.Length)
                {
                    byte modrm = data[i + 1];
                    int mod = (modrm >> 6) & 3;
                    int rm = modrm & 7;
                    int reg = (modrm >> 3) & 7;
                    if (mod == 0 && rm == 5)
                    {
                        uint addr = BitConverter.ToUInt32(data, i + 2);
                        if (addr > imageBase && addr < imageBase + 0x10000000)
                            regValues[reg] = addr;
                        i += 5;
                        continue;
                    }
                }

                // MOV [reg+disp8], imm32: C7 ModRM disp8 imm32
                // MOV [reg+disp32], imm32: C7 ModRM disp32 imm32
                if (data[i] == 0xC7)
                {
                    byte modrm = data[i + 1];
                    int mod = (modrm >> 6) & 3;
                    int rm = modrm & 7;
                    int regOp = (modrm >> 3) & 7;

                    if (regOp == 0 && mod == 1 && rm != 4 && i + 7 <= data.Length)
                    {
                        int disp = (sbyte)data[i + 2];
                        uint imm = BitConverter.ToUInt32(data, i + 3);

                        for (int j = 0; j < MajorFunctionOffsets32.Length; j++)
                        {
                            if (disp == MajorFunctionOffsets32[j] && imm > imageBase)
                            {
                                uint handlerRva = (uint)(imm - imageBase);
                                if (handlerRva < 0x10000000)
                                {
                                    int mjIndex = MajorFunctionIndices[j];
                                    result.HandlerRvas[mjIndex] = handlerRva;
                                    if (mjIndex == 14) result.DeviceControlHandlerRva = handlerRva;
                                    else if (mjIndex == 0) result.CreateHandlerRva = handlerRva;
                                }
                                break;
                            }
                        }
                        i += 6;
                        continue;
                    }

                    if (regOp == 0 && mod == 2 && rm != 4 && i + 10 <= data.Length)
                    {
                        int disp = BitConverter.ToInt32(data, i + 2);
                        uint imm = BitConverter.ToUInt32(data, i + 6);

                        for (int j = 0; j < MajorFunctionOffsets32.Length; j++)
                        {
                            if (disp == MajorFunctionOffsets32[j] && imm > imageBase)
                            {
                                uint handlerRva = (uint)(imm - imageBase);
                                if (handlerRva < 0x10000000)
                                {
                                    int mjIndex = MajorFunctionIndices[j];
                                    result.HandlerRvas[mjIndex] = handlerRva;
                                    if (mjIndex == 14) result.DeviceControlHandlerRva = handlerRva;
                                    else if (mjIndex == 0) result.CreateHandlerRva = handlerRva;
                                }
                                break;
                            }
                        }
                        i += 9;
                        continue;
                    }
                }

                // MOV [reg+disp], src_reg: 89 ModRM disp8/disp32
                if (data[i] == 0x89 && i + 3 <= data.Length)
                {
                    byte modrm = data[i + 1];
                    int mod = (modrm >> 6) & 3;
                    int rm = modrm & 7;
                    int srcReg = (modrm >> 3) & 7;

                    int disp = 0;
                    int instrLen = 0;

                    if (mod == 1 && rm != 4)
                    {
                        disp = (sbyte)data[i + 2];
                        instrLen = 3;
                    }
                    else if (mod == 2 && rm != 4 && i + 6 <= data.Length)
                    {
                        disp = BitConverter.ToInt32(data, i + 2);
                        instrLen = 6;
                    }

                    if (instrLen > 0 && regValues.ContainsKey(srcReg))
                    {
                        for (int j = 0; j < MajorFunctionOffsets32.Length; j++)
                        {
                            if (disp == MajorFunctionOffsets32[j])
                            {
                                uint va = regValues[srcReg];
                                uint handlerRva = (uint)(va - imageBase);
                                if (handlerRva > 0 && handlerRva < 0x10000000)
                                {
                                    int mjIndex = MajorFunctionIndices[j];
                                    result.HandlerRvas[mjIndex] = handlerRva;
                                    if (mjIndex == 14) result.DeviceControlHandlerRva = handlerRva;
                                    else if (mjIndex == 0) result.CreateHandlerRva = handlerRva;
                                }
                                break;
                            }
                        }
                        i += instrLen - 1;
                        continue;
                    }
                }

                if (data[i] == 0xC3 && result.HandlerRvas.Count > 0)
                    break;
            }
        }

        private static List<uint> ExtractIoctlCodesFromHandler(byte[] data, int handlerOffset,
            PeParser.PeHeaders headers)
        {
            var codes = new HashSet<uint>();
            int scanEnd = Math.Min(handlerOffset + 4096, data.Length - 6);

            for (int i = handlerOffset; i < scanEnd; i++)
            {
                uint imm32 = 0;
                bool found = false;

                // CMP EAX, imm32: 3D XX XX XX XX
                if (data[i] == 0x3D && i + 5 <= data.Length)
                {
                    imm32 = BitConverter.ToUInt32(data, i + 1);
                    found = true;
                }
                // CMP r32, imm32: 81 F8-FF XX XX XX XX
                else if (data[i] == 0x81 && i + 6 <= data.Length)
                {
                    byte modrm = data[i + 1];
                    int regOp = (modrm >> 3) & 7;
                    int mod = (modrm >> 6) & 3;
                    if (mod == 3 && regOp == 7) // CMP r32, imm32
                    {
                        imm32 = BitConverter.ToUInt32(data, i + 2);
                        found = true;
                    }
                }
                // SUB EAX, imm32: 2D XX XX XX XX
                else if (data[i] == 0x2D && i + 5 <= data.Length)
                {
                    imm32 = BitConverter.ToUInt32(data, i + 1);
                    found = true;
                }
                // With REX.B prefix: 41 81 F8-FF XX XX XX XX
                else if (data[i] == 0x41 && i + 7 <= data.Length && data[i + 1] == 0x81)
                {
                    byte modrm = data[i + 2];
                    int regOp = (modrm >> 3) & 7;
                    int mod = (modrm >> 6) & 3;
                    if (mod == 3 && regOp == 7)
                    {
                        imm32 = BitConverter.ToUInt32(data, i + 3);
                        found = true;
                    }
                }

                if (found && IsValidIoctlCode(imm32))
                    codes.Add(imm32);

                if (codes.Count >= 30) break;
            }
            return codes.OrderBy(c => c).ToList();
        }

        private static List<CallTarget> FindCallTargets(byte[] data, int funcOffset, int maxScan,
            PeParser.PeHeaders headers)
        {
            var targets = new List<CallTarget>();
            int scanEnd = Math.Min(funcOffset + maxScan, data.Length - 6);
            int retCount = 0;

            for (int i = funcOffset; i < scanEnd; i++)
            {
                // CALL rel32: E8 XX XX XX XX
                if (data[i] == 0xE8 && i + 5 <= data.Length)
                {
                    int disp = BitConverter.ToInt32(data, i + 1);
                    uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                    if (instrRva > 0)
                    {
                        uint targetRva = (uint)(instrRva + 5 + disp);
                        if (targetRva > 0x1000 && targetRva < 0x10000000)
                        {
                            targets.Add(new CallTarget
                            {
                                TargetRva = targetRva,
                                IsIatCall = false,
                                FileOffset = PeParser.RvaToOffset(targetRva, headers.Sections)
                            });
                        }
                    }
                    i += 4;
                    continue;
                }

                // CALL [RIP+disp32]: FF 15 XX XX XX XX
                if (data[i] == 0xFF && i + 6 <= data.Length && data[i + 1] == 0x15)
                {
                    int disp = BitConverter.ToInt32(data, i + 2);
                    uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                    if (instrRva > 0)
                    {
                        uint iatRva = (uint)(instrRva + 6 + disp);
                        targets.Add(new CallTarget
                        {
                            TargetRva = iatRva,
                            IsIatCall = true,
                            FileOffset = -1
                        });
                    }
                    i += 5;
                    continue;
                }

                // JMP [RIP+disp32]: FF 25 XX XX XX XX (tail call through IAT)
                if (data[i] == 0xFF && i + 6 <= data.Length && data[i + 1] == 0x25)
                {
                    int disp = BitConverter.ToInt32(data, i + 2);
                    uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                    if (instrRva > 0)
                    {
                        uint iatRva = (uint)(instrRva + 6 + disp);
                        targets.Add(new CallTarget
                        {
                            TargetRva = iatRva,
                            IsIatCall = true,
                            FileOffset = -1
                        });
                    }
                    i += 5;
                    continue;
                }

                // JMP rel32: E9 XX XX XX XX (tail call to local func)
                if (data[i] == 0xE9 && i + 5 <= data.Length)
                {
                    int disp = BitConverter.ToInt32(data, i + 1);
                    uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                    if (instrRva > 0)
                    {
                        uint targetRva = (uint)(instrRva + 5 + disp);
                        if (targetRva > 0x1000 && targetRva < 0x10000000)
                        {
                            targets.Add(new CallTarget
                            {
                                TargetRva = targetRva,
                                IsIatCall = false,
                                FileOffset = PeParser.RvaToOffset(targetRva, headers.Sections)
                            });
                        }
                    }
                    break; // JMP is typically end of function/block
                }

                // RET: C3 - count returns, stop after second one
                if (data[i] == 0xC3)
                {
                    retCount++;
                    if (retCount >= 3) break;
                }
            }
            return targets;
        }

        private static AccessCheckVerdict AnalyzeCreateHandler(byte[] data, int handlerOffset,
            PeParser.PeHeaders headers, Dictionary<uint, string> iatMap)
        {
            var calls = FindCallTargets(data, handlerOffset, 1024, headers);

            string[] accessCheckFuncs =
            {
                "SeAccessCheck", "SeSinglePrivilegeCheck", "IoCheckShareAccess",
                "SeLockSubjectContext", "SePrivilegeCheck", "ObReferenceObjectByHandle",
                "ExGetPreviousMode", "KeGetPreviousMode", "PsGetCurrentProcessId",
                "IoGetCurrentProcess"
            };

            foreach (var call in calls)
            {
                if (call.IsIatCall && iatMap.ContainsKey(call.TargetRva))
                {
                    string name = iatMap[call.TargetRva];
                    if (accessCheckFuncs.Any(f => f.Equals(name, StringComparison.OrdinalIgnoreCase)))
                        return AccessCheckVerdict.HasChecks;
                }
            }

            // Check if handler is trivially short (just STATUS_SUCCESS + IoCompleteRequest + return)
            int bytesToRet = 0;
            for (int i = handlerOffset; i < Math.Min(handlerOffset + 128, data.Length); i++)
            {
                if (data[i] == 0xC3) { bytesToRet = i - handlerOffset; break; }
            }
            if (bytesToRet > 0 && bytesToRet < 64 && calls.Count <= 2)
                return AccessCheckVerdict.NoChecks;

            return AccessCheckVerdict.Unknown;
        }

        private static BufferValidation CheckBufferValidation(byte[] data, int handlerOffset,
            PeParser.PeHeaders headers, Dictionary<uint, string> iatMap)
        {
            var calls = FindCallTargets(data, handlerOffset, 4096, headers);

            bool hasProbe = false;
            bool hasLenCheck = false;

            foreach (var call in calls)
            {
                if (!call.IsIatCall || !iatMap.ContainsKey(call.TargetRva)) continue;
                string name = iatMap[call.TargetRva];

                if (name == "ProbeForRead" || name == "ProbeForWrite")
                    hasProbe = true;
                if (name == "MmIsAddressValid")
                    hasLenCheck = true;
            }

            // Look for __try/__except pattern (common structured exception handling in x64)
            // .pdata section contains exception handler info - presence of __C_specific_handler import
            if (iatMap.Values.Any(v => v == "__C_specific_handler"))
                hasLenCheck = true;

            if (hasProbe) return BufferValidation.ProperValidation;
            if (hasLenCheck) return BufferValidation.BasicValidation;

            // Check for length comparison patterns (CMP reg, [reg+offset] where offset = InputBufferLength)
            int scanEnd = Math.Min(handlerOffset + 4096, data.Length - 4);
            for (int i = handlerOffset; i < scanEnd; i++)
            {
                // CMP instructions with small immediates could be length checks
                // But this is too imprecise - mark as unknown unless we found probes
            }

            return BufferValidation.NoValidation;
        }

        private static bool CheckPreviousModeCheck(byte[] data, int handlerOffset,
            PeParser.PeHeaders headers, Dictionary<uint, string> iatMap)
        {
            var calls = FindCallTargets(data, handlerOffset, 4096, headers);
            foreach (var call in calls)
            {
                if (!call.IsIatCall || !iatMap.ContainsKey(call.TargetRva)) continue;
                string name = iatMap[call.TargetRva];
                if (name == "ExGetPreviousMode" || name == "KeGetPreviousMode")
                    return true;
            }
            return false;
        }

        private static bool IsValidIoctlCode(uint code)
        {
            uint deviceType = (code >> 16) & 0xFFFF;
            uint function = (code >> 2) & 0xFFF;
            if (function < 1 || function > 0xFFF) return false;
            if (deviceType == 0x22) return true; // FILE_DEVICE_UNKNOWN (most common)
            if (deviceType >= 0x8000 && deviceType <= 0x80FF) return true; // Custom device types
            if (deviceType == 0x09 || deviceType == 0x0C || deviceType == 0x12 ||
                deviceType == 0x21 || deviceType == 0x34) return true; // Other known types
            return false;
        }

    }

using System;

using System.Web.Script.Serialization;

namespace BYOVDHunter
{
    public class LolDriversDb
    {
        private const string DriverJsonUrl = "https://www.loldrivers.io/api/drivers.json";

        private readonly Dictionary<string, LolMatch> _hashLookup =
            new Dictionary<string, LolMatch>(StringComparer.OrdinalIgnoreCase);

        public int Count => _hashLookup.Count;

        public static string GetCachePath()
        {
            string exeDir = AppDomain.CurrentDomain.BaseDirectory;
            return System.IO.Path.Combine(exeDir, "loldrivers_cache.json");
        }

        public bool Download(string cachePath)
        {
            try
            {
                ServicePointManager.SecurityProtocol = SecurityProtocolType.Tls12 | SecurityProtocolType.Tls11;
                using (var client = new WebClient())
                {
                    client.Headers.Add("User-Agent", "BYOVDHunter/1.0");
                    client.DownloadFile(DriverJsonUrl, cachePath);
                }
                return true;
            }
            catch (Exception ex)
            {
                Console.Error.WriteLine($"  Failed to download LOLDrivers DB: {ex.Message}");
                return false;
            }
        }

        public bool Load(string cachePath)
        {
            try
            {
                string json = File.ReadAllText(cachePath);
                var serializer = new JavaScriptSerializer { MaxJsonLength = int.MaxValue };
                var drivers = serializer.Deserialize<object[]>(json);

                if (drivers == null) return false;

                foreach (object driverObj in drivers)
                {
                    if (!(driverObj is Dictionary<string, object> driver))
                        continue;

                    string id = GetString(driver, "Id");
                    string category = GetString(driver, "Category");

                    var cves = new List<string>();
                    var tags = new List<string>();

                    if (driver.ContainsKey("Tags"))
                    {
                        foreach (object tag in AsEnumerable(driver["Tags"]))
                        {
                            string t = tag?.ToString() ?? "";
                            if (string.IsNullOrWhiteSpace(t)) continue;
                            if (t.StartsWith("CVE-", StringComparison.OrdinalIgnoreCase))
                                cves.Add(t);
                            else
                                tags.Add(t);
                        }
                    }

                    if (!driver.ContainsKey("KnownVulnerableSamples")) continue;

                    foreach (object sampleObj in AsEnumerable(driver["KnownVulnerableSamples"]))
                    {
                        if (!(sampleObj is Dictionary<string, object> sample))
                            continue;

                        string sha256 = GetString(sample, "SHA256");
                        if (string.IsNullOrWhiteSpace(sha256)) continue;

                        string filename = GetString(sample, "Filename");

                        var match = new LolMatch
                        {
                            Id = id,
                            Category = category,
                            CVEs = new List<string>(cves),
                            Tags = new List<string>(tags),
                            KnownFileName = filename
                        };

                        _hashLookup[sha256] = match;
                    }
                }

                return true;
            }
            catch (Exception ex)
            {
                Console.Error.WriteLine($"  Failed to parse LOLDrivers DB: {ex.Message}");
                return false;
            }
        }

        public LolMatch Lookup(string sha256Hash)
        {
            if (string.IsNullOrWhiteSpace(sha256Hash)) return null;
            _hashLookup.TryGetValue(sha256Hash, out var match);
            return match;
        }

        private static string GetString(Dictionary<string, object> dict, string key)
        {
            if (dict.ContainsKey(key) && dict[key] != null)
                return dict[key].ToString();
            return "";
        }

        private static IEnumerable<object> AsEnumerable(object obj)
        {
            if (obj is ArrayList al) return al.Cast<object>();
            if (obj is object[] arr) return arr;
            return Enumerable.Empty<object>();
        }
    }

                ? $"Yes ({Trunc(dr.SignerName, 36)}){(dr.IsSignatureValid ? "" : " [INVALID]")}"

                PrintBoxLine($"Cert warning    : {dr.CertFlag.Reason}", ConsoleColor.Red);

                string age = dr.Metadata.AgeYears > 0 ? $" ({dr.Metadata.AgeYears}y old)" : "";

            }

            if (dr.Metadata != null && !string.IsNullOrEmpty(dr.Metadata.PdbPath))
                PrintBoxLine($"PDB path        : {Trunc(dr.Metadata.PdbPath, 48)}", ConsoleColor.DarkGray);

            PrintBoxLine($"Driver model    : {dr.Framework}{(dr.Framework == DriverFramework.WDM ? " (no framework protections)" : "")}", ConsoleColor.Gray);

            if (dr.Mitigations != null && dr.Mitigations.MissingCount > 0)

                    PrintBoxLine($"  IRP_MJ_CREATE          : {dr.Dispatch.CreateAccessCheck}", dr.Dispatch.CreateAccessCheck == AccessCheckVerdict.NoChecks ? ConsoleColor.Red : ConsoleColor.Green);
                if (dr.Dispatch.DeviceControlHandlerRva != 0)

                else
                {

                    foreach (var imp in dr.DangerousImports.Take(8))

                }
            }

            if (dr.ExportedDangerousFunctions.Count > 0)
            {

                foreach (string exp in dr.ExportedDangerousFunctions.Take(5))

            }

            if (dr.SuspiciousStrings.Count > 0)

                PrintBoxLine("Suspicious strings:", ConsoleColor.Magenta);
                foreach (string s in dr.SuspiciousStrings.Take(5))
                    PrintBoxLine($"  - \"{s}\"", ConsoleColor.Magenta);
            }

            {
                var rwx = dr.Sections.Where(s => s.IsRWX).Select(s => s.Name);

            }
            if (dr.Sections.Any(s => s.IsHighEntropy))
            {

                PrintBoxLine($"High entropy    : {string.Join(", ", he)} (possibly packed)", ConsoleColor.Yellow);
            }

            if (dr.Behaviors != BehaviorTag.None)
            {
                var behaviorNames = new List<string>();
                if (dr.Behaviors.HasFlag(BehaviorTag.B1_ProtectedServiceTermination)) behaviorNames.Add("B1:SvcTermination");
                if (dr.Behaviors.HasFlag(BehaviorTag.B2_PrivilegedHandleLeak))       behaviorNames.Add("B2:HandleLeak");
                if (dr.Behaviors.HasFlag(BehaviorTag.B3_UnsafePoolAllocation))       behaviorNames.Add("B3:PoolAlloc");
                if (dr.Behaviors.HasFlag(BehaviorTag.B4_DynamicKernelCodeExec))      behaviorNames.Add("B4:KernelExec");
                if (dr.Behaviors.HasFlag(BehaviorTag.B5_MismatchedMemoryMapping))    behaviorNames.Add("B5:MemMap");
                if (dr.Behaviors.HasFlag(BehaviorTag.B6_CodeIntegrityTampering))     behaviorNames.Add("B6:CITamper");
                if (dr.Behaviors.HasFlag(BehaviorTag.B7_RemoteHandleClosure))        behaviorNames.Add("B7:HandleClose");
                if (dr.Behaviors.HasFlag(BehaviorTag.B8_ArbitraryMemoryWrite))       behaviorNames.Add("B8:MemWrite");
                if (dr.Behaviors.HasFlag(BehaviorTag.B9_ArbitraryMsrWrite))          behaviorNames.Add("B9:MsrWrite");
                if (dr.Behaviors.HasFlag(BehaviorTag.B10_ArbitraryPortIO))           behaviorNames.Add("B10:PortIO");
                if (dr.Behaviors.HasFlag(BehaviorTag.B11_PhysicalMemoryMapping))     behaviorNames.Add("B11:PhysMemMap");
                if (dr.Behaviors.HasFlag(BehaviorTag.B12_TaintedKernelArgs))         behaviorNames.Add("B12:TaintedArgs");
                if (dr.Behaviors.HasFlag(BehaviorTag.B13_ExecWritableRemap))         behaviorNames.Add("B13:ExecRemap");
                if (dr.Behaviors.HasFlag(BehaviorTag.B14_UnsafeRtlQueryRegistry))    behaviorNames.Add("B14:UnsafeRegQuery");
                if (dr.Behaviors.HasFlag(BehaviorTag.B15_RtlQueryRegistryTampering)) behaviorNames.Add("B15:RegTamper");

            }

                PrintBoxLine($"Risk            : {dr.RiskReason}", accent);
            if (!string.IsNullOrEmpty(dr.Impact))
                PrintBoxLine($"Impact          : {dr.Impact}", accent);

            if (!string.IsNullOrEmpty(dr.Note))
                PrintBoxLine($"Note            : {dr.Note}", ConsoleColor.White);

            Console.WriteLine(bot);
            Console.ForegroundColor = prev;

        private static void PrintMediumBlock(DriverResult dr)
        {
            string top = "┌" + new string('─', BoxWidth) + "┐";
            string bot = "└" + new string('─', BoxWidth) + "┘";

            var prev = Console.ForegroundColor;
            Console.ForegroundColor = ConsoleColor.Yellow;
            Console.WriteLine();
            Console.WriteLine(top);

            string title = $"[MEDIUM] {dr.FileName}";
            if (dr.DangerousImports.All(i => i.Reachability == Reachability.NotReachable) && dr.DangerousImports.Count > 0)
                title += " (false positive reduced)";
            PrintBoxLineThin(title, ConsoleColor.Yellow);

            if (!string.IsNullOrEmpty(dr.RiskReason))
                PrintBoxLineThin(dr.RiskReason, ConsoleColor.DarkGray);

            var details = new List<string>();
            details.Add($"Driver: {dr.Framework}");
            details.Add($"Signed: {(dr.IsSigned ? "Yes (" + Trunc(dr.SignerName, 20) + ")" : "No")}");
            details.Add($"Loaded: {(dr.IsLoaded ? "Yes" : "No")}");
            if (dr.Metadata != null && dr.Metadata.AgeYears > 0)
                details.Add($"Age: {dr.Metadata.AgeYears}y");
            PrintBoxLineThin(string.Join(" | ", details), ConsoleColor.DarkGray);

            if (dr.DangerousImports.Count > 0)
            {
                var impNames = dr.DangerousImports.Take(4).Select(i =>
                {
                    string r = i.Reachability != Reachability.Unknown ? $"[{i.Reachability}]" : "";
                    return i.FunctionName + r;
                });
                PrintBoxLineThin("Imports: " + string.Join(", ", impNames), ConsoleColor.DarkYellow);
            }

            if (dr.Behaviors != BehaviorTag.None)
            {
                var bNames = FormatBehaviorTags(dr.Behaviors);
                if (bNames.Length > 0)
                    PrintBoxLineThin("Behaviors: " + bNames, ConsoleColor.Magenta);
            }

            if (!string.IsNullOrEmpty(dr.Note))
                PrintBoxLineThin(dr.Note, ConsoleColor.DarkGray);

            Console.ForegroundColor = ConsoleColor.Yellow;
            Console.WriteLine(bot);
            Console.ForegroundColor = prev;
        }

        public static void PrintExploitableRecap(List<DriverResult> loaded)
        {
            if (loaded.Count == 0) return;
            var prev = Console.ForegroundColor;
            Console.WriteLine();
            Console.ForegroundColor = ConsoleColor.Red;
            string bar = new string('═', 66);
            Console.WriteLine($"  {bar}");
            Console.WriteLine("   LOADED EXPLOITABLE DRIVERS - IMMEDIATE RISK");
            Console.WriteLine($"  {bar}");
            Console.ForegroundColor = prev;
            Console.WriteLine();

            foreach (var dr in loaded)
            {
                Console.ForegroundColor = dr.Risk == RiskLevel.Critical ? ConsoleColor.Red : ConsoleColor.DarkRed;
                Console.Write($"  ● {dr.FileName}");

                if (dr.LolMatch != null) { Console.ForegroundColor = ConsoleColor.Red; Console.Write("  [LOLDrivers]"); }
                if (dr.BlocklistMatch != null) { Console.ForegroundColor = ConsoleColor.Red; Console.Write("  [MS-Blocklist]"); }
                if (!dr.IsSigned) { Console.ForegroundColor = ConsoleColor.Red; Console.Write("  [UNSIGNED]"); }
                Console.WriteLine();

                Console.ForegroundColor = ConsoleColor.DarkGray;
                Console.Write("    Signer  : ");
                Console.ForegroundColor = ConsoleColor.White;
                Console.WriteLine(dr.IsSigned ? dr.SignerName : "NONE");

                if (dr.LolMatch != null && dr.LolMatch.CVEs.Count > 0)
                {
                    Console.ForegroundColor = ConsoleColor.DarkGray;
                    Console.Write("    CVEs    : ");
                    Console.ForegroundColor = ConsoleColor.Yellow;
                    Console.WriteLine(string.Join(", ", dr.LolMatch.CVEs));
                }

                if (dr.Devices.Count > 0)
                {
                    Console.ForegroundColor = ConsoleColor.DarkGray;
                    Console.Write("    Device  : ");
                    Console.ForegroundColor = ConsoleColor.Cyan;
                    Console.WriteLine(dr.Devices[0].DeviceName);
                }

                var reachableImps = dr.DangerousImports
                    .Where(i => i.Reachability == Reachability.Direct || i.Reachability == Reachability.Indirect)
                    .ToList();
                if (reachableImps.Count > 0)
                {
                    Console.ForegroundColor = ConsoleColor.DarkGray;
                    Console.Write("    Reach   : ");
                    Console.ForegroundColor = ConsoleColor.Red;
                    Console.WriteLine(string.Join(", ", reachableImps.Take(4).Select(i => $"{i.FunctionName}({i.Reachability})")));
                }

                var caps = new List<string>();
                bool hasMem = dr.DangerousImports.Any(i => i.Category == "PhysMem");
                bool hasProc = dr.DangerousImports.Any(i => i.Category == "ProcKill" || i.Category == "ProcManip");
                bool hasCb = dr.DangerousImports.Any(i => i.Category == "CallbackUnreg" || i.Category == "CallbackReg");
                if (hasMem) caps.Add("Kernel R/W");
                if (hasProc) caps.Add("Process kill");
                if (hasCb) caps.Add("Callback removal");
                if (caps.Count > 0)
                {
                    Console.ForegroundColor = ConsoleColor.DarkGray;
                    Console.Write("    Exploit : ");
                    Console.ForegroundColor = ConsoleColor.Red;
                    Console.WriteLine(string.Join(" + ", caps));
                }

                var abuses = new List<string>();
                if (hasMem) { abuses.Add("Privilege Escalation"); abuses.Add("EDR bypass"); abuses.Add("Credential theft"); }
                if (hasProc) abuses.Add("EDR kill");
                if (hasCb) abuses.Add("EDR blinding");
                if (hasMem || dr.LolMatch != null) abuses.Add("Persistence");
                if (abuses.Count > 0)
                {
                    Console.ForegroundColor = ConsoleColor.DarkGray;
                    Console.Write("    Abuse   : ");
                    Console.ForegroundColor = ConsoleColor.Magenta;
                    Console.WriteLine(string.Join(", ", abuses));
                }
                Console.WriteLine();
            }
            Console.ForegroundColor = prev;
        }

                            : kv.Key.StartsWith("B1") || kv.Key.StartsWith("B8") || kv.Key.StartsWith("B7")

                }
            }

            Console.ForegroundColor = ConsoleColor.White;
            Console.WriteLine($"  {bar}");
            Console.ForegroundColor = prev;
        }

        public static void ExportJson(List<DriverResult> results, ScanSummary summary, string outPath)
        {
            var export = new Dictionary<string, object>
            {
                { "scanDate", DateTime.UtcNow.ToString("o") },
                { "summary", SummaryToDict(summary) },
                { "findings", results
                    .Where(r => r.Risk >= RiskLevel.Low || r.LolMatch != null || r.BlocklistMatch != null)
                    .Select(DriverToDict).ToList() }
            };
            var serializer = new JavaScriptSerializer { MaxJsonLength = int.MaxValue };
            string json = serializer.Serialize(export);
            File.WriteAllText(outPath, FormatJson(json), Encoding.UTF8);
        }

        {

        private static string FormatJson(string json)
        {
            var sb = new StringBuilder();
            int indent = 0;
            bool inString = false;
            bool escaped = false;
            foreach (char c in json)
            {
                if (escaped) { sb.Append(c); escaped = false; continue; }
                if (c == '\\' && inString) { sb.Append(c); escaped = true; continue; }
                if (c == '"') { inString = !inString; sb.Append(c); continue; }
                if (inString) { sb.Append(c); continue; }
                switch (c)
                {
                    case '{': case '[':
                        sb.Append(c); sb.AppendLine(); indent++; sb.Append(new string(' ', indent * 2)); break;
                    case '}': case ']':
                        sb.AppendLine(); indent--; sb.Append(new string(' ', indent * 2)); sb.Append(c); break;
                    case ',':
                        sb.Append(c); sb.AppendLine(); sb.Append(new string(' ', indent * 2)); break;
                    case ':': sb.Append(c); sb.Append(' '); break;
                    default: if (!char.IsWhiteSpace(c)) sb.Append(c); break;
                }
            }
            return sb.ToString();
        }

        public static CertificateFlag CheckCertificateFlags(string signerName, int ageYears = 0)
        {
            if (string.IsNullOrEmpty(signerName)) return null;
            string lower = signerName.ToLowerInvariant();

            var alwaysCompromised = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase)
            {
                { "ht srl", "HackingTeam leak" },
                { "hackingteam", "HackingTeam leak" },
            };

            foreach (var kv in alwaysCompromised)
            {
                if (lower.Contains(kv.Key.ToLowerInvariant()))
                    return new CertificateFlag { Reason = kv.Value, IsCompromised = true };
            }

            // Realtek/JMicron: only flag old drivers (pre-2015) that may use the leaked Stuxnet-era certs
            var stuxnetSigners = new[] { "realtek semiconductor", "jmicron technology" };
            foreach (var signer in stuxnetSigners)
            {
                if (lower.Contains(signer))
                {
                    if (ageYears > 11)
                        return new CertificateFlag { Reason = "Leaked certificate (Stuxnet-era)", IsCompromised = true };
                    return new CertificateFlag { Reason = "Vendor had historical cert leak (Stuxnet); current cert OK", IsCompromised = false };
                }
            }

            string[] testKeywords = { "test", "debug", "sample", "development" };
            if (testKeywords.Any(k => lower.Contains(k)))
                return new CertificateFlag { Reason = "Test/debug certificate detected", IsTestSigned = true };

            return null;
        }

        private static double CalculateEntropy(byte[] data, int start, int length)
        {
            var freq = new int[256];
            int end = Math.Min(start + length, data.Length);
            int total = end - start;
            if (total <= 0) return 0;

            for (int i = start; i < end; i++)
                freq[data[i]]++;

            double entropy = 0;
            for (int i = 0; i < 256; i++)
            {
                if (freq[i] == 0) continue;
                double p = (double)freq[i] / total;
                entropy -= p * Math.Log(p, 2);
            }
            return Math.Round(entropy, 2);
        }

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text;

namespace BYOVDHunter
{
    public static class PeParser
    {
        public static byte[] ReadFile(string filePath)
        {
            return File.ReadAllBytes(filePath);
        }

        public static bool IsValidPe(byte[] data)
        {
            if (data.Length < 64 || data[0] != 0x4D || data[1] != 0x5A) return false;
            int peOff = BitConverter.ToInt32(data, 0x3C);
            if (peOff < 0 || peOff + 4 > data.Length) return false;
            return data[peOff] == 0x50 && data[peOff + 1] == 0x45 && data[peOff + 2] == 0 && data[peOff + 3] == 0;
        }

        public struct PeHeaders
        {
            public int PeOffset;
            public int CoffOffset;
            public int OptionalOffset;
            public ushort NumberOfSections;
            public ushort SizeOfOptionalHeader;
            public bool IsPe32Plus;
            public uint EntryPointRva;
            public uint TimeDateStamp;
            public ushort MajorLinkerVersion;
            public ushort MinorLinkerVersion;
            public ushort DllCharacteristics;
            public int SectionHeadersStart;
            public int DataDirectoryOffset;
            public ulong ImageBase;
            public List<SectionEntry> Sections;
        }

        public struct SectionEntry
        {
            public string Name;
            public uint VirtualSize;
            public uint VirtualAddress;
            public uint SizeOfRawData;
            public uint PointerToRawData;
            public uint Characteristics;
        }

        public static PeHeaders? ParseHeaders(byte[] data)
        {
            if (!IsValidPe(data)) return null;
            var h = new PeHeaders();
            h.PeOffset = BitConverter.ToInt32(data, 0x3C);
            h.CoffOffset = h.PeOffset + 4;
            if (h.CoffOffset + 20 > data.Length) return null;

            h.NumberOfSections = BitConverter.ToUInt16(data, h.CoffOffset + 2);
            h.TimeDateStamp = BitConverter.ToUInt32(data, h.CoffOffset + 4);
            h.SizeOfOptionalHeader = BitConverter.ToUInt16(data, h.CoffOffset + 16);
            h.OptionalOffset = h.CoffOffset + 20;

            if (h.OptionalOffset + 2 > data.Length) return null;
            ushort magic = BitConverter.ToUInt16(data, h.OptionalOffset);
            h.IsPe32Plus = magic == 0x20B;

            if (h.OptionalOffset + 28 > data.Length) return null;
            h.EntryPointRva = BitConverter.ToUInt32(data, h.OptionalOffset + 16);
            if (h.IsPe32Plus && h.OptionalOffset + 32 <= data.Length)
                h.ImageBase = BitConverter.ToUInt64(data, h.OptionalOffset + 24);
            else if (h.OptionalOffset + 32 <= data.Length)
                h.ImageBase = BitConverter.ToUInt32(data, h.OptionalOffset + 28);
            h.MajorLinkerVersion = data[h.OptionalOffset + 2];
            h.MinorLinkerVersion = data[h.OptionalOffset + 3];

            int dllCharOff = h.IsPe32Plus ? h.OptionalOffset + 70 : h.OptionalOffset + 66;
            if (dllCharOff + 2 <= data.Length)
                h.DllCharacteristics = BitConverter.ToUInt16(data, dllCharOff);

            h.DataDirectoryOffset = h.IsPe32Plus ? h.OptionalOffset + 112 : h.OptionalOffset + 96;
            h.SectionHeadersStart = h.OptionalOffset + h.SizeOfOptionalHeader;

            h.Sections = new List<SectionEntry>();
            for (int i = 0; i < h.NumberOfSections; i++)
            {
                int off = h.SectionHeadersStart + i * 40;
                if (off + 40 > data.Length) break;
                var sec = new SectionEntry();
                var nameBytes = new byte[8];
                Array.Copy(data, off, nameBytes, 0, 8);
                sec.Name = Encoding.ASCII.GetString(nameBytes).TrimEnd('\0');
                sec.VirtualSize = BitConverter.ToUInt32(data, off + 8);
                sec.VirtualAddress = BitConverter.ToUInt32(data, off + 12);
                sec.SizeOfRawData = BitConverter.ToUInt32(data, off + 16);
                sec.PointerToRawData = BitConverter.ToUInt32(data, off + 20);
                sec.Characteristics = BitConverter.ToUInt32(data, off + 36);
                h.Sections.Add(sec);
            }
            return h;
        }

        public static int RvaToOffset(uint rva, List<SectionEntry> sections)
        {
            foreach (var s in sections)
            {
                uint extent = Math.Max(s.VirtualSize, s.SizeOfRawData);
                if (rva >= s.VirtualAddress && rva < s.VirtualAddress + extent)
                    return (int)(rva - s.VirtualAddress + s.PointerToRawData);
            }
            return -1;
        }

        public static uint OffsetToRva(int offset, List<SectionEntry> sections)
        {
            foreach (var s in sections)
            {
                if (offset >= s.PointerToRawData && offset < s.PointerToRawData + s.SizeOfRawData)
                    return (uint)(offset - s.PointerToRawData + s.VirtualAddress);
            }
            return 0;
        }

        public static List<string> GetImportedFunctions(byte[] data)
        {
            var functions = new List<string>();
            try
            {
                var headers = ParseHeaders(data);
                if (headers == null) return functions;
                var h = headers.Value;

                int importDirEntry = h.DataDirectoryOffset + 8;
                if (importDirEntry + 8 > data.Length) return functions;
                uint importRva = BitConverter.ToUInt32(data, importDirEntry);
                if (importRva == 0) return functions;

                int importOff = RvaToOffset(importRva, h.Sections);
                if (importOff < 0) return functions;

                int desc = importOff;
                while (desc + 20 <= data.Length)
                {
                    uint oft = BitConverter.ToUInt32(data, desc);
                    uint nameRva = BitConverter.ToUInt32(data, desc + 12);
                    uint ft = BitConverter.ToUInt32(data, desc + 16);
                    if (oft == 0 && nameRva == 0 && ft == 0) break;

                    uint iltRva = oft != 0 ? oft : ft;
                    int iltOff = RvaToOffset(iltRva, h.Sections);
                    if (iltOff >= 0)
                    {
                        int entrySize = h.IsPe32Plus ? 8 : 4;
                        int pos = iltOff;
                        while (pos + entrySize <= data.Length)
                        {
                            ulong entry = h.IsPe32Plus
                                ? BitConverter.ToUInt64(data, pos)
                                : BitConverter.ToUInt32(data, pos);
                            if (entry == 0) break;
                            bool isOrdinal = h.IsPe32Plus
                                ? (entry & 0x8000000000000000UL) != 0
                                : (entry & 0x80000000UL) != 0;
                            if (!isOrdinal)
                            {
                                uint hintRva = (uint)(entry & 0x7FFFFFFFU);
                                int hintOff = RvaToOffset(hintRva, h.Sections);
                                if (hintOff >= 0 && hintOff + 2 < data.Length)
                                {
                                    string fn = ReadAsciiString(data, hintOff + 2);
                                    if (!string.IsNullOrEmpty(fn))
                                        functions.Add(fn);
                                }
                            }
                            pos += entrySize;
                        }
                    }
                    desc += 20;
                }
            }
            catch { }
            return functions;
        }

                uint exportRva = BitConverter.ToUInt32(data, exportDirEntry);
                if (exportRva == 0) return exports;

                int exportOff = RvaToOffset(exportRva, h.Sections);
                if (exportOff < 0 || exportOff + 40 > data.Length) return exports;

                uint numberOfNames = BitConverter.ToUInt32(data, exportOff + 24);
                uint namePointerRva = BitConverter.ToUInt32(data, exportOff + 32);
                int namePointerOff = RvaToOffset(namePointerRva, h.Sections);
                if (namePointerOff < 0) return exports;

                for (uint i = 0; i < numberOfNames && i < 500; i++)
                {
                    int nOff = namePointerOff + (int)(i * 4);
                    if (nOff + 4 > data.Length) break;
                    uint fnNameRva = BitConverter.ToUInt32(data, nOff);
                    int fnNameOff = RvaToOffset(fnNameRva, h.Sections);
                    if (fnNameOff >= 0)
                    {
                        string name = ReadAsciiString(data, fnNameOff);
                        if (!string.IsNullOrEmpty(name))
                            exports.Add(name);
                    }
                }
            }
            catch { }
            return exports;
        }

        public static List<string> FindDeviceNames(byte[] data)
        {
            var names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            string[] prefixes = { "\\Device\\", "\\DosDevices\\", "\\??\\", "\\\\.\\" };
            foreach (string prefix in prefixes)
            {
                byte[] pattern = Encoding.Unicode.GetBytes(prefix);
                int pos = 0;
                while (pos < data.Length - pattern.Length)
                {
                    pos = IndexOfBytes(data, pattern, pos);
                    if (pos < 0) break;
                    string str = ReadUnicodeString(data, pos);
                    if (str.Length > prefix.Length + 1 && !IsGarbageString(str))
                        names.Add(str);
                    pos += 2;
                }
            }
            return names.Take(15).ToList();
        }

        public static List<uint> FindIoctlCodes(byte[] data)
        {
            var candidates = new List<KeyValuePair<int, uint>>();
            for (int i = 512; i <= data.Length - 4; i++)
            {
                uint val = BitConverter.ToUInt32(data, i);
                uint deviceType = (val >> 16) & 0xFFFF;
                uint function = (val >> 2) & 0xFFF;
                if (deviceType == 0x22 || (deviceType >= 0x8000 && deviceType <= 0x80FF))
                {
                    if (function >= 1 && function <= 0xFFF)
                        candidates.Add(new KeyValuePair<int, uint>(i, val));
                }
            }
            var result = new HashSet<uint>();
            for (int i = 0; i < candidates.Count; i++)
            {
                bool hasNeighbor = false;
                if (i > 0 && candidates[i].Key - candidates[i - 1].Key <= 256) hasNeighbor = true;
                if (i < candidates.Count - 1 && candidates[i + 1].Key - candidates[i].Key <= 256) hasNeighbor = true;
                if (hasNeighbor) result.Add(candidates[i].Value);
                if (result.Count >= 20) break;
            }
            return result.OrderBy(c => c).ToList();
        }

        public static List<string> FindSuspiciousStrings(byte[] data)
        {
            var found = new List<string>();
            string[] patterns =
            {
                "PhysicalMemory", "SeDebugPrivilege", "SeTcbPrivilege",
                "ZwWriteVirtualMemory", "ZwReadVirtualMemory",
                "NtWriteVirtualMemory", "NtReadVirtualMemory",
                "g_CiEnabled", "g_CiOptions", "ci.dll",
            };
            foreach (string pattern in patterns)
            {
                byte[] ascii = Encoding.ASCII.GetBytes(pattern);
                byte[] unicode = Encoding.Unicode.GetBytes(pattern);
                if (IndexOfBytes(data, ascii, 0) >= 0 || IndexOfBytes(data, unicode, 0) >= 0)
                    found.Add(pattern);
            }
            return found;
        }

        public static List<string> FindSddlStrings(byte[] data)
        {
            var found = new List<string>();
            byte[] sddlAscii = Encoding.ASCII.GetBytes("D:P(A;;");
            byte[] sddlAscii2 = Encoding.ASCII.GetBytes("D:(A;;");
            byte[] sddlUni = Encoding.Unicode.GetBytes("D:P(A;;");
            byte[] sddlUni2 = Encoding.Unicode.GetBytes("D:(A;;");

            foreach (var pat in new[] { sddlAscii, sddlAscii2 })
            {
                int pos = IndexOfBytes(data, pat, 0);
                if (pos >= 0)
                {
                    string s = ReadAsciiString(data, pos);
                    if (s.Length > 6) found.Add(s);
                }
            }
            foreach (var pat in new[] { sddlUni, sddlUni2 })
            {
                int pos = IndexOfBytes(data, pat, 0);
                if (pos >= 0)
                {
                    string s = ReadUnicodeString(data, pos);
                    if (s.Length > 6) found.Add(s);
                }
            }
            return found;
        }

        public static string GetPdbPath(byte[] data)
        {
            try
            {
                var headers = ParseHeaders(data);
                if (headers == null) return null;
                var h = headers.Value;

                int debugDirEntry = h.DataDirectoryOffset + 6 * 8;
                if (debugDirEntry + 8 > data.Length) return null;
                uint debugRva = BitConverter.ToUInt32(data, debugDirEntry);
                uint debugSize = BitConverter.ToUInt32(data, debugDirEntry + 4);
                if (debugRva == 0 || debugSize == 0) return null;

                int debugOff = RvaToOffset(debugRva, h.Sections);
                if (debugOff < 0) return null;

                int count = (int)(debugSize / 28);
                for (int i = 0; i < count; i++)
                {
                    int entryOff = debugOff + i * 28;
                    if (entryOff + 28 > data.Length) break;
                    uint type = BitConverter.ToUInt32(data, entryOff + 12);
                    if (type == 2) // IMAGE_DEBUG_TYPE_CODEVIEW
                    {
                        uint dataOffset = BitConverter.ToUInt32(data, entryOff + 24);
                        if (dataOffset + 24 > data.Length) continue;
                        if (data[dataOffset] == 'R' && data[dataOffset + 1] == 'S' &&
                            data[dataOffset + 2] == 'D' && data[dataOffset + 3] == 'S')
                        {
                            int pdbOff = (int)dataOffset + 24;
                            return ReadAsciiString(data, pdbOff);
                        }
                    }
                }
            }
            catch { }
            return null;
        }

        #region Helpers

        public static int IndexOfBytes(byte[] data, byte[] pattern, int start)
        {
            if (pattern.Length == 0 || data.Length < pattern.Length) return -1;
            int end = data.Length - pattern.Length;
            byte first = pattern[0];
            for (int i = start; i <= end; i++)
            {
                if (data[i] != first) continue;
                bool match = true;
                for (int j = 1; j < pattern.Length; j++)
                {
                    if (data[i + j] != pattern[j]) { match = false; break; }
                }
                if (match) return i;
            }
            return -1;
        }

        public static string ReadUnicodeString(byte[] data, int offset)
        {
            var sb = new StringBuilder();
            int pos = offset;
            while (pos + 1 < data.Length && sb.Length < 256)
            {
                ushort ch = BitConverter.ToUInt16(data, pos);
                if (ch == 0 || ch < 0x20 || ch > 0x7E) break;
                sb.Append((char)ch);
                pos += 2;
            }
            return sb.ToString();
        }

        public static string ReadAsciiString(byte[] data, int offset)
        {
            var sb = new StringBuilder();
            while (offset < data.Length && data[offset] != 0 && sb.Length < 512)
            {
                char c = (char)data[offset];
                if (c < 0x20 || c > 0x7E) break;
                sb.Append(c);
                offset++;
            }
            return sb.ToString();
        }

        private static bool IsGarbageString(string s)
        {
            if (s.Length < 4) return true;
            int alphaCount = s.Count(c => char.IsLetterOrDigit(c));
            return alphaCount < s.Length / 3;
        }

        #endregion
    }
}

        public static List<DangerousImport> FindDangerousImports(byte[] data)
        {
            var imports = PeParser.GetImportedFunctions(data);
            var result = new List<DangerousImport>();
            foreach (string fn in imports)
            {
                if (DangerousImportDb.TryGetValue(fn, out var info))
                {
                    result.Add(new DangerousImport
                    {
                        FunctionName = fn,
                        Category = info.Category,
                        Description = info.Description
                    });
                }
            }
            return result;
        }

        public static HashSet<string> GetDangerousImportNames()
        {
            return new HashSet<string>(DangerousImportDb.Keys, StringComparer.OrdinalIgnoreCase);

        public static void SetWhitelistEnabled(bool enabled) { _whitelistEnabled = enabled; }

        public static bool IsWhitelisted(string fileName, string signerName, string filePath)
        {
            if (!_whitelistEnabled) return false;
            if (string.IsNullOrEmpty(fileName)) return false;
            if (!CoreWhitelist.Contains(fileName)) return false;

            if (!string.IsNullOrEmpty(signerName))
            {
                string lower = signerName.ToLowerInvariant();
                bool isMsSigned = lower.Contains("microsoft windows") ||
                                  lower.Contains("microsoft corporation") ||
                                  lower.Contains("catalog-signed");
                if (!isMsSigned) return false;
            }

            if (!string.IsNullOrEmpty(filePath))
            {
                string dir = Path.GetDirectoryName(filePath)?.ToLowerInvariant() ?? "";
                if (!dir.Contains(@"\windows\system32")) return false;
            }

            return true;
        }

        public static bool IsWhitelistedNameOnly(string fileName)
        {
            return _whitelistEnabled && CoreWhitelist.Contains(fileName);
        }

        #region Driver Discovery

        public static List<string> DiscoverDriverPaths(string extraScanPath, bool scanDriverStore)
        {
            var paths = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            string sys32Drivers = Path.Combine(
                Environment.GetFolderPath(Environment.SpecialFolder.Windows), "System32", "drivers");
            string sys32 = Path.Combine(
                Environment.GetFolderPath(Environment.SpecialFolder.Windows), "System32");

            AddDriversFromDir(paths, sys32Drivers);

            try
            {
                foreach (string f in Directory.GetFiles(sys32, "*.sys", SearchOption.TopDirectoryOnly))
                    paths.Add(f);
            }
            catch { }

            AddDriversFromRegistry(paths);

            if (scanDriverStore)
            {
                string driverStore = Path.Combine(
                    Environment.GetFolderPath(Environment.SpecialFolder.Windows),
                    "System32", "DriverStore", "FileRepository");
                if (Directory.Exists(driverStore))
                {
                    try
                    {
                        foreach (string f in Directory.GetFiles(driverStore, "*.sys", SearchOption.AllDirectories))
                            paths.Add(f);
                    }
                    catch { }
                }
            }

            if (!string.IsNullOrEmpty(extraScanPath) && Directory.Exists(extraScanPath))
                AddDriversFromDir(paths, extraScanPath);

            return paths.ToList();
        }

        private static void AddDriversFromDir(HashSet<string> paths, string dir)
        {
            try
            {
                foreach (string f in Directory.GetFiles(dir, "*.sys", SearchOption.TopDirectoryOnly))
                    paths.Add(f);
            }
            catch { }
        }

        private static void AddDriversFromRegistry(HashSet<string> paths)
        {
            try
            {
                using (var services = Microsoft.Win32.Registry.LocalMachine.OpenSubKey(
                    @"SYSTEM\CurrentControlSet\Services"))
                {
                    if (services == null) return;
                    foreach (string svcName in services.GetSubKeyNames())
                    {
                        try
                        {
                            using (var svc = services.OpenSubKey(svcName))
                            {
                                if (svc == null) continue;
                                object typeObj = svc.GetValue("Type");
                                if (typeObj == null) continue;
                                int type = Convert.ToInt32(typeObj);
                                if (type != 1 && type != 2) continue;

                                string imagePath = svc.GetValue("ImagePath") as string;
                                if (string.IsNullOrEmpty(imagePath)) continue;
                                imagePath = imagePath.Trim();

                                if (imagePath.StartsWith(@"\SystemRoot\", StringComparison.OrdinalIgnoreCase))
                                    imagePath = Path.Combine(
                                        Environment.GetFolderPath(Environment.SpecialFolder.Windows),
                                        imagePath.Substring(12));
                                else if (imagePath.StartsWith(@"system32\", StringComparison.OrdinalIgnoreCase) ||
                                         imagePath.StartsWith(@"System32\", StringComparison.OrdinalIgnoreCase))
                                    imagePath = Path.Combine(
                                        Environment.GetFolderPath(Environment.SpecialFolder.Windows), imagePath);
                                else if (imagePath.StartsWith(@"\??\"))
                                    imagePath = imagePath.Substring(4);
                                else if (imagePath.StartsWith(@"\\?\"))
                                    imagePath = imagePath.Substring(4);

                                if (File.Exists(imagePath) &&
                                    imagePath.EndsWith(".sys", StringComparison.OrdinalIgnoreCase))
                                    paths.Add(imagePath);
                            }
                        }
                        catch { }
                    }
                }
            }
            catch { }
        }

        #region Loaded Drivers

        public static HashSet<string> GetLoadedDriverBaseNames()
        {
            var names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            try
            {
                int needed = 0;
                EnumDeviceDrivers(null, 0, out needed);
                if (needed <= 0) return names;
                IntPtr[] addrs = new IntPtr[needed / IntPtr.Size];
                if (!EnumDeviceDrivers(addrs, needed, out _)) return names;
                var sb = new StringBuilder(260);
                foreach (var addr in addrs)
                {
                    if (addr == IntPtr.Zero) continue;
                    sb.Clear();
                    if (GetDeviceDriverBaseName(addr, sb, sb.Capacity) > 0)
                        names.Add(sb.ToString());
                }
            }
            catch { }
            return names;
        }

        [DllImport("psapi.dll", SetLastError = true)]
        private static extern bool EnumDeviceDrivers([Out] IntPtr[] lpImageBase, int cb, out int lpcbNeeded);

        [DllImport("psapi.dll", CharSet = CharSet.Unicode, SetLastError = true)]
        private static extern int GetDeviceDriverBaseName(IntPtr ImageBase, StringBuilder lpBaseName, int nSize);

        [DllImport("psapi.dll", CharSet = CharSet.Unicode, SetLastError = true)]
        private static extern int GetDeviceDriverFileName(IntPtr ImageBase, StringBuilder lpFileName, int nSize);

        public static Dictionary<string, string> GetLoadedDriverFullPaths()
        {
            var result = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
            try
            {
                int needed = 0;
                EnumDeviceDrivers(null, 0, out needed);
                if (needed <= 0) return result;
                IntPtr[] addrs = new IntPtr[needed / IntPtr.Size];
                if (!EnumDeviceDrivers(addrs, needed, out _)) return result;
                var sbName = new StringBuilder(260);
                var sbPath = new StringBuilder(260);
                string winDir = Environment.GetFolderPath(Environment.SpecialFolder.Windows);
                foreach (var addr in addrs)
                {
                    if (addr == IntPtr.Zero) continue;
                    sbName.Clear();
                    if (GetDeviceDriverBaseName(addr, sbName, sbName.Capacity) <= 0) continue;
                    string baseName = sbName.ToString();

                    sbPath.Clear();
                    if (GetDeviceDriverFileName(addr, sbPath, sbPath.Capacity) > 0)
                    {
                        string fullPath = NormalizeKernelPath(sbPath.ToString(), winDir);
                        if (!string.IsNullOrEmpty(fullPath) && File.Exists(fullPath))
                        {
                            if (!result.ContainsKey(baseName))
                                result[baseName] = fullPath;
                        }
                    }
                }
            }
            catch { }
            return result;
        }

        private static string NormalizeKernelPath(string kernelPath, string winDir)
        {
            if (string.IsNullOrEmpty(kernelPath)) return null;

            if (kernelPath.StartsWith(@"\SystemRoot\", StringComparison.OrdinalIgnoreCase))
                return Path.Combine(winDir, kernelPath.Substring(12));
            if (kernelPath.StartsWith(@"\??\"))
                return kernelPath.Substring(4);
            if (kernelPath.StartsWith(@"\\?\"))
                return kernelPath.Substring(4);
            if (kernelPath.StartsWith(@"system32\", StringComparison.OrdinalIgnoreCase))
                return Path.Combine(winDir, kernelPath);

            if (kernelPath.Length >= 3 && kernelPath[1] == ':')
                return kernelPath;

            return null;
        }

        #region Signature Verification

        public static void CheckSignature(string filePath,
            out bool isSigned, out bool isValid, out string signerName)
        {
            isSigned = false; isValid = false; signerName = "";
            try
            {
                Guid actionId = new Guid("00AAC56B-CD44-11d0-8CC2-00C04FC295EE");
                var fileInfo = new WINTRUST_FILE_INFO
                {
                    cbStruct = (uint)Marshal.SizeOf(typeof(WINTRUST_FILE_INFO)),
                    pcwszFilePath = filePath
                };
                IntPtr fileInfoPtr = Marshal.AllocHGlobal(Marshal.SizeOf(fileInfo));
                Marshal.StructureToPtr(fileInfo, fileInfoPtr, false);

                var trustData = new WINTRUST_DATA
                {
                    cbStruct = (uint)Marshal.SizeOf(typeof(WINTRUST_DATA)),
                    dwUIChoice = 2,
                    fdwRevocationChecks = 0,
                    dwUnionChoice = 1,
                    pFile = fileInfoPtr,
                    dwStateAction = 0,
                    dwProvFlags = 0
                };

                int hr = WinVerifyTrust(IntPtr.Zero, ref actionId, ref trustData);
                Marshal.FreeHGlobal(fileInfoPtr);

                if (hr == 0) { isSigned = true; isValid = true; }
                else if (hr == unchecked((int)0x800B0101)) { isSigned = true; isValid = false; }

                if (isSigned)
                {
                    try
                    {
                        var cert = new X509Certificate2(X509Certificate.CreateFromSignedFile(filePath));
                        signerName = cert.GetNameInfo(X509NameType.SimpleName, false);
                    }
                    catch { signerName = "(embedded)"; }
                }

                if (!isSigned)
                {
                    bool catSigned = CheckCatalogSignature(filePath, out string catSigner);
                    if (catSigned)
                    {
                        isSigned = true; isValid = true;
                        signerName = !string.IsNullOrEmpty(catSigner) ? catSigner : "(catalog-signed)";
                    }
                }
            }
            catch { }
        }

        private static bool CheckCatalogSignature(string filePath, out string signerName)
        {
            signerName = "";
            IntPtr catAdmin = IntPtr.Zero;
            try
            {
                Guid subsystem = new Guid("F750E6C3-38EE-11D1-85E5-00C04FC295EE");
                if (!CryptCATAdminAcquireContext(ref catAdmin, ref subsystem, 0)) return false;

                using (var fs = new FileStream(filePath, FileMode.Open, FileAccess.Read, FileShare.Read))
                {
                    int hashSize = 0;
                    CryptCATAdminCalcHashFromFileHandle(fs.SafeFileHandle.DangerousGetHandle(),
                        ref hashSize, IntPtr.Zero, 0);
                    if (hashSize <= 0) return false;
                    IntPtr hashPtr = Marshal.AllocHGlobal(hashSize);
                    try
                    {
                        if (!CryptCATAdminCalcHashFromFileHandle(fs.SafeFileHandle.DangerousGetHandle(),
                            ref hashSize, hashPtr, 0)) return false;
                        IntPtr catInfo = CryptCATAdminEnumCatalogFromHash(
                            catAdmin, hashPtr, hashSize, 0, IntPtr.Zero);
                        if (catInfo != IntPtr.Zero)
                        {
                            CryptCATAdminReleaseCatalogContext(catAdmin, catInfo, 0);
                            signerName = "(catalog-signed)";
                            return true;
                        }
                    }
                    finally { Marshal.FreeHGlobal(hashPtr); }
                }
            }
            catch { }
            finally { if (catAdmin != IntPtr.Zero) CryptCATAdminReleaseContext(catAdmin, 0); }
            return false;
        }

        #region Deep Analysis

        public static void DeepAnalysis(DriverResult dr, byte[] data, bool skipTaint = false)
        {
            try
            {
                var deviceNames = PeParser.FindDeviceNames(data);
                foreach (string dn in deviceNames)
                {
                    var di = new DeviceInfo { DeviceName = dn };
                    if (dn.StartsWith("\\DosDevices\\") || dn.StartsWith("\\\\.\\"))
                        di.SymbolicLink = dn;
                    dr.Devices.Add(di);
                }

                var sddlStrings = PeParser.FindSddlStrings(data);
                if (sddlStrings.Count > 0 && dr.Devices.Count > 0)
                    dr.Devices[0].SddlAccess = ParseSddlAccess(sddlStrings[0]);

                var rawIoctlCodes = PeParser.FindIoctlCodes(data);

                var headers = PeParser.ParseHeaders(data);
                if (headers != null)
                {
                    var iatMap = PeParser.GetImportAddressTable(data);

                    dr.HasDeviceCreation = iatMap.Values.Any(v =>
                        v.Equals("IoCreateDevice", StringComparison.OrdinalIgnoreCase) ||
                        v.Equals("IoCreateSymbolicLink", StringComparison.OrdinalIgnoreCase));

                    var dangerousNames = new HashSet<string>(
                        dr.DangerousImports.Select(i => i.FunctionName), StringComparer.OrdinalIgnoreCase);

                    var dispResult = DispatchAnalyzer.Analyze(data, headers.Value, iatMap, dangerousNames);
                    dr.Dispatch = dispResult;

                    if (dispResult.ExtractedIoctlCodes.Count > 0)
                        foreach (uint c in dispResult.ExtractedIoctlCodes)
                            if (!rawIoctlCodes.Contains(c)) rawIoctlCodes.Add(c);

                    uint reachStartRva = dispResult.DeviceControlHandlerRva;
                    if (reachStartRva == 0)
                    {
                        bool hasCreate = iatMap.Values.Any(v =>
                            v.Equals("IoCreateDevice", StringComparison.OrdinalIgnoreCase));
                        bool hasSymLink = iatMap.Values.Any(v =>
                            v.Equals("IoCreateSymbolicLink", StringComparison.OrdinalIgnoreCase));
                        if (hasCreate && hasSymLink)
                            reachStartRva = headers.Value.EntryPointRva;
                    }

                    if (reachStartRva != 0 && dangerousNames.Count > 0)
                    {
                        var reachability = DispatchAnalyzer.AnalyzeReachability(
                            data, headers.Value, reachStartRva, iatMap, dangerousNames);

                        foreach (var imp in dr.DangerousImports)
                        {
                            if (reachability.TryGetValue(imp.FunctionName, out var reach))
                            {
                                imp.Reachability = reach;
                                if (!skipTaint && (reach == Reachability.Direct || reach == Reachability.Indirect))
                                {
                                    var sites = FindImportCallSites(data, headers.Value, iatMap, imp.FunctionName);
                                    if (sites.Count > 0)
                                        imp.Taint = DispatchAnalyzer.AnalyzeTaint(data, sites[0], headers.Value);
                                }
                            }
                        }
                    }

                    bool hasSecureCreate = iatMap.Values.Any(v =>
                        v.Equals("IoCreateDeviceSecureCreate", StringComparison.OrdinalIgnoreCase) ||
                        v.Equals("WdmlibIoCreateDeviceSecure", StringComparison.OrdinalIgnoreCase));
                    if (hasSecureCreate && dr.Devices.Count > 0)
                        dr.Devices[0].HasSecureOpen = true;
                }

                foreach (uint code in rawIoctlCodes.Take(20))
                    dr.IoctlCodes.Add(IoctlInfo.Decode(code));

                dr.SuspiciousStrings = PeParser.FindSuspiciousStrings(data);

                // Compute behavior tags from reachable dangerous imports
                BehaviorTag behaviors = BehaviorTag.None;
                foreach (var imp in dr.DangerousImports)
                {
                    if (imp.Reachability == Reachability.Direct || imp.Reachability == Reachability.Indirect)
                    {
                        if (CategoryToBehavior.TryGetValue(imp.Category, out var bt))
                        {
                            imp.BehaviorTag = bt;
                            behaviors |= bt;
                        }
                        if (imp.Taint == TaintStatus.Tainted || imp.Taint == TaintStatus.LikelyTainted)
                        {
                            behaviors |= BehaviorTag.B12_TaintedKernelArgs;
                        }
                    }
                }

                // B6 heuristic: CITamper import reachable OR suspicious strings + PhysMem reachable
                if (behaviors.HasFlag(BehaviorTag.B6_CodeIntegrityTampering) ||
                    (dr.SuspiciousStrings.Any(s => s == "g_CiEnabled" || s == "g_CiOptions") &&
                     dr.DangerousImports.Any(i => i.Category == "PhysMem" &&
                         (i.Reachability == Reachability.Direct || i.Reachability == Reachability.Indirect))))
                {
                    behaviors |= BehaviorTag.B6_CodeIntegrityTampering;
                }

                // B14 detection: requires IOCTL reachability + stack buffer pattern
                if (headers != null && dr.DangerousImports.Any(i =>
                    i.FunctionName == "RtlQueryRegistryValues" &&
                    (i.Reachability == Reachability.Direct || i.Reachability == Reachability.Indirect)))
                {
                    var iatMap2 = PeParser.GetImportAddressTable(data);
                    if (DetectUnsafeRtlQueryRegistry(data, headers.Value, iatMap2))
                        behaviors |= BehaviorTag.B14_UnsafeRtlQueryRegistry;
                }

                // Refine B8 for MmMapLockedPagesSpecifyCache: check AccessMode parameter
                if (behaviors.HasFlag(BehaviorTag.B8_ArbitraryMemoryWrite) && headers != null)
                {
                    var reachablePhysMem = dr.DangerousImports.FindAll(i =>
                        i.Category == "PhysMem" &&
                        (i.Reachability == Reachability.Direct || i.Reachability == Reachability.Indirect));
                    if (reachablePhysMem.Count > 0 &&
                        reachablePhysMem.TrueForAll(i => i.FunctionName == "MmMapLockedPagesSpecifyCache"))
                    {
                        var iatMap3 = PeParser.GetImportAddressTable(data);
                        uint hRva = dr.Dispatch != null ? dr.Dispatch.DeviceControlHandlerRva : 0;
                        if (!HasUserModeMapping(data, headers.Value, iatMap3, hRva))
                            behaviors &= ~BehaviorTag.B8_ArbitraryMemoryWrite;
                    }
                }

                dr.Behaviors = behaviors;
            }
            catch { }
        }

        private static List<int> FindImportCallSites(byte[] data, PeParser.PeHeaders headers,
            Dictionary<uint, string> iatMap, string importName)
        {
            var sites = new List<int>();
            uint iatRva = 0;
            foreach (var kv in iatMap)
            {
                if (kv.Value.Equals(importName, StringComparison.OrdinalIgnoreCase))
                { iatRva = kv.Key; break; }
            }
            if (iatRva == 0) return sites;

            for (int i = 0; i < data.Length - 6; i++)
            {
                if (data[i] == 0xFF && data[i + 1] == 0x15)
                {
                    int disp = BitConverter.ToInt32(data, i + 2);
                    uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                    if (instrRva > 0 && (uint)(instrRva + 6 + disp) == iatRva)
                    {
                        sites.Add(i);
                        if (sites.Count >= 5) break;
                    }
                }
            }
            return sites;
        }

        private static bool DetectUnsafeRtlQueryRegistry(byte[] data, PeParser.PeHeaders headers,
            Dictionary<uint, string> iatMap)
        {
            uint targetIatRva = 0;
            foreach (var kv in iatMap)
            {
                if (kv.Value.Equals("RtlQueryRegistryValues", StringComparison.OrdinalIgnoreCase))
                { targetIatRva = kv.Key; break; }
            }
            if (targetIatRva == 0) return false;

            for (int i = 0; i < data.Length - 6; i++)
            {
                if (data[i] != 0xFF || data[i + 1] != 0x15) continue;
                int disp = BitConverter.ToInt32(data, i + 2);
                uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                if (instrRva == 0) continue;
                if ((uint)(instrRva + 6 + disp) != targetIatRva) continue;

                bool hasStackQueryTable = false;
                int searchStart = Math.Max(0, i - 64);
                for (int j = i - 4; j >= searchStart; j--)
                {
                    if (j + 5 <= data.Length && data[j] == 0x4C && data[j + 1] == 0x8D && data[j + 2] == 0x44 && data[j + 3] == 0x24)
                    { hasStackQueryTable = true; break; }
                    if (j + 8 <= data.Length && data[j] == 0x4C && data[j + 1] == 0x8D && data[j + 2] == 0x84 && data[j + 3] == 0x24)
                    { hasStackQueryTable = true; break; }
                    if (j + 4 <= data.Length && data[j] == 0x4C && data[j + 1] == 0x8D && data[j + 2] == 0x45)
                    { hasStackQueryTable = true; break; }
                    if (j + 7 <= data.Length && data[j] == 0x4C && data[j + 1] == 0x8D && data[j + 2] == 0x85)
                    { hasStackQueryTable = true; break; }
                }

                if (!hasStackQueryTable) continue;

                // Check for DIRECT flag (0x20) and TYPECHECK mitigation (0x100)
                // Search MOV DWORD patterns that set the Flags field
                bool hasDirect = false;
                bool hasTypecheck = false;
                bool anyFlagStore = false;
                int flagSearch = Math.Max(0, i - 128);
                for (int j = flagSearch; j < i; j++)
                {
                    if (j + 7 > data.Length) break;
                    // C7 41 xx imm32 - MOV DWORD [RCX+disp8], imm32
                    // C7 45 xx imm32 - MOV DWORD [RBP+disp8], imm32
                    // C7 44 24 xx imm32 - MOV DWORD [RSP+disp8], imm32
                    // C7 84 24 xx xx xx xx imm32 - MOV DWORD [RSP+disp32], imm32
                    uint flagVal = 0;
                    bool found = false;
                    if (data[j] == 0xC7 && data[j + 1] == 0x41 && j + 7 <= data.Length)
                    { flagVal = BitConverter.ToUInt32(data, j + 3); found = true; }
                    else if (data[j] == 0xC7 && data[j + 1] == 0x45 && j + 7 <= data.Length)
                    { flagVal = BitConverter.ToUInt32(data, j + 3); found = true; }
                    else if (data[j] == 0xC7 && data[j + 1] == 0x44 && data[j + 2] == 0x24 && j + 8 <= data.Length)
                    { flagVal = BitConverter.ToUInt32(data, j + 4); found = true; }
                    else if (data[j] == 0xC7 && data[j + 1] == 0x84 && data[j + 2] == 0x24 && j + 11 <= data.Length)
                    { flagVal = BitConverter.ToUInt32(data, j + 7); found = true; }
                    else if (data[j] == 0xC7 && data[j + 1] == 0x85 && j + 10 <= data.Length)
                    { flagVal = BitConverter.ToUInt32(data, j + 6); found = true; }

                    if (found && flagVal < 0x1000)
                    {
                        anyFlagStore = true;
                        if ((flagVal & 0x20) != 0)
                        {
                            hasDirect = true;
                            if ((flagVal & 0x100) != 0) hasTypecheck = true;
                        }
                    }
                }

                if (hasDirect && !hasTypecheck) return true;
                if (anyFlagStore && !hasDirect) continue;
                if (!anyFlagStore && hasStackQueryTable) return true;
            }
            return false;
        }

        private static bool HasUserModeMapping(byte[] data, PeParser.PeHeaders headers,
            Dictionary<uint, string> iatMap, uint handlerRva)
        {
            uint targetIatRva = 0;
            foreach (var kv in iatMap)
            {
                if (kv.Value.Equals("MmMapLockedPagesSpecifyCache", StringComparison.OrdinalIgnoreCase))
                { targetIatRva = kv.Key; break; }
            }
            if (targetIatRva == 0) return false;

            HashSet<uint> reachableFuncs = null;
            if (handlerRva != 0)
                reachableFuncs = BuildReachableSet(data, headers, handlerRva);

            for (int i = 0; i < data.Length - 6; i++)
            {
                if (data[i] != 0xFF || data[i + 1] != 0x15) continue;
                int disp = BitConverter.ToInt32(data, i + 2);
                uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                if (instrRva == 0) continue;
                if ((uint)(instrRva + 6 + disp) != targetIatRva) continue;

                if (reachableFuncs != null)
                {
                    uint funcRva = FindFunctionStart(data, headers, i);
                    if (!reachableFuncs.Contains(funcRva)) continue;
                }

                int searchStart = Math.Max(0, i - 32);
                bool kernelMode = false;
                for (int j = i - 2; j >= searchStart; j--)
                {
                    if (j + 2 <= i && (data[j] == 0x31 || data[j] == 0x33) && data[j + 1] == 0xD2)
                    { kernelMode = true; break; }
                    if (j + 5 <= i && data[j] == 0xBA &&
                        data[j + 1] == 0 && data[j + 2] == 0 && data[j + 3] == 0 && data[j + 4] == 0)
                    { kernelMode = true; break; }
                }
                if (!kernelMode) return true;
            }
            return false;
        }

        private static uint FindFunctionStart(byte[] data, PeParser.PeHeaders headers, int offset)
        {
            for (int i = offset - 1; i >= Math.Max(0, offset - 4096); i--)
            {
                if (data[i] == 0xCC && i > 0 && data[i - 1] == 0xCC)
                    return PeParser.OffsetToRva(i + 1, headers.Sections);
            }
            return PeParser.OffsetToRva(offset, headers.Sections);
        }

        private static HashSet<uint> BuildReachableSet(byte[] data, PeParser.PeHeaders headers, uint handlerRva)
        {
            var reachable = new HashSet<uint>();
            reachable.Add(handlerRva);

            int hOff = PeParser.RvaToOffset(handlerRva, headers.Sections);
            if (hOff <= 0) return reachable;

            var depth1 = CollectCallTargets(data, headers, hOff, 0x3000);
            foreach (uint t in depth1) reachable.Add(t);

            foreach (uint t in depth1)
            {
                int tOff = PeParser.RvaToOffset(t, headers.Sections);
                if (tOff <= 0 || tOff >= data.Length) continue;
                foreach (uint t2 in CollectCallTargets(data, headers, tOff, 0x2000))
                    reachable.Add(t2);
            }
            return reachable;
        }

        private static HashSet<uint> CollectCallTargets(byte[] data, PeParser.PeHeaders headers,
            int startOffset, int maxScan)
        {
            var targets = new HashSet<uint>();
            int end = Math.Min(startOffset + maxScan, data.Length - 5);
            for (int i = startOffset; i < end; i++)
            {
                if (data[i] != 0xE8) continue;
                int disp = BitConverter.ToInt32(data, i + 1);
                uint instrRva = PeParser.OffsetToRva(i, headers.Sections);
                if (instrRva == 0) continue;
                uint target = (uint)(instrRva + 5 + disp);
                if (target > 0x1000 && target < 0x200000)
                    targets.Add(target);
            }
            return targets;
        }

        public static void AssessRisk(DriverResult dr, SystemProtection sysStatus)
        {
            if (dr.IsWhitelisted && dr.LolMatch == null && dr.BlocklistMatch == null)
            {
                dr.Risk = RiskLevel.Info;
                dr.RiskReason = "Whitelisted (Microsoft core driver, signature verified)";
                return;
            }

            bool hasLolOrBlocklist = dr.LolMatch != null || dr.BlocklistMatch != null;
            if (hasLolOrBlocklist)
            {
                if (dr.IsLoaded)
                {
                    dr.Risk = RiskLevel.Critical;
                    dr.RiskReason = FormatDbMatch(dr) + " - LOADED in kernel";
                    dr.Impact = "Known exploitable - check CVE details";
                    dr.ExploitUse = "BYOVD";
                    if (dr.BlocklistMatch != null && !sysStatus.HvciEnabled)
                        dr.Note = "HVCI is OFF - blocklist NOT enforced";
                }
                else
                {
                    dr.Risk = RiskLevel.High;
                    dr.RiskReason = FormatDbMatch(dr) + " - on disk (not loaded)";
                    dr.Impact = "Available for drop-and-load BYOVD attack";
                    dr.ExploitUse = "BYOVD (requires load)";
                }
                return;
            }

            bool hasDeviceExposure = dr.Devices.Count > 0 || dr.HasDeviceCreation;

            bool hasTier1Direct = dr.DangerousImports.Any(i =>
                i.Reachability == Reachability.Direct && Tier1Imports.Contains(i.FunctionName));
            bool hasTier1Indirect = dr.DangerousImports.Any(i =>
                i.Reachability == Reachability.Indirect && Tier1Imports.Contains(i.FunctionName) &&
                i.Taint != TaintStatus.NotTainted);
            bool hasTier2Direct = dr.DangerousImports.Any(i =>
                i.Reachability == Reachability.Direct && Tier2Imports.Contains(i.FunctionName));
            bool hasAnyTierDirect = hasTier1Direct || hasTier2Direct;

            if (dr.IsMicrosoftSigned && !hasAnyTierDirect)
            {
                dr.Risk = dr.DangerousImports.Count > 0 ? RiskLevel.Medium : RiskLevel.Info;
                dr.RiskReason = "Microsoft signed, no Tier 1/Tier 2 import DIRECT";
                return;
            }

            bool hasMethodNeither = dr.IoctlCodes.Any(c => c.Method == 3);
            bool noValidation = dr.Dispatch != null &&
                dr.Dispatch.BufferValidation == BufferValidation.NoValidation;
            bool hasHandlerFound = dr.Dispatch != null && dr.Dispatch.DeviceControlHandlerRva != 0;

            if (hasTier1Direct && hasDeviceExposure && (hasMethodNeither || noValidation) && dr.IsLoaded)
            {
                dr.Risk = RiskLevel.Critical;
                dr.RiskReason = "Tier 1 DIRECT + device exposure + weak validation - LOADED";
                dr.Impact = BuildImpactString(dr);
                dr.ExploitUse = BuildExploitUse(dr);
                if (dr.Dispatch != null && !dr.Dispatch.HasPreviousModeCheck)
                    dr.Note = "No PreviousMode check";
                return;
            }

            if (hasTier1Indirect && hasDeviceExposure && dr.IsLoaded)
            {
                dr.Risk = RiskLevel.High;
                dr.RiskReason = "Tier 1 INDIRECT + device exposure - LOADED";
                dr.Impact = BuildImpactString(dr);
                dr.ExploitUse = BuildExploitUse(dr);
                return;
            }

            if (hasTier1Direct && hasDeviceExposure && !dr.IsLoaded)
            {
                dr.Risk = RiskLevel.High;
                dr.RiskReason = "Tier 1 DIRECT + device exposure - on disk";
                dr.Impact = BuildImpactString(dr);
                dr.ExploitUse = BuildExploitUse(dr) + " (requires load)";
                return;
            }

            if (hasTier2Direct && hasDeviceExposure && dr.IsLoaded)
            {
                dr.Risk = RiskLevel.High;
                dr.RiskReason = "Tier 2 DIRECT + device exposure - LOADED";
                dr.Impact = BuildImpactString(dr);
                dr.ExploitUse = BuildExploitUse(dr);
                return;
            }

            if (!dr.IsSigned && dr.IsLoaded && hasDeviceExposure && hasHandlerFound)
            {
                dr.Risk = RiskLevel.High;
                dr.RiskReason = "Unsigned + loaded + device exposure + handler found";
                dr.Impact = "Unsigned driver with IOCTL handler";
                dr.ExploitUse = "BYOVD";
                return;
            }

            if (dr.DangerousImports.Count > 0 && hasDeviceExposure)
            {
                dr.Risk = RiskLevel.Medium;
                dr.RiskReason = dr.IsLoaded
                    ? "Dangerous imports + device exposure (non-critical path)"
                    : "Dangerous imports + device exposure (not loaded)";
                return;
            }

            if (dr.DangerousImports.Count > 0 && dr.IsLoaded)
            {
                dr.Risk = RiskLevel.Medium;
                dr.RiskReason = "Dangerous imports + loaded (no device exposure)";
                return;
            }

            if (dr.DangerousImports.Count > 0)
            {
                dr.Risk = RiskLevel.Low;
                dr.RiskReason = "Dangerous imports (not loaded, no device exposure)";
                return;
            }

            if (!dr.IsSigned)
            {
                dr.Risk = RiskLevel.Low;
                dr.RiskReason = "Unsigned driver";
                return;
            }

            dr.Risk = RiskLevel.Info;
        }

        private static string FormatDbMatch(DriverResult dr)

        private static string BuildImpactString(DriverResult dr)
        {
            var impacts = new List<string>();
            bool hasMem = dr.DangerousImports.Any(i => i.Category == "PhysMem");
            bool hasProc = dr.DangerousImports.Any(i => i.Category == "ProcKill" || i.Category == "ProcManip");
            bool hasCallbackUnreg = dr.DangerousImports.Any(i => i.Category == "CallbackUnreg");
            bool hasIoPort = dr.DangerousImports.Any(i => i.Category == "IoPort");
            bool hasMsr = dr.DangerousImports.Any(i => i.Category == "MSR");

            if (hasMem) impacts.Add("LPE");
            if (hasMem || hasCallbackUnreg) impacts.Add("EDR bypass");
            if (hasProc) impacts.Add("EDR kill");
            if (hasMem) impacts.Add("credential theft");
            if (hasIoPort || hasMsr) impacts.Add("firmware persistence");
            if (impacts.Count == 0) impacts.Add("potential exploitation");
            return string.Join(" / ", impacts);
        }

        private static string BuildExploitUse(DriverResult dr)
        {
            var uses = new List<string> { "BYOVD" };
            bool hasMem = dr.DangerousImports.Any(i => i.Category == "PhysMem");
            bool hasProc = dr.DangerousImports.Any(i => i.Category == "ProcKill" || i.Category == "ProcManip");
            bool hasCallbackUnreg = dr.DangerousImports.Any(i => i.Category == "CallbackUnreg");

            if (hasProc) uses.Add("EDR killer");
            if (hasCallbackUnreg) uses.Add("EDR blinding");
            if (hasMem) { uses.Add("PPL bypass"); uses.Add("DSE bypass"); }
            uses.Add("persistence");
            return string.Join(" / ", uses);
        }

            if (sddl.Contains("SY") && !sddl.Contains("BA"))
                return "SYSTEM only";
            return "Custom ACL";
        }

            {

                status.HvciEnabled = hvci != null && Convert.ToInt32(hvci) != 0;
            }
            catch { }

            {
                object sb = ReadReg(@"SYSTEM\CurrentControlSet\Control\SecureBoot\State", "UEFISecureBootEnabled");
                status.SecureBootEnabled = sb != null && Convert.ToInt32(sb) != 0;
            }
            catch { }

            {
                object testSign = ReadReg(@"SYSTEM\CurrentControlSet\Control\CI", "TestSigning");
                status.TestSigningEnabled = testSign != null && Convert.ToInt32(testSign) != 0;
                status.DseEnforced = !status.TestSigningEnabled;

            {
                status.DseEnforced = true;
            }

            {
                object vbs = ReadReg(@"SYSTEM\CurrentControlSet\Control\DeviceGuard", "EnableVirtualizationBasedSecurity");
                status.VbsEnabled = vbs != null && Convert.ToInt32(vbs) != 0;
            }
            catch { }

        {

            if (!MsBlocklistHashes.Contains(sha256)) return null;
            return new MsBlocklistMatch
            {
                DriverName = "",
                Description = "Driver hash matches Microsoft Vulnerable Driver Blocklist",
                HvciEnabled = hvciEnabled,
                BlocklistEnabled = blocklistEnabled
            };
        }
