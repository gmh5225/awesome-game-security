    public boolean isFullSolidBlock(Material mat) {
        if (mat == null || mat.isAir() || !mat.isBlock() || !mat.isSolid() || !mat.isOccluding()) {
            return false;
        }
        String name = mat.name();
        return !name.contains("GLASS")
                && !name.contains("LEAVES")
                && !name.contains("FENCE")
                && !name.contains("WALL")
                && !name.contains("STAIRS")
                && !name.contains("SLAB")
                && !name.contains("DOOR")
                && !name.contains("TRAPDOOR")
                && !name.contains("BARS")
                && !name.contains("CHEST")
                && !name.contains("SHULKER")
                && !name.equals("SPAWNER")
                && !name.equals("HOPPER")
                && !name.equals("SLIME_BLOCK")
                && !name.equals("HONEY_BLOCK")
                && !name.equals("ICE");
    }

    public boolean isHiddenByYLayer(Player viewer, double targetY, boolean isPlayerTarget) {
        if (!getConfig().getBoolean("hien-tren-an-duoi.bat", true)) return false;
        if (isPlayerTarget && !getConfig().getBoolean("hien-tren-an-duoi.an-nguoi-choi", true)) return false;
        if (!isPlayerTarget && !getConfig().getBoolean("hien-tren-an-duoi.an-ruong-block", true)) return false;

        double splitY = getConfig().getDouble("hien-tren-an-duoi.moc-do-cao-y", 35.0);
        boolean viewerAbove = playerAboveLayerState.computeIfAbsent(
                viewer.getUniqueId(), k -> viewer.getLocation().getY() > splitY
        );
        boolean targetAbove = targetY > splitY;
        return viewerAbove != targetAbove;
    }

    public boolean isHiddenByYLayer(double viewerY, double targetY, boolean isPlayerTarget) {
        if (!getConfig().getBoolean("hien-tren-an-duoi.bat", true)) return false;
        if (isPlayerTarget && !getConfig().getBoolean("hien-tren-an-duoi.an-nguoi-choi", true)) return false;
        if (!isPlayerTarget && !getConfig().getBoolean("hien-tren-an-duoi.an-ruong-block", true)) return false;

        double splitY = getConfig().getDouble("hien-tren-an-duoi.moc-do-cao-y", 35.0);
        return (viewerY > splitY) != (targetY > splitY);
    }

    public void applyAllowedChunkRadius(Player p) {
        if (!getConfig().getBoolean("hien-tren-an-duoi.bat", true) || p.hasPermission("amberac.bypass")) {
            p.setSendViewDistance(-1);
            return;
        }
        int splitY = getConfig().getInt("hien-tren-an-duoi.moc-do-cao-y", 35);
        boolean isAbove = playerAboveLayerState.computeIfAbsent(p.getUniqueId(), k -> p.getLocation().getBlockY() > splitY);

        int surfaceChunks = Math.max(2, getConfig().getInt("hien-tren-an-duoi.chunk-cho-phep-tren-mat-dat", 6));
        int undergroundChunks = Math.max(2, getConfig().getInt("hien-tren-an-duoi.chunk-cho-phep-duoi-ham", 2));

        int allowedRadius = isAbove ? surfaceChunks : undergroundChunks;
        if (p.getSendViewDistance() != allowedRadius) {
            p.setSendViewDistance(allowedRadius);
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onCrossYThreshold(PlayerMoveEvent e) {
        if (e.getTo() == null || e.getFrom().getBlockY() == e.getTo().getBlockY()) return;
        if (!getConfig().getBoolean("hien-tren-an-duoi.bat", true)) return;

        Player p = e.getPlayer();
        UUID uuid = p.getUniqueId();
        int splitY = getConfig().getInt("hien-tren-an-duoi.moc-do-cao-y", 35);
        int hysteresis = getConfig().getInt("hien-tren-an-duoi.do-tre-do-cao-y", 2);
        int currentY = e.getTo().getBlockY();

        boolean wasAbove = playerAboveLayerState.computeIfAbsent(uuid, k -> e.getFrom().getBlockY() > splitY);
        boolean switched = false;

        if (wasAbove && currentY <= (splitY - hysteresis)) {
            playerAboveLayerState.put(uuid, false);
            switched = true;
        } else if (!wasAbove && currentY >= (splitY + hysteresis)) {
            playerAboveLayerState.put(uuid, true);
            switched = true;
        }

        if (switched) {
            applyAllowedChunkRadius(p);
            if (blockEspListener != null) {
                blockEspListener.clearPlayerCache(uuid);
            }
        }
    }

    private void runAntiPlayerEspTask() {
        boolean wallEspEnabled = getConfig().getBoolean("anti-player-esp.bat", true);
        boolean layerEspEnabled = getConfig().getBoolean("hien-tren-an-duoi.bat", true)
                && getConfig().getBoolean("hien-tren-an-duoi.an-nguoi-choi", true);

        if (!wallEspEnabled && !layerEspEnabled) {
            revealAllHiddenPlayers();
            return;
        }

        double maxViewDist = getConfig().getDouble("anti-player-esp.khoang-cach-toi-da", 64.0);
        double wallHideDist = getConfig().getDouble("anti-player-esp.khoang-cach-xuyen-tuong", 8.0);
        long now = System.currentTimeMillis();
        int rayBudget = 250;

        List<Player> players = new ArrayList<>(Bukkit.getOnlinePlayers());

        for (Player viewer : players) {
            UUID viewerId = viewer.getUniqueId();
            Set<UUID> hiddenSet = hiddenPlayersMap.computeIfAbsent(viewerId, k -> ConcurrentHashMap.newKeySet());

            if (viewer.hasPermission("amberac.bypass")
                    || viewer.getGameMode() == GameMode.SPECTATOR
                    || viewer.getGameMode() == GameMode.CREATIVE) {
                for (UUID targetId : new HashSet<>(hiddenSet)) {
                    Player t = Bukkit.getPlayer(targetId);
                    if (t != null) showEntityModelOnly(viewer, t);
                    hiddenSet.remove(targetId);
                }
                continue;
            }

            Location eye = viewer.getEyeLocation();

            for (Player target : players) {
                if (viewer.equals(target)) continue;
                UUID targetId = target.getUniqueId();

                if (!viewer.getWorld().equals(target.getWorld())) {
                    hiddenSet.remove(targetId);
                    continue;
                }

                String pairKey = viewerId + ":" + targetId;
                boolean occludedNow = false;

                if (!target.isGlowing()) {
                    double dist = eye.distance(target.getLocation());

                    if (layerEspEnabled && isHiddenByYLayer(viewer, target.getLocation().getY(), true)) {
                        if (dist <= 10.0 && !isCompletelyOccluded(viewer, target)) {
                            occludedNow = false;
                        } else {
                            occludedNow = true;
                            lastSeenTimestampMap.remove(pairKey);
                        }
                    } else if (wallEspEnabled) {
                        if (dist > maxViewDist) {
                            occludedNow = true;
                        } else if (dist > wallHideDist && rayBudget > 0) {
                            rayBudget -= 3;
                            occludedNow = isCompletelyOccluded(viewer, target);
                        }
                    }
                }

                if (!occludedNow) {
                    lastSeenTimestampMap.put(pairKey, now);
                    if (hiddenSet.remove(targetId)) {
                        showEntityModelOnly(viewer, target);
                    }
                } else {
                    long lastSeen = lastSeenTimestampMap.getOrDefault(pairKey, 0L);
                    if (now - lastSeen >= 1500L) {
                        if (hiddenSet.add(targetId)) {
                            hideEntityModelKeepTab(viewer, target);
                        }
                    }
                }
            }
        }
    }

    private boolean isCompletelyOccluded(Player viewer, Player target) {
        if (occlusionEngine == null) return false;
        Location eye = viewer.getEyeLocation();
        BoundingBox box = target.getBoundingBox();
        return occlusionEngine.isTargetOccludedAsync(
                viewer.getWorld().getName(),
                eye.getX(), eye.getY(), eye.getZ(),
                box.getMinX(), box.getMinY(), box.getMinZ(),
                box.getMaxX(), box.getMaxY(), box.getMaxZ()
        );
    }

    private void hideEntityModelKeepTab(Player viewer, Player target) {
        viewer.hidePlayer(this, target);
    }

    private void showEntityModelOnly(Player viewer, Player target) {
        viewer.showPlayer(this, target);
    }

    public void revealAllHiddenPlayers() {
        for (Map.Entry<UUID, Set<UUID>> entry : hiddenPlayersMap.entrySet()) {
            Player viewer = Bukkit.getPlayer(entry.getKey());
            if (viewer == null) continue;
            for (UUID targetId : entry.getValue()) {
                Player target = Bukkit.getPlayer(targetId);
                if (target != null) {
                    viewer.showPlayer(this, target);
                }
            }
            entry.getValue().clear();
        }
        hiddenPlayersMap.clear();
        lastSeenTimestampMap.clear();
    }

    public void flag(Player p, String checkName, String detail) {
        if (!getConfig().getBoolean("he-thong.anticheat", true)) return;
        if (p.hasPermission("amberac.bypass")) return;

        int maxPing = getConfig().getInt("chong-ban-nham.ping-toi-da-kiem-tra", 280);
        double minTps = getConfig().getDouble("chong-ban-nham.tps-toi-thieu-kiem-tra", 17.5);
        double currentTps = Bukkit.getTPS()[0];

        if (p.getPing() > maxPing || currentTps < minTps) return;

        UUID uuid = p.getUniqueId();
        int vl = violations.merge(uuid, 1, Integer::sum);

        if (suspectGuiManager != null) {
            suspectGuiManager.recordViolation(p, checkName, detail, vl);
        }

        if (getConfig().getBoolean("he-thong.thong-bao-staff", true)) {
            sendClickableAlert(p, checkName, detail, vl);
        }

        long now = System.currentTimeMillis();
        if (now - logCooldownMap.getOrDefault(uuid, 0L) >= 4000L) {
            logCooldownMap.put(uuid, now);
            logViolationAsync(p.getName(), checkName, detail, vl, p.getPing(), currentTps);
        }

        boolean safeMode = getConfig().getBoolean("chong-ban-nham.che-do-an-toan-khong-ban", true);
        int maxVl = getConfig().getInt("chong-ban-nham.nguong-vi-pham-toi-da", 25);

        if (!safeMode && vl >= maxVl) {
            violations.remove(uuid);
            String cmd = getConfig().getString("chong-ban-nham.lenh-xu-phat", "kick %player% [Amber AntiCheat] Gian lan (%check%)")
                    .replace("%player%", p.getName())
                    .replace("%check%", checkName);
            Bukkit.getScheduler().runTask(this, () -> Bukkit.dispatchCommand(Bukkit.getConsoleSender(), cmd));
        }
    }

    private void logViolationAsync(String playerName, String check, String detail, int vl, int ping, double tps) {
        boolean saveFile = getConfig().getBoolean("he-thong.luu-file-log", true);
        String webhook = getConfig().getString("he-thong.discord-webhook-url", "");

        Bukkit.getScheduler().runTaskAsynchronously(this, () -> {
            String timeStr = logDateFormat.format(new Date());
            String line = String.format("[%s] %s | %s (%s) | VL:%d | Ping:%dms | TPS:%.1f",
                    timeStr, playerName, check, detail, vl, ping, tps);

            if (saveFile) {
                try {
                    File file = new File(getDataFolder(), "violations.log");
                    try (PrintWriter out = new PrintWriter(new FileWriter(file, true))) {
                        out.println(line);
                    }
                } catch (Exception ignored) {
                }
            }

            if (webhook != null && webhook.startsWith("http")) {
                try {
                    HttpURLConnection conn = (HttpURLConnection) URI.create(webhook).toURL().openConnection();
                    conn.setRequestMethod("POST");
                    conn.setRequestProperty("Content-Type", "application/json; charset=UTF-8");
                    conn.setDoOutput(true);
                    String json = "{\"content\": \"**[Amber AntiCheat]** `" + playerName + "` vi phạm **" + check
                            + "** (`" + detail + "`) - **" + vl + " VL** (Ping: `" + ping + "ms`)\"}";
                    try (OutputStream os = conn.getOutputStream()) {
                        os.write(json.getBytes(StandardCharsets.UTF_8));
                    }
                    conn.getResponseCode();
                    conn.disconnect();
                } catch (Exception ignored) {
                }
            }
        });
    }

    @Override
    public void onPluginMessageReceived(String channel, Player player, byte[] message) {
        if (!channel.equals("minecraft:brand") || message == null || message.length <= 1) return;
        String brand = new String(message, 1, message.length - 1, StandardCharsets.UTF_8).trim();
        clientBrands.put(player.getUniqueId(), brand);

        if (!getConfig().getBoolean("kiem-tra-client.bat", true) || player.hasPermission("amberac.bypass")) return;

        String lower = brand.toLowerCase(Locale.ROOT);
        for (String banned : getConfig().getStringList("kiem-tra-client.danh-sach-cam")) {
            if (lower.contains(banned.toLowerCase(Locale.ROOT))) {
                flag(player, "CLIENT-BRAND", "Client cấm: " + brand);
                if (getConfig().getBoolean("kiem-tra-client.tu-dong-kick", false)) {
                    Bukkit.getScheduler().runTask(this, () ->
                            player.kickPlayer(color("&c[Amber AntiCheat] Máy chủ không cho phép sử dụng Client: &e" + brand)));
                }
                break;
            }
        }
    }

    @EventHandler
    public void onJoin(PlayerJoinEvent e) {
        Player p = e.getPlayer();
        quittingPlayers.remove(p.getUniqueId());
        int splitY = getConfig().getInt("hien-tren-an-duoi.moc-do-cao-y", 35);
        playerAboveLayerState.put(p.getUniqueId(), p.getLocation().getBlockY() > splitY);

        Bukkit.getScheduler().runTaskLater(this, () -> {
            if (p.isOnline()) applyAllowedChunkRadius(p);
        }, 5L);
    }

    @EventHandler(priority = EventPriority.LOWEST)
    public void onQuit(PlayerQuitEvent e) {
        UUID uuid = e.getPlayer().getUniqueId();
        quittingPlayers.add(uuid);
        hiddenPlayersMap.remove(uuid);
        playerAboveLayerState.remove(uuid);
        clientBrands.remove(uuid);
        logCooldownMap.remove(uuid);
        if (packetCheckListener != null) {
            packetCheckListener.clearPlayerData(uuid);
        }
    }

    private long getChunkKey(int cx, int cz) {

    private void scanChunk(Chunk chunk) {
        if (chunk.getWorld().getEnvironment() == World.Environment.THE_END) return;
        String worldName = chunk.getWorld().getName();

        Set<BlockPos> set = ConcurrentHashMap.newKeySet();

        for (BlockState state : chunk.getTileEntities()) {
            if (isTargetMaterial(state.getType())) {
                set.add(new BlockPos(worldName, state.getX(), state.getY(), state.getZ()));
            }
        }
        if (!set.isEmpty()) {
            trackedBlocks.computeIfAbsent(worldName, k -> new ConcurrentHashMap<>()).put(key, set);
        }

    private void updateBlocksForAll() {
        boolean wallBlockEsp = plugin.getConfig().getBoolean("anti-block-esp.bat", true);
        boolean layerBlockEsp = plugin.getConfig().getBoolean("hien-tren-an-duoi.bat", true)
                && plugin.getConfig().getBoolean("hien-tren-an-duoi.an-ruong-block", true);

        if (!wallBlockEsp && !layerBlockEsp) {
            restoreAllHiddenBlocks();
            return;
        }

        double minAlwaysShowDist = plugin.getConfig().getDouble("anti-block-esp.khoang-cach-luon-hien", 2.5);
        double minShowDistSq = minAlwaysShowDist * minAlwaysShowDist;
        long now = System.currentTimeMillis();

        for (Player viewer : Bukkit.getOnlinePlayers()) {
            World world = viewer.getWorld();
            if (world.getEnvironment() == World.Environment.THE_END) continue;

            if (viewer.hasPermission("amberac.bypass")
                    || viewer.getGameMode() == GameMode.SPECTATOR
                    || viewer.getGameMode() == GameMode.CREATIVE) {
                revealNearbyForBypass(viewer);
                continue;
            }

            Set<BlockPos> revealedSet = revealedBlocksByPlayer.computeIfAbsent(viewer.getUniqueId(), k -> ConcurrentHashMap.newKeySet());
            Location eye = viewer.getEyeLocation();
            double eyeX = eye.getX(), eyeY = eye.getY(), eyeZ = eye.getZ();
            String worldName = world.getName();

            Iterator<BlockPos> it = revealedSet.iterator();
            while (it.hasNext()) {
                BlockPos pos = it.next();
                if (!pos.world().equals(worldName)) {
                    it.remove();
                    continue;
                }
                double distSq = Math.pow(eyeX - (pos.x() + 0.5), 2)
                        + Math.pow(eyeY - (pos.y() + 0.5), 2)
                        + Math.pow(eyeZ - (pos.z() + 0.5), 2);

                boolean differentLayer = layerBlockEsp && plugin.isHiddenByYLayer(viewer, pos.y(), false);
                if (distSq > MAX_REVEAL_DIST_SQ || differentLayer) {
                    it.remove();
                    viewer.sendBlockChange(new Location(world, pos.x(), pos.y(), pos.z()), getMaskData(world, pos.y()));
                }
            }

            Map<Long, Set<BlockPos>> worldMap = trackedBlocks.get(worldName);
            if (worldMap != null && !worldMap.isEmpty()) {
                int pChunkX = eye.getBlockX() >> 4;
                int pChunkZ = eye.getBlockZ() >> 4;
                int checkedCount = 0;

                outerLoop:
                for (int dx = -2; dx <= 2; dx++) {
                    for (int dz = -2; dz <= 2; dz++) {
                        Set<BlockPos> blocksInChunk = worldMap.get(getChunkKey(pChunkX + dx, pChunkZ + dz));
                        if (blocksInChunk == null || blocksInChunk.isEmpty()) continue;

                        for (BlockPos pos : blocksInChunk) {
                            double distSq = Math.pow(eyeX - (pos.x() + 0.5), 2)
                                    + Math.pow(eyeY - (pos.y() + 0.5), 2)
                                    + Math.pow(eyeZ - (pos.z() + 0.5), 2);

                            if (distSq > MAX_REVEAL_DIST_SQ) continue;
                            if (layerBlockEsp && plugin.isHiddenByYLayer(viewer, pos.y(), false)) continue;

                            if (++checkedCount > MAX_BLOCKS_PER_PLAYER) break outerLoop;

                            Block block = world.getBlockAt(pos.x(), pos.y(), pos.z());
                            if (!isTargetMaterial(block.getType())) {
                                blocksInChunk.remove(pos);
                                revealedSet.remove(pos);
                                continue;
                            }

                            boolean canSee = true;
                            if (wallBlockEsp && distSq >= minShowDistSq && occlusionEngine != null) {
                                canSee = !occlusionEngine.isTargetOccludedAsync(
                                        worldName, eyeX, eyeY, eyeZ,
                                        pos.x(), pos.y(), pos.z(),
                                        pos.x() + 1.0, pos.y() + 1.0, pos.z() + 1.0
                                );
                            }

                            String timeKey = viewer.getUniqueId() + ":" + pos.x() + "," + pos.y() + "," + pos.z();
                            Location blockLoc = block.getLocation();

                            if (canSee) {
                                lastSeenBlockTime.put(timeKey, now);
                                if (revealedSet.add(pos)) {
                                    viewer.sendBlockChange(blockLoc, block.getBlockData());
                                }
                            } else {
                                if (revealedSet.contains(pos)) {
                                    long lastSeen = lastSeenBlockTime.getOrDefault(timeKey, 0L);
                                    if (now - lastSeen >= 1500L) {
                                        revealedSet.remove(pos);
                                        viewer.sendBlockChange(blockLoc, getMaskData(world, pos.y()));
                                    }
                                }
                            }
                        }
                    }
                }
            }

            updateBaseEntitiesForViewer(viewer, wallBlockEsp, layerBlockEsp);
        }
    }

    private void updateBaseEntitiesForViewer(Player viewer, boolean wallEsp, boolean layerEsp) {
        World world = viewer.getWorld();
        Location eye = viewer.getEyeLocation();
        int pChunkX = eye.getBlockX() >> 4;
        int pChunkZ = eye.getBlockZ() >> 4;

        for (int dx = -2; dx <= 2; dx++) {
            for (int dz = -2; dz <= 2; dz++) {
                if (!world.isChunkLoaded(pChunkX + dx, pChunkZ + dz)) continue;
                Chunk chunk = world.getChunkAt(pChunkX + dx, pChunkZ + dz);

                for (Entity ent : chunk.getEntities()) {
                    if (!isBaseEntity(ent)) continue;

                    Location entLoc = ent.getLocation();
                    boolean shouldHide = false;

                    if (layerEsp && plugin.isHiddenByYLayer(viewer, entLoc.getY(), false)) {
                        shouldHide = true;
                    } else if (wallEsp) {
                        double distSq = eye.distanceSquared(entLoc);
                        if (distSq > MAX_REVEAL_DIST_SQ) {
                            shouldHide = true;
                        } else if (distSq > 16.0 && occlusionEngine != null) {
                            BoundingBox box = ent.getBoundingBox();
                            shouldHide = occlusionEngine.isTargetOccludedAsync(
                                    world.getName(), eye.getX(), eye.getY(), eye.getZ(),
                                    box.getMinX(), box.getMinY(), box.getMinZ(),
                                    box.getMaxX(), box.getMaxY(), box.getMaxZ()
                            );
                        }
                    }

                    if (shouldHide) {
                        viewer.hideEntity(plugin, ent);
                    } else {
                        viewer.showEntity(plugin, ent);
                    }
                }
            }
        }
    }

    private void revealNearbyForBypass(Player viewer) {
        World world = viewer.getWorld();
        int cx = viewer.getLocation().getBlockX() >> 4;
        int cz = viewer.getLocation().getBlockZ() >> 4;

        Map<Long, Set<BlockPos>> worldMap = trackedBlocks.get(world.getName());
        Set<BlockPos> rev = revealedBlocksByPlayer.computeIfAbsent(viewer.getUniqueId(), k -> ConcurrentHashMap.newKeySet());

        for (int dx = -2; dx <= 2; dx++) {
            for (int dz = -2; dz <= 2; dz++) {
                if (worldMap != null) {
                    Set<BlockPos> set = worldMap.get(getChunkKey(cx + dx, cz + dz));
                    if (set != null) {
                        for (BlockPos pos : set) {
                            if (rev.add(pos)) {
                                Block b = world.getBlockAt(pos.x(), pos.y(), pos.z());
                                viewer.sendBlockChange(b.getLocation(), b.getBlockData());
                            }
                        }
                    }
                }
                if (world.isChunkLoaded(cx + dx, cz + dz)) {
                    Chunk chunk = world.getChunkAt(cx + dx, cz + dz);
                    for (Entity ent : chunk.getEntities()) {
                        if (isBaseEntity(ent)) {
                            viewer.showEntity(plugin, ent);
                        }
                    }
                }
            }
        }
    }

    private BlockData getMaskData(World world, int y) {
        if (world.getEnvironment() == World.Environment.NETHER) return netherrackData;
        return y < 0 ? deepslateData : stoneData;
    }

    public void restoreAllHiddenBlocks() {
        for (Player p : Bukkit.getOnlinePlayers()) {
            revealNearbyForBypass(p);
        }
        revealedBlocksByPlayer.clear();
        lastSeenBlockTime.clear();
    }

                "ANVIL", "CHIPPED_ANVIL", "DAMAGED_ANVIL"

                    || blockEspList.contains(name)
                    || pieChartExtras.contains(name)
                    || name.endsWith("_BED")
                    || name.endsWith("_SIGN");

            if (isNether) newNetherOres[id] = true;
            if (isStorageOrTile) newStorage[id] = true;
            if (isOw || isNether) {
                newOreMat[mat.ordinal()] = true;
            }

        this.netherOreTable = newNetherOres;

        if (player == null) return false;
        GameMode gm = player.getGameMode();
        if (gm == GameMode.SPECTATOR || gm == GameMode.CREATIVE || player.hasPermission("amberac.bypass")) {
            return false;
        }

        World world = player.getWorld();
        World.Environment env = world.getEnvironment();
        if (env == World.Environment.THE_END) return false;

        boolean xrayEnabled = plugin.getConfig().getBoolean("anti-xray.bat", true);
        boolean hideStorageOnChunk = plugin.getConfig().getBoolean("anti-block-esp.bat", true);
        boolean layerEnabled = plugin.getConfig().getBoolean("hien-tren-an-duoi.bat", true);
        boolean pieChartProtect = layerEnabled && plugin.getConfig().getBoolean("hien-tren-an-duoi.chong-pie-chart-f3", true);

        if (!xrayEnabled && !hideStorageOnChunk && !layerEnabled) return false;

        int maxScanY = plugin.getConfig().getInt("anti-xray.do-cao-quet-toi-da", 64);

        Column column = wrapper.getColumn();
        BaseChunk[] sections = column.getChunks();
        if (sections == null || sections.length == 0) return false;

        int minY = world.getMinHeight();
        boolean isNether = (env == World.Environment.NETHER);
        boolean[] oreTable = isNether ? netherOreTable : overworldOreTable;
        boolean modified = false;

        long[] transparentMask = new long[64];

        for (int secIdx = 0; secIdx < sections.length; secIdx++) {
            int sectionBaseY = minY + (secIdx << 4);
            if (sectionBaseY > maxScanY && !hideStorageOnChunk && !pieChartProtect) break;

            BaseChunk section = sections[secIdx];
            if (section == null || section.isEmpty()) continue;

            boolean maskBuilt = false;

            for (int y = 0; y < 16; y++) {
                int worldY = sectionBaseY + y;
                boolean scanOreHere = xrayEnabled && worldY <= maxScanY;

                if (!scanOreHere && !hideStorageOnChunk && !pieChartProtect) continue;

                int defaultMaskId = isNether ? netherrackId : (worldY < 0 ? deepslateId : stoneId);

                for (int z = 0; z < 16; z++) {
                    for (int x = 0; x < 16; x++) {
                        int currentId = getBlockStateId(section, x, y, z);
                        if (currentId <= 0 || currentId >= MAX_STATE_ID) continue;

                        if ((hideStorageOnChunk || pieChartProtect) && storageBlockTable[currentId]) {
                            section.set(x, y, z, defaultMaskId);
                            modified = true;
                            continue;
                        }

                        if (scanOreHere && oreTable[currentId]) {
                            if (!maskBuilt) {
                                buildTransparentMask(section, transparentMask);
                                maskBuilt = true;
                            }
                            if (!isExposedFast(sections, secIdx, transparentMask, x, y, z)) {
                                section.set(x, y, z, defaultMaskId);
                                modified = true;
                            }
                        }
                    }
                }
            }
        }

        return modified;
    }

    private void buildTransparentMask(BaseChunk section, long[] mask) {
        Arrays.fill(mask, 0L);
        boolean[] trans = this.transparentTable;
        for (int y = 0; y < 16; y++) {
            int yOff = y << 8;
            for (int z = 0; z < 16; z++) {
                int zOff = yOff | (z << 4);
                for (int x = 0; x < 16; x++) {
                    int id = getBlockStateId(section, x, y, z);
                    if (id >= 0 && id < MAX_STATE_ID && trans[id]) {
                        int idx = zOff | x;
                        mask[idx >> 6] |= (1L << (idx & 63));
                    }
                }
            }
        }
    }

    private boolean isExposedFast(BaseChunk[] sections, int secIdx, long[] mask, int x, int y, int z) {
        if (x == 0 || x == 15 || z == 0 || z == 15) return true;

        if (isMaskBitSet(mask, x - 1, y, z)) return true;
        if (isMaskBitSet(mask, x + 1, y, z)) return true;
        if (isMaskBitSet(mask, x, y, z - 1)) return true;
        if (isMaskBitSet(mask, x, y, z + 1)) return true;

        if (y > 0) {
            if (isMaskBitSet(mask, x, y - 1, z)) return true;
        } else if (secIdx > 0) {
            BaseChunk below = sections[secIdx - 1];
            if (below == null || below.isEmpty()) return true;
            int id = getBlockStateId(below, x, 15, z);
            if (id >= 0 && id < MAX_STATE_ID && transparentTable[id]) return true;
        }

        if (y < 15) {
            if (isMaskBitSet(mask, x, y + 1, z)) return true;
        } else if (secIdx + 1 < sections.length) {
            BaseChunk above = sections[secIdx + 1];
            if (above == null || above.isEmpty()) return true;
            int id = getBlockStateId(above, x, 0, z);
            if (id >= 0 && id < MAX_STATE_ID && transparentTable[id]) return true;
        }

        return false;
    }

    private boolean isMaskBitSet(long[] mask, int x, int y, int z) {
        int idx = (y << 8) | (z << 4) | x;
        return (mask[idx >> 6] & (1L << (idx & 63))) != 0L;
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onBlockBreak(BlockBreakEvent e) {
        revealAdjacentBlocks(e.getBlock());
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onEntityExplode(EntityExplodeEvent e) {
        for (Block b : e.blockList()) revealAdjacentBlocks(b);
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onBlockExplode(BlockExplodeEvent e) {
        for (Block b : e.blockList()) revealAdjacentBlocks(b);
    }

    public void revealAdjacentBlocks(Block center) {
        if (!plugin.getConfig().getBoolean("anti-xray.bat", true)) return;

        World world = center.getWorld();
        if (world.getEnvironment() == World.Environment.THE_END) return;

        int by = center.getY();
        int maxScanY = plugin.getConfig().getInt("anti-xray.do-cao-quet-toi-da", 64);
        if (by > maxScanY + 1) return;

        int bx = center.getX();
        int bz = center.getZ();
        double centerX = bx + 0.5;
        double centerY = by + 0.5;
        double centerZ = bz + 0.5;
        boolean[] ores = this.oreMaterialTable;

        for (int[] offset : ADJACENT_OFFSETS) {
            Block adj = world.getBlockAt(bx + offset[0], by + offset[1], bz + offset[2]);
            Material type = adj.getType();
            if (!ores[type.ordinal()]) continue;

            Location adjLoc = adj.getLocation();
            BlockData realData = adj.getBlockData();
            for (Player p : world.getPlayers()) {
                if (plugin.isHiddenByYLayer(p, adjLoc.getY(), false)) continue;
                Location pLoc = p.getLocation();
                double dx = pLoc.getX() - centerX;
                double dy = pLoc.getY() - centerY;
                double dz = pLoc.getZ() - centerZ;
                if ((dx * dx + dy * dy + dz * dz) <= 576.0) {
                    p.sendBlockChange(adjLoc, realData);
                }
            }
        }
    }
}

package vn.amber.anticheat;

import org.bukkit.Bukkit;
import org.bukkit.Chunk;
import org.bukkit.ChunkSnapshot;
import org.bukkit.Material;

import java.util.BitSet;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class AsyncOcclusionEngine implements Listener {

    private static class ChunkBitMask {
        final int minY;
        final int height;
        final BitSet solidBits;

        ChunkBitMask(int minY, int maxY) {
            this.minY = minY;
            this.height = maxY - minY;
            this.solidBits = new BitSet(16 * 16 * this.height);
        }

        int getIndex(int localX, int worldY, int localZ) {
            int relY = worldY - minY;
            if (relY < 0 || relY >= height) return -1;
            return (relY << 8) | (localZ << 4) | (localX & 15);
        }

        synchronized void setSolid(int localX, int worldY, int localZ, boolean solid) {
            int idx = getIndex(localX, worldY, localZ);
            if (idx >= 0) {
                solidBits.set(idx, solid);
            }
        }

        synchronized boolean isSolid(int localX, int worldY, int localZ) {
            int idx = getIndex(localX, worldY, localZ);
            return idx >= 0 && solidBits.get(idx);
        }
    }

    private final AmberAntiCheat plugin;
    private final Map<String, Map<Long, ChunkBitMask>> worldChunkMasks = new ConcurrentHashMap<>();

    public AsyncOcclusionEngine(AmberAntiCheat plugin) {
        this.plugin = plugin;
        for (World world : Bukkit.getWorlds()) {
            for (Chunk chunk : world.getLoadedChunks()) {
                indexChunkAsync(chunk);
            }
        }
    }

    private long getChunkKey(int cx, int cz) {
        return (((long) cx) << 32) | (cz & 0xffffffffL);
    }

    private void indexChunkAsync(Chunk chunk) {
        World world = chunk.getWorld();
        String worldName = world.getName();
        int minY = world.getMinHeight();
        int maxY = world.getMaxHeight();
        long key = getChunkKey(chunk.getX(), chunk.getZ());
        ChunkSnapshot snapshot = chunk.getChunkSnapshot(false, false, false);

        Bukkit.getScheduler().runTaskAsynchronously(plugin, () -> {
            ChunkBitMask mask = new ChunkBitMask(minY, maxY);
            for (int y = minY; y < maxY; y++) {
                for (int z = 0; z < 16; z++) {
                    for (int x = 0; x < 16; x++) {
                        Material mat = snapshot.getBlockType(x, y, z);
                        if (plugin.isFullSolidBlock(mat)) {
                            mask.setSolid(x, y, z, true);
                        }
                    }
                }
            }
            worldChunkMasks.computeIfAbsent(worldName, k -> new ConcurrentHashMap<>()).put(key, mask);
        });
    }

    @EventHandler(priority = EventPriority.MONITOR)
    public void onChunkLoad(ChunkLoadEvent e) {
        indexChunkAsync(e.getChunk());
    }

    @EventHandler(priority = EventPriority.MONITOR)
    public void onChunkUnload(ChunkUnloadEvent e) {
        Map<Long, ChunkBitMask> map = worldChunkMasks.get(e.getWorld().getName());
        if (map != null) {
            map.remove(getChunkKey(e.getChunk().getX(), e.getChunk().getZ()));
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onBlockPlace(BlockPlaceEvent e) {
        updateSingleBlock(e.getBlockPlaced(), e.getBlockPlaced().getType());
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onBlockBreak(BlockBreakEvent e) {
        updateSingleBlock(e.getBlock(), Material.AIR);
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onEntityExplode(EntityExplodeEvent e) {
        for (Block b : e.blockList()) {
            updateSingleBlock(b, Material.AIR);
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onBlockExplode(BlockExplodeEvent e) {
        for (Block b : e.blockList()) {
            updateSingleBlock(b, Material.AIR);
        }
    }

    private void updateSingleBlock(Block b, Material newType) {
        Map<Long, ChunkBitMask> map = worldChunkMasks.get(b.getWorld().getName());
        if (map == null) return;
        ChunkBitMask mask = map.get(getChunkKey(b.getX() >> 4, b.getZ() >> 4));
        if (mask != null) {
            mask.setSolid(b.getX() & 15, b.getY(), b.getZ() & 15, plugin.isFullSolidBlock(newType));
        }
    }

    public boolean isTargetOccludedAsync(
            String worldName,
            double eyeX, double eyeY, double eyeZ,
            double minX, double minY, double minZ,
            double maxX, double maxY, double maxZ) {

        double midX = (minX + maxX) * 0.5;
        double midY = (minY + maxY) * 0.5;
        double midZ = (minZ + maxZ) * 0.5;

        double[][] points = {
                {midX, midY, midZ},
                {midX, maxY - 0.1, midZ},
                {midX, minY + 0.1, midZ},
                {minX + 0.1, midY, minZ + 0.1},
                {maxX - 0.1, midY, maxZ - 0.1}
        };

        for (double[] pt : points) {
            if (!isRayBlockedAsync(worldName, eyeX, eyeY, eyeZ, pt[0], pt[1], pt[2])) {
                return false;
            }
        }
        return true;
    }

    public boolean isRayBlockedAsync(String worldName, double x0, double y0, double z0, double x1, double y1, double z1) {
        Map<Long, ChunkBitMask> worldMap = worldChunkMasks.get(worldName);
        if (worldMap == null) return false;

        double dx = x1 - x0;
        double dy = y1 - y0;
        double dz = z1 - z0;
        double dist = Math.sqrt(dx * dx + dy * dy + dz * dz);
        if (dist < 0.6) return false;

        int steps = (int) Math.ceil(dist * 2.5);
        double stepX = dx / steps;
        double stepY = dy / steps;
        double stepZ = dz / steps;

        int endBlockX = (int) Math.floor(x1);
        int endBlockY = (int) Math.floor(y1);
        int endBlockZ = (int) Math.floor(z1);

        double curX = x0 + stepX;
        double curY = y0 + stepY;
        double curZ = z0 + stepZ;

        long cachedChunkKey = Long.MIN_VALUE;
        ChunkBitMask currentMask = null;

        for (int i = 1; i < steps; i++) {
            int bx = (int) Math.floor(curX);
            int by = (int) Math.floor(curY);
            int bz = (int) Math.floor(curZ);

            if (bx == endBlockX && by == endBlockY && bz == endBlockZ) {
                return false;
            }

            long ck = getChunkKey(bx >> 4, bz >> 4);
            if (ck != cachedChunkKey) {
                cachedChunkKey = ck;
                currentMask = worldMap.get(ck);
            }

            if (currentMask != null && currentMask.isSolid(bx & 15, by, bz & 15)) {
                return true;
            }

            curX += stepX;
            curY += stepY;
            curZ += stepZ;
        }
        return false;
    }
}

package vn.amber.anticheat;

import com.github.retrooper.packetevents.event.PacketListener;
import com.github.retrooper.packetevents.event.PacketReceiveEvent;
import com.github.retrooper.packetevents.event.PacketSendEvent;

public class PacketCheckListener implements PacketListener {

    private final AmberAntiCheat plugin;
    private final AntiXrayManager antiXrayManager;

    private final Map<UUID, Long> timerWindowStart = new ConcurrentHashMap<>();
    private final Map<UUID, Integer> timerPacketCount = new ConcurrentHashMap<>();
    private final Map<UUID, Float> lastPitchMap = new ConcurrentHashMap<>();
    private final Map<UUID, Float> lastDeltaPitchMap = new ConcurrentHashMap<>();
    private final Map<UUID, Integer> gcdStreakMap = new ConcurrentHashMap<>();

    private final Map<UUID, Long> pendingVelocityTime = new ConcurrentHashMap<>();
    private final Map<UUID, Double> velocityStartY = new ConcurrentHashMap<>();
    private final Map<UUID, Integer> velocityFailStreak = new ConcurrentHashMap<>();

    public PacketCheckListener(AmberAntiCheat plugin, AntiXrayManager antiXrayManager) {
        this.plugin = plugin;
        this.antiXrayManager = antiXrayManager;
    }

    public void clearPlayerData(UUID uuid) {
        timerWindowStart.remove(uuid);
        timerPacketCount.remove(uuid);
        lastPitchMap.remove(uuid);
        lastDeltaPitchMap.remove(uuid);
        gcdStreakMap.remove(uuid);
        pendingVelocityTime.remove(uuid);
        velocityStartY.remove(uuid);
        velocityFailStreak.remove(uuid);
    }

    @Override
    public void onPacketSend(PacketSendEvent event) {
        if (!(event.getPlayer() instanceof Player player)) return;

        // 1. Giữ nguyên tên trên bảng Tab khi ẩn mô hình 3D bằng Anti-PlayerESP
        if (event.getPacketType() == PacketType.Play.Server.PLAYER_INFO_REMOVE) {
            WrapperPlayServerPlayerInfoRemove removePacket = new WrapperPlayServerPlayerInfoRemove(event);
            List<UUID> uuids = new ArrayList<>(removePacket.getProfileIds());
            boolean modified = uuids.removeIf(targetId -> {
                Player target = Bukkit.getPlayer(targetId);
                return target != null && target.isOnline() && !plugin.isPlayerQuitting(targetId);
            });
            if (modified) {
                if (uuids.isEmpty()) {
                    event.setCancelled(true);
                } else {
                    removePacket.setProfileIds(uuids);
                    event.markForReEncode(true);
                }
            }
            return;
        }

        boolean bypassVisuals = player.hasPermission("amberac.bypass")
                || player.getGameMode() == GameMode.SPECTATOR
                || player.getGameMode() == GameMode.CREATIVE;

        // 2. Chặn gói NBT của TileEntity khác tầng Y (Chống lộ căn cứ qua Shift+F3 Pie-Chart)
        if (event.getPacketType() == PacketType.Play.Server.BLOCK_ENTITY_DATA) {
            if (!bypassVisuals
                    && plugin.getConfig().getBoolean("hien-tren-an-duoi.bat", true)
                    && plugin.getConfig().getBoolean("hien-tren-an-duoi.chong-pie-chart-f3", true)) {
                WrapperPlayServerBlockEntityData bePacket = new WrapperPlayServerBlockEntityData(event);
                int blockY = bePacket.getPosition().getY();
                if (plugin.isHiddenByYLayer(player, blockY, false)) {
                    event.setCancelled(true);
                }
            }
            return;
        }

        // 3. Chặn âm thanh phát ra từ tầng bị ẩn (Chống SoundESP)
        if (event.getPacketType() == PacketType.Play.Server.SOUND_EFFECT) {
            if (!bypassVisuals
                    && plugin.getConfig().getBoolean("hien-tren-an-duoi.bat", true)
                    && plugin.getConfig().getBoolean("hien-tren-an-duoi.chan-am-thanh-khac-tang", true)) {
                WrapperPlayServerSoundEffect soundPacket = new WrapperPlayServerSoundEffect(event);
                double soundY = soundPacket.getEffectPosition().getY() / 8.0;
                if (plugin.isHiddenByYLayer(player, soundY, false)) {
                    event.setCancelled(true);
                }
            }
            return;
        }

        // 4. Chặn hạt hiệu ứng (Particle) phát ra từ tầng bị ẩn
        if (event.getPacketType() == PacketType.Play.Server.PARTICLE) {
            if (!bypassVisuals && plugin.getConfig().getBoolean("hien-tren-an-duoi.bat", true)) {
                WrapperPlayServerParticle particlePacket = new WrapperPlayServerParticle(event);
                double particleY = particlePacket.getPosition().getY();
                if (plugin.isHiddenByYLayer(player, particleY, false)) {
                    event.setCancelled(true);
                }
            }
            return;
        }

        // 5. Ghi nhận thời điểm gửi gói đẩy lùi (Velocity) để bắt Anti-Knockback
        if (event.getPacketType() == PacketType.Play.Server.ENTITY_VELOCITY) {
            WrapperPlayServerEntityVelocity vel = new WrapperPlayServerEntityVelocity(event);
            if (vel.getEntityId() == player.getEntityId() && vel.getVelocity().getY() > 0.25) {
                UUID uuid = player.getUniqueId();
                pendingVelocityTime.put(uuid, System.currentTimeMillis());
                velocityStartY.put(uuid, player.getLocation().getY());
            }
            return;
        }

        // 6. Xử lý gói CHUNK_DATA (Anti-Xray 1 Mode + Khóa Rương/Máy móc thành Đá)
        if (event.getPacketType() == PacketType.Play.Server.CHUNK_DATA) {
            if (!bypassVisuals && antiXrayManager != null) {
                try {
                    WrapperPlayServerChunkData chunkData = new WrapperPlayServerChunkData(event);
                    if (antiXrayManager.obfuscateChunkPacket(player, chunkData)) {
                        event.markForReEncode(true);
                    }
                } catch (Exception ignored) {
                }
            }
        }
    }

    @Override
    public void onPacketReceive(PacketReceiveEvent event) {
        if (!(event.getPlayer() instanceof Player player)) return;
        if (!plugin.getConfig().getBoolean("he-thong.anticheat", true)) return;
        if (player.hasPermission("amberac.bypass")) return;

        UUID uuid = player.getUniqueId();
        if (plugin.getSuspectGuiManager() != null && plugin.getSuspectGuiManager().isFrozen(uuid)) {
            return;
        }

        if (WrapperPlayClientPlayerFlying.isFlying(event.getPacketType())) {
            long now = System.currentTimeMillis();

            // 1. Check Timer (Tua nhanh tốc độ game)
            long start = timerWindowStart.computeIfAbsent(uuid, k -> now);
            int count = timerPacketCount.merge(uuid, 1, Integer::sum);
            if (now - start >= 1000L) {
                int maxPackets = plugin.getConfig().getInt("checks.movement.timer-max-packets-per-sec", 26);
                if (count > maxPackets && !player.isInsideVehicle() && !player.isGliding()) {
                    int finalCount = count;
                    Bukkit.getScheduler().runTask(plugin, () ->
                            plugin.flag(player, "TIMER", finalCount + " gói/giây"));
                }
                timerWindowStart.put(uuid, now);
                timerPacketCount.put(uuid, 0);
            }

            // 2. Check Velocity (Anti-Knockback)
            Long velTime = pendingVelocityTime.get(uuid);
            if (velTime != null) {
                long elapsed = now - velTime;
                double currentY = player.getLocation().getY();
                if (event.getPacketType() == PacketType.Play.Client.PLAYER_POSITION) {
                    currentY = new WrapperPlayClientPlayerPosition(event).getPosition().getY();
                } else if (event.getPacketType() == PacketType.Play.Client.PLAYER_POSITION_AND_ROTATION) {
                    currentY = new WrapperPlayClientPlayerPositionAndRotation(event).getPosition().getY();
                }

                double startY = velocityStartY.getOrDefault(uuid, currentY);
                if (currentY - startY > 0.08) {
                    pendingVelocityTime.remove(uuid);
                    velocityFailStreak.put(uuid, 0);
                } else if (elapsed > Math.max(350L, player.getPing() + 180L)) {
                    pendingVelocityTime.remove(uuid);
                    if (!player.isInsideVehicle() && !player.isSwimming() && !player.isClimbing()) {
                        int streak = velocityFailStreak.merge(uuid, 1, Integer::sum);
                        if (streak >= 3) {
                            velocityFailStreak.put(uuid, 0);
                            Bukkit.getScheduler().runTask(plugin, () ->
                                    plugin.flag(player, "VELOCITY", "Không nhận lực đẩy lùi (Anti-KB)"));
                        }
                    }
                }
            }

            // 3. Check Aim-GCD (Aimbot khóa tâm)
            float pitch = Float.NaN;
            if (event.getPacketType() == PacketType.Play.Client.PLAYER_ROTATION) {
                pitch = new WrapperPlayClientPlayerRotation(event).getPitch();
            } else if (event.getPacketType() == PacketType.Play.Client.PLAYER_POSITION_AND_ROTATION) {
                pitch = new WrapperPlayClientPlayerPositionAndRotation(event).getPitch();
            }

            if (!Float.isNaN(pitch) && Math.abs(pitch) < 89.0f) {
                Float prevPitch = lastPitchMap.put(uuid, pitch);
                if (prevPitch != null) {
                    float deltaPitch = Math.abs(pitch - prevPitch);
                    if (deltaPitch > 0.5f && deltaPitch < 25.0f) {
                        Float prevDelta = lastDeltaPitchMap.put(uuid, deltaPitch);
                        if (prevDelta != null) {
                            long expandedCurr = (long) (deltaPitch * 16777216.0f);
                            long expandedPrev = (long) (prevDelta * 16777216.0f);
                            long gcd = computeGcd(expandedCurr, expandedPrev);
                            if (gcd > 0 && gcd < 131072L) {
                                int streak = gcdStreakMap.merge(uuid, 1, Integer::sum);
                                if (streak >= 14) {
                                    gcdStreakMap.put(uuid, 0);
                                    Bukkit.getScheduler().runTask(plugin, () ->
                                            plugin.flag(player, "AIM-GCD", "Khóa tâm bất thường"));
                                }
                            } else {
                                gcdStreakMap.computeIfPresent(uuid, (k, v) -> v > 0 ? v - 1 : 0);
                            }
                        }
                    }
                }
            }
        }
    }

    private long computeGcd(long a, long b) {
        while (b != 0) {
            long t = b;
            b = a % b;
            a = t;
        }
        return a;
    }
}

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onFrozenMove(PlayerMoveEvent e) {
        Player p = e.getPlayer();
        if (!frozenPlayers.contains(p.getUniqueId())) return;

        p.setFallDistance(0f);
        Location from = e.getFrom();
        Location to = e.getTo();
        if (to == null) return;

        // Khóa di chuyển ngang (X, Z) và nhảy lên (Y > from.Y), nhưng cho phép rơi chạm đất để không bị Vanilla kick Fly
        if (from.getX() != to.getX() || from.getZ() != to.getZ() || to.getY() > from.getY()) {
            Location locked = from.clone();
            locked.setY(Math.min(from.getY(), to.getY()));
            locked.setYaw(to.getYaw());
            locked.setPitch(to.getPitch());
            e.setTo(locked);
        }
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onFrozenCommand(PlayerCommandPreprocessEvent e) {
        Player p = e.getPlayer();
        if (!frozenPlayers.contains(p.getUniqueId())) return;

        String msg = e.getMessage().toLowerCase(Locale.ROOT);
        // Chỉ cho phép nhắn tin riêng trả lời Staff (/msg, /r, /w, /tell)
        if (!msg.startsWith("/msg ") && !msg.startsWith("/r ") && !msg.startsWith("/w ") && !msg.startsWith("/tell ")) {
            e.setCancelled(true);
            p.sendMessage(color("&8[&6&lAmber AntiCheat&8] &cBạn không thể dùng lệnh khi đang bị đóng băng kiểm tra!"));
        }
    }

    @EventHandler
    public void onSuspectQuit(PlayerQuitEvent e) {
        Player p = e.getPlayer();
        UUID uuid = p.getUniqueId();

        SuspectProfile prof = suspectMap.get(uuid);
        if (prof != null) {
            prof.quitTimestamp = System.currentTimeMillis();
        }

        if (frozenPlayers.remove(uuid)) {
            String alert = color("&4&l[BÁO ĐỘNG] &e&l" + p.getName() + " &c&lĐÃ THOÁT GAME KHI ĐANG BỊ ĐÓNG BĂNG KIỂM TRA!");
            for (Player online : Bukkit.getOnlinePlayers()) {
                if (online.hasPermission("amberac.staff") || online.hasPermission("amberac.admin")) {
                    online.sendMessage(alert);
                    online.playSound(online.getLocation(), Sound.ENTITY_WITHER_SPAWN, 0.8f, 1.2f);
                }
            }
            plugin.getLogger().warning("[FREEZE-QUIT] " + p.getName() + " da thoat game khi dang bi dong bang!");
        }

        staffPageMap.remove(uuid);
        staffFilterMap.remove(uuid);
        staffDisplayedList.remove(uuid);
        staffSelectedSuspect.remove(uuid);
        invseeStaffSet.remove(uuid);
    }

    private ItemStack createSimpleItem(Material mat, String title, List<String> loreLines) {
        ItemStack item = new ItemStack(mat);
        ItemMeta meta = item.getItemMeta();
        if (meta != null) {
            meta.setDisplayName(color(title));
            List<String> colored = new ArrayList<>();
            for (String line : loreLines) colored.add(color(line));
            meta.setLore(colored);
            meta.addItemFlags(ItemFlag.HIDE_ATTRIBUTES);
            item.setItemMeta(meta);
        }
        return item;
    }

    private String color(String text) {
        return ChatColor.translateAlternateColorCodes('&', text);
    }
