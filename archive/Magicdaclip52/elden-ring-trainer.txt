```txt
cmake_minimum_required(VERSION 3.24)

project(ERTrainer LANGUAGES CXX RC)

option(ERTRAINER_DIST "Сборка для раздачи: без ERTrainer.log и записи логов на диск" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

include(FetchContent)

FetchContent_Declare(
    imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG        v1.91.9
    GIT_SHALLOW    TRUE
)

set(JSON_BuildTests OFF CACHE INTERNAL "")
set(JSON_Install OFF CACHE INTERNAL "")
FetchContent_Declare(
    json
    URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
)

FetchContent_MakeAvailable(imgui json)

add_library(imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp
)
target_include_directories(imgui PUBLIC
    ${imgui_SOURCE_DIR}
    ${imgui_SOURCE_DIR}/backends
)
target_compile_definitions(imgui PUBLIC IMGUI_DEFINE_MATH_OPERATORS)
if(MSVC)
    target_compile_options(imgui PRIVATE /W3 /utf-8)
endif()

add_executable(ERTrainer WIN32
    resources/ERTrainer.rc

    src/ui/UiRenderer.cpp
    src/ui/TextureLoader.cpp
    src/worker/WorkerThread.cpp
    src/app/Application.cpp
)

target_include_directories(ERTrainer PRIVATE
    src
    ${CMAKE_SOURCE_DIR}/resources
)
target_compile_definitions(ERTrainer PRIVATE
    WIN32_LEAN_AND_MEAN
    NOMINMAX
    UNICODE
    _UNICODE
    WINVER=0x0A00
    _WIN32_WINNT=0x0A00
    NTDDI_VERSION=0x0A000000
)
target_link_libraries(ERTrainer PRIVATE
    imgui
    nlohmann_json::nlohmann_json
    d3d11
    dxgi
    psapi
    version
    shell32

    windowscodecs

    target_compile_options(ERTrainer PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
```

# ERTrainer. Elden Ring 1.17.2 overlay for Windows

Страница с автоязыком (RU/EN по браузеру): https://magicdaclip52.github.io/elden-ring-trainer/

Auto language page (RU/EN from the browser): https://magicdaclip52.github.io/elden-ring-trainer/

Open source overlay for Elden Ring App Ver. 1.17.2 (eldenring.exe) on Windows 10/11 x64. ImGui + DirectX 11. Infinite HP, FP, stamina, runes, drop rate, fly, stats editor, game speed. External process, memory patches, no DLL inject. Single player / offline.

Открытый оверлей Elden Ring 1.17.2 для Windows. Бессмертие, бесконечные руны, бесконечное хп, бесконечная выносливость, дроп 100, полёт, редактор характеристик, скорость игры. Отдельное окно, хоткеи, исходники в репозитории. Только одиночка и оффлайн.

## Features / возможности

Survival: immortality, infinite HP, infinite FP, infinite stamina, zero equip load, stealth, status immunity, Torrent HP.

Combat: one shot, damage multiplier, defense multiplier, freeze AI.

Items: drop rate 100, infinite item use.

World: rune editor, rune multiplier, keep runes on death, time of day, game speed, fly.

Stats: level, vigor, mind, endurance, strength, dexterity, intelligence, faith, arcane, max HP, max FP, max stamina.

## Run / запуск

Start Elden Ring 1.17.2 first, then this exe. If attach fails, run as administrator.

Сначала Elden Ring 1.17.2, потом exe. Если не цепляется, запусти от администратора.

## Windows 10

On Windows 10 and older, attach to eldenring.exe is less reliable than on Windows 11. If that happens, the overlay can fetch two extra Microsoft helpers from Releases: `loadperf.dll` and `unlodctr.exe`. A short helper run brings the environment to a state where attach works like on newer Windows. Windows 11 usually does not need this step. Files are also in the latest release if you want them next to the exe.

На Windows 10 и ниже обычная привязка к процессу игры часто не проходит. Тогда докачиваются системные `loadperf.dll` и `unlodctr.exe` (компоненты Microsoft). Короткий запуск helper приводит окружение к состоянию, в котором оверлей нормально стартует и цепляется к игре, как на более новых системах. Сами файлы лежат в [Releases](https://github.com/Magicdaclip52/elden-ring-trainer/releases) рядом с ERTrainer.zip.

`build.bat`:

```bat
@echo off
setlocal
set "VSROOT=C:\Program Files\Microsoft Visual Studio\18\Community"
set "CMAKE=%VSROOT%\Common7\IDE\CommonExtensions\Microsoft\CMake\CMake\bin\cmake.exe"
set "VCVARS=%VSROOT%\VC\Auxiliary\Build\vcvars64.bat"

if not exist "%CMAKE%" (
    echo [ERTrainer] cmake.exe not found: %CMAKE%
    exit /b 1
)
if not exist "%VCVARS%" (
    echo [ERTrainer] vcvars64.bat not found: %VCVARS%
    exit /b 1
)

call "%VCVARS%" >nul
cd /d "%~dp0"

"%CMAKE%" -S . -B build -G "Visual Studio 18 2026" -A x64
if errorlevel 1 exit /b 1

"%CMAKE%" --build build --config Release
if errorlevel 1 exit /b 1

echo.
echo [ERTrainer] OK: build\bin\Release\ERTrainer129.exe
endlocal

```

`build_dist.ps1`:

```ps1
# Сборка дистрибутива: ERTrainer\..\dist\ERTrainer.exe (без файлового лога)
$ErrorActionPreference = "Stop"
$cmake = "C:\Program Files\Microsoft Visual Studio\18\Community\Common7\IDE\CommonExtensions\Microsoft\CMake\CMake\bin\cmake.exe"
$root = Split-Path -Parent $MyInvocation.MyCommand.Path
$build = Join-Path $root "build"
$dist = Join-Path (Split-Path -Parent $root) "dist"

& $cmake -S $root -B $build -DERTRAINER_DIST=ON
& $cmake --build $build --config Release

$built = Join-Path $build "bin\Release\ERTrainer129.exe"
if (-not (Test-Path $built)) {
    throw "Не найден: $built"
}

New-Item -ItemType Directory -Force -Path $dist | Out-Null
Copy-Item -Force $built (Join-Path $dist "ERTrainer.exe")
Write-Host "Готово: $(Join-Path $dist 'ERTrainer.exe')"

```

```cpp
    CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);

    if (!Logger::instance().init()) {
        MessageBoxW(nullptr, L"Не удалось открыть ERTrainer.log", L"ERTrainer", MB_ICONERROR);

    }

    Config::instance().load();

    GameAttach game;
    FeatureManager features;
    WorkerThread worker(game, features);
    worker.start();

    UiRenderer ui(game, features);
    const auto& w = Config::instance().window;
    if (!ui.create(w.x, w.y, w.w, w.h)) {
        worker.stop();
        Logger::instance().error("UI не создан");
        Logger::instance().shutdown();

    try {
        code = ui.run();
    } catch (const std::exception& ex) {
        Logger::instance().error(std::string("Исключение UI: ") + ex.what());
        code = 1;
    }

    worker.stop();
    Logger::instance().info("выход");
    Logger::instance().shutdown();
    CoUninitialize();
    return code;
}
```

```cpp
        return {};
    }
    std::filesystem::path dir = std::filesystem::path(roaming) / kAppFolder;

    std::filesystem::create_directories(dir, ec);

bool Config::load() {
    path = wideToUtf8(configPath().wstring());
    for (const auto& info : featureCatalog()) {
        hotkeys[info.id] = info.defaultHotkey;
        if (info.defaultFloat != 0.0f) {
            featureFloat[info.id] = info.defaultFloat;
        }
        if (info.defaultInt != 0) {
            featureInt[info.id] = info.defaultInt;
        }
    }

    std::ifstream in(utf8ToWide(path), std::ios::binary);
    if (!in) {
        Logger::instance().info("Конфиг не найден, создаём по умолчанию: " + path);
        return save();
    }

    nlohmann::json j;
    try {
        in >> j;
    } catch (const std::exception& ex) {
        Logger::instance().warn(std::string("Конфиг битый, перезаписываем: ") + ex.what());
        return save();
    }

    if (j.contains("window")) {
        window.x = j["window"].value("x", window.x);
        window.y = j["window"].value("y", window.y);
        window.w = j["window"].value("w", window.w);
        window.h = j["window"].value("h", window.h);
    }
    if (j.contains("hotkeys") && j["hotkeys"].is_object()) {
        for (auto it = j["hotkeys"].begin(); it != j["hotkeys"].end(); ++it) {
            if (it.value().is_string()) {
                hotkeys[it.key()] = parseHotkey(it.value().get<std::string>());
            }
        }
    }
    if (j.contains("feature_float") && j["feature_float"].is_object()) {
        for (auto it = j["feature_float"].begin(); it != j["feature_float"].end(); ++it) {
            featureFloat[it.key()] = it.value().get<float>();
        }
    }
    if (j.contains("feature_int") && j["feature_int"].is_object()) {
        for (auto it = j["feature_int"].begin(); it != j["feature_int"].end(); ++it) {
            featureInt[it.key()] = it.value().get<int>();
        }
    }
    Logger::instance().info("Конфиг загружен: " + path);
    return true;
}

bool Config::save() const {
    nlohmann::json j;
    j["window"] = {{"x", window.x}, {"y", window.y}, {"w", window.w}, {"h", window.h}};
    nlohmann::json hk = nlohmann::json::object();
    for (const auto& [id, key] : hotkeys) {
        hk[id] = key.toString();
    }
    j["hotkeys"] = hk;
    nlohmann::json ff = nlohmann::json::object();
    for (const auto& [id, v] : featureFloat) {
        ff[id] = v;
    }
    j["feature_float"] = ff;
    nlohmann::json fi = nlohmann::json::object();
    for (const auto& [id, v] : featureInt) {
        fi[id] = v;
    }
    j["feature_int"] = fi;

    const auto p = utf8ToWide(path.empty() ? wideToUtf8(configPath().wstring()) : path);
    std::ofstream out(p, std::ios::binary | std::ios::trunc);
    if (!out) {
        Logger::instance().error("Не удалось записать конфиг");
        return false;
    }
    out << std::setw(4) << j << '\n';
    return true;
}
```

```cpp
namespace er {

RcBlob loadResource(int id) {
    RcBlob out;
    HMODULE mod = GetModuleHandleW(nullptr);
    HRSRC hRes = FindResourceW(mod, MAKEINTRESOURCEW(id), RT_RCDATA);
    if (!hRes) {
        return out;
    }
    HGLOBAL hGlob = LoadResource(mod, hRes);
    if (!hGlob) {
        return out;
    }
    const void* p = LockResource(hGlob);
    const DWORD size = SizeofResource(mod, hRes);
    if (!p || size == 0) {
        return out;
    }
    out.data.assign(static_cast<const u8*>(p), static_cast<const u8*>(p) + size);
    out.ok = true;
    return out;
}

}  // namespace er
```

```cpp
    wchar_t buf[MAX_PATH]{};
    const DWORD n = GetModuleFileNameW(nullptr, buf, MAX_PATH);
    if (n == 0 || n >= MAX_PATH) {
        return std::filesystem::current_path();
    }

std::string timestamp() {
    using namespace std::chrono;
    const auto now = system_clock::now();
    const auto t = system_clock::to_time_t(now);
    const auto ms = duration_cast<milliseconds>(now.time_since_epoch()) % 1000;
    std::tm local{};
    localtime_s(&local, &t);
    char buf[64]{};
    std::snprintf(buf, sizeof(buf), "%04d-%02d-%02d %02d:%02d:%02d.%03d",
                  local.tm_year + 1900, local.tm_mon + 1, local.tm_mday,
                  local.tm_hour, local.tm_min, local.tm_sec, static_cast<int>(ms.count()));
    return buf;
}

#ifdef ERTRAINER_DIST
    return true;
#else
    std::lock_guard lock(mutex_);
    if (file_) {
        return true;
    }

    std::error_code ec;
    auto dir = appDataDir();
    if (dir.empty()) {
        dir = exeDir();
    }
    std::filesystem::create_directories(dir, ec);
    const auto appDataLog = dir / kLogFileName;
    const auto localLog = exeDir() / kLogFileName;

    FILE* f = _wfsopen(appDataLog.c_str(), L"ab", _SH_DENYNO);
    if (!f) {
        f = _wfsopen(localLog.c_str(), L"ab", _SH_DENYNO);
        path_ = wideToUtf8(localLog.wstring());

        if (std::ftell(f) == 0) {
```

```h
#include <string_view>

namespace er {

inline std::string wideToUtf8(std::wstring_view w) {
    if (w.empty()) {
        return {};
    }
    const int n = WideCharToMultiByte(CP_UTF8, 0, w.data(), static_cast<int>(w.size()), nullptr, 0, nullptr, nullptr);
    if (n <= 0) {
        return {};
    }
    std::string s(static_cast<size_t>(n), '\0');
    WideCharToMultiByte(CP_UTF8, 0, w.data(), static_cast<int>(w.size()), s.data(), n, nullptr, nullptr);
    return s;
}

inline std::wstring utf8ToWide(std::string_view u8) {
    if (u8.empty()) {
        return {};
    }
    const int n = MultiByteToWideChar(CP_UTF8, 0, u8.data(), static_cast<int>(u8.size()), nullptr, 0);
    if (n <= 0) {
        return {};
    }
    std::wstring s(static_cast<size_t>(n), L'\0');
    MultiByteToWideChar(CP_UTF8, 0, u8.data(), static_cast<int>(u8.size()), s.data(), n);
    return s;
}
```

```cpp
class PlayerFlagFeature final : public IFeature {
public:
    PlayerFlagFeature(FeatureInfo info, int flagIndex)
        : info_(std::move(info)), flagIndex_(flagIndex) {}

    const FeatureInfo& info() const override { return info_; }
    FeatureStatus status() const override { return status_; }
    std::string_view reason() const override { return reason_; }

    bool resolve(SignatureScanner&) override {
        if (GameHooks::instance().playerSiteFound()) {
            status_ = FeatureStatus::Ready;
            reason_ = "Сигнатура найдена. Хук ставится при включении.";
            return true;
        }
        status_ = FeatureStatus::Unavailable;
        reason_ = "Нет aobplayer на 1.17.2";
        return false;
    }

    void enable() override {
        if (!memory_ || !GameHooks::instance().ensurePlayerHook(*memory_)) {
            status_ = FeatureStatus::Error;
            reason_ = "Не удалось поставить хук HP (cave). Игра не патчилась.";
            enabled_ = false;
            return;
        }
        enabled_ = true;
        status_ = FeatureStatus::Enabled;
        applyFlag(true);
    }

    void disable() override {
        enabled_ = false;
        if (memory_ && status_ != FeatureStatus::Unavailable) {
            applyFlag(false);
            status_ = FeatureStatus::Ready;
        }
    }

    void tick(float) override {}
    void bindMemory(ProcessMemory* m) override { memory_ = m; }

private:
    void applyFlag(bool on) {
        auto& h = GameHooks::instance();
        switch (flagIndex_) {
            case 0:
                h.setHpFlag(*memory_, on);
                break;
            case 1:
                h.setFocusFlag(*memory_, on);
                break;
            case 2:
                h.setStaminaFlag(*memory_, on);
                break;
            default:
                break;
        }
    }

    FeatureInfo info_;
    int flagIndex_;
    FeatureStatus status_ = FeatureStatus::Unavailable;
    std::string reason_ = "Нет хука";
    bool enabled_ = false;
    ProcessMemory* memory_ = nullptr;
};

class ToggleHookFeature final : public IFeature {
public:
    ToggleHookFeature(FeatureInfo info, ToggleKind kind) : info_(std::move(info)), kind_(kind) {}

    const FeatureInfo& info() const override { return info_; }
    FeatureStatus status() const override { return status_; }
    std::string_view reason() const override { return reason_; }

    bool resolve(SignatureScanner&) override {
        auto& h = GameHooks::instance();
        bool ok = false;
        switch (kind_) {
            case ToggleKind::God:
                ok = h.godSiteFound();
                break;
            case ToggleKind::Weight:
                ok = h.weightSiteFound();
                break;
            case ToggleKind::Drop:
                ok = h.dropSiteFound();
                break;
            case ToggleKind::Stealth:
                ok = h.stealthSiteFound();
                break;
            case ToggleKind::Immune:
                ok = h.immuneSiteFound();
                break;
            case ToggleKind::Horse:
                ok = h.horseSiteFound();
                break;
            case ToggleKind::NoSoul:
                ok = h.noSoulLossSiteFound();
                break;
            case ToggleKind::Item:
                ok = h.itemSiteFound();
                break;
            case ToggleKind::FreezeDay:
                ok = h.daySiteFound();
                break;
            case ToggleKind::Fly:
            case ToggleKind::FreezeAi:
                ok = h.locSiteFound();
                break;
            case ToggleKind::OneHit:
            case ToggleKind::DmgMul:
            case ToggleKind::DefMul:
                ok = h.dmgSiteFound();
                break;
            case ToggleKind::SoulMul:
                ok = h.soulsSiteFound();
                break;
            case ToggleKind::GameSpeed:
                ok = h.speedSiteFound();
                break;
        }
        if (ok) {
            status_ = FeatureStatus::Ready;
            reason_ = "Готово. Хук/патч при включении.";
            return true;
        }
        status_ = FeatureStatus::Unavailable;
        reason_ = "Сигнатура не найдена на 1.17.2";
        return false;
    }

    void enable() override {
        if (!memory_) {
            fail("Нет процесса");
            return;
        }
        auto& h = GameHooks::instance();
        auto& cfg = Config::instance();
        bool ok = false;
        switch (kind_) {
            case ToggleKind::God:
                ok = h.ensureGodHook(*memory_);
                if (ok) {
                    h.setGodFlag(*memory_, true);
                }
                break;
            case ToggleKind::Weight:
                ok = h.ensureWeightHook(*memory_);
                break;
            case ToggleKind::Drop:
                ok = h.ensureDropHook(*memory_);
                break;
            case ToggleKind::Stealth:
                ok = h.applyStealth(*memory_, true);
                break;
            case ToggleKind::Immune:
                ok = h.ensureImmuneHook(*memory_);
                if (ok) {
                    h.setImmuneFlag(*memory_, true);
                }
                break;
            case ToggleKind::Horse:
                ok = h.ensureHorseHook(*memory_);
                if (ok) {
                    h.setHorseFlag(*memory_, true);
                }
                break;
            case ToggleKind::NoSoul:
                ok = h.applyNoSoulLoss(*memory_, true);
                break;
            case ToggleKind::Item:
                ok = h.ensureItemHook(*memory_);
                break;
            case ToggleKind::FreezeDay:
                ok = h.ensureDayHook(*memory_);
                if (ok) {
                    h.setFreezeDay(*memory_, true);
                }
                break;
            case ToggleKind::Fly:
                ok = h.ensureLocationHook(*memory_);
                if (ok) {
                    h.setFlyMode(*memory_, true);
                }
                break;
            case ToggleKind::FreezeAi:
                ok = h.ensureLocationHook(*memory_);
                if (ok) {
                    h.setFreezeAi(*memory_, true);
                }
                break;
            case ToggleKind::OneHit:
                ok = h.ensureDmgHook(*memory_);
                if (ok) {
                    h.setOneHit(*memory_, true);
                }
                break;
            case ToggleKind::DmgMul:
                ok = h.ensureDmgHook(*memory_);
                if (ok) {
                    float v = 2.0f;
                    if (const auto it = cfg.featureFloat.find(info_.id); it != cfg.featureFloat.end()) {
                        v = it->second;
                    }
                    h.setDmgMul(*memory_, v);
                }
                break;
            case ToggleKind::DefMul:
                ok = h.ensureDmgHook(*memory_);
                if (ok) {
                    float v = 2.0f;
                    if (const auto it = cfg.featureFloat.find(info_.id); it != cfg.featureFloat.end()) {
                        v = it->second;
                    }
                    h.setDefMul(*memory_, v);
                }
                break;
            case ToggleKind::SoulMul:
                ok = h.ensureSoulsHook(*memory_);
                if (ok) {
                    float v = 2.0f;
                    if (const auto it = cfg.featureFloat.find(info_.id); it != cfg.featureFloat.end()) {
                        v = it->second;

                    h.setSoulMul(*memory_, v);

            return;

        if (kind_ != ToggleKind::DmgMul && kind_ != ToggleKind::DefMul && kind_ != ToggleKind::SoulMul &&
            kind_ != ToggleKind::GameSpeed) {
            return;
        }
        auto& cfg = Config::instance();
        float v = 2.0f;
        if (const auto it = cfg.featureFloat.find(info_.id); it != cfg.featureFloat.end()) {
            v = it->second;
        }
        auto& h = GameHooks::instance();
        if (kind_ == ToggleKind::DmgMul) {
            h.setDmgMul(*memory_, v);
        } else if (kind_ == ToggleKind::DefMul) {
            h.setDefMul(*memory_, v);
        } else if (kind_ == ToggleKind::SoulMul) {
            h.setSoulMul(*memory_, v);
        } else if (kind_ == ToggleKind::GameSpeed) {
            h.setGameSpeed(*memory_, true, v);
        }
    }

    void bindMemory(ProcessMemory* m) override { memory_ = m; }

private:
    void fail(const char* msg) {
        status_ = FeatureStatus::Error;
        reason_ = msg;
        enabled_ = false;
    }

    FeatureInfo info_;
    ToggleKind kind_;
    FeatureStatus status_ = FeatureStatus::Unavailable;
    std::string reason_ = "Нет хука";
    bool enabled_ = false;
    ProcessMemory* memory_ = nullptr;
};

    bool resolve(SignatureScanner&) override {
        auto& h = GameHooks::instance();
        bool ok = false;
        switch (kind_) {
            case ApplyKind::DayPlus:
                ok = h.daySiteFound();
                break;
            case ApplyKind::FlyUp:
            case ApplyKind::FlyDown:
                ok = h.locSiteFound();
                break;
            case ApplyKind::MaxHp:
            case ApplyKind::MaxFp:
            case ApplyKind::MaxSt:
                ok = h.playerSiteFound();
                break;
            default:
                ok = h.soulsSiteFound();
                break;
        }
        if (ok) {
            status_ = FeatureStatus::Ready;
            reason_ = "Готово. Жми ПРИМЕНИТЬ.";
            return true;
        }
        status_ = FeatureStatus::Unavailable;
        reason_ = "Сигнатура не найдена";
        return false;
    }

    void enable() override {
        applyNow();
        if (kind_ == ApplyKind::FlyUp || kind_ == ApplyKind::FlyDown) {
            return;
        }
        if (status_ != FeatureStatus::Error) {
            status_ = FeatureStatus::Enabled;
        }
    }
    void disable() override {
        if (memory_) {
            auto& h = GameHooks::instance();
            if (kind_ == ApplyKind::MaxHp) {
                h.setMaxHp(*memory_, 0);
            } else if (kind_ == ApplyKind::MaxFp) {
                h.setMaxFp(*memory_, 0);
            } else if (kind_ == ApplyKind::MaxSt) {
                h.setMaxStamina(*memory_, 0);
            } else if (kind_ >= ApplyKind::Level && kind_ <= ApplyKind::St8) {
                h.setStat(*memory_, static_cast<int>(kind_) - static_cast<int>(ApplyKind::Level), 0);
            }
        }
        if (status_ != FeatureStatus::Unavailable) {
            status_ = FeatureStatus::Ready;
        }
    }
    void tick(float) override {}
    void bindMemory(ProcessMemory* m) override { memory_ = m; }

    void applyNow() override {
        if (!memory_) {
            return;
        }
        auto& h = GameHooks::instance();
        auto& cfg = Config::instance();
        i32 iv = info_.defaultInt;
        if (const auto it = cfg.featureInt.find(info_.id); it != cfg.featureInt.end()) {
            iv = it->second;
        }
        bool ok = false;
        switch (kind_) {
            case ApplyKind::DayPlus:
                ok = h.ensureDayHook(*memory_);
                if (ok) {
                    h.pulseDayAdd(*memory_);
                }
                break;
            case ApplyKind::FlyUp:
                ok = h.ensureLocationHook(*memory_);
                if (ok) {
                    h.pulseFly(*memory_, 1);
                }
                break;
            case ApplyKind::FlyDown:
                ok = h.ensureLocationHook(*memory_);
                if (ok) {
                    h.pulseFly(*memory_, -1);
                }
                break;
            case ApplyKind::MaxHp:
                ok = h.ensurePlayerHook(*memory_);
                if (ok) {
                    h.setMaxHp(*memory_, iv);
                }
                break;
            case ApplyKind::MaxFp:
                ok = h.ensurePlayerHook(*memory_);
                if (ok) {
                    h.setMaxFp(*memory_, iv);
                }
                break;
            case ApplyKind::MaxSt:
                ok = h.ensurePlayerHook(*memory_);
                if (ok) {
                    h.setMaxStamina(*memory_, iv);
                }
                break;
            default: {
                ok = h.ensureSoulsHook(*memory_);
                if (ok) {
                    int idx = 0;
                    switch (kind_) {
                        case ApplyKind::Level:
                            idx = 0;
                            break;
                        case ApplyKind::St1:
                            idx = 1;
                            break;
                        case ApplyKind::St2:
                            idx = 2;
                            break;
                        case ApplyKind::St3:
                            idx = 3;
                            break;
                        case ApplyKind::St4:
                            idx = 4;
                            break;
                        case ApplyKind::St5:
                            idx = 5;
                            break;
                        case ApplyKind::St6:
                            idx = 6;
                            break;
                        case ApplyKind::St7:
                            idx = 7;
                            break;
                        case ApplyKind::St8:
                            idx = 8;
                            break;
                        default:
                            break;

                    h.setStat(*memory_, idx, iv);
                }
                break;
            }
        }
        if (!ok) {
            status_ = FeatureStatus::Error;
            reason_ = "Не удалось применить";
        }
    }

private:
    FeatureInfo info_;
    ApplyKind kind_;
    FeatureStatus status_ = FeatureStatus::Unavailable;
    std::string reason_;

FeatureManager::FeatureManager() {

        auto f = makeFeature(info);

        wanted_[f->info().id] = false;

    }
}

    memory_ = memory;
    for (auto& f : features_) {
        f->bindMemory(memory);
    }
}

void FeatureManager::resolveAll(SignatureScanner& scanner) {
    if (memory_) {
        GameHooks::instance().resolve(*memory_, scanner);
        bindMemory(memory_);
    }
    for (auto& f : features_) {
        f->resolve(scanner);
    }
}

void FeatureManager::shutdown() {
    disableAll();
    if (memory_) {
        GameHooks::instance().shutdown(*memory_);
    }

void FeatureManager::requestEnable(std::string_view id, bool on) {
    wanted_[std::string(id)] = on;
    syncStates();
}

void FeatureManager::requestApply(std::string_view id) {
    IFeature* f = find(id);
    if (!f || !memory_) {
        return;
    }
    if (id == "edit_runes") {
        auto& cfg = Config::instance();
        i32 val = 99999999;
        if (const auto it = cfg.featureInt.find("edit_runes"); it != cfg.featureInt.end()) {
            val = it->second;
        }
        f->setIntValue(val);
    }
    f->applyNow();
}

IFeature* FeatureManager::find(std::string_view id) {
    const auto it = byId_.find(std::string(id));
    return it == byId_.end() ? nullptr : it->second;
}

void FeatureManager::syncStates() {
    std::lock_guard lock(mu_);
    for (auto& f : features_) {
        const auto& id = f->info().id;
        const bool want = wanted_[id].load();
        const bool isOn = f->status() == FeatureStatus::Enabled;
        if (f->status() == FeatureStatus::Unavailable) {
            continue;
        }
        if (want && !isOn) {
            if (f->status() != FeatureStatus::Error) {
                f->enable();
            }
        } else if (!want && (isOn || f->status() == FeatureStatus::Error)) {
            f->disable();
        }
    }
}

void FeatureManager::disableAll() {
    for (auto& f : features_) {
        if (f->status() == FeatureStatus::Enabled) {
            f->disable();
        }
        wanted_[f->info().id] = false;
    }
}

void FeatureManager::tick(float dt) {
    syncStates();
    if (memory_) {
        GameHooks::instance().tick(*memory_);
    }
    for (auto& f : features_) {
        if (f->status() == FeatureStatus::Enabled) {
            f->tick(dt);
        }
    }
}
```

```cpp
bool nameEquals(std::wstring_view a, std::wstring_view b) {
    if (a.size() != b.size()) {
        return false;
    }
    for (usize i = 0; i < a.size(); ++i) {
        if (std::towlower(a[i]) != std::towlower(b[i])) {
            return false;
        }
    }
    return true;
}

    state_.status = "Ожидание процесса eldenring.exe";

void GameAttach::enableDebugPrivilege() {
    if (privilegeTried_) {
        return;
    }
    privilegeTried_ = true;
    HANDLE token = nullptr;
    if (!OpenProcessToken(GetCurrentProcess(), TOKEN_ADJUST_PRIVILEGES | TOKEN_QUERY, &token)) {
        Logger::instance().warn("OpenProcessToken: " + lastWinError());
        return;
    }
    TOKEN_PRIVILEGES tp{};
    tp.PrivilegeCount = 1;
    tp.Privileges[0].Attributes = SE_PRIVILEGE_ENABLED;
    if (!LookupPrivilegeValueW(nullptr, SE_DEBUG_NAME, &tp.Privileges[0].Luid)) {
        Logger::instance().warn("LookupPrivilegeValue SE_DEBUG: " + lastWinError());
        CloseHandle(token);
        return;
    }
    if (!AdjustTokenPrivileges(token, FALSE, &tp, sizeof(tp), nullptr, nullptr)) {
        Logger::instance().warn("AdjustTokenPrivileges: " + lastWinError());
    }
    CloseHandle(token);
}

bool GameAttach::findProcess(DWORD& pidOut) {
    HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snap == INVALID_HANDLE_VALUE) {
        return false;
    }
    PROCESSENTRY32W pe{};
    pe.dwSize = sizeof(pe);
    bool found = false;
    DWORD first = 0;
    int count = 0;
    if (Process32FirstW(snap, &pe)) {
        do {
            if (nameEquals(pe.szExeFile, kProcessName)) {
                ++count;
                if (!found) {
                    first = pe.th32ProcessID;
                    found = true;
                }
            }
        } while (Process32NextW(snap, &pe));
    }
    CloseHandle(snap);
    if (count > 1) {
        Logger::instance().warn("Найдено несколько eldenring.exe, берём PID " + std::to_string(first));
    }
    pidOut = first;
    return found;
}

bool GameAttach::readModuleInfo() {
    HMODULE mods[512]{};
    DWORD needed = 0;
    if (!EnumProcessModulesEx(memory_.handle(), mods, sizeof(mods), &needed, LIST_MODULES_64BIT)) {
        if (!EnumProcessModules(memory_.handle(), mods, sizeof(mods), &needed)) {
            Logger::instance().error("EnumProcessModules: " + lastWinError());
            return false;
        }
    }
    if (needed < sizeof(HMODULE)) {
        return false;
    }

    wchar_t name[MAX_PATH]{};
    wchar_t path[MAX_PATH]{};
    bool picked = false;
    const usize n = needed / sizeof(HMODULE);
    for (usize i = 0; i < n && i < 512; ++i) {
        name[0] = 0;
        GetModuleBaseNameW(memory_.handle(), mods[i], name, MAX_PATH);
        if (nameEquals(name, kProcessName)) {
            MODULEINFO info{};
            if (!GetModuleInformation(memory_.handle(), mods[i], &info, sizeof(info))) {
                Logger::instance().error("GetModuleInformation: " + lastWinError());
                return false;
            }
            GetModuleFileNameExW(memory_.handle(), mods[i], path, MAX_PATH);
            std::lock_guard lock(mutex_);
            state_.base = reinterpret_cast<uptr>(info.lpBaseOfDll);
            state_.moduleSize = info.SizeOfImage;
            state_.path = wideToUtf8(path);
            picked = true;
            break;
        }
    }
    if (!picked) {
        Logger::instance().error("Модуль eldenring.exe не найден в процессе");
        return false;
    }
    scanner_.setModule(state_.base, state_.moduleSize);
    return true;
}

bool GameAttach::readTextSection() {
    IMAGE_DOS_HEADER dos{};
    if (!memory_.read(state_.base, dos) || dos.e_magic != IMAGE_DOS_SIGNATURE) {
        Logger::instance().error("DOS-заголовок eldenring.exe не прочитан");
        return false;
    }
    IMAGE_NT_HEADERS64 nt{};
    if (!memory_.read(state_.base + static_cast<uptr>(dos.e_lfanew), nt) ||
        nt.Signature != IMAGE_NT_SIGNATURE) {
        Logger::instance().error("NT-заголовок eldenring.exe не прочитан");
        return false;
    }
    const uptr sectionStart = state_.base + static_cast<uptr>(dos.e_lfanew) +
                              offsetof(IMAGE_NT_HEADERS64, OptionalHeader) +
                              nt.FileHeader.SizeOfOptionalHeader;
    for (u16 i = 0; i < nt.FileHeader.NumberOfSections; ++i) {
        IMAGE_SECTION_HEADER sec{};
        if (!memory_.read(sectionStart + static_cast<uptr>(i) * sizeof(IMAGE_SECTION_HEADER), sec)) {
            continue;
        }
        char name[9]{};
        std::memcpy(name, sec.Name, 8);
        if (std::string_view(name) == ".text") {
            std::lock_guard lock(mutex_);
            state_.textBase = state_.base + sec.VirtualAddress;
            state_.textSize = sec.Misc.VirtualSize ? sec.Misc.VirtualSize : sec.SizeOfRawData;
            scanner_.setTextRange(state_.textBase, state_.textSize);

            u8 preview[16]{};
            if (memory_.read(state_.textBase, preview, sizeof(preview))) {
                char hex[64]{};
                int pos = 0;
                for (int b = 0; b < 16 && pos + 4 < static_cast<int>(sizeof(hex)); ++b) {
                    pos += std::snprintf(hex + pos, sizeof(hex) - static_cast<usize>(pos), "%02X ", preview[b]);
                }
                state_.textPreviewHex = hex;
            }
            return true;
        }
    }
    Logger::instance().error("Секция .text не найдена");
    return false;
}

bool GameAttach::detectEac() {
    HMODULE mods[512]{};
    DWORD needed = 0;
    if (!EnumProcessModulesEx(memory_.handle(), mods, sizeof(mods), &needed, LIST_MODULES_64BIT) &&
        !EnumProcessModules(memory_.handle(), mods, sizeof(mods), &needed)) {
        return false;
    }
    const usize n = needed / sizeof(HMODULE);
    wchar_t name[MAX_PATH]{};
    for (usize i = 0; i < n && i < 512; ++i) {
        name[0] = 0;
        GetModuleBaseNameW(memory_.handle(), mods[i], name, MAX_PATH);
        std::wstring lower(name);
        std::transform(lower.begin(), lower.end(), lower.begin(),
                       [](wchar_t ch) { return static_cast<wchar_t>(std::towlower(ch)); });
        if (lower.find(L"easyanticheat") != std::wstring::npos) {
            return true;
        }
    }
    return false;
}

void GameAttach::tick() {
    enableDebugPrivilege();

    DWORD pid = 0;
    if (!findProcess(pid)) {
        if (state_.attached) {
            Logger::instance().warn("Процесс eldenring.exe закрыт, отключаемся");
            detach();
        }
        std::lock_guard lock(mutex_);
        state_.status = "Ожидание процесса eldenring.exe";
        return;
    }

    if (state_.attached && state_.pid == pid && memory_.isOpen()) {
        DWORD exitCode = 0;
        if (GetExitCodeProcess(memory_.handle(), &exitCode) && exitCode == STILL_ACTIVE) {
            return;
        }
        Logger::instance().warn("Процесс больше не активен");
        detach();
    }

    if (!memory_.open(pid)) {
        std::lock_guard lock(mutex_);
        state_.status = "Не удалось OpenProcess";
        return;
    }

    if (!readModuleInfo() || !readTextSection()) {
        memory_.close();
        std::lock_guard lock(mutex_);
        state_.attached = false;
        state_.status = "Модуль не прочитан";
        return;
    }

    const bool eac = detectEac();
    GameVersion ver;
    {
        std::lock_guard lock(mutex_);
        ver = parseFileVersion(utf8ToWide(state_.path));
        state_.pid = pid;
        state_.attached = true;
        state_.eacLoaded = eac;
        state_.version = ver;
        state_.status = eac ? "Подключено, но обнаружен EasyAntiCheat" : "Подключено";
    }

    Logger::instance().info("Подключено к eldenring.exe PID=" + std::to_string(pid));
    Logger::instance().info("Путь: " + state_.path);
    Logger::instance().info("База модуля: 0x" + toHex(state_.base) + " размер=0x" + toHex(state_.moduleSize));
    Logger::instance().info(".text: 0x" + toHex(state_.textBase) + " размер=0x" + toHex(state_.textSize));
    if (!state_.textPreviewHex.empty()) {
        Logger::instance().info(".text[0..15]: " + state_.textPreviewHex);
    }
    Logger::instance().info("FileVersion=" + ver.fileVersion + "  отображаемая=" + ver.displayVersion);
    if (ver.warning) {
        Logger::instance().warn(ver.warningText);
    } else {
        Logger::instance().info("Версия совпадает с ожидаемой App Ver. " + std::string(kExpectedAppVersion));
    }
    if (eac) {
        Logger::instance().error("В процессе загружен EasyAntiCheat. Трейнер не будет писать в память. Запусти игру офлайн без EAC.");
    } else {
        Logger::instance().info("EasyAntiCheat в процессе не найден (офлайн, ок).");
    }
}

    memory_.close();
    std::lock_guard lock(mutex_);
    const std::string keepStatus = "Процесс потерян";
    state_ = {};
    state_.status = keepStatus;
}

    std::lock_guard lock(mutex_);

}
```

```cpp
namespace er {

GameVersion mapKnownVersion(std::string_view fileVersion) {
    GameVersion v;
    v.fileVersion = std::string(fileVersion);
    v.productVersion = v.fileVersion;
    if (fileVersion == kExpectedFileVersion) {
        v.displayVersion = kExpectedAppVersion;
        v.known = true;
        v.warning = false;
    } else if (fileVersion.empty()) {
        v.displayVersion = "неизвестно";
        v.known = false;
        v.warning = true;
        v.warningText = "Не удалось прочитать FileVersion eldenring.exe.";
    } else {
        v.displayVersion = std::string(fileVersion);
        v.known = false;
        v.warning = true;
        v.warningText = "Версия FileVersion " + std::string(fileVersion) +
                        " не совпадает с ожидаемой " + kExpectedFileVersion +
                        " (App Ver. " + kExpectedAppVersion + "). Запуск разрешён, часть функций может быть недоступна.";
    }
    return v;
}

GameVersion parseFileVersion(std::wstring_view exePath) {
    GameVersion empty = mapKnownVersion("");
    if (exePath.empty()) {
        return empty;
    }
    const std::wstring path(exePath);
    DWORD handle = 0;
    const DWORD size = GetFileVersionInfoSizeW(path.c_str(), &handle);
    if (size == 0) {
        return empty;
    }
    std::vector<u8> buf(size);
    if (!GetFileVersionInfoW(path.c_str(), 0, size, buf.data())) {
        return empty;
    }

    VS_FIXEDFILEINFO* info = nullptr;
    UINT len = 0;
    if (!VerQueryValueW(buf.data(), L"\\", reinterpret_cast<LPVOID*>(&info), &len) || !info) {
        return empty;
    }

    char fv[32]{};
    std::snprintf(fv, sizeof(fv), "%u.%u.%u.%u",
                  HIWORD(info->dwFileVersionMS), LOWORD(info->dwFileVersionMS),
                  HIWORD(info->dwFileVersionLS), LOWORD(info->dwFileVersionLS));
    GameVersion v = mapKnownVersion(fv);
    v.productVersion = v.fileVersion;
    return v;
}
```

```h
struct CaveEmit {
    std::vector<u8> b;
    uptr base = 0;

    void bytes(std::initializer_list<u8> x) { b.insert(b.end(), x.begin(), x.end()); }

    void put32(::er::i32 v) {
        u8 t[4];
        std::memcpy(t, &v, 4);
        b.insert(b.end(), t, t + 4);
    }

    void put64(u64 v) {

        std::memcpy(t, &v, 8);
        b.insert(b.end(), t, t + 8);

    void L(const char* n) { lab_[n] = static_cast<int>(b.size()); }

    void jcc8(u8 cc, const char* n) {

    [[nodiscard]] bool finish(std::string* why = nullptr) {
        for (const auto& r : rips_) {
            const std::int64_t delta =
                static_cast<std::int64_t>(r.target) - static_cast<std::int64_t>(base + static_cast<uptr>(r.at + r.len));
            if (delta < INT32_MIN || delta > INT32_MAX) {
                if (why) {
                    *why = "RIP at+" + std::to_string(r.at) + " len=" + std::to_string(r.len) +
                           " tgt=" + toHex(r.target);
                }
                return false;
            }
            const std::int32_t d32 = static_cast<std::int32_t>(delta);
            std::memcpy(b.data() + r.at + r.off, &d32, 4);
        }
        for (const auto& j : j8_) {
            const auto it = lab_.find(j.name);
            if (it == lab_.end()) {
                if (why) {
                    *why = "no label " + j.name;
                }
                return false;
            }
            const int rel = it->second - (j.at + 1);
            if (rel < -128 || rel > 127) {
                if (why) {
                    *why = "rel8 " + j.name + " d=" + std::to_string(rel);
                }
                return false;
            }
            b[static_cast<usize>(j.at)] = static_cast<u8>(rel);
        }
        for (const auto& j : j32_) {
            const auto it = lab_.find(j.name);
            if (it == lab_.end()) {
                if (why) {
                    *why = "no j32 label " + j.name;
                }
                return false;
            }
            const int rel = it->second - (j.at + 4);
            const std::int32_t d32 = static_cast<std::int32_t>(rel);
            std::memcpy(b.data() + j.at, &d32, 4);
        }
        return true;
    }

private:
    struct Rip {
        int at;
        int len;
        int off;
        uptr target;
    };
    struct J8 {
        int at;
        std::string name;
    };
    std::vector<Rip> rips_;
    std::vector<J8> j8_;
    std::vector<J8> j32_;
    std::unordered_map<std::string, int> lab_;
};
```

```cpp
bool commitCave(ProcessMemory& memory, CodeCave& cave, uptr site, usize patchLen, CaveEmit& e,
                const char* name) {
    std::string why;
    if (!e.finish(&why)) {
        Logger::instance().error(std::string(name) + ": " + why + " cave=0x" + toHex(e.base) +
                                 " bytes=" + std::to_string(e.b.size()));
        return false;
    }
    if (!cave.write(memory, 0, e.b) || !cave.installRelJmpHook(memory, site, patchLen)) {
        Logger::instance().error(std::string(name) + ": не удалось записать jmp");
        return false;
    }
    Logger::instance().info(std::string(name) + " ON @ 0x" + toHex(site));
    return true;
}

bool GameHooks::installPlayerHook(ProcessMemory& memory) {
    if (playerHooked_ || playerSite_ == 0) {
        return playerHooked_;
    }
    if (!memory.read(playerSite_ + 2, playerDisp_)) {
        return false;
    }
    if (!playerCave_.allocateNear(memory, playerSite_)) {
        lastError_ += "aobplayer: нет cave. ";
        return false;
    }

    const uptr cave = playerCave_.base();
    const uptr hp = cave + kFlags;
    const uptr fp = cave + kFlags + 4;
    const uptr st = cave + kFlags + 8;
    const uptr maxHp = cave + kFlags + 0x0C;
    const uptr maxFp = cave + kFlags + 0x10;
    const uptr maxSt = cave + kFlags + 0x14;
    const uptr playerData = cave + kPtrs;
    const uptr playerIns = cave + kPtrs + 8;

    CaveEmit e;
    e.base = cave;
    e.bytes({0x52, 0x48, 0x8B, 0xD0});  // push rdx; mov rdx, rax
    e.movRipRdx(playerData);
    e.bytes({0x48, 0x8B, 0x52, 0x08});  // mov rdx, [rdx+8]
    e.movRipRdx(playerIns);
    e.bytes({0x48, 0x8D, 0x90});  // lea rdx, [rax+disp]
    e.put32(playerDisp_);

    e.cmpRip1(hp);
    e.jne8("nohp");
    e.bytes({0x8B, 0x4A, 0x04, 0x89, 0x0A});
    e.L("nohp");
    e.cmpRip1(fp);
    e.jne8("nofp");
    e.bytes({0x8B, 0x4A, 0x14, 0x89, 0x4A, 0x10});
    e.L("nofp");
    e.cmpRip1(st);
    e.jne8("nost");
    e.bytes({0x8B, 0x4A, 0x20, 0x89, 0x4A, 0x1C});
    e.L("nost");

    e.movEcxRip(maxHp);
    e.bytes({0x83, 0xF9, 0x00});
    e.jle8("nomh");
    e.bytes({0x89, 0x4A, 0x08, 0x89, 0x4A, 0x0C});
    e.L("nomh");
    e.movEcxRip(maxFp);
    e.bytes({0x83, 0xF9, 0x00});
    e.jle8("nomf");
    e.bytes({0x89, 0x4A, 0x18});
    e.L("nomf");
    e.movEcxRip(maxSt);
    e.bytes({0x83, 0xF9, 0x00});
    e.jle8("noms");
    e.bytes({0x89, 0x4A, 0x24});
    e.L("noms");

    e.bytes({0x8B, 0x0A, 0x5A});  // mov ecx,[rdx]; pop rdx
    e.absJmp(playerSite_ + 6);

    for (usize off = 0; off <= 0x14; off += 4) {
        playerCave_.writeU32(memory, kFlags + off, 0);
    }
    if (!commitCave(memory, playerCave_, playerSite_, 6, e, "player")) {
        return false;
    }
    playerHooked_ = true;
    return true;
}

bool GameHooks::installSoulsHook(ProcessMemory& memory) {
    if (soulsHooked_ || soulsSite_ == 0) {
        return soulsHooked_;
    }
    if (!soulsCave_.allocateNear(memory, soulsSite_)) {
        lastError_ += "aobsouls: нет cave. ";
        return false;
    }

    const uptr cave = soulsCave_.base();
    const uptr soul = cave + kFlags;
    const uptr mul = cave + kFlags + 4;
    const uptr level = cave + kFlags + 8;

    CaveEmit e;
    e.base = cave;
    e.movRipRcx(cave + kPtrs);
    if (soulsVariant41_) {
        e.movR8dRip(soul);
        e.bytes({0x41, 0x83, 0xF8, 0x00});
        e.jle8("nosoul");
        e.bytes({0x44, 0x89, 0x41, 0x6C});
    } else {
        e.movR9dRip(soul);
        e.bytes({0x41, 0x83, 0xF9, 0x00});
        e.jle8("nosoul");
        e.bytes({0x44, 0x89, 0x49, 0x6C});
    }
    e.L("nosoul");

    e.bytes({0x83, 0xFA, 0x00});
    e.jle8("nomul");
    e.cmpRip0(mul);
    e.je8("nomul");
    e.bytes({0x52, 0xDB, 0x04, 0x24});  // push rdx; fild dword [rsp]
    e.fmulRip(mul);
    e.bytes({0xDB, 0x1C, 0x24, 0x5A});  // fistp dword [rsp]; pop rdx
    e.L("nomul");

    auto emitStat = [&](uptr slot, u8 rcxOff, const char* skip) {
        e.leaR11Rip(slot);
        e.bytes({0x45, 0x8B, 0x0B, 0x41, 0x83, 0xF9, 0x00});
        e.jle8(skip);
        e.bytes({0x44, 0x89, 0x49, rcxOff});
        e.L(skip);
    };
    emitStat(level, 0x68, "nlv");
    emitStat(level + 4, 0x3C, "ns1");
    emitStat(level + 8, 0x40, "ns2");
    emitStat(level + 12, 0x44, "ns3");
    emitStat(level + 16, 0x48, "ns4");
    emitStat(level + 20, 0x4C, "ns5");
    emitStat(level + 24, 0x50, "ns6");
    emitStat(level + 28, 0x54, "ns7");
    emitStat(level + 32, 0x58, "ns8");

    if (soulsVariant41_) {
        e.bytes({0x44, 0x8B, 0x41, 0x6C, 0x45, 0x33, 0xD2});
    } else {
        e.bytes({0x44, 0x8B, 0x49, 0x6C, 0x45, 0x33, 0xDB});
    }
    e.absJmp(soulsSite_ + 7);

    for (usize off = 0; off <= 0x28; off += 4) {
        soulsCave_.writeU32(memory, kFlags + off, 0);
    }
    if (!commitCave(memory, soulsCave_, soulsSite_, 7, e, "souls")) {
        return false;
    }
    soulsHooked_ = true;
    return true;
}

bool GameHooks::installGodHook(ProcessMemory& memory) {
    if (godHooked_ || godSite_ == 0) {
        return godHooked_;
    }
    if (!ensurePlayerHook(memory)) {
        lastError_ += "godmode нужен aobplayer. ";
        return false;
    }
    ensureHorseHook(memory);
    if (!godCave_.allocateNear(memory, godSite_)) {
        lastError_ += "aobgodmode: нет cave. ";
        return false;
    }

    const uptr cave = godCave_.base();
    const uptr flag = cave + kFlags;
    const uptr pins = playerCave_.base() + kPtrs + 8;
    uptr hins = cave + kPtrs;
    if (horseHooked_) {
        hins = horseCave_.base() + kPtrs + 8;
    } else {
        godCave_.writeU64(memory, kPtrs, 0);
    }

    CaveEmit e;
    e.base = cave;
    e.bytes({0x66, 0xC1, 0xE9, 0x08, 0x50});  // shr cx,8; push rax
    e.cmpRip1(flag);
    e.jne8("code");
    e.movRaxRip(pins);
    e.bytes({0x48, 0x3B, 0xC6});  // cmp rax, rsi  (1.17: rsi = ChrIns)
    e.je8("hit");
    e.bytes({0x48, 0x3B, 0x46, 0x08});  // cmp rax, [rsi+8]
    e.je8("hit");
    e.movRaxRip(hins);
    e.bytes({0x48, 0x3B, 0xC6});
    e.je8("hit");
    e.bytes({0x48, 0x3B, 0x46, 0x08});
    e.jne8("code");
    e.L("hit");
    e.bytes({0xB1, 0x01});
    e.L("code");
    e.bytes({0x58, 0x84, 0xC9});
    e.absJmp(godSite_ + 6);

    godCave_.writeU32(memory, kFlags, 0);
    if (!commitCave(memory, godCave_, godSite_, 6, e, "godmode")) {
        return false;
    }
    godHooked_ = true;
    return true;
}

bool GameHooks::installImmuneHook(ProcessMemory& memory) {
    if (immuneHooked_ || immuneSite_ == 0) {

    e.movRaxRip(pins);

    e.jne8("code");
    e.bytes({0x8B, 0x44, 0x91, 0x2C, 0x89, 0x44, 0x91, 0x10});
    e.L("code");
    e.bytes({0x8B, 0xC2, 0x8B, 0x44, 0x81, 0x2C});
    e.absJmp(immuneSite_ + 6);

    immuneCave_.writeU32(memory, kFlags, 0);
    if (!commitCave(memory, immuneCave_, immuneSite_, 6, e, "immunity")) {
        return false;
    }
    immuneHooked_ = true;
    return true;
}

bool GameHooks::installLocationHook(ProcessMemory& memory) {
    if (locHooked_ || locSite_ == 0) {
        return locHooked_;
    }
    if (!ensurePlayerHook(memory)) {
        lastError_ += "location нужен aobplayer. ";
        return false;
    }
    ensureHorseHook(memory);
    if (!locCave_.allocateNear(memory, locSite_)) {
        lastError_ += "aoblocation: нет cave. ";
        return false;
    }

    constexpr usize kLocFlags = 0x600;
    constexpr usize kLocPins = 0x610;
    constexpr usize kLocHins = 0x618;
    constexpr usize kLocPtr = 0x620;
    constexpr usize kSave1 = 0x630;
    constexpr usize kSave0 = 0x640;

    const uptr cave = locCave_.base();
    const uptr fly = cave + kLocFlags;
    const uptr up = cave + kLocFlags + 4;
    const uptr down = cave + kLocFlags + 8;
    const uptr frz = cave + kLocFlags + 12;
    const uptr pins = cave + kLocPins;
    const uptr hins = cave + kLocHins;
    const uptr locPtr = cave + kLocPtr;
    const uptr pdat = cave + 0x628;
    const uptr saveXmm1 = cave + kSave1;
    const uptr saveXmm0 = cave + kSave0;

    CaveEmit e;
    e.base = cave;
    e.movupsRipXmm1(saveXmm1);
    e.bytes({0x50, 0x68, 0x00, 0x00, 0x80, 0x3F});
    e.movRaxRip(pins);
    e.bytes({0x48, 0x39, 0xC3});
    e.je32("pl");
    e.bytes({0x48, 0x39, 0x43, 0x08});
    e.je32("pl");
    e.movRaxRip(hins);
    e.bytes({0x48, 0x39, 0xC3});
    e.je32("pl");
    e.bytes({0x48, 0x39, 0x43, 0x08});
    e.je32("pl");
    e.movRaxRip(pdat);
    e.bytes({0x48, 0x39, 0xC3});
    e.je32("pl");
    e.bytes({0x48, 0x39, 0x43, 0x08});
    e.jne32("notpl");
    e.L("pl");
    e.movRipRbx(locPtr);
    e.cmpRip1(fly);
    e.jne32("code");
    e.bytes({0xC6, 0x83, 0x92, 0x00, 0x00, 0x00, 0x01});
    e.bytes({0xC6, 0x83, 0x93, 0x00, 0x00, 0x00, 0x01});
    e.cmpRip1(up);
    e.jne32("noup");
    e.movImmRip0(up);
    e.bytes({0xF3, 0x0F, 0x10, 0x4B, 0x74, 0xF3, 0x0F, 0x58, 0x0C, 0x24, 0xF3, 0x0F, 0x11, 0x4B, 0x74});
    e.L("noup");
    e.cmpRip1(down);
    e.jne32("nodn");
    e.movImmRip0(down);
    e.bytes({0xF3, 0x0F, 0x10, 0x4B, 0x74, 0xF3, 0x0F, 0x5C, 0x0C, 0x24, 0xF3, 0x0F, 0x11, 0x4B, 0x74});
    e.L("nodn");
    e.movupsRipXmm0(saveXmm0);
    e.movssXmm1Rip(saveXmm0);
    e.bytes({0xF3, 0x0F, 0x11, 0x4B, 0x70});
    e.movssXmm1Rip(saveXmm0 + 8);
    e.bytes({0xF3, 0x0F, 0x11, 0x4B, 0x78, 0xC7, 0x43, 0x7C, 0x00, 0x00, 0x80, 0x3F});
    e.jmp32("skip");
    e.L("notpl");
    e.cmpRip1(frz);
    e.je32("skip");
    e.L("code");
    e.bytes({0x0F, 0x11, 0x43, 0x70, 0xC7, 0x43, 0x7C, 0x00, 0x00, 0x80, 0x3F});
    e.L("skip");
    e.bytes({0x48, 0x83, 0xC4, 0x08, 0x58});
    e.movupsXmm1Rip(saveXmm1);
    e.absJmp(locSite_ + 11);

    locCave_.writeU64(memory, kLocPins, 0);
    locCave_.writeU64(memory, kLocHins, 0);
    locCave_.writeU64(memory, 0x628, 0);
    locCave_.writeU32(memory, kLocFlags, 0);
    locCave_.writeU32(memory, kLocFlags + 4, 0);
    locCave_.writeU32(memory, kLocFlags + 8, 0);
    locCave_.writeU32(memory, kLocFlags + 12, 0);
    if (!commitCave(memory, locCave_, locSite_, 11, e, "location")) {
        return false;
    }
    locHooked_ = true;
    copyLocPtrs(memory);
    Logger::instance().info("location cave bytes=" + std::to_string(e.b.size()));
    return true;
}

bool GameHooks::installDmgHook(ProcessMemory& memory) {
    if (dmgHooked_ || dmgSite_ == 0) {
        return dmgHooked_;
    }
    if (!ensurePlayerHook(memory)) {
        lastError_ += "dmg нужен aobplayer. ";
        return false;
    }
    if (!memory.read(dmgSite_ + 3, dmgDisp_)) {
        return false;
    }
    ensureHorseHook(memory);
    if (!dmgCave_.allocateNear(memory, dmgSite_)) {
        lastError_ += "aobdmg: нет cave. ";
        return false;
    }

    const uptr cave = dmgCave_.base();
    const uptr dmg = cave + kFlags;
    const uptr dmul = cave + kFlags + 4;
    const uptr defm = cave + kFlags + 8;
    const uptr pdata = playerCave_.base() + kPtrs;
    const uptr hdata = horseHooked_ ? (horseCave_.base() + kPtrs) : (cave + kPtrs);

    CaveEmit e;
    e.base = cave;
    e.bytes({0x49, 0x8D, 0x96});  // lea rdx, [r14+disp]
    e.put32(dmgDisp_);
    e.bytes({0x83, 0x3A, 0x00});  // cmp dword [rdx], 0
    e.jle8("code");
    e.bytes({0x48, 0xB9});
    e.put64(pdata);
    e.bytes({0x4C, 0x39, 0x29});  // cmp [rcx], r13
    e.je8("pl");
    e.bytes({0x48, 0xB9});
    e.put64(hdata);
    e.bytes({0x4C, 0x39, 0x29});
    e.jne8("npl");
    e.L("pl");
    e.cmpRip0(defm);
    e.je8("code");
    e.bytes({0xDB, 0x02});
    e.fdivRip(defm);
    e.bytes({0xDB, 0x1A});
    e.jmp8("code");

    e.L("npl");
    e.cmpRip0(dmul);
    e.je8("ohk");
    e.bytes({0xDB, 0x02});
    e.fmulRip(dmul);
    e.bytes({0xDB, 0x1A});
    e.L("ohk");
    e.cmpRip1(dmg);
    e.jne8("code");
    e.bytes({0xC7, 0x02, 0x7F, 0x96, 0x98, 0x00});  // mov [rdx], 9999999

    e.L("code");
    e.bytes({0x8B, 0x12});  // mov edx, [rdx]
    e.absJmp(dmgSite_ + 7);

    if (!horseHooked_) {
        dmgCave_.writeU64(memory, kPtrs, 0);
    }
    dmgCave_.writeU32(memory, kFlags, 0);
    dmgCave_.writeU32(memory, kFlags + 4, 0);
    dmgCave_.writeU32(memory, kFlags + 8, 0);
    if (!commitCave(memory, dmgCave_, dmgSite_, 7, e, "dmg")) {
        return false;
    }
    dmgHooked_ = true;
    return true;
}

void GameHooks::tick(ProcessMemory& memory) {
    if (horseHooked_) {
        u32 flag = 0;
        u64 hdata = 0;
        memory.read(horseCave_.base() + kFlags, flag);
        memory.read(horseCave_.base() + kPtrs, hdata);
        if (flag == 1 && hdata != 0) {
            i32 mx = 0;
            if (memory.read(hdata + static_cast<uptr>(horseDisp_) + 4, mx) && mx > 0) {
                memory.write(hdata + static_cast<uptr>(horseDisp_), mx);
            }
        }
    }
    copyLocPtrs(memory);
}

void GameHooks::copyLocPtrs(ProcessMemory& memory) {
    if (!locHooked_ || !playerHooked_) {
        return;
    }
    u64 pins = 0;
    u64 pdat = 0;
    u64 hins = 0;
    memory.read(playerCave_.base() + kPtrs + 8, pins);
    memory.read(playerCave_.base() + kPtrs, pdat);
    locCave_.writeU64(memory, 0x610, pins);
    locCave_.writeU64(memory, 0x628, pdat);
    if (horseHooked_) {
        memory.read(horseCave_.base() + kPtrs + 8, hins);
        locCave_.writeU64(memory, 0x618, hins);
    }
}

bool GameHooks::installSpeedAt(ProcessMemory& memory, CodeCave& cave, uptr site, const char* name) {
    if (site == 0) {
        return false;
    }
    if (cave.active()) {
        return true;
    }
    if (!cave.allocateNear(memory, site)) {
        lastError_ += std::string(name) + ": нет cave. ";
        return false;
    }
    const uptr base = cave.base();
    const uptr flag = base + kFlags;
    const uptr mul = base + kFlags + 4;

    CaveEmit e;
    e.base = base;
    e.bytes({0xF3, 0x0F, 0x10, 0x88, 0xCC, 0x02, 0x00, 0x00});
    e.bytes({0xF3, 0x0F, 0x59, 0x88, 0x68, 0x02, 0x00, 0x00});
    e.cmpRip1(flag);
    e.jne8("done");
    e.mulssXmm1Rip(mul);
    e.L("done");
    e.absJmp(site + 16);

    cave.writeU32(memory, kFlags, 0);
    cave.writeU32(memory, kFlags + 4, 0x3F800000);
    return commitCave(memory, cave, site, 16, e, name);
}

void GameHooks::resolveCsFlipper(ProcessMemory& memory) {
    if (csFlipper_ != 0 || speedSite_ == 0) {
        return;
    }
    i32 disp = 0;
    if (!memory.read(speedSite_ - 4, disp)) {
        return;
    }
    const uptr slot = speedSite_ + static_cast<uptr>(static_cast<i64>(disp));
    u64 ptr = 0;
    if (!memory.read(slot, ptr) || ptr == 0) {
        return;
    }
    csFlipper_ = ptr;
    Logger::instance().info("CSFlipper=0x" + toHex(csFlipper_));
}

bool GameHooks::installSpeedHook(ProcessMemory& memory) {
    if (speedSite_ == 0 && speedSite2_ == 0) {
        return false;
    }
    const bool a = installSpeedAt(memory, speedCave_, speedSite_, "gamespeed");
    const bool b = installSpeedAt(memory, speedCave2_, speedSite2_, "gamespeed2");
    resolveCsFlipper(memory);
    speedHooked_ = a || b || speedSite_ != 0;
    if (!a && !b) {
        Logger::instance().warn("gamespeed: cave не встал, пишем CSFlipper+0x268");
    }
    return speedHooked_;
}

bool GameHooks::applyStealth(ProcessMemory& memory, bool on) {
    if (stealthSite_ == 0) {
        return false;
    }
    if (on && !stealthOn_) {
        const u8 patch[] = {0x66, 0x90};
        if (!memory.patch(stealthSite_, patch)) {
            return false;
        }
        stealthOn_ = true;
        Logger::instance().info("stealth ON @ 0x" + toHex(stealthSite_));
        return true;
    }
    if (!on && stealthOn_) {
        memory.restore(stealthSite_);
        stealthOn_ = false;
    }
    return true;
}

bool GameHooks::applyNoSoulLoss(ProcessMemory& memory, bool on) {
    if (noSoulLossSite_ == 0) {
        return false;
    }
    if (on && !noSoulOn_) {
        const u8 patch[] = {0xEB, 0x03};
        if (!memory.patch(noSoulLossSite_, patch)) {
            return false;
        }
        noSoulOn_ = true;
        Logger::instance().info("nosoullost ON @ 0x" + toHex(noSoulLossSite_));
        return true;
    }
    if (!on && noSoulOn_) {
        memory.restore(noSoulLossSite_);
        noSoulOn_ = false;
    }
    return true;
}

bool GameHooks::resolve(ProcessMemory& /*memory*/, SignatureScanner& scanner) {
    hooksAttempted_ = true;
    lastError_.clear();
    playerSite_ = soulsSite_ = godSite_ = weightSite_ = dropSite_ = stealthSite_ = 0;
    immuneSite_ = horseSite_ = daySite_ = locSite_ = itemSite_ = dmgSite_ = speedSite_ = speedSite2_ =
        noSoulLossSite_ = 0;
    csFlipper_ = 0;
    speedSaved_ = false;

    logPatternHits(scanner);
    Logger::instance().info("хуки ставлю только когда фичу включают");

    auto take = [&](uptr& dst, const Signature& a, const Signature* b = nullptr) {
        if (auto s = scanner.find(a, true)) {
            dst = *s;
            return;
        }
        if (b) {
            if (auto s = scanner.find(*b, true)) {
                dst = *s;
            }
        }
    };

    Signature pHook{"player_hook",
                    "8B 88 ?? ?? 00 00 89 ?? ?? ?? 00 00 ?? 8B ?? ?? ?? ?? 00 00 48 8B ?? E8 ?? ?? ?? ?? 89 ?? ?? ?? "
                    "00 00"};
    Signature pHookFb{
        "player_hook_fb",
        "8B 88 ?? ?? 00 00 89 ?? ?? ?? 00 00 ?? 8B ?? ?? ?? 00 00 48 8B ?? E8 ?? ?? ?? ?? 89 ?? ?? 00 00"};
    take(playerSite_, pHook, &pHookFb);

    Signature souls{"souls_hook", "44 8B 41 6C 45 33 D2 ?? 89 ?? 24 ?? 45 8D 0C 10"};
    Signature soulsFb{"souls_hook_fb", "44 8B 49 6C 45 33 DB ?? 89 ?? 24 ?? 45 8D 04 11"};
    if (auto hit = scanner.find(souls, true)) {
        soulsSite_ = *hit;
        soulsVariant41_ = true;
    } else if (auto hitFb = scanner.find(soulsFb, true)) {
        soulsSite_ = *hitFb;
        soulsVariant41_ = false;
    }

    Signature god{"godmode",
                  "66 C1 E9 08 84 C9 75 ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? ?? 48 8B CE E8 ?? ?? ?? ?? 83 ?? 03 0F 85 ?? ?? "
                  "00 00"};
    take(godSite_, god);

    Signature w{"weight", "FF C3 83 FB 05 7C ?? ?? ?? ?? ?? ?? 0F 28 C6"};
    take(weightSite_, w);
    Signature dr{"droprate", "41 0F 28 F8 48 85 D2 74 ?? BA ?? 00 00 00 ?? ?? ?? E8 ?? ?? ?? ?? 84 C0"};
    take(dropSite_, dr);
    Signature st{"stealth", "74 0C ?? 88 ?? ?? ?? 00 00 E9 ?? ?? 00 00 80 ?? ?? ?? 00 00 00 74"};
    take(stealthSite_, st);
    Signature im{"immunity", "8B C2 8B 44 81 2C C3"};
    take(immuneSite_, im);
    Signature hs{"horse", "8B 8A ?? ?? 00 00 89 ?? ?? ?? 8B ?? E8 ?? ?? ?? ?? 83 63 ?? FD"};
    take(horseSite_, hs);
    Signature dy{"daytime", "F3 0F 2C D0 85 D2 7E ?? 48 ?? ?? E8"};
    take(daySite_, dy);
    Signature loc{"location", "0F 11 43 70 C7 43 7C 00 00 80 3F 48 8B ?? 24"};
    take(locSite_, loc);

    Signature noLoss{"no_soul_loss", "8B 00 89 ?? 6C 48 8B ?? ?? ?? ?? ?? 48 85 ?? 75"};
    Signature noLoss2{"no_soul_loss2", "8B 00 89 ?? 6C 48 8B ?? ?? ?? ?? 48 85 ?? 75"};
    take(noSoulLossSite_, noLoss, &noLoss2);

    Signature item{"itemusage", "0F 42 E8 3B EB 8B D5 ?? ?? ?? ?? ?? 73 ?? E8"};
    Signature item2{"itemusage2", "0F 42 E8 3B EB 8B D5 ?? ?? ?? ?? ?? 73"};
    take(itemSite_, item, &item2);

    Signature dmg{"dmg1", "41 8B 96 ?? ?? 00 00 48 8B CE ?? 0F B6 ?? ?? ?? 00 00"};
    Signature dmg2{"dmg2", "41 8B 96 ?? ?? 00 00 49 8B CD ?? 0F B6 ?? ?? ?? 00 00"};
    Signature dmg3{"dmg3", "41 8B 96 ?? ?? 00 00 ?? ?? ?? ?? F3 ?? 0F 10 ?? ?? ?? 00 00 85"};
    if (auto hit = scanner.find(dmg, true)) {
        dmgSite_ = *hit;
    } else if (auto hit2 = scanner.find(dmg2, true)) {
        dmgSite_ = *hit2;
    } else if (auto hit3 = scanner.find(dmg3, true)) {
        dmgSite_ = *hit3;
    }

    Signature spd{"gamespeed", "F3 0F 10 88 CC 02 00 00 F3 0F 59 88 68 02 00 00"};
    {
        const auto hits = scanner.findAll(spd);
        if (!hits.empty()) {
            speedSite_ = hits[0];
        }
        if (hits.size() > 1) {
            speedSite2_ = hits[1];
        }
    }

    return playerSite_ != 0 || soulsSite_ != 0;
}

void GameHooks::shutdown(ProcessMemory& memory) {
    applyStealth(memory, false);
    applyNoSoulLoss(memory, false);
    if (speedSaved_ && csFlipper_ != 0) {
        memory.write(csFlipper_ + 0x268, speedOrig268_);
    }
    locCave_.removeHook(memory);
    godCave_.removeHook(memory);
    immuneCave_.removeHook(memory);
    dmgCave_.removeHook(memory);
    horseCave_.removeHook(memory);
    dayCave_.removeHook(memory);
    weightCave_.removeHook(memory);
    dropCave_.removeHook(memory);
    itemCave_.removeHook(memory);
    speedCave_.removeHook(memory);
    speedCave2_.removeHook(memory);
    playerCave_.removeHook(memory);
    soulsCave_.removeHook(memory);
    playerHooked_ = soulsHooked_ = godHooked_ = immuneHooked_ = horseHooked_ = false;
    dayHooked_ = locHooked_ = dmgHooked_ = weightHooked_ = dropHooked_ = itemHooked_ = speedHooked_ = false;
    csFlipper_ = 0;
    speedSaved_ = false;
}
```

```cpp
namespace er {

bool BytePatch::apply(ProcessMemory& memory, uptr addr, std::span<const u8> patched) {
    if (applied_ && addr_ == addr) {
        return true;
    }
    if (applied_) {
        revert(memory);
    }
    original_.assign(patched.size(), 0);
    if (!memory.read(addr, original_.data(), original_.size())) {
        original_.clear();
        return false;
    }
    if (!memory.write(addr, patched.data(), patched.size())) {
        original_.clear();
        return false;
    }
    addr_ = addr;
    applied_ = true;
    return true;
}

bool BytePatch::revert(ProcessMemory& memory) {
    if (!applied_) {
        return true;
    }
    const bool ok = memory.write(addr_, original_.data(), original_.size());
    if (ok) {
        applied_ = false;
        addr_ = 0;
        original_.clear();
    }
    return ok;
}

}  // namespace er
```

```cpp
#include <climits>
#include <cstring>
#include <vector>

namespace er {
namespace {

bool inRel32Range(uptr target, uptr hookSite) {
    const i64 rel = static_cast<i64>(target) - static_cast<i64>(hookSite + 5);
    return rel >= INT32_MIN && rel <= INT32_MAX;
}

}  // namespace

bool CodeCave::ensureAllocated(ProcessMemory& memory, usize size) {
    if (base_ != 0) {
        return true;
    }
    void* p = VirtualAllocEx(memory.handle(), nullptr, size, MEM_COMMIT | MEM_RESERVE, PAGE_EXECUTE_READWRITE);
    if (!p) {
        Logger::instance().error("VirtualAllocEx cave failed: " + lastWinError());
        return false;
    }
    base_ = reinterpret_cast<uptr>(p);
    return true;
}

bool CodeCave::allocateNear(ProcessMemory& memory, uptr hookSite, usize size) {
    if (!memory.isOpen()) {
        return false;
    }
    if (base_ != 0 && inRel32Range(base_, hookSite)) {
        return true;
    }
    if (base_ != 0) {
        VirtualFreeEx(memory.handle(), reinterpret_cast<void*>(base_), 0, MEM_RELEASE);
        base_ = 0;
    }

    SYSTEM_INFO si{};
    GetSystemInfo(&si);
    const usize gran = si.dwAllocationGranularity ? si.dwAllocationGranularity : 0x10000;
    const uptr alignedHook = hookSite & ~(gran - 1);

    auto tryAt = [&](uptr tryAddr) -> bool {
        tryAddr &= ~(gran - 1);
        if (tryAddr < 0x10000 || !inRel32Range(tryAddr, hookSite)) {
            return false;
        }
        void* p = VirtualAllocEx(memory.handle(), reinterpret_cast<void*>(tryAddr), size, MEM_COMMIT | MEM_RESERVE,
                                 PAGE_EXECUTE_READWRITE);
        if (!p) {
            return false;
        }
        const uptr cave = reinterpret_cast<uptr>(p);
        if (inRel32Range(cave, hookSite)) {
            base_ = cave;
            Logger::instance().info("Cave near: 0x" + toHex(cave) + " (hook 0x" + toHex(hookSite) + ")");
            return true;
        }
        VirtualFreeEx(memory.handle(), p, 0, MEM_RELEASE);
        return false;
    };

    for (usize step = 1; step < 0x7F00; ++step) {
        const uptr delta = step * gran;
        if (tryAt(alignedHook + delta)) {
            return true;
        }
        if (alignedHook > delta && tryAt(alignedHook - delta)) {
            return true;
        }
    }

    Logger::instance().error("Нет свободной страницы в ±2GB от хука 0x" + toHex(hookSite) +
                             ", инжект отменён (игра не патчится)");
    return false;
}

bool CodeCave::write(ProcessMemory& memory, usize offset, std::span<const u8> data) {
    if (base_ == 0 || data.empty()) {
        return false;
    }
    return memory.write(base_ + offset, data.data(), data.size());
}

bool CodeCave::writeU32(ProcessMemory& memory, usize offset, u32 value) {
    return write(memory, offset, std::span<const u8>(reinterpret_cast<const u8*>(&value), sizeof(value)));
}

bool CodeCave::writeU64(ProcessMemory& memory, usize offset, u64 value) {
    return write(memory, offset, std::span<const u8>(reinterpret_cast<const u8*>(&value), sizeof(value)));
}

bool CodeCave::installRelJmpHook(ProcessMemory& memory, uptr hookSite, usize patchLen) {
    if (base_ == 0 || patchLen < 5 || !inRel32Range(base_, hookSite)) {
        Logger::instance().error("installRelJmpHook: cave не в диапазоне rel32, отказ, без патча");
        return false;
    }

    hookOriginal_.assign(patchLen, 0);
    if (!memory.read(hookSite, hookOriginal_.data(), hookOriginal_.size())) {
        return false;
    }

    const i32 rel = static_cast<i32>(static_cast<i64>(base_) - static_cast<i64>(hookSite + 5));
    std::vector<u8> patch(patchLen, 0x90);
    patch[0] = 0xE9;
    std::memcpy(&patch[1], &rel, sizeof(rel));
    if (!memory.write(hookSite, patch.data(), patch.size())) {
        return false;
    }

    hookSite_ = hookSite;
    patchLen_ = patchLen;
    hooked_ = true;
    return true;
}

bool CodeCave::removeHook(ProcessMemory& memory) {
    if (!hooked_ || hookSite_ == 0 || hookOriginal_.empty()) {
        hooked_ = false;
        return true;
    }
    const bool ok = memory.write(hookSite_, hookOriginal_.data(), hookOriginal_.size());
    hookOriginal_.clear();
    hooked_ = false;
    hookSite_ = 0;
    if (base_ != 0 && memory.handle()) {
        VirtualFreeEx(memory.handle(), reinterpret_cast<void*>(base_), 0, MEM_RELEASE);
        base_ = 0;
    }
    return ok;
}
```

```cpp
namespace er {

std::optional<uptr> resolvePointerChain(const ProcessMemory& memory, uptr base, std::span<const i32> offsets) {
    if (offsets.empty()) {
        return base;
    }
    uptr addr = base;
    for (usize i = 0; i < offsets.size(); ++i) {
        addr += static_cast<uptr>(static_cast<i64>(offsets[i]));
        if (i + 1 == offsets.size()) {
            return addr;
        }
        uptr next = 0;
        if (!memory.read(addr, next) || next == 0) {
            return std::nullopt;
        }
        addr = next;
    }
    return addr;
}
```

```cpp
namespace er {

std::string lastWinError() {
    const DWORD code = GetLastError();
    wchar_t* buf = nullptr;
    const DWORD n = FormatMessageW(
        FORMAT_MESSAGE_ALLOCATE_BUFFER | FORMAT_MESSAGE_FROM_SYSTEM | FORMAT_MESSAGE_IGNORE_INSERTS,
        nullptr, code, MAKELANGID(LANG_NEUTRAL, SUBLANG_DEFAULT),
        reinterpret_cast<LPWSTR>(&buf), 0, nullptr);
    std::ostringstream o;
    o << "Win32 " << code;
    if (n && buf) {
        std::wstring w(buf, buf + n);
        while (!w.empty() && (w.back() == L'\r' || w.back() == L'\n' || w.back() == L' ')) {
            w.pop_back();
        }
        o << ": " << wideToUtf8(w);
        LocalFree(buf);
    }
    return o.str();
}

ProcessMemory::~ProcessMemory() {
    restoreAll();
    close();
}

bool ProcessMemory::open(DWORD pid) {
    close();
    HANDLE h = OpenProcess(
        PROCESS_QUERY_INFORMATION | PROCESS_VM_READ | PROCESS_VM_WRITE | PROCESS_VM_OPERATION,
        FALSE, pid);
    if (!h) {
        Logger::instance().error("OpenProcess pid=" + std::to_string(pid) + " failed: " + lastWinError());
        return false;
    }
    handle_ = h;
    pid_ = pid;
    return true;
}

void ProcessMemory::close() {
    if (handle_) {
        CloseHandle(handle_);
        handle_ = nullptr;
    }
    pid_ = 0;
    originals_.clear();
}

bool ProcessMemory::read(uptr addr, void* dst, usize size) const {
    if (!handle_ || !dst || size == 0) {
        return false;
    }
    SIZE_T got = 0;
    if (!ReadProcessMemory(handle_, reinterpret_cast<LPCVOID>(addr), dst, size, &got) || got != size) {
        return false;
    }
    return true;
}

bool ProcessMemory::write(uptr addr, const void* src, usize size) {
    if (!handle_ || !src || size == 0) {
        return false;
    }
    SIZE_T put = 0;
    if (WriteProcessMemory(handle_, reinterpret_cast<LPVOID>(addr), src, size, &put) && put == size) {
        FlushInstructionCache(handle_, reinterpret_cast<LPCVOID>(addr), size);
        return true;
    }

    DWORD oldProtect = 0;
    if (!VirtualProtectEx(handle_, reinterpret_cast<LPVOID>(addr), size, PAGE_EXECUTE_READWRITE, &oldProtect)) {
        Logger::instance().error("VirtualProtectEx failed at 0x" + toHex(addr) + " " + lastWinError());
        return false;
    }
    put = 0;
    const bool ok = WriteProcessMemory(handle_, reinterpret_cast<LPVOID>(addr), src, size, &put) && put == size;
    DWORD tmp = 0;
    VirtualProtectEx(handle_, reinterpret_cast<LPVOID>(addr), size, oldProtect, &tmp);
    if (ok) {
        FlushInstructionCache(handle_, reinterpret_cast<LPCVOID>(addr), size);
    } else {
        Logger::instance().error("WriteProcessMemory failed at 0x" + toHex(addr) + " " + lastWinError());
    }
    return ok;
}

bool ProcessMemory::patch(uptr addr, std::span<const u8> bytes) {
    if (bytes.empty()) {
        return false;
    }
    if (!originals_.contains(addr)) {
        std::vector<u8> orig(bytes.size());
        if (!read(addr, orig.data(), orig.size())) {
            Logger::instance().error("patch: cannot read original bytes");
            return false;
        }
        originals_.emplace(addr, std::move(orig));
    }
    return write(addr, bytes.data(), bytes.size());
}

bool ProcessMemory::restore(uptr addr) {
    const auto it = originals_.find(addr);
    if (it == originals_.end()) {
        return true;
    }
    const bool ok = write(addr, it->second.data(), it->second.size());
    if (ok) {
        originals_.erase(it);
    }
    return ok;
}

void ProcessMemory::restoreAll() {
    if (!handle_) {
        originals_.clear();
        return;
    }
    for (auto it = originals_.begin(); it != originals_.end();) {
        if (write(it->first, it->second.data(), it->second.size())) {
            it = originals_.erase(it);
        } else {
            Logger::instance().error("restoreAll failed for one patch");
            ++it;
        }
    }
}
```

```cpp
int hexNibble(char c) {
    if (c >= '0' && c <= '9') {
        return c - '0';
    }
    if (c >= 'a' && c <= 'f') {
        return c - 'a' + 10;
    }
    if (c >= 'A' && c <= 'F') {
        return c - 'A' + 10;
    }
    return -1;
}

std::optional<SignatureScanner::Pattern> SignatureScanner::parseAob(std::string_view aob) {
    Pattern pat;
    usize i = 0;
    while (i < aob.size()) {
        while (i < aob.size() && std::isspace(static_cast<unsigned char>(aob[i]))) {
            ++i;
        }
        if (i >= aob.size()) {
            break;
        }
        if (i + 1 < aob.size() && aob[i] == '?' && aob[i + 1] == '?') {
            pat.bytes.push_back(0);
            pat.mask.push_back(0);
            i += 2;
            continue;
        }
        const int hi = hexNibble(aob[i]);
        const int lo = (i + 1 < aob.size()) ? hexNibble(aob[i + 1]) : -1;
        if (hi < 0 || lo < 0) {
            return std::nullopt;
        }
        pat.bytes.push_back(static_cast<u8>((hi << 4) | lo));
        pat.mask.push_back(0xFF);
        i += 2;
    }
    if (pat.bytes.empty()) {
        return std::nullopt;
    }
    return pat;
}

std::vector<uptr> SignatureScanner::scanAll(const Pattern& pat) const {
    std::vector<uptr> hits;
    if (!memory_.isOpen() || text_.base == 0 || text_.size < pat.bytes.size()) {
        return hits;
    }

    constexpr usize kChunk = 1024 * 1024;
    std::vector<u8> buf;
    buf.resize(kChunk + pat.bytes.size());

    for (usize off = 0; off < text_.size; ) {
        const usize remain = text_.size - off;
        const usize readSize = remain < buf.size() ? remain : buf.size();
        if (!memory_.read(text_.base + off, buf.data(), readSize)) {
            Logger::instance().warn("AOB: не удалось прочитать .text по смещению " + std::to_string(off));
            off += kChunk;
            continue;
        }
        const usize searchLen = readSize >= pat.bytes.size() ? readSize - pat.bytes.size() + 1 : 0;
        for (usize i = 0; i < searchLen; ++i) {
            bool ok = true;
            for (usize j = 0; j < pat.bytes.size(); ++j) {
                if (pat.mask[j] && buf[i + j] != pat.bytes[j]) {
                    ok = false;
                    break;
                }
            }
            if (ok) {
                hits.push_back(text_.base + off + i);
            }
        }
        if (remain <= kChunk) {
            break;
        }
        off += kChunk;
    }
    return hits;
}

std::optional<uptr> SignatureScanner::find(const Signature& sig, bool requireUnique) {
    if (const auto it = cache_.find(sig.name); it != cache_.end()) {
        return it->second;
    }
    const auto pat = parseAob(sig.aob);
    if (!pat) {
        Logger::instance().error("Сигнатура '" + sig.name + "': некорректный AOB");
        return std::nullopt;
    }
    const auto hits = scanAll(*pat);
    if (hits.empty()) {
        Logger::instance().warn("Сигнатура '" + sig.name + "': не найдена");
        return std::nullopt;
    }
    if (requireUnique && hits.size() != 1) {
        Logger::instance().warn("Сигнатура '" + sig.name + "': не уникальна (" +
                                std::to_string(hits.size()) + " совпадений), расширьте маску");
        return std::nullopt;
    }
    cache_[sig.name] = hits.front();
    Logger::instance().info("Сигнатура '" + sig.name + "' найдена: 0x" + toHex(hits.front()));
    return hits.front();
}

std::vector<uptr> SignatureScanner::findAll(const Signature& sig) {
    const auto pat = parseAob(sig.aob);
    if (!pat) {
        Logger::instance().error("Сигнатура '" + sig.name + "': некорректный AOB");
        return {};
    }
    auto hits = scanAll(*pat);
    Logger::instance().info("Сигнатура '" + sig.name + "': " + std::to_string(hits.size()) + " совпадений");
    for (usize i = 0; i < hits.size(); ++i) {
        Logger::instance().info("  [" + std::to_string(i) + "] 0x" + toHex(hits[i]));
    }
    if (!hits.empty()) {
        cache_[sig.name] = hits.front();
    }
    return hits;
}

std::optional<uptr> SignatureScanner::findWithFallback(const Signature& primary, const Signature& fallback) {
    if (auto a = find(primary, true)) {
        return a;
    }
    Logger::instance().info("Пробуем fallback для '" + primary.name + "'");
    return find(fallback, true);
}

std::optional<uptr> SignatureScanner::resolveRip(uptr insnAddr, int dispOffset, int insnLength) const {
    i32 disp = 0;
    if (!memory_.read(insnAddr + static_cast<uptr>(dispOffset), disp)) {
        return std::nullopt;
    }
    return insnAddr + static_cast<uptr>(insnLength) + static_cast<uptr>(static_cast<i64>(disp));
}

std::optional<uptr> SignatureScanner::cached(std::string_view name) const {
    const auto it = cache_.find(std::string(name));
    if (it == cache_.end()) {
        return std::nullopt;
    }
    return it->second;
}
```

```cpp
std::optional<GpuTexture> TextureLoader::createFromRgba(std::span<const u8> rgba, int w, int h) {
    if (!device_ || rgba.empty() || w <= 0 || h <= 0) {
        return std::nullopt;
    }
    D3D11_TEXTURE2D_DESC desc{};
    desc.Width = static_cast<UINT>(w);
    desc.Height = static_cast<UINT>(h);
    desc.MipLevels = 1;
    desc.ArraySize = 1;
    desc.Format = DXGI_FORMAT_R8G8B8A8_UNORM;
    desc.SampleDesc.Count = 1;
    desc.Usage = D3D11_USAGE_IMMUTABLE;
    desc.BindFlags = D3D11_BIND_SHADER_RESOURCE;

    D3D11_SUBRESOURCE_DATA sub{};
    sub.pSysMem = rgba.data();
    sub.SysMemPitch = static_cast<UINT>(w) * 4;

    ID3D11Texture2D* tex = nullptr;
    if (FAILED(device_->CreateTexture2D(&desc, &sub, &tex)) || !tex) {
        return std::nullopt;
    }

    ID3D11ShaderResourceView* srv = nullptr;
    const HRESULT hr = device_->CreateShaderResourceView(tex, nullptr, &srv);
    tex->Release();
    if (FAILED(hr) || !srv) {
        return std::nullopt;
    }
    owned_.push_back(srv);
    return GpuTexture{srv, w, h};
}

std::optional<GpuTexture> TextureLoader::loadFromMemory(std::span<const u8> data) {
    if (!device_ || data.empty()) {
        return std::nullopt;
    }

    IWICImagingFactory* factory = nullptr;
    if (FAILED(CoCreateInstance(CLSID_WICImagingFactory, nullptr, CLSCTX_INPROC_SERVER,
                                IID_PPV_ARGS(&factory)))) {
        return std::nullopt;
    }

    IWICStream* stream = nullptr;
    if (FAILED(factory->CreateStream(&stream))) {
        factory->Release();
        return std::nullopt;
    }
    if (FAILED(stream->InitializeFromMemory(const_cast<BYTE*>(data.data()), static_cast<DWORD>(data.size())))) {
        stream->Release();
        factory->Release();
        return std::nullopt;
    }

    IWICBitmapDecoder* decoder = nullptr;
    if (FAILED(factory->CreateDecoderFromStream(stream, nullptr, WICDecodeMetadataCacheOnLoad, &decoder))) {
        stream->Release();
        factory->Release();
        return std::nullopt;
    }

    IWICBitmapFrameDecode* frame = nullptr;
    if (FAILED(decoder->GetFrame(0, &frame))) {
        decoder->Release();
        stream->Release();
        factory->Release();
        return std::nullopt;
    }

    IWICFormatConverter* converter = nullptr;
    if (FAILED(factory->CreateFormatConverter(&converter))) {
        frame->Release();
        decoder->Release();
        stream->Release();
        factory->Release();
        return std::nullopt;
    }
    if (FAILED(converter->Initialize(frame, GUID_WICPixelFormat32bppRGBA, WICBitmapDitherTypeNone, nullptr, 0.0,
                                     WICBitmapPaletteTypeCustom))) {
        converter->Release();
        frame->Release();
        decoder->Release();
        stream->Release();
        factory->Release();
        return std::nullopt;
    }

    UINT w = 0, h = 0;
    converter->GetSize(&w, &h);
    std::vector<u8> rgba(static_cast<usize>(w) * static_cast<usize>(h) * 4);
    if (FAILED(converter->CopyPixels(nullptr, w * 4, static_cast<UINT>(rgba.size()), rgba.data()))) {
        converter->Release();
        frame->Release();
        decoder->Release();
        stream->Release();
        factory->Release();
        return std::nullopt;
    }

    converter->Release();
    frame->Release();
    decoder->Release();
    stream->Release();
    factory->Release();

    return createFromRgba(rgba, static_cast<int>(w), static_cast<int>(h));
}

std::optional<GpuTexture> TextureLoader::loadBmpFromBlob(std::span<const u8> blob) {
    for (usize i = 0; i + 54 < blob.size(); ++i) {
        if (blob[i] != 'B' || blob[i + 1] != 'M') {
            continue;
        }
        const i32 fileSize = *reinterpret_cast<const i32*>(&blob[i + 2]);
        const i32 dataOff = *reinterpret_cast<const i32*>(&blob[i + 10]);
        const i32 dibSize = *reinterpret_cast<const i32*>(&blob[i + 14]);
        if (dibSize < 40 || dataOff < 14 || fileSize < 1024 || static_cast<usize>(i + fileSize) > blob.size()) {
            continue;
        }
        const int w = *reinterpret_cast<const i32*>(&blob[i + 18]);
        int h = *reinterpret_cast<const i32*>(&blob[i + 22]);
        const u16 bpp = *reinterpret_cast<const u16*>(&blob[i + 28]);
        if (w <= 0 || w > 2048 || bpp != 24 && bpp != 32) {
            continue;
        }
        const bool flip = h > 0;
        const int ah = flip ? h : -h;
        if (ah < 32 || ah > 2048) {
            continue;
        }
        const usize rowStride = ((static_cast<usize>(w) * bpp + 31) / 32) * 4;
        const usize pixelBase = i + static_cast<usize>(dataOff);
        if (pixelBase + rowStride * static_cast<usize>(ah) > blob.size()) {
            continue;
        }
        std::vector<u8> rgba(static_cast<usize>(w) * static_cast<usize>(ah) * 4);
        for (int y = 0; y < ah; ++y) {
            const int srcY = flip ? (ah - 1 - y) : y;
            const u8* row = blob.data() + pixelBase + static_cast<usize>(srcY) * rowStride;
            u8* dst = rgba.data() + static_cast<usize>(y) * static_cast<usize>(w) * 4;
            for (int x = 0; x < w; ++x) {
                if (bpp == 24) {
                    dst[x * 4 + 0] = row[x * 3 + 2];
                    dst[x * 4 + 1] = row[x * 3 + 1];
                    dst[x * 4 + 2] = row[x * 3 + 0];
                    dst[x * 4 + 3] = 255;
                } else {
                    dst[x * 4 + 0] = row[x * 4 + 2];
                    dst[x * 4 + 1] = row[x * 4 + 1];
                    dst[x * 4 + 2] = row[x * 4 + 0];
                    dst[x * 4 + 3] = row[x * 4 + 3];
                }
            }
        }
        if (w > 64 && ah > 64) {
            return createFromRgba(rgba, w, ah);
        }
    }
    return std::nullopt;
}
```

```cpp
ImVec4 v4(ImU32 c) {
    return ImVec4(((c >> 0) & 255) / 255.0f, ((c >> 8) & 255) / 255.0f, ((c >> 16) & 255) / 255.0f,
                  ((c >> 24) & 255) / 255.0f);
}

bool toggleSwitch(const char* id, bool on, bool disabled) {
    ImGui::PushID(id);
    const float w = 40.0f;
    const float h = 22.0f;
    const ImVec2 p = ImGui::GetCursorScreenPos();
    ImGui::InvisibleButton("##sw", ImVec2(w, h));
    const bool clicked = !disabled && ImGui::IsItemClicked();
    const bool hover = ImGui::IsItemHovered();
    ImDrawList* dl = ImGui::GetWindowDrawList();
    const     ImU32 track = disabled ? IM_COL32(32, 36, 38, 255)
                           : (on ? kAccent : (hover ? IM_COL32(48, 56, 58, 255) : IM_COL32(38, 44, 46, 255)));
    dl->AddRectFilled(p, ImVec2(p.x + w, p.y + h), track, h * 0.5f);
    dl->AddRect(p, ImVec2(p.x + w, p.y + h), disabled ? kLine : (on ? kAccentHi : kLine), h * 0.5f, 0, 1.0f);
    const float knob = h - 6.0f;
    const float kx = on ? (p.x + w - 3.0f - knob) : (p.x + 3.0f);
    dl->AddCircleFilled(ImVec2(kx + knob * 0.5f, p.y + h * 0.5f), knob * 0.5f,
                        disabled ? IM_COL32(90, 94, 92, 255) : (on ? kBone : IM_COL32(180, 184, 180, 255)), 20);
    ImGui::PopID();
    return clicked;
}

bool pillButton(const char* id, const char* label, bool disabled) {
    ImGui::PushID(id);
    const ImVec2 ts = ImGui::CalcTextSize(label);
    const ImVec2 size(ts.x + 22.0f, 26.0f);
    const ImVec2 p = ImGui::GetCursorScreenPos();
    ImGui::InvisibleButton("##pb", size);
    const bool hover = ImGui::IsItemHovered();
    const bool down = ImGui::IsItemActive();
    const bool clicked = !disabled && ImGui::IsItemClicked();
    ImDrawList* dl = ImGui::GetWindowDrawList();
    ImU32 fill = hover ? IM_COL32(40, 56, 50, 255) : IM_COL32(28, 36, 34, 255);
    if (down) {
        fill = kAccentDim;
    }
    if (disabled) {
        fill = IM_COL32(24, 26, 28, 255);
    }
    dl->AddRectFilled(p, ImVec2(p.x + size.x, p.y + size.y), fill, 4.0f);
    dl->AddRect(p, ImVec2(p.x + size.x, p.y + size.y), disabled ? kLine : kAccent, 4.0f, 0, 1.0f);
    const ImU32 tc = disabled ? kMuted : kBone;
    dl->AddText(ImVec2(p.x + (size.x - ts.x) * 0.5f, p.y + (size.y - ts.y) * 0.5f), tc, label);
    ImGui::PopID();
    return clicked;
}

std::string windowsFontPath(const char* file) {
    wchar_t dir[MAX_PATH]{};
    if (GetWindowsDirectoryW(dir, MAX_PATH) == 0) {
        return std::string("C:\\Windows\\Fonts\\") + file;
    }
    return wideToUtf8(std::wstring(dir) + L"\\Fonts\\") + file;
}

bool fontFileExists(const std::string& path) {
    const std::wstring w = utf8ToWide(path);
    const DWORD attr = GetFileAttributesW(w.c_str());
    return attr != INVALID_FILE_ATTRIBUTES && (attr & FILE_ATTRIBUTE_DIRECTORY) == 0;
}

ImFont* tryFont(ImGuiIO& io, const char* const* names, int count, float size, const ImWchar* ranges) {
    for (int i = 0; i < count; ++i) {
        const std::string path = windowsFontPath(names[i]);
        if (!fontFileExists(path)) {
            continue;
        }
        if (ImFont* f = io.Fonts->AddFontFromFileTTF(path.c_str(), size, nullptr, ranges)) {
            Logger::instance().info(std::string("Шрифт UI: ") + path);
            return f;
        }
    }
    return nullptr;
}

void clampWindowRect(int& x, int& y, int& w, int& h) {
    RECT wa{};
    if (!SystemParametersInfoW(SPI_GETWORKAREA, 0, &wa, 0)) {
        wa.left = 0;
        wa.top = 0;
        wa.right = GetSystemMetrics(SM_CXSCREEN);
        wa.bottom = GetSystemMetrics(SM_CYSCREEN);
    }
    const int availW = std::max(kMinW, static_cast<int>(wa.right - wa.left));
    const int availH = std::max(kMinH, static_cast<int>(wa.bottom - wa.top));
    if (w <= 0) {
        w = kDefaultW;
    }
    if (h <= 0) {
        h = kDefaultH;
    }
    w = std::clamp(w, kMinW, availW);
    h = std::clamp(h, kMinH, availH);
    if (x < wa.left || y < wa.top || x + 80 > wa.right || y + 40 > wa.bottom) {
        x = wa.left + (availW - w) / 2;
        y = wa.top + (availH - h) / 2;
    }
}

bool captionBtn(const char* id, const char* glyph, ImU32 hoverBg) {
    ImGui::PushID(id);
    const ImVec2 sz(32.0f, 24.0f);
    const ImVec2 p = ImGui::GetCursorScreenPos();
    ImGui::InvisibleButton("##cap", sz);
    const bool hover = ImGui::IsItemHovered();
    const bool click = ImGui::IsItemClicked();
    ImDrawList* dl = ImGui::GetWindowDrawList();
    if (hover) {
        dl->AddRectFilled(p, ImVec2(p.x + sz.x, p.y + sz.y), hoverBg, 3.0f);
    }
    const ImVec2 ts = ImGui::CalcTextSize(glyph);
    dl->AddText(ImVec2(p.x + (sz.x - ts.x) * 0.5f, p.y + (sz.y - ts.y) * 0.5f), hover ? kBone : kMuted, glyph);
    ImGui::PopID();
    return click;
}

bool langChip(const char* label, bool selected) {
    const ImVec2 ts = ImGui::CalcTextSize(label);
    const ImVec2 size(ts.x + 16.0f, 24.0f);
    const ImVec2 p = ImGui::GetCursorScreenPos();
    ImGui::InvisibleButton(label, size);
    const bool clicked = ImGui::IsItemClicked();
    ImDrawList* dl = ImGui::GetWindowDrawList();
    if (selected) {
        dl->AddRectFilled(p, ImVec2(p.x + size.x, p.y + size.y), kAccentDim, 3.0f);
    }
    dl->AddText(ImVec2(p.x + 8.0f, p.y + (size.y - ts.y) * 0.5f), selected ? kBone : kMuted, label);
    return clicked;
}

UiRenderer::UiRenderer(GameAttach& game, FeatureManager& features)
    : game_(game), features_(features) {}

UiRenderer::~UiRenderer() {
    if (hwnd_) {
        ImGui_ImplDX11_Shutdown();
        ImGui_ImplWin32_Shutdown();
        ImGui::DestroyContext();
        destroyDevice();
        DestroyWindow(hwnd_);
        UnregisterClassW(kClassName, GetModuleHandleW(nullptr));
        hwnd_ = nullptr;
    }
}

bool UiRenderer::create(int x, int y, int w, int h) {
    ImGui_ImplWin32_EnableDpiAwareness();
    clampWindowRect(x, y, w, h);

    wc_ = {};
    wc_.cbSize = sizeof(wc_);
    wc_.style = CS_CLASSDC;
    wc_.lpfnWndProc = wndProc;
    wc_.hInstance = GetModuleHandleW(nullptr);
    wc_.lpszClassName = kClassName;
    wc_.hCursor = LoadCursor(nullptr, IDC_ARROW);
    wc_.hIcon = LoadIconW(wc_.hInstance, MAKEINTRESOURCEW(IDI_APP_ICON));
    wc_.hIconSm = static_cast<HICON>(LoadImageW(
        wc_.hInstance, MAKEINTRESOURCEW(IDI_APP_ICON), IMAGE_ICON,
        GetSystemMetrics(SM_CXSMICON), GetSystemMetrics(SM_CYSMICON), 0));
    RegisterClassExW(&wc_);

    hwnd_ = CreateWindowExW(WS_EX_APPWINDOW, kClassName, L"ERTrainer",
                            WS_POPUP | WS_THICKFRAME | WS_MINIMIZEBOX, x, y, w, h, nullptr, nullptr, wc_.hInstance,
                            this);
    if (!hwnd_) {
        Logger::instance().error("CreateWindowEx failed: " + lastWinError());
        return false;
    }
    if (wc_.hIcon) {
        SendMessageW(hwnd_, WM_SETICON, ICON_BIG, reinterpret_cast<LPARAM>(wc_.hIcon));
    }
    if (wc_.hIconSm) {
        SendMessageW(hwnd_, WM_SETICON, ICON_SMALL, reinterpret_cast<LPARAM>(wc_.hIconSm));
    }

    BOOL dark = TRUE;
    DwmSetWindowAttribute(hwnd_, 20, &dark, sizeof(dark)); // Win10 1809+; на старых просто no-op

    if (!createDevice()) {
        Logger::instance().error("Direct3D 11 недоступен");
        return false;
    }

    IMGUI_CHECKVERSION();
    ImGui::CreateContext();
    ImGuiIO& io = ImGui::GetIO();
    io.ConfigFlags |= ImGuiConfigFlags_NavEnableKeyboard;
    io.IniFilename = nullptr;

    applyTheme();
    if (!loadFonts()) {
        Logger::instance().warn("Шрифт с кириллицей не загрузился");
    }

    ImGui_ImplWin32_Init(hwnd_);
    ImGui_ImplDX11_Init(device_, context_);

    ShowWindow(hwnd_, SW_SHOWDEFAULT);
    UpdateWindow(hwnd_);
    running_ = true;
    return true;
}

bool UiRenderer::loadFonts() {
    ImGuiIO& io = ImGui::GetIO();
    const char* body[] = {"segoeui.ttf", "tahoma.ttf", "arial.ttf", "calibri.ttf", "micross.ttf", "times.ttf"};
    const char* title[] = {"seguisb.ttf", "segoeuib.ttf", "tahomabd.ttf", "arialbd.ttf", "segoeui.ttf", "tahoma.ttf"};
    const char* mono[] = {"consola.ttf", "cour.ttf", "lucon.ttf", "consolab.ttf"};

    fontBody_ = tryFont(io, body, IM_ARRAYSIZE(body), 15.5f, io.Fonts->GetGlyphRangesCyrillic());
    if (!fontBody_) {
        fontBody_ = io.Fonts->AddFontDefault();
        Logger::instance().warn("Системный шрифт с кириллицей не найден, берём встроенный");
    }
    fontTitle_ = tryFont(io, title, IM_ARRAYSIZE(title), 20.0f, io.Fonts->GetGlyphRangesCyrillic());
    fontMono_ = tryFont(io, mono, IM_ARRAYSIZE(mono), 13.5f, io.Fonts->GetGlyphRangesDefault());
    if (!fontMono_) {
        fontMono_ = fontBody_;
    }
    io.FontDefault = fontBody_;
    return fontBody_ != nullptr;
}

bool UiRenderer::createDevice() {
    DXGI_SWAP_CHAIN_DESC sd{};
    sd.BufferCount = 2;
    sd.BufferDesc.Format = DXGI_FORMAT_R8G8B8A8_UNORM;
    sd.BufferUsage = DXGI_USAGE_RENDER_TARGET_OUTPUT;
    sd.OutputWindow = hwnd_;
    sd.SampleDesc.Count = 1;
    sd.Windowed = TRUE;

    const D3D_FEATURE_LEVEL levels[] = {D3D_FEATURE_LEVEL_11_0, D3D_FEATURE_LEVEL_10_1, D3D_FEATURE_LEVEL_10_0};
    const D3D_DRIVER_TYPE drivers[] = {D3D_DRIVER_TYPE_HARDWARE, D3D_DRIVER_TYPE_WARP, D3D_DRIVER_TYPE_REFERENCE};
    const DXGI_SWAP_EFFECT effects[] = {DXGI_SWAP_EFFECT_DISCARD, DXGI_SWAP_EFFECT_SEQUENTIAL};

    D3D_FEATURE_LEVEL got{};
    for (const D3D_DRIVER_TYPE drv : drivers) {
        for (const DXGI_SWAP_EFFECT fx : effects) {
            sd.SwapEffect = fx;
            swap_ = nullptr;
            device_ = nullptr;
            context_ = nullptr;
            if (SUCCEEDED(D3D11CreateDeviceAndSwapChain(nullptr, drv, nullptr, 0, levels, 3, D3D11_SDK_VERSION, &sd,
                                                        &swap_, &device_, &got, &context_))) {
                Logger::instance().info(std::string("D3D11: driver=") +
                                        (drv == D3D_DRIVER_TYPE_HARDWARE ? "hardware"
                                         : drv == D3D_DRIVER_TYPE_WARP   ? "WARP"
                                                                         : "reference"));
                createRenderTarget();
                return rtv_ != nullptr;
            }
        }
    }
    return false;
}

void UiRenderer::createRenderTarget() {
    ID3D11Texture2D* back = nullptr;
    swap_->GetBuffer(0, IID_PPV_ARGS(&back));
    if (back) {
        device_->CreateRenderTargetView(back, nullptr, &rtv_);
        back->Release();
    }
}

void UiRenderer::destroyRenderTarget() {
    if (rtv_) {
        rtv_->Release();
        rtv_ = nullptr;
    }
}

void UiRenderer::destroyDevice() {
    destroyRenderTarget();
    if (swap_) {
        swap_->Release();
        swap_ = nullptr;
    }
    if (context_) {
        context_->Release();
        context_ = nullptr;
    }
    if (device_) {
        device_->Release();
        device_ = nullptr;
    }
}

void UiRenderer::requestClose() {
    running_ = false;
}

void UiRenderer::drawBrandMark(float x, float y, float size) {
    ImDrawList* dl = ImGui::GetWindowDrawList();
    const ImVec2 origin(x, y);
    const ImVec2 c(origin.x + size * 0.5f, origin.y + size * 0.5f);
    const float r = size * 0.42f;
    dl->AddCircle(c, r, kAccent, 40, 1.6f);
    dl->AddCircle(c, r * 0.62f, kAccentDim, 32, 1.2f);
    dl->AddLine(ImVec2(c.x, origin.y + 3.0f), ImVec2(c.x, origin.y + size - 3.0f), kAccentHi, 1.4f);
    dl->AddCircleFilled(c, 2.4f, kAccentHi, 12);
}

void UiRenderer::drawHeader(const AttachSnapshot& snap) {
    ImGui::BeginChild("##hdr", ImVec2(0, kHeaderH), ImGuiChildFlags_None, ImGuiWindowFlags_NoScrollbar);
    ImDrawList* dl = ImGui::GetWindowDrawList();
    const ImVec2 p0 = ImGui::GetWindowPos();
    const ImVec2 p1(p0.x + ImGui::GetWindowWidth(), p0.y + kHeaderH);
    dl->AddRectFilled(p0, p1, kHeader);
    dl->AddLine(ImVec2(p0.x, p1.y - 1.0f), ImVec2(p1.x, p1.y - 1.0f), kLine, 1.0f);

    ImGui::SetCursorPos(ImVec2(16.0f, 11.0f));
    const ImVec2 mark = ImGui::GetCursorScreenPos();
    drawBrandMark(mark.x, mark.y, 36.0f);
    ImGui::Dummy(ImVec2(36.0f, 36.0f));
    ImGui::SameLine(0, 12.0f);

    ImGui::BeginGroup();
    if (fontTitle_) {
        ImGui::PushFont(fontTitle_);
    }
    ImGui::SetCursorPosY(10.0f);
    ImGui::TextColored(v4(kBone), "ERTrainer");
    if (fontTitle_) {
        ImGui::PopFont();
    }
    ImGui::SetCursorPosY(32.0f);
    ImGui::TextColored(v4(kMuted), uiLang_ == 0 ? "Elden Ring  1.17.2" : "Elden Ring  1.17.2");
    ImGui::EndGroup();

    const float right = ImGui::GetWindowWidth() - 8.0f;
    float x = right;

    auto place = [&](float w) {
        x -= w;
        ImGui::SetCursorPos(ImVec2(x, 17.0f));
        return x;
    };

    place(32.0f);
    if (captionBtn("cls", "x", IM_COL32(110, 42, 42, 255))) {
        PostMessageW(hwnd_, WM_CLOSE, 0, 0);
    }
    place(32.0f);
    if (captionBtn("min", "-", IM_COL32(42, 50, 52, 255))) {
        ShowWindow(hwnd_, SW_MINIMIZE);
    }

    place(36.0f + 40.0f);
    if (langChip("RU", uiLang_ == 0)) {
        uiLang_ = 0;
    }
    ImGui::SameLine(0, 4.0f);
    if (langChip("EN", uiLang_ == 1)) {
        uiLang_ = 1;
    }

    const bool attached = snap.attached;
    const char* st = attached
                         ? (snap.eacLoaded ? (uiLang_ == 0 ? "EAC" : "EAC")
                                           : (uiLang_ == 0 ? "в игре" : "in-game"))
                         : (uiLang_ == 0 ? "ожидание" : "waiting");
    const ImVec2 sts = ImGui::CalcTextSize(st);
    const float lampW = 12.0f + sts.x + 18.0f;
    place(lampW + 16.0f);
    {
        const ImVec2 lp = ImGui::GetCursorScreenPos();
        const float pulse = attached && !snap.eacLoaded
                                ? 1.0f
                                : (0.45f + 0.55f * (0.5f + 0.5f * std::sinf(static_cast<float>(ImGui::GetTime()) * 3.0f)));
        ImU32 lamp = snap.eacLoaded ? kDanger : (attached ? kAccent : kWarn);
        lamp = (lamp & 0x00FFFFFF) | (static_cast<ImU32>(pulse * 255.0f) << 24);
        dl->AddCircleFilled(ImVec2(lp.x + 6.0f, lp.y + 12.0f), 4.2f, lamp, 16);
        dl->AddText(ImVec2(lp.x + 16.0f, lp.y + 4.0f), attached ? kBone : kMuted, st);
        ImGui::Dummy(ImVec2(lampW, 24.0f));
    }

    // тащим за шапку, кнопки справа не трогаем
    if (ImGui::IsMouseHoveringRect(p0, ImVec2(p1.x - 210.0f, p1.y)) && ImGui::IsMouseClicked(0)) {
        ReleaseCapture();
        SendMessageW(hwnd_, WM_NCLBUTTONDOWN, HTCAPTION, 0);
    }

    ImGui::EndChild();
}

void UiRenderer::drawFooter(const AttachSnapshot& snap) {
    ImGui::BeginChild("##ftr", ImVec2(0, kFooterH), ImGuiChildFlags_None, ImGuiWindowFlags_NoScrollbar);
    ImDrawList* dl = ImGui::GetWindowDrawList();
    const ImVec2 p0 = ImGui::GetWindowPos();
    const ImVec2 p1(p0.x + ImGui::GetWindowWidth(), p0.y + kFooterH);
    dl->AddRectFilled(p0, p1, kHeader);
    dl->AddLine(ImVec2(p0.x, p0.y), ImVec2(p1.x, p0.y), kLine, 1.0f);

    int on = 0;
    int ready = 0;
    for (const auto& f : features_.features()) {
        if (f->status() == FeatureStatus::Enabled) {
            ++on;
        } else if (f->status() == FeatureStatus::Ready) {
            ++ready;
        }
    }

    ImGui::SetCursorPos(ImVec2(16.0f, 8.0f));
    if (snap.attached) {
        ImGui::TextColored(v4(kMuted), "PID %u    %s    %s    %s %d/%d", snap.pid, toHex(snap.base).c_str(),
                           snap.version.displayVersion.c_str(), uiLang_ == 0 ? "вкл" : "on", on, on + ready);
    } else {
        ImGui::TextColored(v4(kMuted), "%s",
                           uiLang_ == 0 ? "Запусти eldenring.exe офлайн, без EAC"
                                        : "Start eldenring.exe offline, without EAC");
    }
    ImGui::EndChild();
}

void UiRenderer::drawFeatureRow(const FeatureInfo& info) {
    IFeature* feat = features_.find(info.id);
    const bool disabled = !feat || feat->status() == FeatureStatus::Unavailable;
    const bool enabled = feat && feat->status() == FeatureStatus::Enabled;
    auto& cfg = Config::instance();
    Hotkey hk = info.defaultHotkey;
    if (const auto it = cfg.hotkeys.find(info.id); it != cfg.hotkeys.end()) {
        hk = it->second;
    }

    ImGui::PushID(info.id.c_str());
    const float w = ImGui::GetContentRegionAvail().x;
    const ImVec2 p = ImGui::GetCursorScreenPos();
    const bool hover = ImGui::IsWindowHovered() &&
                       ImGui::IsMouseHoveringRect(p, ImVec2(p.x + w, p.y + kRowH - 3.0f), false);
    ImDrawList* dl = ImGui::GetWindowDrawList();
    ImU32 bg = enabled ? kRowOn : (hover ? kRowHover : kRow);
    dl->AddRectFilled(p, ImVec2(p.x + w, p.y + kRowH - 3.0f), bg, 5.0f);
    if (enabled) {
        dl->AddRectFilled(p, ImVec2(p.x + 3.0f, p.y + kRowH - 3.0f), kAccent, 2.0f);
    }

    float cx = p.x + 12.0f;
    const float cy = p.y + (kRowH - 3.0f) * 0.5f;

    const std::string hkStr = hk.toString();
    if (fontMono_) {
        ImGui::PushFont(fontMono_);
    }
    const ImVec2 hks = ImGui::CalcTextSize(hkStr.c_str());
    const ImVec2 chip0(cx, cy - 11.0f);
    const ImVec2 chip1(cx + hks.x + 14.0f, cy + 11.0f);
    dl->AddRectFilled(chip0, chip1, kChip, 3.0f);
    dl->AddRect(chip0, chip1, kLine, 3.0f, 0, 1.0f);
    dl->AddText(ImVec2(cx + 7.0f, cy - hks.y * 0.5f), kAccentHi, hkStr.c_str());
    if (fontMono_) {
        ImGui::PopFont();
    }
    cx = chip1.x + 14.0f;

    const char* label = uiLang_ == 0 ? info.nameRu.c_str() : info.nameEn.c_str();
    const ImVec2 ls = ImGui::CalcTextSize(label);
    dl->AddText(ImVec2(cx, cy - ls.y * 0.5f), disabled ? kMuted : kBone, label);

    if (hover) {
        std::string tip = std::string(uiLang_ == 0 ? info.nameRu : info.nameEn);
        tip += "\n";
        tip += feat ? std::string(feat->reason()) : (uiLang_ == 0 ? "нет сигнатуры" : "no signature");
        ImGui::SetTooltip("%s", tip.c_str());
    }

    const bool hasInt = info.kind == FeatureKind::ApplyValue || info.defaultInt != 0;
    const bool hasFloat = info.defaultFloat != 0.0f &&
                          (info.kind == FeatureKind::Toggle || info.id.find("multiplier") != std::string::npos ||
                           info.id == "game_speed");
    const bool hasHold = info.kind == FeatureKind::Hold;
    const bool hasToggle = info.kind == FeatureKind::Toggle;

    float controls = 0.0f;
    if (hasToggle) {
        controls += 48.0f;
    }
    if (hasInt) {
        controls += 188.0f;
    }
    if (hasFloat) {
        controls += 118.0f;
    }
    if (hasHold) {
        controls += 72.0f;
    }

    const float controlsX = p.x + w - controls - 12.0f;
    ImGui::SetCursorScreenPos(ImVec2(controlsX, p.y + (kRowH - 26.0f) * 0.5f - 1.0f));

    if (disabled) {
        ImGui::BeginDisabled();
    }

    if (hasInt) {
        int val = info.defaultInt;
        if (const auto it = cfg.featureInt.find(info.id); it != cfg.featureInt.end()) {
            val = it->second;
        }
        ImGui::SetNextItemWidth(108.0f);
        ImGui::InputInt("##v", &val, 0, 0);
        cfg.featureInt[info.id] = val;
        ImGui::SameLine(0, 6.0f);
        if (pillButton("ap", uiLang_ == 0 ? "Задать" : "Set", disabled)) {
            features_.requestApply(info.id);
        }
    } else if (hasHold) {
        if (pillButton("hold", uiLang_ == 0 ? "Жми" : "Hold", disabled)) {
            features_.requestApply(info.id);
        }
    }

    if (hasFloat) {
        float val = info.defaultFloat;
        if (const auto it = cfg.featureFloat.find(info.id); it != cfg.featureFloat.end()) {
            val = it->second;
        }
        ImGui::SetNextItemWidth(88.0f);
        ImGui::InputFloat("##f", &val, 0.1f, 1.0f, "%.1f");
        cfg.featureFloat[info.id] = val;
        if (hasToggle) {
            ImGui::SameLine(0, 8.0f);
        }
    }

    if (hasToggle) {
        if (toggleSwitch("tog", enabled, disabled) && feat) {
            features_.requestEnable(info.id, !enabled);
        }
    }

    if (disabled) {
        ImGui::EndDisabled();
    }

    if (!disabled && hasToggle && feat && hover && ImGui::IsMouseClicked(ImGuiMouseButton_Left) &&
        ImGui::GetIO().MousePos.x < controlsX) {
        features_.requestEnable(info.id, !enabled);
    }

    ImGui::SetCursorScreenPos(p);
    ImGui::Dummy(ImVec2(w, kRowH));
    ImGui::PopID();
}

void UiRenderer::drawFeatureList() {
    ImGui::BeginChild("##feats", ImVec2(0, 0), ImGuiChildFlags_None);

    const auto snap = game_.snapshot();
    if (snap.version.warning) {
        ImGui::SetCursorPos(ImVec2(18.0f, 8.0f));
        ImGui::TextColored(v4(kWarn), "%s", snap.version.warningText.c_str());
    }
    if (snap.eacLoaded) {
        ImGui::SetCursorPosX(18.0f);
        ImGui::TextColored(v4(kDanger),
                           "%s", uiLang_ == 0 ? "EasyAntiCheat загружен. Играй офлайн, без EAC."
                                              : "EasyAntiCheat is loaded. Play offline, without EAC.");
    }

    const FeatureGroup groups[] = {FeatureGroup::Survival, FeatureGroup::Combat, FeatureGroup::World,
                                   FeatureGroup::Items, FeatureGroup::Stats};

    ImGui::SetCursorPosY(ImGui::GetCursorPosY() + 8.0f);
    for (const FeatureGroup g : groups) {
        ImGui::SetCursorPosX(18.0f);
        const char* title = uiLang_ == 0 ? groupTitleRu(g) : groupTitleEn(g);
        ImGui::TextColored(v4(kAccentHi), "%s", title);
        const ImVec2 lp = ImGui::GetItemRectMax();
        ImDrawList* dl = ImGui::GetWindowDrawList();
        dl->AddLine(ImVec2(lp.x + 10.0f, lp.y - 8.0f),
                    ImVec2(ImGui::GetWindowPos().x + ImGui::GetWindowWidth() - 18.0f, lp.y - 8.0f), kLine, 1.0f);

        ImGui::SetCursorPosX(14.0f);
        ImGui::BeginGroup();
        for (const auto& f : features_.features()) {
            if (f->info().group != g) {
                continue;
            }
            drawFeatureRow(f->info());
        }
        ImGui::EndGroup();
        ImGui::Dummy(ImVec2(0, 10.0f));
    }

    ImGui::Dummy(ImVec2(0, 12.0f));
    ImGui::EndChild();
}

int UiRenderer::run() {
    while (running_) {
        MSG msg{};
        while (PeekMessageW(&msg, nullptr, 0, 0, PM_REMOVE)) {
            TranslateMessage(&msg);
            DispatchMessageW(&msg);
            if (msg.message == WM_QUIT) {
                running_ = false;
            }
        }
        if (!running_) {
            break;
        }

        if (resizeW_ && resizeH_) {
            destroyRenderTarget();
            swap_->ResizeBuffers(0, resizeW_, resizeH_, DXGI_FORMAT_UNKNOWN, 0);
            resizeW_ = resizeH_ = 0;
            createRenderTarget();
        }

        ImGui_ImplDX11_NewFrame();
        ImGui_ImplWin32_NewFrame();
        ImGui::NewFrame();
        draw();
        ImGui::Render();

        const float clear[4] = {12.0f / 255.0f, 14.0f / 255.0f, 16.0f / 255.0f, 1};
        context_->OMSetRenderTargets(1, &rtv_, nullptr);
        context_->ClearRenderTargetView(rtv_, clear);
        ImGui_ImplDX11_RenderDrawData(ImGui::GetDrawData());
        swap_->Present(1, 0);
    }

    RECT rc{};
    if (hwnd_ && GetWindowRect(hwnd_, &rc)) {
        auto& w = Config::instance().window;
        w.x = rc.left;
        w.y = rc.top;
        w.w = rc.right - rc.left;
        w.h = rc.bottom - rc.top;
        Config::instance().save();
    }
    return 0;
}

LRESULT CALLBACK UiRenderer::wndProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam) {
    if (ImGui_ImplWin32_WndProcHandler(hwnd, msg, wParam, lParam)) {
        return true;
    }
    UiRenderer* self = nullptr;
    if (msg == WM_NCCREATE) {
        auto* cs = reinterpret_cast<CREATESTRUCTW*>(lParam);
        self = static_cast<UiRenderer*>(cs->lpCreateParams);
        SetWindowLongPtrW(hwnd, GWLP_USERDATA, reinterpret_cast<LONG_PTR>(self));
    } else {
        self = reinterpret_cast<UiRenderer*>(GetWindowLongPtrW(hwnd, GWLP_USERDATA));
    }

    switch (msg) {
        case WM_NCHITTEST: {
            POINT pt{static_cast<LONG>(static_cast<short>(LOWORD(lParam))),
                     static_cast<LONG>(static_cast<short>(HIWORD(lParam)))};
            RECT wr{};
            GetWindowRect(hwnd, &wr);
            const int b = 6;
            const bool left = pt.x < wr.left + b;
            const bool right = pt.x >= wr.right - b;
            const bool top = pt.y < wr.top + b;
            const bool bottom = pt.y >= wr.bottom - b;
            if (top && left) {
                return HTTOPLEFT;
            }
            if (top && right) {
                return HTTOPRIGHT;
            }
            if (bottom && left) {
                return HTBOTTOMLEFT;
            }
            if (bottom && right) {
                return HTBOTTOMRIGHT;
            }
            if (left) {
                return HTLEFT;
            }
            if (right) {
                return HTRIGHT;
            }
            if (top) {
                return HTTOP;
            }
            if (bottom) {
                return HTBOTTOM;
            }
            return HTCLIENT;
        }
        case WM_GETMINMAXINFO:
            if (lParam) {
                auto* mmi = reinterpret_cast<MINMAXINFO*>(lParam);
                mmi->ptMinTrackSize.x = kMinW;
                mmi->ptMinTrackSize.y = kMinH;
            }
            return 0;
        case WM_SIZE:
            if (self && wParam != SIZE_MINIMIZED) {
                self->resizeW_ = LOWORD(lParam);
                self->resizeH_ = HIWORD(lParam);
            }
            return 0;
        case WM_DESTROY:
            PostQuitMessage(0);
            return 0;
        default:
            break;
    }
    return DefWindowProcW(hwnd, msg, wParam, lParam);
}
```

```cpp
#include "worker/WorkerThread.h"

#include "core/Logger.h"

#include <chrono>

namespace er {

WorkerThread::WorkerThread(GameAttach& game, FeatureManager& features)
    : game_(game), features_(features) {}

WorkerThread::~WorkerThread() {
    stop();
}

void WorkerThread::start() {
    stop_ = false;
    thread_ = std::thread([this] { loop(); });
}

void WorkerThread::stop() {
    stop_ = true;
    if (thread_.joinable()) {
        thread_.join();
    }
}

void WorkerThread::loop() {
    using clock = std::chrono::steady_clock;
    auto nextAttach = clock::now();
    auto nextTick = clock::now();
    bool resolved = false;
    bool hooksResolved = false;

    Logger::instance().info("Рабочий поток запущен");
    while (!stop_) {
        const auto now = clock::now();
        if (now >= nextAttach) {
            game_.tick();
            const auto snap = game_.snapshot();
            if (snap.attached && !snap.eacLoaded) {
                features_.bindMemory(&game_.memory());
                if (!hooksResolved) {
                    features_.resolveAll(game_.scanner());
                    hooksResolved = true;
                    Logger::instance().info("сигнатуры прогнал");
                }
                resolved = true;
            }
            if (!snap.attached) {
                if (resolved) {
                    features_.shutdown();
                }
                resolved = false;
                hooksResolved = false;
            }
            nextAttach = now + std::chrono::milliseconds(500);
        }
        if (now >= nextTick) {
            const auto snap = game_.snapshot();
            if (snap.attached && !snap.eacLoaded) {
                features_.tick(1.0f / 30.0f);
            }
            nextTick = now + std::chrono::milliseconds(33);
        }
        std::this_thread::sleep_for(std::chrono::milliseconds(5));
    }
    features_.shutdown();
    game_.detach();
    Logger::instance().info("Рабочий поток остановлен, патчи восстановлены");
}

}  // namespace er
```
