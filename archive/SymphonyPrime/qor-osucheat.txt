osu! aim assist externe (Windows, Python 3.9+)
Usage prévu : serveur privé / entraînement hors ligne.

Principe :
  - tosu (ou gosumemory) tourne en parallèle et fournit via WebSocket le temps
    de la chanson, le chemin de la beatmap et les mods.
  - On lit le fichier .osu pour connaître la position/temps des hit circles.
  - Pendant que TU bouges et cliques, l'outil ajoute une petite correction de
    trajectoire vers le prochain cercle (réglable).

Dépendances : pip install websocket-client customtkinter
Réglages osu! recommandés : Raw input OFF, "Améliorer la précision du pointeur"
(Windows) OFF, sensibilité osu! à 1.0 -> le curseur du jeu = curseur Windows.

def keep_active_in_background():
    """
    Empêche Windows de brider le script quand sa fenêtre n'est pas au premier
    plan (mode éco / timers dégradés) et l'empêche de passer en veille.
    """
    k32 = ctypes.windll.kernel32
    k32.GetCurrentProcess.restype = wintypes.HANDLE
    h = k32.GetCurrentProcess()

    class PROCESS_POWER_THROTTLING_STATE(ctypes.Structure):
        _fields_ = [("Version", wintypes.ULONG),
                    ("ControlMask", wintypes.ULONG),
                    ("StateMask", wintypes.ULONG)]

    try:  # désactive le mode éco et le bridage de la résolution du timer
        st = PROCESS_POWER_THROTTLING_STATE(1, 0x1 | 0x4, 0)
        k32.SetProcessInformation(h, 4, ctypes.byref(st), ctypes.sizeof(st))
    except Exception:
        pass
    try:  # priorité haute
        k32.SetPriorityClass(h, 0x00000080)
    except Exception:
        pass
    try:  # pas de mise en veille pendant que le script tourne
        k32.SetThreadExecutionState(0x80000000 | 0x00000001 | 0x00000002)
    except Exception:
        pass
    try:  # timers à ~1 ms
        ctypes.windll.winmm.timeBeginPeriod(1)
    except Exception:
        pass

def move_abs(x, y):
    """Vrai évènement souris ABSOLU (visible par osu! même en raw input)."""
    w, h = user32.GetSystemMetrics(0), user32.GetSystemMetrics(1)
    inp = INPUT()
    inp.type = 0
    inp.mi.dx = int(x * 65535 / max(1, w - 1))
    inp.mi.dy = int(y * 65535 / max(1, h - 1))
    inp.mi.dwFlags = MOUSEEVENTF_MOVE | MOUSEEVENTF_ABSOLUTE
    inp.mi.dwExtraInfo = MAGIC
    user32.SendInput(1, ctypes.byref(inp), ctypes.sizeof(INPUT))

def send_key(scan, up=False):
    inp = INPUT()
    inp.type = 1  # INPUT_KEYBOARD
    inp.ki.wVk = 0
    inp.ki.wScan = scan
    inp.ki.dwFlags = KEYEVENTF_SCANCODE | (KEYEVENTF_KEYUP if up else 0)
    user32.SendInput(1, ctypes.byref(inp), ctypes.sizeof(INPUT))


def send_mouse_button(down):
    inp = INPUT()
    inp.type = 0
    inp.mi.dwFlags = 0x0002 if down else 0x0004  # LEFTDOWN / LEFTUP
    user32.SendInput(1, ctypes.byref(inp), ctypes.sizeof(INPUT))


def key_down(vk):

def parse_osu(path):
    """Retourne (circle_size, objs, ends).
    objs = [(temps_ms, x, y), ...] : cercles + têtes de sliders (spinners ignorés)
    ends = [temps_de_fin_ms, ...]  : égal au temps pour un cercle, fin du slider sinon
    """
    cs, slider_mult, section = 5.0, 1.4, ""
    timing, raw = [], []
    with open(path, encoding="utf-8", errors="ignore") as f:
        for line in f:
            line = line.strip()
            if line.startswith("["):
                section = line
                continue
            if not line:
                continue
            if section == "[Difficulty]":
                if line.startswith("CircleSize:"):
                    cs = float(line.split(":")[1])
                elif line.startswith("SliderMultiplier:"):
                    slider_mult = float(line.split(":")[1])
            elif section == "[TimingPoints]":
                p = line.split(",")
                if len(p) >= 2:
                    uninherited = len(p) < 7 or p[6].strip() == "1"
                    timing.append((float(p[0]), float(p[1]), uninherited))
            elif section == "[HitObjects]":
                p = line.split(",")
                if len(p) >= 4:
                    raw.append(p)
    timing.sort(key=lambda x: x[0])

    def slider_duration(t, length, slides):
        beat, sv = 500.0, 1.0
        for tt, bl, uninh in timing:
            if tt > t:
                break
            if uninh:
                beat, sv = bl, 1.0
            else:
                sv = -100.0 / bl if bl < 0 else 1.0
        return length / (slider_mult * 100.0 * sv) * beat * slides

    items = []
    for p in raw:
        x, y, t, typ = float(p[0]), float(p[1]), int(p[2]), int(p[3])
        if typ & 8:  # spinner -> ignoré
            continue
        end = t
        if typ & 2 and len(p) >= 8:  # slider
            try:
                end = t + slider_duration(t, float(p[7]), int(p[6]))
            except Exception:
                end = t
        items.append((t, x, y, end))
    items.sort()
    return cs, [(t, x, y) for t, x, y, _ in items], [e for *_, e in items]


def circle_radius_osu_px(cs, mods):
    if mods & 16:   # HR
        cs = min(cs * 1.3, 10)
    if mods & 2:    # EZ
        cs *= 0.5
    return 54.4 - 4.48 * cs


# ------------------------------------------------------------------ Updater
def _ver(v):
    try:
        return tuple(int(x) for x in str(v).strip().lstrip("vV").split("."))
    except ValueError:
        return (0,)


class Updater:
    """
    Vérifie un manifeste JSON publié par toi et remplace le script (ou l'exe)
    par la nouvelle version, après contrôle HTTPS + SHA-256.

    Manifeste (version.json) :
      {"version": "1.1.0", "url": "https://.../osu_aim_assist.py",
       "sha256": "...", "notes": "ce qui change",
       "exe_url": "...", "exe_sha256": "..."}      <- exe_* seulement si tu distribues un .exe
    """
    MAX_BYTES = 150 * 1024 * 1024

    def __init__(self, eng):
        self.eng = eng
        self.state = "idle"   # idle / checking / available / uptodate / installing / error / disabled
        self.message = ""
        self.manifest = None
        self.busy = threading.Lock()

    @property
    def frozen(self):
        return bool(getattr(sys, "frozen", False))

    def _get(self, url, limit):
        if not url.lower().startswith("https://"):
            raise ValueError("l'adresse de mise à jour doit être en HTTPS")
        req = urllib.request.Request(url, headers={"User-Agent": "aimassist-updater"})
        with urllib.request.urlopen(req, timeout=15) as r:
            data = r.read(limit + 1)
        if len(data) > limit:
            raise ValueError("fichier de mise à jour trop gros")
        return data

    def check(self):
        if not UPDATE_MANIFEST_URL:
            self.state, self.message = "disabled", "Mises à jour non configurées"
            return False
        if not self.busy.acquire(blocking=False):
            return False
        try:
            self.state, self.message = "checking", "Vérification..."
            m = json.loads(self._get(UPDATE_MANIFEST_URL, 1024 * 1024).decode("utf-8"))
            if _ver(m["version"]) > _ver(APP_VERSION):
                self.manifest = m
                self.state = "available"
                self.message = f"Version {m['version']} disponible"
                return True
            self.state, self.message = "uptodate", f"À jour (v{APP_VERSION})"
            return False
        except Exception as e:
            self.state, self.message = "error", f"Échec de la vérification : {type(e).__name__}"
            return False
        finally:
            self.busy.release()

    def install(self):
        """Télécharge, vérifie, remplace, puis relance. Ne revient pas si ça marche."""
        m = self.manifest
        if not m or not self.busy.acquire(blocking=False):
            return
        try:
            self.state, self.message = "installing", "Téléchargement..."
            url = m.get("exe_url") if self.frozen else m.get("url")
            want = (m.get("exe_sha256") if self.frozen else m.get("sha256")) or ""
            if not url or not want:
                raise ValueError("manifeste incomplet (url / sha256)")
            data = self._get(url, self.MAX_BYTES)
            if hashlib.sha256(data).hexdigest().lower() != want.strip().lower():
                raise ValueError("empreinte SHA-256 incorrecte, mise à jour refusée")

            if self.frozen:
                self._apply_exe(data)
            else:
                compile(data.decode("utf-8"), "update", "exec")  # refuse un fichier cassé
                self._apply_script(data)
            self.message = "Redémarrage..."
            self.eng.running = False       # relâche proprement les touches / hooks
            time.sleep(0.4)
            os._exit(0)
        except Exception as e:
            self.state, self.message = "error", f"Mise à jour échouée : {e}"
        finally:
            self.busy.release()

    def _apply_script(self, data):
        path = os.path.abspath(__file__)
        folder = os.path.dirname(path)
        fd, tmp = tempfile.mkstemp(dir=folder, suffix=".new")
        with os.fdopen(fd, "wb") as f:
            f.write(data)
        try:
            os.replace(path, path + ".bak")           # sauvegarde de l'ancienne version
        except OSError:
            pass
        os.replace(tmp, path)
        subprocess.Popen([sys.executable] + sys.argv, cwd=os.getcwd(), close_fds=True)

    def _apply_exe(self, data):
        exe = os.path.abspath(sys.executable)
        new = exe + ".new"
        with open(new, "wb") as f:
            f.write(data)
        bat = os.path.join(tempfile.gettempdir(), "aimassist_update.bat")
        with open(bat, "w") as f:
            f.write('@echo off\r\nping 127.0.0.1 -n 3 >nul\r\n'
                    f'move /y "{new}" "{exe}"\r\n'
                    f'start "" "{exe}"\r\n'
                    'del "%~f0"\r\n')
        subprocess.Popen(["cmd", "/c", bat], creationflags=0x08000000, close_fds=True)

class Engine:
    def __init__(self, cfg):
        self.cfg = cfg
        self.lock = threading.Lock()
        self.playing = False
        self.t, self.stamp, self.rate = 0.0, time.perf_counter(), 1.0
        self._clk_off = 0.0           # décalage horloge locale -> temps de la map (ms)
        self._last_change = 0.0       # dernier instant où le temps de la map a changé
        self._relax_dbg = "—"
        self.mods = 0
        self.map_path = None
        self.times, self.objs, self.cs = [], [], 5.0
        self.ends = []
        self._relax_clicks = 0
        self.status = "En attente de tosu..."
        self.running = True
        self._toggle_latch = False
        self._rx = self._ry = 0.0
        # mode tablette : position réelle du stylo + décalage ajouté par l'assistance
        self._pen = (0.0, 0.0)
        self._last_set = None
        self._ox = self._oy = 0.0
        self._test = False          # F9 : décalage fixe de 150 px pour diagnostiquer
        self._test_latch = False
        self._pen_seen = False      # True dès que le hook a reçu un évènement du stylo
        self._hook_count = 0        # nombre d'évènements du stylo vus par le hook (diagnostic)
        self._dbg = "pas de cible"  # diagnostic : ce que voit le calcul d'assistance

    # ---- WebSocket (tosu)
    def ws_thread(self):
        while self.running:
            try:
                ws = websocket.create_connection(TOSU_URL, timeout=5)
                self.status = "Connecté à tosu"
                while self.running:
                    self.on_data(json.loads(ws.recv()))
            except Exception as e:
                self.status = f"tosu indisponible ({type(e).__name__}), nouvel essai..."
                time.sleep(2)

    def on_data(self, d):
        menu = d.get("menu", {})
        bm = menu.get("bm", {})
        t = float(bm.get("time", {}).get("current", 0))
        mods = int(menu.get("mods", {}).get("num", 0))
        playing = menu.get("state") == 2
        path = None
        try:
            songs = d["settings"]["folders"]["songs"]
            path = os.path.join(songs, bm["path"]["folder"], bm["path"]["file"])
        except KeyError:
            pass
        rate = 1.5 if mods & 64 else (0.75 if mods & 256 else 1.0)
        with self.lock:
            now = time.perf_counter()
            if t != self.t:
                # Les messages de tosu arrivent avec un retard variable. Le décalage
                # (temps_map - horloge_locale) est donc sous-estimé par chaque message :
                # on garde le plus grand (= le moins retardé) et on redescend lentement.
                off_new = t - now * 1000.0 * rate
                if off_new >= self._clk_off or abs(off_new - self._clk_off) > 300:
                    self._clk_off = off_new
                else:
                    self._clk_off += (off_new - self._clk_off) * 0.02
                self.t, self.stamp, self._last_change = t, now, now
            self.playing, self.mods, self.rate = playing, mods, rate
        if path and path != self.map_path and os.path.exists(path):
            self.map_path = path
            cs, objs, ends = parse_osu(path)
            with self.lock:
                self.cs, self.objs, self.ends = cs, objs, ends
                self.times = [o[0] for o in objs]
            self.status = f"Map chargée : {os.path.basename(path)} ({len(objs)} objets)"

    def now_ms(self):
        """Temps de la map en ms, interpolé entre deux messages de tosu."""
        with self.lock:
            t, rate, off, last = self.t, self.rate, self._clk_off, self._last_change
        now = time.perf_counter()
        if now - last > 0.3:          # le temps ne bouge plus : pause / menu
            return t
        return now * 1000.0 * rate + off

    # ---- conversion playfield osu -> écran
    def to_screen(self, x, y, mods):
        c = self.cfg
        w, h = user32.GetSystemMetrics(0), user32.GetSystemMetrics(1)
        s = (h / 480.0) * c["scale_mult"]
        if mods & 16:  # HR : inversion verticale
            y = 384 - y
        sx = (w - 512 * s) / 2 + x * s + c["offset_x"]
        sy = (h - 384 * s) / 2 + y * s + c["offset_y"]
        return sx, sy, s

    def test_keys(self):
        """Envoie quelques appuis Z/X (ou clics) dans 3 s pour vérifier qu'ils arrivent."""
        def run():
            time.sleep(3)
            for _ in range(3):
                if self.cfg["relax_input"] == "mouse":
                    send_mouse_button(True); time.sleep(0.05); send_mouse_button(False)
                else:
                    for scan in (SCAN_Z, SCAN_X):
                        send_key(scan); time.sleep(0.05); send_key(scan, True); time.sleep(0.15)
                time.sleep(0.3)
        threading.Thread(target=run, daemon=True).start()

    # ---- mode Relax : clics automatiques (le joueur ne fait que viser)
    def relax_thread(self):
        """
        Clique à l'instant du hit quand le curseur est sur le cercle.
        Clavier : alternance Z / X (tient la touche pendant un slider).
        Souris  : clic gauche.
        """
        scans = (SCAN_Z, SCAN_X)
        held = [False, False]
        held_until = [0.0, 0.0]
        done, last_t, last_k, latch = -1, 0.0, 1, False

        def press(k):
            if self.cfg["relax_input"] == "mouse":
                send_mouse_button(True)
            else:
                send_key(scans[k], False)
            held[k] = True

        def release(k):
            if self.cfg["relax_input"] == "mouse":
                send_mouse_button(False)
            else:
                send_key(scans[k], True)
            held[k] = False

        while self.running:
            time.sleep(0.001)
            c = self.cfg

            if key_down(c["relax_toggle_vk"]):   # F10
                if not latch:
                    c["relax_enabled"] = not c["relax_enabled"]
                latch = True
            else:
                latch = False

            t = self.now_ms()
            for k in (0, 1):                      # relâcher les touches arrivées à échéance
                if held[k] and t >= held_until[k]:
                    release(k)

            with self.lock:
                times, objs, ends = self.times, self.objs, self.ends
                mods, cs, rate = self.mods, self.cs, self.rate
            if not (c["relax_enabled"] and self.playing and objs):
                self._relax_dbg = ("désactivé" if not c["relax_enabled"] else
                                   "pas en partie" if not self.playing else "aucune map")
                for k in (0, 1):
                    if held[k]:
                        release(k)
                continue

            if t < last_t - 300:                  # retry / retour en arrière dans la map
                done = bisect.bisect_left(times, t - 100) - 1
            last_t = t

            off, win = c["relax_offset_ms"], c["relax_window_ms"]
            i = done + 1
            while i < len(objs) and times[i] + off + win < t:   # trop tard : objet passé
                i += 1
            done = max(done, i - 1)
            if i >= len(objs):
                self._relax_dbg = "fin de la map"
                continue
            ot, ox, oy = objs[i]
            if t < ot + off:
                self._relax_dbg = f"prochain objet dans {ot + off - t:.0f} ms"
                continue

            if c["relax_require_inside"]:
                tx, ty, s = self.to_screen(ox, oy, mods)
                r_px = circle_radius_osu_px(cs, mods) * s
                cx, cy = cursor_pos()
                d = math.hypot(tx - cx, ty - cy)
                if d > c["relax_zone"] * r_px:
                    self._relax_dbg = (f"curseur à {d / max(r_px, 1):.1f} rayons du cercle "
                                       f"(zone {c['relax_zone']:.2f}) — pas de clic")
                    continue                      # pas sur le cercle : on réessaie jusqu'à la fin de la fenêtre

            order = (0,) if c["relax_input"] == "mouse" else ((1, 0) if last_k == 0 else (0, 1))
            k = next((k for k in order if not held[k]), None)
            if k is None:
                continue
            press(k)
            held_until[k] = max(t + c["relax_tap_ms"] * rate, ends[i])
            last_k, done = k, i
            self._relax_clicks += 1
            self._relax_dbg = (f"clic {'Z' if k == 0 else 'X'} envoyé, "
                               f"écart {t - ot:+.0f} ms par rapport au hit")

        for k in (0, 1):                          # à l'arrêt : ne jamais laisser une touche bloquée
            if held[k]:
                release(k)

    # ---- hook souris bas niveau (tablettes à haute fréquence)
    def hook_thread(self):
        """
        Intercepte chaque évènement de déplacement du stylo (jusqu'à 1000/s).
        Pour chacun : on retient la position réelle du stylo, on bloque
        l'évènement d'origine et on place le curseur à (stylo + décalage).
        Le curseur n'est donc jamais "remis" à la position brute du stylo.
        """
        def proc(nCode, wParam, lParam):
            try:
                c = self.cfg
                if (nCode >= 0 and wParam == WM_MOUSEMOVE
                        and c["input_mode"] == "tablet" and c.get("tablet_hook", True)):
                    ms = ctypes.cast(lParam, ctypes.POINTER(MSLLHOOKSTRUCT)).contents
                    pos = (ms.pt.x, ms.pt.y)
                    if ms.dwExtraInfo != MAGIC and pos != self._last_set:
                        self._pen = (float(pos[0]), float(pos[1]))
                        self._pen_seen = True
                        self._hook_count += 1
                        ox, oy = self._ox, self._oy
                        if abs(ox) + abs(oy) >= 0.5:
                            nx, ny = int(round(pos[0] + ox)), int(round(pos[1] + oy))
                            self._last_set = (nx, ny)
                            if c.get("tablet_raw_inject", True):
                                move_abs(nx, ny)  # évènement lu par osu! en raw input
                            user32.SetCursorPos(nx, ny)
                            return 1  # on bloque l'évènement d'origine
            except Exception:
                pass
            return user32.CallNextHookEx(None, nCode, wParam, lParam)

        self._hook_proc = HOOKPROC(proc)  # garder une référence (sinon ramassé par le GC)
        hook = user32.SetWindowsHookExW(WH_MOUSE_LL, self._hook_proc, None, 0)
        if not hook:
            self.status = "Hook souris impossible (essaie en administrateur)"
            return
        msg = wintypes.MSG()
        while self.running and user32.GetMessageW(ctypes.byref(msg), None, 0, 0) > 0:
            user32.TranslateMessage(ctypes.byref(msg))
            user32.DispatchMessageW(ctypes.byref(msg))
        user32.UnhookWindowsHookEx(hook)

    # ---- mode tablette (stylo en mode absolu)
    def tablet_tick(self, dt):
        """
        Le driver Wacom réécrit la position ABSOLUE du curseur à chaque rapport
        du stylo : un déplacement relatif est donc aussitôt écrasé. On fait
        autrement : on retient la position réelle du stylo et on place le
        curseur à (stylo + décalage), le décalage pointant vers le cercle.
        """
        c = self.cfg
        cx, cy = cursor_pos()
        use_hook = self._pen_seen and c.get("tablet_hook", True)
        if not use_hook:
            # Repli : si le curseur n'est plus là où on l'a mis, le stylo a bougé
            if self._last_set is None or (cx, cy) != self._last_set:
                self._pen = (float(cx), float(cy))
        px, py = self._pen

        want_x = want_y = 0.0
        self._dbg = ("aide désactivée" if not c["enabled"] else
                     "pas en jeu (tosu ne signale pas de partie)" if not self.playing else
                     "aucune map chargée" if not self.objs else "aucun cercle proche")
        if c["enabled"] and self.playing and self.objs:
            t = self.now_ms()
            with self.lock:
                i = bisect.bisect_left(self.times, t - 80)
                obj = self.objs[i] if i < len(self.objs) else None
                mods, cs = self.mods, self.cs
            if obj is not None:
                ot, ox, oy = obj
                ahead = ot - t
                if ahead <= c["lookahead_ms"]:
                    tx, ty, s = self.to_screen(ox, oy, mods)
                    r_px = circle_radius_osu_px(cs, mods) * s
                    vx, vy = tx - px, ty - py
                    dist = math.hypot(vx, vy)
                    self._dbg = (f"cible à {dist / max(r_px, 1):.1f} rayons, "
                                 f"hit dans {ahead:.0f} ms (portée {c['range']:.1f})")
                    if c["inner_zone"] * r_px < dist <= c["range"] * r_px:
                        urgency = 1.0 - max(0.0, min(1.0, ahead / c["lookahead_ms"]))
                        w = c["strength"] * (0.3 + 0.7 * urgency)
                        mag = min(dist * w, c["max_offset"] * r_px)
                        want_x, want_y = vx / dist * mag, vy / dist * mag

        if self._test:                         # F9 : décalage fixe, pour tester
            want_x, want_y = 150.0, 0.0

        a = 1 - math.exp(-c["smooth"] * dt)   # lissage : pas de saccades
        self._ox += (want_x - self._ox) * a
        self._oy += (want_y - self._oy) * a

        nx, ny = int(round(px + self._ox)), int(round(py + self._oy))
        if (nx, ny) != (cx, cy):
            self._last_set = (nx, ny)     # avant l'écriture, pour que le hook reconnaisse nos évènements
            if c.get("tablet_raw_inject", True):
                move_abs(nx, ny)          # évènement d'entrée réel (osu! en raw input)
            user32.SetCursorPos(nx, ny)   # position exacte au pixel près
        self._last_set = (nx, ny)

    # ---- boucle d'assistance
    def assist_thread(self):
        last = time.perf_counter()
        while self.running:
            time.sleep(0.001)
            now = time.perf_counter()
            dt, last = now - last, now
            c = self.cfg

            if key_down(c["toggle_vk"]):
                if not self._toggle_latch:
                    c["enabled"] = not c["enabled"]
                self._toggle_latch = True
            else:
                self._toggle_latch = False

            if key_down(0x78):  # F9 : test de diagnostic
                if not self._test_latch:
                    self._test = not self._test
                self._test_latch = True
            else:
                self._test_latch = False

            if c["input_mode"] == "tablet":
                self.tablet_tick(dt)
                continue

            if not c["enabled"] or not self.playing or not self.objs:
                continue

            t = self.now_ms()
            with self.lock:
                i = bisect.bisect_left(self.times, t - 80)  # cercle expiré après 80 ms
                if i >= len(self.objs):
                    continue
                ot, ox, oy = self.objs[i]
                mods, cs = self.mods, self.cs
            ahead = ot - t
            if ahead > c["lookahead_ms"]:
                continue

            tx, ty, s = self.to_screen(ox, oy, mods)
            r_px = circle_radius_osu_px(cs, mods) * s
            cx, cy = cursor_pos()
            vx, vy = tx - cx, ty - cy
            dist = math.hypot(vx, vy)
            if dist > c["range"] * r_px or dist <= c["inner_zone"] * r_px or dist < 1:
                continue

            urgency = 1.0 - max(0.0, min(1.0, ahead / c["lookahead_ms"]))
            k = c["strength"] * 25.0 * (0.3 + 0.7 * urgency)
            step = dist * (1 - math.exp(-k * dt))
            step = min(step, c["max_speed"] * dt)
            self._rx += vx / dist * step * c["move_gain"]
            self._ry += vy / dist * step * c["move_gain"]
            ix, iy = int(self._rx), int(self._ry)
            if ix or iy:
                move_rel(ix, iy)
                self._rx -= ix
                self._ry -= iy

    threading.Thread(target=eng.ws_thread, daemon=True).start()
    threading.Thread(target=eng.assist_thread, daemon=True).start()
    threading.Thread(target=eng.hook_thread, daemon=True).start()
    threading.Thread(target=eng.relax_thread, daemon=True).start()
    sys.setswitchinterval(0.0005)  # le hook doit répondre vite, sinon Windows le coupe

    upd = Updater(eng)

    def startup_update_check():
        if cfg["auto_check_updates"] and upd.check() and cfg["auto_install_updates"]:
            upd.install()
    threading.Thread(target=startup_update_check, daemon=True).start()

    if "--headless" in sys.argv:
        # Sans fenêtre : utilise aimassist_config.json, F8 pour activer/désactiver
        print("Aim assist lancé sans fenêtre. Ctrl+C pour quitter.")
        last_status = ""
        try:
            while True:
                if eng.status != last_status:
                    last_status = eng.status
                    print(last_status)
                if upd.state == "available" and last_status != upd.message:
                    print(f"{upd.message} — relance avec l'option d'installation auto, "
                          "ou lance l'interface graphique.")
                    last_status = upd.message
                time.sleep(0.5)
        except KeyboardInterrupt:
            eng.running = False
        return

    run_gui(cfg, eng, upd)


if __name__ == "__main__":
    main()
