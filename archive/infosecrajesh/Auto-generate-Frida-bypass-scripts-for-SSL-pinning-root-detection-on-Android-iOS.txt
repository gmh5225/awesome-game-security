# frida-script-generator

Static-analysis-driven Frida script generator for Android and iOS mobile app security testing.

Point it at an APK or IPA. It inspects the binary, reports every SSL pinning and root/jailbreak
detection mechanism it finds, then writes a ready-to-run Frida bypass script tailored to that
specific app — no manual class-hunting required.

---

## Features

**SSL pinning bypass**
OkHttp3 · Retrofit · Volley · Cronet · Conscrypt · Flutter/Dart ·
React Native · Xamarin · Cordova · gRPC · Ktor · BouncyCastle ·
Certificate Transparency · Network Security Config · TrustKit ·
Custom TrustManagers · HostnameVerifier · SSLSocket ·
HttpsURLConnection · WebViewClient

**Root / jailbreak detection bypass**
RootBeer · SafetyNet / Play Integrity · Promon SHIELD ·
Talsec / FreeRASP · Approov · Zimperium zShield · Samsung Knox ·
File-based su/Magisk checks · Build.TAG · /proc/mounts ·
Emulator detection · Frida/hook detection · TracerPid spoofing ·
ADB / developer-options detection · Mock location bypass

**Three-layer injection architecture**
- L1 — Native libc hooks (kill, raise, abort, _exit) via Interceptor.replace
- L2 — ActivityThread.handleBindApplication (runs before Application.onCreate)
- L3 — setTimeout retries at 1 s and 3 s for lazy-loaded SDKs

**Reliable on Android 12+ (APEX bionic)**
Uses Interceptor.replace + NativeCallback for all libc hooks — avoids
the "not a function" failure caused by APEX PLT stubs.

## How it works

1. **Static analysis** — Scans the DEX string pool (Android) or Mach-O binary (iOS) for known
   framework signatures and custom security class names. No full decompilation needed.

2. **Targeted script generation** — Only hooks what was detected. An app using gRPC gets gRPC
   hooks; one with no Knox dependency gets none. Keeps the script lean and avoids unnecessary
   Java.use() calls that slow startup.

3. **Layered injection** — L1 native hooks fire at script load. L2 hooks inside
   handleBindApplication fire before Application.onCreate. L3 setTimeout passes catch SDKs
   that initialise after the application starts.

4. **Safe hook patterns** — Avoids known Frida pitfalls:
   - Interceptor.replace (not attach) for all libc symbols — works on Android 12 APEX bionic
   - trustCheckReturn() helper returns ArrayList for List-typed overloads — prevents OkHttp NPE
   - Saved-ref pattern for File.exists / readLine — prevents infinite recursion
   - CountDownLatch.await released without calling original — prevents deadlock

    def _scan_dex_classes(self):
        """Scan all DEX files for class/method name signals."""
        all_class_names = []
        try:
            with zipfile.ZipFile(self.path, "r") as z:
                dex_files = [n for n in z.namelist() if re.match(r'classes\d*\.dex', n)]
                for dex_name in dex_files:
                    raw = z.read(dex_name)
                    # Extract string pool — fast heuristic (no full DEX parse needed)
                    strings = re.findall(rb'L([a-zA-Z][a-zA-Z0-9_$/]{3,})', raw)
                    for s in strings:
                        cls = s.decode("utf-8","ignore").rstrip(";>")
                        # Skip obviously invalid hits (too short, contains junk)
                        if len(cls) > 5 and "/" in cls:
                            all_class_names.append(cls)
        except Exception as e:
            if self.verbose: console.print(f"  DEX scan error: {e}")

            # SSL signals
            for pattern, attr in self.SSL_SIGNALS.items():
                if pattern.lower() in cls.lower():
                    setattr(self.info, attr, True)
            # Root signals
            for pattern, attr in self.ROOT_SIGNALS.items():
                if pattern.lower() in cls.lower():
                    setattr(self.info, attr, True)
            # Custom SSL classes

                if re.search(pat, cls):
                    full = cls.replace("/",".")
                    if full not in self.info.smali_pinning_classes:
                        self.info.smali_pinning_classes.append(full)

            console.print(f"  SSL classes found: {len(self.info.smali_pinning_classes)}")

    def _parse_info_plist(self):
        try:
            with zipfile.ZipFile(self.path,"r") as z:
                plists = [n for n in z.namelist() if n.endswith("Info.plist") and "Payload/" in n]
                if plists:
                    raw = z.read(plists[0])
                    # Extract bundle id and version from plist (XML or binary)
                    txt = raw.decode("utf-8","ignore")
                    bid = re.search(r'CFBundleIdentifier.*?<string>(.*?)</string>', txt, re.DOTALL)
                    ver = re.search(r'CFBundleShortVersionString.*?<string>(.*?)</string>', txt, re.DOTALL)
                    nam = re.search(r'CFBundleName.*?<string>(.*?)</string>', txt, re.DOTALL)
                    if bid: self.info.package_name = bid.group(1).strip()
                    if ver: self.info.version_name = ver.group(1).strip()
                    if nam: self.info.app_name = nam.group(1).strip()
                    # ATS check
                    if "NSAllowsArbitraryLoads" in txt: self.info.uses_network_security_config = True
        except Exception as e:
            if self.verbose: console.print(f"  plist error: {e}")

    def _scan_binary(self):
        """Extract strings from Mach-O binary for framework signals."""
        try:
            with zipfile.ZipFile(self.path,"r") as z:
                # Find main binary
                bins = [n for n in z.namelist()
                        if "Payload/" in n and not n.endswith("/")
                        and "." not in os.path.basename(n)
                        and "__" not in n]
                # Also check frameworks
                frameworks = [n for n in z.namelist() if n.endswith(".framework/") or "Frameworks/" in n]

                all_names = z.namelist()
                fw_str = " ".join(all_names).lower()

                if "afnetworking" in fw_str:   self.info.uses_afnetworking = True
                if "alamofire" in fw_str:      self.info.uses_alamofire = True
                if "trustkit" in fw_str:       self.info.uses_trustkit = True
                if "flutter" in fw_str:        self.info.uses_flutter = True
                if "react" in fw_str:          self.info.uses_react_native = True
                if "cordova" in fw_str:        self.info.uses_cordova = True
                if "xamarin" in fw_str:        self.info.uses_xamarin = True

                # Scan strings in main binary
                for n in bins[:3]:  # first 3 candidates
                    try:
                        raw = z.read(n)
                        strings = re.findall(b'[\x20-\x7e]{6,}', raw)
                        str_joined = b" ".join(strings).decode("utf-8","ignore").lower()
                        if "sslpinning" in str_joined or "certificatepinning" in str_joined:
                            self.info.uses_custom_ssl = True
                        if "afnetworking" in str_joined: self.info.uses_afnetworking = True
                        if "alamofire" in str_joined:    self.info.uses_alamofire = True
                        if "trustkit" in str_joined:     self.info.uses_trustkit = True
                        if "nsurlsession" in str_joined: self.info.uses_nsurlsession = True
                        if "jailbreak" in str_joined:    self.info.uses_jailbreak_detect = True
                        if "frida" in str_joined:        self.info.uses_frida_detect = True
                        if "flutter" in str_joined:      self.info.uses_flutter = True
                        # Scan for custom classes
                        cls_hits = re.findall(r'(?:ssl|pin|trust|cert|root|jailbreak|tamper)\w+', str_joined, re.I)
                        self.info.smali_pinning_classes.extend(list(set(cls_hits))[:15])
                    except Exception: pass
        except Exception as e:
            if self.verbose: console.print(f"  Binary scan error: {e}")

    def __init__(self, info: AppInfo, mode: str = "all"):
        self.info = info

def main():

Examples:

  python3 frida_script_generator.py --apk app.apk --mode root --out root_bypass.js
  python3 frida_script_generator.py --apk app.apk --verbose
  frida -U -f com.example.app -l <generated_script.js>
"""
    )
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--apk", help="Path to Android APK file")
    group.add_argument("--ipa", help="Path to iOS IPA file")
    parser.add_argument("--out",     "-o", default=None,      help="Output .js filename")
    parser.add_argument("--mode",    "-m", default="all",     choices=["all","ssl","root"], help="Bypass mode (default: all)")
    parser.add_argument("--verbose", "-v", action="store_true", help="Verbose output")
