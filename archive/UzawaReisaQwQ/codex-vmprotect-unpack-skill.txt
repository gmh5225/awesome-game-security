		std::string Sha256Hex(std::string_view text) {
			const auto bytes = Utf8Bytes(text);
			BCRYPT_ALG_HANDLE rawAlgorithm = nullptr;
			if (BCryptOpenAlgorithmProvider(&rawAlgorithm, BCRYPT_SHA256_ALGORITHM, nullptr, 0) < 0)
				throw std::runtime_error("Cannot open SHA-256 provider");
			std::unique_ptr<void, AlgorithmClose> algorithm(rawAlgorithm);
			BCRYPT_HASH_HANDLE rawHash = nullptr;
			if (BCryptCreateHash(algorithm.get(), &rawHash, nullptr, 0, nullptr, 0, 0) < 0)
				throw std::runtime_error("Cannot create SHA-256 hash");
			std::unique_ptr<void, HashClose> hash(rawHash);
			// Chunking avoids truncating the Windows ULONG length parameter.
			std::size_t offset = 0;
			while (offset < bytes.size()) {
				const auto remaining = bytes.size() - offset;
				constexpr auto maxChunk = (std::numeric_limits<ULONG>::max)();
				const auto count = static_cast<ULONG>(remaining > maxChunk ? maxChunk : remaining);
				if (BCryptHashData(hash.get(), const_cast<PUCHAR>(bytes.data() + offset), count, 0) < 0)
					throw std::runtime_error("SHA-256 update failed");
				offset += count;
			}
			std::array<unsigned char, 32> digest{};
			if (BCryptFinishHash(hash.get(), digest.data(), static_cast<ULONG>(digest.size()), 0) < 0)
				throw std::runtime_error("SHA-256 finalization failed");
			constexpr char hex[] = "0123456789ABCDEF";
			std::string result;
			for (const auto value : digest) { result += hex[value >> 4]; result += hex[value & 15]; }
			return result;
		}
	}

	std::string MachineID::GetMachineGuid() {
		HKEY rawKey = nullptr;
		auto status = RegOpenKeyExW(HKEY_LOCAL_MACHINE, L"SOFTWARE\\Microsoft\\Cryptography", 0, KEY_READ, &rawKey);
		if (status == ERROR_FILE_NOT_FOUND || status == ERROR_PATH_NOT_FOUND) return {};
		if (status != ERROR_SUCCESS) throw std::runtime_error("Cannot open MachineGuid registry key");
		std::unique_ptr<std::remove_pointer_t<HKEY>, RegistryClose> key(rawKey);
		DWORD type = 0, size = 0;
		status = RegQueryValueExW(key.get(), L"MachineGuid", nullptr, &type, nullptr, &size);
		if (status == ERROR_FILE_NOT_FOUND) return {};
		if (status != ERROR_SUCCESS) throw std::runtime_error("Cannot query MachineGuid");
		std::vector<wchar_t> buffer(size / sizeof(wchar_t) + 2, L'\0');
		status = RegQueryValueExW(key.get(), L"MachineGuid", nullptr, &type, reinterpret_cast<LPBYTE>(buffer.data()), &size);
		if (status != ERROR_SUCCESS) throw std::runtime_error("Cannot read MachineGuid");
		if (type != REG_SZ && type != REG_EXPAND_SZ) throw std::runtime_error("MachineGuid must be a registry string");
		std::wstring value(buffer.data());
		if (type == REG_EXPAND_SZ) {
			const auto length = ExpandEnvironmentStringsW(value.c_str(), nullptr, 0);
			if (!length) throw std::runtime_error("Cannot expand MachineGuid");
			std::vector<wchar_t> expanded(length);
			const auto written = ExpandEnvironmentStringsW(value.c_str(), expanded.data(), length);
			if (!written || written > length) throw std::runtime_error("Cannot expand MachineGuid");
			value.assign(expanded.data());
		}
		return ToUtf8(value);
	}

	std::string MachineID::GetCpuId() {
		try {
			ComScope com;
			if (FAILED(com.status) && com.status != RPC_E_CHANGED_MODE) return {};
			auto status = CoInitializeSecurity(nullptr, -1, nullptr, nullptr, RPC_C_AUTHN_LEVEL_DEFAULT,
				RPC_C_IMP_LEVEL_IMPERSONATE, nullptr, EOAC_NONE, nullptr);
			if (FAILED(status) && status != RPC_E_TOO_LATE) return {};
			IWbemLocator* rawLocator = nullptr;
			status = CoCreateInstance(CLSID_WbemLocator, nullptr, CLSCTX_INPROC_SERVER,
				IID_IWbemLocator, reinterpret_cast<void**>(&rawLocator));
			ComPtr<IWbemLocator> locator(rawLocator);
			if (FAILED(status)) return {};
			IWbemServices* rawServices = nullptr;
			Bstr path(L"ROOT\\CIMV2");
			status = locator->ConnectServer(path.value, nullptr, nullptr, nullptr, 0, nullptr, nullptr, &rawServices);
			ComPtr<IWbemServices> services(rawServices);
			if (FAILED(status)) return {};
			status = CoSetProxyBlanket(services.get(), RPC_C_AUTHN_WINNT, RPC_C_AUTHZ_NONE, nullptr,
				RPC_C_AUTHN_LEVEL_CALL, RPC_C_IMP_LEVEL_IMPERSONATE, nullptr, EOAC_NONE);
			if (FAILED(status)) return {};
			Bstr language(L"WQL"), query(L"SELECT ProcessorId FROM Win32_Processor");
			IEnumWbemClassObject* rawEnumerator = nullptr;
			status = services->ExecQuery(language.value, query.value,
				WBEM_FLAG_FORWARD_ONLY | WBEM_FLAG_RETURN_IMMEDIATELY, nullptr, &rawEnumerator);
			ComPtr<IEnumWbemClassObject> enumerator(rawEnumerator);
			if (FAILED(status)) return {};
			IWbemClassObject* rawObject = nullptr;
			ULONG count = 0;
			status = enumerator->Next(WBEM_INFINITE, 1, &rawObject, &count);
			ComPtr<IWbemClassObject> object(rawObject);
			if (FAILED(status) || count == 0) return {};
			struct Variant {
				VARIANT value;
				Variant() { VariantInit(&value); }
				~Variant() { VariantClear(&value); }
			} property;
			status = object->Get(L"ProcessorId", 0, &property.value, nullptr, nullptr);
			if (FAILED(status) || property.value.vt != VT_BSTR || !property.value.bstrVal) return {};
			std::wstring_view value(property.value.bstrVal, SysStringLen(property.value.bstrVal));
			while (!value.empty() && IsWhitespace(value.front())) value.remove_prefix(1);
			while (!value.empty() && IsWhitespace(value.back())) value.remove_suffix(1);
			return ToUtf8(value);
		}
		catch (...) { return {}; }
	}

		int CheckedLength(std::size_t size) {
			if (size > static_cast<std::size_t>((std::numeric_limits<int>::max)()))
				throw std::length_error("Text exceeds Windows conversion limit");
			return static_cast<int>(size);
		}
		constexpr std::array<std::uint8_t, 8> xorKey{ 0x37, 0xA1, 0x52, 0xC8, 0x19, 0x73, 0xE4, 0x2B };
		constexpr char alphabet[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
	}

	std::wstring FromUtf8(std::string_view text) {
		std::wstring result;
		for (std::size_t i = 0; i < text.size();) {
			const auto first = static_cast<std::uint8_t>(text[i++]);
			if (first < 0x80) { result += static_cast<wchar_t>(first); continue; }
			unsigned length = 0;
			std::uint32_t codepoint = 0;
			if (first >= 0xC2 && first <= 0xDF) { length = 2; codepoint = first & 0x1F; }
			else if (first >= 0xE0 && first <= 0xEF) { length = 3; codepoint = first & 0x0F; }
			else if (first >= 0xF0 && first <= 0xF4) { length = 4; codepoint = first & 0x07; }
			else { result += L'\uFFFD'; continue; }
			bool valid = true;
			for (unsigned j = 1; j < length; ++j) {
				if (i == text.size()) { valid = false; break; }
				const auto next = static_cast<std::uint8_t>(text[i]);
				if (next < 0x80 || next > 0xBF
					|| (j == 1 && ((first == 0xE0 && next < 0xA0)
						|| (first == 0xED && next > 0x9F)
						|| (first == 0xF0 && next < 0x90)
						|| (first == 0xF4 && next > 0x8F)))) {
					valid = false;
					break;
				}
				++i;
				codepoint = (codepoint << 6) | (next & 0x3F);
			}
			if (!valid) result += L'\uFFFD';
			else if (codepoint <= 0xFFFF) result += static_cast<wchar_t>(codepoint);
			else {
				codepoint -= 0x10000;
				result += static_cast<wchar_t>(0xD800 + (codepoint >> 10));
				result += static_cast<wchar_t>(0xDC00 + (codepoint & 0x3FF));
			}
		}
		return result;
	}

	std::string ToUtf8(std::wstring_view text) {
		if (text.empty()) return {};
		const int length = CheckedLength(text.size());
		const int size = WideCharToMultiByte(CP_UTF8, 0, text.data(), length, nullptr, 0, nullptr, nullptr);
		if (!size) throw std::runtime_error("UTF-8 encoding failed");
		std::string result(size, '\0');
		if (!WideCharToMultiByte(CP_UTF8, 0, text.data(), length, result.data(), size, nullptr, nullptr))
			throw std::runtime_error("UTF-8 encoding failed");
		return result;
	}

	std::string DecodeUtf8(const Bytes& bytes) {
		if (bytes.empty()) return {};
		return ToUtf8(FromUtf8(std::string_view(reinterpret_cast<const char*>(bytes.data()), bytes.size())));
	}

	Bytes Utf8Bytes(std::string_view text) {
		const auto normalized = ToUtf8(FromUtf8(text));
		return Bytes(normalized.begin(), normalized.end());
	}

	std::string Base64Encode(const Bytes& data) {
		std::string result;
		result.reserve((data.size() + 2) / 3 * 4);
		for (std::size_t i = 0; i < data.size(); i += 3) {
			const auto value = (static_cast<std::uint32_t>(data[i]) << 16)
				| (i + 1 < data.size() ? static_cast<std::uint32_t>(data[i + 1]) << 8 : 0)
				| (i + 2 < data.size() ? data[i + 2] : 0);
			result += alphabet[(value >> 18) & 63];
			result += alphabet[(value >> 12) & 63];
			result += i + 1 < data.size() ? alphabet[(value >> 6) & 63] : '=';
			result += i + 2 < data.size() ? alphabet[value & 63] : '=';
		}
		return result;
	}

	Bytes Base64Decode(std::string_view text) {
		std::string clean;
		for (char c : text) {
			if (c != ' ' && c != '\t' && c != '\r' && c != '\n') clean += c;
		}
		if (clean.size() % 4) throw std::invalid_argument("Invalid Base64 length");
		Bytes result;
		for (std::size_t i = 0; i < clean.size(); i += 4) {
			std::uint32_t value = 0;
			unsigned padding = 0;
			for (unsigned j = 0; j < 4; ++j) {
				const char c = clean[i + j];
				value <<= 6;
				if (c == '=') {
					if (j < 2 || i + 4 != clean.size()) throw std::invalid_argument("Invalid Base64 padding");
					++padding;
				}
				else {
					const auto position = std::string_view(alphabet, 64).find(c);
					if (padding || position == std::string_view::npos) throw std::invalid_argument("Invalid Base64 character");
					value |= static_cast<std::uint32_t>(position);
				}
			}
			result.push_back(static_cast<std::uint8_t>(value >> 16));
			if (padding < 2) result.push_back(static_cast<std::uint8_t>(value >> 8));
			if (padding < 1) result.push_back(static_cast<std::uint8_t>(value));
		}
		return result;
	}

	std::string XOR::Encode(std::string_view input) {
		auto data = Utf8Bytes(input);
		for (std::size_t i = 0; i < data.size(); ++i) data[i] ^= xorKey[i % xorKey.size()];
		return Base64Encode(data);
	}

	std::string XOR::Decode(std::string_view input) {
		auto data = Base64Decode(input);
		for (std::size_t i = 0; i < data.size(); ++i) data[i] ^= xorKey[i % xorKey.size()];
		return DecodeUtf8(data);
	}

	std::string StringObfuscator::Decode(const Bytes& encoded) {
		auto data = encoded;
		for (auto& value : data) value ^= 0x37;
		return DecodeUtf8(data);
	}

// Adapted from Ma Bingyao's XXTEA.cs, last modified March 10, 2015.
// XXTEA algorithm: David J. Wheeler and Roger M. Needham.
#include "XXTEA.h"
#include <array>
#include <stdexcept>
#include <limits>

namespace Xxtea {
using console_app2::Bytes;
namespace {
constexpr std::uint32_t delta = 0x9E3779B9u;
using Words = std::vector<std::uint32_t>;

Words ToWords(const Bytes& data, bool includeLength) {
    if (data.size() > (std::numeric_limits<std::uint32_t>::max)())
        throw std::length_error("XXTEA input exceeds the 32-bit length format");
    const auto count = (data.size() + 3) / 4;
    Words result(count + (includeLength ? 1 : 0), 0);
    if (includeLength) result[count] = static_cast<std::uint32_t>(data.size());
    for (std::size_t i = 0; i < data.size(); ++i)
        result[i / 4] |= static_cast<std::uint32_t>(data[i]) << ((i % 4) * 8);
    return result;
}

std::optional<Bytes> ToBytes(const Words& data, bool includeLength) {
    auto size = data.size() * 4;
    if (includeLength) {
        const auto length = static_cast<std::size_t>(data.back());
        size -= 4;
        if (length > size || size - length > 3) return std::nullopt;
        size = length;
    }
    Bytes result(size);
    for (std::size_t i = 0; i < size; ++i)
        result[i] = static_cast<std::uint8_t>(data[i / 4] >> ((i % 4) * 8));
    return result;
}

Words FixKey(const Bytes& key) {
    Bytes fixed(16, 0);
    for (std::size_t i = 0; i < key.size() && i < fixed.size(); ++i) fixed[i] = key[i];
    return ToWords(fixed, false);
}

std::uint32_t MX(std::uint32_t sum, std::uint32_t y, std::uint32_t z,
                 std::size_t p, std::uint32_t e, const Words& key) {
    return (((z >> 5) ^ (y << 2)) + ((y >> 3) ^ (z << 4)))
        ^ ((sum ^ y) + (key[(p & 3) ^ e] ^ z));
}

std::string DecodeResult(const std::optional<Bytes>& data) {
    if (!data) throw std::invalid_argument("Invalid XXTEA decrypted length");
    return console_app2::DecodeUtf8(*data);
}
}

Bytes XXTEA::Encrypt(const Bytes& data, const Bytes& key) {
    if (data.empty()) return data;
    auto words = ToWords(data, true);
    const auto fixedKey = FixKey(key);
    const auto n = words.size() - 1;
    std::uint32_t z = words[n], sum = 0;
    auto rounds = 6 + 52 / words.size();
    while (rounds-- > 0) {
        sum += delta;
        const auto e = (sum >> 2) & 3;
        for (std::size_t p = 0; p < n; ++p)
            z = (words[p] += MX(sum, words[p + 1], z, p, e, fixedKey));
        z = (words[n] += MX(sum, words[0], z, n, e, fixedKey));
    }
    return *ToBytes(words, false);
}

std::optional<Bytes> XXTEA::Decrypt(const Bytes& data, const Bytes& key) {
    if (data.empty()) return data;
    auto words = ToWords(data, false);
    const auto n = words.size() - 1;
    if (n >= 1) {
        const auto fixedKey = FixKey(key);
        std::uint32_t y = words[0];
        auto sum = static_cast<std::uint32_t>(6 + 52 / words.size()) * delta;
        while (sum != 0) {
            const auto e = (sum >> 2) & 3;
            for (auto p = n; p > 0; --p)
                y = (words[p] -= MX(sum, y, words[p - 1], p, e, fixedKey));
            y = (words[0] -= MX(sum, y, words[n], 0, e, fixedKey));
            sum -= delta;
        }
    }
    return ToBytes(words, true);
}

Bytes XXTEA::Encrypt(std::string_view data, std::string_view key) { return Encrypt(console_app2::Utf8Bytes(data), console_app2::Utf8Bytes(key)); }
Bytes XXTEA::Encrypt(std::string_view data, const Bytes& key) { return Encrypt(console_app2::Utf8Bytes(data), key); }

---
name: vmprotect-unpack
description: 分析和恢复 VMProtect 保护的 Windows 原生 PE 程序，包括解包、入口定位、导入重建、字符串恢复、反调试与反虚拟机检查分析，以及可运行性验证和逆向报告。适用于脱壳或修复受保护 EXE/DLL；参考多个 x64 版本及 Lite 3.10.4、Ultimate 3.5.0 x86 样本实测，含按需模拟兼容层和 x86/WOW64 验证流程，不保证通用去虚拟化。

# VMProtect 程序恢复

产出可复现的恢复文件及报告。将“内容已解压”“能够运行”“所有保护已移除”分别证明；自动工具成功退出仅代表其中一个阶段。用户已有授权和文件访问边界继续有效，不重复询问已明确的事项。文件路径、工具目录、版本与配置从当前任务获取，不套用案例地址。

## 开始与工具选择

1. 保留原文件，记录 SHA-256、PE 位数、入口、ImageBase、节、数据目录、TLS、导入和异常表。建立任务内的 tools、analysis、intermediate、output 目录。禁止覆盖输入。
2. 优先发现用户提供的工具与快捷方式目标。ExeInfoPE 可拖入样本识别；无 GUI 时可使用 DIE/diec、PE 解析。签名 Unknown 不否定保护存在，也不证明具体版本。区分用户提供版本与工具识别结论。
3. 先识别 native PE、托管程序集、DLL/驱动；本流程默认原生用户态 PE。托管、驱动、虚拟化函数需要调整路线，不能当作普通 EXE 直接运行。
   对 I386/PE32，先阅读 [x86 专用流程](references/x86-workflow.md)，使用 `--32bit` 及匹配的 catalog。不要复用 x64 地址、指针宽度、入口模式或调试上下文假设。
4. 核对保护选项：打包、导入/资源/字符串保护、虚拟化、变异、反调试、反虚拟机。即使未选择函数虚拟化，也不假定无 SDK 或自动生成的保护跳板。
5. 缺少工具时，在当前授权范围内查询官方文档/源仓库并下载到任务目录；记录来源、版本/提交及哈希。不要上传样本到在线脱壳服务，除非用户明确要求。不要为此修改系统安全配置。

静态检查可使用 `scripts/pe_inventory.py`（依赖 pefile），具体调用见脚本 `--help`。只读取输入，结果写入显式指定目录。

## 恢复路线

优先使用与架构、壳版本匹配的模拟解包和导入恢复方案；阅读 [references/workflow.md](references/workflow.md) 获取工具命令、入口误判、重定位和测试细节。模拟不支持样本时再转向调试器捕获 OEP、进程转储及 Scylla 导入重建，保留每步中间产物。

- 模拟中观察恢复页面、读写范围、导入解析、API/系统调用事件和当前地址。长时间运行时以状态变化判断进展，不因日志稀疏而终止，也不无依据地反复重跑。
- 首次执行到恢复节可能只是壳或 SDK 跳板。结合反汇编、调用图、栈与运行验证确认应用入口；EXE 的 CRT 启动、TLS 回调、DLL 的 DllMain 不能互换。
- 将转储中的模拟 API 地址、堆指针、句柄及运行后状态视为不可持久化状态。重建 IAT，并修复直接调用、跳转、指针槽和包装调用；区分调用点修复数与唯一导入数。
- 新增导入必须有来源证据；模拟地址的数值匹配不等于函数指针。交付前及出现跨系统 Entry Point Not Found 时阅读 [导入来源审计](references/import-provenance.md)，重点复核 final_scan_only 额外导入，不能仅凭在当前 Windows 上可运行认定正确。
- 检查文件布局、节权限、异常/unwind 表、TLS、Load Config、资源、重定位、目录边界和初始化状态。**不得默认清空 Load Config/TLS 或禁用 ASLR。** 只有证据支持且测试覆盖时才做针对性调整，并说明功能损失。
- 字符串分为压缩恢复、编码导入名、运行时解密三类。扫描能读到字符串不代表所有字符串已解密；记录解密入口、引用、编码与长度，按实际需要恢复字符串及调用语义。
- 反调试/反虚拟机处理要定位到具体检查或证明启动时绕过相应初始化路径。不把导入中出现 IsDebuggerPresent、VirtualAlloc 或 RtlVirtualUnwind 等名称当作需要删除的证据。不改动业务激活校验来伪装恢复成功。
- 删除壳段前分析直接/间接代码引用、指针、异常处理器、TLS、资源及数据目录。仅改节名或抹掉壳签名不是脱壳；无法证明可删时保留并披露残留。

## 验证与交付门槛

保持原版与候选版的输入、参数、工作目录策略、标准输入 EOF、超时及运行环境一致。先选择不会产生无关外部副作用的测试路径。用户只授权静态分析时不运行样本。

普通命令行程序可用 `scripts/compare_runs.py` 对比，必须显式传 `--execute`、两份文件、输入字节文件和结果目录。它只适用于有界标准输入/输出的单进程测试，不涵盖交互 UI、子进程生命周期或真实调试验证。

至少分别记录：

- PE 结构能解析、文件能装载、是否超时、退出码、标准输出与错误；超时后强制终止不算正常退出。
- 已验证输入及分支；有效输入缺失时保留未测试结论。
- 实际附加调试器后的行为，而非仅在模拟器中设置 BeingDebugged=0。初始系统断点与访问异常分开记录。
- 如要声称移除反虚拟机限制，须提供对应虚拟化环境实测或对具体检查的充分代码证据；普通宿主机运行不够。
- ASLR、首选基址限制和保留的保护依赖。与沙箱强制重定位有关的失败先查实际装载地址，不能只凭 WinError 文案认定原因。

最终文件每次修改后重跑相关验证；记录交付版哈希，保留可重复构建脚本和补丁前后证据。使用 [references/report-outline.md](references/report-outline.md) 写报告。状态只能按证据写成“分析中间件”“已验证路径的可运行恢复版”或有充分证据的“完整去保护版”。未达到可运行门槛时继续定位；遇到工具不支持/缺少必要输入且无可行进展时交付诊断，不用换扩展名掩盖失败。

开始跨版本任务时阅读 [references/version-matrix.md](references/version-matrix.md)，区分实测样本与工具声明支持范围。按需要阅读 [Ultimate 3.5.0 x86 案例](references/vmp-3.5.0-x86-case.md)、[Lite 3.10.4 x86 案例](references/vmp-lite-3.10.4-x86-case.md)、[Lite 3.10.4 x64 案例](references/vmp-lite-3.10.4-case.md)、[3.10.3 案例](references/vmp-3.10.3-case.md)、[3.9.4 案例](references/vmp-3.9.4-case.md)、[3.8.7 案例](references/vmp-3.8.7-case.md)、[3.8.4 案例](references/vmp-3.8.4-case.md) 或 [3.5.0 案例](references/vmp-3.5.0-case.md)。所有 RVA、哈希、节名和统计仅属于相应样本，不能当作新样本补丁。

# 导入来源与跨 Windows 版本审计

在较新 Windows/catalog 下运行通过，不证明旧版 Windows 能装载。新增静态导入会在程序入口执行前就要求目标系统提供对应导出；不能把 Entry Point Not Found 一律归因于用户 DLL 缺失，也不建议用户复制新版系统 DLL。

## 保留来源

x86 适配器支持 `--import-evidence path.json`（在 `--` 前，要求新文件），在重建前保存 bootstrap、解析写入、编码元数据及最终扫描来源。例如：

```powershell
python "$skillDir\scripts\run_x86_compat.py" --tool-root .\tools\vmp-lazy-unpack-main --import-evidence .\analysis\imports-evidence.json -- .\sample.exe sample.stage1.exe --32bit
python "$skillDir\scripts\audit_import_evidence.py" .\sample.stage1.exe --evidence .\analysis\imports-evidence.json --reconstruction .\analysis.imports.json --out .\analysis\imports-audit.json
```

审计工具只列出需要复核的 final_scan_only 候选，不自动删除。单独检查这些候选是否存在原导入描述符、编码元数据、真实 API 解析写入、调用点或可信数据引用。重建器自己生成的 thunk/IAT 不能反过来当作原始程序调用证据。

## 数值碰撞的实测错误

模拟模块的 API 地址只是人工布局中的整数。原代码/位移/字节码里相同的 4 字节并不一定是函数指针。旧版 x86 分类器曾把恢复区中的每个非 bootstrap 扫描匹配都加入额外导入。

Ultimate 3.5.0 x86 的四项误报仅有 final_scan_only 来源，无 bootstrap、编码表或指令补丁证据，也没有对槽地址的绝对引用。其中 RVA `0x16A414` 的 `00 0F 84 70` 跨越比较指令尾部与条件跳转，却匹配模拟 NtAllocateVirtualMemoryEx 地址。另三个匹配 Heap32ListNext、GetUserDefaultGeoName、ZwSetInformationCpuPartition。误报不仅增加不必要依赖，还会改坏原字节；Windows 11 可解析导出，掩盖了问题，用户的 Windows 10 在入口前报错。

缺少绝对引用本身不构成删除证明，编码/间接引用可能存在；需要结合来源、上下文及原值证据。此案例经过联合复核后恢复四处原字节并移除对应导入，而非将它们重命名成旧系统 API 或提供虚假成功桩。

## 修复与验证约束

- 修改导入表时保持所有保留 IAT 单元地址不变。不能删除中间元素后整体平移 IAT，代码仍可能引用旧地址。可按经过审核的布局拆分同 DLL 描述符，并在被移除的 INT/IAT 单元终止原连续段。
- 同时撤回误识别槽对应的字节改写，保留精确前后值与输入哈希。
- 重新解析完整导入表，确认误报身份消失、保留项数量/地址正确，检查资源与装载元数据，再重跑原生功能及调试测试。
- 如果有目标系统的可信导出目录，可做静态依赖审计；不要使用分析机的 Win11 导出存在性代替 Win10 验证。
- 没有真实目标系统实测时，结论写成“已移除该缺失依赖，本机回归通过，目标系统待复测”，不要宣称所有 Windows 10 版本已兼容。

脚本依赖上游 ApiHandlers、catalog 和 synthetic PE 布局。变更工具版本后须重新检查接口与测试，不能保证自动兼容。命令行入口仍拒绝 `--32bit`；共享资源函数已按架构核对 catalog/Magic，并选择 PE32/PE32+ 的资源目录偏移，分别经过回归验证。工具根必须含 unpack/tracer.py，依赖安装在任务环境中。

## 已实现的有限模型

- GetProcessWindowStation：返回稳定的模拟窗口站句柄。GetUserObjectInformationA/W 仅支持此句柄的 UOI_FLAGS，返回 12 字节 USEROBJECTFLAGS，处理短缓冲区、输出长度和无效地址；其他信息类停止。
- CheckRemoteDebuggerPresent：仅支持当前进程伪句柄；成功返回非零，并写入 FALSE，表示所建模的进程没有被调试。不能把返回值零当作“没有调试器”。
- EnumSystemFirmwareTables / GetSystemFirmwareTable：采用与上游固件模型一致的 UEFI 场景，FIRM 返回不支持；RSMB 枚举表 ID 0，内容复用上游 smbios_table()，支持查询长度、短缓冲区和实际复制。未知 provider 停止，不能伪造任意 ACPI 表。
- `--ntdll-resources`：从随工具提供的 ntdll.dll **原样复制版本资源载荷**，重建其三级资源目录到 synthetic ntdll 中。不是伪造任意 Windows 版本，也不读取宿主系统 DLL。仅支持当前观察到的单个数字 ID、单语言版本资源，校验 PE64 头、空间和目标区为空；不匹配就停止。

窗口站模型在本次失败路径调查中调用过；成功流程的关键差异是补充 ntdll 版本资源和缺失的调试/固件 API。脚本没有吞掉 MessageBox，也没有“所有未知 API 一律成功”的兜底。

参考官方语义：[GetProcessWindowStation](https://learn.microsoft.com/en-us/windows/win32/api/winuser/nf-winuser-getprocesswindowstation)、[GetUserObjectInformationW](https://learn.microsoft.com/en-us/windows/win32/api/winuser/nf-winuser-getuserobjectinformationw)、[CheckRemoteDebuggerPresent](https://learn.microsoft.com/en-us/windows/win32/api/debugapi/nf-debugapi-checkremotedebuggerpresent)、[EnumSystemFirmwareTables](https://learn.microsoft.com/en-us/windows/win32/api/sysinfoapi/nf-sysinfoapi-enumsystemfirmwaretables)、[GetSystemFirmwareTable](https://learn.microsoft.com/en-us/windows/win32/api/sysinfoapi/nf-sysinfoapi-getsystemfirmwaretable)。

## 验证要求

验证 API 的成功/失败返回值、输出缓冲区、所需长度、未知信息类停止，以及版本资源载荷与 catalog 原始数据一致。记录新增模型确实执行过的事件；某个方法经过单元检查不代表已被目标调用。之后仍需真实原生运行、调试及功能对比；模拟通过不能替代真实虚拟机验证。

对比脚本记录 stdout/stderr 原始文件及其哈希、输入/EXE 哈希、退出码和超时。`completed_equivalent` 仅在两次正常完成且退出码和输出一致时为真；仍应查看 `both_exit_zero`，两个相同崩溃不算成功。`--cwd` 可固定共同工作目录，默认分别使用各 EXE 的父目录。该脚本不管理目标派生的子进程，目标会派生进程时改用具备进程树生命周期管理的测试器。

PE 检查脚本还输出资源叶节点的类型/ID/语言路径、载荷长度和哈希、异常表实际排序状态及重定位项数。不同版本的末尾节可能是 `.rsrc` 而非 `.reloc`，重建导入时不可凭节序号猜数据用途。对比资源时忽略可合法变化的 RVA，检查资源路径、大小和载荷哈希；若原资源经过加密，则另按解密后的实际内容验证。发现异常范围重叠时记录具体区间与所属节，分析引用及异常语义；不要只为让检查通过而删除条目，也不要把常规输入测试扩大成异常路径完整验证。

## 工具获取与模拟路线

可评估的开源方案：[x86rc/vmp-lazy-unpack](https://github.com/x86rc/vmp-lazy-unpack)。先核对当前 README、提交和支持架构。工具声称的版本支持不是当前样本的成功保证；不提供通用去虚拟化。不要用 Themida/WinLicense 专用工具 unlicense 直接当作 VMProtect 通用解包器。

本技能不捆绑第三方源码、DLL 或样本。将所需版本下载到当前工作区的 tools，校验并保存哈希；依赖使用本地虚拟环境或 `pip --target`，不依赖作者电脑路径。Windows 上必要时将临时目录设在工作区，避免 one-file 工具自解压权限问题；网络失败按执行环境的权限机制处理。

一次验证过的调用形式（运行前以所下载版本的解析器为准）：

```powershell
# 当前目录为任务目录，工具与依赖已经准备好。
$env:PYTHONPATH = "$PWD\tools\pylibs;$PWD\tools\vmp-lazy-unpack"
python -u -m unpack .\sample.exe sample.stage1.exe --diagnostics *> .\analysis\unpack.log
```

此版本第二个参数只接受文件名，不接受目录路径。实际输出可位于工具自身的 output 目录，应读取日志或 summary.json 的 analysis_pe 字段定位，不猜测输出位置。32 位选项和 imports_only 模式以当前源码为准；不把 PE64 用 32 位模式执行。

使用工具随附 catalog 可以避免抓取宿主 DLL，尤其适合文件访问范围受限的任务；先确认 catalog 与模拟架构一致。不要在缺 catalog 时默许工具自动读取范围外的系统目录。

遇到 `unknown_import` 时先读取事件详情，不用统一返回成功绕过。旧版样本可能需要兼容层，见 [legacy-compat.md](legacy-compat.md)。如果进入 MessageBox 错误路径，先提取文本并检查模拟环境是否缺字段；显示“检测到调试器”不必然代表真实附加了调试器。未建模接口与保护主动拒绝执行应分别诊断。

阶段性产物应包含输入信息、progress、events、summary、导入修复明细。监测当前地址、基本块数量、恢复页数和文件扫描指针；重复循环也可能在顺序计算文件 CRC，不能只看同一条指令就认定死循环。若状态不再变化且可复现，分析循环终止条件或未建模 API 后再重试。

如果异常堆栈显示失败发生在可选诊断 JSON 的临时文件替换（如 `ntdll-reads.json.tmp` 的 WinError 5），应将其记录为宿主日志 I/O 问题，不能据此宣布版本不支持或篡改被模拟 API。保留失败目录后，可在新运行目录取消 `--diagnostics` 重试；常规 progress、events、summary 和导入重建记录仍可用于验证。没有证据时不猜测是杀毒软件或其他特定程序锁文件；不要关闭系统防护。详细诊断缺失时在报告中说明，以最终原生测试补充验证。

## 入口与原生调试路线

“进入已写入代码页”是候选入口条件，不是 OEP 证明。反汇编候选位置，跟踪跳转及其上下文，判断是否仍在保护调度、SDK 或环境初始化。

MSVC x64 常见的 CRT 入口包含分配栈、调用 cookie 初始化、释放栈并转入启动代码，但编译优化、链接增量跳板、编译器与版本会改变布局。模式匹配只是候选发现，不能把一次地址或字节签名当作万能入口。绕开初始化可能破坏 TLS、静态构造或 SDK 状态；需要测试和引用分析。

可用随附 `scripts/find_crt_candidates.py sample.stage1.exe --out analysis/crt.json` 按 Machine 选择 MSVC AMD64 或 I386 入口启发式。它仅列出候选 RVA、调用/跳转目标和指令字节，不修改文件；x86 的 call+jmp 模式尤其可能命中无关函数，必须追踪调用目标和 CRT 初始化语义。多个候选可能指向同一启动过程。无匹配不代表没有 CRT 入口，应改用反汇编/动态跟踪。

跨版本时按架构、文件类型、实际保护选项和运行结果建立测试矩阵。Professional/Ultimate 的名称、版本号或相同源程序不能代替样本检查。同一原始程序在不同保护配置下可能恢复出相同 CRT 地址，仍需重新发现并验证。保持输入样本和前一版本交付物分别命名，避免将旧版输出误当成本轮结果。

动态转储时，在原始代码恢复且导入就绪的适当边界保存映像与必要状态，再用 Scylla 或等价工具重建导入。进程地址不能直接写入可重启 EXE；验证转储中没有悬空堆、模块或模拟地址。不要直接保存进程执行到 main 后的全部数据状态而忽略 CRT 重复初始化。

## PE 修复的判定

| 观察 | 检查与处理 |
| --- | --- |
| 自动输出还保留壳入口 | 找到并验证应用入口，不直接信任 stop_reason 的名字 |
| 崩溃访问模拟堆地址 | 确认是否进入了壳跳板、未恢复状态或残留保护调用 |
| 重定位后访问旧 ImageBase | 修复完整重定位覆盖；不能只保留壳的少数 relocation 项 |
| 固定基址不能装载 | 查看实际环境是否要求重定位、地址是否冲突及装载目录有效性 |
| WinError 623 | 不单凭错误文字认定系统 DLL 冲突，结合实际基址与配置排查 |
| 异常路径崩溃 | 校验 .pdata 排序、范围、UNWIND_INFO、链式项和处理器引用 |
| 启动即崩溃或不输出 | 对照 TLS、Load Config、CFG 指针、IAT、CRT 状态及真实入口 |

优先完整恢复重定位。只有完整恢复暂不可行、首选基址原生测试通过且用户可接受该限制时，才把固定基址版作为受限交付：清楚记录 ASLR 不可用、基址和可能的装载失败。不要将此案例退路自动应用于所有文件。

做最小变更对比：先测试仅修正入口的候选，再逐项调整有证据的问题。早期壳跳板在 ASLR 下的失败不自动证明最终 CRT 入口也失败；每次改变入口/初始化路线后重新核验这项假设。不要一次清除多个数据目录后将成功归因于其中某一项。

同样，清除 Load Config 可能影响安全 cookie、CFG/XFG 或其他加载器行为。仅当其包含壳特有无效信息、已逐项评估必要字段并验证替代行为时考虑清除或重建；不要机械复制示例脚本的目录置零。

## 调试与行为测试

真实调试验证推荐现有 x64dbg/WinDbg。若编写调试器，按架构定义 Win32 结构和 API 原型，处理进程退出与 DLL 文件句柄；不可把本例硬编码的 x64 DEBUG_EVENT 布局用于 x86。分别记录自然退出、调试超时、强制终止和异常代码。

VirtualQuery 模型复用模拟器内存映射，输出 28 字节 MEMORY_BASIC_INFORMATION32，按模拟地址上限、长度和写入权限处理失败。kernel32!RtlUnwind 别名仅在现有异常分派处于活动状态时委托给上游 x86 展开器，不是简单返回成功；其他未知 API 仍停止。参考：[VirtualQuery](https://learn.microsoft.com/en-us/windows/win32/api/memoryapi/nf-memoryapi-virtualquery)。

对于 Python 全局基本块回调成为瓶颈的情况，可以显式加 `--selective-hooks`（放在 `--` 前）。此选项只在本 x86 样本实测：按上游 native filter 的观测范围思路，保留映像外、恢复区、入口/已知加速点/栈复制点的 block 回调，其他内存写入、系统调用、异常及独立 SEH 监视钩子保持。它不跳过目标指令，但日志的 blocks/instruction_count 此时是**回调计数**，不是基本块总数；上游模拟 QueryPerformanceCounter 也引用该计数，时间敏感样本可能受影响。需要完整指令诊断或怀疑计数影响时关闭该选项。不要将其视为所有样本均可靠的通用加速器。

`--dump-on-failure` 可指定一个不存在的文件，停止时保留内存 PE 供分析；该转储不自动恢复完整导入或启动状态，不能直接命名为成功脱壳产物。适配器依赖当前工具的内部接口，升级工具后须重新验证。

## 入口与 PE 修复

旧版环境检查缺口可以按需添加 `--legacy-user32` 和 `--ntdll-resources`（均在 `--` 前，默认关闭）。前者提供窗口站/UOI_FLAGS/当前进程调试查询模型，MessageBoxW 只提取文本后停止；后者复制匹配 x86 catalog 的版本资源，使用 PE32 目录偏移。不要套用 x64 偏移或绕过错误消息。

模拟成功和 wrapper resolved 不保证原生改写正确。若崩溃位于新导入指令内部、返回地址或邻接栈操作，应核对精确补丁及控制流。未跟踪证实时，不把推测当作已定位的唯一根因。

有证据时可保守回退 encoded_wrapper/api_wrapper 改写，保留原包装代码和恢复的 IAT/编码槽。随附 `restore_wrapper_calls.py` 要求输入 SHA-256，按对应报告验证所有新字节、长度、边界、不重叠后输出新文件：

使用 Capstone CS_MODE_32 或 x86 调试器重新分析，不照搬 x64 CRT 指令序列、入口 RVA 或安全 cookie 初始化地址。IAT/指针为 4 字节，导入调用通常使用绝对内存操作数；恢复 x86 重定位时检查 HIGHLOW 等实际类型，不能套用 x64 DIR64 的 8 字节修复。

x86 的 SEH/SafeSEH/Load Config 与 x64 的 .pdata/unwind 不同：没有 .pdata 不能直接判定异常信息丢失；不得无条件清除 Load Config 或 SafeSEH。保留 ASLR 测试优先，只有实际换址失败证据才能说明当前候选需要受限固定基址方案。

## 真实调试验证

调试事件结构布局取决于**调试器宿主**位数，而目标寄存器上下文取决于**目标**位数。64 位 Python 的 Win32 DEBUG_EVENT/EXCEPTION_RECORD 不能因为目标 x86 就改成 32 位布局；读取或设置 WOW64 寄存器时才应使用对应上下文 API。

WOW64 进程可能先后出现原生系统断点 `0x80000003` 和 WOW64 断点 `0x4000001F`。记录并区分加载器启动事件、后续断点、访问异常和应用退出，不能把 WOW64 启动断点误判为反调试失败，也不能无条件吞掉后续所有断点。原始测试和候选测试必须使用相同处理方式。

首选现有 x32dbg/WinDbg；自写验证器应核对宿主结构大小、参数/指针类型、EOF、超时和退出事件。参考：[Microsoft WOW64 调试](https://learn.microsoft.com/en-us/windows/win32/winprog64/debugging-wow64)、[调试器异常处理](https://learn.microsoft.com/en-us/windows/win32/debug/debugger-exception-handling)、[LLVM Windows 调试事件实现](https://lldb.llvm.org/cpp_reference/DebuggerThread_8cpp_source.html)。

import argparse,json,struct
from pathlib import Path


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('pe',type=Path)
    parser.add_argument('--evidence',required=True,type=Path)
    parser.add_argument('--reconstruction',required=True,type=Path)
    parser.add_argument('--out',required=True,type=Path)
    args=parser.parse_args()
    import pefile
    evidence=json.loads(args.evidence.read_text(encoding='utf-8'))
    report=json.loads(args.reconstruction.read_text(encoding='utf-8'))
    pe=pefile.PE(data=args.pe.read_bytes());ptr=8 if pe.OPTIONAL_HEADER.Magic==0x20b else 4
    image=pe.get_memory_mapped_image();result=[]
    def key(r):return r['module'].lower(),r['symbol']
    for item in evidence['resolved_imports']:
        if item.get('source')!='final_scan_only':continue
        slot=int(item['slot_va'],0) if isinstance(item['slot_va'],str) else item['slot_va']
        references=image.count(struct.pack('<Q' if ptr==8 else '<I',slot))
        row={'module':item['module'],'symbol':item['symbol'],'slot_rva':item['slot_rva'],'section':item['section'],
             'bootstrap_evidence':sum(key(r)==key(item) for r in evidence['bootstrap_iat']),
             'encoded_metadata_evidence':sum(key(r)==key(item) for r in evidence['encoded_imports']),
             'instruction_patch_count':sum(key(r)==key(item) for r in report['patches']),
             'absolute_slot_reference_count':references,
             'status':'needs_manual_review','note':'Address equality alone is not proof of an API pointer; generated thunks are not independent evidence.'}
        result.append(row)
    args.out.parent.mkdir(parents=True,exist_ok=True)
    with args.out.open('x',encoding='utf-8') as stream:json.dump(result,stream,indent=2)
    print(json.dumps({'scan_only_candidates':len(result),'report':str(args.out.resolve())}))


if __name__=='__main__':main()

`vmprotect-unpack/scripts/compare_runs.py`:

```py
"""Execute two console programs with identical input files; does not debug them."""
import argparse
import hashlib
import json
import os
import subprocess
import time
from pathlib import Path


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('original', type=Path)
    parser.add_argument('candidate', type=Path)
    parser.add_argument('--input', action='append', required=True, type=Path, help='Raw stdin file; repeat for multiple cases')
    parser.add_argument('--out', required=True, type=Path, help='New results directory')
    parser.add_argument('--cwd', type=Path, help='Shared working directory; default is each executable parent')
    parser.add_argument('--arg', action='append', default=[], help='Shared program argument')
    parser.add_argument('--timeout', type=float, default=15)
    parser.add_argument('--execute', action='store_true', help='Explicitly enable target execution')
    args = parser.parse_args()
    if not args.execute:
        parser.error('No execution performed; --execute is required')
    if args.timeout <= 0:
        parser.error('--timeout must be positive')
    paths = [args.original.resolve(strict=True), args.candidate.resolve(strict=True)]
    inputs = [p.resolve(strict=True) for p in args.input]
    if args.cwd:
        args.cwd = args.cwd.resolve(strict=True)
    args.out.mkdir(parents=True, exist_ok=False)
    results = []
    for index, input_path in enumerate(inputs):
        row = {'input': str(input_path), 'input_sha256': digest(input_path), 'runs': []}
        for label, path in zip(('original', 'candidate'), paths):
            stdout_path = args.out / f'{index:03d}-{label}.stdout.bin'
            stderr_path = args.out / f'{index:03d}-{label}.stderr.bin'
            cwd = args.cwd or path.parent
            result = {'file': str(path), 'sha256': digest(path), 'cwd': str(cwd),
                      'arguments': args.arg, 'timed_out': False}
            start = time.monotonic()
            with input_path.open('rb') as stdin, stdout_path.open('wb') as stdout, stderr_path.open('wb') as stderr:
                try:
                    proc = subprocess.run([str(path), *args.arg], cwd=cwd, stdin=stdin,
                                          stdout=stdout, stderr=stderr, timeout=args.timeout,
                                          creationflags=subprocess.CREATE_NO_WINDOW if os.name == 'nt' else 0)
                    result['exit_code'] = proc.returncode
                except subprocess.TimeoutExpired:
                    result['timed_out'] = True
                except OSError as error:
                    result['launch_error'] = str(error)
            result.update(elapsed_seconds=round(time.monotonic()-start, 3),
                          stdout=str(stdout_path.resolve()), stderr=str(stderr_path.resolve()),
                          stdout_sha256=digest(stdout_path), stderr_sha256=digest(stderr_path))
            row['runs'].append(result)
        first, second = row['runs']
        completed = all('exit_code' in r and not r['timed_out'] for r in row['runs'])
        row['completed_equivalent'] = completed and all(first[k] == second[k] for k in ('exit_code', 'stdout_sha256', 'stderr_sha256'))
        row['both_exit_zero'] = completed and all(r['exit_code'] == 0 for r in row['runs'])
        results.append(row)
        (args.out / 'comparison.json').write_text(json.dumps(results, indent=2, ensure_ascii=False), encoding='utf-8')

"""Conservatively restore original wrapper-call bytes using an exact reconstruction report."""
import argparse
import hashlib
import json
from pathlib import Path


def restore(data,report):
    output=bytearray(data)
    selected=[]
    occupied=set()
    for patch in report['patches']:
        if patch['category'] not in ('encoded_wrapper','api_wrapper'): continue
        raw_offset=patch['patch_file_offset']
        offset=int(raw_offset,0) if isinstance(raw_offset,str) else raw_offset
        old=bytes.fromhex(patch['old_bytes']);new=bytes.fromhex(patch['new_bytes'])
        if not old or len(old)!=len(new) or not 0<=offset<=len(data)-len(new):
            raise ValueError('Invalid equal-width wrapper patch')
        positions=set(range(offset,offset+len(new)))
        if occupied & positions: raise ValueError('Overlapping wrapper patches require manual review')
        if data[offset:offset+len(new)]!=new: raise ValueError(f'Patch preimage mismatch at {offset:#x}')
        occupied.update(positions)
        selected.append((offset,old))
    if not selected: raise ValueError('No wrapper patches selected; do not claim this fallback changed anything')
    for offset,old in selected: output[offset:offset+len(old)]=old
    return bytes(output),len(selected)


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('input',type=Path)
    parser.add_argument('--report',required=True,type=Path)
    parser.add_argument('--sha256',required=True,help='Expected hash of this exact input')
    parser.add_argument('--out',required=True,type=Path)
    args=parser.parse_args()
    data=args.input.read_bytes()
    if hashlib.sha256(data).hexdigest()!=args.sha256.lower(): parser.error('Input hash mismatch')
    if args.out.exists(): parser.error('Refusing to overwrite an existing output')
    output,count=restore(data,json.loads(args.report.read_text(encoding='utf-8')))
    args.out.parent.mkdir(parents=True,exist_ok=True)
    with args.out.open('xb') as stream: stream.write(output)
    print(json.dumps({'restored_wrapper_patches':count,'sha256':hashlib.sha256(output).hexdigest(),'output':str(args.out.resolve())}))


if __name__=='__main__': main()

def firmware(self, symbol, rcx, rdx, r8, r9):
    from unpack.api_handlers import smbios_table
    provider = rcx & 0xffffffff
    details = {'compat_model': 'uefi_firmware', 'provider': hex(provider)}
    if provider == 0x4649524d:
        self.last_error = 50
        return 0, details
    if provider != 0x52534d42:
        return None, {**details, 'unsupported_provider': True}
    if symbol == 'enumsystemfirmwaretables':
        payload, output, capacity = struct.pack('<I',0), rdx, r8 & 0xffffffff
    else:
        if (rdx & 0xffffffff) != 0:
            self.last_error = 87
            return 0, details
        payload, output, capacity = smbios_table(), r8, r9 & 0xffffffff
    if not output or capacity < len(payload): return len(payload), details
    if not self.writable_range(output, len(payload)):
        self.last_error = 998
        return 0, details
    self.uc.mem_write(output, payload)
    return len(payload), details

def install_ntdll_resources(tracer):
    import pefile
    original = tracer.build_emu_module_blob
    def build(module_name, symbols, module_base, arch=tracer.MODE64):
        blob, assignments, stubs = original(module_name, symbols, module_base, arch)
        if module_name != 'ntdll.dll': return blob, assignments, stubs
        pe = pefile.PE(data=tracer.require_catalog().ntdll_bytes)
        destination = 0x60000
        versions = [e for e in pe.DIRECTORY_ENTRY_RESOURCE.entries if e.id == 16]
        if len(versions)!=1 or len(versions[0].directory.entries)!=1:
            raise ValueError('Expected one bundled ntdll version resource')
        name = versions[0].directory.entries[0]
        if name.id is None or len(name.directory.entries)!=1:
            raise ValueError('Expected numeric single-language version resource')
        lang = name.directory.entries[0]
        record = lang.data.struct
        data = pe.get_data(record.OffsetToData, record.Size)
        payload = bytearray(88)+data
        for offset in (0,24,48): struct.pack_into('<H',payload,offset+14,1)
        struct.pack_into('<II',payload,16,16,0x80000018)
        struct.pack_into('<II',payload,40,name.id,0x80000030)
        struct.pack_into('<II',payload,64,lang.id,72)
        struct.pack_into('<IIII',payload,72,destination+88,len(data),record.CodePage,0)
        if destination+len(payload)>len(blob): raise ValueError('Version resource too large')
        out = bytearray(blob)
        if any(out[destination:destination+len(payload)]):
            raise ValueError('Resource destination is occupied; unsupported upstream layout')
        out[destination:destination+len(payload)] = payload
        # The upstream synthetic optional header begins at 0x98.
        if pe.FILE_HEADER.Machine != arch.machine or out[0x98:0x9a] != struct.pack('<H',arch.magic):
            raise ValueError('Catalog/synthetic PE architecture mismatch')
        resource_directory = 0x118 if arch.bits == 64 else 0x108
        struct.pack_into('<II', out, resource_directory, destination, len(payload))
        return bytes(out), assignments, stubs
    tracer.build_emu_module_blob = build
