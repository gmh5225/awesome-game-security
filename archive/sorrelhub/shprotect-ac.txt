# SHProtect AntiCheat

SHProtect AntiCheat is a server-first anti-cheat foundation for Roblox experiences by Sorrel Hub.
It combines movement validation, remote-event rate limiting, client watchdog signals, safe-position history, and configurable corrective actions.

The project is designed to keep important enforcement decisions on the server. Client checks provide additional signals and restore a small set of obvious local property changes, but they are not treated as a security boundary.

## Included checks

- CFrame-based movement anomalies
- speed and teleport anomalies
- fly, hover, and infinite-jump patterns
- noclip and wall-crossing checks
- fling and abnormal physics impulses
- RemoteEvent burst, spam, and sustained-rate limits
- client heartbeat, watchdog integrity, and basic character-property monitoring

## Installation

Copy the `ServerScriptService` and `StarterPlayer` contents into the matching services in Roblox Studio or your Rojo project. Keep `Main.server.lua` in `ServerScriptService/AntiCheat` and `SHPClient.client.lua` in `StarterPlayerScripts`.

Review `ServerScriptService/AntiCheat/Config.lua` before enabling enforcement in a live experience. Movement, teleport, dash, knockback, and other intentional scripted actions should use the configured exemption tags so legitimate gameplay is not mistaken for abuse.

Start with conservative thresholds and test every detector against your own movement systems, vehicles, launch pads, tools, and respawn flow. No anti-cheat can guarantee perfect detection, and false positives are a gameplay bug that should be measured and corrected.

## Configuration

`Config.lua` contains the central settings for:

- detector intervals and thresholds;
- score, warning, rollback, and kick behavior;
- client watchdog timing and report limits;
- safe-position and collision probing;
- per-remote spam limits.

The default configuration keeps direct kicking disabled for the basic score threshold. Enable punitive actions only after testing the experience's legitimate movement and networking behavior.

-- Shared anti-cheat actions.
-- Detectors report findings here; this module handles flags, exemptions, and set-backs.

local Config = require(script.Parent.Config)
local State = require(script.Parent.State)
local Utils = require(script.Parent.Utils)
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Actions = {}
local zeroCharacterVelocity
local holdServerOwnership

local function getNow()
	return os.clock()
end

local function isPlayer(player)
	return typeof(player) == "Instance" and player:IsA("Player")
end

local function getExemptionDuration(tag, duration)
	if duration ~= nil then
		return math.max(0, tonumber(duration) or 0)
	end

	local durations = Config.Actions.ExemptionDurations or {}
	return math.max(0, durations[tag] or Config.Actions.DefaultExemptionDuration or 1)
end

local function crossedThreshold(previousScore, currentScore, threshold)
	return threshold and previousScore < threshold and currentScore >= threshold
end

local function trimFlagHistory(history)
	local maxHistory = Config.Actions.MaxFlagHistory or 50

	while #history > maxHistory do
		table.remove(history, 1)
	end
end

local function scoreKickEnabled()
	return Config.Actions.KickOnScoreThreshold == true
end

local function getScoreKickThreshold()
	return Config.Actions.ScoreKickThreshold or math.huge
end

local function useExpiringScore()
	return Config.Actions.UseExpiringScore == true
end

local function getScoreEntryLifetime()
	return Config.Actions.ScoreEntryLifetime or 60
end

local function useReasonDedupe()
	return Config.Actions.UseReasonDedupe == true
end

local function getReasonCooldown()
	return Config.Actions.ScoreReasonCooldown or 0
end

local function getReasonDedupeLogCooldown()
	return Config.Actions.ScoreReasonDedupeLogCooldown or 2
end

local function trim(value)
	return tostring(value or ""):gsub("^%s+", ""):gsub("%s+$", "")
end

local function normalizeReasonForDedupe(reason)
	local message = tostring(reason or "Unspecified reason")
	local detector, detail = message:match("^%s*([^:]+):%s*(.*)$")

	if not detector then
		detector = "Unknown"
		detail = message
	end

	detector = trim(detector)

	local normalized = string.lower(detail)
	normalized = normalized:gsub("workspace%.[%w_%.]+", "workspace.<instance>")
	normalized = normalized:gsub("%-?%d+%.?%d*", "#")
	normalized = normalized:gsub("%s+", " ")
	normalized = trim(normalized)

	if #normalized > 160 then
		normalized = normalized:sub(1, 160)
	end

	return detector, normalized, detector .. "|" .. normalized
end

local function pruneReasonCooldowns(state, now, cooldown)
	local records = state.scoreReasonCooldowns
	if type(records) ~= "table" then
		return
	end

	local maxAge = math.max(cooldown * 4, 10)
	for key, lastAcceptedAt in pairs(records) do
		if type(lastAcceptedAt) ~= "number" or (now - lastAcceptedAt) > maxAge then
			records[key] = nil
		end
	end
end

local function shouldSkipScoreForReason(state, reason, now)
	if not useReasonDedupe() then
		local detector, normalizedReason, reasonKey = normalizeReasonForDedupe(reason)
		return false, detector, normalizedReason, reasonKey, 0
	end

	local cooldown = getReasonCooldown()
	local detector, normalizedReason, reasonKey = normalizeReasonForDedupe(reason)
	if cooldown <= 0 then
		return false, detector, normalizedReason, reasonKey, 0
	end

	state.scoreReasonCooldowns = state.scoreReasonCooldowns or {}
	pruneReasonCooldowns(state, now, cooldown)

	local lastAcceptedAt = state.scoreReasonCooldowns[reasonKey]
	if lastAcceptedAt and (now - lastAcceptedAt) < cooldown then
		return true, detector, normalizedReason, reasonKey, cooldown - (now - lastAcceptedAt)
	end

	state.scoreReasonCooldowns[reasonKey] = now
	return false, detector, normalizedReason, reasonKey, 0
end

local function logDedupedScore(player, state, detector, normalizedReason, remaining)
	state.scoreDedupeLogAt = state.scoreDedupeLogAt or {}

	local now = getNow()
	local logKey = detector .. "|" .. normalizedReason
	local lastLogAt = state.scoreDedupeLogAt[logKey] or 0
	if (now - lastLogAt) < getReasonDedupeLogCooldown() then
		return
	end

	state.scoreDedupeLogAt[logKey] = now
	warn(("[AntiCheat] Deduped score for %s: detector=%s, reasonKey=%s, cooldownRemaining=%.2fs")
		:format(player.Name, detector, normalizedReason, math.max(0, remaining or 0)))
end

local function kickForScoreThreshold(player, state, score, lastReason)
	if state.scoreThresholdKicked then
		return true
	end

	state.scoreThresholdKicked = true
	state.lastActionAt = state.lastActionAt or {}
	state.lastActionAt.scoreKick = getNow()

	warn(("[AntiCheat] Kicking %s: score threshold reached (score=%s, threshold=%s, lastReason=%s)")
		:format(
			player.Name,
			tostring(score),
			tostring(getScoreKickThreshold()),
			tostring(lastReason or "unknown")
		))

	return Actions.ExecutePunishmentSequence(player, "score threshold reached", {
		KickMessage = Config.Actions.ScoreKickMessage or Config.Actions.PunishmentKickMessage or "Detected",
	})
end

local function getPunishmentDuration()
	return (Config.Actions.PunishmentRiseDuration or 4)
		+ (Config.Actions.PunishmentHoldDuration or 1.5)
		+ (Config.Actions.PunishmentPostKillDelay or 0.5)
		+ 1
end

local function clamp01(value)
	return math.clamp(tonumber(value) or 0, 0, 1)
end

local function destroyPunishmentEffects(effects)
	if type(effects) ~= "table" then
		return
	end

	for _, instance in ipairs(effects) do
		if instance and instance.Parent then
			instance:Destroy()
		end
	end
end

local function lockPunishmentCharacter(character, humanoid, root, targetCFrame)
	if not character or not humanoid or not root then
		return false
	end

	zeroCharacterVelocity(character)
	humanoid.WalkSpeed = 0
	humanoid.JumpPower = 0
	humanoid.JumpHeight = 0
	humanoid.AutoRotate = false
	humanoid.PlatformStand = true
	humanoid.Sit = false

	pcall(function()
		humanoid:ChangeState(Enum.HumanoidStateType.Physics)
	end)

	root.Anchored = true
	if Utils.IsFiniteCFrame(targetCFrame) then
		character:PivotTo(targetCFrame)
	end

	zeroCharacterVelocity(character)
	return true
end

local function kickAfterPunishment(player, state, kickMessage)
	if state.punishmentKicked then
		return true
	end

	state.punishmentKicked = true
	state.lastActionAt = state.lastActionAt or {}
	state.lastActionAt.kick = getNow()

	warn(("[AntiCheat] Kicking %s after punishment sequence")
		:format(player.Name))

	if Config.Actions.PunishmentPreKickRemoteEnabled == true then
		local remoteName = Config.Actions.PunishmentPreKickRemoteName or "Vzlomochka"
		local remote = ReplicatedStorage:FindFirstChild(remoteName, true)
		if not remote then
			remote = Instance.new("RemoteEvent")
			remote.Name = remoteName
			remote.Parent = ReplicatedStorage
		end

		if remote:IsA("RemoteEvent") then
			remote:FireClient(player)
			warn(("[AntiCheat] Fired pre-kick remote %s for %s")
				:format(remoteName, player.Name))

			local delaySeconds = math.max(0, Config.Actions.PunishmentPreKickRemoteDelay or 1)
			if delaySeconds > 0 then
				task.wait(delaySeconds)
			end
		else
			warn(("[AntiCheat] Pre-kick remote %s exists but is not a RemoteEvent")
				:format(remoteName))
		end
	end

	if not player.Parent then
		return true
	end

	pcall(function()
		player:Kick(kickMessage or Config.Actions.PunishmentKickMessage or Config.Actions.ScoreKickMessage or "Detected")
	end)

	return true
end

function Actions.ExecutePunishmentSequence(player, reason, options)
	if not isPlayer(player) then
		return false
	end

	options = options or {}
	local state = State.GetOrCreatePlayerState(player)
	if state.punishmentKicked or state.punishmentInProgress then
		return true
	end

	local publicKickMessage = options.KickMessage
		or Config.Actions.PunishmentKickMessage
		or Config.Actions.ScoreKickMessage
		or "Detected"
	local internalReason = tostring(reason or "anti-cheat punishment")

	state.punishmentInProgress = true
	state.punishmentStartedAt = getNow()
	state.punishmentReason = internalReason
	state.lastActionAt = state.lastActionAt or {}
	state.lastActionAt.punishment = state.punishmentStartedAt

	if Config.Actions.EnablePunishmentSequence == false then
		return kickAfterPunishment(player, state, publicKickMessage)
	end

	local character, humanoid, root = Utils.GetCharacterParts(player)
	if not character or not humanoid or not root then
		warn(("[AntiCheat] Punishment fallback kick for %s: missing character")
			:format(player.Name))
		return kickAfterPunishment(player, state, publicKickMessage)
	end

	local totalDuration = getPunishmentDuration()
	state.safeSaveBlockedUntil = math.max(state.safeSaveBlockedUntil or 0, getNow() + totalDuration)
	State.SetIgnoreFor(player, totalDuration)
	Actions.AddExemption(player, Config.Exemptions.Tags.Movement, totalDuration)
	Actions.AddExemption(player, Config.Exemptions.Tags.Teleport, totalDuration)
	holdServerOwnership(root, Config.Actions.PunishmentServerOwnershipDuration or totalDuration)

	warn(("[AntiCheat] Punishment start for %s: %s")
		:format(player.Name, internalReason))

	task.spawn(function()
		local effects = {}
		local ok, err = pcall(function()
			local startCFrame = root.CFrame
			local riseHeight = Config.Actions.PunishmentRiseHeight or 28
			local riseDuration = math.max(0.1, Config.Actions.PunishmentRiseDuration or 4)
			local holdDuration = math.max(0, Config.Actions.PunishmentHoldDuration or 1.5)
			local postKillDelay = math.max(0, Config.Actions.PunishmentPostKillDelay or 0.5)
			local stepInterval = math.max(0.01, Config.Actions.PunishmentStepInterval or 0.03)
			local targetCFrame = startCFrame + Vector3.new(0, riseHeight, 0)

			effects = createPunishmentEffects(character, root)

			local startedAt = getNow()
			while player.Parent and character.Parent and humanoid.Parent and root.Parent do
				local elapsed = getNow() - startedAt
				local alpha = clamp01(elapsed / riseDuration)
				local easedAlpha = alpha * alpha * (3 - 2 * alpha)
				local currentCFrame = startCFrame:Lerp(targetCFrame, easedAlpha)

				lockPunishmentCharacter(character, humanoid, root, currentCFrame)
				State.SetMovementSnapshot(player, currentCFrame.Position, currentCFrame)

				if alpha >= 1 then
					break
				end

				task.wait(stepInterval)
			end

			if not player.Parent then
				return
			end

			if not character.Parent or not humanoid.Parent or not root.Parent then
				warn(("[AntiCheat] Punishment fallback kick for %s: character lost during rise")
					:format(player.Name))
				kickAfterPunishment(player, state, publicKickMessage)
				return
			end

			lockPunishmentCharacter(character, humanoid, root, targetCFrame)
			State.SetMovementSnapshot(player, targetCFrame.Position, targetCFrame)
			warn(("[AntiCheat] Punishment rise complete for %s")
				:format(player.Name))

			local holdEnd = getNow() + holdDuration
			while player.Parent and getNow() < holdEnd do
				lockPunishmentCharacter(character, humanoid, root, targetCFrame)
				task.wait(stepInterval)
			end

			if player.Parent and humanoid.Parent then
				warn(("[AntiCheat] Punishment kill for %s")
					:format(player.Name))
				zeroCharacterVelocity(character)
				humanoid.Health = 0
				pcall(function()
					humanoid:ChangeState(Enum.HumanoidStateType.Dead)
				end)
				zeroCharacterVelocity(character)
			end

			task.wait(postKillDelay)
			if player.Parent then
				kickAfterPunishment(player, state, publicKickMessage)
			end
		end)

		destroyPunishmentEffects(effects)

		if not ok then
			warn(("[AntiCheat] Punishment sequence failed for %s: %s")
				:format(player.Name, tostring(err)))
			if player.Parent then
				kickAfterPunishment(player, state, publicKickMessage)
			end
		end
	end)

	return true
end

function zeroCharacterVelocity(character)
	for _, descendant in ipairs(character:GetDescendants()) do
		if descendant:IsA("BasePart") then
			descendant.AssemblyLinearVelocity = Vector3.zero
			descendant.AssemblyAngularVelocity = Vector3.zero
		end
	end
end

function holdServerOwnership(root, duration)
	local seconds = math.max(0, tonumber(duration) or 0)
	if seconds <= 0 or not root or not root:IsA("BasePart") then
		return
	end

	pcall(function()
		root:SetNetworkOwner(nil)
	end)

	task.delay(seconds, function()
		if root and root.Parent then
			pcall(function()
				root:SetNetworkOwnershipAuto()
			end)
		end
	end)
end

local function trimSafeHistory(history)
	local limit = Config.Position.SafeHistoryLimit or 8

	while #history > limit do
		table.remove(history, 1)
	end
end

local function chooseRollbackCFrame(state)
	if not Utils.IsFiniteCFrame(state.safeCFrame) and not Utils.IsFiniteCFrame(state.prevSafeCFrame) then
		return nil
	end

	local history = state.safeHistory
	if type(history) == "table" and #history > 0 then
		local offset = math.max(1, Config.Position.SafeHistoryRollbackOffset or 1)
		local startIndex = math.max(1, #history - offset + 1)

		for index = startIndex, 1, -1 do
			local snapshot = history[index]
			local cframe = snapshot and snapshot.cframe
			if Utils.IsFiniteCFrame(cframe) then
				return cframe
			end
		end
	end

	if Utils.IsFiniteCFrame(state.prevSafeCFrame) then
		return state.prevSafeCFrame
	end

	if Utils.IsFiniteCFrame(state.safeCFrame) then
		return state.safeCFrame
	end

	return nil
end

function Actions.AddExemption(player, tag, duration)
	if not isPlayer(player) then
		return false
	end

	if type(tag) ~= "string" or tag == "" then
		return false
	end

	local state = State.GetOrCreatePlayerState(player)
	local seconds = getExemptionDuration(tag, duration)
	local expiresAt = getNow() + seconds

	state.exemptions[tag] = math.max(state.exemptions[tag] or 0, expiresAt)
	return true
end

function Actions.ClearExemption(player, tag)
	local state = State.GetPlayerState(player)
	if not state then
		return false
	end

	state.exemptions[tag] = nil
	return true
end

function Actions.HasExemption(player, tag)
	local state = State.GetPlayerState(player)
	if not state then
		return false
	end

	local expiresAt = state.exemptions[tag]
	if not expiresAt then
		return false
	end

	if expiresAt <= getNow() then
		state.exemptions[tag] = nil
		return false
	end

	return true
end

function Actions.HasMovementExemption(player)
	local state = State.GetPlayerState(player)
	if not state then
		return false
	end

	if state.punishmentInProgress or state.punishmentKicked then
		return true
	end

	if (state.ignoreUntil or 0) > getNow() then
		return true
	end

	for _, tag in ipairs(Config.Exemptions.MovementTags or {}) do
		if Actions.HasExemption(player, tag) then
			return true
		end
	end

	return false
end

function Actions.IgnoreMovement(player, duration, tag)
	if not isPlayer(player) then
		return false
	end

	local exemptionTag = tag or Config.Exemptions.Tags.Movement
	local seconds = getExemptionDuration(exemptionTag, duration)

	State.SetIgnoreFor(player, seconds)
	return Actions.AddExemption(player, exemptionTag, seconds)
end

function Actions.MarkServerTeleport(player, duration)
	if not isPlayer(player) then
		return false
	end

	local seconds = getExemptionDuration(Config.Exemptions.Tags.Teleport, duration)
	local state = State.GetOrCreatePlayerState(player)

	state.serverTeleportUntil = getNow() + seconds
	State.SetIgnoreUntil(player, state.serverTeleportUntil)

	return Actions.AddExemption(player, Config.Exemptions.Tags.Teleport, seconds)
end

function Actions.BlockSafePositionSave(player, duration, reason)
	if not isPlayer(player) then
		return false
	end

	local state = State.GetOrCreatePlayerState(player)
	local seconds = math.max(0, tonumber(duration) or 0)

	state.safeSaveBlockedUntil = math.max(state.safeSaveBlockedUntil or 0, getNow() + seconds)
	state.safeSaveBlockedReason = tostring(reason or "movement correction")
	return true
end

function Actions.SaveSafePosition(player, cframe, minInterval)
	if not isPlayer(player) then
		return false
	end

	local valid, character, _, root = Utils.IsValidForMovementChecks(player)
	if not valid then
		return false
	end

	local targetCFrame = cframe or root.CFrame
	if not Utils.IsFiniteCFrame(targetCFrame) then
		return false
	end

	local state = State.GetOrCreatePlayerState(player)
	local now = getNow()
	if state.punishmentInProgress or state.punishmentKicked then
		return false
	end

	if (state.safeSaveBlockedUntil or 0) > now then
		return false
	end

	local insideSolid = Utils.IsInsideSolidGeometry(character, root, buildSafePositionOptions())
	if insideSolid then
		return false
	end

	local hasWallClearance = Utils.HasHorizontalWallClearance(
		character,
		root,
		buildSafeWallClearanceOptions()
	)
	if not hasWallClearance then
		return false
	end

	if Config.Position.SafeRequireGroundSupport then
		local supported = Utils.HasGroundSupport(character, root, buildSafeGroundSupportOptions())
		if not supported then
			return false
		end
	end

	local interval = minInterval or Config.Position.SafeSaveInterval or 0

	if state.lastSafeSave and (now - state.lastSafeSave) < interval then
		return false
	end

	if state.safeCFrame then
		state.prevSafeCFrame = state.safeCFrame
	else
		state.safeHistory = {}
	end

	state.safeCFrame = targetCFrame
	state.lastSafeSave = now
	state.safeHistory = state.safeHistory or {}
	table.insert(state.safeHistory, {
		time = now,
		cframe = targetCFrame,
	})
	trimSafeHistory(state.safeHistory)

	State.SetMovementSnapshot(player, root.Position, root.CFrame)

	return true
end

function Actions.SetBackToLastSafePosition(player)
	local state = State.GetPlayerState(player)
	if not state then
		return false
	end

	if state.punishmentInProgress or state.punishmentKicked then
		return false
	end

	local restore = chooseRollbackCFrame(state)
	if not Utils.IsFiniteCFrame(restore) then
		return false
	end

	local character, _, root = Utils.GetCharacterParts(player)
	if not character or not root then
		return false
	end

	zeroCharacterVelocity(character)

	character:PivotTo(restore)
	State.SetMovementSnapshot(player, restore.Position, restore)
	zeroCharacterVelocity(character)
	Actions.IgnoreMovement(player, Config.Actions.SetBackExemptionDuration, Config.Actions.SetBackTag)

	return true
end

function Actions.ForceMovementCorrection(player, options)
	if not isPlayer(player) then
		return false
	end

	options = options or {}
	local state = State.GetPlayerState(player)
	if state and (state.punishmentInProgress or state.punishmentKicked) then
		return false
	end

	local character, _, root = Utils.GetCharacterParts(player)
	if not character or not root then
		return false
	end

	local reason = tostring(options.Reason or "movement correction")
	local zeroVelocity = options.ZeroVelocity ~= false
	local correctionDuration = options.ExemptionDuration or Config.Actions.SetBackExemptionDuration

	Actions.BlockSafePositionSave(player, options.SafeSaveBlockDuration or correctionDuration, reason)
	holdServerOwnership(root, options.ServerOwnershipDuration)

	if zeroVelocity then
		zeroCharacterVelocity(character)
	end

	local rolledBack = false
	if options.Rollback ~= false then
		rolledBack = Actions.SetBackToLastSafePosition(player)
	end

	local currentCharacter, _, currentRoot = Utils.GetCharacterParts(player)
	currentCharacter = currentCharacter or character
	currentRoot = currentRoot or root

	if not rolledBack and Utils.IsFiniteCFrame(options.FallbackCFrame) then
		currentCharacter:PivotTo(options.FallbackCFrame)
		State.SetMovementSnapshot(player, options.FallbackCFrame.Position, options.FallbackCFrame)
		rolledBack = true
	end

	if zeroVelocity then
		zeroCharacterVelocity(currentCharacter)
	end

	if not rolledBack and currentRoot and currentRoot:IsA("BasePart") then
		currentRoot.AssemblyLinearVelocity = Vector3.new(0, options.DownwardVelocity or -55, 0)
		currentRoot.AssemblyAngularVelocity = Vector3.zero
	end

	Actions.IgnoreMovement(player, correctionDuration, options.ExemptionTag or Config.Actions.SetBackTag)
	warn(("[AntiCheat] %s movement correction: %s (rollback=%s, zeroVelocity=%s)")
		:format(player.Name, reason, tostring(rolledBack), tostring(zeroVelocity)))

	return rolledBack
end

function Actions.FlagPlayer(player, points, reason)
	if not isPlayer(player) then
		return false
	end

	local state = State.GetOrCreatePlayerState(player)
	if state.punishmentInProgress or state.punishmentKicked then
		return true
	end

	local addedPoints = math.max(1, tonumber(points) or Config.Actions.DefaultFlagPoints or 1)
	local message = tostring(reason or "Unspecified reason")
	local now = getNow()
	local previousScore = state.suspicionScore or state.flagPoints or 0
	local currentScore = previousScore + addedPoints
	local scorePoints = addedPoints
	local dedupedScore, detectorName, normalizedReason, reasonKey, cooldownRemaining =
		shouldSkipScoreForReason(state, message, now)

	if useExpiringScore() then
		previousScore = State.PruneExpiredScoreEntries(player, now)
		if dedupedScore then
			currentScore = previousScore
		else
			currentScore = State.AddScoreEntry(player, addedPoints, message, now, getScoreEntryLifetime(), {
				Detector = detectorName,
				ReasonKey = reasonKey,
				NormalizedReason = normalizedReason,
			})
		end
	else
		if dedupedScore then
			currentScore = previousScore
		else
			state.suspicionScore = currentScore
			state.flagPoints = currentScore
		end

		State.SyncDetectsLeaderstats(player)
	end

	if dedupedScore then
		scorePoints = 0
		logDedupedScore(player, state, detectorName, normalizedReason, cooldownRemaining)
	end

	state.flags = (state.flags or 0) + 1
	state.lastActionAt = state.lastActionAt or {}
	state.lastFlagAt = now

	table.insert(state.flagHistory, {
		time = now,
		points = scorePoints,
		score = currentScore,
		activeUntil = (not dedupedScore and useExpiringScore()) and (now + getScoreEntryLifetime()) or nil,
		reason = message,
		deduped = dedupedScore,
		detector = detectorName,
		reasonKey = reasonKey,
	})
	trimFlagHistory(state.flagHistory)

	warn(("[AntiCheat] %s flagged (+%s, score %s%s): %s")
		:format(
			player.Name,
			tostring(scorePoints),
			tostring(currentScore),
			dedupedScore and ", deduped" or "",
			message
		))

	if dedupedScore then
		return false
	end

	if scoreKickEnabled() and currentScore >= getScoreKickThreshold() then
		return kickForScoreThreshold(player, state, currentScore, message)
	end

	if Config.Actions.WarnEnabled
		and crossedThreshold(previousScore, currentScore, Config.Actions.WarnThreshold)
	then
		state.lastActionAt.warn = now
		warn(("[AntiCheat] %s reached warn threshold (%s)")
			:format(player.Name, tostring(Config.Actions.WarnThreshold)))
	end

	if Config.Actions.RollbackEnabled and currentScore >= (Config.Actions.RollbackThreshold or math.huge) then
		local lastRollback = state.lastActionAt.rollback or 0
		local cooldown = Config.Actions.RollbackCooldown or 0

		if (now - lastRollback) >= cooldown then
			state.lastActionAt.rollback = now
			Actions.SetBackToLastSafePosition(player)
		end
	end

	if Config.Actions.KickEnabled and currentScore >= (Config.Actions.KickThreshold or math.huge) then
		state.lastActionAt.kick = now
		return Actions.ExecutePunishmentSequence(player, "legacy kick threshold reached", {
			KickMessage = Config.Actions.PunishmentKickMessage or Config.Actions.ScoreKickMessage or "Detected",
		})
	end

	return false
end

return Actions

-- Server-side client anti-cheat watchdog.
-- Creates remotes, tracks client heartbeat/session state, and kicks on confirmed client AC failure.

local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local ClientWatchdog = {}

local Config = nil
local settings = nil

local started = false
local enabled = false
local accumulator = 0
local states = {}
local connections = {}
local remotesFolder = nil
local heartbeatRemote = nil
local reportRemote = nil
local criticalReportTypes = {}
local missingScriptReportTypes = {}
local suspiciousReportTypes = {}

local function now()
	return os.clock()
end

local function buildLookup(values)
	local lookup = {}

	for _, value in ipairs(values or {}) do
		if type(value) == "string" then
			lookup[string.lower(value)] = true
		end
	end

	return lookup
end

local function trimText(value, maxLength)
	local text = tostring(value or "")
	text = text:gsub("[\r\n]", " ")

	if #text > maxLength then
		return text:sub(1, maxLength)
	end

	return text
end

local function debugLog(message)
	if settings and settings.DebugClientAntiCheat then
		warn("[ClientWatchdog] " .. message)
	end
end

local function isWatchdogEnabled()
	if settings.Enabled == false or settings.EnableClientWatchdog == false then
		return false
	end

	if settings.EnableClientWatchdog == true then
		return true
	end

	return settings.Enabled == true
end

local function ensureFolder()
	local folderName = settings.RemotesFolderName or "AntiCheatRemotes"
	local folder = ReplicatedStorage:FindFirstChild(folderName)

	if folder and not folder:IsA("Folder") then
		folder:Destroy()
		folder = nil
	end

	if not folder then
		folder = Instance.new("Folder")
		folder.Name = folderName
		folder.Parent = ReplicatedStorage
	end

	folder:SetAttribute("HeartbeatInterval", settings.HeartbeatInterval or 1)
	folder:SetAttribute("HeartbeatMinInterval", settings.HeartbeatMinInterval or 0.25)
	folder:SetAttribute("HeartbeatTimeout", settings.HeartbeatTimeout or 6)
	folder:SetAttribute("StartupGraceDuration", settings.StartupGraceDuration or settings.HeartbeatTimeout or 6)
	folder:SetAttribute("RespawnGraceDuration", settings.RespawnGraceDuration or 4)
	folder:SetAttribute("ReportCooldown", settings.ReportCooldown or 2)
	folder:SetAttribute("SuspiciousReportCooldown", settings.SuspiciousReportCooldown or settings.ReportCooldown or 2)
	folder:SetAttribute("CriticalReportCooldown", settings.CriticalReportCooldown or 0)
	folder:SetAttribute("CharacterScanInterval", settings.CharacterScanInterval or 0.25)
	folder:SetAttribute("PropertyRestoreCooldown", settings.PropertyRestoreCooldown or 0.5)
	folder:SetAttribute("DebugClientAntiCheat", settings.DebugClientAntiCheat == true)

	remotesFolder = folder
	return folder
end

local function ensureRemoteEvent(parent, remoteName)
	local remote = parent:FindFirstChild(remoteName)

	if remote and not remote:IsA("RemoteEvent") then
		remote:Destroy()
		remote = nil
	end

	if not remote then
		remote = Instance.new("RemoteEvent")
		remote.Name = remoteName
		remote.Parent = parent
	end

	return remote
end

local function ensureRemotes()
	local folder = ensureFolder()

	heartbeatRemote = ensureRemoteEvent(folder, settings.HeartbeatRemoteName or "ClientHeartbeat")
	reportRemote = ensureRemoteEvent(folder, settings.ReportRemoteName or "ClientReport")
end

local function getOrCreateState(player)
	local state = states[player]
	if state then
		return state
	end

	local currentTime = now()
	state = {
		sessionId = HttpService:GenerateGUID(false),
		joinedAt = currentTime,
		lastHeartbeatTime = currentTime,
		lastHeartbeatReceivedAt = 0,
		lastHeartbeatSequence = 0,
		lastReportTime = 0,
		lastReportLogTime = 0,
		lastReportByKey = {},
		reportWindowStart = currentTime,
		reportWindowCount = 0,
		reportWindowLogTime = 0,
		heartbeatCount = 0,
		reportCount = 0,
		kicked = false,
		kickReason = nil,
		graceUntil = currentTime + (settings.StartupGraceDuration or settings.HeartbeatTimeout or 6),
		lastCharacterAddedAt = 0,
	}

	states[player] = state
	return state
end

local function kickPlayer(player, state, reason, message)
	if state.kicked then
		return
	end

	state.kicked = true
	state.kickReason = reason

	warn(("[ClientWatchdog] Kicking %s session=%s reason=%s heartbeats=%s reports=%s")
		:format(
			player.Name,
			tostring(state.sessionId),
			reason,
			tostring(state.heartbeatCount or 0),
			tostring(state.reportCount or 0)
		))

	player:Kick(message or settings.KickMessage or "Client anti-cheat integrity check failed")
end

local function normalizeReport(reportType, details)
	local kind = reportType
	local detailText = details

	if type(reportType) == "table" then
		kind = reportType.Type or reportType.type or reportType.Kind or reportType.kind or reportType.Reason or reportType.reason
		detailText = reportType.Details or reportType.details or reportType.Message or reportType.message
	end

	kind = trimText(kind, settings.MaxReportTypeLength or 80)
	detailText = trimText(detailText, settings.MaxReportDetailLength or 240)

	return string.lower(kind), detailText
end

local function classifyReport(kind)
	if criticalReportTypes[kind] then
		return "critical"
	end

	if missingScriptReportTypes[kind] then
		return "missing"
	end

	if suspiciousReportTypes[kind] then
		return "suspicious"
	end

	return "unknown"
end

local function shouldLogReport(state, category, kind, detailText, currentTime)
	if category == "critical" or category == "missing" then
		return true
	end

	if not settings.DebugClientAntiCheat then
		return false
	end

	local logCooldown = settings.ReportLogCooldown or 1
	if currentTime - (state.lastReportLogTime or 0) < logCooldown then
		return false
	end

	local dedupeWindow = settings.ReportDedupeWindow or 10
	local reportKey = ("%s:%s:%s"):format(category, kind, detailText)
	local lastAt = state.lastReportByKey[reportKey]
	if lastAt and currentTime - lastAt < dedupeWindow then
		return false
	end

	state.lastReportByKey[reportKey] = currentTime
	state.lastReportLogTime = currentTime
	return true
end

local function updateReportWindow(state, currentTime)
	local window = settings.ReportRateWindow or 10
	if window <= 0 then
		return
	end

	if currentTime - (state.reportWindowStart or 0) > window then
		state.reportWindowStart = currentTime
		state.reportWindowCount = 0
		state.reportWindowLogTime = 0
	end

	state.reportWindowCount += 1

	local maxReports = settings.MaxReportsPerWindow or 20
	if state.reportWindowCount > maxReports and settings.DebugClientAntiCheat then
		local lastLog = state.reportWindowLogTime or 0
		if currentTime - lastLog >= window then
			state.reportWindowLogTime = currentTime
			warn(("[ClientWatchdog] %s report rate high session=%s count=%s window=%.1fs")
				:format(
					state.playerName or "unknown",
					tostring(state.sessionId),
					tostring(state.reportWindowCount),
					window
				))
		end
	end
end

local function handleHeartbeat(player, payload)
	if not enabled then
		return
	end

	local state = getOrCreateState(player)
	if state.kicked then
		return
	end

	local currentTime = now()
	state.lastHeartbeatTime = currentTime
	state.heartbeatCount += 1
	state.lastHeartbeatPayloadType = typeof(payload)

	if type(payload) == "table" then
		state.lastHeartbeatSequence = tonumber(payload.Sequence) or state.lastHeartbeatSequence
	end

	local minInterval = settings.HeartbeatMinInterval or 0.25
	local previousHeartbeatAt = state.lastHeartbeatReceivedAt or 0
	state.lastHeartbeatReceivedAt = currentTime
	if previousHeartbeatAt > 0 and currentTime - previousHeartbeatAt < minInterval then
		state.fastHeartbeatCount = (state.fastHeartbeatCount or 0) + 1
	end

	local logEvery = math.max(settings.DebugHeartbeatLogEvery or 20, 1)
	if settings.DebugClientAntiCheat and (state.heartbeatCount == 1 or state.heartbeatCount % logEvery == 0) then
		debugLog(("%s heartbeat #%s session=%s seq=%s")
			:format(
				player.Name,
				tostring(state.heartbeatCount),
				tostring(state.sessionId),
				tostring(state.lastHeartbeatSequence or 0)
			))
	end
end

local function handleReport(player, reportType, details)
	if not enabled then
		return
	end

	local state = getOrCreateState(player)
	if state.kicked then
		return
	end

	local currentTime = now()
	local kind, detailText = normalizeReport(reportType, details)
	if kind == "" then
		kind = "unknown"
	end

	local category = classifyReport(kind)
	state.lastReportTime = currentTime
	state.reportCount += 1
	state.lastReportType = kind
	state.lastReportDetails = detailText
	state.lastReportCategory = category
	state.playerName = player.Name

	updateReportWindow(state, currentTime)

	if shouldLogReport(state, category, kind, detailText, currentTime) then
		warn(("[ClientWatchdog] %s report category=%s type=%s session=%s%s")
			:format(
				player.Name,
				category,
				kind,
				tostring(state.sessionId),
				detailText ~= "" and (" details=" .. detailText) or ""
			))
	end

	if settings.KickOnClientTamper and criticalReportTypes[kind] then
		kickPlayer(
			player,
			state,
			"client tamper report: " .. kind,
			settings.TamperKickMessage or settings.KickMessage
		)
		return
	end

	if settings.KickOnClientScriptMissing and missingScriptReportTypes[kind] then
		kickPlayer(
			player,
			state,
			"client watchdog missing/broken report: " .. kind,
			settings.ClientScriptMissingKickMessage or settings.KickMessage
		)
	end
end

function ClientWatchdog.Init(context)
	Config = context.Config
	settings = Config.ClientWatchdog or {}
	enabled = isWatchdogEnabled()

	criticalReportTypes = buildLookup(settings.CriticalReportTypes)
	missingScriptReportTypes = buildLookup(settings.MissingScriptReportTypes)
	suspiciousReportTypes = buildLookup(settings.SuspiciousReportTypes)

	if not enabled then
		return false
	end

	ensureRemotes()
	return true
end

function ClientWatchdog.Start()
	if started or not enabled then
		return
	end

	started = true
	ensureRemotes()

	connections.heartbeat = heartbeatRemote.OnServerEvent:Connect(handleHeartbeat)
	connections.report = reportRemote.OnServerEvent:Connect(handleReport)

	for _, player in ipairs(Players:GetPlayers()) do
		ClientWatchdog.OnPlayerAdded(player)
	end

	debugLog("started")
end

function ClientWatchdog.Stop()
	started = false

	for key, connection in pairs(connections) do
		connection:Disconnect()
		connections[key] = nil
	end

	for player in pairs(states) do
		states[player] = nil
	end
end

function ClientWatchdog.OnPlayerAdded(player)
	if not enabled then
		return
	end

	local state = getOrCreateState(player)
	state.playerName = player.Name
	state.graceUntil = math.max(
		state.graceUntil or 0,
		now() + (settings.StartupGraceDuration or settings.HeartbeatTimeout or 6)
	)
	debugLog(("%s session=%s startup grace until %.2f")
		:format(player.Name, tostring(state.sessionId), state.graceUntil))
end

function ClientWatchdog.OnCharacterAdded(player)
	if not enabled then
		return
	end

	local state = getOrCreateState(player)
	local currentTime = now()
	state.lastCharacterAddedAt = currentTime
	state.graceUntil = math.max(
		state.graceUntil or 0,
		currentTime + (settings.RespawnGraceDuration or 4)
	)

	debugLog(("%s session=%s respawn grace until %.2f")
		:format(player.Name, tostring(state.sessionId), state.graceUntil))
end

function ClientWatchdog.OnPlayerRemoving(player)
	states[player] = nil
end

function ClientWatchdog.Update(dt)
	if not started or not enabled or settings.KickOnMissingHeartbeat ~= true then
		return
	end

	accumulator += dt
	local interval = settings.CheckInterval or 0.5
	if interval > 0 and accumulator < interval then
		return
	end
	accumulator = 0

	local currentTime = now()
	local timeout = settings.HeartbeatTimeout or 6

	for _, player in ipairs(Players:GetPlayers()) do
		local state = getOrCreateState(player)
		if state.kicked then
			continue
		end

		if currentTime < (state.graceUntil or 0) then
			continue
		end

		local elapsed = currentTime - (state.lastHeartbeatTime or state.joinedAt or currentTime)
		if elapsed > timeout then
			warn(("[ClientWatchdog] Timeout %s session=%s elapsed=%.2fs timeout=%.2fs graceUntil=%.2f heartbeats=%s reports=%s")
				:format(
					player.Name,
					tostring(state.sessionId),
					elapsed,
					timeout,
					state.graceUntil or 0,
					tostring(state.heartbeatCount or 0),
					tostring(state.reportCount or 0)
				))

			kickPlayer(
				player,
				state,
				("missing heartbeat for %.2fs, heartbeats=%s")
					:format(elapsed, tostring(state.heartbeatCount or 0)),
				settings.MissingHeartbeatKickMessage or settings.KickMessage
			)
		end
	end
end

function ClientWatchdog.GetState(player)
	return states[player]
end

return ClientWatchdog

-- Core anti-cheat runtime.
-- Loads detectors, wires player lifecycle, and runs one shared Heartbeat loop.

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")

local Root = script.Parent

local Config = require(Root.Config)
local State = require(Root.State)
local Actions = require(Root.Actions)
local Utils = require(Root.Utils)
local ClientWatchdog = require(Root.ClientWatchdog)

local Core = {}

local started = false
local detectors = {}
local playerConnections = {}
local heartbeatConnection = nil
local playerAddedConnection = nil
local playerRemovingConnection = nil
local clientWatchdogEnabled = false
local coreAccumulator = 0
local stateCleanupAccumulator = 0

local context = {
	Config = Config,
	State = State,
	Actions = Actions,
	Utils = Utils,
	Services = {
		Players = Players,
		RunService = RunService,
		Workspace = Workspace,
	},
}

local function callDetector(record, methodName, ...)
	local method = record.module[methodName]
	if type(method) ~= "function" then
		return nil
	end

	local ok, result = pcall(method, ...)
	if not ok then
		warn(("[AntiCheat] Detector %s.%s failed: %s")
			:format(record.name, methodName, tostring(result)))
		return nil
	end

	return result
end

local function loadDetectors()
	for index = #detectors, 1, -1 do
		detectors[index] = nil
	end

	local detectorsFolder = Root:WaitForChild("Detectors")

	for _, moduleScript in ipairs(detectorsFolder:GetChildren()) do
		if not moduleScript:IsA("ModuleScript") then
			continue
		end

		local ok, detectorOrError = pcall(require, moduleScript)
		if not ok then
			warn(("[AntiCheat] Failed to load detector %s: %s")
				:format(moduleScript.Name, tostring(detectorOrError)))
			continue
		end

		if type(detectorOrError) ~= "table" then
			warn(("[AntiCheat] Detector %s did not return a module table")
				:format(moduleScript.Name))
			continue
		end

		local detector = detectorOrError
		local name = detector.Name or moduleScript.Name
		local detectorConfig = Config.Detectors and Config.Detectors[name]

		if detectorConfig and detectorConfig.Enabled == false then
			warn(("[AntiCheat] Detector %s is disabled by config"):format(name))
			continue
		end

		local interval = Config.Core.HeartbeatInterval
		if detectorConfig and detectorConfig.CheckInterval then
			interval = detectorConfig.CheckInterval
		end

		local record = {
			name = name,
			module = detector,
			interval = interval or Config.Core.HeartbeatInterval or 0,
			elapsed = 0,
		}

		if type(detector.Init) == "function" then
			local initOk, initResult = pcall(detector.Init, context)
			if not initOk then
				warn(("[AntiCheat] Detector %s.Init failed: %s")
					:format(name, tostring(initResult)))
				continue
			end

			if initResult == false then
				warn(("[AntiCheat] Detector %s chose not to start"):format(name))
				continue
			end
		end

		table.insert(detectors, record)
	end
end

local function handleCharacterAdded(player, character)
	State.ResetCharacterState(player)

	if clientWatchdogEnabled then
		ClientWatchdog.OnCharacterAdded(player)
	end

	for _, record in ipairs(detectors) do
		callDetector(record, "OnCharacterAdded", player, character)
	end
end

local function setupPlayer(player)
	State.CreatePlayerState(player)

	if clientWatchdogEnabled then
		ClientWatchdog.OnPlayerAdded(player)
	end

	for _, record in ipairs(detectors) do
		callDetector(record, "OnPlayerAdded", player)
	end

	if playerConnections[player] then
		playerConnections[player]:Disconnect()
	end

	playerConnections[player] = player.CharacterAdded:Connect(function(character)
		handleCharacterAdded(player, character)
	end)

	if player.Character then
		task.defer(function()
			if State.GetPlayerState(player) then
				handleCharacterAdded(player, player.Character)
			end
		end)
	end
end

local function cleanupPlayer(player)
	if playerConnections[player] then
		playerConnections[player]:Disconnect()
		playerConnections[player] = nil
	end

	for _, record in ipairs(detectors) do
		callDetector(record, "OnPlayerRemoving", player)
	end

	if clientWatchdogEnabled then
		ClientWatchdog.OnPlayerRemoving(player)
	end

	State.RemovePlayerState(player)
end

local function updateDetectorForPlayers(record, dt)
	callDetector(record, "UpdateAll", dt)

	if type(record.module.Update) ~= "function" then
		return
	end

	for _, player in ipairs(Players:GetPlayers()) do
		if State.GetPlayerState(player) then
			callDetector(record, "Update", player, dt)
		end
	end
end

local function updateMovementSnapshots()
	for _, player in ipairs(Players:GetPlayers()) do
		if State.GetPlayerState(player) then
			local valid, _, _, root = Utils.IsValidForMovementChecks(player)
			if valid then
				State.SetMovementSnapshot(player, root.Position, root.CFrame)
			end
		end
	end
end

local function onHeartbeat(dt)
	local coreInterval = Config.Core.HeartbeatInterval or 0

	coreAccumulator += dt
	if coreInterval > 0 and coreAccumulator < coreInterval then
		return
	end

	local stepDt = coreAccumulator
	coreAccumulator = 0
	stateCleanupAccumulator += stepDt

	local cleanupInterval = Config.Core.StateCleanupInterval or 1
	if cleanupInterval <= 0 or stateCleanupAccumulator >= cleanupInterval then
		stateCleanupAccumulator = 0
		State.CleanupAllExpiredExemptions()
	end

	for _, record in ipairs(detectors) do
		record.elapsed += stepDt

		if record.interval <= 0 or record.elapsed >= record.interval then
			local detectorDt = record.elapsed
			record.elapsed = 0
			updateDetectorForPlayers(record, detectorDt)
		end
	end

	if clientWatchdogEnabled then
		ClientWatchdog.Update(stepDt)
	end

	updateMovementSnapshots()
end

function Core.Start()
	if started then
		return
	end

	started = true
	loadDetectors()
	clientWatchdogEnabled = ClientWatchdog.Init(context) ~= false
	if clientWatchdogEnabled then
		ClientWatchdog.Start()
	end

	playerAddedConnection = Players.PlayerAdded:Connect(setupPlayer)
	playerRemovingConnection = Players.PlayerRemoving:Connect(cleanupPlayer)

	for _, player in ipairs(Players:GetPlayers()) do
		setupPlayer(player)
	end

	heartbeatConnection = RunService.Heartbeat:Connect(onHeartbeat)
end

function Core.Stop()
	if not started then
		return
	end

	started = false

	if heartbeatConnection then
		heartbeatConnection:Disconnect()
		heartbeatConnection = nil
	end

	if playerAddedConnection then
		playerAddedConnection:Disconnect()
		playerAddedConnection = nil
	end

	if playerRemovingConnection then
		playerRemovingConnection:Disconnect()
		playerRemovingConnection = nil
	end

	for player, connection in pairs(playerConnections) do
		connection:Disconnect()
		playerConnections[player] = nil
	end

	if clientWatchdogEnabled then
		ClientWatchdog.Stop()
		clientWatchdogEnabled = false
	end

	for _, player in ipairs(Players:GetPlayers()) do
		cleanupPlayer(player)
	end

	for index = #detectors, 1, -1 do
		detectors[index] = nil
	end
end

return Core

local function resetSnapshot(detectorState, root)
	local now = os.clock()

	detectorState.lastPos = root and root.Position or nil
	detectorState.lastCFrame = root and root.CFrame or nil
	detectorState.lastSeenAt = now
end

local function zeroCharacterVelocity(character)
	if not character then
		return
	end

	for _, descendant in ipairs(character:GetDescendants()) do
		if descendant:IsA("BasePart") then
			descendant.AssemblyLinearVelocity = Vector3.zero
			descendant.AssemblyAngularVelocity = Vector3.zero
		end
	end
end

local function ignoreFor(detectorState, duration)
	detectorState.ignoreUntil = math.max(detectorState.ignoreUntil or 0, os.clock() + duration)
	resetSamples(detectorState)
end

local function hasExternalMovementExemption(player, detectorState, now)
	if (detectorState.correctionUntil or 0) > now then
		return false
	end

	return Actions.HasMovementExemption(player)
end

local function getHorizontalVector(vector)
	return Vector3.new(vector.X, 0, vector.Z)
end

local function getGroundSupport(character, root)
	return Utils.HasGroundSupport(character, root, {
		Distance = settings.GroundProbeDistance,
		Radius = settings.GroundProbeRadius,
		MinNormalY = settings.GroundMinNormalY,
	})
end

local function hasWallContact(character, root)
	local clear = Utils.HasHorizontalWallClearance(character, root, {
		Distance = settings.WallContactDistance,
		Heights = settings.WallContactHeights,
		NormalYMax = settings.WallContactNormalYMax,
	})

	return clear == false
end

local function isClimbingState(humanoid)
	return humanoid:GetState() == Enum.HumanoidStateType.Climbing
end

local function isAllowedClimbMovement(character, humanoid, root, motion)
	if settings.AllowClimbMovement ~= true then
		return false
	end

	if motion.verticalSpeed > (settings.ClimbMaxUpwardVelocity or math.huge) then
		return false
	end

	if motion.actualHorizontalSpeed > (settings.ClimbMaxHorizontalSpeed or math.huge) then
		return false
	end

	if isClimbingState(humanoid) then
		return true
	end

	if not hasWallContact(character, root) then
		return false
	end

	return math.abs(motion.verticalSpeed) >= (settings.ClimbMinVerticalSpeed or 0)
end

local function updateJumpGrace(detectorState, humanoid, supported, verticalSpeed, now)
	if supported then
		detectorState.jumpGraceUntil = 0
		return
	end

	local humanoidState = humanoid:GetState()
	if humanoidState == Enum.HumanoidStateType.Jumping
		or verticalSpeed >= settings.JumpUpwardVelocityMin
	then
		local graceUntil = math.max(
			detectorState.jumpGraceUntil or 0,
			now + settings.NormalJumpGrace
		)
		detectorState.jumpGraceUntil = graceUntil
		detectorState.lastJumpGraceUntil = math.max(detectorState.lastJumpGraceUntil or 0, graceUntil)
	end
end

local function getMotionStats(root, lastPos, deltaTime)
	local currentPos = root.Position
	local delta = currentPos - lastPos
	local horizontalDelta = getHorizontalVector(delta)
	local velocity = root.AssemblyLinearVelocity
	local horizontalVelocity = getHorizontalVector(velocity)
	local horizontalDistance = horizontalDelta.Magnitude
	local actualHorizontalSpeed = horizontalDistance / deltaTime
	local actualDistance = delta.Magnitude
	local expectedHorizontalDistance = horizontalVelocity.Magnitude * deltaTime
	local expectedVerticalDistance = velocity.Y * deltaTime
	local horizontalMismatch = math.max(0, horizontalDistance - expectedHorizontalDistance)
	local verticalMismatch = math.abs(delta.Y - expectedVerticalDistance)
	local mismatchRatio = 0

	if horizontalDistance > 0 then
		mismatchRatio = horizontalMismatch / horizontalDistance
	end

	return {
		currentPos = currentPos,
		delta = delta,
		horizontalDistance = horizontalDistance,
		actualDistance = actualDistance,
		actualHorizontalSpeed = actualHorizontalSpeed,
		expectedHorizontalDistance = expectedHorizontalDistance,
		expectedVerticalDistance = expectedVerticalDistance,
		velocity = velocity,
		horizontalVelocity = horizontalVelocity,
		horizontalVelocitySpeed = horizontalVelocity.Magnitude,
		verticalSpeed = velocity.Y,
		horizontalMismatch = horizontalMismatch,
		verticalMismatch = verticalMismatch,
		mismatchRatio = mismatchRatio,
	}
end

local function isWallHitConfirmed(hit, probeStart, direction, horizontalDistance, toPos)
	if not hit or not hit.Instance then
		return false, 0, 0
	end

	if not Utils.IsWorldObstacle(hit.Instance) then
		return false, 0, 0
	end

	if math.abs(hit.Normal.Y) > settings.WallNormalYMax then
		return false, 0, 0
	end

	local horizontalNormal = Vector3.new(hit.Normal.X, 0, hit.Normal.Z)
	if horizontalNormal.Magnitude <= settings.WallNormalEpsilon then
		return false, 0, 0
	end

	local normal = horizontalNormal.Unit
	if direction:Dot(-normal) < settings.WallApproachDotMin then
		return false, 0, 0
	end

	local hitOffset = Vector3.new(hit.Position.X - probeStart.X, 0, hit.Position.Z - probeStart.Z)
	local horizontalHitDistance = hitOffset:Dot(direction)
	if horizontalHitDistance <= 0 then
		return false, 0, 0
	end

	local overrunDistance = horizontalDistance - horizontalHitDistance
	if (horizontalDistance - horizontalHitDistance) < settings.WallCrossOverrunDistance then
		return false, overrunDistance, 0
	end

	local finalOffset = Vector3.new(toPos.X - hit.Position.X, 0, toPos.Z - hit.Position.Z)
	local finalPlaneDistance = finalOffset:Dot(normal)
	if finalPlaneDistance > -settings.WallCrossPlaneTolerance then
		return false, overrunDistance, finalPlaneDistance
	end

	return true, overrunDistance, finalPlaneDistance
end

local function findWallCross(character, fromPos, toPos)
	local horizontal = Vector3.new(toPos.X - fromPos.X, 0, toPos.Z - fromPos.Z)
	local horizontalDistance = horizontal.Magnitude

	if horizontalDistance < settings.MinPositionDelta then
		return false, nil
	end

	local direction = horizontal.Unit
	local side = Vector3.new(-direction.Z, 0, direction.X)
	local params = Utils.MakeWorldRaycastParams(character)
	local hitCount = 0
	local bestHit = nil
	local centerHit = nil
	local bestOverrun = 0
	local centerOverrun = 0

	for _, heightOffset in ipairs(settings.WallProbeHeights) do
		local fromBase = Vector3.new(fromPos.X, fromPos.Y + heightOffset, fromPos.Z)
		local toBase = Vector3.new(toPos.X, toPos.Y + heightOffset, toPos.Z)

		for _, sideOffset in ipairs(settings.WallProbeSideOffsets) do
			local probeStart = fromBase + side * sideOffset
			local probeEnd = toBase + side * sideOffset
			local hit = Workspace:Raycast(probeStart, probeEnd - probeStart, params)

			local confirmed, overrun = isWallHitConfirmed(hit, probeStart, direction, horizontalDistance, toPos)
			if confirmed then
				hitCount += 1
				bestHit = bestHit or hit
				bestOverrun = math.max(bestOverrun, overrun or 0)

				if math.abs(sideOffset) <= settings.CenterProbeSideOffsetTolerance then
					centerHit = centerHit or hit
					centerOverrun = math.max(centerOverrun, overrun or 0)
				end
			end
		end
	end

	if centerHit or hitCount >= settings.RequiredWallProbeHits then
		return true, centerHit or bestHit, {
			hitCount = hitCount,
			overrun = centerHit and centerOverrun or bestOverrun,
			hadCenterHit = centerHit ~= nil,
		}
	end

	return false, nil
end

local function isNonPhysicalSample(detectorState, supported, motion, now)
	if motion.horizontalDistance < settings.MinPositionDelta then
		return false, "tiny movement"
	end

	local horizontalMismatch = motion.horizontalMismatch >= settings.VelocityMismatchDistance
	local mismatchRatio = motion.mismatchRatio >= settings.VelocityMismatchRatio
	local lowVelocityDriven = motion.horizontalVelocitySpeed <= settings.LowVelocityForCFrame
	local suspiciousHorizontal = motion.actualHorizontalSpeed >= settings.MinSuspiciousHorizontalSpeed
		and horizontalMismatch
		and (mismatchRatio or lowVelocityDriven)

	if supported then
		local tweenLikeGround = motion.actualHorizontalSpeed >= (settings.GroundedTweenMinHorizontalSpeed or 23)
			and motion.horizontalVelocitySpeed <= (settings.GroundedTweenMaxVelocity or settings.LowVelocityForCFrame)
			and motion.horizontalMismatch >= (settings.GroundedTweenMismatchDistance or settings.VelocityMismatchDistance)
			and motion.mismatchRatio >= (settings.GroundedTweenMismatchRatio or settings.VelocityMismatchRatio)

		if tweenLikeGround then
			return true, ("ground tween-like cframe delta %.2f/s, velocity %.2f/s, mismatch %.2f ratio %.2f")
				:format(
					motion.actualHorizontalSpeed,
					motion.horizontalVelocitySpeed,
					motion.horizontalMismatch,
					motion.mismatchRatio
				), "groundedTween"
		end

		local groundedMismatchDistance = settings.GroundedVelocityMismatchDistance
			or settings.VelocityMismatchDistance
		local groundedMismatchRatio = settings.GroundedVelocityMismatchRatio
			or settings.VelocityMismatchRatio
		local groundedLowVelocity = settings.GroundedLowVelocityForCFrame
			or settings.LowVelocityForCFrame
		local groundedMismatch = motion.horizontalMismatch >= groundedMismatchDistance
		local groundedRatio = motion.mismatchRatio >= groundedMismatchRatio
		local groundedLowVelocityDriven = motion.horizontalVelocitySpeed <= groundedLowVelocity
		local suspiciousGround = motion.actualHorizontalSpeed > settings.MaxGroundedHorizontalSpeed
			and groundedMismatch
			and (groundedRatio or groundedLowVelocityDriven)

		if suspiciousGround then
			return true, ("ground cframe delta %.2f/s, velocity %.2f/s, mismatch %.2f")
				:format(motion.actualHorizontalSpeed, motion.horizontalVelocitySpeed, motion.horizontalMismatch),
				"groundedVelocityMismatch"
		end

		return false, "grounded physical movement"
	end

	if now < (detectorState.jumpGraceUntil or 0) then
		return false, "normal jump grace"
	end

	if motion.actualHorizontalSpeed > settings.MaxAirHorizontalSpeed and motion.verticalSpeed > settings.FreefallDownwardSpeed then
		return true, ("air horizontal %.2f/s without normal fall, vertical %.2f")
			:format(motion.actualHorizontalSpeed, motion.verticalSpeed),
			"airVelocityMismatch"
	end

	if suspiciousHorizontal then
		return true, ("air cframe delta %.2f/s, velocity %.2f/s, mismatch %.2f")
			:format(motion.actualHorizontalSpeed, motion.horizontalVelocitySpeed, motion.horizontalMismatch),
			"airVelocityMismatch"
	end

	local verticalCFrame = math.abs(motion.delta.Y) >= settings.MinSuspiciousVerticalDelta
		and motion.verticalMismatch >= settings.VerticalMismatchDistance
		and math.abs(motion.verticalSpeed) <= settings.LowVelocityForCFrame

	if verticalCFrame then
		return true, ("vertical cframe delta %.2f, velocity %.2f, mismatch %.2f")
			:format(motion.delta.Y, motion.verticalSpeed, motion.verticalMismatch),
			"verticalCFrame"
	end

	return false, "air movement matches physics"
end

local function recordSuspiciousSample(detectorState, now, distance, supported, reason, motion, kind)
	if detectorState.lastSuspiciousSupported ~= nil
		and detectorState.lastSuspiciousSupported ~= supported
	then
		resetSamples(detectorState)
	end

	detectorState.lastSuspiciousSupported = supported

	table.insert(detectorState.suspiciousSamples, {
		time = now,
		distance = distance,
		supported = supported,
		reason = reason,
		kind = kind,
		speed = motion and motion.actualHorizontalSpeed or 0,
		velocity = motion and motion.horizontalVelocitySpeed or 0,
		mismatch = motion and motion.horizontalMismatch or 0,
		mismatchRatio = motion and motion.mismatchRatio or 0,
	})
	table.insert(detectorState.pathSamples, {
		time = now,
		distance = distance,
		supported = supported,
	})

	trimTimedSamples(detectorState.suspiciousSamples, now, settings.SuspiciousWindow)
	trimTimedSamples(detectorState.pathSamples, now, settings.PathAccumulationWindow)
end

local function isInJumpGrace(detectorState, now)
	return now < (detectorState.jumpGraceUntil or 0)
		or now < ((detectorState.lastJumpGraceUntil or 0) + (settings.AirbornePostJumpKillBlockDuration or 0))
end

local function isAirborneKillGrade(detectorState, now, motion, suspiciousDistance, pathDistance, sampleCount)
	if settings.AirborneKillBlockDuringJumpGrace ~= false and isInJumpGrace(detectorState, now) then
		return false
	end

	return sampleCount >= (settings.AirborneKillRequiredSamples or settings.RequiredSuspiciousSamples)
		and getSampleTimeSpan(detectorState.suspiciousSamples) >= (settings.AirborneKillMinSuspiciousDuration or 0)
		and motion.actualHorizontalSpeed >= (settings.AirborneKillMinHorizontalSpeed or settings.MaxAirHorizontalSpeed)
		and motion.horizontalMismatch >= (settings.AirborneKillMinMismatchDistance or settings.VelocityMismatchDistance)
		and suspiciousDistance >= (settings.AirborneKillMinSuspiciousDistance or settings.AccumulatedSuspiciousDistance)
		and pathDistance >= (settings.AirborneKillMinPathDistance or settings.MaxAccumulatedPathDistance)
end

local function isConfirmedNonPhysical(detectorState, now, supported, motion)
	trimTimedSamples(detectorState.suspiciousSamples, now, settings.SuspiciousWindow)
	trimTimedSamples(detectorState.pathSamples, now, settings.PathAccumulationWindow)

	local suspiciousDistance = sumSampleDistance(detectorState.suspiciousSamples)
	local pathDistance = sumSampleDistance(detectorState.pathSamples)
	local sampleCount = #detectorState.suspiciousSamples
	local confirmed = false
	local confirmKind = nil
	local killGrade = false

	if supported then
		local tweenSampleCount = countSamplesByKind(detectorState.suspiciousSamples, "groundedTween")
		local tweenRequiredSamples = settings.GroundedTweenRequiredSamples or settings.RequiredSuspiciousSamples
		local tweenDistance = settings.GroundedTweenAccumulatedDistance or settings.AccumulatedSuspiciousDistance
		local tweenPath = settings.GroundedTweenPathDistance or settings.MaxAccumulatedPathDistance
		local requiredSamples = settings.GroundedRequiredSuspiciousSamples
			or settings.RequiredSuspiciousSamples
		local accumulatedDistance = settings.GroundedAccumulatedSuspiciousDistance
			or settings.AccumulatedSuspiciousDistance
		local accumulatedPath = settings.GroundedMaxAccumulatedPathDistance
			or settings.MaxAccumulatedPathDistance

		if tweenSampleCount >= tweenRequiredSamples
			and suspiciousDistance >= tweenDistance
			and pathDistance >= tweenPath
		then
			confirmed = true
			confirmKind = "groundedTweenMovement"
		end

		if sampleCount >= requiredSamples or suspiciousDistance >= accumulatedDistance then
			confirmed = true
			confirmKind = confirmKind or "groundedVelocityMismatch"
		end

		if pathDistance >= accumulatedPath then
			confirmed = true
			confirmKind = confirmKind or "groundedPathAccumulation"
		end

		if confirmed and confirmKind == "groundedTweenMovement" then
			local killRequiredSamples = settings.GroundedTweenKillRequiredSamples or tweenRequiredSamples
			killGrade = settings.KillOnGroundedTweenMovement == true
				and tweenSampleCount >= killRequiredSamples
				and motion.actualHorizontalSpeed >= settings.GroundedTweenKillMinHorizontalSpeed
				and motion.horizontalMismatch >= settings.GroundedTweenKillMinMismatchDistance
				and suspiciousDistance >= settings.GroundedTweenKillMinSuspiciousDistance
				and pathDistance >= settings.GroundedTweenKillMinPathDistance
		elseif confirmed and settings.KillOnGroundedVelocityMismatch == true then
			local killRequiredSamples = settings.GroundedKillRequiredSamples or requiredSamples
			killGrade = sampleCount >= killRequiredSamples
				and motion.actualHorizontalSpeed >= settings.GroundedKillMinHorizontalSpeed
				and motion.horizontalMismatch >= settings.GroundedKillMinMismatchDistance
				and suspiciousDistance >= settings.GroundedKillMinSuspiciousDistance
				and pathDistance >= settings.GroundedKillMinPathDistance
		end
	else
		if sampleCount >= settings.RequiredSuspiciousSamples then
			confirmed = true
			confirmKind = "velocityMismatch"
		end

		if suspiciousDistance >= settings.AccumulatedSuspiciousDistance then
			confirmed = true
			confirmKind = "velocityMismatch"
		end

		if pathDistance >= settings.MaxAccumulatedPathDistance then
			confirmed = true
			confirmKind = "pathAccumulation"
		end

		killGrade = confirmed
			and isAirborneKillGrade(detectorState, now, motion, suspiciousDistance, pathDistance, sampleCount)
	end

	return confirmed, suspiciousDistance, pathDistance, confirmKind, killGrade, sampleCount
end

local function isWallCrossKillGrade(motion, wallInfo)
	wallInfo = wallInfo or {}

	if motion.horizontalDistance < (settings.WallCrossKillMinDelta or settings.MinPositionDelta) then
		return false
	end

	if (wallInfo.overrun or 0) < (settings.WallCrossKillMinOverrun or settings.WallCrossOverrunDistance) then
		return false
	end

	if (wallInfo.hitCount or 0) < (settings.WallCrossKillMinProbeHits or settings.RequiredWallProbeHits) then
		return false
	end

	return true
end

local function recordAbuse(player, detectorState, now, reason)
	table.insert(detectorState.abuseEvents, {
		time = now,
		distance = 0,
	})
	trimTimedSamples(detectorState.abuseEvents, now, settings.RepeatedAbuseWindow)

	if #detectorState.abuseEvents < settings.RepeatedAbuseThreshold then
		return
	end

	if (now - (detectorState.lastRepeatedFlagAt or 0)) < settings.RepeatedFlagCooldown then
		return
	end

	detectorState.lastRepeatedFlagAt = now
	Actions.FlagPlayer(
		player,
		settings.RepeatedFlagPoints,
		("AntiCFrameMovement: repeated non-physical movement (%s)"):format(reason)
	)
end

local function shouldKillConfirmed(confirmKind)
	if settings.KillOnConfirmedExploit ~= true then
		return false
	end

	if settings.KillRequiresConfirmedState ~= true then
		return true
	end

	if confirmKind == "wallCross" then
		return settings.KillOnWallCross == true
	end

	if confirmKind == "velocityMismatch" then
		return settings.KillOnConfirmedVelocityMismatch == true
	end

	if confirmKind == "pathAccumulation" then
		return settings.KillOnAccumulatedPath == true
	end

	if confirmKind == "groundedVelocityMismatch"
		or confirmKind == "groundedPathAccumulation"
	then
		return settings.KillOnGroundedVelocityMismatch == true
	end

	if confirmKind == "groundedTweenMovement" then
		return settings.KillOnGroundedTweenMovement == true
	end

	return false
end

local function logDecision(player, detectorState, classification, reason, supported, motion, suspiciousDistance, pathDistance, sampleCount)
	if classification == "weak suspicious" and settings.DebugLogWeakSamples ~= true then
		return
	end

	local now = os.clock()
	local cooldown = settings.ConfirmedLogCooldown or 0.5
	local key = "lastConfirmedLogAt"

	if classification == "weak suspicious" then
		cooldown = settings.WeakLogCooldown or 2
		key = "lastWeakLogAt"
	elseif classification == "instant kill" then
		key = "lastKillLogAt"
	end

	if (now - (detectorState[key] or 0)) < cooldown then
		return
	end

	detectorState[key] = now

	local movementState = supported and "grounded" or "airborne"
	local delta = motion and motion.actualDistance or 0
	local expected = motion and motion.expectedHorizontalDistance or 0
	local mismatch = motion and motion.horizontalMismatch or 0
	local mismatchRatio = motion and motion.mismatchRatio or 0

	warn(("[AntiCheat] AntiCFrameMovement %s for %s: %s; state=%s; delta=%.2f; expectedVelocityDelta=%.2f; mismatch=%.2f (ratio %.2f); suspicious=%.2f; path=%.2f; samples=%s")
		:format(
			classification,
			player.Name,
			reason,
			movementState,
			delta,
			expected,
			mismatch,
			mismatchRatio,
			suspiciousDistance or 0,
			pathDistance or 0,
			tostring(sampleCount or 0)
		))
end

local function killPlayer(player, reason)
	local character, humanoid = Utils.GetCharacterParts(player)
	if not character or not humanoid then
		return false
	end

	if settings.ZeroVelocityBeforeKill ~= false then
		zeroCharacterVelocity(character)
	end

	humanoid.Health = 0
	pcall(function()
		humanoid:ChangeState(Enum.HumanoidStateType.Dead)
	end)

	if settings.ZeroVelocityBeforeKill ~= false then
		zeroCharacterVelocity(character)
	end

	warn(("[AntiCheat] %s killed by AntiCFrameMovement: %s")
		:format(player.Name, reason))

	return true
end

local function correctPlayer(player, detectorState, root, now, points, reason, confirmKind, killGrade)
	Actions.BlockSafePositionSave(player, settings.SafeSaveBlockDuration, "AntiCFrameMovement: " .. reason)
	recordAbuse(player, detectorState, now, reason)

	if (now - (detectorState.lastFlagAt or 0)) >= settings.FlagCooldown then
		detectorState.lastFlagAt = now
		Actions.FlagPlayer(player, points, ("AntiCFrameMovement: %s"):format(reason))
	end

	if killGrade and shouldKillConfirmed(confirmKind) and killPlayer(player, reason) then
		resetSamples(detectorState)
		resetSnapshot(detectorState, root)
		return
	end

	detectorState.correctionUntil = math.max(
		detectorState.correctionUntil or 0,
		now + settings.CorrectionExemptionDuration
	)

	Actions.ForceMovementCorrection(player, {
		Reason = "AntiCFrameMovement: " .. reason,
		Rollback = settings.RollbackOnDetect,
		ZeroVelocity = settings.ZeroVelocityOnDetect,
		ExemptionDuration = settings.CorrectionExemptionDuration,
		ServerOwnershipDuration = settings.ServerOwnershipDuration,
		DownwardVelocity = settings.DownwardCorrectionVelocity,
		SafeSaveBlockDuration = settings.SafeSaveBlockDuration,
		FallbackCFrame = detectorState.lastClearCFrame,
	})

	resetSamples(detectorState)
	resetSnapshot(detectorState, root)
end

local function updatePlayer(player)
	local detectorState = getDetectorState(player)
	local valid, character, humanoid, root = Utils.IsValidForMovementChecks(player)

	if not valid then
		detectorState.lastPos = nil
		detectorState.lastCFrame = nil
		detectorState.lastSeenAt = 0
		ignoreFor(detectorState, settings.MissingCharacterGraceDuration)
		return
	end

	local now = os.clock()

	if (detectorState.ignoreUntil or 0) > now or hasExternalMovementExemption(player, detectorState, now) then
		resetSnapshot(detectorState, root)
		return
	end

	if (detectorState.correctionUntil or 0) > now then
		Actions.BlockSafePositionSave(player, settings.SafeSaveBlockDuration, "AntiCFrameMovement correction")
		return
	end

	local lastPos = detectorState.lastPos
	local lastSeenAt = detectorState.lastSeenAt
	if not lastPos or not lastSeenAt or lastSeenAt <= 0 then
		resetSnapshot(detectorState, root)
		Actions.SaveSafePosition(player, root.CFrame, settings.SafeSaveInterval)
		return
	end

	local deltaTime = now - lastSeenAt
	if deltaTime < settings.MinSampleDeltaTime then
		return
	end

	if deltaTime > settings.MaxSampleDeltaTime then
		resetSamples(detectorState)
		resetSnapshot(detectorState, root)
		return
	end

	local motion = getMotionStats(root, lastPos, deltaTime)
	local supported = getGroundSupport(character, root)

	if isAllowedClimbMovement(character, humanoid, root, motion) then
		detectorState.climbGraceUntil = math.max(
			detectorState.climbGraceUntil or 0,
			now + (settings.ClimbGraceDuration or 0.5)
		)
		resetSamples(detectorState)
		resetSnapshot(detectorState, root)
		return
	end

	if now < (detectorState.climbGraceUntil or 0) then
		resetSamples(detectorState)
		resetSnapshot(detectorState, root)
		return
	end

	updateJumpGrace(detectorState, humanoid, supported, motion.verticalSpeed, now)

	local wallCrossed, wallHit, wallInfo = findWallCross(character, lastPos, motion.currentPos)
	if wallCrossed then
		local killGrade = isWallCrossKillGrade(motion, wallInfo)
		local reason = ("non-physical wall-cross through %s, delta %.2f studs, overrun %.2f, probes %s")
			:format(
				Utils.SafeGetFullName(wallHit.Instance),
				motion.horizontalDistance,
				(wallInfo and wallInfo.overrun) or 0,
				tostring((wallInfo and wallInfo.hitCount) or 0)
			)
		logDecision(player, detectorState, killGrade and "instant kill" or "confirmed exploit", reason, supported, motion, 0, 0, 0)
		correctPlayer(player, detectorState, root, now, settings.WallCrossFlagPoints, reason, "wallCross", killGrade)
		return
	end

	local suspicious, sampleReason, sampleKind = isNonPhysicalSample(detectorState, supported, motion, now)
	if suspicious then
		recordSuspiciousSample(detectorState, now, motion.actualDistance, supported, sampleReason, motion, sampleKind)
		Actions.BlockSafePositionSave(player, settings.SafeSaveBlockDuration, "AntiCFrameMovement suspicious sample")

		local confirmed, suspiciousDistance, pathDistance, confirmKind, killGrade, sampleCount =
			isConfirmedNonPhysical(detectorState, now, supported, motion)
		if confirmed then
			local reason = ("%s, suspicious %.2f studs, path %.2f studs")
				:format(sampleReason, suspiciousDistance, pathDistance)
			logDecision(
				player,
				detectorState,
				killGrade and "instant kill" or "confirmed exploit",
				reason,
				supported,
				motion,
				suspiciousDistance,
				pathDistance,
				sampleCount
			)
			correctPlayer(player, detectorState, root, now, settings.FlagPoints, reason, confirmKind, killGrade)
			return
		end

		logDecision(
			player,
			detectorState,
			"weak suspicious",
			sampleReason,
			supported,
			motion,
			suspiciousDistance,
			pathDistance,
			sampleCount
		)
	else
		resetSamples(detectorState)
		if supported and Actions.SaveSafePosition(player, root.CFrame, settings.SafeSaveInterval) then
			detectorState.lastClearCFrame = root.CFrame
		end
	end

	resetSnapshot(detectorState, root)
end

function Detector.Init(context)
	Config = context.Config
	State = context.State

-- Stops server-observed fling behavior and stabilizes victims without trusting the client.

local Workspace = game:GetService("Workspace")

Detector.Name = "AntiFling"

local State = nil
local Actions = nil
local Utils = nil
local settings = nil

local collisionGroupName = nil

local function getDetectorState(player)
	return State.GetDetectorState(player, Detector.Name, {
		lastSeenAt = 0,
		lastVelocity = nil,
		ignoreUntil = 0,
		lastFlagAt = 0,
		lastRepeatedFlagAt = 0,
		lastVictimCorrectionAt = 0,
		abuseEvents = {},
		collisionConnection = nil,
	})

	detectorState.ignoreUntil = math.max(detectorState.ignoreUntil or 0, os.clock() + duration)

	detectorState.lastSeenAt = os.clock()

		return false
	end

		return true
	end

	collisionGroupName = settings.CharacterCollisionGroupName or "AntiCheatCharacters"

	pcall(function()
		PhysicsService:RegisterCollisionGroup(collisionGroupName)
	end)

	local ok = pcall(function()
		PhysicsService:CollisionGroupSetCollidable(collisionGroupName, collisionGroupName, false)
	end)

	collisionGroupReady = ok
	return ok

	if not settings.CharacterNoCollisionEnabled or not collisionGroupReady then
		return
	end

	if not part:IsA("BasePart") then
		return
	end

	pcall(function()
		part.CollisionGroup = collisionGroupName
	end)
end

local function disconnectCollisionConnection(detectorState)
	local connection = detectorState.collisionConnection
	if connection then
		connection:Disconnect()
		detectorState.collisionConnection = nil
	end

	if not settings.CharacterNoCollisionEnabled or not setupCollisionGroup() then
		return
	end

	local detectorState = getDetectorState(player)
	disconnectCollisionConnection(detectorState)

	for _, descendant in ipairs(character:GetDescendants()) do
		setCharacterPartCollisionGroup(descendant)

		setCharacterPartCollisionGroup(descendant)

		if instance:IsA(className) then

	return false

	local maxAngularSpeed = root.AssemblyAngularVelocity.Magnitude

		return 0

		return 0

		Actions.SetBackToLastSafePosition(player)
	end

	local currentCharacter = Utils.GetCharacter(player)
	if currentCharacter then
		zeroCharacterVelocity(currentCharacter)

	local correctionGrace = settings.PostCorrectionExemptionDuration or 0
	if correctionGrace > 0 then
		Actions.IgnoreMovement(player, correctionGrace, Config.Exemptions.Tags.Movement)
		ignoreFor(detectorState, correctionGrace)
	end

	recordSourceAbuse(player, detectorState, now, reason)

	if (now - (detectorState.lastFlagAt or 0)) >= settings.FlagCooldown then
		detectorState.lastFlagAt = now
		Actions.FlagPlayer(player, points, reason)
	end

	applyCorrection(player, detectorState, character)

	if (now - (detectorState.lastVictimCorrectionAt or 0)) < settings.VictimCorrectionCooldown then
		return

	detectorState.lastVictimCorrectionAt = now
	applyCorrection(player, detectorState, character)
end

local function getSourceFlingReason(character, humanoid, root, motion, bodyMoverCount, impulseDelta)
	if bodyMoverCount > 0 then
		return settings.ExtremeFlagPoints, ("AntiFling: removed %d dangerous body mover(s)"):format(bodyMoverCount)
	end

	if motion.angularSpeed >= settings.ExtremeAngularVelocity then
		return settings.ExtremeFlagPoints,
			("AntiFling: extreme angular velocity %.2f"):format(motion.angularSpeed)
	end

	if motion.angularSpeed >= settings.MaxAngularVelocity then
		return settings.FlagPoints,
			("AntiFling: abnormal angular velocity %.2f"):format(motion.angularSpeed)
	end

	local sourceLikeLinear = isSourceLikeLinearMotion(character, humanoid, root, motion)
	if sourceLikeLinear and motion.linearSpeed >= settings.ExtremeLinearVelocity then
		return settings.ExtremeFlagPoints,
			("AntiFling: extreme source linear velocity %.2f"):format(motion.linearSpeed)
	end

	if sourceLikeLinear and motion.linearSpeed >= settings.MaxLinearVelocity then
		return settings.FlagPoints,
			("AntiFling: abnormal source linear velocity %.2f"):format(motion.linearSpeed)
	end

	if sourceLikeLinear and impulseDelta >= settings.ViolentImpulseDeltaVelocity then
		return settings.FlagPoints,
			("AntiFling: violent source impulse %.2f"):format(impulseDelta)
	end

	return nil, nil
end

local function isVictimLaunch(motion, impulseDelta)
	local upwardLaunch = motion.verticalSpeed >= settings.VictimVerticalLaunchSpeed
	local horizontalLaunch = motion.horizontalSpeed >= settings.VictimLaunchSpeed
	local violentImpulse = impulseDelta >= settings.ViolentImpulseDeltaVelocity

	return upwardLaunch
		or horizontalLaunch
		or violentImpulse
end

local function updatePlayer(player)
	local detectorState = getDetectorState(player)
	local valid, character, humanoid, root = Utils.IsValidForMovementChecks(player)

	if not valid then
		detectorState.lastVelocity = nil
		detectorState.lastSeenAt = 0
		ignoreFor(detectorState, settings.MissingCharacterGraceDuration)
		return
	end

	local now = os.clock()
	local ignored = shouldIgnore(player, detectorState)
	local bodyMoverCount = removeDangerousBodyMovers(character, true)
	local motion = getCharacterMotion(character, root)
	local impulseDelta = getImpulseDelta(detectorState, motion, now)

	if ignored then
		if bodyMoverCount > 0 then
			zeroCharacterVelocity(character)
		end

		resetSample(detectorState, root)
		return
	end

	local points, sourceReason = getSourceFlingReason(
		character,
		humanoid,
		root,
		motion,
		bodyMoverCount,
		impulseDelta
	)

	if sourceReason then
		flagSourceFling(player, detectorState, character, now, points, sourceReason)
		resetSample(detectorState, root)
		return
	end

	if isVictimLaunch(motion, impulseDelta) then
		correctVictimLaunch(player, detectorState, character, now)
	end

	resetSample(detectorState, root)
end

function Detector.Init(context)
	Config = context.Config
	State = context.State
	Actions = context.Actions
	Utils = context.Utils

	settings = Config.Detectors and Config.Detectors[Detector.Name]
	if not settings or settings.Enabled == false then
		return false
	end

	setupCollisionGroup()
	return true
end

	local detectorState = getDetectorState(player)

	detectorState.lastVelocity = nil
	detectorState.lastSeenAt = 0
	ignoreFor(detectorState, settings.SpawnGraceDuration)
end

function Detector.OnCharacterAdded(player, character)
	local detectorState = getDetectorState(player)

	detectorState.lastVelocity = nil
	detectorState.lastSeenAt = 0
	ignoreFor(detectorState, settings.SpawnGraceDuration)
	applyCharacterCollisionProtection(player, character)
end

function Detector.OnPlayerRemoving(player)
	local detectorState = State.GetDetectorState(player, Detector.Name)
	disconnectCollisionConnection(detectorState)
	State.ClearDetectorState(player, Detector.Name)
end

function Detector.Update(player)
	if not settings or settings.Enabled == false then
		return
	end

	updatePlayer(player)
end

return Detector

	return state == Enum.HumanoidStateType.Jumping
end

	if supported then
		detectorState.jumpGraceUntil = 0
		detectorState.apexGraceUntil = 0
		return
	end

	if isJumping(humanoid) or verticalSpeed >= settings.JumpUpwardVelocityMin then
		detectorState.jumpGraceUntil = math.max(
			detectorState.jumpGraceUntil or 0,
			now + settings.NormalJumpAirGrace
		)
	end

	local lastVerticalSpeed = detectorState.lastVerticalSpeed
	if lastVerticalSpeed
		and lastVerticalSpeed >= settings.ApexUpwardVelocityMin
		and math.abs(verticalSpeed) <= settings.HoverVerticalSpeedAbsMax
	then
		detectorState.apexGraceUntil = math.max(
			detectorState.apexGraceUntil or 0,
			now + settings.JumpApexGraceDuration
		)
	end
end

local function isNormalAirborneGrace(detectorState, verticalSpeed, now)
	if now < (detectorState.jumpGraceUntil or 0) then
		return true
	end

	if now < (detectorState.apexGraceUntil or 0) then
		return true
	end

	return verticalSpeed <= settings.FreefallDownwardSpeed
end

local function getHorizontalSpeed(detectorState, root, now)
	local lastPos = detectorState.lastPos
	local lastSeenAt = detectorState.lastSeenAt

	if not lastPos or not lastSeenAt or lastSeenAt <= 0 then
		return 0
	end

	local deltaTime = now - lastSeenAt
	if deltaTime <= 0 then
		return 0
	end

	local delta = root.Position - lastPos
	return Vector3.new(delta.X, 0, delta.Z).Magnitude / deltaTime
end

		NormalYMax = settings.WallContactNormalYMax,

		return 0
	end

	local minY = samples[1].y
	local maxY = samples[1].y

	for _, sample in ipairs(samples) do
		minY = math.min(minY, sample.y)
		maxY = math.max(maxY, sample.y)

local function recordFlyAbuse(player, detectorState, now, reason)
	detectorState.abuseEvents = detectorState.abuseEvents or {}
	table.insert(detectorState.abuseEvents, now)
	trimAbuseEvents(detectorState.abuseEvents, now)

	local threshold = settings.RepeatedAbuseThreshold or math.huge
	if #detectorState.abuseEvents < threshold then
		return
	end

	local cooldown = settings.RepeatedFlagCooldown or settings.FlagCooldown or 0
	if (now - (detectorState.lastRepeatedFlagAt or 0)) < cooldown then
		return
	end

	detectorState.lastRepeatedFlagAt = now
	Actions.FlagPlayer(

		settings.RepeatedFlagPoints or settings.FlyFlagPoints,

	)

	if not root or not root:IsA("BasePart") then
		return false
	end

	local correctionReason = reason or "AntiFly correction"
	local correctionGrace = settings.PostCorrectionExemptionDuration or 0
	local rolledBack = Actions.ForceMovementCorrection(player, {
		Reason = correctionReason,
		Rollback = settings.RollbackOnDetect,
		ZeroVelocity = settings.ZeroVelocityOnDetect,
		ExemptionDuration = correctionGrace,
		ServerOwnershipDuration = settings.ServerOwnershipDuration,
		DownwardVelocity = settings.CorrectionDownwardVelocity,
		SafeSaveBlockDuration = settings.SafeSaveBlockDuration,
	})

	ignoreFor(detectorState, correctionGrace)

	return rolledBack

	end

	applyFlyCorrection(player, detectorState, root, reason)
end

local function shouldIgnore(player, detectorState)
	if Actions.HasExemption(player, Config.Exemptions.Tags.Knockback) then
		ignoreFor(detectorState, settings.GraceAfterKnockback)
		return true
	end

	if Actions.HasMovementExemption(player) then
		return true
	end

	return (detectorState.ignoreUntil or 0) > os.clock()

	if supported then
		detectorState.airStartedAt = nil
		detectorState.lastGroundedAt = now
		detectorState.climbGraceUntil = 0
		clearAirViolationCandidates(detectorState)
		return 0
	end

	if not detectorState.airStartedAt then
		detectorState.airStartedAt = now
	end

	return now - detectorState.airStartedAt

local function updatePlayer(player)
	local detectorState = getDetectorState(player)
	local valid, character, humanoid, root = Utils.IsValidForMovementChecks(player)

	if not valid then
		detectorState.airStartedAt = nil
		detectorState.lastPos = nil
		detectorState.lastSeenAt = 0
		ignoreFor(detectorState, settings.MissingCharacterGraceDuration)
		return
	end

	local now = os.clock()
	local supported = hasSupportBelow(character, root)
	local inWater = isInWater(character, root)
	local velocity = root.AssemblyLinearVelocity
	local verticalSpeed = velocity.Y
	local swimming = isSwimmingState(humanoid)

	if shouldIgnore(player, detectorState) then
		resetSample(detectorState, root, verticalSpeed)
		if supported then
			Actions.SaveSafePosition(player, root.CFrame)
		end
		return
	end

	if inWater then
		resetSample(detectorState, root, verticalSpeed)
		clearAirViolationCandidates(detectorState)
		return
	end

	if swimming then
		detectorState.swimStartedAt = detectorState.swimStartedAt or now

		if (now - detectorState.swimStartedAt) >= settings.SwimStateConfirmDuration then
			flagFly(
				player,
				detectorState,
				root,
				now,
				settings.SwimFlagPoints,
				("AntiFly: swim state outside water %.2fs, supported=%s")
					:format(now - detectorState.swimStartedAt, tostring(supported))
			)
			resetSample(detectorState, root, verticalSpeed)
			return
		end

		if supported then
			resetSample(detectorState, root, verticalSpeed)
			return
		end
	else
		detectorState.swimStartedAt = nil
	end

	updateJumpGrace(detectorState, humanoid, verticalSpeed, supported, now)
	local airTime = updateAirState(detectorState, supported, now)
	local horizontalSpeed = getHorizontalSpeed(detectorState, root, now)
	local inNormalAirGrace = isNormalAirborneGrace(detectorState, verticalSpeed, now)

	if isAllowedClimbMovement(character, humanoid, root, verticalSpeed, horizontalSpeed) then
		detectorState.climbGraceUntil = math.max(
			detectorState.climbGraceUntil or 0,
			now + (settings.ClimbGraceDuration or 0.5)
		)
		clearAirViolationCandidates(detectorState)
		resetSample(detectorState, root, verticalSpeed)
		return
	end

	if now < (detectorState.climbGraceUntil or 0) then
		clearAirViolationCandidates(detectorState)
		resetSample(detectorState, root, verticalSpeed)
		return
	end

	if supported then
		resetSample(detectorState, root, verticalSpeed)
		clearAirViolationCandidates(detectorState)
		Actions.SaveSafePosition(player, root.CFrame)
		return
	end

	if verticalSpeed > settings.MaxUpwardVelocity then
		flagFly(
			player,
			detectorState,
			root,
			now,
			settings.FlyFlagPoints,
			("AntiFly: impossible upward velocity %.2f"):format(verticalSpeed)
		)
		resetSample(detectorState, root, verticalSpeed)
		return
	end

	if inNormalAirGrace then
		clearAirViolationCandidates(detectorState)
		resetSample(detectorState, root, verticalSpeed)

		and not recentlyGrounded(detectorState, now, settings.InitialJumpGrace)
	then
		return ("airborne jump transition after %.2fs without support")
			:format(now - (detectorState.lastGroundedAt or now))
	end

	if lastVerticalSpeed then

	local detectorState = getDetectorState(player)
	local valid, character, humanoid, root = Utils.IsValidForMovementChecks(player)

	if not valid then
		detectorState.airStartedAt = nil
		detectorState.lastVerticalSpeed = nil
		detectorState.lastHumanoidState = nil
		ignoreFor(detectorState, settings.MissingCharacterGraceDuration)
		return
	end

	local now = os.clock()
	local humanoidState = humanoid:GetState()
	local verticalSpeed = root.AssemblyLinearVelocity.Y
	local supported = hasSupport(character, root)
	local inWater = isInWater(character, root)
	local climbSupported = hasClimbSupport(character, root, humanoidState, verticalSpeed)

	if shouldIgnore(player, detectorState) then
		detectorState.lastHumanoidState = humanoidState
		detectorState.lastVerticalSpeed = verticalSpeed
		return
	end

	if supported or inWater or climbSupported then
		resetAirState(detectorState, now)
		if climbSupported then
			detectorState.climbSupportUntil = math.max(
				detectorState.climbSupportUntil or 0,
				now + (settings.ClimbSupportGrace or 0.35)
			)
		end

		detectorState.lastHumanoidState = humanoidState
		detectorState.lastVerticalSpeed = verticalSpeed

		if supported then
			Actions.SaveSafePosition(player, root.CFrame, settings.SafeSaveInterval)
		end

		return
	end

	if now < (detectorState.climbSupportUntil or 0) then
		detectorState.lastHumanoidState = humanoidState
		detectorState.lastVerticalSpeed = verticalSpeed
		return
	end

	if not detectorState.airStartedAt then
		detectorState.airStartedAt = now
	end

	local readyToCheck = isAirborneLongEnough(detectorState, now)
	if not readyToCheck or recentlyGrounded(detectorState, now, settings.GroundedResetGrace) then
		detectorState.lastHumanoidState = humanoidState
		detectorState.lastVerticalSpeed = verticalSpeed
		return
	end

	if verticalSpeed >= settings.ExtremeUpwardVelocity
		and not recentlyGrounded(detectorState, now, settings.InitialJumpGrace)
	then
		correctPlayer(
			player,
			detectorState,
			now,
			settings.ExtremeFlagPoints,
			("extreme airborne upward velocity %.2f without support"):format(verticalSpeed)
		)
		detectorState.lastHumanoidState = humanoidState
		detectorState.lastVerticalSpeed = verticalSpeed
		return
	end

	local invalidReason = getInvalidJumpReason(detectorState, humanoidState, verticalSpeed, now)
	if invalidReason then
		recordInvalidJump(detectorState, now, invalidReason)
		Actions.BlockSafePositionSave(player, settings.SafeSaveBlockDuration, "AntiInfiniteJump: " .. invalidReason)

		if #detectorState.invalidJumpEvents >= settings.InvalidJumpThreshold then

				player,
				detectorState,
				now,

					:format(#detectorState.invalidJumpEvents, settings.AirJumpWindow, invalidReason)
			)
		end
	end

	detectorState.lastHumanoidState = humanoidState
	detectorState.lastVerticalSpeed = verticalSpeed
end

	Config = context.Config

	if not hit or not hit.Instance then
		return false
	end

	if not Utils.IsWorldObstacle(hit.Instance) then
		return false
	end

	if math.abs(hit.Normal.Y) > settings.WallNormalYMax then
		return false
	end

	local horizontalNormal = Vector3.new(hit.Normal.X, 0, hit.Normal.Z)
	if horizontalNormal.Magnitude <= settings.WallNormalEpsilon then
		return false
	end

	local normal = horizontalNormal.Unit
	local approachDot = direction:Dot(-normal)
	if approachDot < settings.WallApproachDotMin then
		return false
	end

	local hitDistance = hit.Distance or horizontalDistance
	local overrunDistance = horizontalDistance - hitDistance
	if overrunDistance < settings.WallCrossOverrunDistance then
		return false
	end

	local finalOffset = Vector3.new(toPos.X - hit.Position.X, 0, toPos.Z - hit.Position.Z)
	local finalPlaneDistance = finalOffset:Dot(normal)
	if finalPlaneDistance > -settings.WallCrossPlaneTolerance then
		return false
	end

	return true
end

local function findConfirmedWallHit(params, fromPos, toPos, direction, horizontalDistance)
	local horizontal = Vector3.new(toPos.X - fromPos.X, 0, toPos.Z - fromPos.Z)
	local side = Vector3.new(-direction.Z, 0, direction.X)
	local hitCount = 0
	local bestHit = nil
	local centerHit = nil

	for _, heightOffset in ipairs(settings.ProbeHeights) do
		local baseStart = Vector3.new(fromPos.X, fromPos.Y + heightOffset, fromPos.Z)

		for _, sideOffset in ipairs(settings.ProbeSideOffsets) do
			local hit = Workspace:Raycast(baseStart + side * sideOffset, horizontal, params)
			if isWallHitConfirmed(hit, direction, horizontalDistance, toPos) then
				hitCount += 1
				bestHit = bestHit or hit

				if math.abs(sideOffset) <= settings.CenterProbeSideOffsetTolerance then
					centerHit = centerHit or hit
				end
			end
		end
	end

	if centerHit then
		return true, centerHit or bestHit, hitCount
	end

	if settings.RequireCenterProbeForWallCross ~= false then
		return false, nil, hitCount
	end

	if hitCount >= settings.RequiredWallProbeHits then

	end

local function getPointPenetrationInfo(part, worldPoint)
	local localPoint = part.CFrame:PointToObjectSpace(worldPoint)
	local halfSize = part.Size * 0.5

	local xDepth = halfSize.X - math.abs(localPoint.X)
	local yDepth = halfSize.Y - math.abs(localPoint.Y)
	local zDepth = halfSize.Z - math.abs(localPoint.Z)

	if xDepth <= 0 or yDepth <= 0 or zDepth <= 0 then
		return 0, nil
	end

	local localNormal = Vector3.new(localPoint.X >= 0 and 1 or -1, 0, 0)
	local penetration = xDepth

	if yDepth < penetration then
		penetration = yDepth
		localNormal = Vector3.new(0, localPoint.Y >= 0 and 1 or -1, 0)
	end

	if zDepth < penetration then
		penetration = zDepth
		localNormal = Vector3.new(0, 0, localPoint.Z >= 0 and 1 or -1)
	end

	if not humanoid or not root then
		return
	end

	if not Utils.IsAlive(humanoid) then
		return
	end

		state.lastPos = currentPos

			clearInsideWallState(state)

			saveSafePosition(player, root)
		else
			clearInsideWallState(state)
		end

		return

	if not lastPos then
		state.lastPos = currentPos

		if not insideWall then
			clearInsideWallState(state)
			state.lastClearCFrame = root.CFrame
			saveSafePosition(player, root)
		end

		return
	end

	if moveDistance > settings.MaxStepDistance then
		local crossedWall, hit = movementCrossedWall(character, lastPos, currentPos)
		if crossedWall then
			flagPlayer(player, "Moved through wall: " .. Utils.SafeGetFullName(hit.Instance))
			setBackFromHit(player, character, root, hit, lastPos, currentPos)
			return
		end

		if settings.HugeDeltaRouteToCFrame ~= false then
			routeHugeNonWallDelta(player, state, moveDistance)
			clearInsideWallState(state)
			state.lastPos = currentPos
			return
		end

		flagHugeDeltaWithCooldown(player, state, moveDistance)
		setBackToSafe(player, character, root)
		return
	end

	local crossedWall, hit = movementCrossedWall(character, lastPos, currentPos)
	if crossedWall then
		flagPlayer(player, "Moved through wall: " .. Utils.SafeGetFullName(hit.Instance))
		setBackFromHit(player, character, root, hit, lastPos, currentPos)
		return
	end

	local insideWall, insidePart, penetration, insideNormal = findInsideWall(character, root)
	if insideWall then
		local airborne = isAirborneCharacter(humanoid, root)
		local approachDot = getHorizontalApproachDot(lastPos, currentPos, insideNormal)

		if isContactOnlyPenetration(penetration, airborne, approachDot) then
			clearInsideWallState(state)

			setBackToSafe(player, character, root)
		else
			state.lastPos = currentPos
		end

		return
	end

	clearInsideWallState(state)
	state.lastPos = currentPos
	state.lastClearCFrame = root.CFrame
	saveSafePosition(player, root)
end

	local firstValidIndex = 1

		firstValidIndex += 1
	end

	if firstValidIndex <= 1 then
		return
	end

	for index = firstValidIndex, #list do
		list[index - firstValidIndex + 1] = list[index]

			extremeCount += 1

		return
	end

	local sustained = highCount >= settings.RequiredHighSamples
		and highRatio >= settings.RequiredHighRatio
		and averageSpeed > settings.AverageSpeedLimit

		return

		flagged = flagSpeed(

			detectorState,
			"extreme",
			now,
			settings.ExtremeFlagPoints,
			("AntiSpeed: repeated extreme horizontal speed (avg %.2f, %d/%d high samples)")

			settings.FlagCooldown
		) or flagged
	end

		flagged = flagSpeed(

			detectorState,
			"sustained",
			now,

		end

	end

	if deltaTime < settings.MinSampleDeltaTime then
		return false
	end

	if deltaTime > settings.MaxSampleDeltaTime then
		return false
	end

	if distance < settings.MinTeleportDistance then
		return false
	end

	local speed = distance / deltaTime
	return speed > settings.MaxAllowedSpeed, speed
end

local function isShortWallHitConfirmed(hit, probeStart, direction, horizontalDistance, toPos)
	if not hit or not hit.Instance then
		return false
	end

	if not Utils.IsSolidPart(hit.Instance) then
		return false
	end

	if math.abs(hit.Normal.Y) > settings.ShortWallNormalYMax then
		return false
	end

	local horizontalNormal = Vector3.new(hit.Normal.X, 0, hit.Normal.Z)
	if horizontalNormal.Magnitude <= settings.ShortWallNormalEpsilon then
		return false
	end

	local normal = horizontalNormal.Unit
	if direction:Dot(-normal) < settings.ShortWallTeleportApproachDotMin then
		return false
	end

	local hitOffset = Vector3.new(hit.Position.X - probeStart.X, 0, hit.Position.Z - probeStart.Z)
	local horizontalHitDistance = hitOffset:Dot(direction)
	if horizontalHitDistance <= 0 then
		return false
	end

	local overrunDistance = horizontalDistance - horizontalHitDistance
	if overrunDistance < settings.ShortWallTeleportOverrun then
		return false
	end

	local finalOffset = Vector3.new(toPos.X - hit.Position.X, 0, toPos.Z - hit.Position.Z)
	local finalPlaneDistance = finalOffset:Dot(normal)
	if finalPlaneDistance > -settings.ShortWallTeleportPlaneTolerance then
		return false
	end

	return true
end

local function findShortWallTeleport(character, fromPos, toPos, deltaTime)
	local horizontalDistance = getHorizontalDistance(fromPos, toPos)
	local minDistance = settings.ShortWallTeleportMinDistance or settings.ShortWallTeleportDistance or 0
	local maxDistance = settings.ShortWallTeleportMaxDistance or settings.MinTeleportDistance or math.huge

	if horizontalDistance < minDistance or horizontalDistance > maxDistance then
		return false, nil
	end

	if deltaTime < settings.MinSampleDeltaTime or deltaTime > settings.MaxSampleDeltaTime then
		return false, nil
	end

	local horizontalSpeed = horizontalDistance / deltaTime
	if horizontalSpeed < (settings.ShortWallTeleportMinSpeed or 0) then
		return false, nil
	end

	local horizontal = Vector3.new(toPos.X - fromPos.X, 0, toPos.Z - fromPos.Z)
	if horizontal.Magnitude <= 0 then
		return false, nil
	end

	local direction = horizontal.Unit
	local side = Vector3.new(-direction.Z, 0, direction.X)
	local params = Utils.MakeRaycastParams(character)

	for _, heightOffset in ipairs(settings.ShortWallProbeHeights) do
		local fromBase = Vector3.new(fromPos.X, fromPos.Y + heightOffset, fromPos.Z)
		local toBase = Vector3.new(toPos.X, toPos.Y + heightOffset, toPos.Z)

		for _, sideOffset in ipairs(settings.ShortWallProbeSideOffsets) do
			local probeStart = fromBase + side * sideOffset
			local probeEnd = toBase + side * sideOffset
			local probeDelta = probeEnd - probeStart

			local hit = Workspace:Raycast(probeStart, probeDelta, params)
			if isShortWallHitConfirmed(hit, probeStart, direction, horizontalDistance, toPos) then
				return true, hit
			end
		end
	end

	return false, nil
end

	return (now - lastFlagAt) >= cooldown
end

local function flag(player, remoteState, kind, now, points, reason, cooldown)
	if not canFlag(remoteState, kind, now, cooldown) then
		return false
	end

	remoteState.lastFlagAt[kind] = now
	remoteState.totalFlags += 1

	Actions.FlagPlayer(player, points, reason)
	return true
end

local function checkRepeatedAbuse(player, remoteState, remoteName, now, limits)
	trimWindow(remoteState.abuseEvents, now, limits.RepeatedAbuseWindow)

	if #remoteState.abuseEvents < limits.RepeatedAbuseThreshold then
		return false
	end

	return flag(
		player,
		remoteState,
		"repeated",
		now,
		limits.RepeatedAbuseFlagPoints,
		("RemoteSpam repeated abuse on %s (%d violations in %.2fs)")
			:format(remoteName, #remoteState.abuseEvents, limits.RepeatedAbuseWindow),
		limits.RepeatedAbuseFlagCooldown
	)
end

local function cleanupPlayerState(player, now)
	local detectorState = State.GetDetectorState(player, Detector.Name, {
		remotes = {},
		lastCleanup = 0,
	})

	if (now - detectorState.lastCleanup) < settings.CleanupInterval then
		return
	end

	detectorState.lastCleanup = now

	for remoteName, remoteState in pairs(detectorState.remotes) do
		if (now - (remoteState.lastSeen or 0)) > settings.RemoteStateTtl then
			detectorState.remotes[remoteName] = nil
		end
	end
end

function Detector.Record(player, remoteName)
	if not ensureReady() then
		return {
			recorded = false,
			reason = "disabled",
		}
	end

	if not isPlayer(player) then
		return {
			recorded = false,
			reason = "invalid_player",
		}
	end

	local normalizedName = normalizeRemoteName(remoteName)
	local limits = getLimits(normalizedName)
	local now = os.clock()
	local detectorState = getDetectorState(player)
	local remoteState = getRemoteState(detectorState, normalizedName)

	remoteState.totalCalls += 1
	remoteState.lastSeen = now

	local burstCount = appendTimestamp(remoteState.burstCalls, now, limits.BurstWindow)
	local spamCount = appendTimestamp(remoteState.spamCalls, now, limits.SpamWindow)
	local sustainedCount = appendTimestamp(remoteState.sustainedCalls, now, limits.SustainedWindow)
	local flagged = false

	if burstCount > limits.BurstLimit then
		flagged = flag(
			player,
			remoteState,
			"burst",
			now,
			limits.BurstFlagPoints,
			("RemoteSpam burst on %s (%d calls in %.2fs)")
				:format(normalizedName, burstCount, limits.BurstWindow),
			limits.FlagCooldown
		) or flagged
	end

	if spamCount > limits.SpamLimit then
		flagged = flag(
			player,
			remoteState,
			"spam",
			now,
			limits.SpamFlagPoints,
			("RemoteSpam rate on %s (%d calls in %.2fs)")
				:format(normalizedName, spamCount, limits.SpamWindow),
			limits.FlagCooldown
		) or flagged
	end

	if sustainedCount > limits.SustainedLimit then
		flagged = flag(
			player,
			remoteState,
			"sustained",
			now,
			limits.SustainedFlagPoints,
			("RemoteSpam sustained on %s (%d calls in %.2fs)")
				:format(normalizedName, sustainedCount, limits.SustainedWindow),
			limits.FlagCooldown
		) or flagged
	end

	if flagged then
		table.insert(remoteState.abuseEvents, now)
		checkRepeatedAbuse(player, remoteState, normalizedName, now, limits)
	end

	return {
		recorded = true,
		remoteName = normalizedName,
		flagged = flagged,
		counts = {
			burst = burstCount,
			spam = spamCount,
			sustained = sustainedCount,
		},
	}
end

function Detector.Init(context)
	Config = context.Config
	State = context.State
	Actions = context.Actions

	settings = Config.Detectors and Config.Detectors[Detector.Name]
	if not settings or settings.Enabled == false then
		return false
	end

	return true
end

function Detector.OnPlayerAdded(player)
	getDetectorState(player)
end

function Detector.OnPlayerRemoving(player)
	State.ClearDetectorState(player, Detector.Name)
end

function Detector.UpdateAll(dt)
	if not ensureReady() then
		return
	end

	cleanupAccumulator += dt
	if cleanupAccumulator < settings.CleanupInterval then
		return
	end

	cleanupAccumulator = 0
	local now = os.clock()

	for player in pairs(State.Players) do
		cleanupPlayerState(player, now)
	end
end

return Detector

	if type(value) ~= "table" then
		return value

function State.PruneExpiredScoreEntries(player, now)
	local state = State.GetPlayerState(player)
	if not state then
		return 0, 0
	end

	now = now or os.clock()
	state.activeScoreEntries = state.activeScoreEntries or {}

	for index = #state.activeScoreEntries, 1, -1 do
		local entry = state.activeScoreEntries[index]
		if entry.expiresAt and entry.expiresAt <= now then
			table.remove(state.activeScoreEntries, index)
			removed += 1
		end

function Utils.GetPointPenetration(part, worldPoint)
	if not Utils.IsWorldObstacle(part) or not Utils.IsFiniteVector3(worldPoint) then
		return 0
	end

	local localPoint = part.CFrame:PointToObjectSpace(worldPoint)
	local halfSize = part.Size * 0.5

	local xDepth = halfSize.X - math.abs(localPoint.X)
	local yDepth = halfSize.Y - math.abs(localPoint.Y)
	local zDepth = halfSize.Z - math.abs(localPoint.Z)

	if xDepth <= 0 or yDepth <= 0 or zDepth <= 0 then
		return 0
	end

	return math.min(xDepth, yDepth, zDepth)
end

function Utils.IsInsideSolidGeometry(character, root, options)
	if typeof(character) ~= "Instance" or typeof(root) ~= "Instance" then
		return false, nil, 0
	end

	if not root:IsA("BasePart") then
		return false, nil, 0
	end

	options = options or {}

	local heightOffset = options.HeightOffset or 0
	local boxSize = options.BoxSize or root.Size
	local maxParts = options.MaxParts
	local sampleHalfHeight = options.SampleHalfHeight or (root.Size.Y * 0.25)
	local tolerance = options.PenetrationTolerance or 0

	local params = Utils.MakeWorldOverlapParams(character, maxParts)
	local checkCFrame = root.CFrame + Vector3.new(0, heightOffset, 0)
	local parts = Workspace:GetPartBoundsInBox(checkCFrame, boxSize, params)

	local basePoint = root.Position + Vector3.new(0, heightOffset, 0)
	local bestPart = nil
	local bestPenetration = 0

	for _, part in ipairs(parts) do
		if Utils.IsWorldObstacle(part) then
			local penetration = Utils.GetPointPenetration(part, basePoint)

			if sampleHalfHeight > 0 then
				penetration = math.max(
					penetration,
					Utils.GetPointPenetration(part, basePoint + Vector3.new(0, sampleHalfHeight, 0)),
					Utils.GetPointPenetration(part, basePoint - Vector3.new(0, sampleHalfHeight, 0))
				)
			end

			if penetration > bestPenetration then
				bestPart = part
				bestPenetration = penetration
			end
		end
	end

	if bestPart and bestPenetration >= tolerance then
		return true, bestPart, bestPenetration
	end

	return false, nil, 0
end

function Utils.HasHorizontalWallClearance(character, root, options)
	if typeof(character) ~= "Instance" or typeof(root) ~= "Instance" then
		return false
	end

	if not root:IsA("BasePart") then
		return false
	end

	options = options or {}

	local distance = options.Distance or 0
	if distance <= 0 then
		return true
	end

	local heights = options.Heights or { 0 }
	local normalYMax = options.NormalYMax or 0.35
	local params = Utils.MakeWorldRaycastParams(character)
	local directions = {
		Vector3.new(1, 0, 0),
		Vector3.new(-1, 0, 0),
		Vector3.new(0, 0, 1),
		Vector3.new(0, 0, -1),
		Vector3.new(1, 0, 1).Unit,
		Vector3.new(-1, 0, 1).Unit,
		Vector3.new(1, 0, -1).Unit,
		Vector3.new(-1, 0, -1).Unit,

		local origin = root.Position + Vector3.new(0, heightOffset, 0)

		for _, direction in ipairs(directions) do

			if hit and Utils.IsWorldObstacle(hit.Instance) and math.abs(hit.Normal.Y) <= normalYMax then

function Utils.HasGroundSupport(character, root, options)
	if typeof(character) ~= "Instance" or typeof(root) ~= "Instance" then
		return false, nil
	end

	if not root:IsA("BasePart") then
		return false, nil
	end

	options = options or {}

	local distance = options.Distance or 4
	local radius = options.Radius or 0
	local minNormalY = options.MinNormalY or 0
	local diagonal = radius * 0.7
	local params = Utils.MakeWorldRaycastParams(character)
	local offsets = {
		Vector3.zero,
		Vector3.new(radius, 0, 0),
		Vector3.new(-radius, 0, 0),
		Vector3.new(0, 0, radius),
		Vector3.new(0, 0, -radius),
		Vector3.new(diagonal, 0, diagonal),
		Vector3.new(-diagonal, 0, diagonal),
		Vector3.new(diagonal, 0, -diagonal),
		Vector3.new(-diagonal, 0, -diagonal),
	}

	for _, offset in ipairs(offsets) do
		local hit = Workspace:Raycast(root.Position + offset, Vector3.new(0, -distance, 0), params)
		if hit and Utils.IsWorldObstacle(hit.Instance) and hit.Normal.Y >= minNormalY then
			return true, hit
		end
	end

	return false, nil
end

function Utils.IsInWater(character, root, options)
	if typeof(root) ~= "Instance" or not root:IsA("BasePart") then
		return false
	end

	options = options or {}

	local terrain = Workspace.Terrain
	if not terrain then
		return false
	end

	local size = options.Size or Vector3.new(4, 5, 4)
	local resolution = options.Resolution or 4
	local minOccupancy = options.MinOccupancy or 0.2
	local region = Region3.new(root.Position - size * 0.5, root.Position + size * 0.5)
	region = region:ExpandToGrid(resolution)

	local ok, materials, occupancies = pcall(function()
		return terrain:ReadVoxels(region, resolution)
	end)

	if not ok or type(materials) ~= "table" or type(occupancies) ~= "table" then
		return false
	end

	for x, column in ipairs(materials) do
		for y, row in ipairs(column) do
			for z, material in ipairs(row) do
				if material == Enum.Material.Water then
					local occupancy = occupancies[x] and occupancies[x][y] and occupancies[x][y][z] or 0
					if occupancy >= minOccupancy then
						return true
					end
				end
			end
		end
	end

	return false
end

function Utils.SafeGetFullName(instance)
	if typeof(instance) ~= "Instance" then
		return tostring(instance)
	end

	local ok, fullName = pcall(function()
		return instance:GetFullName()
	end)

	if ok then
		return fullName
	end

	return instance.Name
end

-- Server-side chat logger.
-- Mirrors player chat into the server output without trusting client-side code.

local Players = game:GetService("Players")
local TextChatService = game:GetService("TextChatService")

local playerConnections = {}
local recentMessages = {}
local textChatConnection = nil

local function cleanupRecent(now)
	for key, timestamp in pairs(recentMessages) do
		if (now - timestamp) > 2 then
			recentMessages[key] = nil
		end
	end
end

local function logChat(player, message)
	if not player or type(message) ~= "string" then
		return
	end

	local cleanMessage = message:gsub("[\r\n]", " ")
	local now = os.clock()
	local dedupeKey = tostring(player.UserId) .. "\0" .. cleanMessage

	if recentMessages[dedupeKey] and (now - recentMessages[dedupeKey]) < 0.35 then
		return
	end

	recentMessages[dedupeKey] = now
	cleanupRecent(now)

	print(("[Chat] %s: %s"):format(player.Name, cleanMessage))
end

local function hookPlayer(player)
	if playerConnections[player] then
		return
	end

	playerConnections[player] = player.Chatted:Connect(function(message)
		logChat(player, message)
	end)
end

local function unhookPlayer(player)
	local connection = playerConnections[player]
	if connection then
		connection:Disconnect()
		playerConnections[player] = nil
	end
end

local function hookTextChatService()
	local ok, messageReceived = pcall(function()
		return TextChatService.MessageReceived
	end)

	if not ok or typeof(messageReceived) ~= "RBXScriptSignal" then
		return
	end

	local connectOk, connection = pcall(function()
		return messageReceived:Connect(function(textChatMessage)
			local textSource = textChatMessage.TextSource
			if not textSource then
				return
			end

			local player = Players:GetPlayerByUserId(textSource.UserId)
			if player then
				logChat(player, textChatMessage.Text or "")
			end
		end)
	end)

	if connectOk then
		textChatConnection = connection
	end
end

for _, player in ipairs(Players:GetPlayers()) do
	hookPlayer(player)
end

Players.PlayerAdded:Connect(hookPlayer)
Players.PlayerRemoving:Connect(unhookPlayer)
hookTextChatService()

script.Destroying:Connect(function()
	for player in pairs(playerConnections) do
		unhookPlayer(player)
	end

	if textChatConnection then
		textChatConnection:Disconnect()
		textChatConnection = nil
	end
end)

-- SHPClient client watchdog.
-- Keeps server heartbeat alive and locally restores obvious client-side character property tampering.
-- Enforcement decisions and kicks stay server-side in ClientWatchdog.

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local localPlayer = Players.LocalPlayer

local REMOTES_FOLDER_NAME = "AntiCheatRemotes"
local HEARTBEAT_REMOTE_NAME = "ClientHeartbeat"
local REPORT_REMOTE_NAME = "ClientReport"

local REMOTE_WAIT_TIMEOUT = 15
local DEFAULT_HEARTBEAT_INTERVAL = 1
local DEFAULT_CHARACTER_SCAN_INTERVAL = 0.25
local DEFAULT_REPORT_COOLDOWN = 2
local DEFAULT_RESTORE_COOLDOWN = 0.5

local remotesFolder = nil
local heartbeatRemote = nil
local reportRemote = nil

local heartbeatInterval = DEFAULT_HEARTBEAT_INTERVAL
local characterScanInterval = DEFAULT_CHARACTER_SCAN_INTERVAL
local reportCooldown = DEFAULT_REPORT_COOLDOWN
local suspiciousReportCooldown = DEFAULT_REPORT_COOLDOWN
local criticalReportCooldown = 0
local propertyRestoreCooldown = DEFAULT_RESTORE_COOLDOWN
local debugEnabled = false

local alive = true
local startedAt = os.clock()
local heartbeatSequence = 0
local lastReportAt = {}
local lastRestoreAt = {}
local currentBaseline = nil

local function debugLog(message)
	if debugEnabled then
		warn("[SHPClient] " .. message)
	end
end

local function cleanText(value, maxLength)
	local text = tostring(value or "")
	text = text:gsub("[\r\n]", " ")

	if #text > maxLength then
		return text:sub(1, maxLength)
	end

	return text
end

local function readNumberAttribute(attributeName, fallback, minValue, maxValue)
	if not remotesFolder then
		return fallback
	end

	local value = tonumber(remotesFolder:GetAttribute(attributeName))
	if not value then
		return fallback
	end

	if minValue or maxValue then
		value = math.clamp(value, minValue or value, maxValue or value)
	end

	return value
end

local function refreshServerSettings()
	if not remotesFolder then
		return
	end

	heartbeatInterval = readNumberAttribute("HeartbeatInterval", DEFAULT_HEARTBEAT_INTERVAL, 0.25, 10)
	characterScanInterval = readNumberAttribute("CharacterScanInterval", DEFAULT_CHARACTER_SCAN_INTERVAL, 0.1, 5)
	reportCooldown = readNumberAttribute("ReportCooldown", DEFAULT_REPORT_COOLDOWN, 0, 60)
	suspiciousReportCooldown = readNumberAttribute("SuspiciousReportCooldown", reportCooldown, 0, 60)
	criticalReportCooldown = readNumberAttribute("CriticalReportCooldown", 0, 0, 10)
	propertyRestoreCooldown = readNumberAttribute("PropertyRestoreCooldown", DEFAULT_RESTORE_COOLDOWN, 0.1, 10)
	debugEnabled = remotesFolder:GetAttribute("DebugClientAntiCheat") == true
end

local function waitForChildOfClass(parent, childName, className, timeout)
	local deadline = os.clock() + timeout

	while os.clock() < deadline do
		local child = parent:FindFirstChild(childName)
		if child and child:IsA(className) then
			return child
		end

		task.wait(0.1)
	end

	return nil
end

local function resolveRemotes()
	remotesFolder = waitForChildOfClass(ReplicatedStorage, REMOTES_FOLDER_NAME, "Folder", REMOTE_WAIT_TIMEOUT)
	if not remotesFolder then
		warn("[SHPClient] AntiCheatRemotes folder was not found")
		return false
	end

	heartbeatRemote = waitForChildOfClass(remotesFolder, HEARTBEAT_REMOTE_NAME, "RemoteEvent", REMOTE_WAIT_TIMEOUT)
	reportRemote = waitForChildOfClass(remotesFolder, REPORT_REMOTE_NAME, "RemoteEvent", REMOTE_WAIT_TIMEOUT)

	if not heartbeatRemote or not reportRemote then
		warn("[SHPClient] Client watchdog remotes were not found")
		return false
	end

	refreshServerSettings()
	remotesFolder:GetAttributeChangedSignal("HeartbeatInterval"):Connect(refreshServerSettings)
	remotesFolder:GetAttributeChangedSignal("CharacterScanInterval"):Connect(refreshServerSettings)
	remotesFolder:GetAttributeChangedSignal("ReportCooldown"):Connect(refreshServerSettings)
	remotesFolder:GetAttributeChangedSignal("SuspiciousReportCooldown"):Connect(refreshServerSettings)
	remotesFolder:GetAttributeChangedSignal("CriticalReportCooldown"):Connect(refreshServerSettings)
	remotesFolder:GetAttributeChangedSignal("PropertyRestoreCooldown"):Connect(refreshServerSettings)
	remotesFolder:GetAttributeChangedSignal("DebugClientAntiCheat"):Connect(refreshServerSettings)

	debugLog("remotes resolved heartbeatInterval=" .. tostring(heartbeatInterval))
	return true
end

local function shouldRunWithCooldown(key, cooldown, store)
	local currentTime = os.clock()
	local lastAt = store[key]

	if lastAt and currentTime - lastAt < cooldown then
		return false
	end

	store[key] = currentTime
	return true
end

local function sendReport(reportType, details, cooldown, dedupeKey)
	if not reportRemote then
		return false
	end

	local safeType = cleanText(reportType, 80)
	local safeDetails = cleanText(details, 240)
	local reportKey = safeType .. ":" .. tostring(dedupeKey or safeDetails)

	if not shouldRunWithCooldown(reportKey, cooldown or reportCooldown, lastReportAt) then
		return false
	end

	local ok, err = pcall(function()
		reportRemote:FireServer({
			Type = safeType,
			Details = safeDetails,
			ClientClock = os.clock(),
			Uptime = os.clock() - startedAt,
			ScriptName = script.Name,
		})
	end)

	if not ok then
		warn("[SHPClient] Failed to send report: " .. tostring(err))
		return false
	end

	debugLog("report sent type=" .. safeType .. " details=" .. safeDetails)
	return true
end

local function criticalFailure(reason)
	debugLog("critical failure: " .. tostring(reason))
	sendReport("critical_failure", reason, criticalReportCooldown)
end

local function sendHeartbeat()
	if not heartbeatRemote then
		return false
	end

	heartbeatSequence += 1

	local ok, err = pcall(function()
		heartbeatRemote:FireServer({
			Sequence = heartbeatSequence,
			ClientClock = os.clock(),
			Uptime = os.clock() - startedAt,
			ScriptName = script.Name,
		})
	end)

	if not ok then
		criticalFailure("heartbeat send failed: " .. tostring(err))
		return false
	end

	return true
end

local function heartbeatLoop()
	debugLog("heartbeat started interval=" .. tostring(heartbeatInterval))

	while alive do
		local ok, err = pcall(sendHeartbeat)
		if not ok then
			criticalFailure("heartbeat loop error: " .. tostring(err))
		end

		task.wait(heartbeatInterval)
	end

	debugLog("heartbeat stopped")
end

local function captureBaseline(character, humanoid)
	currentBaseline = {
		Character = character,
		WalkSpeed = humanoid.WalkSpeed,
		JumpPower = humanoid.JumpPower,
		JumpHeight = humanoid.JumpHeight,
		HipHeight = humanoid.HipHeight,
		UseJumpPower = humanoid.UseJumpPower,
	}

	debugLog(("baseline captured WalkSpeed=%.2f JumpPower=%.2f JumpHeight=%.2f HipHeight=%.2f")
		:format(
			currentBaseline.WalkSpeed,
			currentBaseline.JumpPower,
			currentBaseline.JumpHeight,
			currentBaseline.HipHeight
		))
end

local function getCurrentHumanoid()
	local character = localPlayer.Character
	if not character then
		return nil, nil
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then
		return character, nil
	end

	if not currentBaseline or currentBaseline.Character ~= character then
		captureBaseline(character, humanoid)
	end

	return character, humanoid
end

local function restoreHumanoidProperty(humanoid, propertyName, currentValue, expectedValue, threshold)
	if expectedValue == nil or math.abs(currentValue - expectedValue) <= threshold then
		return
	end

	local key = "humanoid:" .. propertyName
	if shouldRunWithCooldown(key, propertyRestoreCooldown, lastRestoreAt) then
		pcall(function()
			humanoid[propertyName] = expectedValue
		end)

		debugLog(("%s restored from %.2f to %.2f"):format(propertyName, currentValue, expectedValue))
	end

	sendReport(
		"suspicious_property",
		("%s %.2f expected %.2f"):format(propertyName, currentValue, expectedValue),
		suspiciousReportCooldown,
		propertyName
	)
end

local function restoreBooleanProperty(instance, propertyName, expectedValue, reportName)
	local ok, currentValue = pcall(function()
		return instance[propertyName]
	end)

	if not ok or currentValue == expectedValue then
		return
	end

	local key = instance.Name .. ":" .. propertyName
	if shouldRunWithCooldown(key, propertyRestoreCooldown, lastRestoreAt) then
		pcall(function()
			instance[propertyName] = expectedValue
		end)

		debugLog(("%s restored on %s to %s"):format(propertyName, instance.Name, tostring(expectedValue)))
	end

	sendReport(
		"suspicious_property",
		reportName .. ": " .. instance.Name,
		suspiciousReportCooldown,
		reportName .. ":" .. instance.Name
	)
end

local function checkAnchoredCoreParts(character)
	local corePartNames = {
		HumanoidRootPart = true,
		UpperTorso = true,
		LowerTorso = true,
		Torso = true,
	}

	for _, descendant in ipairs(character:GetDescendants()) do
		if descendant:IsA("BasePart") and corePartNames[descendant.Name] then
			restoreBooleanProperty(descendant, "Anchored", false, "anchored core part")
		end
	end
end

local function checkCharacterProperties()
	local character, humanoid = getCurrentHumanoid()
	if not character or not humanoid or not currentBaseline then
		return
	end

	restoreHumanoidProperty(humanoid, "WalkSpeed", humanoid.WalkSpeed, currentBaseline.WalkSpeed, 2.5)

	if humanoid.UseJumpPower or currentBaseline.UseJumpPower then
		restoreHumanoidProperty(humanoid, "JumpPower", humanoid.JumpPower, currentBaseline.JumpPower, 8)
	else
		restoreHumanoidProperty(humanoid, "JumpHeight", humanoid.JumpHeight, currentBaseline.JumpHeight, 1.5)
	end

	restoreHumanoidProperty(humanoid, "HipHeight", humanoid.HipHeight, currentBaseline.HipHeight, 2)
	restoreBooleanProperty(humanoid, "PlatformStand", false, "PlatformStand enabled")
	checkAnchoredCoreParts(character)
end

local function characterLoop()
	while alive do
		local ok, err = pcall(checkCharacterProperties)
		if not ok then
			criticalFailure("property monitor error: " .. tostring(err))
		end

		task.wait(characterScanInterval)
	end
end

local function hookSelfIntegrity()
	local ancestryConnection
	ancestryConnection = script.AncestryChanged:Connect(function()
		if not script:IsDescendantOf(game) then
			alive = false
			if ancestryConnection then
				ancestryConnection:Disconnect()
			end

			debugLog("integrity failure: script removed from DataModel")
			sendReport("client_script_missing", "SHPClient removed from DataModel", criticalReportCooldown)
		end
	end)

	local enabledOk, signal = pcall(function()
		return script:GetPropertyChangedSignal("Enabled")
	end)

	if enabledOk and signal then
		signal:Connect(function()
			local ok, isEnabled = pcall(function()
				return script.Enabled
			end)

			if ok and isEnabled == false then
				debugLog("integrity failure: script disabled")
				sendReport("watchdog_broken", "SHPClient disabled", criticalReportCooldown)
			end
		end)
	end
end

local function hookCharacterLifecycle()
	localPlayer.CharacterAdded:Connect(function(character)
		currentBaseline = nil
		table.clear(lastRestoreAt)

		local humanoid = character:WaitForChild("Humanoid", 10)
		if humanoid then
			captureBaseline(character, humanoid)
		end
	end)

	local character, humanoid = getCurrentHumanoid()
	if character and humanoid then
		captureBaseline(character, humanoid)
	end
end

local function start()

		return
	end

	hookSelfIntegrity()
	hookCharacterLifecycle()

	task.spawn(heartbeatLoop)
	task.spawn(characterLoop)

end

local ok, err = pcall(start)
if not ok then
	criticalFailure("SHPClient startup error: " .. tostring(err))
end
