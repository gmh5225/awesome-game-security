# OmniClutch
OmniClutch is an advanced, lightweight client-side utility mod for Minecraft Java Edition designed to save you from sudden lethal falls. Whether you're flying with an elytra, caught in a surprise trap, or knocked off a cliff, OmniClutch reacts with smart, anti-cheat-safe automation to secure your survival.

package com.omniclutch.event;

import com.omniclutch.config.ClutchConfig;
import com.omniclutch.logic.ClutchStateMachine;
import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;
import net.minecraft.client.MinecraftClient;

public class ClientTickHandler {

    private final ClutchStateMachine stateMachine;
    private final ClutchConfig config;

    public ClientTickHandler(ClutchConfig config) {
        this.config = config;
        this.stateMachine = new ClutchStateMachine(config);
    }

    public void register() {
        ClientTickEvents.END_CLIENT_TICK.register(this::onEndTick);
    }

    private void onEndTick(MinecraftClient client) {
        stateMachine.tick(client);
    }

    public ClutchStateMachine getStateMachine() {
        return stateMachine;
    }
}

package com.omniclutch.logic;

import com.omniclutch.config.ClutchConfig;
import com.omniclutch.util.InventoryUtils;
import com.omniclutch.util.MathUtils;
import net.minecraft.block.FluidBlock;
import net.minecraft.client.MinecraftClient;
import net.minecraft.client.network.ClientPlayerEntity;
import net.minecraft.entity.player.PlayerInventory;
import net.minecraft.fluid.Fluids;
import net.minecraft.item.BlockItem;
import net.minecraft.item.ItemPlacementContext;
import net.minecraft.item.ItemStack;
import net.minecraft.registry.tag.FluidTags;
import net.minecraft.util.Hand;
import net.minecraft.util.hit.BlockHitResult;
import net.minecraft.util.math.BlockPos;
import net.minecraft.util.math.Vec3d;
import net.minecraft.world.RaycastContext;

public class ClutchStateMachine {

    public enum State {
        IDLE,
        SCANNING,
        PREPARING,
        EXECUTING,
        RECOVERING
    }

    private State currentState = State.IDLE;
    private final ClutchConfig config;
    private int reactionDelayTicksRemaining = 0;
    private int interpolationTicksElapsed = 0;
    private int totalInterpolationTicks = 0;
    private float startPitch = 0.0f;
    private float targetPitch = 90.0f;
    private int selectedSlot = -1;
    private boolean hasAttemptedPlacement = false;

    public ClutchStateMachine(ClutchConfig config) {
        this.config = config;
    }

    public void tick(MinecraftClient client) {
        if (!config.enabled || client.player == null || client.world == null) {
            reset();
            return;
        }

        ClientPlayerEntity player = client.player;

        switch (currentState) {
            case IDLE:
                updateIdleState(player);
                break;
            case SCANNING:
                updateScanningState(player);
                break;
            case PREPARING:
                updatePreparingState(player);
                break;
            case EXECUTING:
                updateExecutingState(client, player);
                break;
            case RECOVERING:
                updateRecoveringState(player);
                break;
        }
    }

    private void updateIdleState(ClientPlayerEntity player) {
        double fallDistance = player.fallDistance;
        if (fallDistance >= config.fallDistanceThreshold) {
            currentState = State.SCANNING;
            if (config.debugLogging) {
                System.out.println("[OmniClutch] Fall detected: " + fallDistance + " blocks");
            }
        }
    }

    private void updateScanningState(ClientPlayerEntity player) {
        PlayerInventory inventory = player.getInventory();
        selectedSlot = InventoryUtils.findBestClutchItem(
                inventory,
                config.prioritizeWaterBucket,
                config.prioritizeBoat,
                config.prioritizeHayBale,
                config.prioritizeSlimeBlock
        );

        if (selectedSlot != -1) {
            if (isSafeLandingNeeded(player)) {
                currentState = State.PREPARING;
                int meanDelay = (config.minReactionDelayMs + config.maxReactionDelayMs) / 2;
                int stdDev = (config.maxReactionDelayMs - config.minReactionDelayMs) / 4;
                int delayMs = MathUtils.gaussianRandom(meanDelay, stdDev);
                reactionDelayTicksRemaining = (int) Math.ceil(delayMs / 50.0);
                startPitch = player.getPitch();
                targetPitch = 90.0f;
                totalInterpolationTicks = MathUtils.clampToInt(
                        MathUtils.gaussianRandom(
                                (config.minInterpolationTicks + config.maxInterpolationTicks) / 2,
                                (config.maxInterpolationTicks - config.minInterpolationTicks) / 4
                        ),
                        config.minInterpolationTicks,
                        config.maxInterpolationTicks
                );
                interpolationTicksElapsed = 0;
                hasAttemptedPlacement = false;
                if (config.debugLogging) {
                    System.out.println("[OmniClutch] Item found in slot " + selectedSlot + ", preparing clutch");
                }
            } else {
                currentState = State.IDLE;
                if (config.debugLogging) {
                    System.out.println("[OmniClutch] Safe landing detected, aborting");
                }
            }
        } else {
            currentState = State.IDLE;
            if (config.debugLogging) {
                System.out.println("[OmniClutch] No clutch item available");
            }
        }
    }

    private boolean isSafeLandingNeeded(ClientPlayerEntity player) {
        Vec3d startPos = player.getPos().add(0.0, player.getEyeHeight(player.getPose()), 0.0);
        Vec3d endPos = startPos.add(0.0, -config.raycastMaxDistance, 0.0);
        RaycastContext context = new RaycastContext(
                startPos,
                endPos,
                RaycastContext.ShapeType.COLLIDER,
                RaycastContext.FluidHandling.NONE,
                player
        );
        BlockHitResult hitResult = player.getWorld().raycast(context);
        BlockPos hitPos = hitResult.getBlockPos();

        double remainingFall = startPos.y - hitResult.getPos().y;

        if (remainingFall <= config.safeLandingThreshold) {
            return false;
        }

        if (player.getWorld().getFluidState(hitPos).isIn(FluidTags.WATER)) {
            return false;
        }

        return true;
    }

    private void updatePreparingState(ClientPlayerEntity player) {
        if (reactionDelayTicksRemaining > 0) {
            reactionDelayTicksRemaining--;
            return;
        }

        player.getInventory().selectedSlot = selectedSlot;

        interpolationTicksElapsed++;
        double t = (double) interpolationTicksElapsed / (double) totalInterpolationTicks;
        float newPitch = MathUtils.lerpFloat(startPitch, targetPitch, t);
        player.setPitch(newPitch);

        if (interpolationTicksElapsed >= totalInterpolationTicks) {
            player.setPitch(targetPitch);
            currentState = State.EXECUTING;
            if (config.debugLogging) {
                System.out.println("[OmniClutch] Rotation complete, executing placement");
            }
        }
    }

    private void updateExecutingState(MinecraftClient client, ClientPlayerEntity player) {
        if (hasAttemptedPlacement) {
            currentState = State.RECOVERING;
            return;
        }

        ItemStack stack = player.getInventory().getMainHandStack();
        if (stack.isEmpty()) {
            currentState = State.RECOVERING;
            return;
        }

        Vec3d startPos = player.getPos().add(0.0, player.getEyeHeight(player.getPose()), 0.0);
        Vec3d lookVec = player.getRotationVec(1.0f);
        Vec3d endPos = startPos.add(lookVec.multiply(config.raycastMaxDistance));

        RaycastContext context = new RaycastContext(
                startPos,
                endPos,
                RaycastContext.ShapeType.COLLIDER,
                RaycastContext.FluidHandling.NONE,
                player
        );
        BlockHitResult hitResult = player.getWorld().raycast(context);

        Hand hand = Hand.MAIN_HAND;

        if (stack.getItem() instanceof BucketItem) {
            client.interactionManager.interactItem(player, hand);
            hasAttemptedPlacement = true;
            if (config.debugLogging) {
                System.out.println("[OmniClutch] Used water bucket");
            }
        } else if (stack.getItem() instanceof BlockItem) {
            client.interactionManager.interactBlock(player, hand, hitResult);
            hasAttemptedPlacement = true;
            if (config.debugLogging) {
                System.out.println("[OmniClutch] Placed block at " + hitResult.getBlockPos());
            }
        } else {
            client.interactionManager.interactItem(player, hand);
            hasAttemptedPlacement = true;
            if (config.debugLogging) {
                System.out.println("[OmniClutch] Used item");
            }
        }

        currentState = State.RECOVERING;
    }

    private void updateRecoveringState(ClientPlayerEntity player) {
        reset();
    }

    public void reset() {
        currentState = State.IDLE;
        reactionDelayTicksRemaining = 0;
        interpolationTicksElapsed = 0;
        totalInterpolationTicks = 0;
        startPitch = 0.0f;
        targetPitch = 90.0f;
        selectedSlot = -1;
        hasAttemptedPlacement = false;
    }

    public State getCurrentState() {
        return currentState;
    }
}

package com.omniclutch.util;

import net.minecraft.block.Block;
import net.minecraft.entity.player.PlayerInventory;
import net.minecraft.item.BlockItem;
import net.minecraft.item.BoatItem;

import net.minecraft.item.ChestBoatItem;
import net.minecraft.item.Item;
import net.minecraft.registry.Registries;
import net.minecraft.util.Identifier;

public class InventoryUtils {

    public static int findWaterBucket(PlayerInventory inventory) {
        for (int i = 0; i < inventory.size(); i++) {
            Item item = inventory.getStack(i).getItem();
            if (item instanceof BucketItem) {
                Identifier id = Registries.ITEM.getId(item);
                if (id != null && id.getPath().equals("water_bucket")) {
                    return i;
                }
            }
        }
        return -1;
    }

    public static int findBoat(PlayerInventory inventory) {
        for (int i = 0; i < inventory.size(); i++) {
            Item item = inventory.getStack(i).getItem();
            if (item instanceof BoatItem || item instanceof ChestBoatItem) {
                return i;
            }
        }
        return -1;
    }

    public static int findHayBale(PlayerInventory inventory) {
        for (int i = 0; i < inventory.size(); i++) {
            Item item = inventory.getStack(i).getItem();
            if (item instanceof BlockItem) {
                Block block = ((BlockItem) item).getBlock();
                Identifier id = Registries.BLOCK.getId(block);
                if (id != null && id.getPath().equals("hay_block")) {
                    return i;
                }
            }
        }
        return -1;
    }

    public static int findSlimeBlock(PlayerInventory inventory) {
        for (int i = 0; i < inventory.size(); i++) {
            Item item = inventory.getStack(i).getItem();
            if (item instanceof BlockItem) {
                Block block = ((BlockItem) item).getBlock();
                Identifier id = Registries.BLOCK.getId(block);
                if (id != null && id.getPath().equals("slime_block")) {
                    return i;
                }
            }
        }
        return -1;
    }

    public static int findBestClutchItem(PlayerInventory inventory, boolean prioritizeWaterBucket,
                                          boolean prioritizeBoat, boolean prioritizeHayBale,
                                          boolean prioritizeSlimeBlock) {
        if (prioritizeWaterBucket) {
            int slot = findWaterBucket(inventory);
            if (slot != -1) {
                return slot;
            }
        }
        if (prioritizeBoat) {
            int slot = findBoat(inventory);
            if (slot != -1) {
                return slot;
            }
        }
        if (prioritizeHayBale) {
            int slot = findHayBale(inventory);
            if (slot != -1) {
                return slot;
            }
        }
        if (prioritizeSlimeBlock) {
            int slot = findSlimeBlock(inventory);
            if (slot != -1) {
                return slot;
            }
        }
        return -1;
    }
}

package com.omniclutch.util;

import java.util.Random;

public class MathUtils {
    private static final Random RANDOM = new Random();

    public static int gaussianRandom(int mean, int stdDev) {
        double gaussian = RANDOM.nextGaussian();
        double value = mean + (gaussian * stdDev);
        return (int) Math.round(Math.max(0, value));
    }

    public static double cubicHermiteSpline(double t, double p0, double p1, double m0, double m1) {
        double t2 = t * t;
        double t3 = t2 * t;
        double h00 = 2.0 * t3 - 3.0 * t2 + 1.0;
        double h10 = t3 - 2.0 * t2 + t;
        double h01 = -2.0 * t3 + 3.0 * t2;
        double h11 = t3 - t2;
        return h00 * p0 + h10 * m0 + h01 * p1 + h11 * m1;
    }

    public static double smoothInterpolate(double start, double end, double t) {
        double clampedT = Math.max(0.0, Math.min(1.0, t));
        double smoothT = clampedT * clampedT * (3.0 - 2.0 * clampedT);
        return start + (end - start) * smoothT;
    }

    public static double bezierQuadratic(double t, double p0, double p1, double p2) {
        double u = 1.0 - t;
        double tt = t * t;
        double uu = u * u;
        return uu * p0 + 2.0 * u * t * p1 + tt * p2;
    }

    public static float lerpFloat(float start, float end, double t) {
        double clampedT = Math.max(0.0, Math.min(1.0, t));
        return (float) (start + (end - start) * clampedT);
    }

    public static int clampToInt(int value, int min, int max) {
        return Math.max(min, Math.min(max, value));
    }

    public static double clampToDouble(double value, double min, double max) {
        return Math.max(min, Math.min(max, value));
    }
}
