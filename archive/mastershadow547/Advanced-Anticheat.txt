local RunService = game:GetService("RunService")

local Configurations = {}

Configurations.USE_FLAG_SYSTEM = true
Configurations.USE_IMMEDIATE_ACTION_SYSTEM = false
Configurations.ACTIVATE_BANWAVE = false
Configurations.ENABLE_SERVICE_OBFUSCATION = true
Configurations.VERBOSE = true
Configurations.MINIMUM_ACCOUNT_AGE = 7

local function ValidateConfig()
	local function AssertType(key, expectedType)
		local val = Configurations[key]
		if val == nil then
			error("AntiCheat Configuration Error: " .. key .. " cannot be nil.")
		end
		if typeof(val) ~= expectedType then
			error("AntiCheat Configuration Error: " .. key .. " must be " .. expectedType .. ", got " .. typeof(val))
		end
	end

	AssertType("USE_FLAG_SYSTEM", "boolean")
	AssertType("USE_IMMEDIATE_ACTION_SYSTEM", "boolean")
	AssertType("ACTIVATE_BANWAVE", "boolean")
	AssertType("ENABLE_SERVICE_OBFUSCATION", "boolean")
	AssertType("VERBOSE", "boolean")
	AssertType("MINIMUM_ACCOUNT_AGE", "number")

	if Configurations.USE_FLAG_SYSTEM and Configurations.USE_IMMEDIATE_ACTION_SYSTEM then
		warn("AntiCheat Error: Both systems enabled. Defaulting to Flag System only.")
		Configurations.USE_IMMEDIATE_ACTION_SYSTEM = false
	elseif not Configurations.USE_FLAG_SYSTEM and not Configurations.USE_IMMEDIATE_ACTION_SYSTEM then
		error("AntiCheat Configuration Error: No system selected! Enable either Flag or Immediate Action.")
	end

	if Configurations.VERBOSE and not RunService:IsStudio() then
		warn("AntiCheat Warning: Verbose mode enabled in production environment. Turning off!")
		Configurations.VERBOSE = false
	end
end

ValidateConfig()

return Configurations

	local direction = Vector3.new(0, -(humanoid.HipHeight + 1.2), 0)

	if type(message) ~= "string" then return false end
	local lower = string.lower(message)
	RecentlyFlagged[player.UserId] = RecentlyFlagged[player.UserId] or {}
	local flags = RecentlyFlagged[player.UserId]

	for _, term in ipairs(BlacklistedTerms) do
		if string.find(lower, term, 1, true) then
			if flags[term] then return false end
			flags[term] = true
			task.delay(5, function()
				if flags then flags[term] = nil end
			end)
			AppendErrorLog(player, "BlacklistedTerm", message, source)
			if Config.USE_FLAG_SYSTEM then AntiCheatData.GetData(player).Flagged = true end
			if Config.USE_IMMEDIATE_ACTION_SYSTEM then
				player:Kick("Blacklisted term detected (" .. term .. ")")
			end
			return true
		end
	end

	for _, pattern in ipairs(BlacklistedPatterns) do
		if string.find(lower, pattern) then
			local key = "pattern_" .. pattern
			if flags[key] then return false end
			flags[key] = true
			task.delay(5, function()
				if flags then flags[key] = nil end
			end)
			AppendErrorLog(player, "SuspiciousPattern", message, source)
			if Config.USE_FLAG_SYSTEM then AntiCheatData.GetData(player).Flagged = true end
			if Config.USE_IMMEDIATE_ACTION_SYSTEM then
				player:Kick("Suspicious pattern detected")
			end
			return true
		end
	end

	return false
end

local function HandleGhostScript(player)
	RecentlyFlagged[player.UserId] = RecentlyFlagged[player.UserId] or {}
	local flags = RecentlyFlagged[player.UserId]
	if flags["GhostScript"] then return end
	flags["GhostScript"] = true
	task.delay(5, function()
		if flags then flags["GhostScript"] = nil end
	end)

	AppendErrorLog(player, "GhostScript", "[GhostScript]", "NilScript")
	if Config.USE_FLAG_SYSTEM then AntiCheatData.GetData(player).Flagged = true end
	if Config.USE_IMMEDIATE_ACTION_SYSTEM then
		player:Kick("Error fired from a nil/external script environment.")
	end
end

local function ResetMovementData(player)
	PlayerMovementData[player.UserId] = {
		lastGoodPosition = nil,
		immuneUntil = tick() + 3,
		jumpImmunityUntil = 0,
		lastCheckPos = nil,
		lastCheckTime = nil,
		airTime = 0,
		flyViolations = 0,
		speedViolations = 0,
		speedSamples = {},
		wasGrounded = false,
		lastRubberbandTime = nil,
	}
end

	RecentlyFlagged[player.UserId] = {}

	task.wait(1)

		local watchdog = script:FindFirstChild("LocalScript")

	player.CharacterAdded:Connect(function()

		task.wait(2)
		StartMovementChecks(player)
	end)

	if type(reason) ~= "string" then return end
	vPrint("[AC:Client] ExploitDetected from", player.Name, "—", reason)
	local data = AntiCheatData.GetData(player)
	if not data then return end
	AppendErrorLog(player, "ExploitDetected", reason, "ClientWatchdog")
	if Config.USE_FLAG_SYSTEM then data.Flagged = true end
	if Config.USE_IMMEDIATE_ACTION_SYSTEM then player:Kick(reason) end
end)

Remotes.LogReport.OnServerEvent:Connect(function(player, message, source)
	if IsSpamming(player) then return end
	if type(message) ~= "string" or type(source) ~= "string" then return end
	if message == "__actest__" then return end

	if message == "[GhostScript]" and source == "NilScript" then
		HandleGhostScript(player)
		return
	end

	vPrint("[AC:Log]", player.Name, "|", source, "|", message)
	AppendErrorLog(player, "LogReport", message, source)
	CheckMessageForExploits(player, message, source)
end)

	vWarn("[AC:Honeypot] GiveItem fired by", player.Name, "item:", tostring(item))

-- Showcase https://drive.google.com/file/d/15AiSCMoJgrgyQgWNhKH2uqJQCnLPmnCe/view?usp=sharing

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("RunService")
local RunService = game:GetService("RunService")
local ServerScriptService = game:GetService("ServerScriptService")

local AntiCheatDataTemplate = require(script.AntiCheatTemplate)
local Config = require(script.Parent.Configurations)
local ProfileStore = require(ServerScriptService.Libraries.ProfileStore)

local DATA_STORE_KEY = RunService:IsStudio() and "TEST_ANTICHEAT7" or "OFFICIAL_ANTICHEAT"
local AntiCheatStore = ProfileStore.New(DATA_STORE_KEY, AntiCheatDataTemplate.DEFAULT_ANTI_CHEAT_DATA)
local Profiles: {[Player]: typeof(AntiCheatStore:StartSessionAsync())} = {}

local Local = {}
local Shared = {}

function Local.OnStart()
	for _, Player in Players:GetPlayers() do
		task.spawn(Local.LoadProfile, Player)
	end
	
	Players.PlayerAdded:Connect(Local.LoadProfile)
	Players.PlayerRemoving:Connect(Local.RemoveProfile)
end

function Local.LoadProfile(Player: Player)
	if Player.AccountAge < Config.MINIMUM_ACCOUNT_AGE then
		Player:Kick("Your account must be at least "..Config.MINIMUM_ACCOUNT_AGE.." days old to play this game.")
	end
	
	local Profile = AntiCheatStore:StartSessionAsync(`{Player.UserId}`, {
		Cancel = function()
			return Player.Parent ~= Players
		end,
	})
	
	if Profile == nil then
		return Player:Kick("Profile load fail! Please rejoin or contact devs.")
	end
	
	Profile:AddUserId(Player.UserId)
	Profile:Reconcile()
	
	Profile.OnSessionEnd:Connect(function()
		Profiles[Player] = nil
		Player:Kick("Profile session ended! Please rejoin.")
	end)
	
	local IsIngame = Player.Parent == Players
	if IsIngame then
		Profiles[Player] = Profile
	else
		Profile:EndSession()
	end
	
	local HasBeenKicked = false
	
	if RunService:IsServer() then
		print(`Loaded AntiCheat profile for {Player.Name}`, Profile.Data)
	end

	if Profile.Data.Banned == true then
		Player:Kick("You have been banned from this game.")
		HasBeenKicked = true
	end
	
	if Config.USE_FLAG_SYSTEM == true then
		if Config.ACTIVATE_BANWAVE == true then
			if HasBeenKicked == false then
				Profile.Data.Banned = true
				Player:Kick("You have been banned from this game.")
			end
		end
	end
	
end

function Local.RemoveProfile(Player: Player)
	local Profile = Profiles[Player]
	if Profile ~= nil then
		Profile:EndSession()
	end
end

function Shared.GetData(Player: Player): AntiCheatDataTemplate.AntiCheatData?
	local Profile = Profiles[Player]
	if not Profile then return end
	
	return Profile.Data
end

Local.OnStart()

return Shared

# Advanced-Anticheat
A robust anti-cheat system for Roblox designed to detect movement exploits, unauthorized remote execution, and environment tampering.

# 🛠️ How to Setup
Follow these steps to integrate the system into your project.

# 1. Dependencies
1. Inside ServerScriptService, create a folder named Libraries.

2. Place the ProfileStore ModuleScript inside this folder.

- Note: If you already use ProfileStore elsewhere, simply update the ProfileStore variable in your AntiCheatData script to point to your existing location.

# 2. Core Structure
1. Create a folder inside ServerScriptService named AntiCheatLoader.

2. Inside AntiCheatLoader, create a folder named Scripts.

# 3. File Hierarchy
Organize your scripts exactly as shown below:

ServerScriptService
 ├── Libraries
 │    └── ProfileStore (ModuleScript)
 └── AntiCheatLoader (Folder)
      ├── Loader (Script)
      └── Scripts (Folder)
           ├── Configurations (ModuleScript)
           ├── AntiCheatData (ModuleScript)
           │    └── AntiCheatTemplate (ModuleScript)
           └── ServerCore (ModuleScript)
                └── ClientCore (LocalScript)
```

- ClientCore must be a LocalScript.
