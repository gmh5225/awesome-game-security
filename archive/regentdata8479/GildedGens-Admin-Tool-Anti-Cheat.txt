## Folder layout
GildedGens_Admin_BP/
  manifest.json
  scripts/main.js       entry: joins/leaves, monitoring, panel triggers, persistence flush
  scripts/config.js     defaults: PANEL_ITEM, OWNER_BOOTSTRAP_NAMES, guard presets, default blacklist

  scripts/ui.js         all dashboards (native forms)

## Install on a Realm
1. BEFORE installing: edit OWNER_BOOTSTRAP_NAMES in scripts/config.js with your gamertag (or use /scriptevent gg:claim later).
2. Import GildedGens_Admin.mcpack on a Minecraft client (double-click / open with Minecraft).
3. Realm settings -> Edit -> your world -> Add-ons -> Behavior Packs -> activate it -> save. (Realms Pro: upload via the same path.)
   No Beta APIs experiment is needed. No external networking is used.
4. Open the panel: rename a compass to "GG Panel" in an anvil and use it, or run /scriptevent gg:panel (needs commands enabled).
5. First owner: gamertag in OWNER_BOOTSTRAP_NAMES, or an operator runs /scriptevent gg:claim while no OWNER exists.

Ships in DRY-RUN (scan.dryRun = true) and Guard LOG-ONLY. Switch to enforcement in Security -> Scan settings after reviewing alerts.

## Support matrix
FULLY SUPPORTED: online player list/search, coords, dimension, health, session time; inventory + hotbar + armor + offhand
inspection (id, qty, enchants, durability, name tag, lore, tags); blacklist with wildcard ids and role/ID exemptions;
auto-removal with dry-run; manual confiscation with confirmation; roles; persistent audit log with filters.
PARTIAL: game mode (needs getGameMode in your API version, else shown n/a); "custom item properties" (only what the API
exposes); kick/ban (uses the kick command from script - verify it works on your Realm; otherwise failures are logged and
staff are alerted); all Guard checks are heuristics (see below); offline players = last known data only.
UNSUPPORTED (not faked): personal Ender Chest contents (no Script API access); contents of shulker boxes/bundles carried
in inventory; packet-level detection; reading anything of offline players' inventories.
Mitigation: items leaving an Ender Chest are caught on the next scan once in the inventory.

## Guard honesty notes
Realm packs only see sampled positions and hit/break/place events. Reach, KillAura and NoFall are approximate; speed/fly
use block-geometry airborne detection (independent of the client's onGround flag). False-positive protection: buffers,
grace after join/teleport/damage/knockback/pearl/elytra/gamemode change, effect awareness, lag pause, VL decay, exemptions,
log-only mode, three presets. Tune on log-only first.

## Operators
Realm operators automatically get ADMIN access (OPERATOR_ROLE in config.js; 0 turns it off, 3 makes them OWNER). Stored roles still apply if higher.

// GILDEDGENS GUARD - behaviour heuristics built on observable Script API state.
// NOT packet-level and NOT authoritative: a Realm behaviour pack only sees what the API exposes (positions sampled
// every few ticks, hit/break/place events). Every check is a heuristic guarded by buffers, grace periods and decay.
import { world, system } from "@minecraft/server";
import { S, log, mark } from "./store.js";
import { roleOf, notify } from "./perms.js";
import { alive } from "./inventory.js";
import { PRESETS } from "./config.js";

export const CHECKS = {
  speed: { group: "movement", pts: 3, label: "Speed" }, fly: { group: "movement", pts: 4, label: "Fly" },
  hover: { group: "movement", pts: 3, label: "Hover" }, waterWalk: { group: "movement", pts: 3, label: "Water walk" },
  noFall: { group: "movement", pts: 2, label: "NoFall (heuristic)" },
  reach: { group: "combat", pts: 3, label: "Reach" }, attackSpeed: { group: "combat", pts: 2, label: "Attack speed / pattern" },
  multiTarget: { group: "combat", pts: 3, label: "Multi-target" }, killAura: { group: "combat", pts: 3, label: "Attack direction" },
  nuker: { group: "world", pts: 3, label: "Nuker" }, fastPlace: { group: "world", pts: 2, label: "FastPlace" },
  itemSpam: { group: "world", pts: 1, label: "Item-use spam" }, interactSpam: { group: "world", pts: 1, label: "Interaction spam" },
  invIllegalStack: { group: "inventory", pts: 4, label: "Illegal stack size" }, invBannedItem: { group: "inventory", pts: 5, label: "Banned item" },
  invEnchant: { group: "inventory", pts: 4, label: "Illegal enchant level" }, invQuantity: { group: "inventory", pts: 2, label: "Suspicious quantity" },
  invRestricted: { group: "inventory", pts: 3, label: "Restricted equipment" },
};
export const GROUPS = ["movement", "combat", "world", "inventory"];
const ACTIONS = ["warn", "setback", "restrict", "kick", "ban"];
const BIG = new Set(["minecraft:ender_dragon", "minecraft:ghast", "minecraft:ravager", "minecraft:warden", "minecraft:wither", "minecraft:iron_golem", "minecraft:hoglin", "minecraft:zoglin", "minecraft:elder_guardian", "minecraft:slime", "minecraft:magma_cube", "minecraft:horse", "minecraft:donkey", "minecraft:mule", "minecraft:llama", "minecraft:camel", "minecraft:sniffer", "minecraft:cow", "minecraft:polar_bear"]);
const SOFT = /water|web|ladder|vine|scaffolding|slime|hay|honey|powder_snow|bubble|bed|berry|lava|twisting|weeping/;
const GRACE_ITEMS = new Set(["minecraft:ender_pearl", "minecraft:wind_charge", "minecraft:trident", "minecraft:firework_rocket", "minecraft:chorus_fruit", "minecraft:mace"]);

const ST = new Map();
let lagUntil = 0;
const T = () => PRESETS[S.cfg.guard.preset] ?? PRESETS.balanced;
const on = (c) => S.cfg.guard.enabled && S.cfg.guard.checks[c];
export const isLagging = () => system.currentTick < lagUntil;
export const dropState = (id) => ST.delete(id);
function st(p) {
  let s = ST.get(p.id);
  if (!s) { s = { last: null, air: 0, peak: 0, buf: {}, grace: system.currentTick + 100, lastGlide: -999, effUntil: 0, gm: null, hits: [], hitLog: [], targets: [], breaks: [], places: [], uses: [], ints: [], safe: null, pend: null, fallHurt: -999, alertAt: {}, pun: {}, lastFlag: -999 }; ST.set(p.id, s); }
  return s;
}
const exempt = (p) => !!S.cfg.guard.exempt[p.id] || (S.cfg.guard.exemptStaff && roleOf(p) >= 2);
const gmode = (p) => { try { return String(p.getGameMode()).toLowerCase(); } catch { return "unknown"; } };
const eff = (p, id) => { try { return p.getEffect(id); } catch { return undefined; } };

export const totalVL = (v) => Object.values(v.vl).reduce((a, b) => a + b, 0);
export const groupVL = (v, g) => Object.entries(v.vl).reduce((a, [k, x]) => a + (CHECKS[k]?.group === g ? x : 0), 0);
const hhmm = () => new Date().toISOString().slice(11, 19) + "Z";

function recommend(total) {
  if (total >= 40) return "Review live / consider kick or ban";
  if (total >= 20) return "Spectate and inspect inventory";
  if (total >= 8) return "Keep watching";
  return "Likely benign - monitor";
}
export function flag(p, check, detail, mult = 1) {
  const meta = CHECKS[check]; if (!meta || !on(check) || exempt(p)) return;
  const s = st(p), now = system.currentTick;
  const pts = Math.round(meta.pts * T().ptsMult * mult * 10) / 10;
  const v = (S.viol[p.id] ??= { name: p.name, vl: {}, hist: [], pun: [], last: 0 });
  v.name = p.name; v.vl[check] = (v.vl[check] || 0) + pts; v.last = Date.now(); s.lastFlag = now;
  const total = totalVL(v);
  v.hist.push({ t: Date.now(), check, pts, vl: Math.round(total * 10) / 10, d: String(detail).slice(0, 90) });
  if (v.hist.length > 25) v.hist.shift();
  mark("viol");
  if (now - (s.alertAt[check] ?? -999) >= 100) {
    s.alertAt[check] = now;
    log({ sev: meta.group === "inventory" ? 2 : 1, act: "guard:" + check, player: p.name, playerId: p.id, detail: `${detail} (+${pts}, VL ${total.toFixed(1)})` });
    notify("viewAlerts", `§c[GG GUARD] §f${p.name} §7${meta.label} §8| §f${detail} §8| §e+${pts} §7(VL §e${total.toFixed(1)}§7) §8| §7${hhmm()} §8| §a${recommend(total)}`);
  }
  punish(p, v, meta.group, s);
}
function hit(p, s, check, cond, detail, need = T().buffer) {
  if (cond) { s.buf[check] = (s.buf[check] || 0) + 1; if (s.buf[check] >= need) { s.buf[check] = 0; flag(p, check, detail); } }
  else s.buf[check] = Math.max(0, (s.buf[check] || 0) - 0.5);
}

// ---------- punishments ----------
function setback(p, s) {
  if (!s.safe) return;
  try { p.teleport({ x: s.safe.x, y: s.safe.y, z: s.safe.z }, { dimension: world.getDimension(s.safe.dim) }); s.grace = system.currentTick + 40; s.last = null; } catch { /* ignore */ }
}
export function kick(p, reason) {
  try { p.runCommand(`kick "${p.name}" ${reason}`); return true; } catch (e) {
    notify("viewAlerts", `§c[GG GUARD] Kick of ${p.name} failed (command unsupported here?): ${e}`);
    log({ sev: 2, act: "kick_failed", player: p.name, playerId: p.id, detail: String(e) });
    return false;
  }
}
function punish(p, v, group, s) {
  const g = S.cfg.guard; if (g.mode !== "punish") return;
  const th = g.punish[group]; if (!th) return;
  const gt = groupVL(v, group), lv = s.pun[group] ?? 0;
  for (let i = ACTIONS.length - 1; i >= 0; i--) {
    const a = ACTIONS[i], t = th[a];
    if (!(t > 0 && gt >= t)) continue;
    if (a === "setback") { setback(p, s); return; }
    if (i + 1 <= lv) return;
    s.pun[group] = i + 1;
    S.stats.punished++; v.pun.push({ t: Date.now(), a, g: group }); if (v.pun.length > 15) v.pun.shift(); mark("viol");
    log({ sev: 3, act: "punish:" + a, player: p.name, playerId: p.id, detail: `${group} VL ${gt.toFixed(1)}` });
    notify("viewAlerts", `§6[GG GUARD] §f${p.name} §7-> §e${a.toUpperCase()} §7(${group} VL ${gt.toFixed(1)})`);
    if (a === "warn") p.sendMessage("§c[GildedGens Guard] §fSuspicious activity detected on your account. Continued violations may lead to removal.");
    else if (a === "restrict") { try { p.addEffect("slowness", 400, { amplifier: 1, showParticles: false }); p.addEffect("weakness", 400, { amplifier: 1, showParticles: false }); } catch { /* ignore */ } }
    else if (a === "kick") kick(p, "Removed by GildedGens Guard (suspicious activity)");
    else if (a === "ban") { banPlayer(p.id, p.name, "Anti-cheat auto-ban", "GUARD"); kick(p, "Banned by GildedGens Guard"); }
    return;
  }
}
export function banPlayer(id, name, reason, by) {
  S.cfg.guard.banned[id] = { name, reason, by, t: Date.now() }; mark("cfg");
  log({ sev: 3, act: "guard_ban", player: name, playerId: id, admin: by, detail: reason });
}
export function unban(id, by) { const b = S.cfg.guard.banned[id]; delete S.cfg.guard.banned[id]; mark("cfg"); log({ sev: 2, act: "guard_unban", player: b?.name, playerId: id, admin: by }); }
export function clearVL(id, by) {
  const v = S.viol[id]; if (!v) return;
  v.vl = {}; mark("viol");
  log({ sev: 1, act: "vl_clear", player: v.name, playerId: id, admin: by });
  const s = ST.get(id); if (s) { s.pun = {}; s.buf = {}; }
}
export function checkBan(p) {
  const b = S.cfg.guard.banned[p.id];
  if (b) { notify("viewAlerts", `§c[GG GUARD] Banned player ${p.name} joined (${b.reason}). Attempting kick.`); system.runTimeout(() => { if (alive(p)) kick(p, "You are banned: " + b.reason); }, 40); }
}

// ---------- movement sampling ----------
function blk(p, y) { try { return p.dimension.getBlock({ x: Math.floor(p.location.x), y: Math.floor(y), z: Math.floor(p.location.z) }); } catch { return undefined; } }
function sample(p, now) {
  const s = st(p), L = p.location, dim = p.dimension.id, last = s.last;
  s.last = { x: L.x, y: L.y, z: L.z, tick: now, dim };
  const gm = gmode(p); if (s.gm !== null && s.gm !== gm) s.grace = now + 60; s.gm = gm;
  if (!last || last.dim !== dim || now - last.tick > 60) { s.grace = Math.max(s.grace, now + 40); s.air = 0; s.pend = null; return; }
  const dtT = now - last.tick, dt = dtT / 20, dx = L.x - last.x, dz = L.z - last.z, dy = L.y - last.y, hd = Math.hypot(dx, dz);
  if (hd > 20 || Math.abs(dy) > 24) { s.grace = now + 60; s.air = 0; s.pend = null; return; } // teleport
  if (p.isGliding === true) s.lastGlide = now;
  if (gm === "creative" || gm === "spectator" || p.isFlying === true) { s.air = 0; s.pend = null; return; }
  if (isLagging() || now < s.grace || now - s.lastGlide < 60 || p.getComponent("minecraft:riding")) { s.air = 0; s.pend = null; return; }

  const sp = eff(p, "speed"), special = eff(p, "levitation") || eff(p, "jump_boost") || eff(p, "slow_falling");
  if (special) s.effUntil = now + 60;
  const bF = blk(p, L.y), bB = blk(p, L.y - 0.3), bB2 = blk(p, L.y - 1.3);
  if (!bF || !bB) return;
  const below = bB.typeId, t = T();
  const airborne = bF.isAir && bB.isAir && bB2?.isAir === true;

  // speed
  const mult = 1 + (sp ? 0.2 * (sp.amplifier + 1) : 0);
  const slippery = /ice|soul/.test(below);
  hit(p, s, "speed", !slippery && hd / dt > t.maxSpeed * mult, `${(hd / dt).toFixed(1)} b/s (limit ${(t.maxSpeed * mult).toFixed(1)})`);

  // fly / hover / nofall (own ground detection from blocks, so spoofed onGround is irrelevant)
  const vertExempt = now < s.effUntil;
  if (airborne && !vertExempt && !p.isInWater && !p.isClimbing) {
    s.air += dtT; s.peak = Math.max(s.peak, L.y);
    if (s.air / 20 > t.flyTime) {
      const vy = dy / dt;
      hit(p, s, "fly", vy > 2.0, `rising ${vy.toFixed(1)} b/s airborne ${(s.air / 20).toFixed(1)}s`);
      hit(p, s, "hover", Math.abs(vy) < 1.0, `hovering airborne ${(s.air / 20).toFixed(1)}s`);
    }
  } else {
    if (s.air > 0 && !vertExempt && !SOFT.test(below) && !p.isInWater) {
      const fall = s.peak - L.y;
      if (fall >= t.noFall && !eff(p, "resistance") && (gm === "survival" || gm === "adventure")) s.pend = { tick: now, fall };
    }
    s.air = 0; s.peak = L.y;
  }
  if (s.pend && now - s.pend.tick >= 12) {
    if (s.fallHurt < s.pend.tick && on("noFall")) flag(p, "noFall", `fell ${s.pend.fall.toFixed(1)} blocks, no fall damage`);
    s.pend = null;
  }

  // water walking
  const wTop = below === "minecraft:water" && bF.isAir && !p.isInWater && !p.isSwimming;
  hit(p, s, "waterWalk", wTop && hd / dt > 1.0, `moving ${(hd / dt).toFixed(1)} b/s on water`);

  // remember a safe, grounded position for setbacks
  if (!airborne && !bB.isAir && !/water|lava/.test(below) && now - s.lastFlag > 40) s.safe = { x: L.x, y: L.y, z: L.z, dim };
}

function onHit(ev) {

  const n = new Set(s.targets.map((x) => x.id)).size;
  hit(p, s, "multiTarget", n >= th.multi, `${n} targets within 0.5s`, 1);

  }, 200);

// Re-reads the slot and only clears it if it still holds the expected item type (prevents removing a swapped item).
export function removeSlot(p, ref, expectId) {
  if (!alive(p)) return { ok: false, reason: "player offline" };
  try {
    const cur = readRef(p, ref);
    if (!cur) return { ok: false, reason: "slot empty" };
    if (cur.typeId !== expectId) return { ok: false, reason: "slot changed" };
    const qty = cur.amount;
    if (ref.loc === "inventory") p.getComponent("minecraft:inventory").container.setItem(ref.index, undefined);
    else p.getComponent("minecraft:equippable").setEquipment(ref.slot, undefined);
    return { ok: true, qty };
  } catch (e) { return { ok: false, reason: String(e) }; }
}

async function logFilter(p) {
  const acts = ["(all)", ...new Set(S.logs.map((e) => e.act))].slice(0, 40);
  const r = await show(p, new ModalFormData().title(TITLE("FILTER")).textField("Player contains", "").textField("Item contains", "").dropdown("Action", acts).dropdown("Min severity", ["info", "low", "medium", "high"]).textField("Last N days (0 = all)", "0", { defaultValue: "0" }));
  if (r.canceled) return admin(p);
  const [pl, item, ai, sev, days] = r.formValues;
  await logs(p, { player: String(pl).toLowerCase(), item: String(item).toLowerCase(), acts: ai > 0 ? [acts[ai]] : undefined, sev, days: Number(days) || 0 }, 0, () => admin(p));

async function logs(p, flt, page, backFn) {
  if (!need(p, "viewLogs")) return;
  const cut = flt.days ? Date.now() - flt.days * 86400000 : 0;
  const all = S.logs.filter((e) => (!flt.player || (e.player ?? "").toLowerCase().includes(flt.player)) && (!flt.item || (e.item ?? "").toLowerCase().includes(flt.item)) && (!flt.acts || flt.acts.includes(e.act)) && (!flt.actsPrefix || e.act.startsWith(flt.actsPrefix)) && (e.sev ?? 0) >= (flt.sev ?? 0) && e.t >= cut).reverse();
  const per = 8, pages = Math.max(1, Math.ceil(all.length / per)), pg = Math.min(page, pages - 1);
  const items = [];
  if (pg < pages - 1) items.push({ label: "§aOlder ▶", fn: () => logs(p, flt, pg + 1, backFn) });
  if (pg > 0) items.push({ label: "§a◀ Newer", fn: () => logs(p, flt, pg - 1, backFn) });
  items.push({ label: "§2◀ Back", fn: backFn });
  await menu(p, "LOGS", `§7${all.length} entries - page ${pg + 1}/${pages}\n\n` + (all.slice(pg * per, pg * per + per).map(logLine).join("\n") || "§7no entries"), items);
}
