use std::collections::{HashMap, HashSet};

use super::constant_pool::ConstantPool;
use super::opcodes::{
    next_pc, OP_INVOKEINTERFACE, OP_INVOKEVIRTUAL, OP_LOOKUPSWITCH, OP_TABLESWITCH,
};
use super::reader::ByteReader;
use super::RawClassData;

const MAX_ANNOTATION_DEPTH: usize = 32;

pub fn parse_attributes(
    reader: &mut ByteReader,
    cp: &ConstantPool,
    data: &mut RawClassData,
    method_sig: Option<&str>,
) -> Option<()> {
    let attr_count = reader.read_u16()? as usize;
    for _ in 0..attr_count {
        let name_idx = reader.read_u16()?;
        let length = reader.read_u32()? as usize;
        let attr_data = reader.read_bytes(length)?;

        let attr_name = cp.get_utf8(name_idx).unwrap_or("");
        // Code attribute is only valid on methods (§4.7.3)
        if attr_name == "Code" && method_sig.is_some() {
            scan_code_attribute(attr_data, cp, data, method_sig);
        } else if attr_name == "RuntimeVisibleAnnotations"
            || attr_name == "RuntimeInvisibleAnnotations"
        {
            parse_annotation_attribute(attr_data, cp, &mut data.annotations);
        }
    }
    Some(())
}

fn scan_code_attribute(
    code_attr: &[u8],
    cp: &ConstantPool,
    data: &mut RawClassData,
    method_sig: Option<&str>,
) -> Option<()> {
    let mut reader = ByteReader::new(code_attr);
    let _max_stack = reader.read_u16()?;
    let _max_locals = reader.read_u16()?;
    let code_len = reader.read_u32()? as usize;
    let bytecode = reader.read_bytes(code_len)?;

    let mut offset = 0;
    let mut has_switch = false;
    let mut m_refs: HashSet<String> = HashSet::new();

    while offset < bytecode.len() {
        let op = bytecode[offset];
        if (OP_INVOKEVIRTUAL..=OP_INVOKEINTERFACE).contains(&op) && offset + 2 < bytecode.len() {
            let cp_idx = u16::from_be_bytes([bytecode[offset + 1], bytecode[offset + 2]]);
            if let Some((owner, name, desc)) = cp.resolve_member_ref(cp_idx, true) {
                m_refs.insert(format!("{owner}.{name}"));
                if owner == "java/lang/StringBuilder"
                    && name == "append"
                    && desc == "(C)Ljava/lang/StringBuilder;"
                {
                    data.char_append_count += 1;
                }
            }
        } else if op == OP_TABLESWITCH || op == OP_LOOKUPSWITCH {
            has_switch = true;
        }

        let next = next_pc(op, bytecode, offset);
        if next <= offset || next > bytecode.len() {
            break;
        }
        offset = next;
    }

    if has_switch {
        data.switch_methods += 1;
    }
    if let Some(sig) = method_sig {
        if !m_refs.is_empty() {
            data.method_refs_by_method.insert(sig.to_string(), m_refs);
        }
    }
    Some(())
}

fn parse_annotation_attribute(
    attr_data: &[u8],
    cp: &ConstantPool,
    out: &mut Vec<(String, HashMap<String, Vec<String>>)>,
) -> Option<()> {
    let mut reader = ByteReader::new(attr_data);
    let count = reader.read_u16()? as usize;
    for _ in 0..count {
        let ann = parse_one_annotation(&mut reader, cp)?;
        out.push(ann);
    }
    Some(())
}

fn parse_one_annotation(
    reader: &mut ByteReader,
    cp: &ConstantPool,
) -> Option<(String, HashMap<String, Vec<String>>)> {
    let type_idx = reader.read_u16()?;
    let type_name = cp.get_utf8(type_idx)?.to_string();
    let num_pairs = reader.read_u16()? as usize;
    let mut elements: HashMap<String, Vec<String>> = HashMap::new();

    for _ in 0..num_pairs {
        let name_idx = reader.read_u16()?;
        let name = cp.get_utf8(name_idx).unwrap_or("").to_string();
        let mut vals = Vec::new();
        read_element_value(reader, cp, &mut vals, 0)?;
        if !name.is_empty() && !vals.is_empty() {
            elements.insert(name, vals);
        }
    }
    Some((type_name, elements))
}

fn read_element_value(
    reader: &mut ByteReader,
    cp: &ConstantPool,
    out: &mut Vec<String>,
    depth: usize,
) -> Option<()> {
    if depth >= MAX_ANNOTATION_DEPTH {
        return None;
    }
    let tag = reader.read_u8()?;
    match tag {
        b's' | b'c' => {
            let idx = reader.read_u16()?;
            if let Some(s) = cp.get_utf8(idx) {
                out.push(s.to_string());
            }
        }
        b'e' => {
            let _enum_type_idx = reader.read_u16()?;
            let const_name_idx = reader.read_u16()?;
            if let Some(s) = cp.get_utf8(const_name_idx) {
                out.push(s.to_string());
            }
        }
        b'[' => {
            let num = reader.read_u16()? as usize;
            for _ in 0..num {
                read_element_value(reader, cp, out, depth + 1)?;
            }
        }
        b'@' => {
            let _type_idx = reader.read_u16()?;
            let pairs = reader.read_u16()? as usize;
            for _ in 0..pairs {
                let _name_idx = reader.read_u16()?;
                read_element_value(reader, cp, out, depth + 1)?;
            }
        }
        b'B' | b'C' | b'D' | b'F' | b'I' | b'J' | b'S' | b'Z' => {
            // Primitive tags have a 2-byte const_value_index
            let _ = reader.read_u16()?;
        }
        _ => return None,
    }
    Some(())
}

use std::collections::HashMap;

use super::reader::{decode_modified_utf8, ByteReader};

// Constant pool tags (JVM Spec §4.4)
pub const CONSTANT_UTF8: u8 = 1;
pub const CONSTANT_INTEGER: u8 = 3;
pub const CONSTANT_FLOAT: u8 = 4;
pub const CONSTANT_LONG: u8 = 5;
pub const CONSTANT_DOUBLE: u8 = 6;
pub const CONSTANT_CLASS: u8 = 7;
pub const CONSTANT_STRING: u8 = 8;
pub const CONSTANT_FIELDREF: u8 = 9;
pub const CONSTANT_METHODREF: u8 = 10;
pub const CONSTANT_INTERFACE_METHODREF: u8 = 11;
pub const CONSTANT_NAME_AND_TYPE: u8 = 12;
pub const CONSTANT_METHOD_HANDLE: u8 = 15;
pub const CONSTANT_METHOD_TYPE: u8 = 16;
pub const CONSTANT_DYNAMIC: u8 = 17;
pub const CONSTANT_INVOKE_DYNAMIC: u8 = 18;
pub const CONSTANT_MODULE: u8 = 19;
pub const CONSTANT_PACKAGE: u8 = 20;

#[derive(Default)]
pub struct ConstantPool {
    pub utf8: HashMap<u16, String>,
    pub classes: HashMap<u16, u16>,
    pub strings: HashMap<u16, u16>,
    pub field_refs: HashMap<u16, (u16, u16)>,
    pub method_refs: HashMap<u16, (u16, u16)>,
    pub name_and_types: HashMap<u16, (u16, u16)>,
    pub float_constants: Vec<f64>,
}

impl ConstantPool {
    #[inline]
    pub fn get_utf8(&self, idx: u16) -> Option<&str> {
        self.utf8.get(&idx).map(|s| s.as_str())
    }

    #[inline]
    pub fn get_class_name(&self, idx: u16) -> Option<&str> {
        let name_idx = self.classes.get(&idx)?;
        self.get_utf8(*name_idx)
    }

    pub fn resolve_member_ref(&self, idx: u16, is_method: bool) -> Option<(&str, &str, &str)> {
        let map = if is_method {
            &self.method_refs
        } else {
            &self.field_refs
        };
        let (class_idx, nt_idx) = map.get(&idx)?;
        let owner = self.get_class_name(*class_idx)?;
        let (name_idx, desc_idx) = self.name_and_types.get(nt_idx)?;
        let name = self.get_utf8(*name_idx)?;
        let desc = self.get_utf8(*desc_idx)?;
        Some((owner, name, desc))
    }
}

pub fn parse_constant_pool(reader: &mut ByteReader) -> Option<ConstantPool> {
    let count = reader.read_u16()? as usize;
    if count == 0 {
        return None;
    }

    let mut cp = ConstantPool::default();
    let mut idx = 1;
    while idx < count {
        let tag = reader.read_u8()?;
        match tag {
            CONSTANT_UTF8 => {
                let len = reader.read_u16()? as usize;
                let bytes = reader.read_bytes(len)?;
                cp.utf8.insert(idx as u16, decode_modified_utf8(bytes));
            }
            CONSTANT_INTEGER => {
                reader.skip(4)?;
            }
            CONSTANT_FLOAT => {
                let bits = reader.read_u32()?;
                cp.float_constants.push(f32::from_bits(bits) as f64);
            }
            CONSTANT_DOUBLE => {
                let bytes = reader.read_bytes(8)?;
                let bits = u64::from_be_bytes(bytes.try_into().ok()?);
                cp.float_constants.push(f64::from_bits(bits));
                idx += 1; // 8-byte constants take two pool entries
            }
            CONSTANT_LONG => {
                reader.skip(8)?;
                idx += 1;
            }
            CONSTANT_CLASS => {
                cp.classes.insert(idx as u16, reader.read_u16()?);
            }
            CONSTANT_STRING => {
                cp.strings.insert(idx as u16, reader.read_u16()?);
            }
            CONSTANT_FIELDREF => {
                let class_idx = reader.read_u16()?;
                let nt_idx = reader.read_u16()?;
                cp.field_refs.insert(idx as u16, (class_idx, nt_idx));
            }
            CONSTANT_METHODREF | CONSTANT_INTERFACE_METHODREF => {
                let class_idx = reader.read_u16()?;
                let nt_idx = reader.read_u16()?;
                cp.method_refs.insert(idx as u16, (class_idx, nt_idx));
            }
            CONSTANT_NAME_AND_TYPE => {
                let name_idx = reader.read_u16()?;
                let desc_idx = reader.read_u16()?;
                cp.name_and_types.insert(idx as u16, (name_idx, desc_idx));
            }
            CONSTANT_METHOD_HANDLE => {
                reader.skip(3)?;
            }
            CONSTANT_METHOD_TYPE => {
                reader.skip(2)?;
            }
            CONSTANT_DYNAMIC | CONSTANT_INVOKE_DYNAMIC => {
                reader.skip(4)?;
            }
            CONSTANT_MODULE | CONSTANT_PACKAGE => {
                reader.skip(2)?;
            }
            _ => return None,
        }
        idx += 1;
    }
    Some(cp)
}

    #[test]

        let empty = [0x00, 0x00];

    }

use std::collections::{HashMap, HashSet};

pub mod attributes;
pub mod constant_pool;
pub mod opcodes;
pub mod reader;

pub use attributes::parse_attributes;
pub use constant_pool::{parse_constant_pool, ConstantPool};
pub use opcodes::*;
pub use reader::{decode_modified_utf8, ByteReader};

/// JVM class file magic number (0xCAFEBABE).
pub const JVM_MAGIC: &[u8; 4] = b"\xCA\xFE\xBA\xBE";

#[derive(Default)]
pub struct RawClassData {
    pub this_class: String,
    pub simple_name: String,
    pub class_refs: HashSet<String>,
    pub field_refs: HashSet<String>,
    pub method_refs: HashSet<String>,
    pub strings: HashSet<String>,
    pub float_constants: Vec<f64>,
    pub member_names: HashSet<String>,
    pub annotations: Vec<(String, HashMap<String, Vec<String>>)>,
    pub char_append_count: usize,
    pub method_refs_by_method: HashMap<String, HashSet<String>>,
    pub total_methods: usize,
    pub switch_methods: usize,
    pub encrypted_string_count: usize,
}

/// Check if an identifier consists of visually confusable homoglyphs ([Il][Il1]+ | [O][O0]+)
/// or stealthy mixed-script characters (Latin + Cyrillic lookalikes).
pub fn is_homoglyph(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.len() < 2 {
        return false;
    }

    match bytes[0] {
        b'I' | b'l' if bytes[1..].iter().all(|&b| matches!(b, b'I' | b'l' | b'1')) => {
            return true;
        }
        b'O' if bytes[1..].iter().all(|&b| matches!(b, b'O' | b'0')) => {
            return true;
        }
        _ => {}
    }

    if name.is_ascii() {
        return false;
    }

    // Mixed script homoglyphs (e.g. disguising Latin 'player' with Cyrillic 'р')
    let mut has_latin = false;
    let mut has_cyrillic = false;
    for c in name.chars() {
        if c.is_ascii_alphabetic() {
            has_latin = true;
        } else if ('\u{0400}'..='\u{04FF}').contains(&c) {
            has_cyrillic = true;
        }
        if has_latin && has_cyrillic {
            return true;
        }
    }
    false
}

pub fn parse_class(raw: &[u8]) -> Option<RawClassData> {
    if raw.len() < 10 || &raw[0..4] != JVM_MAGIC {
        return None;
    }

    let mut reader = ByteReader::new(&raw[8..]); // skip magic and version
    let cp = parse_constant_pool(&mut reader)?;

    let _access_flags = reader.read_u16()?;
    let this_class_idx = reader.read_u16()?;
    let _super_class_idx = reader.read_u16()?;

    let this_class = cp.get_class_name(this_class_idx)?.to_string();
    let simple_name = this_class
        .rsplit('/')
        .next()
        .unwrap_or(&this_class)
        .to_string();

    let mut data = RawClassData {
        this_class,
        simple_name,
        ..Default::default()
    };

    populate_constant_pool_facts(&cp, &mut data);

    // Skip interfaces
    let interfaces_count = reader.read_u16()? as usize;
    reader.skip(interfaces_count * 2)?;

    parse_fields(&mut reader, &cp, &mut data)?;
    parse_methods(&mut reader, &cp, &mut data)?;
    parse_attributes(&mut reader, &cp, &mut data, None)?;

    Some(data)
}

fn populate_constant_pool_facts(cp: &ConstantPool, data: &mut RawClassData) {
    for &name_idx in cp.classes.values() {
        if let Some(s) = cp.get_utf8(name_idx) {
            data.class_refs.insert(s.to_string());
        }
    }
    for &str_idx in cp.strings.values() {
        if let Some(s) = cp.get_utf8(str_idx) {
            if crate::is_encrypted_string(s) {
                data.encrypted_string_count += 1;
            }
            data.strings.insert(s.to_string());
        }
    }
    for &idx in cp.field_refs.keys() {
        if let Some((owner, name, _)) = cp.resolve_member_ref(idx, false) {
            data.field_refs.insert(format!("{owner}.{name}"));
        }
    }
    for &idx in cp.method_refs.keys() {
        if let Some((owner, name, _)) = cp.resolve_member_ref(idx, true) {
            data.method_refs.insert(format!("{owner}.{name}"));
        }
    }
    data.float_constants = cp.float_constants.clone();
}

fn parse_fields(reader: &mut ByteReader, cp: &ConstantPool, data: &mut RawClassData) -> Option<()> {
    let fields_count = reader.read_u16()? as usize;
    for _ in 0..fields_count {
        let _f_access = reader.read_u16()?;
        let name_idx = reader.read_u16()?;
        let _desc_idx = reader.read_u16()?;
        if let Some(name) = cp.get_utf8(name_idx) {
            data.member_names.insert(name.to_string());
        }
        parse_attributes(reader, cp, data, None)?;
    }
    Some(())
}

fn parse_methods(
    reader: &mut ByteReader,
    cp: &ConstantPool,
    data: &mut RawClassData,
) -> Option<()> {
    let methods_count = reader.read_u16()? as usize;
    data.total_methods = methods_count;
    for _ in 0..methods_count {
        let _m_access = reader.read_u16()?;
        let name_idx = reader.read_u16()?;
        let desc_idx = reader.read_u16()?;
        let m_name = cp.get_utf8(name_idx).unwrap_or("");
        let m_desc = cp.get_utf8(desc_idx).unwrap_or("");
        if !m_name.is_empty() {
            data.member_names.insert(m_name.to_string());
        }
        let m_sig = format!("{m_name}{m_desc}");
        parse_attributes(reader, cp, data, Some(&m_sig))?;
    }
    Some(())
}

// Selected JVM opcodes (JVM Spec §6.5)
pub const OP_TABLESWITCH: u8 = 0xAA;
pub const OP_LOOKUPSWITCH: u8 = 0xAB;
pub const OP_INVOKEVIRTUAL: u8 = 0xB6;
pub const OP_INVOKEINTERFACE: u8 = 0xB9;
pub const OP_WIDE: u8 = 0xC4;

#[inline]
pub fn align_to_4_bytes(offset: usize) -> usize {
    (offset + 3) & !3
}

pub fn next_pc(op: u8, code: &[u8], offset: usize) -> usize {
    if op == OP_TABLESWITCH {
        let i = align_to_4_bytes(offset + 1);
        if i + 12 > code.len() {
            return code.len();
        }
        let low = i32::from_be_bytes([code[i + 4], code[i + 5], code[i + 6], code[i + 7]]) as i64;
        let high =
            i32::from_be_bytes([code[i + 8], code[i + 9], code[i + 10], code[i + 11]]) as i64;
        let count = (high - low + 1).max(0) as usize;
        let Some(table_size) = count.checked_mul(4) else {
            return code.len();
        };
        return i.saturating_add(12).saturating_add(table_size);
    }
    if op == OP_LOOKUPSWITCH {
        let i = align_to_4_bytes(offset + 1);
        if i + 8 > code.len() {
            return code.len();
        }
        let npairs = i32::from_be_bytes([code[i + 4], code[i + 5], code[i + 6], code[i + 7]]).max(0)
            as usize;
        let Some(pairs_size) = npairs.checked_mul(8) else {
            return code.len();
        };
        return i.saturating_add(8).saturating_add(pairs_size);
    }
    if op == OP_WIDE {
        if offset + 1 >= code.len() {
            return code.len();
        }
        return if code[offset + 1] == 0x84 {
            offset + 6
        } else {
            offset + 4
        };
    }

    let len = match op {
        // 2-byte opcodes (bipush, ldc, xload, xstore, ret, newarray)
        0x10 | 0x12 | 0x15..=0x19 | 0x36..=0x3A | 0xA9 | 0xBC => 2,
        // 3-byte opcodes (sipush, ldc_w, ldc2_w, iinc, branches, get/put, invokes, new, checkcast, etc.)
        0x11
        | 0x13
        | 0x14
        | 0x84
        | 0x99..=0xA8
        | 0xB2..=0xB8
        | 0xBB
        | 0xBD
        | 0xC0
        | 0xC1
        | 0xC6
        | 0xC7 => 3,
        // 4-byte opcodes (multianewarray)
        0xC5 => 4,
        // 5-byte opcodes (invokeinterface, invokedynamic, goto_w, jsr_w)
        0xB9 | 0xBA | 0xC8 | 0xC9 => 5,
        _ => 1,
    };
    offset + len
}

/// Safe, zero-allocation byte reader over borrowed binary data.
#[derive(Clone, Copy, Debug)]
pub struct ByteReader<'a> {
    data: &'a [u8],
}

impl<'a> ByteReader<'a> {
    #[inline]
    pub fn new(data: &'a [u8]) -> Self {
        Self { data }
    }

    #[inline]
    pub fn len(&self) -> usize {
        self.data.len()
    }

    #[inline]
    pub fn is_empty(&self) -> bool {
        self.data.is_empty()
    }

    #[inline]
    pub fn read_u8(&mut self) -> Option<u8> {
        let (&first, rest) = self.data.split_first()?;
        self.data = rest;
        Some(first)
    }

    #[inline]
    pub fn read_u16(&mut self) -> Option<u16> {
        let (&chunk, rest) = self.data.split_first_chunk::<2>()?;
        self.data = rest;
        Some(u16::from_be_bytes(chunk))
    }

    #[inline]
    pub fn read_u32(&mut self) -> Option<u32> {
        let (&chunk, rest) = self.data.split_first_chunk::<4>()?;
        self.data = rest;
        Some(u32::from_be_bytes(chunk))
    }

    #[inline]
    pub fn read_bytes(&mut self, len: usize) -> Option<&'a [u8]> {
        let (bytes, rest) = self.data.split_at_checked(len)?;
        self.data = rest;
        Some(bytes)
    }

    #[inline]
    pub fn skip(&mut self, len: usize) -> Option<()> {
        let (_, rest) = self.data.split_at_checked(len)?;
        self.data = rest;
        Some(())
    }

    #[inline]
    pub fn remaining(&self) -> &'a [u8] {
        self.data
    }
}

/// Decode standard and JVM Modified UTF-8 strings, preserving surrogate pairs (e.g. emojis).
pub fn decode_modified_utf8(bytes: &[u8]) -> String {
    // Fast path: standard UTF-8 without embedded NUL
    if let Ok(s) = std::str::from_utf8(bytes) {
        if !s.contains('\0') {
            return s.to_string();
        }
    }

    let mut u16_buf: Vec<u16> = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b & 0x80 == 0 {
            u16_buf.push(b as u16);
            i += 1;
        } else if b & 0xE0 == 0xC0 && i + 1 < bytes.len() {
            let b2 = bytes[i + 1];
            if b == 0xC0 && b2 == 0x80 {
                u16_buf.push(0x0000);
            } else {
                let code_unit = (((b & 0x1F) as u16) << 6) | ((b2 & 0x3F) as u16);
                u16_buf.push(code_unit);
            }
            i += 2;
        } else if b & 0xF0 == 0xE0 && i + 2 < bytes.len() {
            let b2 = bytes[i + 1];
            let b3 = bytes[i + 2];
            let code_unit =
                (((b & 0x0F) as u16) << 12) | (((b2 & 0x3F) as u16) << 6) | ((b3 & 0x3F) as u16);
            u16_buf.push(code_unit);
            i += 3;
        } else {
            // Invalid byte, represent as replacement character
            u16_buf.push(0xFFFD);
            i += 1;
        }
    }
    String::from_utf16_lossy(&u16_buf)
}

use std::fs::File;
use std::path::Path;

pub fn detect_loader(jar_path: &Path) -> String {
    let Ok(file) = File::open(jar_path) else {
        return "any".to_string();
    };
    let Ok(mut archive) = zip::ZipArchive::new(file) else {
        return "any".to_string();
    };

    let mut has_fabric = false;
    let mut has_forge = false;
    let mut has_neoforge = false;
    let mut has_quilt = false;

    for i in 0..archive.len() {
        if let Ok(entry) = archive.by_index(i) {
            if let Ok(name) = entry.name() {
                match name.as_ref() {
                    "fabric.mod.json" => has_fabric = true,
                    "quilt.mod.json" => has_quilt = true,
                    "META-INF/neoforge.mods.toml" => has_neoforge = true,
                    "META-INF/mods.toml" => has_forge = true,
                    _ => {}
                }
            }
        }
    }

    if has_quilt {
        "quilt".to_string()
    } else if has_fabric {
        "fabric".to_string()
    } else if has_neoforge {
        "neoforge".to_string()
    } else if has_forge {
        "forge".to_string()
    } else {
        "any".to_string()
    }
}

pub fn scan_single_jar(
    jar_path: &Path,
    engine: &RuleEngine,
    forced_loader: Option<&str>,
) -> Result<SingleScanReport, JarScanError> {
    let path_str = jar_path.to_string_lossy();
    let facts = scan_jar(&path_str)?;
    let loader = forced_loader
        .map(|s| s.to_string())
        .unwrap_or_else(|| detect_loader(jar_path));

    let matches = engine.evaluate(&facts, &loader, true);
    let score = calculate_score(&matches);

    let status = if score >= 7 {
        "detected"
    } else if score >= 2 {
        "suspicious"
    } else {
        "clean"
    }
    .to_string();

    let file_size = std::fs::metadata(jar_path).map(|m| m.len()).unwrap_or(0);

    Ok(SingleScanReport {

            .map(|s| s.to_string_lossy().to_string())
            .unwrap_or_else(|| path_str.to_string()),
        file_size_bytes: file_size,
        loader,
        classes_count: facts.classes.len(),
        status,
        score,
        entropy: facts.entropy,
        missing_entrypoints: facts.missing_entrypoints,
        matches,
    })
}

fn calculate_score(matches: &[RawMatch]) -> i64 {
    let mut event_scores: HashMap<(&str, &str), i64> = HashMap::new();
    for m in matches {
        let key = (m.category.as_str(), m.file.as_str());
        event_scores
            .entry(key)
            .and_modify(|current| *current = (*current).max(m.score))
            .or_insert(m.score);
    }
    event_scores.values().sum()
}

    }
}

use std::collections::BTreeMap;
use std::path::Path;

use crate::rules::engine::RuleEngine;
use crate::rules::models::RawRule;

pub const DEFAULT_RULES_JSON: &str = include_str!("../default_rules.json");

pub fn load_rules(custom_path: Option<&Path>) -> Result<(RuleEngine, usize), String> {
    match custom_path {
        None => {
            let engine = RuleEngine::from_json(DEFAULT_RULES_JSON)
                .map_err(|e| format!("Failed to parse default rules: {e}"))?;
            let count = engine.rules().len();
            Ok((engine, count))
        }
        Some(path) => {
            if path.is_file() {
                let content = std::fs::read_to_string(path)
                    .map_err(|e| format!("Failed to read {}: {e}", path.display()))?;
                let engine = RuleEngine::from_json(&content)
                    .map_err(|e| format!("Failed to compile rules from {}: {e}", path.display()))?;
                let count = engine.rules().len();
                Ok((engine, count))
            } else if path.is_dir() {
                let mut merged: BTreeMap<String, RawRule> = BTreeMap::new();
                collect_json_rules(path, &mut merged)?;
                let engine = RuleEngine::from_raw_rules(merged).map_err(|e| {
                    format!(
                        "Failed to compile merged rules from {}: {e}",
                        path.display()
                    )
                })?;
                let count = engine.rules().len();
                Ok((engine, count))
            } else {
                Err(format!("Rules path not found: {}", path.display()))
            }
        }
    }
}

fn collect_json_rules(dir: &Path, merged: &mut BTreeMap<String, RawRule>) -> Result<(), String> {
    let entries =
        std::fs::read_dir(dir).map_err(|e| format!("Failed to read dir {}: {e}", dir.display()))?;
    for entry in entries.flatten() {
        let p = entry.path();
        if p.is_dir() {
            collect_json_rules(&p, merged)?;
        } else if p.is_file() && p.extension().and_then(|s| s.to_str()) == Some("json") {
            let content = std::fs::read_to_string(&p)
                .map_err(|e| format!("Failed to read {}: {e}", p.display()))?;
            let rules: BTreeMap<String, RawRule> = serde_json::from_str(&content)
                .map_err(|e| format!("Error in rule file {}: {e}", p.display()))?;
            for (k, v) in rules {
                merged.insert(k, v);
            }
        }
    }
    Ok(())
}

use sha2::{Digest, Sha512};
use std::fs::File;
use std::io::{BufReader, Read};

const BUFFER_CAPACITY: usize = 512 * 1024;
const READ_CHUNK_SIZE: usize = 256 * 1024;

#[inline]
fn to_hex(bytes: &[u8]) -> String {
    const HEX_CHARS: &[u8; 16] = b"0123456789abcdef";
    let mut hex = String::with_capacity(bytes.len() * 2);
    for &b in bytes {
        hex.push(HEX_CHARS[(b >> 4) as usize] as char);
        hex.push(HEX_CHARS[(b & 0x0f) as usize] as char);
    }
    hex
}

/// Computes SHA-512 hex digest of a file at the given path.
pub fn sha512_file(file_path: &str) -> std::io::Result<String> {
    let file = File::open(file_path)?;
    let mut reader = BufReader::with_capacity(BUFFER_CAPACITY, file);
    let mut hasher = Sha512::new();
    let mut buffer = [0u8; READ_CHUNK_SIZE];

    loop {
        let bytes_read = reader.read(&mut buffer)?;
        if bytes_read == 0 {
            break;
        }
        hasher.update(&buffer[..bytes_read]);
    }

    Ok(to_hex(&hasher.finalize()))
}

/// Computes SHA-512 hex digest of an in-memory byte slice.
pub fn sha512_bytes(data: &[u8]) -> String {
    let mut hasher = Sha512::new();
    hasher.update(data);
    to_hex(&hasher.finalize())
}

pub fn round_to(val: f64, decimals: u32) -> f64 {
    let factor = 10_f64.powi(decimals as i32);
    (val * factor).round() / factor
}

pub fn classify_obfuscation(
    high_entropy_classes: usize,
    high_entropy_strings: usize,
    short_ratio: f64,
) -> &'static str {
    if high_entropy_classes > 0 {
        "packed"
    } else if high_entropy_strings >= 5 {
        "encrypted"
    } else if short_ratio >= 0.35 {
        "renamed"
    } else {
        "none"
    }
}

pub fn build_entropy_metrics(
    entropies: &[f64],
    total_classes: usize,
    short_names: usize,
    high_entropy_strings: usize,
    total_methods: usize,
    switch_methods: usize,
) -> EntropyMetricsData {
    let (sum_entropy, max_val, high_entropy_classes) = entropies
        .iter()
        .fold((0.0_f64, 0.0_f64, 0_usize), |(sum, max_e, count), &e| {
            (sum + e, max_e.max(e), count + usize::from(e >= 7.0))
        });

    let total_len = entropies.len();
    let avg_entropy = if total_len > 0 {
        round_to(sum_entropy / total_len as f64, 2)
    } else {
        0.0
    };
    let max_entropy = round_to(max_val, 2);

    let short_ratio = if total_classes > 0 {
        round_to(short_names as f64 / total_classes as f64, 3)
    } else {
        0.0
    };
    let switch_density = if total_methods > 0 {
        round_to(switch_methods as f64 / total_methods as f64, 3)
    } else {
        0.0
    };

    let obfuscation_type =
        classify_obfuscation(high_entropy_classes, high_entropy_strings, short_ratio);

    EntropyMetricsData {
        byte_entropy_avg: avg_entropy,
        byte_entropy_max: max_entropy,
        high_entropy_classes,
        high_entropy_strings,
        short_name_ratio: short_ratio,
        switch_density,
        obfuscation_type: obfuscation_type.to_string(),
    }
}

pub fn parse_fabric_entrypoint_value(s: &str) -> Option<String> {
    let class_part = s.split("::").next()?.trim();
    if class_part.is_empty() {
        return None;
    }
    let internal = class_part.replace('.', "/");
    (!internal.ends_with('/')).then_some(internal)
}

pub fn parse_fabric_entrypoints(fabric_json: &str, declared: &mut Vec<String>) {
    let clean_json = fabric_json.strip_prefix('\u{FEFF}').unwrap_or(fabric_json);
    let Ok(val) = serde_json::from_str::<serde_json::Value>(clean_json) else {
        return;
    };
    let Some(ep_obj) = val.get("entrypoints").and_then(|v| v.as_object()) else {
        return;
    };

    for (_key, val) in ep_obj {
        if let Some(arr) = val.as_array() {
            for item in arr {
                let candidate = item
                    .as_str()
                    .or_else(|| item.get("value").and_then(|v| v.as_str()));
                if let Some(s) = candidate.and_then(parse_fabric_entrypoint_value) {
                    declared.push(s);
                }
            }
        } else if let Some(s) = val.as_str() {
            if let Some(ep) = parse_fabric_entrypoint_value(s) {
                declared.push(ep);
            }
        }
    }
}

pub fn unfold_manifest_lines(manifest: &str) -> Vec<String> {
    let mut unfolded: Vec<String> = Vec::new();
    for line in manifest.lines() {
        if line.starts_with(' ') || line.starts_with('\t') {
            if let Some(last) = unfolded.last_mut() {
                last.push_str(&line[1..]);
                continue;
            }
        }
        unfolded.push(line.to_string());
    }
    unfolded
}

pub fn parse_manifest_entrypoints(manifest: &str, declared: &mut Vec<String>) {
    for line in unfold_manifest_lines(manifest) {
        let trimmed = line.trim();
        let lower = trimmed.to_ascii_lowercase();

        if let Some(rest) = line.strip_prefix("Fabric-Loom-Client-Only-Entries:") {
            for token in rest.split(|c: char| c == ';' || c == ',' || c.is_whitespace()) {
                let t = token.trim();
                if let Some(stripped) = t.strip_suffix(".class") {
                    declared.push(stripped.to_string());
                }
            }
        } else if lower.starts_with("main-class:")
            || lower.starts_with("premain-class:")
            || lower.starts_with("agent-class:")
            || lower.starts_with("launcher-agent-class:")
        {
            if let Some((_, val)) = trimmed.split_once(':') {
                let val = val.trim();
                if !val.is_empty() {
                    let internal = val.replace('.', "/");
                    declared.push(internal);
                }
            }
        }
    }
}

pub fn parse_entrypoints(fabric_json: Option<&str>, manifest: Option<&str>) -> Vec<String> {
    let mut declared = Vec::new();
    if let Some(json_str) = fabric_json {
        parse_fabric_entrypoints(json_str, &mut declared);
    }
    if let Some(mf) = manifest {
        parse_manifest_entrypoints(mf, &mut declared);
    }
    declared
}

pub fn find_missing_entrypoints(
    declared: Vec<String>,
    entry_names: &HashSet<String>,
) -> Vec<String> {
    let mut missing: Vec<String> = declared
        .into_iter()
        .filter(|name| {
            let class_name = format!("{name}.class");
            !entry_names.contains(&class_name) && !entry_names.contains(name)
        })
        .collect();
    missing.sort();
    missing.dedup();
    missing
}

use super::entropy::build_entropy_metrics;
use super::entrypoints::{find_missing_entrypoints, parse_entrypoints};

pub fn scan_jar(jar_path: &str) -> Result<JarFactsResult, JarScanError> {
    let file = File::open(Path::new(jar_path))?;
    let reader = BufReader::with_capacity(64 * 1024, file);
    let mut zip = ZipArchive::new(reader)?;

    let mut entry_names = HashSet::with_capacity(zip.len());
    let mut manifest_content = None;
    let mut fabric_mod_content = None;
    let mut acc = ClassScanAccumulator::default();

    for i in 0..zip.len() {
        let Ok(mut file) = zip.by_index(i) else {
            continue;
        };
        let Ok(name_cow) = file.name() else {
            continue;
        };
        let is_dir = name_cow.ends_with('/');
        let name = name_cow.to_string();

        if !is_dir {
            entry_names.insert(name.clone());
        }

        if name == "META-INF/MANIFEST.MF" {
            let mut buf = Vec::new();
            if file.read_to_end(&mut buf).is_ok() {
                manifest_content = Some(String::from_utf8_lossy(&buf).to_string());
            }
        } else if name == "fabric.mod.json" {
            let mut buf = Vec::new();
            if file.read_to_end(&mut buf).is_ok() {
                fabric_mod_content = Some(String::from_utf8_lossy(&buf).to_string());
            }
        } else if let Some(raw_bytes) = read_candidate_class_entry(&mut file) {
            acc.record_class(&name, &raw_bytes);
        }
    }

    let declared_entrypoints =
        parse_entrypoints(fabric_mod_content.as_deref(), manifest_content.as_deref());
    let missing_entrypoints = find_missing_entrypoints(declared_entrypoints, &entry_names);

    let entropy = build_entropy_metrics(
        &acc.byte_entropies,
        acc.classes.len(),
        acc.short_names,
        acc.high_entropy_strings,
        acc.total_methods,
        acc.switch_methods,
    );

    Ok(JarFactsResult {
        classes: acc.classes,
        unparsed: acc.unparsed,
        hidden_class_entries: acc.hidden_class_entries,
        missing_entrypoints,
        entropy,
    })
}

fn read_candidate_class_entry<R: Read>(file: &mut zip::read::ZipFile<'_, R>) -> Option<Vec<u8>> {
    if file.is_dir() {
        return None;
    }
    let mut magic = [0u8; 4];
    if file.read_exact(&mut magic).is_err() || &magic != JVM_MAGIC {
        return None;
    }

    let mut raw_bytes = Vec::with_capacity(4096);
    raw_bytes.extend_from_slice(&magic);
    file.read_to_end(&mut raw_bytes).ok()?;
    Some(raw_bytes)
}

#[derive(Default)]
struct ClassScanAccumulator {
    classes: Vec<ClassFactsData>,
    unparsed: Vec<String>,
    hidden_class_entries: usize,
    byte_entropies: Vec<f64>,
    short_names: usize,
    high_entropy_strings: usize,
    total_methods: usize,
    switch_methods: usize,
}

impl ClassScanAccumulator {
    fn record_class(&mut self, entry_name: &str, raw_bytes: &[u8]) {
        if !entry_name.ends_with(".class") {
            self.hidden_class_entries += 1;
        }

        self.byte_entropies.push(crate::entropy(raw_bytes));

        match classfile::parse_class(raw_bytes) {
            Some(data) => {
                let simple_stripped = data
                    .simple_name
                    .rsplit('$')
                    .next()
                    .unwrap_or(&data.simple_name);

                let is_numeric = !simple_stripped.is_empty()
                    && simple_stripped.chars().all(|c| c.is_ascii_digit());
                if simple_stripped.len() <= 2 && !is_numeric {
                    self.short_names += 1;
                }

                self.high_entropy_strings += data.encrypted_string_count;
                self.total_methods += data.total_methods;
                self.switch_methods += data.switch_methods;

                self.classes.push(build_class_facts_data(data));
            }
            None => {
                self.unparsed.push(entry_name.to_string());
            }
        }
    }
}

fn extract_homoglyphs(simple_name: &str, member_names: &HashSet<String>) -> Vec<String> {
    let simple_is_homoglyph = is_homoglyph(simple_name);
    let any_member_homoglyph = member_names.iter().any(|m| is_homoglyph(m));

    if !simple_is_homoglyph && !any_member_homoglyph {
        return Vec::new();
    }

    let mut homoglyphs = std::collections::BTreeSet::new();
    if simple_is_homoglyph {
        homoglyphs.insert(simple_name.to_string());
    }
    for m in member_names {
        if is_homoglyph(m) {
            homoglyphs.insert(m.clone());
        }
    }
    homoglyphs.into_iter().collect()
}

/// Calculate entropy in bits per byte (0.0 to 8.0) rounded to 4 decimals.
#[cfg_attr(feature = "python", pyfunction)]
pub fn entropy(data: &[u8]) -> f64 {
    if data.is_empty() {
        return 0.0;
    }
    let mut counts = [0usize; 256];
    for &b in data {
        counts[b as usize] += 1;
    }
    let total = data.len() as f64;
    let val = counts
        .iter()
        .filter(|&&c| c > 0)
        .map(|&c| {
            let p = c as f64 / total;
            -p * p.log2()
        })
        .sum::<f64>();
    (val * 10000.0).round() / 10000.0
}

/// Determine if a string looks like an encrypted or packed pool literal.
#[cfg_attr(feature = "python", pyfunction)]
pub fn is_encrypted_string(s: &str) -> bool {
    let char_count = s.chars().count();
    if char_count < 20 {
        return false;
    }

    let non_ascii_count = s.chars().filter(|&ch| !ch.is_ascii()).count();
    if non_ascii_count >= 8 {
        let ratio = (non_ascii_count as f64) / (char_count as f64);
        if ratio >= 0.4 && entropy(s.as_bytes()) >= 4.2 {
            return true;
        }
    }
    if char_count >= 64
        && !s.contains(' ')
        && !s.contains('.')
        && !s.contains('/')
        && entropy(s.as_bytes()) >= 5.0
    {
        return true;
    }
    false
}

        pyo3::exceptions::PyValueError::new_err("Invalid or unparseable Java class bytes")
    })?;
    let facts_data = jar::build_class_facts_data(raw_data);
    let facts_mod = py.import("app.bytecode.facts")?;
    let class_facts_cls = facts_mod.getattr("ClassFacts")?;
    py_bridge::build_py_class_facts(py, &class_facts_cls, &facts_data)
}

fn run(cli: &Cli, style: &Style) -> Result<(), Box<dyn std::error::Error>> {
    let start_time = Instant::now();

    if !cli.json {
        print_banner(style);
    }

    let (engine, rules_count) =
        load_rules(cli.rules.as_deref()).map_err(|e| -> Box<dyn std::error::Error> { e.into() })?;

    let jars = collect_jars(&cli.path);
    if jars.is_empty() {

            style.yellow("Warning:"),
            cli.path.display()
        );
        return Ok(());
    }

    let is_batch = jars.len() > 1;
    if !cli.json && is_batch {
        print_batch_header(jars.len(), &cli.path, rules_count, style);
    }

    let (reports, stats) = execute_scan(&jars, &engine, cli, is_batch, style);

    if cli.json {
        print_json_reports(&reports, is_batch);
    } else if is_batch {
        print_batch_summary(start_time, &stats, style);
    }

    Ok(())
}

fn execute_scan(
    jars: &[PathBuf],
    engine: &RuleEngine,
    cli: &Cli,
    is_batch: bool,
    style: &Style,
) -> (Vec<SingleScanReport>, ScanStats) {
    let mut reports = Vec::with_capacity(jars.len());
    let mut stats = ScanStats::default();

    for jar in jars {
        match scan_single_jar(jar, engine, cli.loader.as_deref()) {
            Ok(report) => {
                stats.record(&report.status);
                if !cli.json {
                    if is_batch {
                        print_batch_row(&report, style);
                    } else {
                        print_single_report(&report, style);
                    }
                }
                reports.push(report);
            }
            Err(e) => eprintln!(
                " [{}] Failed to scan {}: {e}",
                style.red("ERROR"),
                jar.display()
            ),
        }
    }

    (reports, stats)
}

fn print_batch_row(report: &SingleScanReport, style: &Style) {
    let status_tag = match report.status.as_str() {
        "detected" => style.red("DETECTED  "),
        "suspicious" => style.yellow("SUSPICIOUS"),
        _ => style.green("CLEAN     "),
    };
    let display_name = format_file_column(&report.file);
    let rules_info = if report.matches.is_empty() {
        format!("entropy: {:.2}", report.entropy.byte_entropy_max)
    } else {
        let mut categories: Vec<&str> =
            report.matches.iter().map(|m| m.category.as_str()).collect();
        categories.sort_unstable();
        categories.dedup();
        categories.join(", ")
    };
    println!(
        " [{}] {} (score: {:>2}, classes: {:>4}) | {}",
        status_tag,
        style.bold(&display_name),
        report.score,
        report.classes_count,
        style.dim(&rules_info)
    );
}

        format!("{}...", &filename[..33])
    } else {
        format!("{filename:<36}")
    }
}

use pyo3::prelude::*;
use pyo3::types::{PyDict, PyFrozenSet, PyList, PyTuple};

use crate::jar::{self, ClassFactsData, JarFactsResult};
use crate::rules::{self, RawMatch};

pub fn build_py_class_facts<'py>(
    py: Python<'py>,
    class_facts_cls: &Bound<'py, PyAny>,
    c: &ClassFactsData,
) -> PyResult<Bound<'py, PyAny>> {
    let method_refs = PyFrozenSet::new(py, &c.method_refs)?;
    let field_refs = PyFrozenSet::new(py, &c.field_refs)?;
    let class_refs = PyFrozenSet::new(py, &c.class_refs)?;
    let strings = PyFrozenSet::new(py, &c.strings)?;
    let float_constants = PyFrozenSet::new(py, &c.float_constants)?;
    let member_names = PyFrozenSet::new(py, &c.member_names)?;
    let member_homoglyphs = PyTuple::new(py, &c.member_homoglyphs)?;

    let mut ann_pairs = Vec::new();
    for (ann_type, elements) in &c.annotations {
        let py_elements = PyDict::new(py);
        for (k, v) in elements {
            py_elements.set_item(k, v)?;
        }
        ann_pairs.push((ann_type, py_elements));
    }
    let annotations = PyTuple::new(py, ann_pairs)?;

    let py_m_by_m = PyDict::new(py);
    for (sig, refs) in &c.method_refs_by_method {
        let f_set = PyFrozenSet::new(py, refs)?;
        py_m_by_m.set_item(sig, f_set)?;
    }

    class_facts_cls.call1((
        &c.class_name,
        &c.simple_name,
        method_refs,
        field_refs,
        class_refs,
        strings,
        float_constants,
        annotations,
        member_names,
        c.char_append_count,
        member_homoglyphs,
        py_m_by_m,
    ))
}

    let entropy_cls = entropy_mod.getattr("EntropyMetrics")?;

        ent.byte_entropy_avg,
        ent.byte_entropy_max,
        ent.high_entropy_classes,
        ent.high_entropy_strings,

pub fn build_py_jar_facts<'py>(
    py: Python<'py>,
    result: &JarFactsResult,
) -> PyResult<Bound<'py, PyAny>> {
    let facts_mod = py.import("app.bytecode.facts")?;
    let class_facts_cls = facts_mod.getattr("ClassFacts")?;
    let jar_facts_cls = facts_mod.getattr("JarFacts")?;

    let py_classes = PyList::empty(py);
    for c in &result.classes {
        let class_facts = build_py_class_facts(py, &class_facts_cls, c)?;
        py_classes.append(class_facts)?;
    }

    let entropy_obj = build_py_entropy(py, &result.entropy)?;

    jar_facts_cls.call1((
        py_classes,
        &result.unparsed,
        result.hidden_class_entries,
        &result.missing_entrypoints,
        entropy_obj,
    ))
}

pub fn build_py_matches<'py>(
    py: Python<'py>,
    raw_matches: &[RawMatch],
) -> PyResult<Bound<'py, PyList>> {
    let result_mod = py.import("app.models.result")?;
    let match_cls = result_mod.getattr("Match")?;
    let list = PyList::empty(py);

    for m in raw_matches {
        let kwargs = PyDict::new(py);
        kwargs.set_item("category", &m.category)?;
        kwargs.set_item("trigger", &m.trigger)?;
        kwargs.set_item("file", &m.file)?;
        kwargs.set_item("line_num", m.line_num)?;
        kwargs.set_item("code_snippet", &m.code_snippet)?;
        kwargs.set_item("severity", &m.severity)?;
        kwargs.set_item("score", m.score)?;
        let py_match = match_cls.call((), Some(&kwargs))?;
        list.append(py_match)?;
    }
    Ok(list)
}

pub struct RuleEngine {
    inner: rules::RuleEngine,
}

#[pymethods]
impl RuleEngine {
    #[new]
    pub fn new(rules_json: &str) -> PyResult<Self> {
        let inner = rules::RuleEngine::from_json(rules_json)
            .map_err(|e| pyo3::exceptions::PyValueError::new_err(e.to_string()))?;
        Ok(Self { inner })
    }

    /// Evaluates rules on a JAR file directly in Rust without holding GIL.
    pub fn scan_jar<'py>(
        &self,
        py: Python<'py>,
        jar_path: &str,
        loader: &str,
    ) -> PyResult<Bound<'py, PyList>> {
        let raw_matches = py
            .detach(|| {
                let facts = jar::scan_jar(jar_path)?;
                Ok::<_, jar::JarScanError>(self.inner.evaluate(&facts, loader, true))
            })
            .map_err(|e| pyo3::exceptions::PyValueError::new_err(e.to_string()))?;

        build_py_matches(py, &raw_matches)
    }

    /// Evaluates rules on a JAR file directly in Rust and returns Matches and EntropyMetrics
    /// WITHOUT the expensive overhead of marshalling hundreds of ClassFacts into Python.
    pub fn scan_jar_with_entropy<'py>(
        &self,
        py: Python<'py>,
        jar_path: &str,
        loader: &str,
    ) -> PyResult<(Bound<'py, PyList>, Bound<'py, PyAny>)> {
        let (raw_matches, entropy) = py
            .detach(|| {
                let facts = jar::scan_jar(jar_path)?;
                let raw_matches = self.inner.evaluate(&facts, loader, true);
                Ok::<_, jar::JarScanError>((raw_matches, facts.entropy))
            })
            .map_err(|e| pyo3::exceptions::PyValueError::new_err(e.to_string()))?;

        let py_matches = build_py_matches(py, &raw_matches)?;
        let py_entropy = build_py_entropy(py, &entropy)?;
        Ok((py_matches, py_entropy))
    }

    /// Evaluates rules on a JAR file directly in Rust and returns both Matches and JarFacts.
    pub fn scan_jar_with_facts<'py>(
        &self,
        py: Python<'py>,
        jar_path: &str,
        loader: &str,
    ) -> PyResult<(Bound<'py, PyList>, Bound<'py, PyAny>)> {
        let (raw_matches, facts) = py
            .detach(|| {
                let facts = jar::scan_jar(jar_path)?;
                let raw_matches = self.inner.evaluate(&facts, loader, true);
                Ok::<_, jar::JarScanError>((raw_matches, facts))
            })
            .map_err(|e| pyo3::exceptions::PyValueError::new_err(e.to_string()))?;

        let py_matches = build_py_matches(py, &raw_matches)?;
        let py_jar_facts = build_py_jar_facts(py, &facts)?;
        Ok((py_matches, py_jar_facts))
    }
}

use std::collections::{BTreeMap, HashMap};

use super::matcher::class_matchers_hold;
use super::models::{CompiledMatcher, CompiledRule, MatcherFact, RawMatch, RawRule, RuleScope};
use super::triggers::{class_trigger, jar_trigger, sanitize_path};
use crate::jar::JarFactsResult;

pub const MAX_MATCHES_PER_CATEGORY: usize = 5;

#[derive(Debug, thiserror::Error)]
pub enum RuleCompileError {
    #[error("Rule '{0}' must declare at least one matcher in all_of")]
    EmptyAllOf(String),
    #[error("Invalid regex in pattern '{0}': {1}")]
    RegexError(String, Box<fancy_regex::Error>),
    #[error("JSON deserialization error: {0}")]
    Json(#[from] serde_json::Error),
}

#[derive(Debug, Clone)]
pub struct RuleEngine {
    rules: Vec<CompiledRule>,
}

impl RuleEngine {
    pub fn new(rules: Vec<CompiledRule>) -> Self {
        Self { rules }
    }

    /// Compile engine from a JSON string containing rule definitions.
    pub fn from_json(json_str: &str) -> Result<Self, RuleCompileError> {
        let raw_rules: BTreeMap<String, RawRule> = serde_json::from_str(json_str)?;
        Self::from_raw_rules(raw_rules)
    }

    /// Compile engine directly from a map of parsed raw rules without JSON serialization roundtrip.
    pub fn from_raw_rules(raw_rules: BTreeMap<String, RawRule>) -> Result<Self, RuleCompileError> {
        let mut compiled = Vec::with_capacity(raw_rules.len());

        for (name, raw) in raw_rules {
            if raw.all_of.is_empty() {
                return Err(RuleCompileError::EmptyAllOf(name));
            }
            let scope = match raw.scope.as_str() {
                "jar" => RuleScope::Jar,
                _ => RuleScope::Class,
            };

            let mut matchers = Vec::with_capacity(raw.all_of.len());
            for m in raw.all_of {
                let fact = MatcherFact::from(m.fact.as_str());
                let regex = if let Some(ref pat) = m.pattern {
                    Some(
                        fancy_regex::Regex::new(pat)
                            .map_err(|e| RuleCompileError::RegexError(pat.clone(), Box::new(e)))?,
                    )
                } else {
                    None
                };
                matchers.push(CompiledMatcher {
                    fact,
                    any: m.any,
                    value: m.value,
                    pattern: m.pattern,
                    regex,
                    same_method: m.same_method,
                });
            }

            compiled.push(CompiledRule {
                name,
                scope,
                loaders: raw.loaders,
                severity: raw.severity,
                score: raw.score,
                all_of: matchers,
            });
        }

        Ok(Self { rules: compiled })
    }

    pub fn rules(&self) -> &[CompiledRule] {
        &self.rules
    }

    pub fn evaluate(
        &self,
        facts: &JarFactsResult,
        loader: &str,
        target_ext_java: bool,
    ) -> Vec<RawMatch> {
        let mut matches = Vec::new();
        let mut per_category = HashMap::new();

        for rule in &self.rules {
            if !rule.is_applicable(loader) {
                continue;
            }

            match rule.scope {
                RuleScope::Jar => {
                    if rule.all_of.iter().all(|m| m.holds_for_jar(facts)) {
                        let trigger = jar_trigger(rule, facts);
                        emit_match(
                            &mut matches,
                            &mut per_category,
                            rule,
                            "archive",
                            1,
                            trigger,
                            target_ext_java,
                        );
                    }
                }
                RuleScope::Class => {
                    for class_facts in &facts.classes {
                        if !class_matchers_hold(rule, class_facts) {
                            continue;
                        }
                        let trigger = class_trigger(rule, class_facts);
                        emit_match(
                            &mut matches,
                            &mut per_category,
                            rule,
                            &class_facts.class_name,
                            1,
                            trigger,
                            target_ext_java,
                        );
                        if per_category.get(rule.name.as_str()).copied().unwrap_or(0)
                            >= MAX_MATCHES_PER_CATEGORY
                        {
                            break;
                        }
                    }
                }
            }
        }

        matches
    }
}

fn emit_match<'a>(
    matches: &mut Vec<RawMatch>,
    per_category: &mut HashMap<&'a str, usize>,
    rule: &'a CompiledRule,
    file_name: &str,
    line: usize,
    trigger: String,
    target_ext_java: bool,
) {
    let count = per_category.entry(rule.name.as_str()).or_insert(0);
    if *count >= MAX_MATCHES_PER_CATEGORY {
        return;
    }
    *count += 1;

    let sanitized = sanitize_path(file_name);
    let target_file = if sanitized == "archive" {
        "archive".to_string()
    } else if target_ext_java {
        let base = sanitized.strip_suffix(".class").unwrap_or(&sanitized);
        format!("{}.java", base)
    } else if sanitized.ends_with(".class") {
        sanitized
    } else {
        format!("{}.class", sanitized)
    };

    matches.push(RawMatch {
        category: rule.name.clone(),
        trigger,
        file: target_file,
        line_num: line.max(1),
        code_snippet: String::new(),
        severity: rule.severity.clone(),
        score: rule.score.max(0),
    });
}

use std::collections::HashSet;

use super::models::{CompiledMatcher, CompiledRule, MatcherFact};
use crate::jar::{ClassFactsData, JarFactsResult};

pub fn is_gibberish_name(name: &str) -> bool {
    let char_count = name.chars().count();
    if !(2..=6).contains(&char_count) {
        return false;
    }
    if name.chars().any(|ch| ch.is_ascii_digit()) {
        return true;
    }
    !name.chars().any(|ch| {
        matches!(
            ch,
            'a' | 'e' | 'i' | 'o' | 'u' | 'y' | 'A' | 'E' | 'I' | 'O' | 'U' | 'Y'
        )
    })
}

pub fn suffix_match(haystack: &HashSet<String>, needles: &[String]) -> bool {
    if needles.is_empty() || haystack.is_empty() {
        return false;
    }
    haystack
        .iter()
        .any(|ref_str| needles.iter().any(|needle| ref_str.ends_with(needle)))
}

pub fn annotation_match(
    annotations: &[(String, std::collections::HashMap<String, Vec<String>>)],
    type_suffixes: &[String],
    rx: Option<&fancy_regex::Regex>,
) -> bool {
    for (type_name, elements) in annotations {
        if !type_suffixes
            .iter()
            .any(|suffix| type_name.ends_with(suffix))
        {
            continue;
        }
        let Some(rx) = rx else {
            return true;
        };
        for values in elements.values() {
            if values.iter().any(|v| rx.is_match(v).unwrap_or(false)) {
                return true;
            }
        }
    }
    false
}

impl CompiledMatcher {
    pub fn holds_for_class(&self, facts: &ClassFactsData) -> bool {
        match self.fact {
            MatcherFact::MethodRef if !self.same_method => {
                suffix_match(&facts.method_refs, &self.any)
            }
            MatcherFact::FieldRef => suffix_match(&facts.field_refs, &self.any),
            MatcherFact::ClassRef => suffix_match(&facts.class_refs, &self.any),
            MatcherFact::String => {
                let Some(rx) = &self.regex else {
                    return false;
                };
                facts
                    .strings
                    .iter()
                    .any(|s| rx.is_match(s).unwrap_or(false))
            }
            MatcherFact::Annotation => {
                annotation_match(&facts.annotations, &self.any, self.regex.as_ref())
            }
            MatcherFact::CharConcat => {
                let threshold = self.value.unwrap_or(60.0) as usize;
                facts.char_append_count >= threshold
            }
            MatcherFact::MemberHomoglyphsMin => {
                let threshold = self.value.unwrap_or(0.0) as usize;
                facts.member_homoglyphs.len() >= threshold
            }
            MatcherFact::FloatConstantMin => {
                let threshold = self.value.unwrap_or(0.0);
                facts.float_constants.iter().any(|c| c.abs() >= threshold)
            }
            _ => false,
        }
    }

    fn holds_for_jar_entropy(&self, facts: &JarFactsResult) -> bool {
        match self.fact {
            MatcherFact::ShortNameRatioMin => {
                facts.entropy.short_name_ratio >= self.value.unwrap_or(0.0)
            }
            MatcherFact::ByteEntropyMaxMin => {
                facts.entropy.byte_entropy_max >= self.value.unwrap_or(0.0)
            }
            MatcherFact::HighEntropyStringsMin => {
                let threshold = self.value.unwrap_or(0.0) as usize;
                facts.entropy.high_entropy_strings >= threshold
            }
            MatcherFact::HighEntropyClassesMin => {
                let threshold = self.value.unwrap_or(0.0) as usize;
                facts.entropy.high_entropy_classes >= threshold
            }
            MatcherFact::SwitchDensityMin => {
                facts.entropy.switch_density >= self.value.unwrap_or(0.0)
            }
            _ => false,
        }
    }

    pub fn holds_for_jar(&self, facts: &JarFactsResult) -> bool {
        match self.fact {
            MatcherFact::ClassesMin => {
                let threshold = self.value.unwrap_or(0.0) as usize;
                facts.classes.len() >= threshold
            }
            MatcherFact::GibberishRatioMin => {
                if facts.classes.is_empty() {
                    return false;
                }
                let gibberish = facts
                    .classes
                    .iter()
                    .filter(|c| is_gibberish_name(&c.simple_name))
                    .count();
                let ratio = (gibberish as f64) / (facts.classes.len() as f64);
                ratio >= self.value.unwrap_or(0.0)
            }
            MatcherFact::SingleLetterRatioMin => {
                if facts.classes.is_empty() {
                    return false;
                }
                let singles = facts
                    .classes
                    .iter()
                    .filter(|c| c.simple_name.len() == 1)
                    .count();
                let ratio = (singles as f64) / (facts.classes.len() as f64);
                ratio >= self.value.unwrap_or(0.0)
            }
            MatcherFact::HiddenClassEntriesMin => {
                let threshold = self.value.unwrap_or(0.0) as usize;
                facts.hidden_class_entries >= threshold
            }
            MatcherFact::UnparsedClassesMin => {
                let threshold = self.value.unwrap_or(0.0) as usize;
                facts.unparsed.len() >= threshold
            }
            MatcherFact::MissingEntrypointsMin => {
                let threshold = self.value.unwrap_or(1.0) as usize;
                facts.missing_entrypoints.len() >= threshold
            }
            _ => self.holds_for_jar_entropy(facts),
        }
    }
}

pub fn class_matchers_hold(rule: &CompiledRule, facts: &ClassFactsData) -> bool {
    let same_method: Vec<&CompiledMatcher> = rule
        .all_of
        .iter()
        .filter(|m| m.same_method && m.fact == MatcherFact::MethodRef)
        .collect();

    let other_hold = rule
        .all_of
        .iter()
        .filter(|m| !(m.same_method && m.fact == MatcherFact::MethodRef))
        .all(|m| m.holds_for_class(facts));

    if !other_hold {
        return false;
    }
    if same_method.is_empty() {
        return true;
    }
    facts
        .method_refs_by_method
        .values()
        .any(|refs| same_method.iter().all(|m| suffix_match(refs, &m.any)))
}

pub fn sanitize_path(name: &str) -> String {
    let mut cleaned = String::with_capacity(name.len());
    for ch in name.chars() {
        let code = ch as u32;
        if (32..127).contains(&code) || code > 159 {
            cleaned.push(ch);
        } else {
            cleaned.push('_');
        }
    }
    let trimmed = cleaned.trim();
    if trimmed.is_empty() {
        "unknown".to_string()
    } else {
        trimmed.to_string()
    }
}

pub fn class_trigger(rule: &CompiledRule, facts: &ClassFactsData) -> String {
    let mut parts = Vec::new();
    for matcher in &rule.all_of {
        if !matcher.any.is_empty() {
            let slice: Vec<&str> = matcher.any.iter().take(3).map(|s| s.as_str()).collect();
            parts.push(slice.join(" or "));
        } else if matcher.fact == MatcherFact::String {
            if let Some(rx) = &matcher.regex {
                for s in &facts.strings {
                    if let Ok(Some(m)) = rx.find(s) {
                        parts.push(format!("literal URL: {}", m.as_str()));
                        break;
                    }
                }
            }
        }
    }
    let sanitized_cls = sanitize_path(&facts.class_name);
    if parts.is_empty() {
        return format!("{}: matched", sanitized_cls);
    }
    if parts.len() == 1 && parts[0].starts_with("literal URL:") {
        return parts[0].clone();
    }
    format!("{}: {}", sanitized_cls, parts.join(" + "))
}

pub fn jar_trigger(rule: &CompiledRule, facts: &JarFactsResult) -> String {
    for matcher in &rule.all_of {
        match matcher.fact {
            MatcherFact::ByteEntropyMaxMin | MatcherFact::HighEntropyClassesMin => {
                return format!(
                    "High entropy: max {} bits/byte ({} packed classes)",
                    facts.entropy.byte_entropy_max, facts.entropy.high_entropy_classes
                );
            }
            MatcherFact::HighEntropyStringsMin => {
                return format!(
                    "Encrypted string pool: {} high-entropy strings",
                    facts.entropy.high_entropy_strings
                );
            }
            MatcherFact::ShortNameRatioMin => {
                let pct = (facts.entropy.short_name_ratio * 100.0) as i64;
                return format!("Aggressive renaming: {}% short class names", pct);
            }
            MatcherFact::SwitchDensityMin => {
                let pct = (facts.entropy.switch_density * 100.0) as i64;
                return format!("Control flow flattening: {}% switch dispatch methods", pct);
            }
            _ => {}
        }
    }

    let gibberish = facts
        .classes
        .iter()
        .filter(|c| is_gibberish_name(&c.simple_name))
        .count();
    let mut parts = vec![
        format!(
            "{}/{} machine-generated class names",
            gibberish,
            facts.classes.len()
        ),
        format!("{} unparsable", facts.unparsed.len()),
    ];
    if !facts.missing_entrypoints.is_empty() {
        let shown = facts
            .missing_entrypoints
            .iter()
            .take(3)
            .map(|s| s.as_str())
            .collect::<Vec<_>>()
            .join(", ");
        parts.push(format!("declared but absent: {}", shown));
    }
    parts.join(", ")
}
