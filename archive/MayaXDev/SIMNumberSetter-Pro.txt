## ⚡ Overview

**SIM Number Setter Pro** (`com.mayaxdev.simnumbersetter`) is an open-source, dual-vector Android telephony utility designed to read, write, and permanently program your phone number (MSISDN) directly into the SIM card hardware chip (`EF_MSISDN`).

Eliminates the pervasive **"Unknown SIM Number" / "Unknown MSISDN"** status across Android, unlocking instantaneous OTP and SMS-based device verification in:
- **Banking & UPI Apps:** PhonePe, Google Pay, Paytm, BHIM, Cred, Navi, Amazon Pay
- **RCS & Carrier Provisioning:** Google Messages RCS chats, VoLTE / VoNR network registration
- **Android Settings:** `Settings > About Phone > Status Information > SIM Status`

Engineered and maintained by **[@MayaXDev](https://t.me/MayaXDev)** & **[@Itcalldeveloper](https://t.me/Itcalldeveloper)**.

## 🛠️ Step-by-Step Operator Guide

### Path A: Direct Root Mode (Fastest — Zero Setup)
1. Install and launch **SIM Number Setter Pro**.
2. Tap **"AUTO-GRANT PRIVILEGES"** when prompted for Superuser access (APatch / KernelSU / Magisk).
3. Select your target slot: **SIM 1 (Slot 0)** or **SIM 2 (Slot 1)**.
4. Input your mobile number with country code (e.g. `+919876543210` or `+12025550143`).
5. Tap **"WRITE & SAVE TO SIM CARD"**. The app executes direct root NVRAM commit and refreshes the telephony stack.

### Path B: LSPosed System Framework Hook Mode
1. Open **LSPosed Manager** (`Zygisk`).
2. Navigate to **Modules** → select **SIM Number Setter Pro**.
3. Toggle **Enable Module** and check the following scopes:
   - `Phone (com.android.phone)`
   - `System Framework (android)`
4. Tap **"Auto-Restart Telephony Stack"** inside the app. The hook will bind instantly (`Telephony Hook: Connected & Ready`).
5. Write your number and save.

```kt
package com.mayaxdev.simnumbersetter

import android.content.Intent
import com.mayaxdev.simnumbersetter.service.PhoneNumberSetterService
import com.mayaxdev.simnumbersetter.utils.extensions.checkSecurity
import de.robv.android.xposed.IXposedHookLoadPackage
import de.robv.android.xposed.XC_MethodHook
import de.robv.android.xposed.XposedHelpers
import de.robv.android.xposed.callbacks.XC_LoadPackage

/**
 *  Xposed hook to allow binding to TelephonyDebugService (an exported, but protected service in
 *  com.android.phone), returning [IPhoneNumberSetter]. This can then be accessed via the UI
 *  and used to query and set the number.
 */
class Xposed: IXposedHookLoadPackage {

    override fun handleLoadPackage(lpparam: XC_LoadPackage.LoadPackageParam) {
        if(lpparam.packageName != "com.android.phone") return
        XposedHelpers.findAndHookMethod(
            "com.android.phone.TelephonyDebugService",
            lpparam.classLoader,
            "onBind",
            Intent::class.java,
            object: XC_MethodHook() {
                override fun beforeHookedMethod(param: MethodHookParam) {
                    super.beforeHookedMethod(param)
                    param.onBind(param.args[0] as Intent)
                }
            }
        )
    }

    private fun XC_MethodHook.MethodHookParam.onBind(intent: Intent) {
        if(intent.checkSecurity()){
            result = PhoneNumberSetterService()
        }
    }

}
```

```kt
package com.mayaxdev.simnumbersetter.repositories

import com.topjohnwu.superuser.Shell
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext

interface RootRepository {

    suspend fun isRooted(): Boolean
    suspend fun runRootCommand(command: String): Shell.Result
    suspend fun restartTelephony(): Boolean
    suspend fun getLine1NumberViaRoot(slotIndex: Int): String?
    suspend fun setLine1NumberViaRoot(slotIndex: Int, number: String): Boolean

}

class RootRepositoryImpl: RootRepository {

    override suspend fun isRooted(): Boolean {
        return withContext(Dispatchers.IO) {
            Shell.rootAccess()
        }
    }

    override suspend fun runRootCommand(command: String): Shell.Result {
        return withContext(Dispatchers.IO){
            Shell.su(command).exec()
        }
    }

    override suspend fun restartTelephony(): Boolean {
        return withContext(Dispatchers.IO) {
            val result = Shell.su("killall com.android.phone || pkill -f com.android.phone").exec()
            result.isSuccess
        }
    }

    override suspend fun getLine1NumberViaRoot(slotIndex: Int): String? {
        return withContext(Dispatchers.IO) {
            try {
                val result = Shell.su("content query --uri content://telephony/siminfo --projection number --where \"sim_id=$slotIndex\"").exec()
                if (!result.isSuccess) return@withContext null
                for (line in result.out) {
                    // Output format: Row: 0 number=91XXXXXXXXXX
                    val regex = Regex("number=([^,\\s]*)")
                    val match = regex.find(line)
                    if (match != null) {
                        val num = match.groupValues[1]
                        return@withContext if (num == "null") "" else num
                    }
                }
                ""
            } catch (t: Throwable) {
                null
            }
        }
    }

    override suspend fun setLine1NumberViaRoot(slotIndex: Int, number: String): Boolean {
        return withContext(Dispatchers.IO) {
            try {
                val cmd = "content update --uri content://telephony/siminfo --bind number:s:$number --where \"sim_id=$slotIndex\""
                val result = Shell.su(cmd).exec()
                result.isSuccess
            } catch (t: Throwable) {
                false
            }
        }
    }

}
```

```kt
package com.mayaxdev.simnumbersetter.repositories

import android.content.ComponentName
import android.content.Context
import android.content.Intent
import android.content.ServiceConnection
import android.os.IBinder
import com.mayaxdev.simnumbersetter.IPhoneNumberSetter
import com.mayaxdev.simnumbersetter.utils.extensions.applySecurity
import kotlinx.coroutines.*
import kotlinx.coroutines.sync.Mutex
import java.lang.IllegalArgumentException
import java.util.concurrent.TimeUnit
import kotlin.coroutines.resume

interface ServiceRepository {

    suspend fun <T> runWithService(block: suspend (IPhoneNumberSetter) -> T): T?
    fun <T> runWithServiceIfAvailable(block: (IPhoneNumberSetter) -> T?): T?
    fun unbindServiceIfNeeded()

}

class ServiceRepositoryImpl(private val context: Context): ServiceRepository {

    companion object {
        // Fast 1.5s timeout: if Xposed hook is not intercepting onBind, failover cleanly to direct root
        private const val SERVICE_TIMEOUT_MS = 1500L
    }

    private var serviceInstance: IPhoneNumberSetter? = null
    private var serviceConnection: ServiceConnection? = null
    private val getServiceMutex = Mutex()

    private val serviceIntent = Intent("com.android.phone.TelephonyDebugService").apply {
        `package` = "com.android.phone"
        applySecurity(context)
    }

    override suspend fun <T> runWithService(block: suspend (IPhoneNumberSetter) -> T): T? {
        val service = withContext(Dispatchers.IO) {
            getServiceLocked()
        } ?: return null
        return block(service)
    }

    override fun <T> runWithServiceIfAvailable(block: (IPhoneNumberSetter) -> T?): T? {
        return serviceInstance?.let {
            block(it)
        }
    }

    override fun unbindServiceIfNeeded() {
        serviceConnection?.let {
            try {
                context.unbindService(it)
            }catch (e: IllegalArgumentException){
                //Already unregistered
            }
        }
    }

    private suspend fun getServiceLocked(): IPhoneNumberSetter? {
        return withTimeoutOrNull(SERVICE_TIMEOUT_MS) {
            suspendCancellableCoroutine { continuation ->
                CoroutineScope(Dispatchers.IO).launch {
                    getServiceMutex.lock()
                    serviceInstance?.let {
                        if (continuation.isActive) continuation.resume(it)
                        getServiceMutex.unlock()
                        return@launch
                    }

                    var isBound = false
                    val conn = object : ServiceConnection {
                        override fun onServiceConnected(component: ComponentName, binder: IBinder) {
                            serviceInstance = IPhoneNumberSetter.Stub.asInterface(binder)
                            binder.pingBinder()
                            serviceConnection = this
                            if (continuation.isActive) {
                                continuation.resume(serviceInstance!!)
                            }
                            getServiceMutex.unlock()
                        }

                        override fun onServiceDisconnected(component: ComponentName) {
                            serviceInstance = null
                            serviceConnection = null
                        }

                        override fun onNullBinding(name: ComponentName?) {
                            // In Android 13 without active Xposed hook, onBind returns null
                            if (continuation.isActive) {
                                continuation.resume(null)
                            }
                            try {
                                context.unbindService(this)
                            } catch (e: Exception) {}
                            getServiceMutex.unlock()
                        }
                    }

                    continuation.invokeOnCancellation {
                        try {
                            if (isBound) context.unbindService(conn)
                        } catch (e: Exception) {}
                        if (getServiceMutex.isLocked) {
                            getServiceMutex.unlock()
                        }
                    }

                    withContext(Dispatchers.Main) {
                        try {
                            isBound = context.bindService(serviceIntent, conn, Context.BIND_AUTO_CREATE)
                            if (!isBound) {
                                if (continuation.isActive) continuation.resume(null)
                                getServiceMutex.unlock()
                            }
                        } catch (t: Throwable) {
                            if (continuation.isActive) continuation.resume(null)
                            getServiceMutex.unlock()
                        }
                    }
                }
            }
        }
    }

}
```

```kt
        if (slotIndex == 0) return defaultPhone
        return try {
            PhoneFactory.getPhone(slotIndex) ?: defaultPhone
        } catch (t: Throwable) {
            defaultPhone
        }
    }

    // ORIGINAL CORE METHOD 1: 100% identical to v1.0
    override fun setLine1Number(number: String, onComplete: Message?): Boolean {
        var tag = defaultPhone.line1AlphaTag
        if(tag.isNullOrEmpty()){
            tag = "Voice Line 1"
        }
        return defaultPhone.setLine1Number(tag, number, onComplete)
    }

    // ORIGINAL CORE METHOD 2: 100% identical to v1.0
    override fun getLine1Number(): String {
        return defaultPhone.line1Number ?: ""
    }

    // UPGRADED METHOD 3: Multi-slot setter (falls back to defaultPhone if slot 0)
    override fun setLine1NumberForSlot(slotIndex: Int, number: String, onComplete: Message?): Boolean {
        if (slotIndex == 0) return setLine1Number(number, onComplete)
        val phone = getPhoneForSlot(slotIndex)
        var tag = try {
            phone.line1AlphaTag
        } catch (e: Exception) {
            null
        }
        if(tag.isNullOrEmpty()){
            tag = "Voice Line ${slotIndex + 1}"
        }
        return phone.setLine1Number(tag, number, onComplete)
    }

    // UPGRADED METHOD 4: Multi-slot getter
    override fun getLine1NumberForSlot(slotIndex: Int): String {
        if (slotIndex == 0) return getLine1Number()
        val phone = getPhoneForSlot(slotIndex)
        return try {
            phone.line1Number ?: ""
        } catch (e: Exception) {
            ""
        }
    }

    // UPGRADED METHOD 5: Phone count getter
    override fun getPhoneCount(): Int {
        return try {
            val phones = PhoneFactory.getPhones()
            phones?.size ?: 1
        } catch (e: Exception) {
            1
        }
    }
```

```kt
class MainViewModelImpl(

            else -> return

    private fun load() = viewModelScope.launch {
        state.emit(State.Loading(LoadType.LOADING))
        val rooted = try {
            rootRepository.isRooted()
        } catch (t: Throwable) {
            false
        }
        isRooted.emit(rooted)
        
        val slot = selectedSlot.value

        // Try primary path: LSPosed Xposed Hook via TelephonyDebugService
        var num: String? = null
        try {
            num = serviceRepository.runWithService { service ->
                isHookConnected.emit(true)
                try {
                    phoneCount.emit(service.phoneCount)
                } catch (t: Throwable) {
                    phoneCount.emit(2)
                }
                try {
                    service.getLine1NumberForSlot(slot)
                } catch (t: Throwable) {
                    service.line1Number
                }
            }
        } catch (e: Exception) {
            num = null
        }

        // Secondary fallback path: Direct Root Telephony Provider query
        if (num == null && rooted) {
            isHookConnected.emit(false)
            phoneCount.emit(2)
            num = rootRepository.getLine1NumberViaRoot(slot)
        }

        if (num == null) {
            isHookConnected.emit(false)
            state.emit(State.Error(ErrorType.NO_XPOSED))
            return@launch
        }
        state.emit(State.Loaded(num))
    }

    private fun save(number: String) = viewModelScope.launch {
        state.emit(State.Loading(LoadType.SAVING))
        val slot = selectedSlot.value
        val rooted = isRooted.value

        var success: Boolean? = null

        // Try primary path: LSPosed Hook
        try {
            success = serviceRepository.runWithService { service ->
                try {
                    service.setLine1NumberForSlot(slot, number, null)
                } catch (t: Throwable) {
                    service.setLine1Number(number, null)
                }
            }
        } catch (e: Exception) {
            success = null
        }

        // Secondary fallback path: Direct Root Telephony Content Provider Update
        if (success == null && rooted) {
            success = rootRepository.setLine1NumberViaRoot(slot, number)
            if (success) {
                // Kick telephony cache so changes take effect
                rootRepository.restartTelephony()
            }
        }

        if (success == null) {
            state.emit(State.Error(ErrorType.NO_XPOSED))
            return@launch
        }

        customNumber.emit(null)
        delay(1200L)
        if (success) {
            load()
        } else {
            state.emit(State.Error(ErrorType.SAVE_FAILED))
        }
    }

    init {
        load()
    }

    override fun onCleared() {
        serviceRepository.unbindServiceIfNeeded()
        super.onCleared()
    }

}
```

```kt
import kotlinx.coroutines.suspendCancellableCoroutine
import kotlinx.coroutines.withTimeoutOrNull
import kotlin.coroutines.Continuation

suspend inline fun <T> suspendCoroutineWithTimeout(timeout: Long, crossinline block: (Continuation<T>) -> Unit ) : T? {
    var finalValue : T? = null
    withTimeoutOrNull(timeout) {
        finalValue = suspendCancellableCoroutine(block = block)
    }
    return finalValue
```

```kt
package com.mayaxdev.simnumbersetter.utils.extensions

import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import com.mayaxdev.simnumbersetter.BuildConfig

private const val KEY_PENDING_INTENT = "SECURITY_PENDING_INTENT"
private const val PENDING_INTENT_REQUEST_CODE = 1001

fun Intent.applySecurity(context: Context) {
    putExtra(
        KEY_PENDING_INTENT, PendingIntent.getActivity(
        context,
        PENDING_INTENT_REQUEST_CODE,
        Intent(),
        PendingIntent.FLAG_IMMUTABLE
    ))
}

fun Intent.checkSecurity(): Boolean {
    return try {
        setExtrasClassLoader(PendingIntent::class.java.classLoader)
        val pendingIntent = getParcelableExtra<PendingIntent>(KEY_PENDING_INTENT) ?: return false
        pendingIntent.creatorPackage == BuildConfig.APPLICATION_ID
    } catch (t: Throwable) {
        false
    }
}
```

```java
package com.android.internal.telephony;

import android.os.Message;

public interface PhoneInternalInterface {

    /**
     * Returns the alpha tag associated with the msisdn number.
     * If there is no alpha tag associated or the record is not yet available,
     * returns a default localized string. <p>
     */
    String getLine1AlphaTag();

    /**
     * Sets the MSISDN phone number in the SIM card.
     *
     * @param alphaTag the alpha tag associated with the MSISDN phone number
     *        (see getMsisdnAlphaTag)
     * @param number the new MSISDN phone number to be set on the SIM.
     * @param onComplete a callback message when the action is completed.
     *
     * @return true if req is sent, false otherwise. If req is not sent there will be no response,
     * that is, onComplete will never be sent.
     */
    boolean setLine1Number(String alphaTag, String number, Message onComplete);

    /**
     * Get the line 1 phone number (MSISDN). For CDMA phones, the MDN is returned
     * and {#getMsisdn()} will return the MSISDN on CDMA/LTE phones.<p>
     *
     * @return phone number. May return null if not
     * available or the SIM is not ready
     */
    String getLine1Number();

}
```
