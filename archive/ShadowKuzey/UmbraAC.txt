```java
package dev.umbra.ac.checks.fly;

import dev.umbra.ac.UmbraAC;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerMoveEvent;

import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

public class FlyListener implements Listener {

    private final UmbraAC plugin;
    private final FlyManager manager;
    private final Map<UUID, Integer> violationLevels = new HashMap<>();

    public FlyListener(UmbraAC plugin, FlyManager manager) {
        this.plugin = plugin;
        this.manager = manager;
    }

    @EventHandler
    public void onMove(PlayerMoveEvent event) {
        if (!manager.isFlyEnabled()) return;

        Player player = event.getPlayer();
        if (manager.isExempt(player)) return;

        if (player.isOnGround()) {
            UUID uuid = player.getUniqueId();
            int level = violationLevels.getOrDefault(uuid, 0);
            if (level > 0) violationLevels.put(uuid, level - 1);
            return;
        }

        double deltaY = event.getTo().getY() - event.getFrom().getY();

        if (deltaY > 0.5) {
            UUID uuid = player.getUniqueId();
            int level = violationLevels.getOrDefault(uuid, 0) + 1;
            violationLevels.put(uuid, level);

            if (level >= 3) {
                plugin.getAlertManager().sendAlert(player, "Fly", "Havada yukselme (DeltaY: " + String.format("%.2f", deltaY) + ")");
                violationLevels.put(uuid, 0);
            }
        }
    }
}
```

```java
        return false;
```

```java
package dev.umbra.ac.checks.killaura;

import dev.umbra.ac.UmbraAC;
import org.bukkit.entity.Entity;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.entity.EntityDamageByEntityEvent;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

public class KillAuraListener implements Listener {

    private final UmbraAC plugin;
    private final KillAuraManager manager;
    private final Map<UUID, Deque<Long>> clickHistory = new HashMap<>();

    public KillAuraListener(UmbraAC plugin, KillAuraManager manager) {
        this.plugin = plugin;
        this.manager = manager;
    }

    @EventHandler
    public void onDamage(EntityDamageByEntityEvent event) {
        if (!manager.isKillAuraEnabled()) return;
        if (!(event.getDamager() instanceof Player player)) return;

        Entity target = event.getEntity();
        UUID uuid = player.getUniqueId();
        long now = System.currentTimeMillis();

        Deque<Long> history = clickHistory.computeIfAbsent(uuid, k -> new ArrayDeque<>());
        history.addLast(now);

        while (!history.isEmpty() && now - history.peekFirst() > 1000) {
            history.pollFirst();
        }

        if (history.size() > manager.getMaxCps()) {
            plugin.getAlertManager().sendAlert(player, "KillAura", "Cok yuksek CPS (Saniyede " + history.size() + " vurus)");
            history.clear();
            return;
        }

        org.bukkit.util.Vector lookDir = player.getEyeLocation().getDirection().normalize();
        org.bukkit.util.Vector toTarget = target.getLocation().add(0, target.getHeight() / 2, 0)
                .subtract(player.getEyeLocation()).toVector().normalize();

        double dot = lookDir.dot(toTarget);
        dot = Math.max(-1.0, Math.min(1.0, dot));
        double angle = Math.toDegrees(Math.acos(dot));

        if (angle > manager.getMaxAngle()) {
            plugin.getAlertManager().sendAlert(player, "KillAura", "Arkasina bakarak vurdu (Aci: " + String.format("%.1f", angle) + " derece)");
        }
    }
}
```

```java
    private final NoFallManager manager;

    public NoFallListener(UmbraAC plugin, NoFallManager manager) {
        this.plugin = plugin;
        this.manager = manager;
    }

    @EventHandler
    public void onDamage(EntityDamageEvent event) {
        if (!manager.isNoFallEnabled()) return;
        if (!(event.getEntity() instanceof Player player)) return;
        if (event.getCause() != EntityDamageEvent.DamageCause.FALL) return;

        double fallDistance = player.getFallDistance();

        if (fallDistance >= manager.getMinFallDistance() && event.getDamage() <= 0.5) {
            plugin.getAlertManager().sendAlert(player, "NoFall", "Dusus Mesafesi: " + String.format("%.2f", fallDistance) + " blok | Alinan Hasar: " + event.getDamage());
        }
    }
}
```

```java
package dev.umbra.ac.checks.xray;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.HashMap;
import java.util.Map;

public class PlayerData {

    private int suspicionPoints = 0;
    private int stoneMined = 0;
    private final Map<String, Integer> oreMined = new HashMap<>();
    private final Deque<AngleEvent> angleEvents = new ArrayDeque<>();
    private float lastYaw = 0f;

    public int getSuspicionPoints() { return suspicionPoints; }
    public void addSuspicionPoints(int points) { this.suspicionPoints += points; }
    public void clearSuspicionPoints() {
        this.suspicionPoints = 0;
    }
    public int getStoneMined() { return stoneMined; }
    public void incrementStone() { this.stoneMined++; }

    public Map<String, Integer> getOreMined() { return oreMined; }
    public void incrementOre(String type) {
        oreMined.merge(type, 1, Integer::sum);
    }

    public Deque<AngleEvent> getAngleEvents() { return angleEvents; }

    public void addAngleEvent(double angle, String oreType) {
        angleEvents.addLast(new AngleEvent(System.currentTimeMillis(), angle, oreType));
        pruneOldEvents();
    }

    private void pruneOldEvents() {
        long cutoff = System.currentTimeMillis() - 10 * 60 * 1000L;
        while (!angleEvents.isEmpty() && angleEvents.peekFirst().timestamp < cutoff) {
            angleEvents.pollFirst();
        }
    }

    public float getLastYaw() { return lastYaw; }
    public void setLastYaw(float lastYaw) { this.lastYaw = lastYaw; }

    public static class AngleEvent {
        public final long timestamp;
        public final double angle;
        public final String oreType;

        public AngleEvent(long timestamp, double angle, String oreType) {
            this.timestamp = timestamp;
            this.angle = angle;
            this.oreType = oreType;
        }
    }
}
```

```md
# UmbraAC

**UmbraAC** is a lightweight, modern, and modular anti-cheat plugin for Minecraft servers. It focuses on detecting the most common cheats without causing unnecessary lag, making it perfect for small to medium-sized communities.

## ✨ Features
- **X-Ray Detection:** Uses statistical and behavioral analysis to find players mining ores abnormally.
- **Fly Check:** Detects hovering and unauthorized vertical movement.
- **Reach Check:** Prevents players from attacking entities from impossible distances.
- **Speed Check:** Catches players moving faster than vanilla limits.
- **KillAura Check:** Detects multi-aura (attacking multiple targets), high CPS (Clicks Per Second), and impossible attack angles.
- **NoFall Check:** Identifies players who take no fall damage from high altitudes.
- **Interactive Alerts:** Staff members (with OP/admin permissions) receive a clickable `[BAN]` button in chat when a player is flagged. No need to constantly watch the console!

## 📥 Installation
1. Download the latest `.jar` file from the [Releases](https://github.com/yourusername/UmbraAC/releases) page.
2. Drop the `.jar` file into your server's `plugins/` folder.
3. Start or restart your server.
4. Configure the checks and thresholds in `plugins/UmbraAC/config.yml`.

## ⚙️ Configuration
UmbraAC comes with a fully customizable `config.yml`. You can enable/disable specific checks and adjust the sensitivity for each one.

## 🌍 Language Support
UmbraAC now supports multiple languages! You can change the plugin's language in the `config.yml` file (Default: Turkish).
Currently supported languages: English (`en`), Turkish (`tr`).

Thats All!
```

```java
package dev.umbra.ac.checks.speed;

import dev.umbra.ac.UmbraAC;
import org.bukkit.Location;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerMoveEvent;

import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

public class SpeedListener implements Listener {

    private final UmbraAC plugin;
    private final SpeedManager manager;
    private final Map<UUID, Integer> violationLevels = new HashMap<>();

    public SpeedListener(UmbraAC plugin, SpeedManager manager) {
        this.plugin = plugin;
        this.manager = manager;
    }

    @EventHandler
    public void onMove(PlayerMoveEvent event) {
        if (!manager.isSpeedEnabled()) return;

        Player player = event.getPlayer();
        if (manager.isExempt(player)) return;

        Location from = event.getFrom();
        Location to = event.getTo();

        double distance = Math.sqrt(Math.pow(to.getX() - from.getX(), 2) + Math.pow(to.getZ() - from.getZ(), 2));

        if (distance > 10.0) return;

        double maxSpeed = manager.getMaxSpeed(player);

        if (distance > maxSpeed) {
            UUID uuid = player.getUniqueId();
            int level = violationLevels.getOrDefault(uuid, 0) + 1;
            violationLevels.put(uuid, level);

            if (level >= 3) {
                plugin.getAlertManager().sendAlert(player, "Speed", "Hiz: " + String.format("%.2f", distance) + " blok/tick");
                violationLevels.put(uuid, 0);
            }
        } else {
            UUID uuid = player.getUniqueId();
            int level = violationLevels.getOrDefault(uuid, 0);
            if (level > 0) violationLevels.put(uuid, level - 1);
        }
    }
}
```

```java
package dev.umbra.ac.checks.speed;

import dev.umbra.ac.UmbraAC;
import org.bukkit.GameMode;
import org.bukkit.entity.Player;

public class SpeedManager {

    private final UmbraAC plugin;

    public SpeedManager(UmbraAC plugin) {
        this.plugin = plugin;
    }

    public boolean isSpeedEnabled() {
        return plugin.getConfig().getBoolean("speed.enabled", true);
    }

    // Oyuncunun hızlı olmasına izin verilen durumlar
    public boolean isExempt(Player player) {
        if (player.getGameMode() == GameMode.CREATIVE || player.getGameMode() == GameMode.SPECTATOR) return true;
        if (player.isFlying() || player.isGliding() || player.isInsideVehicle()) return true;
        // Buz, ruh kumu gibi hızlandıran bloklarda veya suda yüzerken kontrolü esnet
        if (player.getLocation().getBlock().isLiquid()) return true;

        return false;
    }

    // Oyuncunun sahip olduğu hız iksiri seviyesine göre limiti ayarla
    public double getMaxSpeed(Player player) {
        double baseSpeed = plugin.getConfig().getDouble("speed.max-speed", 0.45); // Saniyedeki blok cinsinden limit

        // Hız iksiri varsa limiti arttır (Seviye 1 = %20, Seviye 2 = %40 vb.)
        if (player.hasPotionEffect(org.bukkit.potion.PotionEffectType.SPEED)) {
            int amplifier = player.getPotionEffect(org.bukkit.potion.PotionEffectType.SPEED).getAmplifier();
            baseSpeed += (baseSpeed * 0.2 * (amplifier + 1));
        }

        return baseSpeed;
    }

    public void alert(String playerName, double distance) {
        plugin.getLogger().warning("[Speed Suphesi] " + playerName
                + " | Hiz: " + String.format("%.2f", distance) + " blok/tick");
    }
}
```

```java
package dev.umbra.ac.checks.xray;

import dev.umbra.ac.UmbraAC;
import org.bukkit.Location;
import org.bukkit.Material;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.block.BlockBreakEvent;
import org.bukkit.event.player.PlayerMoveEvent;

public class XRayListener implements Listener {

    private final UmbraAC plugin;
    private final XRayManager manager;

    public XRayListener(UmbraAC plugin, XRayManager manager) {
        this.plugin = plugin;
        this.manager = manager;
    }

    @EventHandler
    public void onMove(PlayerMoveEvent event) {
        if (!manager.isXRayEnabled()) return;
        Player player = event.getPlayer();
        PlayerData data = manager.getPlayerData(player.getUniqueId());
        data.setLastYaw(player.getLocation().getYaw());
    }

    @EventHandler
    public void onBlockBreak(BlockBreakEvent event) {
        if (!manager.isXRayEnabled()) return;

        Player player = event.getPlayer();
        Material type = event.getBlock().getType();
        PlayerData data = manager.getPlayerData(player.getUniqueId());

        if (isStone(type)) {
            data.incrementStone();
            return;
        }

        if (!isOre(type)) return;

        String oreName = type.name();
        data.incrementOre(oreName);

        // 1. Temel Puan
        int basePoints = manager.getOrePoints(oreName);
        data.addSuspicionPoints(basePoints);

        // 2. Yön Sapması Puanı
        double angle = calculateAngle(player, event.getBlock().getLocation());
        if (angle > manager.getAngleThreshold()) {
            data.addAngleEvent(angle, oreName);
            data.addSuspicionPoints(manager.getAngleBonus());
        }

        // 3. Oran Kontrolü
        checkRatio(data);

        // 4. Alarm Kontrolü (Merkezi AlertManager kullanarak)
        if (data.getSuspicionPoints() >= manager.getAlertThreshold()) {
            plugin.getAlertManager().sendAlert(player, "X-Ray",
                    "Puan: " + data.getSuspicionPoints() +
                            " | Elmas: " + data.getOreMined().getOrDefault("DIAMOND_ORE", 0) +
                            " | Tas: " + data.getStoneMined());

            // Aynı oyuncu için sürekli spam atmaması için puanı sıfırlayalım
            data.clearSuspicionPoints(); // Bunu PlayerData'ya ekleyeceğiz
        }
    }

    private void checkRatio(PlayerData data) {
        int totalOre = 0;
        for (int count : data.getOreMined().values()) {
            totalOre += count;
        }

        int totalStone = data.getStoneMined();
        if (totalStone < 20) return;

        double ratio = (double) totalOre / totalStone;
        if (ratio > manager.getRatioThreshold()) {
            data.addSuspicionPoints(manager.getRatioBonus());
        }
    }

    private double calculateAngle(Player player, Location oreLocation) {
        Location eye = player.getEyeLocation();
        org.bukkit.util.Vector lookDir = eye.getDirection().normalize();
        org.bukkit.util.Vector toOre = oreLocation.clone().add(0.5, 0.5, 0.5)
                .subtract(eye).toVector().normalize();
        double dot = lookDir.dot(toOre);
        dot = Math.max(-1.0, Math.min(1.0, dot));
        return Math.toDegrees(Math.acos(dot));
    }

    private boolean isStone(Material type) {
        return type == Material.STONE || type == Material.DEEPSLATE
                || type == Material.NETHERRACK || type == Material.TUFF;
    }

    private boolean isOre(Material type) {
        return type.name().endsWith("_ORE") || type == Material.ANCIENT_DEBRIS;
    }
}
```

```java
package dev.umbra.ac.checks.xray;

import dev.umbra.ac.UmbraAC;

    public PlayerData getPlayerData(UUID uuid) {
        return playerDataMap.computeIfAbsent(uuid, k -> new PlayerData());
    }

    public boolean isXRayEnabled() {
        return plugin.getConfig().getBoolean("xray.enabled", true);
    }

    public double getAngleThreshold() {
        return plugin.getConfig().getDouble("xray.angle-threshold", 47.0);

    }

    }

        return plugin.getConfig().getInt("xray.ratio-bonus", 10);
    }

    public double getRatioThreshold() {
        return plugin.getConfig().getDouble("xray.ratio-threshold", 0.15);
    }

    public int getOrePoints(String materialName) {
        ConfigurationSection section = plugin.getConfig().getConfigurationSection("xray.ore-points");
        if (section != null && section.contains(materialName)) {
            return section.getInt(materialName);
        }
        return 0;
    }
```
