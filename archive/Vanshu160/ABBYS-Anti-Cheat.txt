`README.md`:

```md
# ABBYS Anti-Cheat

Admin generates a 6-digit PIN on a dashboard, the player enters it in the Android app, the app
scans the device and sends the result back to the admin.

## Deploy on Render

1. Render dashboard -> New -> Blueprint -> select this repo. `render.yaml` creates a free web service named `abbys-anticheat`.
2. Open the service URL. No admin key is needed: anyone can generate a PIN, and only the browser tab that created it can read its report.
3. Free tier sleeps after inactivity (first request may take ~30-50s) and sessions are in memory, so a restart clears pending PINs.

## Server (`server/`)

```sh
cd server
npm install
npm start
```

- Serve it over HTTPS (reverse proxy such as Caddy or nginx). Set `TRUST_PROXY=1` when behind one,
  so rate limits use the real client address.
- PINs are random, valid for 30 minutes, usable once, and rate limited.
- Anyone can create a PIN (limited to 5 open PINs and 10 requests per minute per address). Creating one returns a secret session token that stays in that browser tab and is required to read the report, so guessing PINs does not expose reports. Closing the tab loses access to that PIN's report.
- Data is held in memory only and is deleted after the PIN expires.

## Android app (`app/`)

`Config.SERVER_URL` in `ReportClient.kt` points at `https://abbys-anticheat.onrender.com`; change it if your Render service has a different name. The app asks for
consent text up front, verifies the PIN with the server, scans, then uploads the report.

Severity policy: HIGH is reserved for strong native signals (root files, hooks, debugger). App-name
matches are WARN and say "name match only". Common innocent things (sideloaded apps, accessibility
services, ADB) are INFO. Results are indicators for an admin to review, not proof.

#include <jni.h>
#include <string>
#include <vector>
#include <fstream>
#include <sstream>
#include <unistd.h>

// Helper to check if a file or directory exists at the Linux level
bool fileExists(const char* path) {
    return access(path, F_OK) == 0;
}

// Helper to scan a /proc file for suspicious keywords
bool scanProcFile(const char* procPath, const std::vector<std::string>& keywords) {
    std::ifstream file(procPath);
    if (!file.is_open()) return false;

    std::string line;
    while (std::getline(file, line)) {
        for (const auto& kw : keywords) {
            if (line.find(kw) != std::string::npos) {
                return true;
            }
        }
    }
    return false;
}

// Check if a debugger is attached via TracerPid in /proc/self/status
bool isDebuggerAttached() {
    std::ifstream statusFile("/proc/self/status");
    std::string line;
    while (std::getline(statusFile, line)) {
        if (line.rfind("TracerPid:", 0) == 0) {
            int pid = std::stoi(line.substr(10));
            return pid > 0;
        }
    }
    return false;
}

extern "C" JNIEXPORT jstring JNICALL
Java_com_example_mobilescanner_NativeBridge_runNativeChecks(
        JNIEnv* env,
        jobject /* this */) {

    std::vector<std::string> flags;

    // 1. Check Root / Magisk / KernelSU binaries
    const char* rootPaths[] = {
        "/system/bin/su",
        "/system/xbin/su",
        "/sbin/su",
        "/data/adb/magisk",
        "/data/adb/ksu",
        "/data/adb/ap"
    };
    for (const char* path : rootPaths) {
        if (fileExists(path)) {
            flags.push_back("HIGH|Root or Magisk Binary Found|" + std::string(path));
        }
    }

    // 2. Check /proc/self/mounts for Zygisk / Magisk / Shamiko mounts
    if (scanProcFile("/proc/self/mounts", {"magisk", "zygisk", "kernelsu", "apatch"})) {
        flags.push_back("HIGH|Root Mount Detected|/proc/self/mounts contains Magisk/Zygisk");
    }

    // 3. Check /proc/self/maps for Frida / Xposed / LSPosed injection
    if (scanProcFile("/proc/self/maps", {"frida", "xposed", "lsposed", "substrate"})) {
        flags.push_back("HIGH|Memory Hook Detected|Frida or LSPosed injected into scanner");
    }

    // 4. Check TracerPid for active debugger
    if (isDebuggerAttached()) {
        flags.push_back("HIGH|Debugger Attached|TracerPid is non-zero in /proc/self/status");
    }

    if (flags.empty()) {
        return env->NewStringUTF("CLEAN");
    }

    std::ostringstream joined;
    for (size_t i = 0; i < flags.size(); ++i) {
        if (i > 0) joined << ";";
        joined << flags[i];
    }
    return env->NewStringUTF(joined.str().c_str());
}

package com.example.mobilescanner

import android.accessibilityservice.AccessibilityServiceInfo
import android.app.usage.UsageStatsManager
import android.content.Context
import android.content.pm.ApplicationInfo
import android.net.Uri
import android.os.Build
import android.provider.Settings
import android.view.accessibility.AccessibilityManager
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

data class ScanFinding(
    val severity: String,
    val title: String,
    val detail: String
)

/**
 * Severity policy: HIGH is reserved for strong, hard-to-explain signals (the native root/hook/debugger
 * checks, or a mod archive containing a known cheat client's code). A name match on an app or file is only WARN and says so. Things that are common and
 * innocent (sideloaded apps, accessibility services, ADB) are INFO for the admin's context.
 */
object CheatScanner {

    /** Distinctive enough to match inside the lowercase letters-only form of package + label. */
    private val DISTINCTIVE = listOf("gameguardian", "luckypatcher", "lsposed", "kernelsu", "aimbot", "wallhack", "autoclicker")

    /** Short words match only as whole words, never as substrings ("esp" must not hit "espn"). */
    private val WHOLE_WORDS = setOf("xposed", "magisk", "injector", "modmenu")

    private val TRUSTED_INSTALLERS = setOf(
        "com.android.vending", "com.sec.android.app.samsungapps", "com.amazon.venezia", "org.fdroid.fdroid"
    )

    fun runFullScan(context: Context, safTree: Uri? = null): List<ScanFinding> {
        val findings = mutableListOf<ScanFinding>()

        findings.addAll(NativeBridge.getNativeFindings())
        findings.addAll(checkInstalledPackages(context))
        findings.addAll(checkAccessibility(context))
        findings.addAll(checkRecentUsage(context))
        val storage = StorageScanner.scan(context)
        findings.addAll(storage.findings)
        findings.addAll(MinecraftScanner.scan(context, storage.gameDirs, safTree))
        findings.addAll(SafFolderScanner.scan(context, safTree))
        findings.addAll(checkDeveloperSettings(context))

        return capFindings(findings)
    }

    /** The server rejects reports over 200 findings, so keep the most severe ones. */
    private fun capFindings(all: List<ScanFinding>, max: Int = 190): List<ScanFinding> {
        if (all.size <= max) return all
        val order = listOf("HIGH", "WARN", "INFO", "CLEAN")
        val sorted = all.sortedBy { val i = order.indexOf(it.severity); if (i < 0) 2 else i }
        val kept = sorted.take(max - 1)
        return kept + ScanFinding("INFO", "Report truncated", "${all.size - kept.size} lower-priority findings were omitted to fit the report limit.")
    }

    private fun matchesSuspiciousName(pkgName: String, label: String): Boolean {
        val letters = (pkgName + label).lowercase().filter { it.isLetter() }
        if (DISTINCTIVE.any { letters.contains(it) }) return true
        val words = (pkgName + "." + label).lowercase().split(Regex("[^a-z0-9]+")).filter { it.isNotEmpty() }
        return words.any { it in WHOLE_WORDS }
    }

    private fun checkInstalledPackages(context: Context): List<ScanFinding> {
        val results = mutableListOf<ScanFinding>()
        val pm = context.packageManager
        val packages = pm.getInstalledPackages(0)

        for (pkg in packages) {
            val pkgName = pkg.packageName.lowercase()
            val appLabel = pkg.applicationInfo?.loadLabel(pm)?.toString() ?: pkgName

            if (matchesSuspiciousName(pkgName, appLabel)) {
                results.add(
                    ScanFinding(
                        "WARN", "App name resembles a known cheat/root tool",
                        "$appLabel ($pkgName). Name match only; verify the app before drawing conclusions."
                    )
                )
            }

            val isSystem = ((pkg.applicationInfo?.flags ?: 0) and ApplicationInfo.FLAG_SYSTEM) != 0
            if (!isSystem && pkgName != context.packageName) {
                val installer = try {
                    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                        pm.getInstallSourceInfo(pkg.packageName).installingPackageName
                    } else {
                        @Suppress("DEPRECATION")
                        pm.getInstallerPackageName(pkg.packageName)
                    }
                } catch (e: Exception) {
                    null
                }

                if (installer !in TRUSTED_INSTALLERS) {
                    results.add(
                        ScanFinding("INFO", "App not from a major store", "$appLabel (installer: ${installer ?: "none / APK file"})")
                    )
                }
            }
        }
        return results
    }

    private fun checkAccessibility(context: Context): List<ScanFinding> {
        val results = mutableListOf<ScanFinding>()
        val am = context.getSystemService(Context.ACCESSIBILITY_SERVICE) as AccessibilityManager
        val enabledServices = am.getEnabledAccessibilityServiceList(AccessibilityServiceInfo.FEEDBACK_ALL_MASK)
        for (service in enabledServices) {
            results.add(
                ScanFinding(
                    "INFO", "Accessibility service enabled",
                    "${service.id ?: "unknown"} (often assistive technology such as a screen reader; not suspicious by itself)"
                )
            )
        }
        return results
    }

    private fun checkRecentUsage(context: Context): List<ScanFinding> {
        val results = mutableListOf<ScanFinding>()
        if (!hasUsageStatsPermission(context)) {
            results.add(ScanFinding("INFO", "Recent-app check skipped", "Usage Access was not granted, so this part of the scan is incomplete."))
            return results
        }

        val usm = context.getSystemService(Context.USAGE_STATS_SERVICE) as UsageStatsManager
        val endTime = System.currentTimeMillis()
        val startTime = endTime - (15 * 60 * 1000)

        val stats = usm.queryUsageStats(UsageStatsManager.INTERVAL_DAILY, startTime, endTime)
        val formatter = SimpleDateFormat("HH:mm:ss", Locale.getDefault())

        val recentApps = stats
            .filter { it.lastTimeUsed >= startTime && it.packageName != context.packageName }
            .sortedByDescending { it.lastTimeUsed }
            .take(8)

        for (app in recentApps) {
            val timeStr = formatter.format(Date(app.lastTimeUsed))
            results.add(ScanFinding("INFO", "Recent App ($timeStr)", app.packageName))
        }
        return results
    }

    private fun checkDeveloperSettings(context: Context): List<ScanFinding> {
        val results = mutableListOf<ScanFinding>()
        val adbEnabled = Settings.Global.getInt(context.contentResolver, Settings.Global.ADB_ENABLED, 0)
        if (adbEnabled == 1) {
            results.add(ScanFinding("INFO", "USB debugging (ADB) enabled", "Common on developer and hobbyist devices; not suspicious by itself."))
        }
        return results
    }
}

enum class ScanStep {
    ENTER_PIN,
    GRANT_PERMISSIONS,
    SCANNING,
    COMPLETED
}

@Composable
fun ScannerApp() {
    var currentStep by remember { mutableStateOf(ScanStep.ENTER_PIN) }
    var pinCode by remember { mutableStateOf("") }
    var scanProgress by remember { mutableFloatStateOf(0f) }
    var scanStatusText by remember { mutableStateOf("Initializing...") }
    var scanResults by remember { mutableStateOf<List<ScanFinding>>(emptyList()) }
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    var verifying by remember { mutableStateOf(false) }
    var pinError by remember { mutableStateOf<String?>(null) }
    var uploadStatus by remember { mutableStateOf("") }
    var permTick by remember { mutableIntStateOf(0) }
    var safTree by remember { mutableStateOf<Uri?>(null) }

    // Re-check permissions every time the player comes back from a settings screen.
    val lifecycleOwner = LocalLifecycleOwner.current
    DisposableEffect(lifecycleOwner) {
        val observer = LifecycleEventObserver { _, event ->
            if (event == Lifecycle.Event.ON_RESUME) permTick++
        }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose { lifecycleOwner.lifecycle.removeObserver(observer) }
    }
    val usageOk = remember(permTick) { hasUsageStatsPermission(context) }
    val filesOk = remember(permTick) { hasAllFilesPermission(context) }
    val treeLauncher = rememberLauncherForActivityResult(ActivityResultContracts.OpenDocumentTree()) { uri -> safTree = uri }
    val storagePermLauncher = rememberLauncherForActivityResult(ActivityResultContracts.RequestPermission()) { permTick++ }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(20.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = if (currentStep == ScanStep.COMPLETED) Arrangement.Top else Arrangement.Center
    ) {
        Text(
            text = "ABBYS ANTI-CHEAT",
            fontSize = 24.sp,
            fontWeight = FontWeight.Bold,
            color = MaterialTheme.colorScheme.primary
        )
        Spacer(modifier = Modifier.height(24.dp))

        when (currentStep) {
            ScanStep.ENTER_PIN -> {
                Text("Enter the 6-digit PIN provided by the admin:")
                Spacer(modifier = Modifier.height(8.dp))
                Text(
                    "This scan reads: names of installed apps, enabled accessibility services, apps used in the " +
                        "last 15 minutes, whether USB debugging is on, and basic root/debugger indicators. With All " +
                        "Files Access it also lists file and folder names (with last-modified times) across your storage and trash, " +
                        "and, if you pick one, the folder you choose. It also looks " +
                        "inside Minecraft Java and Bedrock folders (mod file lists, configs, resource packs and recent game logs). It does not read " +
                        "messages, photos, launcher login data or game worlds. " +
                        "The result is sent to the admin who gave you this PIN.",
                    fontSize = 12.sp,
                    color = Color.LightGray
                )
                Spacer(modifier = Modifier.height(16.dp))

                OutlinedTextField(
                    value = pinCode,
                    onValueChange = { if (it.length <= 6 && it.all { c -> c.isDigit() }) pinCode = it },
                    label = { Text("Session PIN") },
                    singleLine = true,
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                    modifier = Modifier.fillMaxWidth()
                )
                Spacer(modifier = Modifier.height(24.dp))

                pinError?.let {
                    Text(it, color = Color(0xFFD32F2F), fontSize = 13.sp)
                    Spacer(modifier = Modifier.height(8.dp))
                }
                Button(
                    onClick = {
                        verifying = true
                        pinError = null
                        scope.launch {
                            val ok = withContext(Dispatchers.IO) { ReportClient.verifyPin(pinCode) }
                            verifying = false
                            if (ok) currentStep = ScanStep.GRANT_PERMISSIONS
                            else pinError = "PIN not accepted. It may be wrong, already used, expired, or the server is unreachable."
                        }
                    },
                    enabled = pinCode.length == 6 && !verifying,
                    modifier = Modifier.fillMaxWidth().height(50.dp)
                ) {
                    Text(if (verifying) "Checking..." else "Verify PIN")
                }
            }

            ScanStep.GRANT_PERMISSIONS -> {
                Text("Required Permissions", fontSize = 18.sp, fontWeight = FontWeight.SemiBold)
                Spacer(modifier = Modifier.height(16.dp))

                PermissionRow("Usage Access", usageOk) {
                    context.startActivity(Intent(Settings.ACTION_USAGE_ACCESS_SETTINGS))
                }
                Spacer(modifier = Modifier.height(12.dp))
                PermissionRow("All Files Access", filesOk) { requestAllFiles(context, storagePermLauncher) }
                Spacer(modifier = Modifier.height(20.dp))

                OutlinedButton(
                    onClick = { treeLauncher.launch(androidDataHint()) },
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Text(
                        if (safTree == null) "3. Select Launcher / Minecraft Folder (Android/data)"
                        else "Folder selected: ${folderLabel(safTree!!)} (tap to change)"
                    )
                }
                Text(
                    "Optional. Pick the launcher's .minecraft folder or the com.mojang folder (Bedrock). Mods, configs, logs and " +
                        "packs inside are checked, plus recently changed .so, .cfg, .lua and .pak files. Android 11+ may refuse " +
                        "Android/data itself; folders under /games usually work.",
                    fontSize = 11.sp,
                    color = Color.LightGray
                )
                Spacer(modifier = Modifier.height(24.dp))

                Button(
                    onClick = { currentStep = ScanStep.SCANNING },
                    enabled = usageOk && filesOk,
                    modifier = Modifier.fillMaxWidth().height(50.dp),
                    colors = ButtonDefaults.buttonColors(containerColor = Color(0xFF2E7D32))
                ) {
                    Text("Start Scan", color = Color.White)
                }
                if (!(usageOk && filesOk)) {
                    Spacer(modifier = Modifier.height(8.dp))
                    Text("Grant both permissions to start the scan.", fontSize = 12.sp, color = Color(0xFFEF9A9A))
                }
            }

            ScanStep.SCANNING -> {
                LaunchedEffect(Unit) {
                    scanStatusText = "Scanning packages & overlays..."
                    scanProgress = 0.35f
                    delay(600)

                    scanStatusText = "Checking recent apps, storage, trash and Minecraft folders (can take a minute)..."
                    scanProgress = 0.70f
                    val results = withContext(Dispatchers.IO) {
                        CheatScanner.runFullScan(context, safTree)
                    }

                    scanStatusText = "Finalizing report..."
                    scanProgress = 1.0f
                    delay(400)

                    scanResults = results
                    scanStatusText = "Sending report to admin..."
                    val sent = withContext(Dispatchers.IO) { ReportClient.submit(pinCode, results) }
                    uploadStatus = if (sent) "Report sent to the admin."
                    else "Could not send the report. Show this screen to the admin."
                    currentStep = ScanStep.COMPLETED
                }

                Text(scanStatusText, fontSize = 16.sp)
                Spacer(modifier = Modifier.height(16.dp))
                LinearProgressIndicator(
                    progress = { scanProgress },
                    modifier = Modifier.fillMaxWidth().height(8.dp)
                )
            }

            ScanStep.COMPLETED -> {
                Text(
                    "Scan Complete (PIN: $pinCode)",
                    fontSize = 18.sp,
                    fontWeight = FontWeight.Bold,
                    color = Color(0xFF4CAF50)
                )
                Text(uploadStatus, fontSize = 12.sp, color = Color.LightGray)
                Spacer(modifier = Modifier.height(12.dp))

                LazyColumn(
                    modifier = Modifier
                        .weight(1f)
                        .fillMaxWidth(),
                    verticalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    items(scanResults) { item ->
                        val badgeColor = when (item.severity) {
                            "HIGH" -> Color(0xFFD32F2F)
                            "WARN" -> Color(0xFFF57C00)
                            "CLEAN" -> Color(0xFF388E3C)
                            else -> Color(0xFF1976D2)
                        }
                        Column(
                            modifier = Modifier
                                .fillMaxWidth()
                                .background(Color(0xFF1E1E1E), RoundedCornerShape(8.dp))
                                .padding(12.dp)
                        ) {
                            Text(
                                text = "[${item.severity}] ${item.title}",
                                color = badgeColor,
                                fontWeight = FontWeight.Bold,
                                fontSize = 14.sp
                            )
                            Spacer(modifier = Modifier.height(4.dp))
                            Text(
                                text = item.detail,
                                color = Color.LightGray,
                                fontSize = 12.sp
                            )

/** One file or folder, either a real file or a document inside a folder tree the player picked. */
internal interface McEntry {
    val name: String
    val path: String
    val isDir: Boolean
    val modified: Long
    val size: Long
    fun children(): List<McEntry>?
    fun readTail(maxBytes: Int): ByteArray?

private fun isLink(f: File): Boolean = try {
    Files.isSymbolicLink(f.toPath())
} catch (e: Exception) {
    false
}

private class FileEntry(private val f: File) : McEntry {
    override val name: String get() = f.name
    override val path: String get() = f.path
    override val isDir: Boolean get() = f.isDirectory
    override val modified: Long get() = f.lastModified()
    override val size: Long get() = f.length()

    private var loaded = false
    private var kids: List<McEntry>? = null

    override fun children(): List<McEntry>? {
        if (!loaded) {
            loaded = true
            kids = try {
                f.listFiles()?.filter { !isLink(it) }?.map { FileEntry(it) }
            } catch (e: SecurityException) {
                null
            }
        }
        return kids
    }

    override fun readTail(maxBytes: Int): ByteArray? = try {
        RandomAccessFile(f, "r").use { raf ->
            val len = raf.length()
            val n = minOf(len, maxBytes.toLong()).toInt()
            raf.seek(len - n)
            val b = ByteArray(n)
            raf.readFully(b)
            b
        }
    } catch (e: Exception) {
        null
    }

    override fun asFile(): File? = f
}

private class SafEntry(
    private val context: Context,
    private val tree: Uri,
    private val docId: String,
    override val name: String,
    override val isDir: Boolean,
    override val modified: Long,
    override val size: Long,
    override val path: String
) : McEntry {
    private var loaded = false
    private var kids: List<McEntry>? = null

    override fun children(): List<McEntry>? {
        if (!loaded) {
            loaded = true
            kids = query()
        }
        return kids
    }

    private fun query(): List<McEntry>? {
        val uri = DocumentsContract.buildChildDocumentsUriUsingTree(tree, docId)
        val cursor = try {
            context.contentResolver.query(
                uri,
                arrayOf(
                    DocumentsContract.Document.COLUMN_DOCUMENT_ID,
                    DocumentsContract.Document.COLUMN_DISPLAY_NAME,
                    DocumentsContract.Document.COLUMN_MIME_TYPE,
                    DocumentsContract.Document.COLUMN_LAST_MODIFIED,
                    DocumentsContract.Document.COLUMN_SIZE
                ),
                null, null, null
            )
        } catch (e: Exception) {
            null
        }
        if (cursor == null) return null
        val out = ArrayList<McEntry>()
        cursor.use { c ->
            while (c.moveToNext()) {
                val id = c.getString(0) ?: continue
                val n = c.getString(1) ?: continue
                val dir = c.getString(2) == DocumentsContract.Document.MIME_TYPE_DIR
                val mod = if (c.isNull(3)) 0L else c.getLong(3)
                val sz = if (c.isNull(4)) 0L else c.getLong(4)
                out.add(SafEntry(context, tree, id, n, dir, mod, sz, "$path/$n"))
            }
        }
        return out
    }

    override fun readTail(maxBytes: Int): ByteArray? = try {
        val uri = DocumentsContract.buildDocumentUriUsingTree(tree, docId)
        context.contentResolver.openInputStream(uri)?.use { ins ->
            var toSkip = if (size > maxBytes) size - maxBytes else 0L
            while (toSkip > 0) {
                val s = ins.skip(toSkip)
                if (s <= 0) break
                toSkip -= s
            }
            val buf = ByteArrayOutputStream()
            val tmp = ByteArray(8192)
            while (buf.size() < 4 * 1024 * 1024) {
                val n = ins.read(tmp)
                if (n < 0) break
                buf.write(tmp, 0, n)
            }
            val all = buf.toByteArray()
            if (all.size > maxBytes) all.copyOfRange(all.size - maxBytes, all.size) else all
        }
    } catch (e: Exception) {
        null
    }

    override fun asFile(): File? = null
}

    fun scan(context: Context, storageGameDirs: List<File>, safTree: Uri?): List<ScanFinding> {
        val sink = McSink()
        val run = Run(sink, System.currentTimeMillis() + TIME_BUDGET_MS)
        val installed = packageChecks(context, sink)

        val fsOk = hasAllFilesPermission(context)
        if (fsOk) {
            val base = storageBase()
            val candidates = LinkedHashMap<String, File>()
            val add = { f: File -> if (f.isDirectory) candidates.putIfAbsent(canon(f), f) }

            for ((pkg, label) in JAVA_LAUNCHERS) {
                if (pkg in installed) probeData(File(base, "Android/data/$pkg/files"), pkg, label, sink, add)

            if (BEDROCK_PKG in installed) {

        if (sink.overflow > 0) {
            out.add(ScanFinding("INFO", "More Minecraft findings omitted", "${sink.overflow} further matches were left out to keep the report a reasonable size."))

    private fun packageChecks(context: Context, sink: FindingSink): Set<String> {

    private fun locate(entry: McEntry, depth: Int, maxDepth: Int, run: Run) {
        if (run.expired()) return
        run.visits++
        val kids = entry.children() ?: return
        val dirs = kids.filter { it.isDir }
        val names = dirs.map { it.name.lowercase() }.toSet()
        val javaLike = entry.name.equals(".minecraft", true) || "mods" in names ||
            ("versions" in names && names.any { it in GAME_SUBDIRS })
        val bedrockLike = entry.name.equals("com.mojang", true) || "resource_packs" in names ||
            "behavior_packs" in names || "minecraftpe" in names
        if (javaLike || bedrockLike) {
            if (run.analyzed.add(entry.path)) {
                if (javaLike) {
                    run.javaDirs++
                    analyzeJava(entry, kids, run)
                }
                if (bedrockLike) {
                    run.bedrockDirs++
                    analyzeBedrock(entry, kids, run)
                }
            }
            return
        }
        if (depth >= maxDepth) return
        for (d in dirs) {
            if (run.expired()) return
            locate(d, depth + 1, maxDepth, run)
        }
    }

    private fun scanTree(context: Context, tree: Uri, run: Run) {
        val sink = run.sink
        val rootId = try {
            DocumentsContract.getTreeDocumentId(tree)
        } catch (e: Exception) {
            sink.flag("saf:bad", ScanFinding("INFO", "Selected Minecraft folder could not be used", "Android did not give usable access to that folder."))
            return
        }
        // A folder on internal storage gets the same path as the real file system, so it is not analyzed twice.
        val label = if (rootId.startsWith("primary:")) File(storageBase(), rootId.removePrefix("primary:")).path else rootId
        val root = SafEntry(context, tree, rootId, label.substringAfterLast('/'), true, 0L, 0L, label)
        val before = run.analyzed.size
        locate(root, 0, MAX_SAF_DEPTH, run)
        if (run.analyzed.size == before) {
            sink.flag(
                "saf:none",
                ScanFinding(
                    "INFO", "No Minecraft folder found in the selected folder",
                    "Searched $label to a depth of $MAX_SAF_DEPTH (or it was already scanned). Pick the .minecraft folder, or the com.mojang folder for Bedrock."
                )
            )
        }
    }

                } else {
                    sink.flag(
                        "mj:" + j.path,
                        ScanFinding(
                            "INFO", "Mod file name contains a word some cheat clients use",
                            "${j.path} matched \"${hit.term}\". Modified ${ts(j.modified)}. Weak match: it is also an ordinary word (for example a harmless library)."
                        )
                    )
                }
            }
            j.asFile()?.let { inspectJar(it, sink, ev, modKeys) }
            if (hit == null) {
                val disp = j.name.removeSuffix(".disabled").removeSuffix(".jar")
                if (KNOWN_OK.any { disp.lowercase().contains(it) }) known.add(disp) else others.add(disp)
            }
        }
        if (mods != null) {
            sink.flag(
                "mf:" + game.path,
                ScanFinding("INFO", "Minecraft mods folder", "${mods.path}: ${jars.size} mod file(s); newest modified ${ts(newest)}.")
            )
            emitChunks(sink, "ml:" + game.path, "Known PvP/performance mods (check the server allows them)", mods.path, known)
            emitChunks(sink, "mo:" + game.path, "Other installed mods", mods.path, others)
        }

        // Versions
        val versions = sub(kids, "versions")?.children()?.filter { it.isDir }
        if (versions != null) {
            for (v in versions.take(200)) {
                if (run.expired()) break
                hackMatch(v.name)?.let { h ->
                    ev.add(h.key, "NAME")
                    sink.flag(
                        "vn:" + v.path,
                        ScanFinding(
                            if (h.strong) "WARN" else "INFO", "Game version name resembles a cheat client",
                            "${v.path} matched \"${h.term}\". Modified ${ts(v.modified)}. Name match only."
                        )
                    )
                }
                for (f in v.children() ?: emptyList()) {
                    if (f.isDir) continue
                    val lowerName = f.name.lowercase()
                    if (lowerName.endsWith(".json")) {
                        hackMatch(f.name)?.let { h -> ev.add(h.key, "NAME") }
                        val text = f.readTail(262_144)?.toString(Charsets.UTF_8)?.lowercase()
                        if (text != null) {
                            val marker = JSON_MARKERS.firstOrNull { text.contains(it.first) }
                            if (marker != null) {
                                ev.add(marker.second, "ARTIFACT")
                                sink.flag(
                                    "vj:" + f.path,
                                    ScanFinding(
                                        "WARN", "Version profile references cheat client code",
                                        "${f.path} mentions \"${marker.first}\". Modified ${ts(f.modified)}. Based on the file's text, not just its name."
                                    )
                                )
                            }
                            if (text.contains("-javaagent:")) {
                                sink.flag(
                                    "va:" + f.path,
                                    ScanFinding(
                                        "WARN", "Version profile starts a Java agent",
                                        "${f.path} contains -javaagent. Agents can be legitimate (authentication or profiling) but can also inject cheats."
                                    )
                                )
                            }
                        }
                    } else if (lowerName.endsWith(".jar")) {
                        f.asFile()?.let { inspectJar(it, sink, ev, modKeys) }
                    }
                }
            }
        }

        // Leftover configs of deleted clients
        sub(kids, "config")?.children()?.forEach { c ->
            val h = hackMatch(c.name.substringBeforeLast('.', c.name)) ?: return@forEach
            ev.add(h.key, "CONFIG")
            if (h.key !in modKeys) {
                sink.flag(
                    "cf:" + c.path,
                    ScanFinding(
                        if (h.strong) "WARN" else "INFO", "Config for a cheat client with no matching mod file",
                        "${c.path} matched \"${h.term}\". Modified ${ts(c.modified)}. The client may have been deleted; config files often outlast it. Name match only."
                    )
                )
            }
        }

        // Resource packs
        sub(kids, "resourcepacks")?.children()?.forEach { p ->
            if (hackMatch(p.name)?.key == "xray" || bedrockMatch(p.name) == "X-Ray") {
                ev.add("xray", "NAME")
                sink.flag(
                    "rp:" + p.path,
                    ScanFinding("WARN", "Resource pack name suggests X-ray", "${p.path}. Modified ${ts(p.modified)}. Name match only; some servers allow X-ray packs.")
                )
            }
        }

        // Logs
        val logFiles = ArrayList<McEntry>()
        sub(kids, "logs")?.children()?.let { lk ->
            lk.firstOrNull { !it.isDir && it.name.equals("latest.log", true) }?.let { logFiles.add(it) }
            lk.filter {
                !it.isDir && it.name.endsWith(".log", true) && !it.name.equals("latest.log", true) && now - it.modified <= RECENT_MS
            }.sortedByDescending { it.modified }.take(MAX_RECENT_LOGS).forEach { logFiles.add(it) }
        }
        kids.firstOrNull { !it.isDir && it.name.equals("latestlog.txt", true) }?.let { logFiles.add(it) }
        for (lf in logFiles) {
            if (run.expired()) break
            scanLog(lf, game.path, sink, ev)
        }

        emitCorrelation(game.path, ev, sink)
    }

    private fun collectJars(dir: McEntry, depth: Int, out: MutableList<McEntry>) {
        val kids = dir.children() ?: return
        for (k in kids) {
            if (out.size >= 1000) return
            val n = k.name.lowercase()
            if (k.isDir) {
                if (depth < 1) collectJars(k, depth + 1, out)
            } else if (n.endsWith(".jar") || n.endsWith(".jar.disabled")) {
                out.add(k)
            }
        }
    }

    /** Reads only the archive's file list (never inflates entries), so zip bombs cannot hurt. */
    private fun inspectJar(jar: File, sink: FindingSink, ev: Evidence, modKeys: MutableSet<String>) {
        if (jar.length() > MAX_JAR_BYTES) return
        try {
            ZipFile(jar).use { zip ->
                var n = 0
                val entries = zip.entries()
                while (entries.hasMoreElements()) {
                    if (++n > MAX_JAR_ENTRIES) return@use
                    val entry = entries.nextElement().name.lowercase()
                    val hit = CLASS_MARKERS.firstOrNull { entry.startsWith(it.first) }
                    if (hit != null) {
                        modKeys.add(hit.second)
                        ev.add(hit.second, "ARTIFACT")
                        sink.flag(
                            "jh:" + jar.path,
                            ScanFinding(
                                "HIGH", "Archive contains a known cheat client's code",
                                "${jar.path} (modified ${ts(jar.lastModified())}) contains classes under \"${hit.first}\" (${hit.third}). Based on the archive's contents, not just its name; an admin should confirm."
                            )
                        )
                        return@use
                    }
                }
            }
        } catch (e: Exception) {
            // not a readable archive: nothing to add
        }
    }

    private fun scanLog(log: McEntry, gamePath: String, sink: FindingSink, ev: Evidence) {
        val bytes = log.readTail(LOG_BYTES) ?: return
        val lines = String(bytes, Charsets.UTF_8).split('\n').takeLast(LOG_LINES)
        for ((i, line) in lines.withIndex()) {
            val lower = line.lowercase()
            val agentAt = line.indexOf("-javaagent:", ignoreCase = true)
            if (agentAt >= 0) {
                val rest = line.substring(agentAt + "-javaagent:".length)
                val end = rest.indexOfFirst { it == ' ' || it == '"' || it == '=' || it == ',' || it == ']' }
                val token = (if (end < 0) rest else rest.substring(0, end)).substringAfterLast('/').substringAfterLast('\\').take(100)
                ev.add("javaagent", "LOG")
                sink.flag(
                    "la:$gamePath:$token",
                    ScanFinding(
                        "WARN", "Game log shows a Java agent",
                        "${log.path} (log line ${i + 1} of the last ${lines.size}): -javaagent:$token. Agents can be legitimate (for example authentication or profiling tools) but can also inject cheats."
                    )
                )
            }
            val marker = LOG_MARKERS.firstOrNull { lower.contains(it.first) } ?: continue
            val isMixin = lower.contains("mixin")
            ev.add(marker.second, "LOG")
            val kind = if (isMixin) "mixin" else "mention"
            sink.flag(
                "ll:$gamePath:${marker.second}:$kind",
                ScanFinding(
                    "WARN",
                    if (isMixin) "Game log shows a mixin from a cheat client (${marker.second})" else "Game log mentions a cheat client (${marker.second})",
                    "${log.path} (log line ${i + 1} of the last ${lines.size}, modified ${ts(log.modified)}). Matched \"${marker.first}\". " +
                        "A log can name a client without it having been used; the line itself is not sent."
                )
            )
        }
    }

    private fun emitCorrelation(gamePath: String, ev: Evidence, sink: FindingSink) {
        for ((key, families) in ev.m) {
            if (families.size >= 2) {
                sink.flag(
                    "ev:$gamePath:$key",
                    ScanFinding(
                        "WARN", "Several independent traces of \"$key\"",
                        "Under $gamePath: ${families.sorted().joinToString(", ")}. Traces from different places together are stronger than one name match; an admin should review them."
                    )
                )
            }
        }
    }

    private fun emitChunks(sink: FindingSink, key: String, title: String, where: String, items: List<String>) {
        if (items.isEmpty()) return
        val groups = ArrayList<ArrayList<String>>()
        var len = 0
        for (item in items) {
            if (groups.isEmpty() || len + item.length > 300) {
                groups.add(ArrayList())
                len = 0
            }
            groups.last().add(item)
            len += item.length + 2
        }
        val shown = minOf(groups.size, 6)
        for (i in 0 until shown) {
            val t = if (shown > 1) "$title (${i + 1}/$shown)" else title
            sink.flag("$key:$i", ScanFinding("INFO", t, "$where: ${groups[i].joinToString(", ")}"))
        }
        if (groups.size > shown) {
            val more = groups.drop(shown).sumOf { it.size }
            sink.flag("$key:more", ScanFinding("INFO", "$title (more)", "$more more not listed."))
        }
    }

    // ---- Bedrock ------------------------------------------------------------------------------

    private fun analyzeBedrock(root: McEntry, kids: List<McEntry>, run: Run) {
        val sink = run.sink
        val ev = Evidence()
        val packNames = HashMap<String, String>()

        for (folder in listOf("resource_packs", "behavior_packs", "development_resource_packs", "development_behavior_packs")) {
            val packs = sub(kids, folder)?.children()?.filter { it.isDir } ?: continue
            for (p in packs.take(300)) {
                if (run.expired()) return
                var pname = ""
                var desc = ""
                val manifest = p.children()?.firstOrNull { !it.isDir && it.name.equals("manifest.json", true) }
                val text = manifest?.readTail(65_536)?.toString(Charsets.UTF_8)
                if (text != null) {
                    try {
                        val header = JSONObject(text).optJSONObject("header")
                        pname = header?.optString("name") ?: ""
                        desc = header?.optString("description") ?: ""
                        val uuid = (header?.optString("uuid") ?: "").lowercase()
                        if (uuid.isNotEmpty()) packNames[uuid] = p.name + " " + pname
                    } catch (e: Exception) {
                        // unreadable manifest: fall back to the folder name
                    }
                }
                val label = bedrockMatch(p.name + " " + pname + " " + desc)
                if (label != null) {
                    ev.add(label.lowercase(), "NAME")
                    sink.flag(
                        "bk:" + p.path,
                        ScanFinding(
                            "WARN", "Bedrock pack name suggests $label",
                            "${p.path}${if (pname.isNotBlank()) " (named \"${pname.take(60)}\")" else ""}, modified ${ts(p.modified)}. Name match only; some servers allow packs like Night Vision or Fullbright."
                        )
                    )
                }
            }
        }

        val mc = sub(kids, "minecraftpe")?.children()
        val globalPacks = mc?.firstOrNull { !it.isDir && it.name.equals("global_resource_packs.json", true) }
        if (globalPacks != null) {
            val text = globalPacks.readTail(65_536)?.toString(Charsets.UTF_8)
            if (text != null) {
                try {
                    val arr = JSONArray(text)
                    var unknown = 0
                    for (i in 0 until arr.length()) {
                        val id = (arr.optJSONObject(i)?.optString("pack_id") ?: "").lowercase()
                        val nm = packNames[id]
                        if (nm == null) {
                            unknown++
                            continue
                        }
                        val label = bedrockMatch(nm) ?: continue
                        ev.add(label.lowercase(), "CONFIG")
                        sink.flag(
                            "bg:" + root.path + id,
                            ScanFinding(
                                "WARN", "Active global resource pack suggests $label",
                                "${globalPacks.path} activates a pack named \"${nm.take(80)}\" for every world. Name match only."
                            )
                        )
                    }
                    sink.flag(
                        "bgs:" + root.path,
                        ScanFinding("INFO", "Bedrock global resource packs", "${arr.length()} active; $unknown not found in the local pack folders (may be built in). Read from ${globalPacks.path}.")
                    )
                } catch (e: Exception) {
                    // not valid JSON: ignore
                }
            }
        }
        val options = mc?.firstOrNull { !it.isDir && it.name.equals("options.txt", true) }
        if (options != null) {
            val text = options.readTail(131_072)?.toString(Charsets.UTF_8)
            if (text != null) {
                val seenLabels = HashSet<String>()
                for (line in text.split('\n')) {
                    val key = line.substringBefore(':').trim()
                    // Only the setting name is checked and reported: values can hold personal data such as the player name.
                    val label = bedrockMatch(key.replace('_', ' ')) ?: continue
                    if (!seenLabels.add(label)) continue
                    ev.add(label.lowercase(), "CONFIG")
                    sink.flag(
                        "bo:" + root.path + label,
                        ScanFinding("WARN", "Bedrock options.txt has a setting named like $label", "${options.path}: setting \"${key.take(60)}\". Name match only.")
                    )
                }
            }
        }
        emitCorrelation(root.path, ev, sink)
    }

    // ---- Downloads ----------------------------------------------------------------------------

    private fun downloads(sink: FindingSink) {
        @Suppress("DEPRECATION")
        val dir = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)
        val stack = ArrayDeque<Pair<File, Int>>()
        stack.addLast(Pair(dir, 0))
        var count = 0
        var overflow = 0
        while (stack.isNotEmpty()) {
            val (d, depth) = stack.removeLast()
            val kids = try {
                d.listFiles()
            } catch (e: SecurityException) {
                null
            }
            if (kids == null) continue
            for (f in kids) {
                if (isLink(f)) continue
                if (f.isDirectory) {
                    if (depth < 3) stack.addLast(Pair(f, depth + 1))
                    continue
                }
                val ext = f.name.substringAfterLast('.', "").lowercase()
                if (ext !in DOWNLOAD_EXT) continue
                count++
                if (count > MAX_DOWNLOAD_FLAGS) {
                    overflow++
                    continue
                }
                val hit = hackMatch(f.name)
                val warn = ext == "so" || ext == "lua" || (hit != null && hit.strong)
                sink.flag(
                    "dl:" + f.path,
                    ScanFinding(
                        if (warn) "WARN" else "INFO", "File in Downloads: .$ext",
                        "${f.path} modified ${ts(f.lastModified())}, ${f.length() / 1024} KB." +
                            (if (hit != null) " Name matched \"${hit.term}\" (name match only)." else "") +
                            (if (ext == "so" || ext == "lua") " Native or script files here are unusual; not proof by itself." else "")
                    )
                )
            }
        }
        if (overflow > 0) {
            sink.flag("dl:more", ScanFinding("INFO", "More Downloads files omitted", "$overflow further Minecraft-related files were left out."))
        }
    }
}

        }

/** Talks to server/server.js. Blocking: call from Dispatchers.IO. */
object ReportClient {
    private const val TIMEOUT_MS = 45_000 // Render free tier can take ~30-50s to wake from sleep

    /** True only if the server says this PIN exists, is unused and has not expired. */
    fun verifyPin(pin: String): Boolean {
        if (!pin.matches(Regex("\\d{6}"))) return false
        return try {
            val c = open("/api/verify-pin/$pin", "GET")
            c.responseCode == 200
        } catch (e: Exception) {
            false
        }
    }

    fun submit(pin: String, findings: List<ScanFinding>): Boolean {
        return try {
            val arr = JSONArray()
            findings.forEach {
                arr.put(JSONObject().put("severity", it.severity).put("title", it.title).put("detail", it.detail))
            }
            val body = JSONObject().put("report", arr).toString().toByteArray(Charsets.UTF_8)
            val c = open("/api/submit-report/$pin", "POST")
            c.doOutput = true
            c.setRequestProperty("Content-Type", "application/json")
            c.outputStream.use { it.write(body) }
            c.responseCode == 200
        } catch (e: Exception) {
            false
        }
    }

    private fun open(path: String, method: String): HttpURLConnection {
        val c = URL(Config.SERVER_URL + path).openConnection() as HttpURLConnection
        c.requestMethod = method
        c.connectTimeout = TIMEOUT_MS
        c.readTimeout = TIMEOUT_MS
        return c
    }

package com.example.mobilescanner

import android.content.Context
import android.net.Uri
import android.provider.DocumentsContract
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

/**
 * Scans a folder tree the player chose themselves with the system folder picker (Storage Access
 * Framework), for example a game's OBB or data folder. Read-only; file contents are never opened.
 * Android may refuse to let the picker select Android/data or Android/obb on newer versions; if so
 * nothing is scanned and the report says it was skipped.
 */
object SafFolderScanner {
    private const val MAX_DOCS = 50_000
    private const val MAX_DEPTH = 10
    private const val TIME_BUDGET_MS = 30_000L
    private const val MAX_FLAGS = 40
    private const val RECENT_MS = 7L * 24 * 3600 * 1000

    private val CODE_EXT = setOf("so", "lua")
    private val DATA_EXT = setOf("cfg", "pak")

    private class Item(val docId: String, val depth: Int, val path: String)

    fun scan(context: Context, tree: Uri?): List<ScanFinding> {
        if (tree == null) {
            return listOf(
                ScanFinding("INFO", "Game folder check skipped", "No OBB/Data folder was selected (this step is optional).")
            )
        }
        val rootId = try {
            DocumentsContract.getTreeDocumentId(tree)
        } catch (e: Exception) {
            return listOf(ScanFinding("INFO", "Game folder check failed", "The selected folder could not be used."))
        }

        val resolver = context.contentResolver
        val deadline = System.currentTimeMillis() + TIME_BUDGET_MS
        val now = System.currentTimeMillis()
        val fmt = SimpleDateFormat("yyyy-MM-dd HH:mm", Locale.getDefault())
        val out = ArrayList<ScanFinding>()
        val counts = HashMap<String, Int>()
        var docs = 0
        var dirs = 0
        var unreadable = 0
        var recent = 0
        var overflow = 0
        var stop: String? = null
        val label = rootId.substringAfter(':', rootId).ifEmpty { "storage root" }

        val stack = ArrayDeque<Item>()
        stack.addLast(Item(rootId, 0, label))
        while (stack.isNotEmpty()) {
            if (System.currentTimeMillis() > deadline) { stop = "time limit reached"; break }
            if (docs >= MAX_DOCS) { stop = "file limit reached"; break }
            val item = stack.removeLast()
            val childrenUri = DocumentsContract.buildChildDocumentsUriUsingTree(tree, item.docId)
            val cursor = try {
                resolver.query(
                    childrenUri,
                    arrayOf(
                        DocumentsContract.Document.COLUMN_DOCUMENT_ID,
                        DocumentsContract.Document.COLUMN_DISPLAY_NAME,
                        DocumentsContract.Document.COLUMN_MIME_TYPE,
                        DocumentsContract.Document.COLUMN_LAST_MODIFIED,
                        DocumentsContract.Document.COLUMN_SIZE
                    ),
                    null, null, null
                )
            } catch (e: Exception) {
                null
            }
            if (cursor == null) { unreadable++; continue }
            dirs++
            cursor.use { c ->
                while (c.moveToNext()) {
                    val id = c.getString(0) ?: continue
                    val name = c.getString(1) ?: continue
                    val mime = c.getString(2) ?: ""
                    val path = item.path + "/" + name
                    if (mime == DocumentsContract.Document.MIME_TYPE_DIR) {
                        if (item.depth < MAX_DEPTH) stack.addLast(Item(id, item.depth + 1, path))
                        continue
                    }
                    docs++
                    val ext = name.substringAfterLast('.', "").lowercase()
                    val isCode = ext in CODE_EXT
                    if (!isCode && ext !in DATA_EXT) continue
                    counts[ext] = (counts[ext] ?: 0) + 1
                    val modified = if (c.isNull(3)) 0L else c.getLong(3)
                    if (modified <= 0L || now - modified > RECENT_MS) continue
                    recent++
                    if (out.size >= MAX_FLAGS) { overflow++; continue }
                    val size = if (c.isNull(4)) 0L else c.getLong(4)
                    out.add(
                        ScanFinding(
                            if (isCode) "WARN" else "INFO",
                            "Recently modified .$ext file in the selected folder",
                            "$path modified ${fmt.format(Date(modified))}, ${size / 1024} KB. Game updates also change these files; compare with the game's update time."
                        )
                    )
                }
            }
        }

        if (overflow > 0) out.add(ScanFinding("INFO", "More selected-folder matches omitted", "$overflow further recent files were left out."))
        val kinds = listOf("so", "lua", "cfg", "pak").joinToString(", ") { ".$it: ${counts[it] ?: 0}" }
        out.add(
            ScanFinding(
                "INFO", "Selected-folder coverage",
                "Folder \"$label\": $docs files in $dirs folders ($unreadable not readable). Counts by type: $kinds. " +
                    "$recent modified in the last 7 days.${stop?.let { " Stopped early: $it." } ?: ""}"
            )
        )
        return out
    }
}

import android.content.Context

/**

 * folders, and finds Minecraft game folders for MinecraftScanner to analyze. Read-only: it never opens, changes or deletes user files
 * other than reading mod archives' file lists and the tail of game logs.
 *
 * Limits are real and are reported in the summary finding: Android hides other apps' private data

object StorageScanner {
    private const val MAX_FILES = 300_000
    private const val MAX_DEPTH = 14
    private const val TIME_BUDGET_MS = 90_000L
    private const val MAX_FLAGS = 100
    private const val MAX_HIDDEN = 25

        "liquidbounce", "meteorclient", "bleachhack", "salhack", "kamiblue", "aristois",
        "futureclient", "impactclient", "rusherhack", "thunderhack", "autoclicker"
    )

    private val WHOLE_WORDS = setOf("wurst", "vape")

    private val EXTENSIONS = setOf("jar", "zip", "apk", "exe", "dll", "so", "lua", "json", "mcpack", "mcaddon", "rar", "7z")

    private val BENIGN_HIDDEN = setOf(".android_secure", ".thumbnails", ".cache", ".nomedia", ".estrongs", ".face", ".data", ".gs_fs0", ".dcim")
    private val HIDDEN_CODE_EXT = setOf("apk", "so", "jar", "dex", "lua")
    private val TRASH_NAMES = setOf(".recycle", ".recyclebin", "recyclebin", "recycle_bin", "trash", "recently deleted")
    private val TRASHED_FILE = Regex("^\\.trashed-(\\d+)-(.+)$")

    class Result(val findings: List<ScanFinding>, val gameDirs: List<File>)

    private class Node(val dir: File, val depth: Int, val inTrash: Boolean, val hidden: Boolean)
    private class Trashed(val original: String, val expirySeconds: Long)

    private class State(val deadline: Long) : FindingSink {
        var files = 0
        var dirs = 0
        var unreadable = 0
        var trashItems = 0
        var hiddenN = 0
        var apkN = 0
        var soN = 0
        var stopReason: String? = null
        var overflow = 0
        val flags = ArrayList<ScanFinding>()
        val high = ArrayList<ScanFinding>()
        val seen = HashSet<String>()
        val gameDirs = ArrayList<File>()

        fun expired() = System.currentTimeMillis() > deadline

        override fun flag(key: String, f: ScanFinding) {
            if (!seen.add(key)) return
            if (f.severity == "HIGH") {
                if (high.size < MAX_HIGH) high.add(f) else overflow++
            } else {
                if (flags.size < MAX_FLAGS) flags.add(f) else overflow++
            }
        }
    }

    fun scan(context: Context): Result {
        if (!hasAllFilesPermission(context)) {
            return Result(
                listOf(
                    ScanFinding(
                        "INFO", "Storage scan skipped",
                        "All Files Access was not granted, so folders, trash and launcher data were not inspected. This part of the scan is incomplete."
                    )
                ),
                emptyList()
            )
        }

        val st = State(System.currentTimeMillis() + TIME_BUDGET_MS)
        val roots = storageRoots(context)
        for (root in roots) {
            if (st.stopReason != null) break
            walk(root, st)
        }
        mediaStoreTrash(context, st)

        val out = ArrayList<ScanFinding>()
        out.addAll(st.high)
        out.addAll(st.flags)
        if (st.overflow > 0) {
            out.add(ScanFinding("INFO", "More storage findings omitted", "${st.overflow} further matches were left out to keep the report a reasonable size."))
        }
        val stopped = st.stopReason?.let { " Scan stopped early: $it." } ?: ""
        out.add(
            ScanFinding(
                "INFO", "Storage scan coverage",
                "Looked at ${st.files} files in ${st.dirs} folders on ${roots.size} storage volume(s); ${st.unreadable} folders were not readable " +
                    "(normal for other apps' private data); ${st.trashItems} trash items seen.$stopped " +
                    "Permanently deleted files cannot be recovered, so no findings is not proof of a clean device."
            )
        )
        return Result(out, st.gameDirs)
    }

    private fun storageRoots(context: Context): List<File> {
        val roots = LinkedHashSet<File>()
        @Suppress("DEPRECATION")
        roots.add(Environment.getExternalStorageDirectory())
        for (d in context.getExternalFilesDirs(null)) {
            val p = d?.absolutePath ?: continue
            val i = p.indexOf("/Android/")
            if (i > 0) roots.add(File(p.substring(0, i)))
        }
        return roots.toList()
    }

    private fun isSymlink(f: File): Boolean = try {
        Files.isSymbolicLink(f.toPath())
    } catch (e: Exception) {
        false
    }

    private fun isTrashDirName(lower: String) = lower.startsWith(".trash") || lower in TRASH_NAMES

    private fun looksLikeGameDir(dir: File): Boolean =
        File(dir, "versions").isDirectory &&
            (File(dir, "mods").isDirectory || File(dir, "resourcepacks").isDirectory ||
                File(dir, "saves").isDirectory || File(dir, "logs").isDirectory)

    private fun matchName(name: String): String? {
        val lower = name.lowercase()
        val letters = lower.filter { it.isLetter() }
        DISTINCTIVE.firstOrNull { letters.contains(it) }?.let { return it }
        val words = lower.split(Regex("[^a-z0-9]+")).filter { it.isNotEmpty() }.toSet()
        return WHOLE_WORDS.firstOrNull { it in words }
    }

    private fun matchFile(name: String): String? {
        val ext = name.substringAfterLast('.', "").lowercase()
        return if (ext in EXTENSIONS) matchName(name) else null
    }

    private fun parseTrashed(name: String): Trashed? {
        val m = TRASHED_FILE.matchEntire(name) ?: return null
        val expiry = m.groupValues[1].toLongOrNull() ?: return null
        return Trashed(m.groupValues[2], expiry)
    }

    private fun walk(root: File, st: State) {
        val stack = ArrayDeque<Node>()
        stack.addLast(Node(root, 0, false, false))
        while (stack.isNotEmpty()) {
            if (st.expired()) { st.stopReason = "time limit reached"; return }
            if (st.files >= MAX_FILES) { st.stopReason = "file limit reached"; return }
            val node = stack.removeLast()
            val children = try {
                node.dir.listFiles()
            } catch (e: SecurityException) {
                null
            }
            if (children == null) { st.unreadable++; continue }
            st.dirs++
            if (st.gameDirs.size < MAX_GAME_DIRS && (node.dir.name.equals(".minecraft", true) || looksLikeGameDir(node.dir))) st.gameDirs.add(node.dir)

            for (child in children) {
                if (isSymlink(child)) continue
                if (child.isDirectory) {
                    matchName(child.name)?.let {
                        st.flag(
                            "d:" + child.path,
                            ScanFinding("WARN", "Folder name resembles a Minecraft cheat client", "${child.path} matched \"$it\". Name match only; verify before drawing conclusions.")
                        )
                    }
                    val lower = child.name.lowercase()
                    val hiddenDir = lower.startsWith(".") && !isTrashDirName(lower) && lower !in BENIGN_HIDDEN
                    if (hiddenDir && !node.hidden && node.depth <= 3 && st.hiddenN < MAX_HIDDEN) {
                        st.hiddenN++
                        st.flag(
                            "h:" + child.path,
                            ScanFinding("INFO", "Hidden folder", "${child.path} (last modified ${ts(child)}). Hidden folders are common; this is context, not a finding by itself.")
                        )
                    }
                    if (node.depth < MAX_DEPTH) {
                        stack.addLast(Node(child, node.depth + 1, node.inTrash || isTrashDirName(lower), node.hidden || hiddenDir))
                    }
                } else {
                    st.files++
                    handleFile(child, node.inTrash, node.hidden, st)
                }
            }
        }
    }

    private fun ts(f: File): String =
        SimpleDateFormat("yyyy-MM-dd HH:mm", Locale.getDefault()).format(Date(f.lastModified()))

    private fun handleFile(f: File, inTrash: Boolean, hidden: Boolean, st: State) {
        val trashed = parseTrashed(f.name)
        if (trashed != null) {
            st.trashItems++
            checkTrashed(trashed, st)
            return
        }
        if (inTrash) {
            st.trashItems++
            matchFile(f.name)?.let {
                st.flag(
                    "t:" + f.name.lowercase(),
                    ScanFinding("WARN", "Deleted file (in a trash folder) resembles a cheat client", "${f.path} matched \"$it\". Name match only.")
                )
            }
            return
        }
        val modified by lazy { ts(f) }
        matchFile(f.name)?.let {
            st.flag(
                "f:" + f.path,
                ScanFinding("WARN", "File name resembles a Minecraft cheat client", "${f.path} matched \"$it\". Modified $modified. Name match only; verify before drawing conclusions.")
            )
        }
        val ext = f.name.substringAfterLast('.', "").lowercase()
        val kb = f.length() / 1024
        if (ext == "apk" && !f.path.contains("/Download/") && st.apkN < MAX_APK && System.currentTimeMillis() - f.lastModified() <= RECENT_MS) {
            st.apkN++
            st.flag(
                "a:" + f.path,
                ScanFinding("INFO", "Recently modified APK", "${f.path} modified $modified, $kb KB. Downloads and installers create these routinely; context only.")
            )
        }
        if (ext == "so" && st.soN < MAX_SO && !f.path.contains("/Android/") && !f.path.contains("/Download/")) {
            st.soN++
            st.flag(
                "s:" + f.path,
                ScanFinding("WARN", "Native library (.so) in shared storage", "${f.path} modified $modified, $kb KB. Apps normally keep these in private folders; unusual here but not proof of anything.")
            )
        }
        if (hidden && ext in HIDDEN_CODE_EXT) {
            st.flag(
                "hc:" + f.path,
                ScanFinding("WARN", "Code or native file inside a hidden folder", "${f.path} modified $modified, $kb KB. Hidden folders can hold legitimate app data; verify before drawing conclusions.")
            )
        }
    }

    /** Android 11+ trash keeps files as ".trashed-<expiry epoch seconds>-<original name>". */
    private fun checkTrashed(t: Trashed, st: State) {
        val match = matchFile(t.original) ?: return
        val estimate = SimpleDateFormat("yyyy-MM-dd HH:mm", Locale.getDefault())
            .format(Date((t.expirySeconds - TRASH_RETENTION_SECONDS) * 1000))
        st.flag(
            "t:" + t.original.lowercase(),
            ScanFinding(
                "WARN", "Deleted file (in trash) resembles a cheat client",
                "${t.original} matched \"$match\". Estimated deletion around $estimate (assumes the default 30-day trash retention, so this is an estimate). Name match only."
            )
        )
    }

    private fun mediaStoreTrash(context: Context, st: State) {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.R) return
        try {
            val args = Bundle().apply { putInt(MediaStore.QUERY_ARG_MATCH_TRASHED, MediaStore.MATCH_ONLY) }
            val cols = arrayOf(MediaStore.MediaColumns.DISPLAY_NAME)
            context.contentResolver.query(MediaStore.Files.getContentUri("external"), cols, args, null)?.use { c ->
                val idx = c.getColumnIndexOrThrow(MediaStore.MediaColumns.DISPLAY_NAME)
                var n = 0
                while (c.moveToNext() && n < 50_000) {
                    n++
                    val name = c.getString(idx) ?: continue
                    val t = parseTrashed(name)
                    if (t != null) {
                        checkTrashed(t, st)
                    } else {
                        matchFile(name)?.let {
                            st.flag(
                                "t:" + name.lowercase(),
                                ScanFinding("WARN", "Deleted file (in trash) resembles a cheat client", "$name matched \"$it\". Name match only.")
                            )
                        }
                    }
                }
            }
        } catch (e: Exception) {
            // MediaStore trash is a bonus source; the file walk above already covers visible trash files.
        }
    }
}

const path = require("path");

const MAX_SESSIONS = 1000;
const MAX_SESSIONS_PER_IP = 5;

const SEVERITIES = new Set(["HIGH", "WARN", "INFO", "CLEAN"]);

const app = express();
app.disable("x-powered-by");
app.set("trust proxy", Number(process.env.TRUST_PROXY || 0));
app.use(express.json({ limit: "256kb" }));

// slipped through sanitising it still could not execute script in the admin's browser.
app.use((req, res, next) => {
  res.set("Content-Security-Policy",
    "default-src 'none'; script-src 'self'; style-src 'self'; connect-src 'self'; base-uri 'none'; form-action 'none'; frame-ancestors 'none'");
  res.set("X-Content-Type-Options", "nosniff");
  res.set("Referrer-Policy", "no-referrer");
  res.set("Cache-Control", "no-store");
  next();
});
app.use(express.static(path.join(__dirname, "public")));

/** pin -> { pin, token, ip, admin, status: PENDING|COMPLETED, report, createdAt } */
const sessions = new Map();

setInterval(() => {
  const now = Date.now();
  for (const [pin, s] of sessions) {
    if (now - s.createdAt > SESSION_TTL_MS) sessions.delete(pin);
  }
  for (const [k, v] of hits) {
    if (now > v.reset) hits.delete(k);
  }
}, 60 * 1000).unref();

// --- helpers ---------------------------------------------------------------------------------

function safeEqual(a, b) {
  const x = Buffer.from(String(a));
  const y = Buffer.from(String(b));
  return x.length === y.length && crypto.timingSafeEqual(x, y);
}

const hits = new Map();
function rateLimit(name, max, windowMs) {
  return (req, res, next) => {
    const key = name + ":" + req.ip;
    const now = Date.now();
    let h = hits.get(key);
    if (!h || now > h.reset) {
      h = { count: 0, reset: now + windowMs };
      hits.set(key, h);
    }
    if (++h.count > max) {
      res.set("Retry-After", String(Math.ceil((h.reset - now) / 1000)));
      return res.status(429).json({ error: "too many requests" });
    }
    next();
  };
}

function liveSession(pin) {
  if (typeof pin !== "string" || !/^\d{6}$/.test(pin)) return null;
  const s = sessions.get(pin);
  if (!s) return null;
  if (Date.now() - s.createdAt > SESSION_TTL_MS) {
    sessions.delete(pin);
    return null;
  }
  return s;
}

function clean(value, max) {
  // plain text only, no control characters; the dashboard also renders via textContent
  return String(value == null ? "" : value).replace(/[\u0000-\u001f\u007f]/g, " ").slice(0, max);
}

function sanitizeReport(report) {
  if (!Array.isArray(report) || report.length > MAX_FINDINGS) return null;
  return report.map((item) => {
    const sev = item && typeof item.severity === "string" ? item.severity.toUpperCase() : "INFO";
    return {
      severity: SEVERITIES.has(sev) ? sev : "INFO",
      title: clean(item && item.title, 120),
      detail: clean(item && item.detail, 500),
    };
  });
}

// --- API -------------------------------------------------------------------------------------

// 1. Anyone can create a PIN. The reply carries a secret token that only the creator's browser
//    holds; it is required to read the report, so other people cannot read it by guessing PINs.
app.post("/api/create-pin", rateLimit("create", 10, 60 * 1000), (req, res) => {
  if (sessions.size >= MAX_SESSIONS) return res.status(503).json({ error: "too many open sessions" });
  let open = 0;
  for (const v of sessions.values()) if (v.ip === req.ip && v.status === "PENDING") open++;
  if (open >= MAX_SESSIONS_PER_IP) return res.status(429).json({ error: "too many open PINs; use or wait for the earlier ones to expire" });
  let pin;
  do {
    pin = String(crypto.randomInt(100000, 1000000));
  } while (sessions.has(pin));
  const token = crypto.randomBytes(24).toString("hex");
  const admin = clean(req.body && req.body.admin, 60) || "Admin";
  sessions.set(pin, { pin, token, ip: req.ip, admin, status: "PENDING", report: null, createdAt: Date.now() });
  res.json({ pin, token, expiresInSeconds: SESSION_TTL_MS / 1000 });
});

// 2. Player app checks the PIN (rate limited: a 6-digit PIN must not be guessable)
app.get("/api/verify-pin/:pin", rateLimit("verify", 10, 60 * 1000), (req, res) => {
  const s = liveSession(req.params.pin);
  if (!s || s.status !== "PENDING") return res.status(404).json({ valid: false });
  res.json({ valid: true });
});

// 3. Player app uploads its report, once per PIN
app.post("/api/submit-report/:pin", rateLimit("submit", 10, 60 * 1000), (req, res) => {
  const s = liveSession(req.params.pin);
  if (!s || s.status !== "PENDING") return res.status(404).json({ error: "Invalid or already used PIN" });
  const report = sanitizeReport(req.body && req.body.report);
  if (!report) return res.status(400).json({ error: "report must be an array of at most " + MAX_FINDINGS + " findings" });
  s.status = "COMPLETED";
  s.report = report;
  res.json({ success: true });
});

// 4. The creator's dashboard polls for the result, proving itself with the session token
app.get("/api/session/:pin", rateLimit("session", 120, 60 * 1000), (req, res) => {
  const s = liveSession(req.params.pin);
  if (!s || !safeEqual(req.get("x-session-token") || "", s.token)) return res.status(404).json({ error: "Not found or expired" });
  res.json({ pin: s.pin, admin: s.admin, status: s.status, report: s.report, createdAt: s.createdAt });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log("ABBYS Anti-Cheat running on port " + PORT));
