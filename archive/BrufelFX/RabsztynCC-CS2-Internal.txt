`README.md`:

```md
# RabsztynCC - CS2 Internal Cheat & Research Framework

Hey! Welcome to **RabsztynCC**, my custom internal C++ framework built for Counter-Strike 2 (CS2) on 64-bit Windows. I created this project to explore game hacking internals, direct NT syscall execution, kernel driver IPC, custom memory protection mechanisms, and DX11 overlay rendering using ImGui.

> [!WARNING]
> **Educational & Research Purpose Only**
> This project was developed strictly for learning, security research, and educational purposes. Do not use this on official VAC-secured servers or in competitive matches. I am not responsible for any bans or misuse of this code.

---

## 🏗️ Project Architecture

Here is how I structured the codebase:

```
RabsztynCC/
├── src/                      # Main internal DLL source
│   ├── Protection/           # Anti-cheat evasion, direct syscalls, PE header wiping, thread hiding
│   ├── SDK/                  # CS2 structures, entity definitions & math helpers
│   ├── Hooks/                # DirectX 11 / Present hooks using MinHook
│   ├── Features/             # ESP, Chams, Triggerbot, Misc features
│   ├── Menu/                 # ImGui menu layout & rendering logic
│   ├── AntiVAC.h             # Memory protection layer
│   └── dllmain.cpp           # DLL entry point & initialization sequence
├── kernel/                   # Optional Windows Kernel Driver & IPC
│   ├── Driver/               # Kernel driver source
│   └── Shared/               # Shared memory structs & IOCTL definitions
├── tools/ Loader/            # Custom DLL injector / loader tool
├── docs/                     # Technical documentation & notes
└── Offsets.h                 # CS2 memory offsets
```

---

## 📦 Dependencies & Setup

Third-party dependencies in `deps/` are excluded from Git. Before building, download and place these libraries into the `deps/` folder:

1. **ImGui (Dear ImGui)**
   - Put ImGui source files into `deps/imgui/`
   - Make sure `imgui_impl_dx11.cpp` and `imgui_impl_win32.cpp` are included.

2. **MinHook**
   - Place MinHook files into `deps/minhook/` (`MinHook.h`, `trampoline.h`, `buffer.h`, etc.).

---

## 🛠️ How to Build (Visual Studio 2026)

1. **Prerequisites**:
   - Visual Studio 2026 (with Desktop development with C++)
   - Windows 10/11 SDK
   - Build Target: **x64 Release** or **x64 Debug**

2. **Steps**:
   - Clone the repo:
     ```bash
     git clone https://github.com/BrufelFX/RabsztynCC-CS2-Internal.git
     ```
   - Add the `deps/` files mentioned above.
   - Open `RabsztynCC.sln` in Visual Studio 2026.
   - Select **Release | x64** and press `Ctrl + Shift + B` to build.

---

#include "driver.h"

PSHARED_MEMORY g_SharedMemory  = nullptr;
HANDLE         g_hSection      = nullptr;
PVOID          g_SectionObject = nullptr;
BOOLEAN        g_Running       = FALSE;

extern "C" NTSTATUS DriverEntry(PVOID lpBaseAddress, PVOID lpParameter) {
    UNREFERENCED_PARAMETER(lpBaseAddress);
    UNREFERENCED_PARAMETER(lpParameter);

    DbgPrint("[RabsztynCC] DriverEntry via kdmapper\n");

    UNICODE_STRING    sectionName;
    OBJECT_ATTRIBUTES objAttr;
    LARGE_INTEGER     sectionSize;

    RtlInitUnicodeString(&sectionName, SHARED_SECTION_NAME);

    SECURITY_DESCRIPTOR sd;
    RtlCreateSecurityDescriptor(&sd, SECURITY_DESCRIPTOR_REVISION);
    RtlSetDaclSecurityDescriptor(&sd, TRUE /*DaclPresent*/, NULL /*Dacl*/, FALSE);

    InitializeObjectAttributes(
        &objAttr,
        &sectionName,
        OBJ_CASE_INSENSITIVE | OBJ_KERNEL_HANDLE,
        nullptr,
        &sd
    );
    sectionSize.QuadPart = SHARED_MEMORY_SIZE;

    NTSTATUS status = ZwCreateSection(
        &g_hSection,
        SECTION_ALL_ACCESS,
        &objAttr,
        &sectionSize,
        PAGE_READWRITE,
        SEC_COMMIT,
        nullptr
    );
    if (!NT_SUCCESS(status)) {
        DbgPrint("[RabsztynCC] ZwCreateSection failed: 0x%08X\n", status);
        return status;
    }

    status = ObReferenceObjectByHandle(
        g_hSection,
        SECTION_ALL_ACCESS,
        nullptr,
        KernelMode,
        &g_SectionObject,
        nullptr
    );
    if (!NT_SUCCESS(status)) {
        DbgPrint("[RabsztynCC] ObReferenceObjectByHandle failed: 0x%08X\n", status);
        ZwClose(g_hSection);
        return status;
    }

    PVOID  baseAddr  = nullptr;
    SIZE_T viewSize  = SHARED_MEMORY_SIZE;
    status = MmMapViewInSystemSpace(g_SectionObject, &baseAddr, &viewSize);
    if (!NT_SUCCESS(status)) {
        DbgPrint("[RabsztynCC] MmMapViewInSystemSpace failed: 0x%08X\n", status);
        ObDereferenceObject(g_SectionObject);
        ZwClose(g_hSection);
        return status;
    }

    g_SharedMemory = reinterpret_cast<PSHARED_MEMORY>(baseAddr);
    RtlZeroMemory(g_SharedMemory, SHARED_MEMORY_SIZE);
    g_SharedMemory->Status = STATUS_COMM_IDLE;

    DbgPrint("[RabsztynCC] Shared memory at: %p\n", g_SharedMemory);

    g_Running = TRUE;
    HANDLE hThread = nullptr;
    status = PsCreateSystemThread(
        &hThread,
        THREAD_ALL_ACCESS,
        nullptr,
        nullptr,
        nullptr,
        WorkerThread,
        nullptr
    );
    if (!NT_SUCCESS(status)) {
        DbgPrint("[RabsztynCC] PsCreateSystemThread failed: 0x%08X\n", status);
        MmUnmapViewInSystemSpace(baseAddr);
        ObDereferenceObject(g_SectionObject);
        ZwClose(g_hSection);
        return status;
    }
    ZwClose(hThread);

    ULONG actualRes = ExSetTimerResolution(5000, TRUE);
    DbgPrint("[RabsztynCC] Timer resolution: %u (100ns units)\n", actualRes);

    InitThreadListOffset();

    DbgPrint("[RabsztynCC] Ready\n");
    return STATUS_SUCCESS;
}

VOID WorkerThread(PVOID context) {
    UNREFERENCED_PARAMETER(context);

    DbgPrint("[RabsztynCC] Worker thread running\n");

    LARGE_INTEGER idleInterval;
    idleInterval.QuadPart = -5000;

    static const ULONG HOT_SPIN_LIMIT = 50000;
    ULONG hotSpinsLeft = 0;

    while (g_Running) {
        if (g_SharedMemory->Status != STATUS_COMM_PENDING) {
            if (hotSpinsLeft > 0) {
                hotSpinsLeft--;
                YieldProcessor();
                continue;
            }
            KeDelayExecutionThread(KernelMode, FALSE, &idleInterval);
            continue;
        }

        NTSTATUS result = STATUS_INVALID_PARAMETER;

        switch (g_SharedMemory->Command) {
            case CMD_READ:          result = HandleRead(g_SharedMemory);           break;
            case CMD_WRITE:         result = HandleWrite(g_SharedMemory);          break;
            case CMD_READ_BATCH:    result = HandleReadBatch(g_SharedMemory);      break;
            case CMD_GET_BASE:      result = HandleGetBase(g_SharedMemory);        break;
            case CMD_GET_PID:       result = HandleGetPid(g_SharedMemory);         break;
            case CMD_HIDE_THREAD:        result = HandleHideThread(g_SharedMemory, TRUE);  break;
            case CMD_UNHIDE_THREAD:      result = HandleHideThread(g_SharedMemory, FALSE); break;
            case CMD_HIDE_THREAD_STRONG: result = HandleHideThreadStrong(g_SharedMemory);  break;
            case CMD_PING:          result = STATUS_SUCCESS;                       break;
            default:
                DbgPrint("[RabsztynCC] Unknown cmd: %u\n", g_SharedMemory->Command);
                break;
        }

        g_SharedMemory->Status = NT_SUCCESS(result)
            ? STATUS_COMM_OK
            : STATUS_COMM_FAIL;

        hotSpinsLeft = HOT_SPIN_LIMIT;
    }

    PsTerminateSystemThread(STATUS_SUCCESS);
}

NTSTATUS HandleRead(PSHARED_MEMORY shared) {
    if (!shared->ProcessId || !shared->Address || !shared->Size)
        return STATUS_INVALID_PARAMETER;

    if (shared->Size > sizeof(shared->Buffer))
        return STATUS_BUFFER_OVERFLOW;

    if (shared->Address < 0x10000 || shared->Address > 0x00007FFFFFFFFFFF)
        return STATUS_INVALID_PARAMETER;

    PEPROCESS process = nullptr;
    NTSTATUS  status  = PsLookupProcessByProcessId(
        reinterpret_cast<HANDLE>(static_cast<ULONG_PTR>(shared->ProcessId)),
        &process
    );
    if (!NT_SUCCESS(status)) return status;

    SIZE_T bytes = 0;
    __try {
        status = MmCopyVirtualMemory(
            process,
            reinterpret_cast<PVOID>(shared->Address),
            PsGetCurrentProcess(),
            shared->Buffer,
            static_cast<SIZE_T>(shared->Size),
            KernelMode,
            &bytes
        );
    }
    __except (EXCEPTION_EXECUTE_HANDLER) {
        status = STATUS_ACCESS_VIOLATION;
    }

    ObDereferenceObject(process);
    return status;
}

NTSTATUS HandleWrite(PSHARED_MEMORY shared) {
    if (!shared->ProcessId || !shared->Address || !shared->Size)
        return STATUS_INVALID_PARAMETER;

    if (shared->Size > sizeof(shared->Buffer))
        return STATUS_BUFFER_OVERFLOW;

    if (shared->Address < 0x10000 || shared->Address > 0x00007FFFFFFFFFFF)
        return STATUS_INVALID_PARAMETER;

    PEPROCESS process = nullptr;
    NTSTATUS  status  = PsLookupProcessByProcessId(
        reinterpret_cast<HANDLE>(static_cast<ULONG_PTR>(shared->ProcessId)),
        &process
    );
    if (!NT_SUCCESS(status)) return status;

    SIZE_T bytes = 0;
    __try {
        status = MmCopyVirtualMemory(
            PsGetCurrentProcess(),
            shared->Buffer,
            process,
            reinterpret_cast<PVOID>(shared->Address),
            static_cast<SIZE_T>(shared->Size),
            KernelMode,
            &bytes
        );
    }
    __except (EXCEPTION_EXECUTE_HANDLER) {
        status = STATUS_ACCESS_VIOLATION;
    }

    ObDereferenceObject(process);
    return status;
}

NTSTATUS HandleReadBatch(PSHARED_MEMORY shared) {
    if (!shared->ProcessId) return STATUS_INVALID_PARAMETER;
    if (shared->BatchCount == 0 || shared->BatchCount > MAX_BATCH_ENTRIES)
        return STATUS_INVALID_PARAMETER;

    PEPROCESS process = nullptr;
    NTSTATUS  status  = PsLookupProcessByProcessId(
        reinterpret_cast<HANDLE>(static_cast<ULONG_PTR>(shared->ProcessId)),
        &process
    );
    if (!NT_SUCCESS(status)) return status;

    const unsigned int bufSize = sizeof(shared->Buffer);
    unsigned int okCount = 0;

    for (unsigned int i = 0; i < shared->BatchCount; i++) {
        BATCH_ENTRY& e = shared->BatchEntries[i];
        e.Success = 0;

        if (e.Size == 0 || e.Size > bufSize) continue;
        if ((ULONG64)e.ResultOffset + e.Size > bufSize) continue;
        if (e.Address < 0x10000 || e.Address > 0x00007FFFFFFFFFFF) continue;

        SIZE_T bytes = 0;
        NTSTATUS entryStatus = STATUS_ACCESS_VIOLATION;
        __try {
            entryStatus = MmCopyVirtualMemory(
                process,
                reinterpret_cast<PVOID>(e.Address),
                PsGetCurrentProcess(),
                shared->Buffer + e.ResultOffset,
                static_cast<SIZE_T>(e.Size),
                KernelMode,
                &bytes
            );
        }
        __except (EXCEPTION_EXECUTE_HANDLER) {
            entryStatus = STATUS_ACCESS_VIOLATION;
        }

        if (NT_SUCCESS(entryStatus) && bytes == e.Size) {
            e.Success = 1;
            okCount++;
        }
    }

    ObDereferenceObject(process);

    return STATUS_SUCCESS;
}

static NTSTATUS SafeReadUserMemory(PEPROCESS process, PVOID src, PVOID dst, SIZE_T size) {
    SIZE_T bytes = 0;
    return MmCopyVirtualMemory(process, src, PsGetCurrentProcess(), dst, size, KernelMode, &bytes);
}

NTSTATUS HandleGetBase(PSHARED_MEMORY shared) {
    if (!shared->ProcessId || !shared->Name[0])
        return STATUS_INVALID_PARAMETER;

    PEPROCESS process = nullptr;
    NTSTATUS  status  = PsLookupProcessByProcessId(
        reinterpret_cast<HANDLE>(static_cast<ULONG_PTR>(shared->ProcessId)),
        &process
    );
    if (!NT_SUCCESS(status)) return status;

    status = STATUS_NOT_FOUND;

    PPEB pebAddress = PsGetProcessPeb(process);
    if (!pebAddress) {
        ObDereferenceObject(process);
        return STATUS_NOT_FOUND;
    }

    PPEB_LDR_DATA ldrAddress = nullptr;
    ULONG_PTR ldrOffset = FIELD_OFFSET(PEB, Ldr);
    status = SafeReadUserMemory(
        process,
        reinterpret_cast<PUCHAR>(pebAddress) + ldrOffset,
        &ldrAddress,
        sizeof(ldrAddress)
    );
    if (!NT_SUCCESS(status) || !ldrAddress) {
        ObDereferenceObject(process);
        return STATUS_NOT_FOUND;
    }

    LIST_ENTRY head = {};
    ULONG_PTR headOffset = FIELD_OFFSET(PEB_LDR_DATA, InMemoryOrderModuleList);
    status = SafeReadUserMemory(
        process,
        reinterpret_cast<PUCHAR>(ldrAddress) + headOffset,
        &head,
        sizeof(head)
    );
    if (!NT_SUCCESS(status) || !head.Flink) {
        ObDereferenceObject(process);
        return STATUS_NOT_FOUND;
    }

    PLIST_ENTRY currentPtr = head.Flink;
    PLIST_ENTRY headPtr    = reinterpret_cast<PLIST_ENTRY>(reinterpret_cast<PUCHAR>(ldrAddress) + headOffset);
    ULONG maxIter = 512;
    status = STATUS_NOT_FOUND;

    while (currentPtr != headPtr && maxIter--) {

        LDR_DATA_TABLE_ENTRY entry = {};
        NTSTATUS readStatus = SafeReadUserMemory(
            process,
            CONTAINING_RECORD(currentPtr, LDR_DATA_TABLE_ENTRY, InMemoryOrderLinks),
            &entry,
            sizeof(entry)
        );
        if (!NT_SUCCESS(readStatus)) break;

        if (entry.BaseDllName.Buffer && entry.BaseDllName.Length > 0 && entry.DllBase) {
            USHORT nameLen = min(entry.BaseDllName.Length, (USHORT)512);
            WCHAR  nameBuf[256] = {};
            SafeReadUserMemory(process, entry.BaseDllName.Buffer, nameBuf, nameLen);

            if (_wcsicmp(nameBuf, shared->Name) == 0) {
                shared->OutAddress = reinterpret_cast<ULONG64>(entry.DllBase);
                shared->OutSize    = static_cast<ULONG64>(entry.SizeOfImage);
                status = STATUS_SUCCESS;
                break;
            }
        }

        LIST_ENTRY nextLink = {};
        if (!NT_SUCCESS(SafeReadUserMemory(process, currentPtr, &nextLink, sizeof(nextLink))))
            break;
        currentPtr = nextLink.Flink;
    }

    ObDereferenceObject(process);
    return status;
}

NTSTATUS HandleGetPid(PSHARED_MEMORY shared) {
    if (!shared->Name[0])
        return STATUS_INVALID_PARAMETER;

    ULONG    bufSize = 1024 * 1024;
    PVOID    buf     = ExAllocatePool2(POOL_FLAG_NON_PAGED, bufSize, 'RPID');
    if (!buf) return STATUS_INSUFFICIENT_RESOURCES;

    ULONG    retSize = 0;
    NTSTATUS status  = ZwQuerySystemInformation(
        5,
        buf, bufSize, &retSize
    );
    if (!NT_SUCCESS(status)) {
        ExFreePoolWithTag(buf, 'RPID');
        return status;
    }

    WCHAR target[256] = {};
    for (int i = 0; i < 255 && shared->Name[i]; i++)
        target[i] = shared->Name[i];
    WCHAR* dot = wcsrchr(target, L'.');
    if (dot && _wcsicmp(dot, L".exe") == 0)
        *dot = L'\0';

    status = STATUS_NOT_FOUND;
    PSYSTEM_PROCESS_INFORMATION spi = (PSYSTEM_PROCESS_INFORMATION)buf;

    while (TRUE) {
        if (spi->ImageName.Buffer && spi->ImageName.Length > 0) {
            WCHAR nameBuf[256] = {};
            USHORT copyLen = min(spi->ImageName.Length, (USHORT)(255 * sizeof(WCHAR)));
            RtlCopyMemory(nameBuf, spi->ImageName.Buffer, copyLen);

            WCHAR* nameDot = wcsrchr(nameBuf, L'.');
            if (nameDot && _wcsicmp(nameDot, L".exe") == 0)
                *nameDot = L'\0';

            if (_wcsicmp(nameBuf, target) == 0) {
                shared->OutAddress = (ULONG64)(ULONG_PTR)spi->UniqueProcessId;
                status = STATUS_SUCCESS;
                break;
            }
        }

        if (spi->NextEntryOffset == 0) break;
        spi = (PSYSTEM_PROCESS_INFORMATION)((PUCHAR)spi + spi->NextEntryOffset);
    }

    ExFreePoolWithTag(buf, 'RPID');
    return status;
}

NTSTATUS HandleHideThread(PSHARED_MEMORY shared, BOOLEAN hide) {
    if (!shared->Address) return STATUS_INVALID_PARAMETER;
    HANDLE tid = (HANDLE)(ULONG_PTR)shared->Address;

    PETHREAD thread = NULL;
    NTSTATUS st = PsLookupThreadByThreadId(tid, &thread);
    if (!NT_SUCCESS(st)) return st;

    if (!hide) {
        ObDereferenceObject(thread);
        return STATUS_NOT_IMPLEMENTED;
    }

    HANDLE hThread = NULL;
    st = ObOpenObjectByPointer(thread,
                               OBJ_KERNEL_HANDLE,
                               NULL,
                               THREAD_ALL_ACCESS,
                               *PsThreadType,
                               KernelMode,
                               &hThread);
    ObDereferenceObject(thread);
    if (!NT_SUCCESS(st)) return st;

    st = ZwSetInformationThread(hThread,
                                (THREADINFOCLASS)ThreadHideFromDebugger,
                                NULL, 0);
    ZwClose(hThread);
    return st;
}

ULONG g_ThreadListOffset = 0;

static BOOLEAN TryDetectOffset(PETHREAD sample) {
    if (!sample) return FALSE;

    for (ULONG offset = 0x400; offset < 0x800; offset += 8) {
        PLIST_ENTRY entry = (PLIST_ENTRY)((PUCHAR)sample + offset);

        __try {
            if (!MmIsAddressValid(entry))         continue;
            if (!MmIsAddressValid(entry->Flink))  continue;
            if (!MmIsAddressValid(entry->Blink))  continue;
            if (entry->Flink->Blink != entry)     continue;
            if (entry->Blink->Flink != entry)     continue;

            int threadHits = 0;
            int hops       = 0;
            PLIST_ENTRY cur = entry->Flink;

            while (cur != entry && hops < 8) {
                PVOID maybeEthread = (PUCHAR)cur - offset;
                if (MmIsAddressValid(maybeEthread)) {
                    UCHAR type = *(UCHAR*)maybeEthread;
                    if (type == 6) threadHits++;
                }
                if (!MmIsAddressValid(cur->Flink)) break;
                cur = cur->Flink;
                hops++;
            }

            if (threadHits >= 1) {
                g_ThreadListOffset = offset;
                DbgPrint("[RabsztynCC] ThreadListEntry offset: 0x%X "
                         "(peer threads seen: %d)\n", offset, threadHits);
                return TRUE;
            }
        }
        __except (EXCEPTION_EXECUTE_HANDLER) {
            continue;
        }
    }
    return FALSE;
}

VOID InitThreadListOffset(VOID) {
    if (g_ThreadListOffset) return;
    if (!TryDetectOffset(PsGetCurrentThread())) {
        DbgPrint("[RabsztynCC] offset auto-detect deferred to first request\n");
    }
}

NTSTATUS HandleHideThreadStrong(PSHARED_MEMORY shared) {
    if (!shared->Address) return STATUS_INVALID_PARAMETER;
    HANDLE tid = (HANDLE)(ULONG_PTR)shared->Address;

    PETHREAD thread = NULL;
    NTSTATUS st = PsLookupThreadByThreadId(tid, &thread);
    if (!NT_SUCCESS(st)) return st;

    if (g_ThreadListOffset == 0) {
        TryDetectOffset(thread);
        if (g_ThreadListOffset == 0) {
            ObDereferenceObject(thread);
            return STATUS_NOT_SUPPORTED;
        }
    }

    HANDLE hThread = NULL;
    if (NT_SUCCESS(ObOpenObjectByPointer(thread,
                                         OBJ_KERNEL_HANDLE,
                                         NULL,
                                         THREAD_ALL_ACCESS,
                                         *PsThreadType,
                                         KernelMode,
                                         &hThread))) {
        ZwSetInformationThread(hThread,
                               (THREADINFOCLASS)ThreadHideFromDebugger,
                               NULL, 0);
        ZwClose(hThread);
    }

    PLIST_ENTRY entry = (PLIST_ENTRY)((PUCHAR)thread + g_ThreadListOffset);

    KIRQL oldIrql;
    KeRaiseIrql(DISPATCH_LEVEL, &oldIrql);

    st = STATUS_SUCCESS;
    __try {
        if (MmIsAddressValid(entry) &&
            MmIsAddressValid(entry->Flink) &&
            MmIsAddressValid(entry->Blink) &&
            entry->Flink->Blink == entry &&
            entry->Blink->Flink == entry &&
            entry->Flink != entry) {
            RemoveEntryList(entry);
            entry->Flink = entry;
            entry->Blink = entry;
        } else {
            st = STATUS_INVALID_PARAMETER;
        }
    }
    __except (EXCEPTION_EXECUTE_HANDLER) {
        st = GetExceptionCode();
    }

    KeLowerIrql(oldIrql);
    ObDereferenceObject(thread);
    return st;
}

    inline bool HideModuleFromPEB(HMODULE hModule) {
        #ifdef _WIN64
        FULL_PEB* peb = (FULL_PEB*)__readgsqword(0x60);
        #else
        FULL_PEB* peb = (FULL_PEB*)__readfsdword(0x30);
        #endif

        if (!peb) return false;

        FULL_PEB_LDR_DATA* ldr = peb->Ldr;
        if (!ldr) return false;

        LIST_ENTRY* head = &ldr->InLoadOrderModuleList;
        LIST_ENTRY* current = head->Flink;

        while (current != head) {
            FULL_LDR_DATA_TABLE_ENTRY* entry =
                CONTAINING_RECORD(current, FULL_LDR_DATA_TABLE_ENTRY, InLoadOrderLinks);

            if (entry->DllBase == (PVOID)hModule) {

                entry->InLoadOrderLinks.Blink->Flink = entry->InLoadOrderLinks.Flink;
                entry->InLoadOrderLinks.Flink->Blink = entry->InLoadOrderLinks.Blink;

                entry->InMemoryOrderLinks.Blink->Flink = entry->InMemoryOrderLinks.Flink;
                entry->InMemoryOrderLinks.Flink->Blink = entry->InMemoryOrderLinks.Blink;

                entry->InInitializationOrderLinks.Blink->Flink = entry->InInitializationOrderLinks.Flink;
                entry->InInitializationOrderLinks.Flink->Blink = entry->InInitializationOrderLinks.Blink;

                return true;
            }

            current = current->Flink;
        }

        return false;

    void BumpMatchedSig(uint32_t indexCount, uint32_t stride) {
        for (int i = 0; i < g_stats.matchedSigCount; ++i) {
            auto& s = g_stats.matchedSigs[i];
            if (s.indexCount == indexCount && s.stride == stride) {
                ++s.count;
                return;
            }
        }
        if (g_stats.matchedSigCount < Chams::kMaxMatchedSigs) {
            auto& s = g_stats.matchedSigs[g_stats.matchedSigCount++];
            s.indexCount = indexCount;
            s.stride     = stride;
            s.count      = 1;
        }

    }

    uintptr_t GetCurrentVSId(ID3D11DeviceContext* ctx) {
        ID3D11VertexShader* vs = nullptr;
        ctx->VSGetShader(&vs, nullptr, nullptr);
        uintptr_t id = reinterpret_cast<uintptr_t>(vs);
        if (vs) vs->Release();
        return id;
    }

    bool IsVSBlacklisted(uintptr_t vs) {
        if (!vs) return false;
        const auto& cfg = Config::chams3d;
        for (int i = 0; i < cfg.vsBlacklistCount; ++i) {
            if (cfg.vsBlacklist[i] == vs) return true;
        }
        return false;
    }

    void BumpMatchedShader(uintptr_t vs) {
        if (!vs) return;
        for (int i = 0; i < g_stats.matchedShaderCount; ++i) {
            auto& e = g_stats.matchedShaders[i];
            if (e.vs == vs) { ++e.count; return; }
        }
        if (g_stats.matchedShaderCount < Chams::kMaxMatchedShaders) {
            auto& e = g_stats.matchedShaders[g_stats.matchedShaderCount++];
            e.vs    = vs;
            e.count = 1;
        }
    }

    bool CompileFlatColorPS() {
        ID3DBlob* code = nullptr;
        ID3DBlob* err  = nullptr;
        HRESULT hr = D3DCompile(kFlatColorPS, sizeof(kFlatColorPS) - 1,
                                "ChamsFlatPS", nullptr, nullptr,
                                "main", "ps_5_0",
                                D3DCOMPILE_OPTIMIZATION_LEVEL3, 0,
                                &code, &err);
        if (FAILED(hr)) {
            const char* msg = err ? static_cast<const char*>(err->GetBufferPointer())
                                  : "(no error blob)";
            Log::Write("[Chams] D3DCompile ps_5_0 failed 0x%08X: %s", hr, msg);
            if (err)  err->Release();
            if (code) code->Release();
            return false;
        }
        hr = g_device->CreatePixelShader(code->GetBufferPointer(),
                                         code->GetBufferSize(),
                                         nullptr, &g_flatPS);
        if (err)  err->Release();
        if (code) code->Release();
        if (FAILED(hr)) {
            Log::Write("[Chams] CreatePixelShader failed 0x%08X", hr);
            return false;
        }
        return true;
    }

    bool CreateDepthStates() {
        D3D11_DEPTH_STENCIL_DESC ds{};
        ds.DepthEnable    = TRUE;

        ds.DepthWriteMask = D3D11_DEPTH_WRITE_MASK_ZERO;
        ds.DepthFunc      = D3D11_COMPARISON_ALWAYS;
        ds.StencilEnable  = FALSE;
        HRESULT hr = g_device->CreateDepthStencilState(&ds, &g_dssAlways);
        if (FAILED(hr)) {
            Log::Write("[Chams] CreateDepthStencilState(ALWAYS) 0x%08X", hr);
            return false;
        }

        ds.DepthFunc = D3D11_COMPARISON_LESS_EQUAL;
        hr = g_device->CreateDepthStencilState(&ds, &g_dssLessEq);
        if (FAILED(hr)) {
            Log::Write("[Chams] CreateDepthStencilState(LESS_EQ) 0x%08X", hr);
            return false;
        }
        return true;
    }

    bool CreateBlendState() {
        D3D11_BLEND_DESC bd{};
        bd.AlphaToCoverageEnable  = FALSE;
        bd.IndependentBlendEnable = FALSE;
        auto& rt = bd.RenderTarget[0];
        rt.BlendEnable    = FALSE;
        rt.RenderTargetWriteMask = D3D11_COLOR_WRITE_ENABLE_ALL;
        HRESULT hr = g_device->CreateBlendState(&bd, &g_opaqueBlend);
        if (FAILED(hr)) {
            Log::Write("[Chams] CreateBlendState failed 0x%08X", hr);
            return false;
        }
        return true;
    }

    bool CreateColorBuffer() {
        D3D11_BUFFER_DESC bd{};
        bd.ByteWidth      = 16;
        bd.Usage          = D3D11_USAGE_DYNAMIC;
        bd.BindFlags      = D3D11_BIND_CONSTANT_BUFFER;
        bd.CPUAccessFlags = D3D11_CPU_ACCESS_WRITE;
        HRESULT hr = g_device->CreateBuffer(&bd, nullptr, &g_cbColor);
        if (FAILED(hr)) {
            Log::Write("[Chams] CreateBuffer(color) 0x%08X", hr);
            return false;
        }
        return true;
    }

    void UpdateColor(const float rgba[4]) {
        D3D11_MAPPED_SUBRESOURCE map{};
        if (SUCCEEDED(g_context->Map(g_cbColor, 0,
                                     D3D11_MAP_WRITE_DISCARD, 0, &map))) {
            std::memcpy(map.pData, rgba, 16);
            g_context->Unmap(g_cbColor, 0);
        }
    }

    UINT GetSlot0Stride(ID3D11DeviceContext* ctx) {
        ID3D11Buffer* vb = nullptr;
        UINT stride = 0;
        UINT offset = 0;
        ctx->IAGetVertexBuffers(0, 1, &vb, &stride, &offset);
        if (vb) vb->Release();
        return stride;
    }

    bool PassesFilter(ID3D11DeviceContext* ctx, UINT indexCount,
                      UINT* outStride, D3D11_PRIMITIVE_TOPOLOGY* outTopo) {
        const auto& cfg = Config::chams3d;

        if (static_cast<int>(indexCount) < cfg.minIndexCount ||
            static_cast<int>(indexCount) > cfg.maxIndexCount) {
            return false;
        }

        UINT stride = GetSlot0Stride(ctx);
        *outStride  = stride;
        if (static_cast<int>(stride) < cfg.minStride ||
            static_cast<int>(stride) > cfg.maxStride) {
            return false;
        }

        D3D11_PRIMITIVE_TOPOLOGY topo = D3D11_PRIMITIVE_TOPOLOGY_UNDEFINED;
        ctx->IAGetPrimitiveTopology(&topo);
        *outTopo = topo;
        if (cfg.triangleListOnly &&
            topo != D3D11_PRIMITIVE_TOPOLOGY_TRIANGLELIST) {
            return false;
        }
        return true;
    }

    template <typename DrawFn>
    void IssueTwoPass(ID3D11DeviceContext* ctx, DrawFn&& draw) {
        const auto& cfg = Config::chams3d;

        ID3D11PixelShader*       origPS       = nullptr;
        ID3D11DepthStencilState* origDSS      = nullptr;
        UINT                     origStRef    = 0;
        ID3D11Buffer*            origCB       = nullptr;
        ID3D11BlendState*        origBlend    = nullptr;
        float                    origBlendF[4]= { 1.f, 1.f, 1.f, 1.f };
        UINT                     origSampMask = 0xFFFFFFFFu;

        ctx->PSGetShader(&origPS, nullptr, nullptr);
        ctx->OMGetDepthStencilState(&origDSS, &origStRef);
        ctx->PSGetConstantBuffers(kColorCBufferSlot, 1, &origCB);
        if (cfg.forceOpaqueBlend) {
            ctx->OMGetBlendState(&origBlend, origBlendF, &origSampMask);
        }

        ctx->PSSetShader(g_flatPS, nullptr, 0);
        ctx->PSSetConstantBuffers(kColorCBufferSlot, 1, &g_cbColor);
        if (cfg.forceOpaqueBlend) {

            const float unitFactors[4] = { 1.f, 1.f, 1.f, 1.f };
            ctx->OMSetBlendState(g_opaqueBlend, unitFactors, 0xFFFFFFFFu);
        }

        UpdateColor(cfg.xqzColor);
        ctx->OMSetDepthStencilState(g_dssAlways, 0);
        draw();

        UpdateColor(cfg.visibleColor);
        ctx->OMSetDepthStencilState(g_dssLessEq, 0);
        draw();

        ctx->PSSetShader(origPS, nullptr, 0);
        ctx->OMSetDepthStencilState(origDSS, origStRef);
        ctx->PSSetConstantBuffers(kColorCBufferSlot, 1, &origCB);
        if (cfg.forceOpaqueBlend) {
            ctx->OMSetBlendState(origBlend, origBlendF, origSampMask);
        }

        if (origPS)    origPS->Release();
        if (origDSS)   origDSS->Release();
        if (origCB)    origCB->Release();
        if (origBlend) origBlend->Release();
    }

    template <typename DrawFn>
    bool TryChams(ID3D11DeviceContext* ctx,
                  const char* fnName,
                  UINT indexCount,
                  uint32_t& outMatched,
                  uint32_t& outOverridden,
                  DrawFn&& draw)
    {
        const auto& cfg = Config::chams3d;

        if (cfg.debugMode && g_totalDrawsSeen <= kDebugLogFirstN) {
            UINT dbgStride = GetSlot0Stride(ctx);
            D3D11_PRIMITIVE_TOPOLOGY dbgTopo = D3D11_PRIMITIVE_TOPOLOGY_UNDEFINED;
            ctx->IAGetPrimitiveTopology(&dbgTopo);
            Log::Write("[Chams] draw #%u %s idx=%u stride=%u topo=%d",
                       g_totalDrawsSeen, fnName, indexCount, dbgStride,
                       static_cast<int>(dbgTopo));
        }

        UINT stride = 0;
        D3D11_PRIMITIVE_TOPOLOGY topo = D3D11_PRIMITIVE_TOPOLOGY_UNDEFINED;
        if (!PassesFilter(ctx, indexCount, &stride, &topo)) {
            return false;
        }

        const uintptr_t vsId = GetCurrentVSId(ctx);

        if (IsVSBlacklisted(vsId)) {
            ++g_stats.drawsSkippedVSBlacklist;
            return false;
        }

        ++outMatched;
        BumpMatchedSig(indexCount, stride);
        BumpMatchedShader(vsId);

        if (cfg.debugMode && outMatched <= 32) {
            Log::Write("[Chams] MATCH %s idx=%u stride=%u topo=%d vs=0x%p",
                       fnName, indexCount, stride, static_cast<int>(topo),
                       reinterpret_cast<void*>(vsId));
        }

        uint32_t totalOverride = g_stats.diOverridden + g_stats.diiOverridden;
        if (static_cast<int>(totalOverride) >= cfg.maxDrawsPerFrame) {
            ++g_stats.drawsSkippedCap;
            return false;
        }

        IssueTwoPass(ctx, std::forward<DrawFn>(draw));
        ++outOverridden;
        return true;
    }

    void STDMETHODCALLTYPE hkDrawIndexed(ID3D11DeviceContext* ctx,
                                        UINT indexCount,
                                        UINT startIndex,
                                        INT  baseVertex) {
        ++g_stats.diSeen;
        ++g_totalDrawsSeen;

        if (!g_initialised || !oDrawIndexed) return;
        if (!Config::chams3d.enabled) {
            return oDrawIndexed(ctx, indexCount, startIndex, baseVertex);
        }
        if (g_imguiRendering) {
            ++g_stats.drawsSkippedImGui;
            return oDrawIndexed(ctx, indexCount, startIndex, baseVertex);
        }
        if (ctx != g_context) {
            return oDrawIndexed(ctx, indexCount, startIndex, baseVertex);
        }

        bool handled = TryChams(ctx, "DI", indexCount,
                                g_stats.diMatched, g_stats.diOverridden,
                                [&](){ oDrawIndexed(ctx, indexCount, startIndex, baseVertex); });
        if (!handled) {
            oDrawIndexed(ctx, indexCount, startIndex, baseVertex);
        }
    }

    void STDMETHODCALLTYPE hkDrawIndexedInstanced(ID3D11DeviceContext* ctx,
                                                 UINT indexCountPerInstance,
                                                 UINT instanceCount,
                                                 UINT startIndex,
                                                 INT  baseVertex,
                                                 UINT startInstanceLocation) {
        ++g_stats.diiSeen;
        ++g_totalDrawsSeen;

        if (!g_initialised || !oDrawIndexedInstanced) return;
        if (!Config::chams3d.enabled) {
            return oDrawIndexedInstanced(ctx, indexCountPerInstance,
                                         instanceCount, startIndex,
                                         baseVertex, startInstanceLocation);
        }
        if (g_imguiRendering) {
            ++g_stats.drawsSkippedImGui;
            return oDrawIndexedInstanced(ctx, indexCountPerInstance,
                                         instanceCount, startIndex,
                                         baseVertex, startInstanceLocation);
        }
        if (ctx != g_context) {
            return oDrawIndexedInstanced(ctx, indexCountPerInstance,
                                         instanceCount, startIndex,
                                         baseVertex, startInstanceLocation);
        }

        const int cap = Config::chams3d.maxInstanceCount;
        if (cap > 0 && static_cast<int>(instanceCount) > cap) {
            ++g_stats.drawsSkippedInstance;
            return oDrawIndexedInstanced(ctx, indexCountPerInstance,
                                         instanceCount, startIndex,
                                         baseVertex, startInstanceLocation);
        }

        bool handled = TryChams(ctx, "DII", indexCountPerInstance,
                                g_stats.diiMatched, g_stats.diiOverridden,
                                [&](){ oDrawIndexedInstanced(ctx,
                                        indexCountPerInstance, instanceCount,
                                        startIndex, baseVertex,
                                        startInstanceLocation); });
        if (!handled) {
            oDrawIndexedInstanced(ctx, indexCountPerInstance,
                                  instanceCount, startIndex,
                                  baseVertex, startInstanceLocation);
        }
    }

    bool InstallHooks() {
        void** vt = *reinterpret_cast<void***>(g_context);

        g_hookTargetDI = vt[kSlotDrawIndexed];
        MH_STATUS s = MH_CreateHook(g_hookTargetDI,
                                    reinterpret_cast<void*>(&hkDrawIndexed),
                                    reinterpret_cast<void**>(&oDrawIndexed));
        if (s != MH_OK) {
            Log::Write("[Chams] MH_CreateHook(DrawIndexed) failed %d", static_cast<int>(s));
            g_hookTargetDI = nullptr;
            return false;
        }
        s = MH_EnableHook(g_hookTargetDI);
        if (s != MH_OK) {
            Log::Write("[Chams] MH_EnableHook(DrawIndexed) failed %d", static_cast<int>(s));
            return false;
        }

        g_hookTargetDII = vt[kSlotDrawIndexedInstanced];
        s = MH_CreateHook(g_hookTargetDII,
                          reinterpret_cast<void*>(&hkDrawIndexedInstanced),
                          reinterpret_cast<void**>(&oDrawIndexedInstanced));
        if (s != MH_OK) {
            Log::Write("[Chams] MH_CreateHook(DrawIndexedInstanced) failed %d", static_cast<int>(s));
            g_hookTargetDII = nullptr;
        } else {
            s = MH_EnableHook(g_hookTargetDII);
            if (s != MH_OK) {
                Log::Write("[Chams] MH_EnableHook(DrawIndexedInstanced) failed %d", static_cast<int>(s));
            } else {
                Log::Write("[Chams] DrawIndexedInstanced hooked at %p", g_hookTargetDII);
            }
        }

        Log::Write("[Chams] DrawIndexed hooked at %p", g_hookTargetDI);
        return true;
    }

    void ReleaseResources() {
        if (g_flatPS)      { g_flatPS->Release();      g_flatPS      = nullptr; }
        if (g_dssAlways)   { g_dssAlways->Release();   g_dssAlways   = nullptr; }
        if (g_dssLessEq)   { g_dssLessEq->Release();   g_dssLessEq   = nullptr; }
        if (g_opaqueBlend) { g_opaqueBlend->Release(); g_opaqueBlend = nullptr; }
        if (g_cbColor)     { g_cbColor->Release();     g_cbColor     = nullptr; }
    }
}

namespace Chams {

bool Initialize(ID3D11Device* device, ID3D11DeviceContext* context) {
    if (g_initialised) return true;
    if (!device || !context) {
        Log::Write("[Chams] Initialize called with null device/context");
        return false;
    }

    g_device  = device;
    g_context = context;

    auto fail = [&]() {
        ReleaseResources();
        g_device  = nullptr;
        g_context = nullptr;
        return false;
    };

    if (!CompileFlatColorPS()) return fail();
    if (!CreateDepthStates())  return fail();
    if (!CreateBlendState())   return fail();
    if (!CreateColorBuffer())  return fail();
    if (!InstallHooks())       return fail();

    g_initialised       = true;
    g_stats.initialised = true;
    Log::Write("[Chams] Initialized OK");
    return true;
}

void Shutdown() {
    if (!g_initialised) return;
    Log::Write("[Chams] Shutdown");

    g_initialised       = false;
    g_stats.initialised = false;

    ReleaseResources();
    g_device              = nullptr;
    g_context             = nullptr;
    oDrawIndexed          = nullptr;
    oDrawIndexedInstanced = nullptr;
    g_hookTargetDI        = nullptr;
    g_hookTargetDII       = nullptr;
}

void OnFrameBegin() {
    g_stats.diSeen                  = 0;
    g_stats.diMatched               = 0;
    g_stats.diOverridden            = 0;
    g_stats.diiSeen                 = 0;
    g_stats.diiMatched              = 0;
    g_stats.diiOverridden           = 0;
    g_stats.drawsSkippedCap         = 0;
    g_stats.drawsSkippedImGui       = 0;
    g_stats.drawsSkippedInstance    = 0;
    g_stats.drawsSkippedVSBlacklist = 0;
    g_stats.matchedSigCount         = 0;
    g_stats.matchedShaderCount      = 0;
    g_totalDrawsSeen                = 0;
}

void SetImGuiRendering(bool on) {
    g_imguiRendering = on;
}

const Stats& GetStats() {
    return g_stats;
}

        constexpr int SCAN_MAX = 8192;

                if (!chunkPtr) { i += 511; continue; }
            }

            uintptr_t ent = Interfaces::GetEntityByIndex(i);
            if (!ent) continue;

            uint8_t lifeState = Memory::Read<uint8_t>(ent + Offsets::Player::m_lifeState);
            if (lifeState != 0) continue;

            int hp = Memory::Read<int>(ent + Offsets::Player::m_iHealth);
            if (hp <= 0 || hp > 100) continue;

            uint8_t team = Memory::Read<uint8_t>(ent + Offsets::Player::m_iTeamNum);
            if (team != Teams::Terrorist && team != Teams::CT) continue;

            uintptr_t sceneNode = Memory::Read<uintptr_t>(ent + Offsets::Player::m_pGameSceneNode);
            if (!sceneNode) continue;

            Vec3 feet = Memory::Read<Vec3>(sceneNode + Offsets::SceneNode::m_vecAbsOrigin);

            ++dbg.entitiesSeen;
            if (ent == localPawn) continue;

            bool isEnemy = (team != local.Team());
            if (isEnemy) ++dbg.enemiesSeen;

            if (!Config::esp.enabled) continue;
            if (Config::esp.teamCheck && !isEnemy) continue;

            if (Config::esp.maxDistance > 0.f) {
                float dm = (feet - localOrigin).Length() * 0.01905f;
                if (dm > Config::esp.maxDistance) continue;
            }

            Vec3 head = { feet.x, feet.y, feet.z + 72.f };
            Vec2 sFeet, sHead;
            if (!Math::WorldToScreen(feet, vm, sw, sh, sFeet)) continue;
            if (!Math::WorldToScreen(head, vm, sw, sh, sHead)) continue;

            float boxH = sFeet.y - sHead.y;
            if (boxH <= 2.f) continue;
            float boxW = boxH * 0.50f;

            float x1 = sHead.x - boxW * 0.5f;
            float y1 = sHead.y;
            float x2 = sHead.x + boxW * 0.5f;
            float y2 = sFeet.y;

            const float* boxCol = isEnemy ? Config::esp.enemyColor : Config::esp.boxColor;
            ImU32 boxColor = ToU32(boxCol);

            if (Config::esp.drawChams) {
                float chamsRgba[4] = {
                    Config::esp.chamsColor[0],
                    Config::esp.chamsColor[1],
                    Config::esp.chamsColor[2],
                    Config::esp.chamsColor[3]
                };
                if (isEnemy && Config::esp.chamsEnemyColor) {

                    chamsRgba[0] = Config::esp.enemyColor[0];
                    chamsRgba[1] = Config::esp.enemyColor[1];
                    chamsRgba[2] = Config::esp.enemyColor[2];
                }
                DrawChams(dl, x1, y1, x2, y2,
                          ToU32(chamsRgba), Config::esp.chamsStyle);
            }

            if (Config::esp.drawBox) {
                DrawBox(dl, x1, y1, x2, y2, boxColor);
                ++dbg.drawnBoxes;
            }

            if (Config::esp.drawHealthBar)
                DrawHealthBar(dl, x1, y1, x2, y2, hp);

            float textY = y2 + 2.f;
            if (Config::esp.drawHealth) {
                char buf[32];
                snprintf(buf, sizeof(buf), "HP %d", hp);
                DrawText(dl, ImVec2(sHead.x - 15.f, textY), IM_COL32_WHITE, buf);
                textY += 14.f;
            }

            if (Config::esp.drawName) {
                auto it = pawnToController.find(ent);
                if (it != pawnToController.end()) {
                    char nameBuf[128];
                    if (ReadControllerName(it->second, nameBuf, sizeof(nameBuf))) {
                        ImVec2 tsz = ImGui::CalcTextSize(nameBuf);
                        DrawText(dl, ImVec2(sHead.x - tsz.x * 0.5f, y1 - tsz.y - 2.f),
                                 IM_COL32_WHITE, nameBuf);
                    }
                }
            }

            if (Config::esp.drawDistance) {
                float dm = (feet - localOrigin).Length() * 0.01905f;
                char buf[32];
                snprintf(buf, sizeof(buf), "%.0f m", dm);
                DrawText(dl, ImVec2(sHead.x - 15.f, textY), IM_COL32_WHITE, buf);
                textY += 14.f;
            }

            if (Config::esp.drawSnaplines) {
                dl->AddLine(ImVec2(sw * 0.5f, static_cast<float>(sh)),
                            ImVec2(sHead.x, y2), boxColor, 1.f);
            }
        }
    }

namespace Triggerbot {
    void Tick() {
        if (!Config::trigger.enabled) return;
        if (!Interfaces::InGame()) return;
        if (!(GetAsyncKeyState(Config::trigger.hotkey) & 0x8000)) return;

        uintptr_t localPawn = Interfaces::LocalPawn();
        if (!localPawn) return;

        LocalPlayer local(localPawn);
        if (!local.IsAlive()) return;

        int targetIdx = Memory::Read<int>(localPawn + Offsets::Player::m_iIDEntIndex);
        if (targetIdx <= 0 || targetIdx > 2048) return;

        uintptr_t target = Interfaces::GetEntityByIndex(targetIdx);
        if (!target || target == localPawn) return;

        Entity ent(target);

        uint8_t lifeState = Memory::Read<uint8_t>(target + Offsets::Player::m_lifeState);
        if (lifeState != 0) return;

        int hp = ent.Health();
        if (hp <= 0 || hp > 100) return;

        int team = ent.Team();
        if (team != Teams::Terrorist && team != Teams::CT) return;

        if (Config::trigger.teamCheck && team == local.Team()) return;

        auto now = std::chrono::steady_clock::now();
        auto sinceLast = std::chrono::duration_cast<std::chrono::milliseconds>(
            now - g_lastShot).count();
        if (sinceLast < Config::trigger.delayMs) return;

        INPUT down{}, up{};
        down.type = INPUT_MOUSE;
        down.mi.dwFlags = MOUSEEVENTF_LEFTDOWN;
        up.type = INPUT_MOUSE;
        up.mi.dwFlags = MOUSEEVENTF_LEFTUP;

        SendInput(1, &down, sizeof(INPUT));
        Sleep(15);
        SendInput(1, &up, sizeof(INPUT));

        g_lastShot = std::chrono::steady_clock::now();
    }
}

    bool GetSwapChainVTable(void** vtable) {
        WNDCLASSEXA wc{};
        wc.cbSize        = sizeof(wc);
        wc.lpfnWndProc   = DefWindowProcA;
        wc.hInstance     = GetModuleHandleA(nullptr);
        wc.lpszClassName = "RabsztynDummy";

                                  0, 0, 100, 100, nullptr, nullptr, wc.hInstance, nullptr);

            return false;
        }

        DXGI_SWAP_CHAIN_DESC scd{};
        scd.BufferCount        = 1;
        scd.BufferDesc.Format  = DXGI_FORMAT_R8G8B8A8_UNORM;
        scd.BufferUsage        = DXGI_USAGE_RENDER_TARGET_OUTPUT;
        scd.OutputWindow       = hwnd;
        scd.SampleDesc.Count   = 1;
        scd.Windowed           = TRUE;
        scd.SwapEffect         = DXGI_SWAP_EFFECT_DISCARD;

        HRESULT hr = D3D11CreateDeviceAndSwapChain(nullptr, D3D_DRIVER_TYPE_HARDWARE,
                                                   nullptr, 0, levels, _countof(levels),
                                                   D3D11_SDK_VERSION, &scd, &swap,
                                                   &dev, &fl, &ctx);
        if (FAILED(hr)) {
            Log::Write("[Hooks] D3D11CreateDeviceAndSwapChain failed 0x%08X", hr);
            DestroyWindow(hwnd);
            UnregisterClassA(wc.lpszClassName, wc.hInstance);
            return false;
        }

        memcpy(vtable, *reinterpret_cast<void***>(swap), sizeof(void*) * 18);

        swap->Release();
        ctx->Release();
        dev->Release();
        DestroyWindow(hwnd);
        UnregisterClassA(wc.lpszClassName, wc.hInstance);

    inline std::mutex g_mutex;

    inline void Write(const char* fmt, ...) {
        std::lock_guard<std::mutex> lk(g_mutex);
        FILE* f = nullptr;
        fopen_s(&f, PathBuf().c_str(), "a");
        if (!f) return;

        SYSTEMTIME st;
        GetLocalTime(&st);
        fprintf(f, "[%02d:%02d:%02d.%03d] ", st.wHour, st.wMinute, st.wSecond, st.wMilliseconds);

        va_list ap;
        va_start(ap, fmt);
        vfprintf(f, fmt, ap);
        va_end(ap);

        fputc('\n', f);
        fclose(f);
    }

    inline void Clear() {
        std::lock_guard<std::mutex> lk(g_mutex);
        FILE* f = nullptr;
        fopen_s(&f, PathBuf().c_str(), "w");
        if (f) fclose(f);
    }

    struct EnumWindowData {

    static BOOL CALLBACK EnumProc(HWND hwnd, LPARAM lParam) {
        auto* data = reinterpret_cast<EnumWindowData*>(lParam);
        DWORD wndPid = 0;
        GetWindowThreadProcessId(hwnd, &wndPid);
        if (wndPid == data->pid && GetWindow(hwnd, GW_OWNER) == nullptr && IsWindowVisible(hwnd)) {
            data->hwnd = hwnd;

        }

    }

        return data.hwnd;

    static void ParsePattern(const char* pattern, std::vector<int>& bytes) {
        for (const char* p = pattern; *p;) {
            if (*p == ' ') { ++p; continue; }
            if (*p == '?') {
                bytes.push_back(-1);
                ++p;
                if (*p == '?') ++p;
            } else {
                bytes.push_back(static_cast<int>(strtoul(p, const_cast<char**>(&p), 16)));
            }
        }
    }

    uintptr_t PatternScan(HMODULE module, const char* pattern) {
        if (!module || !pattern) return 0;

        auto dos = reinterpret_cast<PIMAGE_DOS_HEADER>(module);
        auto nt  = reinterpret_cast<PIMAGE_NT_HEADERS>(
            reinterpret_cast<uint8_t*>(module) + dos->e_lfanew);
        const size_t size = nt->OptionalHeader.SizeOfImage;
        auto base = reinterpret_cast<uint8_t*>(module);

        std::vector<int> bytes;
        ParsePattern(pattern, bytes);
        if (bytes.empty()) return 0;

        for (size_t i = 0; i + bytes.size() <= size; ++i) {
            bool found = true;
            for (size_t j = 0; j < bytes.size(); ++j) {
                if (bytes[j] != -1 && base[i + j] != static_cast<uint8_t>(bytes[j])) {
                    found = false;
                    break;
                }
            }
            if (found)
                return reinterpret_cast<uintptr_t>(base + i);
        }
        return 0;
    }
}

    template <typename T>
    inline T Read(uintptr_t address) {
        if (!address) return T{};
        if (DriverRouting()) {
            T value{};
            if (DriverReadRaw(address, &value, sizeof(T)))
                return value;

        }
        __try {
            return *reinterpret_cast<T*>(address);
        }
        __except (EXCEPTION_EXECUTE_HANDLER) {
            return T{};
        }
    }

    template <typename T>
    inline void Write(uintptr_t address, const T& value) {
        if (!address) return;
        if (DriverRouting()) {
            if (DriverWriteRaw(address, &value, sizeof(T)))
                return;
        }
        __try {
            *reinterpret_cast<T*>(address) = value;
        }
        __except (EXCEPTION_EXECUTE_HANDLER) {}
    }

};

static inline uint8_t Xtime(uint8_t x) {
    return uint8_t((x << 1) ^ ((x & 0x80) ? 0x1B : 0x00));
}

void Aes128::SetKey(const uint8_t key[16]) {
    std::memcpy(rk, key, 16);
    for (int i = 4; i < 44; ++i) {
        uint8_t t0 = rk[(i - 1) * 4 + 0];
        uint8_t t1 = rk[(i - 1) * 4 + 1];
        uint8_t t2 = rk[(i - 1) * 4 + 2];
        uint8_t t3 = rk[(i - 1) * 4 + 3];
        if (i % 4 == 0) {
            uint8_t r0 = kSbox[t1] ^ kRcon[i / 4];
            uint8_t r1 = kSbox[t2];
            uint8_t r2 = kSbox[t3];
            uint8_t r3 = kSbox[t0];
            t0 = r0; t1 = r1; t2 = r2; t3 = r3;
        }
        rk[i * 4 + 0] = rk[(i - 4) * 4 + 0] ^ t0;
        rk[i * 4 + 1] = rk[(i - 4) * 4 + 1] ^ t1;
        rk[i * 4 + 2] = rk[(i - 4) * 4 + 2] ^ t2;
        rk[i * 4 + 3] = rk[(i - 4) * 4 + 3] ^ t3;
    }
}

static void SubBytes(uint8_t s[16]) {
    for (int i = 0; i < 16; ++i) s[i] = kSbox[s[i]];
}
static void ShiftRows(uint8_t s[16]) {
    uint8_t t;

    t = s[1]; s[1] = s[5]; s[5] = s[9]; s[9] = s[13]; s[13] = t;

    t = s[2]; s[2] = s[10]; s[10] = t;
    t = s[6]; s[6] = s[14]; s[14] = t;

    t = s[15]; s[15] = s[11]; s[11] = s[7]; s[7] = s[3]; s[3] = t;
}
static void MixColumns(uint8_t s[16]) {
    for (int c = 0; c < 4; ++c) {
        uint8_t* col = s + c * 4;
        uint8_t a0 = col[0], a1 = col[1], a2 = col[2], a3 = col[3];
        uint8_t x  = a0 ^ a1 ^ a2 ^ a3;
        col[0] ^= Xtime(a0 ^ a1) ^ x;
        col[1] ^= Xtime(a1 ^ a2) ^ x;
        col[2] ^= Xtime(a2 ^ a3) ^ x;
        col[3] ^= Xtime(a3 ^ a0) ^ x;
    }
}
static void AddRoundKey(uint8_t s[16], const uint8_t* rk) {
    for (int i = 0; i < 16; ++i) s[i] ^= rk[i];
}

void Aes128::EncryptBlock(const uint8_t in[16], uint8_t out[16]) const {
    uint8_t s[16];
    std::memcpy(s, in, 16);
    AddRoundKey(s, rk);
    for (int round = 1; round < 10; ++round) {
        SubBytes(s);
        ShiftRows(s);
        MixColumns(s);
        AddRoundKey(s, rk + round * 16);
    }
    SubBytes(s);
    ShiftRows(s);
    AddRoundKey(s, rk + 10 * 16);
    std::memcpy(out, s, 16);
}

void Ctr128(const uint8_t key[16], const uint8_t iv[16],
            const void* src, void* dst, size_t bytes) {
    Aes128 aes;
    aes.SetKey(key);

    uint8_t counter[16];
    std::memcpy(counter, iv, 16);

    auto* in  = static_cast<const uint8_t*>(src);
    auto* out = static_cast<uint8_t*>(dst);

    uint8_t block[16];
    while (bytes) {
        aes.EncryptBlock(counter, block);
        size_t chunk = bytes < 16 ? bytes : 16;
        for (size_t i = 0; i < chunk; ++i)
            out[i] = uint8_t(in[i] ^ block[i]);
        in     += chunk;
        out    += chunk;
        bytes  -= chunk;

        for (int i = 15; i >= 0; --i) {
            if (++counter[i]) break;
        }
    }
}

void XorRegion(void* buf, size_t bytes,
               const uint8_t* key, size_t keyLen) {
    if (!key || !keyLen) return;
    auto* p = static_cast<uint8_t*>(buf);
    for (size_t i = 0; i < bytes; ++i)
        p[i] = uint8_t(p[i] ^ key[i % keyLen]);
}

namespace Region {

bool Scramble(void* addr, size_t bytes,
              const uint8_t* key, size_t keyLen) {
    if (!addr || !bytes || !key || !keyLen) return false;

    ULONG oldProt = 0;

    if (Protection::Syscalls::IsReady()) {
        PVOID  base = addr;
        SIZE_T sz   = bytes;
        NTSTATUS st = Protection::Syscalls::NtProtectVirtualMemory(
            GetCurrentProcess(), &base, &sz, PAGE_READWRITE, &oldProt);
        if (!Protection::Syscalls::NT_SUCCESS_(st)) return false;

        XorRegion(addr, bytes, key, keyLen);

        ULONG dummy = 0;
        Protection::Syscalls::NtProtectVirtualMemory(
            GetCurrentProcess(), &base, &sz, oldProt, &dummy);
        FlushInstructionCache(GetCurrentProcess(), addr, bytes);
        return true;
    }

    DWORD prevProt = 0;
    if (!VirtualProtect(addr, bytes, PAGE_READWRITE, &prevProt)) return false;
    XorRegion(addr, bytes, key, keyLen);
    DWORD dummy = 0;
    VirtualProtect(addr, bytes, prevProt, &dummy);
    FlushInstructionCache(GetCurrentProcess(), addr, bytes);
    return true;
}

bool HasHardwareBreakpoint() {

    HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPTHREAD, 0);
    if (snap == INVALID_HANDLE_VALUE) return false;

    DWORD ownPid = GetCurrentProcessId();
    DWORD selfTid = GetCurrentThreadId();
    THREADENTRY32 te{ sizeof(te) };
    bool tripped = false;

    if (Thread32First(snap, &te)) {
        do {
            if (te.th32OwnerProcessID != ownPid) continue;
            if (te.th32ThreadID       == selfTid) continue;

            HANDLE ht = OpenThread(THREAD_GET_CONTEXT, FALSE, te.th32ThreadID);
            if (!ht) continue;

            CONTEXT ctx{};
            ctx.ContextFlags = CONTEXT_DEBUG_REGISTERS;

            NTSTATUS st = Protection::Syscalls::IsReady()
                ? Protection::Syscalls::NtGetContextThread(ht, &ctx)
                : (GetThreadContext(ht, &ctx) ? 0 : -1);

            if (st == 0 && (ctx.Dr0 | ctx.Dr1 | ctx.Dr2 | ctx.Dr3)) {
                tripped = true;
            }
            CloseHandle(ht);
            if (tripped) break;
        } while (Thread32Next(snap, &te));
    }
    CloseHandle(snap);
    return tripped;

    bool DispatchLocked(unsigned int cmd, unsigned timeoutMs) {
        SHARED_MEMORY* shm = g_state.shm;
        if (!shm) return false;

        shm->Command = cmd;
        shm->Status  = STATUS_COMM_PENDING;

        for (int i = 0; i < 2000; ++i) {
            if (shm->Status != STATUS_COMM_PENDING) {
                bool ok = shm->Status == STATUS_COMM_OK;
                (ok ? g_state.successCount : g_state.failureCount)
                    .fetch_add(1, std::memory_order_relaxed);
                return ok;
            }
            _mm_pause();
        }

        DWORD start = GetTickCount();
        while (shm->Status == STATUS_COMM_PENDING) {
            if (GetTickCount() - start > timeoutMs) {
                g_state.failureCount.fetch_add(1, std::memory_order_relaxed);
                return false;
            }
            if (!SwitchToThread()) _mm_pause();
        }
        bool ok = shm->Status == STATUS_COMM_OK;
        (ok ? g_state.successCount : g_state.failureCount)
            .fetch_add(1, std::memory_order_relaxed);
        return ok;
    }

    unsigned Timeout() { return g_state.defaultTimeoutMs; }

    bool KernelServiced() {
        return g_state.mode == Mode::Client &&
               g_state.driverPresent.load(std::memory_order_relaxed);
    }

}

bool Init(const InitOptions& opts) {
    if (g_state.mode != Mode::Uninitialized) return true;
    g_state.defaultTimeoutMs = opts.defaultTimeoutMs;

    NTSTATUS st = TryOpenExisting(opts);
    if (!Protection::Syscalls::NT_SUCCESS_(st)) {
        if (st != kStatusObjectNameNotFound &&
            st != kStatusObjectTypeMismatch) {
            Log::Write("[Driver] NtOpenSection failed 0x%08lX", st);
        }
        if (!opts.createIfMissing) return false;

        st = TryCreateNew(opts);
        if (!Protection::Syscalls::NT_SUCCESS_(st)) {
            Log::Write("[Driver] NtCreateSection failed 0x%08lX", st);
            return false;
        }

        std::memset(g_state.shm, 0, sizeof(SHARED_MEMORY));
        g_state.shm->Status = STATUS_COMM_IDLE;
    }

    g_state.report.ok         = true;
    g_state.report.mode       = g_state.mode;
    g_state.report.sharedBase = g_state.viewBase;
    g_state.report.sharedSize = g_state.viewSize;

    bool live = false;
    if (g_state.mode == Mode::Client) {
        std::lock_guard<std::mutex> lk(g_state.txMutex);
        live = DispatchLocked(CMD_PING, 200);
    }
    g_state.driverPresent.store(live, std::memory_order_relaxed);

    Log::Write("[Driver] Init ok: mode=%s base=%p size=%zu driverLive=%d",
               g_state.mode == Mode::Client ? "Client" : "Standalone",
               g_state.viewBase, g_state.viewSize, int(live));
    return true;
}

void Shutdown() {
    if (g_state.viewBase) {
        Protection::Syscalls::NtUnmapViewOfSection(GetCurrentProcess(),
                                                   g_state.viewBase);
        g_state.viewBase = nullptr;
        g_state.shm      = nullptr;
    }
    if (g_state.sectionHandle) {
        Protection::Syscalls::NtClose(g_state.sectionHandle);
        g_state.sectionHandle = nullptr;
    }
    g_state.mode          = Mode::Uninitialized;
    g_state.driverPresent.store(false, std::memory_order_relaxed);
    g_state.successCount.store(0, std::memory_order_relaxed);
    g_state.failureCount.store(0, std::memory_order_relaxed);
    g_state.report        = Report{};
}

bool IsAvailable()      { return g_state.shm != nullptr; }
bool IsDriverPresent()  { return KernelServiced(); }

const Report& GetReport() {
    g_state.report.successCount = g_state.successCount.load(std::memory_order_relaxed);
    g_state.report.failureCount = g_state.failureCount.load(std::memory_order_relaxed);
    return g_state.report;
}

bool Ping() {
    if (!IsAvailable()) return false;
    std::lock_guard<std::mutex> lk(g_state.txMutex);
    bool ok = DispatchLocked(CMD_PING, Timeout());

    g_state.driverPresent.store(ok, std::memory_order_relaxed);
    return ok;
}

DWORD GetProcessId(const wchar_t* imageName) {
    if (!KernelServiced() || !imageName) return 0;
    std::lock_guard<std::mutex> lk(g_state.txMutex);

    std::wcsncpy(g_state.shm->Name, imageName,
                 sizeof(g_state.shm->Name) / sizeof(wchar_t) - 1);
    g_state.shm->Name[sizeof(g_state.shm->Name) / sizeof(wchar_t) - 1] = 0;

    if (!DispatchLocked(CMD_GET_PID, Timeout())) return 0;
    return static_cast<DWORD>(g_state.shm->OutAddress);
}

ModuleInfo GetModuleBase(DWORD pid, const wchar_t* moduleName) {
    ModuleInfo r{ 0, 0 };
    if (!KernelServiced() || !moduleName) return r;
    std::lock_guard<std::mutex> lk(g_state.txMutex);

    g_state.shm->ProcessId = pid;
    std::wcsncpy(g_state.shm->Name, moduleName,
                 sizeof(g_state.shm->Name) / sizeof(wchar_t) - 1);
    g_state.shm->Name[sizeof(g_state.shm->Name) / sizeof(wchar_t) - 1] = 0;

    if (!DispatchLocked(CMD_GET_BASE, Timeout())) return r;
    r.base = g_state.shm->OutAddress;
    r.size = g_state.shm->OutSize;
    return r;
}

bool ReadMemory(DWORD pid, uint64_t address, void* out, size_t bytes) {
    if (!KernelServiced() || !out || !bytes) return false;
    if (bytes > sizeof(g_state.shm->Buffer)) return false;
    std::lock_guard<std::mutex> lk(g_state.txMutex);

    g_state.shm->ProcessId = pid;
    g_state.shm->Address   = address;
    g_state.shm->Size      = bytes;

    if (!DispatchLocked(CMD_READ, Timeout())) return false;
    std::memcpy(out, g_state.shm->Buffer, bytes);
    return true;
}

    if (!KernelServiced() || !tid) return false;

    return DispatchLocked(CMD_HIDE_THREAD, Timeout());

    g_state.shm->Address = tid;
    return DispatchLocked(CMD_HIDE_THREAD_STRONG, Timeout());

bool WriteMemory(DWORD pid, uint64_t address, const void* in, size_t bytes) {
    if (!KernelServiced() || !in || !bytes) return false;
    if (bytes > sizeof(g_state.shm->Buffer)) return false;
    std::lock_guard<std::mutex> lk(g_state.txMutex);

    g_state.shm->ProcessId = pid;
    g_state.shm->Address   = address;
    g_state.shm->Size      = bytes;
    std::memcpy(const_cast<unsigned char*>(g_state.shm->Buffer), in, bytes);
    return DispatchLocked(CMD_WRITE, Timeout());
}

int BatchBuilder::Add(uint64_t address, unsigned bytes) {
    if (!IsAvailable()) return -1;
    if (count >= MAX_BATCH_ENTRIES) return -1;
    if (bytes == 0 || address == 0) return -1;
    if (bufferUsed + bytes > sizeof(g_state.shm->Buffer)) return -1;

    BATCH_ENTRY& e = const_cast<BATCH_ENTRY&>(
        g_state.shm->BatchEntries[count]);
    e.Address      = address;
    e.Size         = bytes;
    e.ResultOffset = bufferUsed;
    e.Success      = 0;
    e._pad         = 0;
    offsets[count] = bufferUsed;
    bufferUsed += bytes;
    return static_cast<int>(count++);
}

bool BatchBuilder::Dispatch(DWORD pid) {
    if (count == 0) return true;
    if (!KernelServiced()) return false;
    std::lock_guard<std::mutex> lk(g_state.txMutex);

    g_state.shm->ProcessId  = pid;
    g_state.shm->BatchCount = count;
    return DispatchLocked(CMD_READ_BATCH, Timeout());
}

bool BatchBuilder::Ok(int id) const {
    if (id < 0 || static_cast<unsigned>(id) >= count) return false;
    if (!IsAvailable()) return false;
    return g_state.shm->BatchEntries[id].Success != 0;
}

bool BatchBuilder::GetRaw(int id, void* dst, unsigned bytes) const {
    if (!Ok(id) || !dst) return false;
    if (bytes > g_state.shm->BatchEntries[id].Size) return false;
    std::memcpy(dst, g_state.shm->Buffer + offsets[id], bytes);
    return true;
}

BatchBuilder BeginBatch() { return BatchBuilder{}; }

        LIST_ENTRY     InMemoryOrderModuleList;
        LIST_ENTRY     InInitializationOrderModuleList;
        PVOID          BaseAddress;

    uint32_t HashLowerAscii(const char* s) {
        uint32_t h = hash_detail::FNV_OFFSET;
        while (*s) {
            h ^= static_cast<uint8_t>(hash_detail::ToLowerAscii(*s));
            h *= hash_detail::FNV_PRIME;
            ++s;
        }
        return h;
    }

    struct ResolvedCache {
        std::mutex mtx;
        std::unordered_map<uint64_t, void*> map;
    };

uint32_t HashWideLower(const wchar_t* s, size_t chars) {
    uint32_t h = hash_detail::FNV_OFFSET;
    for (size_t i = 0; i < chars; ++i) {
        wchar_t c = s[i];
        if (c >= L'A' && c <= L'Z') c = wchar_t(c + 32);

        h ^= static_cast<uint8_t>(c & 0xFF);
        h *= hash_detail::FNV_PRIME;
    }
    return h;
}

void* ResolveModule(uint32_t moduleHash) {
    auto* peb = reinterpret_cast<PEB_MIN*>(__readgsqword(0x60));
    if (!peb || !peb->Ldr) return nullptr;

    LIST_ENTRY* head = &peb->Ldr->InLoadOrderModuleList;
    for (LIST_ENTRY* it = head->Flink; it && it != head; it = it->Flink) {
        auto* mod = CONTAINING_RECORD(it, LDR_MODULE_MIN, InLoadOrderModuleList);
        if (!mod->BaseAddress || !mod->BaseDllName.Buffer) continue;

        size_t chars = mod->BaseDllName.Length / sizeof(wchar_t);
        if (HashWideLower(mod->BaseDllName.Buffer, chars) == moduleHash)
            return mod->BaseAddress;
    }
    return nullptr;
}

void* ResolveExport(void* moduleBase, uint32_t functionHash) {
    if (!moduleBase) return nullptr;

    auto* base = static_cast<uint8_t*>(moduleBase);
    auto* dos  = reinterpret_cast<IMAGE_DOS_HEADER*>(base);
    if (dos->e_magic != IMAGE_DOS_SIGNATURE) return nullptr;

    auto* nt = reinterpret_cast<IMAGE_NT_HEADERS*>(base + dos->e_lfanew);
    if (nt->Signature != IMAGE_NT_SIGNATURE) return nullptr;

    auto& d = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXPORT];
    if (!d.VirtualAddress || !d.Size) return nullptr;

    auto* exp = reinterpret_cast<IMAGE_EXPORT_DIRECTORY*>(base + d.VirtualAddress);
    auto* names = reinterpret_cast<uint32_t*>(base + exp->AddressOfNames);
    auto* funcs = reinterpret_cast<uint32_t*>(base + exp->AddressOfFunctions);
    auto* ords  = reinterpret_cast<uint16_t*>(base + exp->AddressOfNameOrdinals);

    for (DWORD i = 0; i < exp->NumberOfNames; ++i) {
        const char* name = reinterpret_cast<const char*>(base + names[i]);
        if (HashLowerAscii(name) != functionHash) continue;

        DWORD funcRva = funcs[ords[i]];
        void* addr    = base + funcRva;

        if (funcRva >= d.VirtualAddress && funcRva < d.VirtualAddress + d.Size) {
            const char* fwd = reinterpret_cast<const char*>(addr);
            const char* dot = std::strchr(fwd, '.');
            if (!dot) return nullptr;

            char modBuf[64]{};
            size_t mlen = static_cast<size_t>(dot - fwd);
            if (mlen >= sizeof(modBuf)) return nullptr;
            std::memcpy(modBuf, fwd, mlen);
            modBuf[mlen] = 0;

            char withExt[80]{};
            std::snprintf(withExt, sizeof(withExt), "%s.dll", modBuf);

            uint32_t fwdModHash  = HashLowerAscii(withExt);
            uint32_t fwdFuncHash = HashLowerAscii(dot + 1);

            void* fwdMod = ResolveModule(fwdModHash);
            if (!fwdMod) return nullptr;
            return ResolveExport(fwdMod, fwdFuncHash);
        }
        return addr;
    }
    return nullptr;
}

void* Resolve(uint32_t moduleHash, uint32_t functionHash) {
    uint64_t key = (static_cast<uint64_t>(moduleHash) << 32) | functionHash;

    auto& c = cache();
    {
        std::lock_guard<std::mutex> lk(c.mtx);
        auto it = c.map.find(key);
        if (it != c.map.end()) return it->second;
    }

    void* mod = ResolveModule(moduleHash);
    void* fn  = ResolveExport(mod, functionHash);

    {
        std::lock_guard<std::mutex> lk(c.mtx);
        c.map.emplace(key, fn);
    }
    return fn;
}

void ResetCache() {
    auto& c = cache();
    std::lock_guard<std::mutex> lk(c.mtx);
    c.map.clear();
}

        }

bool InitEarly(const Config& cfg) {
    if (g_earlyReady.load()) return true;
    g_cfg = cfg;

    if (g_cfg.enableSyscalls) {
        if (!Protection::Syscalls::Init()) {
            Log::Write("[Manager] Syscalls::Init failed - continuing without direct syscalls");
        }
    }

    if (g_cfg.enableDriver) {
        if (!Protection::Driver::Init()) {
            Log::Write("[Manager] Driver::Init failed - no kernel channel");
        }
    }
    RefreshDriverStats();

    if (g_cfg.enableDetection) {
        Log::Write("[Manager] Detection engine armed");
    }

    }

    Memory::SetTargetPid(selfPid);
    g_snap.targetPid = selfPid;

               int(g_snap.syscallsReady), int(g_snap.driverAvailable));

    if (!g_earlyReady.load()) {
        Log::Write("[Manager] InitPostHooks called before InitEarly");
        return;

    if (g_postReady.load()) return;
    g_selfMod = selfModule;

    if (g_cfg.enablePEWipe && selfModule) {
        if (!Protection::PEWipe::Wipe(selfModule)) {
            Log::Write("[Manager] PEWipe::Wipe failed (non-fatal)");
        }

bool Tick() {
    if (g_shutdownReq.load()) return false;

    ++g_tickIndex;
    g_snap.tickCount = g_tickIndex;

    if ((g_tickIndex & 0x0F) == 0) {
        RefreshDriverStats();
    }

    if (g_cfg.enableDetection &&
        g_cfg.detectionEveryTicks > 0 &&
        (g_tickIndex % g_cfg.detectionEveryTicks) == 0) {
        RunDetectionSweep();
    }

    if (g_cfg.enableSignatureBuster) {
        Protection::SignatureBuster::Tick();
        const auto& br = Protection::SignatureBuster::GetReport();
        g_snap.busterCycles         = br.scrambleCycles;
        g_snap.busterLastScrambled  = br.totalScrambled;
        g_snap.busterRegionCount    = br.registeredCount;
    }

    return !g_shutdownReq.load();
}

    Protection::Driver::Shutdown();
    Protection::Syscalls::Shutdown();

    g_earlyReady.store(false);
    g_postReady.store(false);
    g_snap = Snapshot{};
}

    constexpr LONG kStatusSuccess       = 0x00000000L;
    constexpr LONG kStatusInvalidImage  = static_cast<LONG>(0xC000007B);
    constexpr LONG kStatusNoMemory      = static_cast<LONG>(0xC0000017);
    constexpr LONG kStatusUnsuccessful  = static_cast<LONG>(0xC0000001);

    void* AllocImage(size_t size) {
        if (Protection::Syscalls::IsReady()) {
            PVOID  base = nullptr;
            SIZE_T sz   = size;
            NTSTATUS st = Protection::Syscalls::NtAllocateVirtualMemory(
                GetCurrentProcess(), &base, 0, &sz,
                MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
            if (Protection::Syscalls::NT_SUCCESS_(st)) return base;
        }
        return VirtualAlloc(nullptr, size, MEM_COMMIT | MEM_RESERVE,
                            PAGE_READWRITE);
    }

    void FreeImage(void* base, size_t size) {
        if (!base) return;
        if (Protection::Syscalls::IsReady()) {
            PVOID  p  = base;
            SIZE_T sz = 0;
            (void)size;
            Protection::Syscalls::NtFreeVirtualMemory(
                GetCurrentProcess(), &p, &sz, MEM_RELEASE);
            return;
        }
        VirtualFree(base, 0, MEM_RELEASE);
    }

    bool Protect(void* addr, size_t size, ULONG newProt) {
        if (Protection::Syscalls::IsReady()) {
            PVOID  p  = addr;
            SIZE_T sz = size;
            ULONG  old = 0;
            NTSTATUS st = Protection::Syscalls::NtProtectVirtualMemory(
                GetCurrentProcess(), &p, &sz, newProt, &old);
            return Protection::Syscalls::NT_SUCCESS_(st);
        }
        DWORD old = 0;
        return VirtualProtect(addr, size, newProt, &old) != 0;
    }

    ULONG SectionProt(DWORD c) {
        bool r = (c & IMAGE_SCN_MEM_READ)    != 0;
        bool w = (c & IMAGE_SCN_MEM_WRITE)   != 0;
        bool x = (c & IMAGE_SCN_MEM_EXECUTE) != 0;
        if (x && w) return PAGE_EXECUTE_READWRITE;
        if (x && r) return PAGE_EXECUTE_READ;
        if (x)      return PAGE_EXECUTE;
        if (w)      return PAGE_READWRITE;
        return PAGE_READONLY;
    }

    bool ApplyRelocations(uint8_t* base, IMAGE_NT_HEADERS* nt, uint64_t delta) {
        if (delta == 0) return true;
        auto& d = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_BASERELOC];
        if (!d.VirtualAddress || !d.Size) return true;

        auto* block = reinterpret_cast<IMAGE_BASE_RELOCATION*>(base + d.VirtualAddress);
        auto* end   = reinterpret_cast<IMAGE_BASE_RELOCATION*>(
            reinterpret_cast<uint8_t*>(block) + d.Size);

        while (block < end && block->SizeOfBlock) {
            DWORD count = (block->SizeOfBlock - sizeof(*block)) / sizeof(WORD);
            auto* entries = reinterpret_cast<WORD*>(block + 1);
            uint8_t* pageBase = base + block->VirtualAddress;

            for (DWORD i = 0; i < count; ++i) {
                WORD e = entries[i];
                WORD type   = WORD(e >> 12);
                WORD offset = WORD(e & 0x0FFF);

                switch (type) {
                    case IMAGE_REL_BASED_ABSOLUTE:
                        break;
                    case IMAGE_REL_BASED_HIGHLOW: {
                        auto* p = reinterpret_cast<uint32_t*>(pageBase + offset);
                        *p += static_cast<uint32_t>(delta);
                        break;
                    }
                    case IMAGE_REL_BASED_DIR64: {
                        auto* p = reinterpret_cast<uint64_t*>(pageBase + offset);
                        *p += delta;
                        break;
                    }
                    default:
                        Log::Write("[Mapper] unsupported reloc type %u", type);
                        return false;
                }
            }
            block = reinterpret_cast<IMAGE_BASE_RELOCATION*>(
                reinterpret_cast<uint8_t*>(block) + block->SizeOfBlock);
        }
        return true;
    }

    bool ResolveImports(uint8_t* base, IMAGE_NT_HEADERS* nt) {
        auto& d = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_IMPORT];
        if (!d.VirtualAddress || !d.Size) return true;

        auto* desc = reinterpret_cast<IMAGE_IMPORT_DESCRIPTOR*>(base + d.VirtualAddress);
        for (; desc->Name; ++desc) {
            const char* modName = reinterpret_cast<const char*>(base + desc->Name);

            void* mod = Protection::Imports::ResolveModule(
                Protection::Imports::Hash(modName));

            if (!mod) {

                mod = LoadLibraryA(modName);
                if (!mod) {
                    Log::Write("[Mapper] failed to load import module '%s'", modName);
                    return false;
                }
            }

            auto* origThunk = reinterpret_cast<IMAGE_THUNK_DATA*>(
                base + (desc->OriginalFirstThunk ? desc->OriginalFirstThunk
                                                 : desc->FirstThunk));
            auto* iat = reinterpret_cast<IMAGE_THUNK_DATA*>(base + desc->FirstThunk);

            for (; origThunk->u1.AddressOfData; ++origThunk, ++iat) {
                void* fn = nullptr;
                if (IMAGE_SNAP_BY_ORDINAL(origThunk->u1.Ordinal)) {

                    WORD ord = IMAGE_ORDINAL(origThunk->u1.Ordinal);
                    fn = reinterpret_cast<void*>(
                        GetProcAddress(static_cast<HMODULE>(mod),
                                       MAKEINTRESOURCEA(ord)));
                } else {
                    auto* import = reinterpret_cast<IMAGE_IMPORT_BY_NAME*>(
                        base + origThunk->u1.AddressOfData);
                    fn = Protection::Imports::ResolveExport(
                        mod, Protection::Imports::Hash(import->Name));
                }
                if (!fn) {
                    Log::Write("[Mapper] unresolved import in '%s'", modName);
                    return false;
                }
                iat->u1.Function = reinterpret_cast<ULONGLONG>(fn);
            }
        }
        return true;
    }

    bool ApplySectionProtections(uint8_t* base, IMAGE_NT_HEADERS* nt) {
        auto* sec = IMAGE_FIRST_SECTION(nt);
        for (WORD i = 0; i < nt->FileHeader.NumberOfSections; ++i, ++sec) {
            if (!sec->Misc.VirtualSize) continue;
            void* p = base + sec->VirtualAddress;
            ULONG prot = SectionProt(sec->Characteristics);
            if (!Protect(p, sec->Misc.VirtualSize, prot)) {
                Log::Write("[Mapper] failed to protect section %.*s",
                           8, sec->Name);
                return false;
            }
        }
        return true;
    }

}

MapResult MapPeImage(const void* peBytes, size_t peSize) {
    MapResult r{ kStatusInvalidImage, nullptr, 0, 0 };
    if (!peBytes || peSize < sizeof(IMAGE_DOS_HEADER)) return r;

    auto* dos = static_cast<const IMAGE_DOS_HEADER*>(peBytes);
    if (dos->e_magic != IMAGE_DOS_SIGNATURE) return r;

    auto* nt = reinterpret_cast<const IMAGE_NT_HEADERS*>(
        static_cast<const uint8_t*>(peBytes) + dos->e_lfanew);
    if (nt->Signature != IMAGE_NT_SIGNATURE) return r;
    if (nt->OptionalHeader.Magic != IMAGE_NT_OPTIONAL_HDR_MAGIC) return r;

    const size_t imageSize = nt->OptionalHeader.SizeOfImage;
    if (imageSize < peSize && peSize < nt->OptionalHeader.SizeOfHeaders) return r;

    uint8_t* base = static_cast<uint8_t*>(AllocImage(imageSize));
    if (!base) { r.status = kStatusNoMemory; return r; }

    std::memcpy(base, peBytes, nt->OptionalHeader.SizeOfHeaders);

    auto* mutableNt = reinterpret_cast<IMAGE_NT_HEADERS*>(base + dos->e_lfanew);
    auto* sec = IMAGE_FIRST_SECTION(mutableNt);
    for (WORD i = 0; i < mutableNt->FileHeader.NumberOfSections; ++i, ++sec) {
        if (!sec->SizeOfRawData) continue;
        if (sec->PointerToRawData + sec->SizeOfRawData > peSize) {
            FreeImage(base, imageSize);
            r.status = kStatusInvalidImage; r.base = nullptr;
            return r;
        }
        std::memcpy(base + sec->VirtualAddress,
                    static_cast<const uint8_t*>(peBytes) + sec->PointerToRawData,
                    sec->SizeOfRawData);
    }

    uint64_t delta = reinterpret_cast<uint64_t>(base)
                   - mutableNt->OptionalHeader.ImageBase;
    if (!ApplyRelocations(base, mutableNt, delta)) {
        FreeImage(base, imageSize);
        r.status = kStatusUnsuccessful; r.base = nullptr; return r;
    }
    mutableNt->OptionalHeader.ImageBase = reinterpret_cast<ULONGLONG>(base);

    if (!ResolveImports(base, mutableNt)) {
        FreeImage(base, imageSize);
        r.status = kStatusUnsuccessful; r.base = nullptr; return r;
    }

    if (!ApplySectionProtections(base, mutableNt)) {
        FreeImage(base, imageSize);
        r.status = kStatusUnsuccessful; r.base = nullptr; return r;
    }

    FlushInstructionCache(GetCurrentProcess(), base, imageSize);

    r.status    = kStatusSuccess;
    r.base      = base;
    r.entryRva  = mutableNt->OptionalHeader.AddressOfEntryPoint;
    r.imageSize = imageSize;
    return r;
}

using DllMain_t = BOOL (WINAPI*)(HINSTANCE, DWORD, LPVOID);

BOOL CallDllMain(const MapResult& img, DWORD reason) {
    if (!img.base || !img.entryRva) return TRUE;
    auto* entry = reinterpret_cast<DllMain_t>(
        static_cast<uint8_t*>(img.base) + img.entryRva);
    return entry(static_cast<HINSTANCE>(img.base), reason, nullptr);
}

NTSTATUS UnmapPeImage(const MapResult& img) {
    if (!img.base) return kStatusSuccess;
    FreeImage(img.base, img.imageSize);
    return kStatusSuccess;
}

}

#include "PEWipe.h"
#include "Syscalls.h"
#include "../Log.h"

#include <vector>
#include <cstring>
#include <cstdint>

namespace Protection::PEWipe {

namespace {

    struct State {
        bool             wiped             = false;
        void*            base              = nullptr;
        size_t           regionSize        = 0;
        std::vector<uint8_t> backup;
        PRUNTIME_FUNCTION funcTable        = nullptr;
        DWORD            funcTableCount    = 0;
        bool             funcTableRegistered = false;
        Report           report{};
    };

    State g_state;

    bool ProtectRegion(void* addr, size_t size, ULONG newProt, PULONG oldProt) {
        if (Protection::Syscalls::IsReady()) {
            PVOID   base = addr;
            SIZE_T  sz   = size;
            ULONG   old  = 0;
            NTSTATUS st = Protection::Syscalls::NtProtectVirtualMemory(
                GetCurrentProcess(), &base, &sz, newProt, &old);
            if (Protection::Syscalls::NT_SUCCESS_(st)) {
                if (oldProt) *oldProt = old;
                return true;
            }
            Log::Write("[PEWipe] NtProtectVirtualMemory failed status=0x%08lX, "
                       "falling back to VirtualProtect", st);
        }
        DWORD old = 0;
        if (!VirtualProtect(addr, size, newProt, &old)) {
            Log::Write("[PEWipe] VirtualProtect failed err=%lu", GetLastError());
            return false;
        }
        if (oldProt) *oldProt = old;
        return true;
    }

    bool FindExceptionTable(uint8_t* base, PRUNTIME_FUNCTION& outTable,
                            DWORD& outCount) {
        auto* dos = reinterpret_cast<IMAGE_DOS_HEADER*>(base);
        if (dos->e_magic != IMAGE_DOS_SIGNATURE) return false;

        auto* nt = reinterpret_cast<IMAGE_NT_HEADERS*>(base + dos->e_lfanew);
        if (nt->Signature != IMAGE_NT_SIGNATURE) return false;

        auto& d = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXCEPTION];
        if (!d.VirtualAddress || !d.Size) return false;

        outTable = reinterpret_cast<PRUNTIME_FUNCTION>(base + d.VirtualAddress);
        outCount = d.Size / sizeof(RUNTIME_FUNCTION);
        return outCount > 0;
    }

}

bool Wipe(HMODULE mod, const Options& opts) {
    if (g_state.wiped) return true;
    if (!mod) {
        Log::Write("[PEWipe] Wipe called with null module");
        return false;
    }

    auto* base = reinterpret_cast<uint8_t*>(mod);

    auto* dos = reinterpret_cast<IMAGE_DOS_HEADER*>(base);
    if (dos->e_magic != IMAGE_DOS_SIGNATURE) {
        Log::Write("[PEWipe] not a valid DOS header at %p", base);
        return false;
    }
    auto* nt = reinterpret_cast<IMAGE_NT_HEADERS*>(base + dos->e_lfanew);
    if (nt->Signature != IMAGE_NT_SIGNATURE) {
        Log::Write("[PEWipe] not a valid NT header at %p", nt);
        return false;
    }

    const size_t regionSize = nt->OptionalHeader.SizeOfHeaders;
    if (regionSize == 0) {
        Log::Write("[PEWipe] SizeOfHeaders is zero");
        return false;
    }

    PRUNTIME_FUNCTION funcTable  = nullptr;
    DWORD             funcCount  = 0;
    bool              tableOk    = false;
    if (opts.registerFunctionTable) {
        if (FindExceptionTable(base, funcTable, funcCount)) {

            if (RtlAddFunctionTable(funcTable, funcCount,
                                    reinterpret_cast<DWORD64>(base))) {
                tableOk = true;
            } else {
                Log::Write("[PEWipe] RtlAddFunctionTable failed - proceeding "
                           "without SEH safety net");
            }
        } else {
            Log::Write("[PEWipe] no exception directory found - skipping "
                       "function table registration");
        }
    }

    std::vector<uint8_t> backup;
    backup.assign(base, base + regionSize);

    ULONG oldProt = 0;
    if (!ProtectRegion(base, regionSize, PAGE_READWRITE, &oldProt)) {
        if (tableOk) RtlDeleteFunctionTable(funcTable);
        return false;
    }

    std::memset(base, 0, regionSize);
    if (opts.preserveDosMz && regionSize >= 2) {
        base[0] = 'M';
        base[1] = 'Z';
    }

    ULONG dummy = 0;
    ProtectRegion(base, regionSize, oldProt, &dummy);

    FlushInstructionCache(GetCurrentProcess(), base, regionSize);

    g_state.wiped               = true;
    g_state.base                = base;
    g_state.regionSize          = regionSize;
    g_state.backup              = std::move(backup);
    g_state.funcTable           = funcTable;
    g_state.funcTableCount      = funcCount;
    g_state.funcTableRegistered = tableOk;

    g_state.report.ok                      = true;
    g_state.report.functionTableRegistered = tableOk;
    g_state.report.moduleBase              = base;
    g_state.report.regionSize              = regionSize;
    g_state.report.backupSize              = g_state.backup.size();

    Log::Write("[PEWipe] wiped %zu bytes at %p (funcTable=%s count=%lu preserveMZ=%d)",
               regionSize, base,
               tableOk ? "registered" : "skipped",
               funcCount, int(opts.preserveDosMz));
    return true;
}

bool Restore() {
    if (!g_state.wiped) return true;

    auto* base = reinterpret_cast<uint8_t*>(g_state.base);
    if (!base || g_state.backup.empty()) {
        Log::Write("[PEWipe] Restore called with invalid state");
        return false;
    }

    ULONG oldProt = 0;
    if (!ProtectRegion(base, g_state.regionSize, PAGE_READWRITE, &oldProt)) {
        return false;
    }

    std::memcpy(base, g_state.backup.data(), g_state.regionSize);

    ULONG dummy = 0;
    ProtectRegion(base, g_state.regionSize, oldProt, &dummy);
    FlushInstructionCache(GetCurrentProcess(), base, g_state.regionSize);

    if (g_state.funcTableRegistered && g_state.funcTable) {
        RtlDeleteFunctionTable(g_state.funcTable);
    }

    Log::Write("[PEWipe] restored %zu bytes at %p", g_state.regionSize, base);

    g_state = State{};
    return true;
}

bool          IsWiped()   { return g_state.wiped; }
const Report& GetReport() { return g_state.report; }

}

#include "SignatureBuster.h"
#include "Syscalls.h"
#include "../Log.h"

#include <intrin.h>
#include <atomic>
#include <cstring>
#include <mutex>

namespace Protection::SignatureBuster {

namespace {

    struct Region {
        uint8_t* base;
        size_t   size;
    };

    constexpr int kMaxRegions = 64;

    struct State {
        std::mutex mtx;
        Config     cfg;
        Report     report;
        uint64_t   key[2] = { 0, 0 };
        void*      junk   = nullptr;
        size_t     junkSize = 0;
        int        regionCount = 0;
        Region     regions[kMaxRegions]{};
        std::atomic<uint64_t> tickCounter{ 0 };
    };
    State g_state;

    uint64_t Splitmix64(uint64_t& s) {
        s += 0x9E3779B97F4A7C15ULL;
        uint64_t z = s;
        z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
        z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
        return z ^ (z >> 31);
    }

    void RerollKey() {

        uint64_t s = g_state.key[0] ^ __rdtsc() ^ (uint64_t)&g_state;
        g_state.key[0] = Splitmix64(s);
        g_state.key[1] = Splitmix64(s);
    }

    void ScrambleRegion(uint8_t* p, size_t bytes) {
        if (!p || !bytes) return;

        const uint8_t* k = reinterpret_cast<const uint8_t*>(g_state.key);

        size_t i = 0;
        for (; i + 16 <= bytes; i += 16) {
            for (int j = 0; j < 16; ++j) p[i + j] ^= k[j];
        }
        for (; i < bytes; ++i) p[i] ^= k[i & 15];
    }

    void* AllocJunkArena(size_t bytes) {
        if (!bytes) return nullptr;

        void* base = nullptr;

        if (Protection::Syscalls::IsReady()) {
            PVOID   b  = nullptr;
            SIZE_T  sz = bytes;
            NTSTATUS st = Protection::Syscalls::NtAllocateVirtualMemory(
                GetCurrentProcess(), &b, 0, &sz,
                MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
            if (Protection::Syscalls::NT_SUCCESS_(st)) base = b;
        }
        if (!base) {
            base = VirtualAlloc(nullptr, bytes,
                                MEM_COMMIT | MEM_RESERVE,
                                PAGE_READWRITE);
        }
        if (!base) return nullptr;

        uint8_t* p = reinterpret_cast<uint8_t*>(base);
        uint64_t s = __rdtsc();
        for (size_t i = 0; i < bytes; i += 8) {
            uint64_t v = Splitmix64(s);
            size_t n = (bytes - i) < 8 ? (bytes - i) : 8;
            std::memcpy(p + i, &v, n);
        }
        return base;
    }

    void FreeJunkArena(void* base) {
        if (!base) return;
        if (Protection::Syscalls::IsReady()) {
            PVOID  p  = base;
            SIZE_T sz = 0;
            Protection::Syscalls::NtFreeVirtualMemory(
                GetCurrentProcess(), &p, &sz, MEM_RELEASE);
            return;
        }
        VirtualFree(base, 0, MEM_RELEASE);
    }

}

bool Init(const Config& cfg) {
    std::lock_guard<std::mutex> lk(g_state.mtx);
    if (g_state.report.initialized) return true;

    g_state.cfg = cfg;

    RerollKey();

    if (cfg.junkArenaBytes) {
        g_state.junk     = AllocJunkArena(cfg.junkArenaBytes);
        g_state.junkSize = g_state.junk ? cfg.junkArenaBytes : 0;
        if (g_state.junk) {

            g_state.regions[g_state.regionCount++] = Region{
                reinterpret_cast<uint8_t*>(g_state.junk),
                g_state.junkSize
            };
        } else {
            Log::Write("[SignatureBuster] junk arena allocation failed");
        }
    }

    g_state.report.initialized     = true;
    g_state.report.junkBase        = g_state.junk;
    g_state.report.junkSize        = g_state.junkSize;
    g_state.report.registeredCount = g_state.regionCount;

    Log::Write("[SignatureBuster] Init: junk=%p size=%zu regions=%d",
               g_state.junk, g_state.junkSize, g_state.regionCount);
    return true;
}

void Shutdown() {
    std::lock_guard<std::mutex> lk(g_state.mtx);
    if (!g_state.report.initialized) return;

    FreeJunkArena(g_state.junk);
    g_state.junk = nullptr;
    g_state.junkSize = 0;
    g_state.regionCount = 0;
    g_state.report = Report{};
}

bool RegisterRegion(void* base, size_t bytes) {
    if (!base || !bytes) return false;
    std::lock_guard<std::mutex> lk(g_state.mtx);
    if (g_state.regionCount >= kMaxRegions) return false;

    auto* p1 = reinterpret_cast<uint8_t*>(base);
    for (int i = 0; i < g_state.regionCount; ++i) {
        auto* p2 = g_state.regions[i].base;
        size_t s2 = g_state.regions[i].size;
        if (p1 < p2 + s2 && p2 < p1 + bytes) return false;
    }

    g_state.regions[g_state.regionCount++] = Region{ p1, bytes };
    g_state.report.registeredCount = g_state.regionCount;
    return true;
}

void Tick() {
    if (!g_state.report.initialized) return;

    uint64_t t = g_state.tickCounter.fetch_add(1, std::memory_order_relaxed);
    unsigned every = g_state.cfg.scrambleEveryTicks
                   ? g_state.cfg.scrambleEveryTicks : 1;
    if ((t % every) != 0) return;

    std::lock_guard<std::mutex> lk(g_state.mtx);

    RerollKey();

    size_t total = 0;
    for (int i = 0; i < g_state.regionCount; ++i) {
        ScrambleRegion(g_state.regions[i].base, g_state.regions[i].size);
        total += g_state.regions[i].size;
    }
    g_state.report.totalScrambled  = total;
    g_state.report.scrambleCycles  += 1;
}

const Report& GetReport()  { return g_state.report; }
void*         JunkArenaBase() { return g_state.junk; }

}

    bool WStringEqualsAsciiCI(const wchar_t* s, size_t len, const char* ascii) {
        size_t alen = 0;
        while (ascii[alen]) ++alen;
        if (len != alen) return false;
        for (size_t i = 0; i < len; ++i) {
            wchar_t a = s[i];
            char    b = ascii[i];
            if (a >= L'A' && a <= L'Z') a = wchar_t(a + 32);
            if (b >= 'A' && b <= 'Z')   b = char(b + 32);
            if (a != wchar_t(b)) return false;
        }
        return true;
    }

    void* FindNtdllBase() {
        auto* peb = reinterpret_cast<PEB_MIN*>(__readgsqword(0x60));
        if (!peb || !peb->Ldr) return nullptr;

        LIST_ENTRY* head = &peb->Ldr->InLoadOrderModuleList;
        for (LIST_ENTRY* it = head->Flink; it && it != head; it = it->Flink) {
            auto* mod = CONTAINING_RECORD(it, LDR_MODULE, InLoadOrderModuleList);
            if (!mod->BaseAddress || !mod->BaseDllName.Buffer) continue;

            size_t chars = mod->BaseDllName.Length / sizeof(wchar_t);
            if (WStringEqualsAsciiCI(mod->BaseDllName.Buffer, chars, "ntdll.dll"))
                return mod->BaseAddress;
        }
        return nullptr;
    }

    struct Export {
        const char* name;
        void*       address;
        uint32_t    rva;
        uint32_t    ssn;
        bool        hooked;
    };

    bool CollectNtExports(void* base, std::vector<Export>& out) {
        auto* dos = reinterpret_cast<IMAGE_DOS_HEADER*>(base);
        if (dos->e_magic != IMAGE_DOS_SIGNATURE) return false;

        auto* nt = reinterpret_cast<IMAGE_NT_HEADERS*>(
            reinterpret_cast<uint8_t*>(base) + dos->e_lfanew);
        if (nt->Signature != IMAGE_NT_SIGNATURE) return false;

        auto& expDir = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXPORT];
        if (!expDir.VirtualAddress || !expDir.Size) return false;

        auto* exp = reinterpret_cast<IMAGE_EXPORT_DIRECTORY*>(
            reinterpret_cast<uint8_t*>(base) + expDir.VirtualAddress);

        auto* names = reinterpret_cast<uint32_t*>(
            reinterpret_cast<uint8_t*>(base) + exp->AddressOfNames);
        auto* funcs = reinterpret_cast<uint32_t*>(
            reinterpret_cast<uint8_t*>(base) + exp->AddressOfFunctions);
        auto* ords = reinterpret_cast<uint16_t*>(
            reinterpret_cast<uint8_t*>(base) + exp->AddressOfNameOrdinals);

        out.reserve(exp->NumberOfNames);
        for (DWORD i = 0; i < exp->NumberOfNames; ++i) {
            const char* name = reinterpret_cast<const char*>(
                reinterpret_cast<uint8_t*>(base) + names[i]);

            if (name[0] != 'N' || name[1] != 't') continue;
            if (name[2] == 'd')                    continue;

            uint32_t rva  = funcs[ords[i]];
            void*    addr = reinterpret_cast<uint8_t*>(base) + rva;
            out.push_back(Export{ name, addr, rva, 0u, false });
        }
        return !out.empty();
    }

    bool ExtractSSNDirect(void* stub, uint32_t& ssn) {
        auto* p = reinterpret_cast<uint8_t*>(stub);
        if (p[0] == 0x4C && p[1] == 0x8B && p[2] == 0xD1 && p[3] == 0xB8) {
            std::memcpy(&ssn, p + 4, sizeof(uint32_t));
            return true;
        }
        return false;
    }

    bool ResolveViaNeighbours(std::vector<Export>& exports, size_t idx) {

        for (size_t j = idx + 1; j < exports.size(); ++j) {
            if (!exports[j].hooked) {
                exports[idx].ssn = exports[j].ssn - static_cast<uint32_t>(j - idx);
                return true;
            }
        }

        for (size_t j = idx; j-- > 0;) {
            if (!exports[j].hooked) {
                exports[idx].ssn = exports[j].ssn + static_cast<uint32_t>(idx - j);
                return true;
            }
        }
        return false;
    }

    constexpr size_t kStubSize = 16;

    void EmitStub(uint8_t* dst, uint32_t ssn) {

        dst[0] = 0x4C; dst[1] = 0x8B; dst[2] = 0xD1;
        dst[3] = 0xB8;
        std::memcpy(dst + 4, &ssn, sizeof(uint32_t));
        dst[8] = 0x0F; dst[9] = 0x05;
        dst[10] = 0xC3;

        for (size_t i = 11; i < kStubSize; ++i) dst[i] = 0xCC;
    }

    Table       g_table{};
    InitReport  g_report{};
    void*       g_stubBuffer     = nullptr;
    size_t      g_stubBufferSize = 0;
    bool        g_ready          = false;

}

        const char* name;

    };

        { "NtProtectVirtualMemory",     reinterpret_cast<void**>(&g_table.NtProtectVirtualMemory)    },
        { "NtAllocateVirtualMemory",    reinterpret_cast<void**>(&g_table.NtAllocateVirtualMemory)   },

        { "NtDelayExecution",           reinterpret_cast<void**>(&g_table.NtDelayExecution)          },
        { "NtClose",                    reinterpret_cast<void**>(&g_table.NtClose)                   },
    };
    constexpr size_t kWantedCount = sizeof(wanted) / sizeof(wanted[0]);
    g_report.totalRequested = static_cast<int>(kWantedCount);

    g_stubBufferSize = kWantedCount * kStubSize;

    SYSTEM_INFO si{};
    GetSystemInfo(&si);
    size_t alloc = ((g_stubBufferSize + si.dwPageSize - 1) / si.dwPageSize) * si.dwPageSize;
    g_stubBuffer = VirtualAlloc(nullptr, alloc,
                                MEM_COMMIT | MEM_RESERVE,
                                PAGE_READWRITE);
    if (!g_stubBuffer) {
        Log::Write("[Syscalls] VirtualAlloc for stub buffer failed (err=%lu)",
                   GetLastError());
        return false;
    }
    g_report.stubBuffer     = g_stubBuffer;
    g_report.stubBufferSize = alloc;

    auto* buf = reinterpret_cast<uint8_t*>(g_stubBuffer);
    for (size_t i = 0; i < kWantedCount; ++i) {

        const Export* found = nullptr;
        for (const auto& e : exports) {
            if (std::strcmp(e.name, wanted[i].name) == 0) { found = &e; break; }
        }

        if (!found || found->ssn == 0) {
            Log::Write("[Syscalls] MISSING: %s (found=%d ssn=%u hooked=%d)",
                       wanted[i].name,
                       found ? 1 : 0,
                       found ? found->ssn : 0u,
                       found ? int(found->hooked) : -1);
            *wanted[i].slot = nullptr;
            g_report.failed++;
            continue;
        }

        uint8_t* stubAddr = buf + (i * kStubSize);
        EmitStub(stubAddr, found->ssn);
        *wanted[i].slot = stubAddr;

        if (found->hooked) g_report.resolvedHalo++;
        else               g_report.resolvedDirect++;
    }

    DWORD oldProt = 0;
    if (!VirtualProtect(g_stubBuffer, alloc, PAGE_EXECUTE_READ, &oldProt)) {
        Log::Write("[Syscalls] VirtualProtect(RX) failed (err=%lu)", GetLastError());
        VirtualFree(g_stubBuffer, 0, MEM_RELEASE);
        g_stubBuffer = nullptr;
        return false;
    }
    FlushInstructionCache(GetCurrentProcess(), g_stubBuffer, alloc);

    g_ready      = (g_report.failed == 0);
    g_report.ok  = g_ready;

    Log::Write("[Syscalls] Init: ntdll=%p buf=%p size=%zu direct=%d halo=%d failed=%d",
               ntdll, g_stubBuffer, alloc,
               g_report.resolvedDirect,
               g_report.resolvedHalo,
               g_report.failed);

    return g_ready;
}

void Shutdown() {
    if (g_stubBuffer) {
        VirtualFree(g_stubBuffer, 0, MEM_RELEASE);
        g_stubBuffer = nullptr;
    }
    g_table  = Table{};
    g_report = InitReport{};
    g_ready  = false;
}

bool              IsReady()   { return g_ready; }
const Table&      Get()       { return g_table; }
const InitReport& GetReport() { return g_report; }

}

    bool TryDriverHide(uint64_t tid) {
        if (!Protection::Driver::IsDriverPresent()) return false;
        if (Protection::Driver::HideThreadStrong(tid)) return true;

        return Protection::Driver::HideThread(tid);
    }

    bool UserModeHide(HANDLE thread) {
        if (!Protection::Syscalls::IsReady()) return false;
        NTSTATUS st = Protection::Syscalls::NtSetInformationThread(
            thread,
            static_cast<THREADINFOCLASS>(
                Protection::Syscalls::Const::ThreadHideFromDebugger_),
            nullptr, 0);
        return Protection::Syscalls::NT_SUCCESS_(st);
    }

    uint64_t TidFromHandle(HANDLE thread) {

        return static_cast<uint64_t>(GetThreadId(thread));
    }

}

void RegisterOwnThread(DWORD tid) {
    if (!tid) return;
    std::lock_guard<std::mutex> lk(g_regMutex);

    if (g_ownThreads.size() >= 64) return;
    for (DWORD existing : g_ownThreads) {
        if (existing == tid) return;
    }
    g_ownThreads.push_back(tid);
}

bool Hide(HANDLE thread) {
    if (!thread || thread == INVALID_HANDLE_VALUE) return false;

    uint64_t tid = TidFromHandle(thread);
    if (tid && TryDriverHide(tid)) return true;

    return UserModeHide(thread);
}

bool HideCurrent() {

    RegisterOwnThread(GetCurrentThreadId());
    return Hide(GetCurrentThread());
}

int HideOwnRegistered() {
    std::vector<DWORD> copy;
    {
        std::lock_guard<std::mutex> lk(g_regMutex);
        copy = g_ownThreads;
    }

    int hidden = 0;
    for (DWORD tid : copy) {

        if (TryDriverHide(tid)) { ++hidden; continue; }

        HANDLE h = OpenThread(THREAD_SET_INFORMATION | THREAD_QUERY_LIMITED_INFORMATION,
                              FALSE, tid);
        if (!h) continue;
        if (UserModeHide(h)) ++hidden;
        CloseHandle(h);
    }

    Log::Write("[Threads] HideOwnRegistered: %d/%zu threads hidden (%s)",
               hidden, copy.size(),
               Protection::Driver::IsDriverPresent() ? "driver" : "user-mode");
    return hidden;
}

int HideAllInProcess() {
    HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPTHREAD, 0);
    if (snap == INVALID_HANDLE_VALUE) return 0;

    DWORD ownPid = GetCurrentProcessId();
    THREADENTRY32 te{ sizeof(te) };
    int hidden = 0;

    if (Thread32First(snap, &te)) {
        do {
            if (te.th32OwnerProcessID != ownPid) continue;
            HANDLE ht = OpenThread(THREAD_SET_INFORMATION | THREAD_QUERY_LIMITED_INFORMATION,
                                   FALSE, te.th32ThreadID);
            if (!ht) continue;
            if (Hide(ht)) ++hidden;
            CloseHandle(ht);
        } while (Thread32Next(snap, &te));
    }
    CloseHandle(snap);
    Log::Write("[Threads] HideAllInProcess: %d threads hidden (DANGEROUS)", hidden);
    return hidden;
}

struct Vec3 {
    float x{ 0 }, y{ 0 }, z{ 0 };

    Vec3() = default;
    Vec3(float x_, float y_, float z_) : x(x_), y(y_), z(z_) {}

    Vec3  operator+(const Vec3& o) const { return { x + o.x, y + o.y, z + o.z }; }
    Vec3  operator-(const Vec3& o) const { return { x - o.x, y - o.y, z - o.z }; }
    Vec3  operator*(float s)       const { return { x * s,  y * s,  z * s }; }
    Vec3& operator+=(const Vec3& o) { x += o.x; y += o.y; z += o.z; return *this; }
    Vec3& operator-=(const Vec3& o) { x -= o.x; y -= o.y; z -= o.z; return *this; }

    float Length()   const { return std::sqrt(x * x + y * y + z * z); }
    float Length2D() const { return std::sqrt(x * x + y * y); }
    float Dot(const Vec3& o) const { return x * o.x + y * o.y + z * o.z; }

    bool IsZero() const {
        return x == 0.f && y == 0.f && z == 0.f;
    }
};

struct Vec2 {
    float x{ 0 }, y{ 0 };
    Vec2() = default;
    Vec2(float x_, float y_) : x(x_), y(y_) {}
};

struct ViewMatrix {
    float m[4][4];
};

namespace Math {
    constexpr float PI = 3.14159265358979323846f;

    inline float RadToDeg(float r) { return r * (180.f / PI); }
    inline float DegToRad(float d) { return d * (PI / 180.f); }

    inline void NormalizeAngles(Vec3& a) {
        while (a.x >  89.0f)  a.x -= 180.f;
        while (a.x < -89.0f)  a.x += 180.f;
        while (a.y >  180.f)  a.y -= 360.f;
        while (a.y < -180.f)  a.y += 360.f;
        a.z = 0.f;
    }

    inline Vec3 CalcAngle(const Vec3& src, const Vec3& dst) {
        Vec3 delta = dst - src;
        float hyp  = delta.Length2D();
        Vec3 angle{
            -RadToDeg(std::atan2(delta.z, hyp)),
             RadToDeg(std::atan2(delta.y, delta.x)),
             0.f
        };
        NormalizeAngles(angle);
        return angle;
    }

    inline bool WorldToScreen(const Vec3& world, const ViewMatrix& vm,
                              int screenW, int screenH, Vec2& out) {
        float w = vm.m[3][0] * world.x + vm.m[3][1] * world.y + vm.m[3][2] * world.z + vm.m[3][3];
        if (w < 0.01f) return false;

        float x = vm.m[0][0] * world.x + vm.m[0][1] * world.y + vm.m[0][2] * world.z + vm.m[0][3];
        float y = vm.m[1][0] * world.x + vm.m[1][1] * world.y + vm.m[1][2] * world.z + vm.m[1][3];

        float invW = 1.f / w;
        float nx = x * invW;
        float ny = y * invW;

        out.x = (screenW * 0.5f) + (nx * screenW * 0.5f);
        out.y = (screenH * 0.5f) - (ny * screenH * 0.5f);
        return true;
    }
}

    }

            Globals::shutdown = true;

    bool EqualsIgnoreCase(const wchar_t* a, const wchar_t* b) {
        while (*a && *b) {
            wchar_t ca = *a, cb = *b;
            if (ca >= L'A' && ca <= L'Z') ca = wchar_t(ca + 32);
            if (cb >= L'A' && cb <= L'Z') cb = wchar_t(cb + 32);
            if (ca != cb) return false;
            ++a; ++b;
        }
        return *a == 0 && *b == 0;
    }

    DWORD FindProcess(const wchar_t* name) {

        if (snap == INVALID_HANDLE_VALUE) return 0;

        PROCESSENTRY32W pe{ sizeof(pe) };

            do {
                if (EqualsIgnoreCase(pe.szExeFile, name)) {
                    pid = pe.th32ProcessID;
                    break;
                }
            } while (Process32NextW(snap, &pe));
        }
        CloseHandle(snap);

    bool ApplyRelocations(uint8_t* image, IMAGE_NT_HEADERS* nt, uint64_t delta) {
        if (delta == 0) return true;
        auto& d = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_BASERELOC];
        if (!d.VirtualAddress || !d.Size) return true;

        auto* block = reinterpret_cast<IMAGE_BASE_RELOCATION*>(image + d.VirtualAddress);
        auto* end   = reinterpret_cast<IMAGE_BASE_RELOCATION*>(
            reinterpret_cast<uint8_t*>(block) + d.Size);

        while (block < end && block->SizeOfBlock) {
            DWORD count = (block->SizeOfBlock - sizeof(*block)) / sizeof(WORD);
            auto* entries = reinterpret_cast<WORD*>(block + 1);
            uint8_t* pageBase = image + block->VirtualAddress;

            for (DWORD i = 0; i < count; ++i) {
                WORD e = entries[i];
                WORD type   = WORD(e >> 12);
                WORD offset = WORD(e & 0x0FFF);
                switch (type) {
                    case IMAGE_REL_BASED_ABSOLUTE:
                        break;
                    case IMAGE_REL_BASED_HIGHLOW:
                        *reinterpret_cast<uint32_t*>(pageBase + offset) +=
                            static_cast<uint32_t>(delta);
                        break;
                    case IMAGE_REL_BASED_DIR64:
                        *reinterpret_cast<uint64_t*>(pageBase + offset) += delta;
                        break;
                    default:
                        return false;
                }
            }
            block = reinterpret_cast<IMAGE_BASE_RELOCATION*>(
                reinterpret_cast<uint8_t*>(block) + block->SizeOfBlock);
        }
        return true;
    }

    HMODULE RemoteLoadLibraryA(HANDLE proc, const char* name) {
        auto pLL = reinterpret_cast<LPTHREAD_START_ROUTINE>(

        if (!pLL) return nullptr;

        DWORD code = 0;
        GetExitCodeThread(th, &code);
        CloseHandle(th);
        VirtualFreeEx(proc, remoteStr, 0, MEM_RELEASE);

    void* ResolveRemoteExport(HANDLE proc, void* modBase, const char* funcName) {

        IMAGE_DOS_HEADER dos{};
        if (!ReadRemote(proc, modBase, &dos, sizeof(dos))) return nullptr;
        if (dos.e_magic != IMAGE_DOS_SIGNATURE) return nullptr;

        IMAGE_NT_HEADERS nt{};
        if (!ReadRemote(proc,
                        static_cast<uint8_t*>(modBase) + dos.e_lfanew,
                        &nt, sizeof(nt))) return nullptr;
        if (nt.Signature != IMAGE_NT_SIGNATURE) return nullptr;

        auto d = nt.OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXPORT];
        if (!d.VirtualAddress || !d.Size) return nullptr;

        std::vector<uint8_t> expBuf(d.Size);
        if (!ReadRemote(proc,
                        static_cast<uint8_t*>(modBase) + d.VirtualAddress,
                        expBuf.data(), d.Size)) return nullptr;

        auto* exp = reinterpret_cast<IMAGE_EXPORT_DIRECTORY*>(expBuf.data());
        auto rvaInExpBuf = [&](DWORD rva) -> void* {
            if (rva < d.VirtualAddress) return nullptr;
            DWORD off = rva - d.VirtualAddress;
            if (off >= d.Size) return nullptr;
            return expBuf.data() + off;
        };

        auto* names = static_cast<uint32_t*>(rvaInExpBuf(exp->AddressOfNames));
        auto* funcs = static_cast<uint32_t*>(rvaInExpBuf(exp->AddressOfFunctions));
        auto* ords  = static_cast<uint16_t*>(rvaInExpBuf(exp->AddressOfNameOrdinals));
        if (!names || !funcs || !ords) return nullptr;

        for (DWORD i = 0; i < exp->NumberOfNames; ++i) {
            const char* n = static_cast<const char*>(rvaInExpBuf(names[i]));
            if (!n) continue;
            if (std::strcmp(n, funcName) != 0) continue;

            DWORD fnRva = funcs[ords[i]];
            void* addr  = static_cast<uint8_t*>(modBase) + fnRva;

            if (fnRva >= d.VirtualAddress && fnRva < d.VirtualAddress + d.Size) {
                const char* fwd = static_cast<const char*>(rvaInExpBuf(fnRva));
                if (!fwd) return nullptr;
                const char* dot = std::strchr(fwd, '.');
                if (!dot) return nullptr;

                char modName[80]{};
                size_t mn = static_cast<size_t>(dot - fwd);
                if (mn + 5 >= sizeof(modName)) return nullptr;
                std::memcpy(modName, fwd, mn);
                std::memcpy(modName + mn, ".dll", 5);

                void* fwdBase = FindRemoteModule(GetProcessId(proc), modName);
                if (!fwdBase) {
                    RemoteLoadLibraryA(proc, modName);
                    fwdBase = FindRemoteModule(GetProcessId(proc), modName);
                }
                if (!fwdBase) return nullptr;
                return ResolveRemoteExport(proc, fwdBase, dot + 1);
            }
            return addr;
        }
        return nullptr;
    }

    bool ResolveImports(HANDLE proc, DWORD pid,
                        uint8_t* image, IMAGE_NT_HEADERS* nt) {
        auto& d = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_IMPORT];
        if (!d.VirtualAddress || !d.Size) return true;

        auto* desc = reinterpret_cast<IMAGE_IMPORT_DESCRIPTOR*>(image + d.VirtualAddress);
        for (; desc->Name; ++desc) {
            const char* modName = reinterpret_cast<const char*>(image + desc->Name);

            HMODULE localMod = GetModuleHandleA(modName);
            if (!localMod) localMod = LoadLibraryA(modName);

            void* remoteBase = FindRemoteModule(pid, modName);
            if (!remoteBase) {
                RemoteLoadLibraryA(proc, modName);
                remoteBase = FindRemoteModule(pid, modName);
            }

            if (!localMod && !remoteBase) {
                std::fwprintf(stderr,
                    L"[Loader] import module not resolvable: %hs\n", modName);
                return false;
            }

            bool sameBase = localMod && remoteBase &&
                            (void*)localMod == remoteBase;
            if (g_verbose) {
                std::wprintf(L"[Loader] import module: %hs local=%p remote=%p %s\n",
                             modName,
                             (void*)localMod, remoteBase,
                             sameBase ? L"(same base, local resolve)"
                                      : (remoteBase ? L"(diff base, remote resolve)"
                                                    : L"(local-only fallback)"));
            }

            auto* origThunk = reinterpret_cast<IMAGE_THUNK_DATA*>(
                image + (desc->OriginalFirstThunk ? desc->OriginalFirstThunk
                                                  : desc->FirstThunk));
            auto* iat = reinterpret_cast<IMAGE_THUNK_DATA*>(image + desc->FirstThunk);

            for (; origThunk->u1.AddressOfData; ++origThunk, ++iat) {
                void* fn = nullptr;
                char  symBuf[128]{};

                if (IMAGE_SNAP_BY_ORDINAL(origThunk->u1.Ordinal)) {
                    WORD ord = IMAGE_ORDINAL(origThunk->u1.Ordinal);
                    std::snprintf(symBuf, sizeof(symBuf), "#%u", ord);
                    if (sameBase && localMod) {
                        fn = reinterpret_cast<void*>(
                            GetProcAddress(localMod, MAKEINTRESOURCEA(ord)));
                    }

                } else {
                    auto* import = reinterpret_cast<IMAGE_IMPORT_BY_NAME*>(
                        image + origThunk->u1.AddressOfData);
                    std::snprintf(symBuf, sizeof(symBuf), "%s", import->Name);

                    if (sameBase && localMod) {
                        fn = reinterpret_cast<void*>(
                            GetProcAddress(localMod, import->Name));
                    } else if (remoteBase) {
                        fn = ResolveRemoteExport(proc, remoteBase, import->Name);
                    } else if (localMod) {

                        fn = reinterpret_cast<void*>(
                            GetProcAddress(localMod, import->Name));
                    }
                }

                if (!fn) {
                    std::fwprintf(stderr,
                        L"[Loader] unresolved import: %hs!%hs\n",
                        modName, symBuf);
                    return false;
                }
                iat->u1.Function = reinterpret_cast<ULONGLONG>(fn);
            }
        }
        return true;
    }

    DWORD SectionProt(DWORD c) {
        bool r = (c & IMAGE_SCN_MEM_READ)    != 0;

        if (g_verbose)
            std::wprintf(L"[Loader] RtlAddFunctionTable count=%u -> %s\n",
                         count, exitCode ? L"registered" : L"failed");
        return exitCode != 0;
    }

    bool RunRemoteTlsCallbacks(HANDLE proc, void* remoteBase,
                               IMAGE_NT_HEADERS* nt) {
        auto& d = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_TLS];
        if (!d.VirtualAddress || !d.Size) return true;

        IMAGE_TLS_DIRECTORY tls{};
        SIZE_T got = 0;
        if (!ReadProcessMemory(proc,
                static_cast<uint8_t*>(remoteBase) + d.VirtualAddress,
                &tls, sizeof(tls), &got) || got != sizeof(tls))
            return false;

        if (!tls.AddressOfCallBacks) return true;

        for (uint64_t addr = tls.AddressOfCallBacks; ; addr += sizeof(PVOID)) {
            PVOID cb = nullptr;
            if (!ReadProcessMemory(proc,
                    reinterpret_cast<void*>(addr), &cb, sizeof(cb), &got) ||
                got != sizeof(cb))
                return false;
            if (!cb) break;

            uint64_t base = reinterpret_cast<uint64_t>(remoteBase);
            uint64_t api  = reinterpret_cast<uint64_t>(cb);
            uint8_t code[40] = {
                0x48, 0xB9, 0,0,0,0,0,0,0,0,
                0xBA, 0x01, 0,0,0,
                0x4D, 0x31, 0xC0,
                0x48, 0xB8, 0,0,0,0,0,0,0,0,
                0x48, 0x83, 0xEC, 0x28,
                0xFF, 0xD0,
                0x48, 0x83, 0xC4, 0x28,
                0xC3
            };
            std::memcpy(code +  2, &base, 8);
            std::memcpy(code + 20, &api,  8);

            void* stub = VirtualAllocEx(proc, nullptr, sizeof(code),
                                        MEM_COMMIT | MEM_RESERVE,
                                        PAGE_EXECUTE_READWRITE);
            if (!stub) return false;
            WriteProcessMemory(proc, stub, code, sizeof(code), nullptr);
            HANDLE th = CreateRemoteThread(proc, nullptr, 0,
                reinterpret_cast<LPTHREAD_START_ROUTINE>(stub),
                nullptr, 0, nullptr);
            if (th) { WaitForSingleObject(th, 10000); CloseHandle(th); }
            VirtualFreeEx(proc, stub, 0, MEM_RELEASE);
            if (g_verbose)
                std::wprintf(L"[Loader] TLS callback %p invoked\n", cb);
        }
        return true;
    }

    bool ApplyRemoteProtections(HANDLE proc, void* remoteBase,
                                IMAGE_NT_HEADERS* nt) {
        auto* sec = IMAGE_FIRST_SECTION(nt);
        for (WORD i = 0; i < nt->FileHeader.NumberOfSections; ++i, ++sec) {
            if (!sec->Misc.VirtualSize) continue;
            void* addr = static_cast<uint8_t*>(remoteBase) + sec->VirtualAddress;
            DWORD prot = SectionProt(sec->Characteristics);
            DWORD old  = 0;
            if (!VirtualProtectEx(proc, addr, sec->Misc.VirtualSize,
                                  prot, &old)) return false;
        }
        return true;
    }

    static constexpr size_t kTrampolineSize = 34;

    void BuildTrampoline(uint8_t* buf, uint64_t base, uint64_t entry) {
        size_t i = 0;
        buf[i++] = 0x48; buf[i++] = 0xB9;
        std::memcpy(buf + i, &base, 8); i += 8;
        buf[i++] = 0xBA; buf[i++] = 0x01; buf[i++] = 0x00;
        buf[i++] = 0x00; buf[i++] = 0x00;
        buf[i++] = 0x4D; buf[i++] = 0x31; buf[i++] = 0xC0;
        buf[i++] = 0x48; buf[i++] = 0xB8;
        std::memcpy(buf + i, &entry, 8); i += 8;
        buf[i++] = 0x48; buf[i++] = 0x83; buf[i++] = 0xEC; buf[i++] = 0x28;
        buf[i++] = 0xFF; buf[i++] = 0xD0;
        buf[i++] = 0x48; buf[i++] = 0x83; buf[i++] = 0xC4; buf[i++] = 0x28;
        buf[i++] = 0xC3;

        while (i < kTrampolineSize) buf[i++] = 0xCC;
    }

}

InjectResult Inject(const InjectOptions& opts) {
    InjectResult r{};
    auto fail = [&](const wchar_t* msg) {
        r.ok = false;
        r.errorText = msg;
        return r;
    };

    g_verbose = opts.verbose;

    std::vector<uint8_t> file;
    std::wstring resolvedPath = opts.dllPath;

    auto tryRead = [&](const std::wstring& p) {
        if (ReadWholeFile(p, file)) { resolvedPath = p; return true; }
        return false;
    };

    if (!tryRead(opts.dllPath)) {
        wchar_t exeDir[MAX_PATH]{};
        DWORD n = GetModuleFileNameW(nullptr, exeDir, MAX_PATH);
        if (n && n < MAX_PATH) {
            for (int i = int(n) - 1; i >= 0; --i) {
                if (exeDir[i] == L'\\' || exeDir[i] == L'/') {
                    exeDir[i + 1] = 0;
                    break;
                }
            }
            std::wstring dir = exeDir;

            if (!tryRead(dir + opts.dllPath)) {

                const wchar_t* bn = opts.dllPath.c_str();
                for (const wchar_t* p = bn; *p; ++p) {
                    if (*p == L'\\' || *p == L'/') bn = p + 1;
                }
                if (bn != opts.dllPath.c_str())
                    tryRead(dir + bn);
            }
        }
    }

    if (file.empty()) {
        std::wstring msg = L"failed to read DLL file: ";
        msg += opts.dllPath;
        return fail(msg.c_str());
    }
    if (opts.verbose) {
        std::wprintf(L"[Loader] resolved DLL: %s (%zu bytes)\n",
                     resolvedPath.c_str(), file.size());
    }

    if (file.size() < sizeof(IMAGE_DOS_HEADER))
        return fail(L"file too small to be a PE");
    auto* dos = reinterpret_cast<IMAGE_DOS_HEADER*>(file.data());
    if (dos->e_magic != IMAGE_DOS_SIGNATURE)
        return fail(L"not a PE (missing MZ)");
    auto* nt = reinterpret_cast<IMAGE_NT_HEADERS*>(file.data() + dos->e_lfanew);
    if (nt->Signature != IMAGE_NT_SIGNATURE)
        return fail(L"not a PE (missing PE\\0\\0)");
    if (nt->OptionalHeader.Magic != IMAGE_NT_OPTIONAL_HDR_MAGIC)
        return fail(L"not a 64-bit PE");

    const size_t imageSize = nt->OptionalHeader.SizeOfImage;
    r.imageSize = imageSize;

    DWORD pid = opts.waitTimeoutMs
        ? WaitForProcess(opts.targetProcess.c_str(), opts.waitTimeoutMs)
        : FindProcess(opts.targetProcess.c_str());
    if (!pid) return fail(L"target process not found");
    r.targetPid = pid;

    HANDLE proc = OpenProcess(PROCESS_ALL_ACCESS, FALSE, pid);
    if (!proc) return fail(L"OpenProcess PROCESS_ALL_ACCESS failed");

    void* remoteBase = VirtualAllocEx(proc, nullptr, imageSize,
                                      MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
    if (!remoteBase) { CloseHandle(proc); return fail(L"VirtualAllocEx failed"); }
    r.remoteBase = remoteBase;

    std::vector<uint8_t> image(imageSize, 0);
    std::memcpy(image.data(), file.data(), nt->OptionalHeader.SizeOfHeaders);

    auto* localNt = reinterpret_cast<IMAGE_NT_HEADERS*>(image.data() + dos->e_lfanew);
    auto* sec = IMAGE_FIRST_SECTION(localNt);
    for (WORD i = 0; i < localNt->FileHeader.NumberOfSections; ++i, ++sec) {
        if (!sec->SizeOfRawData) continue;
        if (sec->PointerToRawData + sec->SizeOfRawData > file.size()) {
            VirtualFreeEx(proc, remoteBase, 0, MEM_RELEASE);
            CloseHandle(proc);
            return fail(L"section extends past file size");
        }
        std::memcpy(image.data() + sec->VirtualAddress,
                    file.data() + sec->PointerToRawData,
                    sec->SizeOfRawData);
    }

    uint64_t delta = reinterpret_cast<uint64_t>(remoteBase)
                   - localNt->OptionalHeader.ImageBase;
    if (!ApplyRelocations(image.data(), localNt, delta)) {
        VirtualFreeEx(proc, remoteBase, 0, MEM_RELEASE);
        CloseHandle(proc);
        return fail(L"base relocations failed");
    }
    localNt->OptionalHeader.ImageBase = reinterpret_cast<ULONGLONG>(remoteBase);

    if (!ResolveImports(proc, pid, image.data(), localNt)) {
        VirtualFreeEx(proc, remoteBase, 0, MEM_RELEASE);
        CloseHandle(proc);
        return fail(L"import resolution failed");
    }

    if (opts.wipeHeaders) {
        std::memset(image.data(), 0, localNt->OptionalHeader.SizeOfHeaders);
    }

    SIZE_T written = 0;
    if (!WriteProcessMemory(proc, remoteBase, image.data(),
                            imageSize, &written) || written != imageSize) {
        VirtualFreeEx(proc, remoteBase, 0, MEM_RELEASE);
        CloseHandle(proc);
        return fail(L"WriteProcessMemory failed");
    }

    if (!ApplyRemoteProtections(proc, remoteBase, localNt)) {
        VirtualFreeEx(proc, remoteBase, 0, MEM_RELEASE);
        CloseHandle(proc);
        return fail(L"VirtualProtectEx per section failed");
    }

    RegisterRemoteFunctionTable(proc, remoteBase, localNt);

    RunRemoteTlsCallbacks(proc, remoteBase, localNt);

    if (opts.wipeHeaders) {

        DWORD old = 0;
        VirtualProtectEx(proc, remoteBase,
                         localNt->OptionalHeader.SizeOfHeaders,
                         PAGE_READONLY, &old);
    }

    uint64_t base  = reinterpret_cast<uint64_t>(remoteBase);
    uint64_t entry = base + localNt->OptionalHeader.AddressOfEntryPoint;

    void* trampoline = VirtualAllocEx(proc, nullptr, 0x1000,
                                      MEM_COMMIT | MEM_RESERVE,
                                      PAGE_EXECUTE_READWRITE);
    if (!trampoline) {
        VirtualFreeEx(proc, remoteBase, 0, MEM_RELEASE);
        CloseHandle(proc);
        return fail(L"VirtualAllocEx trampoline failed");
    }

    uint8_t code[64]{};
    BuildTrampoline(code, base, entry);
    if (!WriteProcessMemory(proc, trampoline, code, sizeof(code), nullptr)) {
        VirtualFreeEx(proc, trampoline, 0, MEM_RELEASE);
        VirtualFreeEx(proc, remoteBase, 0, MEM_RELEASE);
        CloseHandle(proc);
        return fail(L"WriteProcessMemory trampoline failed");
    }

    HANDLE th = CreateRemoteThread(proc, nullptr, 0,
                                   reinterpret_cast<LPTHREAD_START_ROUTINE>(trampoline),
                                   nullptr, 0, nullptr);
    if (!th) {
        VirtualFreeEx(proc, trampoline, 0, MEM_RELEASE);
        VirtualFreeEx(proc, remoteBase, 0, MEM_RELEASE);
        CloseHandle(proc);
        return fail(L"CreateRemoteThread failed");
    }
    WaitForSingleObject(th, INFINITE);
    GetExitCodeThread(th, &r.entryExitCode);
    CloseHandle(th);

    VirtualFreeEx(proc, trampoline, 0, MEM_RELEASE);
    CloseHandle(proc);

    r.ok = true;
    return r;
}

}

```cpp


#include "Injector.h"

#include <Windows.h>
#include <cstdio>
#include <cwchar>
#include <string>

static void PrintUsage() {
    std::wprintf(
        L"Usage: Loader.exe --target <exe> --dll <path>\n"
        L"                  [--wait <seconds>] [--no-wipe] [--verbose]\n"
        L"\n"
        L"Example:\n"
        L"    Loader.exe --target cs2.exe --dll RabsztynCC.dll --wait 120\n");
}

int wmain(int argc, wchar_t** argv) {
    Loader::InjectOptions opts;
    opts.targetProcess = L"cs2.exe";

    for (int i = 1; i < argc; ++i) {
        std::wstring a = argv[i];
        auto need = [&](const wchar_t* what) -> const wchar_t* {
            if (i + 1 >= argc) {
                std::fwprintf(stderr, L"missing value for %s\n", what);
                std::exit(1);
            }
            return argv[++i];
        };
        if (a == L"--target")      opts.targetProcess = need(L"--target");
        else if (a == L"--dll")    opts.dllPath       = need(L"--dll");
        else if (a == L"--wait")   opts.waitTimeoutMs = static_cast<unsigned>(
                                       std::wcstoul(need(L"--wait"), nullptr, 10)) * 1000;
        else if (a == L"--no-wipe") opts.wipeHeaders  = false;
        else if (a == L"--wipe")    opts.wipeHeaders  = true;
        else if (a == L"--verbose") opts.verbose      = true;
        else if (a == L"--help" || a == L"-h" || a == L"/?") {
            PrintUsage();
            return 0;
        }
        else {
            std::fwprintf(stderr, L"unknown argument: %s\n", a.c_str());
            PrintUsage();
            return 1;
        }
    }

    if (opts.dllPath.empty()) {
        std::fwprintf(stderr, L"--dll is required\n");
        PrintUsage();
        return 1;
    }

    wchar_t abs[MAX_PATH]{};
    DWORD gt = GetFullPathNameW(opts.dllPath.c_str(), MAX_PATH, abs, nullptr);
    std::wprintf(L"[Loader] target=%s\n", opts.targetProcess.c_str());
    std::wprintf(L"[Loader] dll   =%s (--dll)\n", opts.dllPath.c_str());
    if (gt && gt < MAX_PATH)
        std::wprintf(L"[Loader] absolute=%s (cwd-relative)\n", abs);
    if (opts.verbose) {
        std::wprintf(L"[Loader] wait=%ums wipe=%d\n",
                     opts.waitTimeoutMs, int(opts.wipeHeaders));
    }

    Loader::InjectResult r = Loader::Inject(opts);
    if (!r.ok) {
        std::fwprintf(stderr, L"[Loader] injection failed: %s\n",
                      r.errorText.c_str());
        return 2;
    }

    std::wprintf(L"[Loader] injected pid=%lu base=%p size=%zu entryExit=%lu\n",
                 r.targetPid, r.remoteBase, r.imageSize, r.entryExitCode);
    return 0;
}
