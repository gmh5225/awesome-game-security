## Hashing

https://godbolt.org/z/jPeaWTno1
```C
#include <stdint.h>
#include <stddef.h>

uint64_t FNV_1a(const uint8_t *value, size_t length)
{
    uint64_t hash = UINT64_C(0xcbf29ce484222325);

    for (size_t index = 0; index < length; ++index)
    {
        hash ^= (uint64_t)value[index];
        hash *= UINT64_C(0x00000100000001b3);
    }

    return hash;
}
```

https://godbolt.org/z/oKxnvxh94
```c
unsigned long djb2_hash(const unsigned char *str) 
{ 
	unsigned long hash = 5381; 
	int c; 
	while ((c = *str++)) 
	{ // hash * 33 + c 
		hash = ((hash << 5) + hash) + c; 
	} 
	return hash; 
}

void returnStruct (struct employee_t* employee) {
    printf("Employee: %s\n", employee->name);

    printf("Age: %d\n", employee->age);
    printf("Salary: %f\n", employee->salary);

int main() {

    struct employee_t employees[MAX_EMPLOYEES];

    int i = 0;

int main(void)
{
    char source[] = "hello";

    while (source[length])
        length++;

    char *copy = malloc(length + 1);
    if (!copy)
        return 1;

    size_t i = 0;
    while (i < length)
    {
        copy[i] = source[i];
