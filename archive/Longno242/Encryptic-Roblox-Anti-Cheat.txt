# Encryptic Roblox Anti-Cheat

Server-authoritative anti-cheat for **Roblox** experiences. Pure Luau — drop into `ServerScriptService`, configure limits, and wire your remotes/combat.

Part of the [Encryptic Anti-Cheat](https://github.com/Longno242/Encryptic-Anti-Cheat) family (Windows client protection lives in that repo).

---

## Quick install (Studio)

1. Download or clone this repo.
2. Import **`demo/EncrypticDemo.rbxmx`** (File → Import, or drag into Explorer).
3. Move instances:
   - **`Encryptic`** folder → `ServerScriptService`
   - **`DemoGame`** → `ServerScriptService`
   - **`DemoClient`** → `StarterPlayer → StarterPlayerScripts`
4. Press **Play**. Click **AC** (top-right) or press **RightControl** to open the test panel.
5. Watch **View → Output → Server** for `[Encryptic]` violation lines.

Full walkthrough: [`demo/DEMO_SETUP.md`](demo/DEMO_SETUP.md)

---

## Use in your game

Copy **`ServerScriptService/Encryptic/`** into your place, then in a server Script:

```lua
local Encryptic = require(game.ServerScriptService.Encryptic.init)

Encryptic.start({
	maxSpeed = 48,
	maxWalkSpeed = 24,
	maxFireRate = 12,
	maxRemotePerSecond = 40,
	maxStrikes = 5,
	strikeDecaySeconds = 60,
	spawnGraceSeconds = 2.5,
	enableFlyGuard = true,
	enableNoclipGuard = true,
	enablePhysicsGuard = true,
	enableHealthGuard = true,
	enableExploitGuard = true,
})

-- Admins / playtesters
Encryptic.whitelist(123456789)

-- Before a legitimate server teleport / cutscene
Encryptic.allowTeleport(player, 2)

-- Legitimate medkit heal
Encryptic.HealthGuard.allowHeal(player, function()
	player.Character.Humanoid.Health += 50
end)
```

## Guards

| Guard | Detects |
|-------|---------|
| **MovementGuard** | Sustained horizontal speed / velocity hacks (lag-tolerant) |
| **TeleportGuard** | Large position jumps; soft-corrects + strike |
| **HumanoidGuard** | WalkSpeed, JumpPower, JumpHeight, HipHeight tamper |
| **FlyGuard** | PlatformStand fly, hover, illegal ascent |
| **NoclipGuard** | `HumanoidRootPart` collision off + phasing through solid geometry |
| **PhysicsGuard** | Injected `BodyVelocity` / movers / HRP constraints |
| **HealthGuard** | Godmode MaxHealth + illegal heal bursts |
| **FireRateGuard** | Shot rate limits |
| **RemoteGuard** | Global + per-remote spam limits |
| **ToolGuard** | Tool equip spam |
| **HitGuard** | Hit distance / reach + optional line-of-sight |
| **ExploitGuard** | Suspicious remote names |
| **BanManager** | Strikes → warn / kick, with idle decay + whitelist |

## Wire your systems

```lua
-- Guns
if not Encryptic.FireRateGuard.recordShot(player) then return end

-- Remotes
Encryptic.RemoteGuard.wrapRemoteEvent(myRemote, function(player, ...)
	-- your handler
end)

-- Melee / hitscan reach
if not Encryptic.HitGuard.validateHit(attacker, targetChar, hitPos) then return end

-- Tools
if not Encryptic.ToolGuard.trackEquip(player) then return end

-- Vehicles: mark movers as trusted
Encryptic.PhysicsGuard.trust(linearVelocityConstraint)
```

## Stability notes (anti-false-positive)

- **Spawn grace** — checks pause briefly after respawn so load-in isn't punished.
- **Seat / vehicle skip** — seated players aren't speed/teleport flagged.
- **Noclip** — only watches collision-critical parts (R15 limbs are often non-colliding by default).
- **Speed** — needs several consecutive overspeed frames before a strike.
- **`start()` once** — duplicate `Encryptic.start()` calls are ignored.
- **Strike decay** — idle players slowly lose strike points (`strikeDecaySeconds`).

```lua
-- Central violation tracking, strikes, kick/ban with decay + whitelist

local Players = game:GetService("Players")

local Shared = require(script.Parent.Shared)

local BanManager = {}
BanManager._strikes = {} :: { [Player]: { [string]: number } }
BanManager._totalStrikes = {} :: { [Player]: number }
BanManager._lastStrikeAt = {} :: { [Player]: number }

export type ViolationAction = "warn" | "kick" | "ignored"
export type ViolationInfo = {
	player: Player,
	guard: string,
	reason: string,
	severity: number,
}

local DEFAULT_MAX_STRIKES = 5
local DEFAULT_DECAY_SECONDS = 60
local maxStrikes = DEFAULT_MAX_STRIKES
local decaySeconds = DEFAULT_DECAY_SECONDS
local onViolation: ((ViolationInfo) -> ())? = nil
local onKick: ((Player, string) -> ())? = nil
local kicked: { [Player]: boolean } = {}
local started = false

local function decayStrikes(player: Player)
	if decaySeconds <= 0 then
		return
	end
	local last = BanManager._lastStrikeAt[player]
	local total = BanManager._totalStrikes[player]
	if not last or not total or total <= 0 then
		return
	end
	local elapsed = Shared.now() - last
	if elapsed < decaySeconds then
		return
	end
	-- Drop one strike point per decay window of inactivity
	local steps = math.floor(elapsed / decaySeconds)
	local nextTotal = math.max(0, total - steps)
	BanManager._totalStrikes[player] = nextTotal
	BanManager._lastStrikeAt[player] = Shared.now() - (elapsed % decaySeconds)
	if nextTotal == 0 then
		BanManager._strikes[player] = nil
	end
end

function BanManager.configure(opts: {
	maxStrikes: number?,
	decaySeconds: number?,
	onViolation: ((ViolationInfo) -> ())?,
	onKick: ((Player, string) -> ())?,
})
	maxStrikes = opts.maxStrikes or DEFAULT_MAX_STRIKES
	decaySeconds = if opts.decaySeconds ~= nil then opts.decaySeconds else DEFAULT_DECAY_SECONDS
	onViolation = opts.onViolation
	onKick = opts.onKick
end

function BanManager.record(violation: ViolationInfo): ViolationAction
	local player = violation.player
	if not player or not player.Parent or kicked[player] then
		return "ignored"
	end
	if Shared.isWhitelisted(player) then
		return "ignored"
	end

	decayStrikes(player)

	local guardStrikes = BanManager._strikes[player]
	if not guardStrikes then
		guardStrikes = {}
		BanManager._strikes[player] = guardStrikes
	end

	local severity = math.max(1, math.floor(violation.severity or 1))
	guardStrikes[violation.guard] = (guardStrikes[violation.guard] or 0) + severity
	BanManager._totalStrikes[player] = (BanManager._totalStrikes[player] or 0) + severity
	BanManager._lastStrikeAt[player] = Shared.now()

	warn(string.format(
		"[Encryptic] %s | %s | %s | strikes=%d/%d",
		violation.guard,
		player.Name,
		violation.reason,
		BanManager._totalStrikes[player],
		maxStrikes
	))

	if onViolation then
		onViolation(violation)
	end

	local total = BanManager._totalStrikes[player]
	if total >= maxStrikes then
		kicked[player] = true
		if onKick then
			onKick(player, violation.reason)
		else
			player:Kick("[Encryptic] " .. violation.reason)
		end
		return "kick"
	end

	return "warn"
end

function BanManager.getStrikes(player: Player): number
	decayStrikes(player)
	return BanManager._totalStrikes[player] or 0
end

function BanManager.getGuardStrikes(player: Player, guard: string): number
	local map = BanManager._strikes[player]
	return if map then (map[guard] or 0) else 0
end

function BanManager.reset(player: Player)
	BanManager._strikes[player] = nil
	BanManager._totalStrikes[player] = nil
	BanManager._lastStrikeAt[player] = nil
	kicked[player] = nil
end

function BanManager.whitelist(userId: number, allowed: boolean?)
	Shared.setWhitelisted(userId, if allowed == nil then true else allowed)
end

function BanManager.isWhitelisted(player: Player): boolean
	return Shared.isWhitelisted(player)
end

if not started then
	started = true
	Players.PlayerRemoving:Connect(function(player)
		BanManager.reset(player)
	end)
end

return BanManager
```

```lua
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

local Shared = require(script.Parent.Shared)

local ExploitGuard = {}
local suspiciousPatterns = {
	"execute", "loadstring", "getgenv", "syn", "fluxus", "infinite",
	"godmode", "backdoor", "bypass", "exploit", "dex", "remote spy",
}
local onViolation: ((Player, string) -> ())? = nil
local hooked = {} :: { [Instance]: boolean }
local started = false

-- Names that look scary but are common in legitimate games — don't false-flag these alone
local ALLOWLIST = {
	speed = true,
	teleport = true,
	admin = true,
	kill = true,
	ban = true,
	noclip = true, -- demo may create TestExploit; validateRemote still usable per-call
}

local function isSuspicious(name: string): boolean
	local lower = string.lower(name)
	if ALLOWLIST[lower] then
		return false
	end
	for _, pattern in suspiciousPatterns do
		if string.find(lower, pattern, 1, true) then
			return true
		end
	end
	return false
end

local function hookRemote(remote: Instance)
	if not remote:IsA("RemoteEvent") and not remote:IsA("RemoteFunction") then
		return
	end
	if hooked[remote] then
		return
	end
	hooked[remote] = true

	if isSuspicious(remote.Name) then
		warn("[Encryptic] Suspicious remote registered:", remote:GetFullName())
	end
end

local function scan(container: Instance)
	for _, inst in container:GetDescendants() do
		hookRemote(inst)
	end
	container.DescendantAdded:Connect(hookRemote)
end

function ExploitGuard.start(opts: {
	onViolation: ((Player, string) -> ())?,
	extraPatterns: { string }?,
})
	onViolation = opts.onViolation
	if opts.extraPatterns then
		for _, p in opts.extraPatterns do
			table.insert(suspiciousPatterns, string.lower(p))
		end
	end

	if started then
		return
	end
	started = true

	scan(ReplicatedStorage)

	local workspaceRemotes = game:GetService("Workspace"):FindFirstChild("Remotes")
	if workspaceRemotes then
		scan(workspaceRemotes)
	end

	-- Catch remotes created later under Workspace.Remotes
	game:GetService("Workspace").ChildAdded:Connect(function(child)
		if child.Name == "Remotes" then
			scan(child)
		end
	end)

	Players.PlayerRemoving:Connect(function()
		-- hooked remotes are Instance-keyed; no player cleanup needed
	end)
end

function ExploitGuard.validateRemote(player: Player, remote: Instance): boolean
	if Shared.isWhitelisted(player) then
		return true
	end
	if not remote then
		return false
	end
	if isSuspicious(remote.Name) then
		if onViolation then
			onViolation(player, "Blocked suspicious remote: " .. remote.Name)
		end
		return false
	end
	return true
end

function ExploitGuard.isSuspiciousName(name: string): boolean
	return isSuspicious(name)
end

return ExploitGuard
```

```lua
local Players = game:GetService("Players")

local Shared = require(script.Parent.Shared)

local FireRateGuard = {}
local maxRate = 12
local lastFire: { [Player]: { number } } = {}
local onViolation: ((Player, string) -> ())? = nil
local started = false

function FireRateGuard.start(opts: number | { maxFireRate: number?, onViolation: ((Player, string) -> ())? })
	if typeof(opts) == "number" then
		maxRate = opts
	else
		maxRate = opts.maxFireRate or maxRate
		onViolation = opts.onViolation
	end

	if started then
		return
	end
	started = true

	Players.PlayerRemoving:Connect(function(player)
		lastFire[player] = nil
	end)
end

function FireRateGuard.recordShot(player: Player): boolean
	if Shared.isWhitelisted(player) then
		return true
	end

	if not Shared.rateLimit(lastFire, player, 1, maxRate) then
		if onViolation then
			local count = #(lastFire[player] or {})
			onViolation(player, string.format("Fire rate exploit shots=%d max=%d", count, maxRate))
		end
		return false
	end

	return true
end

function FireRateGuard.reset(player: Player)
	lastFire[player] = nil
end

return FireRateGuard
```

`ServerScriptService/Encryptic/FlyGuard.lua`:

```lua
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")

local Shared = require(script.Parent.Shared)

local FlyGuard = {}
local maxAirTime = 4
local airTime: { [Player]: number } = {}
local lastViolation: { [Player]: number } = {}
local violationCooldown = 1.0
local onViolation: ((Player, string) -> ())? = nil
local started = false

-- States where mid-air is expected; we still watch for hover within these
local CLIMB_OR_SWIM = {
	[Enum.HumanoidStateType.Climbing] = true,
	[Enum.HumanoidStateType.Swimming] = true,
}

local function report(player: Player, reason: string)
	if not onViolation or Shared.shouldSkipPlayer(player) then
		return
	end
	if Shared.cooldownOk(lastViolation, player, violationCooldown) then
		onViolation(player, reason)
	end
end

local function isGrounded(character: Model, root: BasePart): boolean
	local params = RaycastParams.new()
	params.FilterDescendantsInstances = { character }
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.IgnoreWater = false
	-- Slightly longer ray + slight forward sample reduces edge false positives
	local down = Workspace:Raycast(root.Position, Vector3.new(0, -5.5, 0), params)
	if down then
		return true
	end
	local offset = root.CFrame.LookVector * 1.2
	return Workspace:Raycast(root.Position + offset, Vector3.new(0, -5.5, 0), params) ~= nil
end

function FlyGuard.start(opts: {
	maxAirTime: number?,
	onViolation: ((Player, string) -> ())?,
})
	maxAirTime = opts.maxAirTime or maxAirTime
	onViolation = opts.onViolation

	if started then
		return
	end
	started = true

	RunService.Heartbeat:Connect(function(dt)
		for _, player in Players:GetPlayers() do
			if Shared.shouldSkipPlayer(player) then
				airTime[player] = 0
				continue
			end

			local character = player.Character
			local root = Shared.getRoot(character)
			local humanoid = Shared.getHumanoid(character)
			if not root or not humanoid then
				airTime[player] = 0
				continue
			end

			if humanoid.SeatPart or humanoid:GetState() == Enum.HumanoidStateType.Seated then
				airTime[player] = 0
				continue
			end

			local state = humanoid:GetState()
			if CLIMB_OR_SWIM[state] then
				airTime[player] = 0
				continue
			end

			-- Classic executor fly
			if humanoid.PlatformStand then
				report(player, "Fly exploit PlatformStand")
				humanoid.PlatformStand = false
				local v = root.AssemblyLinearVelocity
				root.AssemblyLinearVelocity = Vector3.new(v.X, math.min(v.Y, -40), v.Z)
				airTime[player] = 0
				continue
			end

			if isGrounded(character, root) then
				airTime[player] = 0
				continue
			end

			local velocity = root.AssemblyLinearVelocity
			local verticalSpeed = velocity.Y
			local horizontalSpeed = Shared.horizontalSpeed(velocity)

			-- Hover / float: not falling, not a normal jump peak
			local hovering = math.abs(verticalSpeed) < 5 and horizontalSpeed < (humanoid.WalkSpeed + 8)
			-- Ascending without jump state for too long (anti-gravity fly)
			local risingHack = verticalSpeed > 8
				and state ~= Enum.HumanoidStateType.Jumping
				and state ~= Enum.HumanoidStateType.Freefall

			if hovering or risingHack then
				airTime[player] = (airTime[player] or 0) + dt
				if airTime[player] > maxAirTime then
					local kind = if hovering then "hover" else "ascent"
					report(player, string.format("Fly exploit %s airTime=%.1fs", kind, airTime[player]))
					root.AssemblyLinearVelocity = Vector3.new(velocity.X, -80, velocity.Z)
					airTime[player] = 0
				end
			else
				-- Decay instead of hard reset so brief jitter doesn't wipe evidence
				airTime[player] = math.max(0, (airTime[player] or 0) - dt * 1.5)
			end
		end
	end)

	Players.PlayerRemoving:Connect(function(player)
		airTime[player] = nil
		lastViolation[player] = nil
	end)
end

return FlyGuard
```

```lua
--[[
	HealthGuard — godmode / illegal heal detection.
	Tracks sudden health increases and MaxHealth inflation.
]]

local Players = game:GetService("Players")

local Shared = require(script.Parent.Shared)

local HealthGuard = {}
local maxHealthCap = 100
local maxHealPerSecond = 40
local onViolation: ((Player, string) -> ())? = nil
local lastViolation: { [Player]: number } = {}
local lastHealth: { [Player]: number } = {}
local healBudget: { [Player]: number } = {}
local lastTick: { [Player]: number } = {}
local trustedHeal: { [Player]: boolean } = {}
local violationCooldown = 1.0
local started = false

local function report(player: Player, reason: string)
	if not onViolation or Shared.shouldSkipPlayer(player) then
		return
	end
	if Shared.cooldownOk(lastViolation, player, violationCooldown) then
		onViolation(player, reason)
	end
end

local function bindHumanoid(player: Player, humanoid: Humanoid)
	lastHealth[player] = humanoid.Health
	healBudget[player] = 0
	lastTick[player] = Shared.now()

	humanoid.HealthChanged:Connect(function(health)
		if Shared.shouldSkipPlayer(player) or trustedHeal[player] then
			lastHealth[player] = health
			return
		end

		local prev = lastHealth[player] or health
		lastHealth[player] = health

		if humanoid.MaxHealth > maxHealthCap + 0.5 then
			report(player, string.format("MaxHealth tamper: %.1f", humanoid.MaxHealth))
			humanoid.MaxHealth = maxHealthCap
			if humanoid.Health > maxHealthCap then
				humanoid.Health = maxHealthCap
			end
			return
		end

		if health <= prev then
			return
		end

		local gained = health - prev
		local now = Shared.now()
		local dt = math.max(now - (lastTick[player] or now), 1 / 60)
		lastTick[player] = now

		-- Decay budget
		healBudget[player] = math.max(0, (healBudget[player] or 0) - maxHealPerSecond * dt)
		healBudget[player] = (healBudget[player] or 0) + gained

		if healBudget[player] > maxHealPerSecond * 1.5 then
			report(player, string.format("Illegal heal +%.1f (budget=%.1f)", gained, healBudget[player]))
			humanoid.Health = prev
			healBudget[player] = 0
		end
	end)
end

function HealthGuard.start(opts: {
	maxHealth: number?,
	maxHealPerSecond: number?,
	onViolation: ((Player, string) -> ())?,
})
	maxHealthCap = opts.maxHealth or maxHealthCap
	maxHealPerSecond = opts.maxHealPerSecond or maxHealPerSecond
	onViolation = opts.onViolation

	if started then
		return
	end
	started = true

	local function onCharacter(player: Player, character: Model)
		task.spawn(function()
			local humanoid = character:WaitForChild("Humanoid", 5) :: Humanoid?
			if humanoid then
				Shared.grantGrace(player, 2)
				bindHumanoid(player, humanoid)
			end
		end)
	end

	local function onPlayer(player: Player)
		player.CharacterAdded:Connect(function(character)
			onCharacter(player, character)
		end)
		if player.Character then
			onCharacter(player, player.Character)
		end
	end

	for _, player in Players:GetPlayers() do
		onPlayer(player)
	end
	Players.PlayerAdded:Connect(onPlayer)

	Players.PlayerRemoving:Connect(function(player)
		lastViolation[player] = nil
		lastHealth[player] = nil
		healBudget[player] = nil
		lastTick[player] = nil
		trustedHeal[player] = nil
	end)
end

-- Wrap legitimate heals / medkits
function HealthGuard.allowHeal(player: Player, fn: () -> ())
	trustedHeal[player] = true
	local ok, err = pcall(fn)
	trustedHeal[player] = nil
	if not ok then
		warn("[Encryptic] HealthGuard.allowHeal error:", err)
	end
end

return HealthGuard
```

```lua
local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")

local Shared = require(script.Parent.Shared)

local HitGuard = {}
local maxHitDistance = 20
local requireLineOfSight = true
local onViolation: ((Player, string) -> ())? = nil
local lastViolation: { [Player]: number } = {}
local violationCooldown = 0.5
local started = false

function HitGuard.start(opts: {
	maxHitDistance: number?,
	requireLineOfSight: boolean?,
	onViolation: ((Player, string) -> ())?,
})
	maxHitDistance = opts.maxHitDistance or maxHitDistance
	if opts.requireLineOfSight ~= nil then
		requireLineOfSight = opts.requireLineOfSight
	end
	onViolation = opts.onViolation

	if started then
		return
	end
	started = true

	Players.PlayerRemoving:Connect(function(player)
		lastViolation[player] = nil
	end)
end

local function hasLineOfSight(attackerChar: Model, from: Vector3, to: Vector3, targetChar: Model): boolean
	local params = RaycastParams.new()
	params.FilterDescendantsInstances = { attackerChar }
	params.FilterType = Enum.RaycastFilterType.Exclude
	local dir = to - from
	local result = Workspace:Raycast(from, dir, params)
	if not result then
		return true
	end
	return result.Instance:IsDescendantOf(targetChar)
end

-- Call from your combat system before applying damage
function HitGuard.validateHit(attacker: Player, targetChar: Model, hitPosition: Vector3?): boolean
	if Shared.isWhitelisted(attacker) then
		return true
	end

	local attackerChar = attacker.Character
	local attackerRoot = Shared.getRoot(attackerChar)
	local targetRoot = Shared.getRoot(targetChar)
	if not attackerRoot or not targetRoot then
		return false
	end

	local from = hitPosition or targetRoot.Position
	local distRoot = (attackerRoot.Position - targetRoot.Position).Magnitude
	local distHit = (attackerRoot.Position - from).Magnitude
	local dist = math.min(distRoot, distHit)

	-- Small latency cushion
	local limit = maxHitDistance + 2
	if dist > limit then
		if onViolation and Shared.cooldownOk(lastViolation, attacker, violationCooldown) then
			onViolation(attacker, string.format("Hit distance exploit dist=%.1f max=%.1f", dist, limit))
		end
		return false
	end

	if requireLineOfSight and attackerChar then
		local origin = attackerRoot.Position + Vector3.new(0, 1.5, 0)
		if not hasLineOfSight(attackerChar, origin, from, targetChar) then
			if onViolation and Shared.cooldownOk(lastViolation, attacker, violationCooldown) then
				onViolation(attacker, "Hit LOS exploit (wallbang)")
			end
			return false
		end
	end

	return true
end
```

```lua
-- Shared helpers for Encryptic guards (grace windows, cleanup, rate limits)

local Players = game:GetService("Players")

local Shared = {}

local graceUntil: { [Player]: number } = {}
local trustedUntil: { [Player]: number } = {}
local whitelist: { [number]: boolean } = {} -- UserId

function Shared.now(): number
	return os.clock()
end

function Shared.isWhitelisted(player: Player): boolean
	return whitelist[player.UserId] == true
end

function Shared.setWhitelisted(userId: number, allowed: boolean)
	if allowed then
		whitelist[userId] = true
	else
		whitelist[userId] = nil
	end
end

function Shared.grantGrace(player: Player, seconds: number?)
	local untilTime = Shared.now() + (seconds or 2)
	graceUntil[player] = math.max(graceUntil[player] or 0, untilTime)
end

function Shared.trustMovement(player: Player, seconds: number?)
	local untilTime = Shared.now() + (seconds or 1.5)
	trustedUntil[player] = math.max(trustedUntil[player] or 0, untilTime)
	Shared.grantGrace(player, seconds or 1.5)
end

function Shared.inGrace(player: Player): boolean
	if Shared.isWhitelisted(player) then
		return true
	end
	local t = Shared.now()
	return (graceUntil[player] or 0) > t or (trustedUntil[player] or 0) > t
end

function Shared.shouldSkipPlayer(player: Player): boolean
	return not player.Parent or Shared.inGrace(player)
end

-- Sliding window rate limiter helper
function Shared.pruneOlderThan(list: { number }, cutoff: number)
	while #list > 0 and list[1] < cutoff do
		table.remove(list, 1)
	end
end

function Shared.rateLimit(
	historyMap: { [Player]: { number } },
	player: Player,
	windowSeconds: number,
	maxCount: number
): boolean
	local now = Shared.now()
	local history = historyMap[player]
	if not history then
		history = {}
		historyMap[player] = history
	end
	table.insert(history, now)
	Shared.pruneOlderThan(history, now - windowSeconds)
	return #history <= maxCount
end

function Shared.cooldownOk(
	lastMap: { [Player]: number },
	player: Player,
	cooldown: number
): boolean
	local now = Shared.now()
	if (now - (lastMap[player] or 0)) < cooldown then
		return false
	end
	lastMap[player] = now
	return true
end

function Shared.clearPlayer(player: Player)
	graceUntil[player] = nil
	trustedUntil[player] = nil
end

function Shared.bindCharacterGrace(seconds: number?)
	local grace = seconds or 2.5
	local function onPlayer(player: Player)
		player.CharacterAdded:Connect(function()
			Shared.grantGrace(player, grace)
		end)
		if player.Character then
			Shared.grantGrace(player, grace)
		end
	end
	for _, player in Players:GetPlayers() do
		onPlayer(player)
	end
	Players.PlayerAdded:Connect(onPlayer)
	Players.PlayerRemoving:Connect(Shared.clearPlayer)
end

function Shared.getRoot(character: Model?): BasePart?
	if not character then
		return nil
	end
	return character:FindFirstChild("HumanoidRootPart") :: BasePart?
end

function Shared.getHumanoid(character: Model?): Humanoid?
	if not character then
		return nil
	end
	return character:FindFirstChildOfClass("Humanoid")
end

function Shared.horizontalSpeed(velocity: Vector3): number
	return Vector3.new(velocity.X, 0, velocity.Z).Magnitude
end

return Shared
```

```lua
					continue

			lastTime[player] = now
```

```rbxmx
local function setNoclip(player: Player, enabled: boolean)
	local character = player.Character
	if not character then
		return
	end

	if not enabled then
		for _, part in noclipParts[player] or {} do
			if part.Parent then
				part.CanCollide = true
			end
		end
		noclipParts[player] = nil
		return
	end

	-- Only disable critical parts (matches what NoclipGuard watches)
	noclipParts[player] = {}
	for _, name in { "HumanoidRootPart", "Head", "UpperTorso", "LowerTorso", "Torso" } do
		local part = character:FindFirstChild(name)
		if part and part:IsA("BasePart") then
			part.CanCollide = false
			table.insert(noclipParts[player], part)
		end
	end
end

		return

					return

			end)
		end
	elseif action == "noclip" then
		setNoclip(player, enabled == true)
	elseif action == "teleport" then
		root.CFrame = root.CFrame + root.CFrame.LookVector * 200
	elseif action == "jump" then
		humanoid.UseJumpPower = true
		humanoid.JumpPower = 150
		humanoid.JumpHeight = 20
		humanoid.HipHeight = 12
	elseif action == "velocity" then
		root.AssemblyLinearVelocity = root.CFrame.LookVector * 120 + Vector3.new(0, 30, 0)
	elseif action == "godmode" then
		humanoid.MaxHealth = if enabled then 9999 else 100
		humanoid.Health = humanoid.MaxHealth
	elseif action == "physics" then
		local existing = root:FindFirstChild("DemoFlyForce")
		if existing then
			existing:Destroy()

		if enabled then
```

```py
#!/usr/bin/env python3
"""Build EncrypticDemo.rbxmx for drag-and-drop import into Roblox Studio."""

from pathlib import Path
import xml.sax.saxutils as sax

ROOT = Path(__file__).resolve().parent.parent
ENCRYPTIC = ROOT / "ServerScriptService" / "Encryptic"
OUT = Path(__file__).resolve().parent / "EncrypticDemo.rbxmx"

MODULES = [
    "init",
    "Shared",
    "BanManager",
    "MovementGuard",
    "TeleportGuard",
    "HumanoidGuard",
    "FlyGuard",
    "NoclipGuard",
    "PhysicsGuard",
    "HealthGuard",
    "FireRateGuard",
    "RemoteGuard",
    "ExploitGuard",
    "ToolGuard",
    "HitGuard",
]

SCRIPTS = [
    ("ModuleScript", ENCRYPTIC / f"{name}.lua", name)
    for name in MODULES
] + [
    ("Script", Path(__file__).resolve().parent / "ServerScriptService" / "DemoGame.server.lua", "DemoGame"),
    (
        "LocalScript",
        Path(__file__).resolve().parent / "StarterPlayer" / "StarterPlayerScripts" / "DemoClient.client.lua",
        "DemoClient",
    ),
]


def cdata(text: str) -> str:
    return text.replace("]]>", "]]]]><![CDATA[>")


def item(class_name: str, name: str, source: str | None, referent: str, indent: str) -> str:
    lines = [
        f'{indent}<Item class="{class_name}" referent="{referent}">',
        f"{indent}  <Properties>",
        f"{indent}    <string name=\"Name\">{sax.escape(name)}</string>",
    ]
    if source is not None:
        lines.append(f"{indent}    <ProtectedString name=\"Source\"><![CDATA[{cdata(source)}]]></ProtectedString>")
    lines.extend([f"{indent}  </Properties>", f"{indent}</Item>"])
    return "\n".join(lines)


def main() -> None:
    ref = 0
    parts = ['<?xml version="1.0" encoding="utf-8"?>', '<roblox version="4">']

    ref += 1
    model_ref = f"RBX{ref}"
    parts.append(f'  <Item class="Model" referent="{model_ref}">')
    parts.append("    <Properties>")
    parts.append("      <string name=\"Name\">EncrypticDemo</string>")
    parts.append("    </Properties>")

    ref += 1
    folder_ref = f"RBX{ref}"
    parts.append(f'    <Item class="Folder" referent="{folder_ref}">')
    parts.append("      <Properties>")
    parts.append("        <string name=\"Name\">Encryptic</string>")
    parts.append("      </Properties>")

    module_count = len(MODULES)
    for class_name, path, name in SCRIPTS[:module_count]:
        ref += 1
        source = path.read_text(encoding="utf-8")
        parts.append(item(class_name, name, source, f"RBX{ref}", "      "))

    parts.append("    </Item>")

    for class_name, path, name in SCRIPTS[module_count:]:
        ref += 1
        source = path.read_text(encoding="utf-8")
        parts.append(item(class_name, name, source, f"RBX{ref}", "    "))

    parts.append("  </Item>")
    parts.append("</roblox>")

    OUT.write_text("\n".join(parts) + "\n", encoding="utf-8")
    print(f"Wrote {OUT}")


if __name__ == "__main__":
    main()
```
