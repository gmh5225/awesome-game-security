# UsbDetectionBypass

[简体中文](README.zh-CN.md)

UsbDetectionBypass is an LSPosed/Xposed module focused only on bypassing USB connection and USB debugging detection inside scoped Android apps. It masks common Java/framework and native signals so the target process sees a disconnected, non-debuggable USB state.

Use it only for authorized testing and research on devices and apps you are allowed to inspect.

## Features

- Masks `adb_enabled` and `development_settings_enabled`.
- Masks `android.os.SystemProperties` and native `__system_property_get`.
- Sanitizes `USB_STATE` and `ACTION_BATTERY_CHANGED` broadcasts.
- Hooks `Intent` extra reads for USB and battery state.
- Hooks common `UsbManager` APIs so scoped apps see no attached USB devices or enabled USB functions.
- Rewrites selected command checks including `getprop`, `settings`, `dumpsys usb`, `dumpsys battery`, and common sysfs reads.
- Hooks native `open`, `openat`, and `fopen` for USB-related sysfs paths such as `/sys/class/power_supply/*usb*`, `/sys/class/android_usb/*`, and `/config/usb_gadget/*`.
- Logs module loading and hook hits through LSPosed logs, logcat, and a per-app diagnostic file.

## Project Layout

- `app/src/main/java/io/github/ryttermohn/usbdetectionbypass/HookEntry.kt`  
  Main LSPosed Java entry point. Installs Java/framework hooks and diagnostic logging.

- `app/src/main/java/io/github/ryttermohn/usbdetectionbypass/LogUtil.kt`  
  Bridges native logs into LSPosed and logcat.

- `app/src/main/cpp/usb_detection_bypass.cpp`  
  LSPosed native entry point. Installs native USB property and file-read hooks.

- `app/src/main/assets/xposed_init`  
  Xposed module entry declaration.

- `app/src/main/assets/native_init`  
  LSPosed native module declaration.

## Build

Use JDK 17 or newer. Android Studio's bundled JBR also works.

```powershell
.\gradlew.bat assembleDebug
```

The debug APK is generated at:

```text
app/build/outputs/apk/debug/app-debug.apk
```

## Usage

1. Install the APK on a rooted device with LSPosed enabled.
2. Enable the module in LSPosed.
3. Select your authorized target app in the module scope.
4. Reboot the device, or fully stop and restart the target app after changing scope.
5. Check logs with:

```bash
adb logcat -s UsbBypass UsbBypassNative UsbDetectionBypassNative
```

The module also writes `UsbDetectionBypass.log` under the target app's external files directory when `Application.attach()` is reached.

## Notes

- The module package is `io.github.ryttermohn.usbdetectionbypass`.
- The default LSPosed scope resource is intentionally empty. Select target apps from LSPosed or add package names to `app/src/main/res/values/arrays.xml`.
- If no `UsbBypass` log appears, the module has not entered the target process. Recheck LSPosed scope, module enablement, package/process name, and whether the target app was restarted after module changes.

#include <jni.h>
#include <android/log.h>
#include <cerrno>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <dlfcn.h>
#include <fcntl.h>
#include <stdarg.h>
#include <sys/stat.h>
#include <sys/types.h>
#include <unistd.h>
#include "native_api.h"

#define LOG_TAG "UsbDetectionBypassNative"
#define LOGI(...) __android_log_print(ANDROID_LOG_INFO, LOG_TAG, __VA_ARGS__)

static JavaVM *g_JavaVm = nullptr;
static jclass g_LogUtilClass = nullptr;
static jmethodID g_LogFromNativeMethod = nullptr;
static HookFunType g_HookFunction = nullptr;

using SystemPropertyGet = int (*)(const char *name, char *value);
using OpenFunc = int (*)(const char *pathname, int flags, ...);
using OpenAtFunc = int (*)(int dirfd, const char *pathname, int flags, ...);
using FopenFunc = FILE *(*)(const char *pathname, const char *mode);

static SystemPropertyGet g_OriginalSystemPropertyGet = nullptr;
static OpenFunc g_OriginalOpen = nullptr;
static OpenAtFunc g_OriginalOpenAt = nullptr;
static FopenFunc g_OriginalFopen = nullptr;

static bool g_SystemPropertyHookInstalled = false;
static bool g_FileAccessHooksInstalled = false;

static void XLog(const char *format, ...) {
    va_list args;
    va_start(args, format);
    char buf[1024];
    vsnprintf(buf, sizeof(buf), format, args);
    va_end(args);

    LOGI("[XPOSED] %s", buf);

    JNIEnv *env = nullptr;
    if (g_JavaVm && g_LogUtilClass && g_LogFromNativeMethod &&
        g_JavaVm->GetEnv(reinterpret_cast<void **>(&env), JNI_VERSION_1_6) == JNI_OK) {
        jstring jmsg = env->NewStringUTF(buf);
        env->CallStaticVoidMethod(g_LogUtilClass, g_LogFromNativeMethod, jmsg);
        env->DeleteLocalRef(jmsg);
    }
}

static bool str_contains(const char *s, const char *sub) {
    return s && sub && strstr(s, sub) != nullptr;
}

static const char *fake_property_value(const char *name) {
    if (!name) return nullptr;

    if (strcmp(name, "sys.usb.config") == 0) return "none";
    if (strcmp(name, "sys.usb.state") == 0) return "none";
    if (strcmp(name, "persist.sys.usb.config") == 0) return "none";
    if (strcmp(name, "vendor.usb.config") == 0) return "none";
    if (strcmp(name, "vendor.usb.state") == 0) return "none";
    if (strcmp(name, "persist.vendor.usb.config") == 0) return "none";
    if (strcmp(name, "sys.usb.ffs.ready") == 0) return "0";
    if (strcmp(name, "init.svc.adbd") == 0) return "stopped";
    if (strcmp(name, "init.svc_debug_pid.adbd") == 0) return "0";
    if (strcmp(name, "service.adb.tcp.port") == 0) return "-1";
    if (strcmp(name, "persist.adb.tcp.port") == 0) return "-1";
    if (strcmp(name, "persist.service.adb.enable") == 0) return "0";
    if (strcmp(name, "persist.sys.usb.adb") == 0) return "0";
    if (strcmp(name, "ro.debuggable") == 0) return "0";
    if (strcmp(name, "ro.secure") == 0) return "1";

    return nullptr;
}

static int write_property_value(char *value, const char *fake) {
    if (!value || !fake) return 0;
    size_t len = strlen(fake);
    memcpy(value, fake, len + 1);
    return static_cast<int>(len);
}

static int replaced_system_property_get(const char *name, char *value) {
    const char *fake = fake_property_value(name);
    if (fake) {
        XLog("hook hit: native __system_property_get, %s -> %s", name ? name : "(null)", fake);
        return write_property_value(value, fake);
    }

    if (g_OriginalSystemPropertyGet) {
        return g_OriginalSystemPropertyGet(name, value);
    }
    return 0;
}

static const char *fake_file_content(const char *path) {
    if (!path) return nullptr;

    if (str_contains(path, "/sys/class/power_supply/") && str_contains(path, "usb") &&
        (str_contains(path, "/online") || str_contains(path, "/present"))) {
        return "0\n";
    }
    if (str_contains(path, "/sys/class/power_supply/") && str_contains(path, "usb") &&
        (str_contains(path, "/type") || str_contains(path, "/real_type"))) {
        return "Unknown\n";
    }
    if (str_contains(path, "/sys/class/power_supply/") && str_contains(path, "usb") &&
        (str_contains(path, "/current_now") || str_contains(path, "/voltage_now"))) {
        return "0\n";
    }
    if (str_contains(path, "/sys/class/power_supply/usb/online")) return "0\n";
    if (str_contains(path, "/sys/class/power_supply/usb/present")) return "0\n";
    if (str_contains(path, "/sys/class/power_supply/usb/type")) return "Unknown\n";
    if (str_contains(path, "/sys/class/power_supply/usb/real_type")) return "Unknown\n";
    if (str_contains(path, "/sys/class/power_supply/battery/status")) return "Discharging\n";
    if (str_contains(path, "/sys/class/android_usb") && str_contains(path, "state")) {
        return "DISCONNECTED\n";
    }
    if (str_contains(path, "/sys/class/android_usb") && str_contains(path, "functions")) {
        return "none\n";
    }
    if (str_contains(path, "/config/usb_gadget") && str_contains(path, "UDC")) {
        return "\n";
    }
    if (str_contains(path, "/config/usb_gadget") && str_contains(path, "functions")) {
        return "none\n";
    }
    if (str_contains(path, "/dev/usb-ffs/adb") || str_contains(path, "/dev/mtp_usb")) {
        return "\n";
    }

    return nullptr;
}

static bool is_read_only_open(int flags) {
    return (flags & O_ACCMODE) == O_RDONLY && (flags & O_DIRECTORY) == 0;
}

static int create_fake_read_fd(const char *path, const char *content) {
    int pipe_fds[2];
    if (pipe(pipe_fds) != 0) {
        XLog("fake file pipe failed for %s: %s", path ? path : "(null)", strerror(errno));
        errno = ENOENT;
        return -1;
    }

    size_t remaining = strlen(content);
    const char *cursor = content;
    while (remaining > 0) {
        ssize_t written = write(pipe_fds[1], cursor, remaining);
        if (written <= 0) break;
        cursor += written;
        remaining -= static_cast<size_t>(written);
    }
    close(pipe_fds[1]);

    XLog("hook hit: native file read, %s -> %s", path ? path : "(null)", content);
    return pipe_fds[0];
}

static int replaced_open(const char *pathname, int flags, ...) {
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list args;
        va_start(args, flags);
        mode = static_cast<mode_t>(va_arg(args, int));
        va_end(args);
    }

    const char *fake = fake_file_content(pathname);
    if (fake && is_read_only_open(flags)) {
        return create_fake_read_fd(pathname, fake);
    }

    if (!g_OriginalOpen) {
        errno = ENOENT;
        return -1;
    }
    if (flags & O_CREAT) return g_OriginalOpen(pathname, flags, mode);
    return g_OriginalOpen(pathname, flags);
}

static int replaced_openat(int dirfd, const char *pathname, int flags, ...) {
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list args;
        va_start(args, flags);
        mode = static_cast<mode_t>(va_arg(args, int));
        va_end(args);
    }

    const char *fake = fake_file_content(pathname);
    if (fake && is_read_only_open(flags)) {
        return create_fake_read_fd(pathname, fake);
    }

    if (!g_OriginalOpenAt) {
        errno = ENOENT;
        return -1;
    }
    if (flags & O_CREAT) return g_OriginalOpenAt(dirfd, pathname, flags, mode);
    return g_OriginalOpenAt(dirfd, pathname, flags);
}

static FILE *replaced_fopen(const char *pathname, const char *mode) {
    const char *fake = fake_file_content(pathname);
    if (fake && mode && mode[0] == 'r') {
        int fd = create_fake_read_fd(pathname, fake);
        if (fd < 0) return nullptr;
        FILE *fp = fdopen(fd, mode);
        if (!fp) close(fd);
        return fp;
    }

    if (!g_OriginalFopen) {
        errno = ENOENT;
        return nullptr;
    }
    return g_OriginalFopen(pathname, mode);
}

static void *find_libc_symbol(const char *symbol_name) {
    void *libc = dlopen("libc.so", RTLD_NOW);
    void *symbol = libc ? dlsym(libc, symbol_name) : nullptr;
    if (!symbol) {
        symbol = dlsym(RTLD_DEFAULT, symbol_name);
    }
    return symbol;
}

static bool hook_libc_symbol(const char *symbol_name, void *replacement, void **backup) {
    void *symbol = find_libc_symbol(symbol_name);
    if (!symbol) {
        XLog("failed to find %s", symbol_name);
        return false;
    }

    int result = g_HookFunction(symbol, replacement, backup);
    if (result == 0) {
        XLog("native %s hook installed", symbol_name);
        return true;
    }

    XLog("native %s hook failed: %d", symbol_name, result);
    return false;
}

static void install_system_property_hook() {
    if (g_SystemPropertyHookInstalled || !g_HookFunction) return;

    if (hook_libc_symbol(
            "__system_property_get",
            reinterpret_cast<void *>(replaced_system_property_get),

        g_SystemPropertyHookInstalled = true;
    }
}

    private fun replacementShellCommand(command: Any?): Any? {

    private fun shellArrayForOutput(output: String): Array<String> {

    private fun fakeShellOutput(commandText: String): String? {
        val normalized = commandText.lowercase(Locale.US)
        fakeSysfsCommandOutput(normalized)?.let { return it }

        if (normalized.contains("dumpsys usb")) {
            return "USB connected: false"
        }
        if (normalized.contains("dumpsys battery")) {
            return "USB powered: false"
        }
        if (normalized.contains("lsusb") || normalized.contains("/sys/bus/usb/devices")) {
            return ""
        }
        if (!looksLikeUsbConnectionCheck(normalized)) return null

        return when {
            normalized.contains("adb_enabled") -> "0"
            normalized.contains("development_settings_enabled") -> "0"
            normalized.contains("init.svc.adbd") -> "stopped"
            normalized.contains("init.svc_debug_pid.adbd") -> "0"
            normalized.contains("adb.tcp.port") -> "-1"
            normalized.contains("ro.debuggable") -> "0"
            normalized.contains("ro.secure") -> "1"
            normalized.contains("sys.usb.config") -> "none"
            normalized.contains("sys.usb.state") -> "none"
            normalized.contains("persist.sys.usb.config") -> "none"
            normalized.contains("vendor.usb.config") -> "none"
            normalized.contains("vendor.usb.state") -> "none"
            normalized.contains("persist.vendor.usb.config") -> "none"
            normalized.contains("sys.usb.ffs.ready") -> "0"
            else -> ""
        }
    }

    private fun fakeSysfsCommandOutput(commandText: String): String? {
        return when {
            commandText.contains("/sys/class/power_supply/") &&
                commandText.contains("usb") &&
                (commandText.contains("/online") || commandText.contains("/present")) -> "0"
            commandText.contains("/sys/class/power_supply/") &&
                commandText.contains("usb") &&
                (commandText.contains("/type") || commandText.contains("/real_type")) -> "Unknown"
            commandText.contains("/sys/class/power_supply/") &&

            commandText.contains("/sys/class/power_supply/usb/online") -> "0"
            commandText.contains("/sys/class/power_supply/usb/present") -> "0"
            commandText.contains("/sys/class/power_supply/usb/type") -> "Unknown"
            commandText.contains("/sys/class/power_supply/usb/real_type") -> "Unknown"
            commandText.contains("/sys/class/power_supply/battery/status") -> "Discharging"
            commandText.contains("/sys/class/android_usb") && commandText.contains("state") -> "DISCONNECTED"
            commandText.contains("/sys/class/android_usb") && commandText.contains("functions") -> "none"
            commandText.contains("/config/usb_gadget") && commandText.contains("udc") -> ""
            commandText.contains("/config/usb_gadget") && commandText.contains("functions") -> "none"
            commandText.contains("/dev/usb-ffs/adb") -> ""
            commandText.contains("/dev/mtp_usb") -> ""
            else -> null
        }

    private fun looksLikeUsbConnectionCheck(commandText: String): Boolean {

            USB_CONNECTION_TOKENS.any { commandText.contains(it) }

    private fun commandToText(command: Any?): String {
        return when (command) {
            is String -> command
            is Array<*> -> command.joinToString(" ") { it?.toString().orEmpty() }
            is Iterable<*> -> command.joinToString(" ") { it?.toString().orEmpty() }
            else -> command?.toString().orEmpty()
        }
    }

    private fun sanitizeUsbStateIntent(intent: Intent?): Boolean {
        if (intent?.action != USB_STATE_ACTION) return false
        USB_STATE_BOOLEAN_KEYS.forEach { intent.putExtra(it, false) }
        USB_STATE_STRING_KEYS.forEach { intent.putExtra(it, "none") }
        return true
    }

    private fun sanitizeBatteryIntent(intent: Intent?): Boolean {
        if (intent?.action != Intent.ACTION_BATTERY_CHANGED) return false
        intent.putExtra(BatteryManager.EXTRA_PLUGGED, 0)
        intent.putExtra(BatteryManager.EXTRA_STATUS, BatteryManager.BATTERY_STATUS_DISCHARGING)
        intent.putExtra("usb_online", 0)
        intent.putExtra("usb_present", 0)
        intent.putExtra("charger_type", "none")
        return true
    }

    private fun sanitizeExtrasBundle(action: String?, bundle: Bundle): Boolean {
        return when (action) {
            USB_STATE_ACTION -> {
                USB_STATE_BOOLEAN_KEYS.forEach { bundle.putBoolean(it, false) }
                USB_STATE_STRING_KEYS.forEach { bundle.putString(it, "none") }
                true
            }
            Intent.ACTION_BATTERY_CHANGED -> {
                bundle.putInt(BatteryManager.EXTRA_PLUGGED, 0)
                bundle.putInt(BatteryManager.EXTRA_STATUS, BatteryManager.BATTERY_STATUS_DISCHARGING)
                bundle.putInt("usb_online", 0)
                bundle.putInt("usb_present", 0)
                bundle.putString("charger_type", "none")
                true
            }
            else -> false
        }
    }

    private fun fakeIntentIntExtra(action: String?, key: String): Int? {
        if (action != Intent.ACTION_BATTERY_CHANGED) return null
        return when (key) {
            BatteryManager.EXTRA_PLUGGED -> 0
            BatteryManager.EXTRA_STATUS -> BatteryManager.BATTERY_STATUS_DISCHARGING
            "usb_online", "usb_present" -> 0
            else -> null
        }
    }

    private fun fakeIntentBooleanExtra(action: String?, key: String): Boolean? {
        if (action == USB_STATE_ACTION && USB_STATE_BOOLEAN_KEYS.contains(key)) return false
        if (action == Intent.ACTION_BATTERY_CHANGED && key.startsWith("usb", ignoreCase = true)) return false
        return null
    }

    private fun fakeIntentStringExtra(action: String?, key: String): String? {
        if (action == USB_STATE_ACTION && USB_STATE_STRING_KEYS.contains(key)) return "none"
        if (action == Intent.ACTION_BATTERY_CHANGED && key == "charger_type") return "none"
        return null
    }

    private fun isUsbDebugSetting(name: String): Boolean {
        return SETTINGS_KEYS.contains(name.lowercase(Locale.US))
    }

    private fun fakeSystemProperty(key: String): String? {
        return PROPERTY_VALUES[key.lowercase(Locale.US)]
    }

    private fun isSensitiveUsbPath(path: String): Boolean {
        val normalized = path.lowercase(Locale.US)
        if (normalized.contains("/sys/class/power_supply/") && normalized.contains("usb")) return true
        return USB_PATH_TOKENS.any { normalized.contains(it) }
    }

    private fun contentResolverClass(): Class<*> {
        return android.content.ContentResolver::class.java
    }

    private fun log(message: String) {
        val line = "[$TAG] $message"
        try {
            XposedBridge.log(line)
        } catch (_: Throwable) {
        }
        try {
            Log.i(TAG, message)
        } catch (_: Throwable) {
        }
        writeFileLog(message)
    }

    private fun logHookHit(source: String, key: String, fakeValue: String) {
        log("hook hit: $source, $key -> $fakeValue")
    }

    private fun initFileLogger(context: Context) {
        synchronized(LOG_LOCK) {
            if (diagnosticLogFile != null) return

            val dir = context.getExternalFilesDir(null) ?: context.filesDir
            diagnosticLogFile = File(dir, "UsbDetectionBypass.log")
            writeFileLogLocked("file logger initialized: ${diagnosticLogFile?.absolutePath}")
            pendingFileLogs.forEach { writeFileLogLocked(it) }
            pendingFileLogs.clear()
        }
    }

    private fun writeFileLog(message: String) {
        synchronized(LOG_LOCK) {
            if (diagnosticLogFile == null) {
                if (pendingFileLogs.size >= MAX_PENDING_FILE_LOGS) {
                    pendingFileLogs.removeAt(0)
                }
                pendingFileLogs.add(message)
                return
            }
            writeFileLogLocked(message)
        }
    }

    private fun writeFileLogLocked(message: String) {
        val file = diagnosticLogFile ?: return
        try {
            val timestamp = LOG_TIME_FORMAT.format(Date())
            FileWriter(file, true).use { writer ->
                writer.append(timestamp)
                    .append(' ')
                    .append(message)
                    .append('\n')
            }
        } catch (_: Throwable) {
        }
    }

##############################################################################
##
##  Gradle start up script for UN*X
##
##############################################################################

# Attempt to set APP_HOME
# Resolve links: $0 may be a link
PRG="$0"
# Need this for relative symlinks.
while [ -h "$PRG" ] ; do
    ls=`ls -ld "$PRG"`
    link=`expr "$ls" : '.*-> \(.*\)$'`
    if expr "$link" : '/.*' > /dev/null; then
        PRG="$link"
    else
        PRG=`dirname "$PRG"`"/$link"
    fi
done
SAVED="`pwd`"
cd "`dirname \"$PRG\"`/" >/dev/null
APP_HOME="`pwd -P`"
cd "$SAVED" >/dev/null

APP_NAME="Gradle"
APP_BASE_NAME=`basename "$0"`

# Add default JVM options here. You can also use JAVA_OPTS and GRADLE_OPTS to pass JVM options to this script.
DEFAULT_JVM_OPTS='"-Xmx64m" "-Xms64m"'

# Use the maximum available, or set MAX_FD != -1 to use that value.
MAX_FD="maximum"

warn () {
    echo "$*"
}

die () {
    echo
    echo "$*"
    echo
    exit 1
}

# OS specific support (must be 'true' or 'false').
cygwin=false
msys=false
darwin=false
nonstop=false
case "`uname`" in
  CYGWIN* )
    cygwin=true
    ;;
  Darwin* )
    darwin=true
    ;;
  MINGW* )
    msys=true
    ;;
  NONSTOP* )
    nonstop=true
    ;;
esac

CLASSPATH=$APP_HOME/gradle/wrapper/gradle-wrapper.jar


# Determine the Java command to use to start the JVM.
if [ -n "$JAVA_HOME" ] ; then
    if [ -x "$JAVA_HOME/jre/sh/java" ] ; then
        # IBM's JDK on AIX uses strange locations for the executables
        JAVACMD="$JAVA_HOME/jre/sh/java"
    else
        JAVACMD="$JAVA_HOME/bin/java"
    fi
    if [ ! -x "$JAVACMD" ] ; then
        die "ERROR: JAVA_HOME is set to an invalid directory: $JAVA_HOME

Please set the JAVA_HOME variable in your environment to match the
location of your Java installation."
    fi
else
    JAVACMD="java"
    which java >/dev/null 2>&1 || die "ERROR: JAVA_HOME is not set and no 'java' command could be found in your PATH.

Please set the JAVA_HOME variable in your environment to match the
location of your Java installation."
fi

# Increase the maximum file descriptors if we can.
if [ "$cygwin" = "false" -a "$darwin" = "false" -a "$nonstop" = "false" ] ; then
    MAX_FD_LIMIT=`ulimit -H -n`
    if [ $? -eq 0 ] ; then
        if [ "$MAX_FD" = "maximum" -o "$MAX_FD" = "max" ] ; then
            MAX_FD="$MAX_FD_LIMIT"
        fi
        ulimit -n $MAX_FD
        if [ $? -ne 0 ] ; then
            warn "Could not set maximum file descriptor limit: $MAX_FD"
        fi
    else
        warn "Could not query maximum file descriptor limit: $MAX_FD_LIMIT"
    fi
fi

# For Darwin, add options to specify how the application appears in the dock
if $darwin; then
    GRADLE_OPTS="$GRADLE_OPTS \"-Xdock:name=$APP_NAME\" \"-Xdock:icon=$APP_HOME/media/gradle.icns\""
fi

# For Cygwin or MSYS, switch paths to Windows format before running java
if [ "$cygwin" = "true" -o "$msys" = "true" ] ; then
    APP_HOME=`cygpath --path --mixed "$APP_HOME"`
    CLASSPATH=`cygpath --path --mixed "$CLASSPATH"`

    JAVACMD=`cygpath --unix "$JAVACMD"`

    # We build the pattern for arguments to be converted via cygpath
    ROOTDIRSRAW=`find -L / -maxdepth 1 -mindepth 1 -type d 2>/dev/null`
    SEP=""
    for dir in $ROOTDIRSRAW ; do
        ROOTDIRS="$ROOTDIRS$SEP$dir"
        SEP="|"
    done
    OURCYGPATTERN="(^($ROOTDIRS))"
    # Add a user-defined pattern to the cygpath arguments
    if [ "$GRADLE_CYGPATTERN" != "" ] ; then
        OURCYGPATTERN="$OURCYGPATTERN|($GRADLE_CYGPATTERN)"
    fi
    # Now convert the arguments - kludge to limit ourselves to /bin/sh
    i=0
    for arg in "$@" ; do
        CHECK=`echo "$arg"|egrep -c "$OURCYGPATTERN" -`
        CHECK2=`echo "$arg"|egrep -c "^-"`                                 ### Determine if an option

        if [ $CHECK -ne 0 ] && [ $CHECK2 -eq 0 ] ; then                    ### Added a condition
            eval `echo args$i`=`cygpath --path --ignore --mixed "$arg"`
        else
            eval `echo args$i`="\"$arg\""
        fi
        i=`expr $i + 1`
    done
    case $i in
        0) set -- ;;
        1) set -- "$args0" ;;
        2) set -- "$args0" "$args1" ;;
        3) set -- "$args0" "$args1" "$args2" ;;
        4) set -- "$args0" "$args1" "$args2" "$args3" ;;
        5) set -- "$args0" "$args1" "$args2" "$args3" "$args4" ;;
        6) set -- "$args0" "$args1" "$args2" "$args3" "$args4" "$args5" ;;
        7) set -- "$args0" "$args1" "$args2" "$args3" "$args4" "$args5" "$args6" ;;
        8) set -- "$args0" "$args1" "$args2" "$args3" "$args4" "$args5" "$args6" "$args7" ;;
        9) set -- "$args0" "$args1" "$args2" "$args3" "$args4" "$args5" "$args6" "$args7" "$args8" ;;
    esac
fi

# Escape application args
save () {
    for i do printf %s\\n "$i" | sed "s/'/'\\\\''/g;1s/^/'/;\$s/\$/' \\\\/" ; done
    echo " "
}
APP_ARGS=`save "$@"`

# Collect all arguments for the java command, following the shell quoting and substitution rules
eval set -- $DEFAULT_JVM_OPTS $JAVA_OPTS $GRADLE_OPTS "\"-Dorg.gradle.appname=$APP_BASE_NAME\"" -classpath "\"$CLASSPATH\"" org.gradle.wrapper.GradleWrapperMain "$APP_ARGS"

exec "$JAVACMD" "$@"
