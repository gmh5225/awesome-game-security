`README.md`:

```md
<img src="images/banner.png">

# Saturn Anti Cheat
## Features
### Kernel driver
- Flags threads not from inside images
- More to come
### Launcher
- Starts the game with -insecure.
- Handles the driver; loading and unloading.
- Clean saturn colored native gui.

<img src="images/gui.png">

## Install
### Requirements
- CMake 3.28 (`winget install -e --id Kitware.CMake`).
- Visual Studio Build Tools with the "Desktop development with C++" workload (`winget install -e --id Microsoft.VisualStudio.BuildTools`).
- Windows SDK 10.0.28000 (`winget install -e --id Microsoft.WindowsSDK.10.0.28000`).
- Windows SDK 10.0.26100, for `signtool` (`winget install -e --id Microsoft.WindowsSDK.10.0.26100`).
- Windows driver kit (`winget install Microsoft.WindowsWDK.10.0.28000`).
- just (optional, `winget install -e --id Casey.Just`).

### PC Setup
> [!CAUTION]
> Make sure to save your BitLocker recovery key for your Microsoft account before changing secure boot (`https://account.microsoft.com/devices/recoverykey`).
>
> If you do not have access to the key you may get prompted for it, If it's not provided you will not be able to access any of your encrypted files.
>
> You will not be able to play games that require test signing or secure boot (Fortnite, R6 Siege, Valorant, FaceIT, etc) until you re-enable secure boot and disable testsigning (below), it is also not reccomended due to how the driver can [look like an active kernel cheat](https://www.reddit.com/r/Battlefield/comments/1mlwcbl/battlefield_6_just_told_me_to_uninstall_valorant/).

> [!NOTE]
> While secure boot is disabled you are more vulnerable to malware that can start before the OS.
- Disable `Secure Boot` in your bios.
- Save and reboot.
- Open an admin powershell window and run `bcdedit /set testsigning on`.
- Restart your pc.

### Reverse Setup
- Enable `Secure Boot` in your bios.
- Save and reboot.
- Open an admin powershell window and run:
```bash
# Remove driver traces
sc.exe stop SaturnAC
sc.exe delete SaturnAC
# Disable test mode
# Turning on secure boot may also disable test mode
bcdedit /set testsigning off
```
- Restart your pc.
- Disable `Secure Boot` in your bios.
- Save and reboot.

### Clone the source code
```bash
git clone https://github.com/optccopa/SaturnAntiCheat
cd SaturnAntiCheat
```
> [!NOTE]
> Signing cert is created on first build as `SaturnAC` in `CurrentUser\My`
### Build with `just`:
```bash
just configure && just run
```

### Build with `CMake`:
```bash
cmake -B build -A x64
cmake --build build --config Release
& "./build/bin/Release/SaturnAntiCheatLauncher.exe"
```
## View Debug Console
- Install Dbgview
- Open Dbgview.exe
- Capture > Capture kernel
- Filter / Highlight (under Help) > Include: `saturn:*`

```

`src/driver/CmakeLists.txt`:

```txt
string(REPLACE "/RTC1" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")

cmake_host_system_information(RESULT WDK QUERY WINDOWS_REGISTRY
    "HKLM/SOFTWARE/Microsoft/Windows Kits/Installed Roots" VALUE KitsRoot10)
file(TO_CMAKE_PATH "${WDK}" WDK)
set(WDK_VERSION ${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION})

add_library(SaturnAntiCheatDriver SHARED main.c)

target_include_directories(SaturnAntiCheatDriver SYSTEM BEFORE PRIVATE
    "${WDK}/Include/${WDK_VERSION}/km"
    "${WDK}/Include/${WDK_VERSION}/shared"
)

target_compile_definitions(SaturnAntiCheatDriver PRIVATE 
    _AMD64_ UNICODE _UNICODE POOL_NX_OPTIN=1
)

target_compile_options(SaturnAntiCheatDriver PRIVATE 
    /W4 /WX
    /GR- /EHs-c-
    /kernel
)

target_link_directories(SaturnAntiCheatDriver PRIVATE 
    "${WDK}/Lib/${WDK_VERSION}/km/x64"
)

target_link_options(SaturnAntiCheatDriver PRIVATE
    /DRIVER
    /INTEGRITYCHECK
    /ENTRY:DriverEntry
    /SUBSYSTEM:NATIVE
    /NODEFAULTLIB
    /IGNORE:4099

    /MANIFEST:NO
    /INCREMENTAL:NO
)

target_link_libraries(SaturnAntiCheatDriver 
    ntoskrnl hal BufferOverflowFastFailK
)

set_target_properties(SaturnAntiCheatDriver PROPERTIES SUFFIX ".sys")

# Test-signs with a self-signed cert from CurrentUser\My, created on first configure
set(SIGN_CERT_SUBJECT "SaturnAC" CACHE STRING "Subject (CN) of the code signing cert")

execute_process(
    COMMAND powershell -NoProfile -Command
        "$ErrorActionPreference = 'Stop'; $env:PSModulePath = [Environment]::GetEnvironmentVariable('PSModulePath', 'Machine'); if (-not (Get-ChildItem Cert:\\CurrentUser\\My |Where-Object { $_.Subject -eq 'CN=${SIGN_CERT_SUBJECT}' -and $_.NotAfter -gt (Get-Date) })) { New-SelfSignedCertificate -Type CodeSigningCert -Subject 'CN=${SIGN_CERT_SUBJECT}' -CertStoreLocation Cert:\\CurrentUser\\My | Out-Null; 'created' }"
    OUTPUT_VARIABLE CERT_CREATED
    OUTPUT_STRIP_TRAILING_WHITESPACE
    COMMAND_ERROR_IS_FATAL ANY
)
if(CERT_CREATED)
    message(STATUS "Created code signing cert CN=${SIGN_CERT_SUBJECT} in CurrentUser\\My")
endif()

# signtool isn't in every SDK version's bin, so search all of them, newest first
file(GLOB SIGNTOOL_DIRS "${WDK}/bin/*/x64")
list(SORT SIGNTOOL_DIRS COMPARE NATURAL ORDER DESCENDING)

find_program(SIGNTOOL signtool.exe
    HINTS ${SIGNTOOL_DIRS}
    NO_DEFAULT_PATH
    REQUIRED
)

add_custom_command(TARGET SaturnAntiCheatDriver POST_BUILD
    COMMAND ${SIGNTOOL} sign /fd sha256 /a /s My /n ${SIGN_CERT_SUBJECT} "$<TARGET_FILE:SaturnAntiCheatDriver>"
    COMMENT "Signing driver..."
)

static VOID SetRoot(PUNICODE_STRING Root, PCUNICODE_STRING Path, PCUNICODE_STRING Rel)
{
    if (!RtlSuffixUnicodeString(Rel, Path, TRUE)) return;
    USHORT len = Path->Length - Rel->Length;
    PWCH buf = (PWCH)ExAllocatePool2(POOL_FLAG_NON_PAGED, len, 'tRaS');
    if (!buf) return;
    RtlCopyMemory(buf, Path->Buffer, len);
    Root->Buffer = buf;
    Root->Length = Root->MaximumLength = len;
}

static VOID FreeRoot(PUNICODE_STRING Root)
{
    if (Root->Buffer) ExFreePoolWithTag(Root->Buffer, 'tRaS');
    RtlZeroMemory(Root, sizeof(*Root));
}

static BOOLEAN MatchSet(PCUNICODE_STRING Path, PCUNICODE_STRING Root,
                        const UNICODE_STRING *Set, ULONG Count)
{
    USHORT rl = Root->Length;
    if (!rl || Path->Length <= rl) return FALSE;

    UNICODE_STRING head = { rl, rl, Path->Buffer };
    if (!RtlEqualUnicodeString(&head, Root, TRUE)) return FALSE;

    USHORT tl = (USHORT)(Path->Length - rl);
    UNICODE_STRING tail = { tl, tl, Path->Buffer + rl / sizeof(WCHAR) };
    for (ULONG i = 0; i < Count; i++)
        if (RtlEqualUnicodeString(&tail, &Set[i], TRUE)) return TRUE;
    return FALSE;
}

static BOOLEAN IsAllowlisted(PCUNICODE_STRING Path)
{
    return MatchSet(Path, &g_GameRoot,  g_GameRel,  ARRAYSIZE(g_GameRel))
        || MatchSet(Path, &g_Sys32Root, g_Sys32Rel, ARRAYSIZE(g_Sys32Rel))
        || MatchSet(Path, &g_SteamRoot, g_SteamRel, ARRAYSIZE(g_SteamRel));
}

VOID createProcessRoutine(
    IN PEPROCESS Process,
    IN HANDLE ProcessId,
    IN PPS_CREATE_NOTIFY_INFO CreateInfo) {
    UNREFERENCED_PARAMETER(Process);

    if (CreateInfo == NULL) { // exit
        if (ProcessId == g_Cs2Pid) {
            InterlockedExchangePointer((PVOID volatile*)&g_Cs2Pid, NULL);

            FreeRoot(&g_GameRoot);
            FreeRoot(&g_Sys32Root);
            FreeRoot(&g_SteamRoot);

            DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
                "saturn: cs2 stopped, pid %lu\n", (ULONG)(ULONG_PTR)ProcessId
            );
        }
        return;
    }

    if (CreateInfo->ImageFileName == NULL) {
        return;
    }

    if (!RtlSuffixUnicodeString(&cs2Suffix, (PUNICODE_STRING)CreateInfo->ImageFileName, TRUE)) {
        return;
    }

    InterlockedExchangePointer((PVOID volatile*)&g_Cs2Pid, ProcessId);

    PEPROCESS parent;
    if (NT_SUCCESS(PsLookupProcessByProcessId(CreateInfo->ParentProcessId, &parent))) {
        PUNICODE_STRING parentName;
        if (NT_SUCCESS(SeLocateProcessImageName(parent, &parentName))) {
            SetRoot(&g_SteamRoot, parentName, &kSteamRel);
            ExFreePool(parentName);
        }
        ObDereferenceObject(parent);
    }

    DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
        "saturn: cs2 launched, pid %lu\n", (ULONG)(ULONG_PTR)ProcessId
    );
}

// ISSUE: if the driver loads after cs2 starts
// the first legit remote thread gets logged
VOID createThreadRoutine(
    IN HANDLE ProcessId,
    IN HANDLE ThreadId,
    IN BOOLEAN Created
) {
    if (Created) { 
        if (ProcessId == g_Cs2Pid) {
            HANDLE currentProcessId = PsGetCurrentProcessId();
            if (currentProcessId != ProcessId) {
                PETHREAD thread;
                if (!NT_SUCCESS(PsLookupThreadByThreadId(ThreadId, &thread))) {
                    return;
                }

                HANDLE threadHandle;
                NTSTATUS status = ObOpenObjectByPointer(thread, OBJ_KERNEL_HANDLE, NULL,
                    THREAD_QUERY_INFORMATION, *PsThreadType, KernelMode, &threadHandle);

                if (NT_SUCCESS(status)) {
                    PVOID startAddress = NULL;

                    status = ZwQueryInformationThread(threadHandle, ThreadQuerySetWin32StartAddress,
                        &startAddress, sizeof(startAddress), NULL);

                    ZwClose(threadHandle);

                    if (NT_SUCCESS(status)) {
                        MEMORY_BASIC_INFORMATION mbi;
                        KAPC_STATE apcState;
                        KeStackAttachProcess(PsGetThreadProcess(thread), &apcState);

                        status = ZwQueryVirtualMemory(ZwCurrentProcess(), startAddress,
                            MemoryBasicInformation, &mbi, sizeof(mbi), NULL);

                        KeUnstackDetachProcess(&apcState);
                        if (NT_SUCCESS(status)) {
                            if (mbi.Type != MEM_IMAGE) {
                                DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
                                    "saturn: suspicous thread %lu from pid %lu starts outside an image at %p (type 0x%lx, protect 0x%lx)\n",
                                    (ULONG)(ULONG_PTR)ThreadId, (ULONG)(ULONG_PTR)currentProcessId,
                                    startAddress, mbi.Type, mbi.Protect);
                            }                        
                        }

                    }
                }
                ObDereferenceObject(thread);
            }
        }
    }
}

VOID loadImageRoutine(
    IN PUNICODE_STRING FullImageName,
    IN HANDLE ProcessId,
    IN PIMAGE_INFO ImageInfo
) {
    if (!ProcessId || !FullImageName || ProcessId != g_Cs2Pid) {
        return;
    }

    if (!g_GameRoot.Length) {
        SetRoot(&g_GameRoot, FullImageName, &kCs2Rel);
        return;
    }

    if (!g_Sys32Root.Length) {
        SetRoot(&g_Sys32Root, FullImageName, &kNtdllRel);
        return;
    }

    if (ImageInfo->ImageSignatureLevel > 8 || IsAllowlisted(FullImageName)) {
        return;
    }

    DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
        "saturn: unverified image loaded: %wZ at Address: %p\n", 
        FullImageName, ImageInfo->ImageBase
    );
}

NTSTATUS DriverCreateClose(PDEVICE_OBJECT DeviceObject, PIRP Irp) {
    UNREFERENCED_PARAMETER(DeviceObject);

    Irp->IoStatus.Status = STATUS_SUCCESS;
    Irp->IoStatus.Information = 0;
    IoCompleteRequest(Irp, IO_NO_INCREMENT);
    return STATUS_SUCCESS;
}

NTSTATUS DriverControl(PDEVICE_OBJECT DeviceObject, PIRP Irp) {
    UNREFERENCED_PARAMETER(DeviceObject);

    PIO_STACK_LOCATION stack = IoGetCurrentIrpStackLocation(Irp);
    NTSTATUS status = STATUS_SUCCESS;
    ULONG written = 0;

    switch (stack->Parameters.DeviceIoControl.IoControlCode) {
        case IOCTL_HEARTBEAT: {
            if (stack->Parameters.DeviceIoControl.OutputBufferLength < sizeof(ULONG)) {
                status = STATUS_BUFFER_TOO_SMALL;
                break;
            }

            if (!g_Cs2Pid) {
                *(PULONG)Irp->AssociatedIrp.SystemBuffer = HEARTBEAT_STATUS_CS2_NOT_FOUND;
                written = sizeof(ULONG);
                break;
            } else {
                *(PULONG)Irp->AssociatedIrp.SystemBuffer = HEARTBEAT_MAGIC;
                written = sizeof(ULONG);                
            }

            break;
        }
        
        default: {
            status = STATUS_INVALID_DEVICE_REQUEST;
        }
    }

    Irp->IoStatus.Status = status;
    Irp->IoStatus.Information = written;
    IoCompleteRequest(Irp, IO_NO_INCREMENT);

    return status;
}

VOID DriverUnload(IN PDRIVER_OBJECT DriverObject) {
    UNICODE_STRING symlink = RTL_CONSTANT_STRING(SATURN_SYMLINK_NAME);
    IoDeleteSymbolicLink(&symlink);
    IoDeleteDevice(DriverObject->DeviceObject);

    NTSTATUS threadStatus;
    threadStatus = PsRemoveCreateThreadNotifyRoutine(createThreadRoutine);
    if (!NT_SUCCESS(threadStatus)) {
        DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
            "saturn: failed to unset ThreadNotifyRoutine, err: %lu\n", threadStatus);
    }

    NTSTATUS processStatus;
    processStatus = PsSetCreateProcessNotifyRoutineEx(createProcessRoutine, TRUE);
    if (!NT_SUCCESS(processStatus)) {
        DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
            "saturn: failed to unset ProcessNotifyRoutineEx, err: %lu\n", processStatus);
    }

    NTSTATUS imageStatus;
    imageStatus = PsRemoveLoadImageNotifyRoutine(loadImageRoutine);
    if (!NT_SUCCESS(imageStatus)) {
        DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
            "saturn: failed to unset ImageNotifyRoutineEx, err: %ld\n", processStatus);
    }

    DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL, "saturn: unloaded\n");
}

NTSTATUS DriverEntry(
    IN PDRIVER_OBJECT  DriverObject,
    IN PUNICODE_STRING RegistryPath
) {
    UNREFERENCED_PARAMETER(RegistryPath);
    
    DriverObject->DriverUnload = DriverUnload;
    DriverObject->MajorFunction[IRP_MJ_CREATE] = DriverCreateClose;
    DriverObject->MajorFunction[IRP_MJ_CLOSE] = DriverCreateClose;
    DriverObject->MajorFunction[IRP_MJ_DEVICE_CONTROL] = DriverControl;

    UNICODE_STRING devName = RTL_CONSTANT_STRING(SATURN_DEVICE_NAME);
    UNICODE_STRING symlink = RTL_CONSTANT_STRING(SATURN_SYMLINK_NAME);
    PDEVICE_OBJECT device;

    NTSTATUS deviceStatus = IoCreateDevice(DriverObject, 0, &devName,
        0x8000, FILE_DEVICE_SECURE_OPEN, FALSE, &device);
    if (!NT_SUCCESS(deviceStatus)) {
        DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
            "saturn: failed to create device, err: %ld\n", deviceStatus);
        return deviceStatus;
    }

    deviceStatus = IoCreateSymbolicLink(&symlink, &devName);
    if (!NT_SUCCESS(deviceStatus)) {
        DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
            "saturn: failed to create symlink, err: %ld\n", deviceStatus);
        IoDeleteDevice(device);
        return deviceStatus;
    }

    NTSTATUS threadStatus;
    threadStatus = PsSetCreateThreadNotifyRoutine(createThreadRoutine);
    if (!NT_SUCCESS(threadStatus)) {
        DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
            "saturn: failed to set ThreadNotifyRoutine, err: %ld\n", threadStatus);
    }

    NTSTATUS processStatus;
    processStatus = PsSetCreateProcessNotifyRoutineEx(createProcessRoutine, FALSE);
    if (!NT_SUCCESS(threadStatus)) {
        DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
            "saturn: failed to set ProcessNotifyRoutineEx, err: %ld\n", processStatus);
    }

    NTSTATUS imageStatus;
    imageStatus = PsSetLoadImageNotifyRoutineEx(loadImageRoutine, 0);
    if (!NT_SUCCESS(imageStatus)) {
        DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL,
            "saturn: failed to set LoadImageNotifyRoutineEx, err: %ld\n", processStatus);
    }
    
    DbgPrintEx(DPFLTR_IHVDRIVER_ID, DPFLTR_ERROR_LEVEL, "saturn: loaded\n");

    return STATUS_SUCCESS;
}

DWORD startDriver(const wchar_t *sysPath) {
    SC_HANDLE scm = OpenSCManagerW(NULL, NULL, SC_MANAGER_ALL_ACCESS);
    if (!scm) {
        return GetLastError();
    }

    SC_HANDLE svc = OpenServiceW(scm, L"SaturnAC", SERVICE_ALL_ACCESS);
    if (!svc) {
        svc = CreateServiceW(scm, L"SaturnAC", L"SaturnAC",
            SERVICE_ALL_ACCESS, SERVICE_KERNEL_DRIVER,
            SERVICE_DEMAND_START, SERVICE_ERROR_NORMAL,
            sysPath, NULL, NULL, NULL, NULL, NULL
        );
    }

    DWORD err = 0;
    if (!svc || !StartServiceW(svc, 0, NULL)) {
        err = GetLastError();
    }

    if (err == ERROR_SERVICE_ALREADY_RUNNING) {
        err = 0;
    }

    if (svc) {
        CloseServiceHandle(svc);
    }

    CloseServiceHandle(scm);
    return err;
}

void stopDriver(void) {
    SC_HANDLE scm = OpenSCManagerW(NULL, NULL, SC_MANAGER_ALL_ACCESS);
    SC_HANDLE svc = OpenServiceW(scm, L"SaturnAC", SERVICE_ALL_ACCESS);
    SERVICE_STATUS status;
    ControlService(svc, SERVICE_CONTROL_STOP, &status);
    DeleteService(svc);
    CloseServiceHandle(svc);
    CloseServiceHandle(scm);
}

typedef enum  {
    HEARTBEAT_SUCCESS, HEARTBEAT_FAILED, HEARTBEAT_CS2_NOT_FOUND, HEARTBEAT_PENDING
} HEARTBEAT_STATUS;

HEARTBEAT_STATUS pingDriver(void) {
    HANDLE dev = CreateFileW(SATURN_USER_PATH, GENERIC_READ | GENERIC_WRITE,
        0, NULL, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, NULL);
    if (dev == INVALID_HANDLE_VALUE) {
        return HEARTBEAT_FAILED;
    }

    ULONG reply = 0;
    DWORD got = 0;
    BOOL ok = DeviceIoControl(dev, IOCTL_HEARTBEAT, NULL, 0,
        &reply, sizeof reply, &got, NULL);

    CloseHandle(dev);

    if (reply == HEARTBEAT_STATUS_CS2_NOT_FOUND) {
        return HEARTBEAT_CS2_NOT_FOUND;
    } else if (reply == HEARTBEAT_MAGIC && ok && got == sizeof reply) {
        return HEARTBEAT_SUCCESS;
    } else {
        return HEARTBEAT_FAILED;
    }
}

    if (err) {
        if (err == ERROR_INVALID_IMAGE_HASH) {
            updateStatus(hwnd, L"Test signing is not enabled\nPlease refer to README.md > Install > PC Setup");
        } else {
            updateStatus(hwnd, L"Driver failed to load (error %lu)", err);
        }
        
    } else {
        updateStatus(hwnd, L"Anticheat running ...");
        SetTimer(hwnd, HEARTBEAT_TIMER, 1000, NULL);

        INT_PTR r = (INT_PTR)ShellExecuteW(NULL, L"open",
            L"C:\\Program Files (x86)\\Steam\\steam.exe",
            L"-applaunch 730 -insecure", NULL, SW_SHOWNORMAL);
        if (r <= 32) {
            updateStatus(hwnd, L"Failed to launch game with -insecure, error: %i", (int)r);
        }
    }
    InvalidateRect(hwnd, NULL, TRUE);

    MSG msg;
    while (GetMessageW(&msg, NULL, 0, 0)) {
        TranslateMessage(&msg);
        DispatchMessageW(&msg);
    }

    stopDriver();

    return 0;
}
