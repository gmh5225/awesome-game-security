`README.md`:

```md
# Fission Plugin API

A Rust-first, headless plugin SDK and cross-tool compatibility runtime for [Fission](https://github.com/fission-systems/Fission).

**Status: experimental bootstrap (protocol v0).** The goal is to run existing IDA Pro, Ghidra, Binary Ninja and angr scripts with minimal modification, without requiring those products for supported operations. This is not full compatibility or binary ABI emulation.

## Architecture

```text
Existing scripts / native SDK clients
          |             |
 IDA / Ghidra / BN / angr profiles
          |
 Versioned JSON-lines protocol + capabilities
          |
 Rust host (fixture or Fission CLI)
          |
 Fission analysis engine
```

The host uses Fission's existing `--contract v1 list --json` for real function discovery; fixture mode additionally supports mapped byte reads. All other capabilities must be negotiated. No silent stubs, invented IR, or fabricated analysis results.

## Quick start

```sh
cargo build --workspace
PYTHONPATH=python python3 -m fission_plugin_api --fixture fixtures/minimal.json --profile ida examples/ida_functions.py
PYTHONPATH=python python3 -m fission_plugin_api --binary ./target.bin --profile ida examples/ida_functions.py
```

Set `FISSION_PLUGIN_HOST` to the built host executable if it is not at `target/debug/fission-plugin-host`. Set `FISSION_CLI` to the Fission CLI executable (default: `fission_cli`).

Run only trusted scripts: the worker is a separate process but **not a security sandbox**. No plugin code is downloaded or executed automatically.

## Current coverage

| Profile | Experimental implementation | Not implemented |
| --- | --- | --- |
| IDA Python | idautils.Functions, idc.get_func_name, ida_bytes.get_bytes (fixture only) | plugin lifecycle, Hex-Rays, Qt, native modules |
| Binary Ninja | BinaryView.functions, get_function_at, read (fixture only) | IL, analysis completion, UI, native modules |
| Ghidra | Python script globals: currentProgram, getFirstFunction, getFunctionAt, toAddr | JVM/Java APIs, transactions, analyzers, HighFunction |
| angr | Project(binary), kb.functions lookup | CFG analyses, CLE, SimState, SimProcedure, AIL |

These are small source-level compatibility subsets; real third-party plugin compatibility has not been measured. Unsupported operations raise errors rather than masquerading as successful.

//! Read-only reference host. Production Fission integration is a later milestone.
use fission_plugin_protocol::{address, parse_address, Error, Function, Outcome, Request, Response, MAX_READ, VERSION};
use serde::Deserialize;
use serde_json::{json, Value};
use std::{env, fs, io::{self, BufRead, Write}, process::Command};

#[derive(Deserialize)]
struct Fixture {
    functions: Vec<Function>,
    base: String,
    bytes: Vec<u8>,
}

enum Backend {
    Fixture(Fixture),
    Fission { cli: String, binary: String },
}

impl Backend {
    fn functions(&self) -> Result<Vec<Function>, Error> {
        match self {
            Self::Fixture(f) => Ok(f.functions.clone()),
            Self::Fission { cli, binary } => {
                let output = Command::new(cli).args(["--contract", "v1", "list", binary, "--json"])
                    .output().map_err(|e| Error::new("backend", e.to_string()))?;
                let envelope: Value = serde_json::from_slice(&output.stdout)
                    .map_err(|e| Error::new("backend", e.to_string()))?;
                if !output.status.success() || envelope["ok"] != true {
                    return Err(Error::new("backend", format!("Fission list failed: {}", envelope["error"])));
                }
                #[derive(Deserialize)]
                struct Row { addr: u64, name: String, size: u64 }
                #[derive(Deserialize)]
                struct Data { functions: Vec<Row>, analyzing: bool }
                let data: Data = serde_json::from_value(envelope["data"].clone())
                    .map_err(|e| Error::new("backend", e.to_string()))?;
                if data.analyzing {
                    return Err(Error::new("backend", "function discovery incomplete"));
                }
                Ok(data.functions.into_iter().map(|f| Function {
                    address: address(f.addr), name: f.name, size: f.size
                }).collect())
            }
        }
    }

    fn call(&self, req: &Request) -> Result<Value, Error> {
        if req.version != VERSION {
            return Err(Error::new("version_mismatch", "only protocol v0 is supported"));
        }
        match req.method.as_str() {
            "hello" => {
                let (name, capabilities) = match self {
                    Self::Fixture(_) => ("fixture", vec!["functions.list", "memory.read"]),
                    Self::Fission { .. } => ("fission-cli", vec!["functions.list"]),
                };
                Ok(json!({"backend": name, "capabilities": capabilities}))
            }
            "functions.list" => Ok(json!(self.functions()?)),
            "memory.read" => {
                let Self::Fixture(f) = self else {
                    return Err(Error::new("unsupported", "memory.read is not implemented by this backend"));
                };
                let invalid = || Error::new("invalid_params", "expected address and size <= MAX_READ");
                let addr = parse_address(req.params["address"].as_str().ok_or_else(invalid)?)?;
                let base = parse_address(&f.base)?;
                let size = req.params["size"].as_u64().filter(|n| *n <= MAX_READ as u64)
                    .ok_or_else(invalid)? as usize;
                let unmapped = || Error::new("unmapped", "range is not wholly mapped");
                let offset = addr.checked_sub(base).and_then(|n| usize::try_from(n).ok()).ok_or_else(unmapped)?;
                let end = offset.checked_add(size).ok_or_else(unmapped)?;
                let bytes = f.bytes.get(offset..end).ok_or_else(unmapped)?;
                Ok(json!(bytes))
            }
            _ => Err(Error::new("unsupported", format!("unknown method {}", req.method))),
        }
    }
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let args: Vec<String> = env::args().skip(1).collect();
    let backend = match args.as_slice() {
        [flag, path] if flag == "--fixture" => Backend::Fixture(serde_json::from_str(&fs::read_to_string(path)?)?),
        [flag, binary] if flag == "--binary" => Backend::Fission {
            cli: env::var("FISSION_CLI").unwrap_or_else(|_| "fission_cli".into()),
            binary: binary.clone(),
        },
        _ => return Err("usage: fission-plugin-host (--fixture FILE | --binary FILE)".into()),
    };
    let mut out = io::stdout().lock();
    for line in io::stdin().lock().lines() {
        let line = line?;
        let req: Request = serde_json::from_str(&line)?;
        let outcome = match backend.call(&req) {
            Ok(result) => Outcome::Ok { result },
            Err(error) => Outcome::Err { error },
        };
        serde_json::to_writer(&mut out, &Response { version: VERSION, id: req.id, outcome })?;
        out.write_all(b"\n")?;
        out.flush()?;
    }
    Ok(())
}

//! Versioned wire types. No dependency on Fission internals.
use serde::{Deserialize, Serialize};
use serde_json::Value;

pub const VERSION: u32 = 0;
pub const MAX_READ: usize = 1 << 20;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Request {
    pub version: u32,
    pub id: u64,
    pub method: String,
    #[serde(default)]
    pub params: Value,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Error {
    pub code: String,
    pub message: String,
}

impl Error {
    pub fn new(code: &str, message: impl Into<String>) -> Self {
        Self { code: code.into(), message: message.into() }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Response {
    pub version: u32,
    pub id: u64,
    #[serde(flatten)]
    pub outcome: Outcome,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(untagged)]
pub enum Outcome {
    Ok { result: Value },
    Err { error: Error },
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct Function {
    pub address: String,
    pub name: String,
    pub size: u64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Hello {
    pub capabilities: Vec<String>,
    pub backend: String,
}

pub fn address(n: u64) -> String { format!("0x{n:x}") }

pub fn parse_address(s: &str) -> Result<u64, Error> {
    let n = s.strip_prefix("0x").and_then(|s| u64::from_str_radix(s, 16).ok())
        .ok_or_else(|| Error::new("invalid_params", "expected hexadecimal u64 address"))?;
    if address(n) != s {
        return Err(Error::new("invalid_params", "address must be canonical lowercase hex"));
    }
    Ok(n)
}

#[cfg(test)]
mod tests {
    use super::*;
    #[test]
    fn address_roundtrip() {
        for n in [0, 0x1000, u64::MAX] {
            assert_eq!(parse_address(&address(n)).unwrap(), n);
        }
        for s in ["123", "0x01", "0xA", "0x10000000000000000"] {
            assert!(parse_address(s).is_err());
        }
    }
}

//! Synchronous native SDK over a caller-provided JSON-lines transport.
pub use fission_plugin_protocol::{Error, Function, Hello};
use fission_plugin_protocol::{Outcome, Request, Response, VERSION};
use serde_json::{json, Value};
use std::io::{BufRead, Write};

pub struct Client<R, W> {
    reader: R,
    writer: W,
    next_id: u64,
}

impl<R: BufRead, W: Write> Client<R, W> {
    pub fn new(reader: R, writer: W) -> Self {
        Self { reader, writer, next_id: 0 }
    }

    pub fn call(&mut self, method: &str, params: Value) -> Result<Value, Error> {
        let id = self.next_id;
        self.next_id = id.checked_add(1).ok_or_else(|| Error::new("transport", "id exhausted"))?;
        let req = Request { version: VERSION, id, method: method.into(), params };
        serde_json::to_writer(&mut self.writer, &req).map_err(transport)?;
        self.writer.write_all(b"\n").map_err(transport)?;
        self.writer.flush().map_err(transport)?;
        let mut line = String::new();
        self.reader.read_line(&mut line).map_err(transport)?;
        let resp: Response = serde_json::from_str(&line).map_err(transport)?;
        if resp.version != VERSION || resp.id != id {
            return Err(Error::new("transport", "response version or id mismatch"));
        }
        match resp.outcome {
            Outcome::Ok { result } => Ok(result),
            Outcome::Err { error } => Err(error),
        }
    }

    pub fn hello(&mut self) -> Result<Hello, Error> {
        serde_json::from_value(self.call("hello", json!({}))?).map_err(transport)
    }

    pub fn functions(&mut self) -> Result<Vec<Function>, Error> {
        serde_json::from_value(self.call("functions.list", json!({}))?).map_err(transport)
    }
}

fn transport(e: impl std::fmt::Display) -> Error { Error::new("transport", e.to_string()) }

#[cfg(test)]
mod tests {
    use super::*;
    use std::io::Cursor;
    #[test]
    fn refuses_mismatched_id() {
        let response = Cursor::new(b"{\"version\":0,\"id\":9,\"result\":[]}\n");
        let mut c = Client::new(response, Vec::new());
        assert_eq!(c.functions().unwrap_err().code, "transport");
    }
}

`docs/ARCHITECTURE.md`:

```md
# Architecture

## ADR-0001: compatibility above a stable semantic host boundary

Fission's internal PluginAPI currently exposes binary metadata, functions, memory, decompilation and disassembly as Rust traits. It is not a stable cross-language ABI. Keep the core authoritative; do not expose Rust trait object layout or pin consumers to internal IR structures.

- Protocol: typed values, version, request IDs, capability discovery, explicit failures.
- Native SDK: Rust transport and typed data; Python SDK and future Java binding share the wire contract.
- Host: fixture for reproducible conformance; CLI bridge for initial actual analysis; eventually an in-process Fission adapter with immutable snapshots and transactions.
- Profiles: distinct source-compatible namespaces and object models. Never map one tool's IR to another by pretending the semantics are identical.
- Worker: separate process for failure containment. OS-level sandboxing, resource limits, timeouts and permission enforcement are future work. Trusted scripts only for this bootstrap.

## Semantic boundaries

Use stable address-space-qualified handles in the future (v0 only supports a single flat 64-bit VA space). Distinguish file bytes, mapped bytes and runtime memory. Distinguish not-found from unsupported. Future mutations require a session revision, optimistic concurrency and atomic commit/rollback. Events require ordering and cancellation semantics.

For decompiler IR, represent provenance and loss explicitly. Hex-Rays microcode/CTree, Ghidra P-code/HighFunction, BN IL and angr AIL are different contracts; an adapter must document unsupported constructs. Do not synthesize a plausible but wrong analysis result.

## Integration

Current CLI bridge uses the existing Fission v1 envelope and FunctionsResponse. The native host should wrap fission-plugin contracts and fission-analysis-db rather than fork analysis implementations. Avoid a dependency cycle: shared protocol types are independent of the Fission engine.

## Security / licensing

Use independently authored shims against documented public interfaces. Do not redistribute proprietary SDK binaries, headers or plugin code without permission. Each plugin retains its license. Native binary plugins and GUI-dependent plugins are out of scope for v0. No product license bypass is intended.

| Target | Priority interfaces | Conformance criteria |
| --- | --- | --- |

| Ghidra | GhidraScript, FlatProgramAPI, Program, Address, FunctionManager | address spaces, iterators, transactions; separate JVM runner |
| Binary Ninja | BinaryView, Function, Symbol, Type, PluginCommand | object lifetime, IL mapping, analysis lifecycle |
| angr | Project, KnowledgeBase, CFG, SimStatePlugin, SimProcedure | CFG, state and solver semantics; do not claim unsupported analysis |

Public API references:
- https://python.docs.hex-rays.com/

`docs/PROTOCOL.md`:

```md
# Wire protocol v0 (experimental)

One UTF-8 JSON request per line on stdin, one response per line on stdout. Logs to stderr. Synchronous single-client transport; malformed JSON terminates the reference host. A production host must bound frame sizes, support cancellation and apply resource budgets.

Request:
```json
{"version":0,"id":0,"method":"hello","params":{}}
```

Response:
```json
{"version":0,"id":0,"result":{"backend":"fixture","capabilities":["functions.list","memory.read"]}}
```

Error:
```json
{"version":0,"id":1,"error":{"code":"unsupported","message":"unknown method example"}}
```

Exactly one result/error; version and ID echo the supported protocol and request ID. Addresses use canonical lowercase hexadecimal strings with no leading zeroes (except 0x0). Sizes are nonnegative integers. No floating-point address conversion.

| Method | Parameters | Result |
| --- | --- | --- |
| hello | {} | backend, capabilities |
| functions.list | {} | array of address/name/size records |
| memory.read | address, size (max 1 MiB) | byte array; wholly mapped range or error |

Error codes: version_mismatch, invalid_params, unmapped, unsupported, backend, transport (client-side). Clients must negotiate capability before use; hosts still enforce it. No arbitrary execution or mutation RPC.

"""Experimental Python client; no third-party dependencies."""
import json
import subprocess

class ProtocolError(RuntimeError):
    def __init__(self, code, message):
        super().__init__(f"{code}: {message}")
        self.code = code

class Client:
    def __init__(self, command):
        self.process = subprocess.Popen(command, stdin=subprocess.PIPE,
                                        stdout=subprocess.PIPE, text=True, encoding="utf-8")
        self.next_id = 0
        try:
            self.hello = self.call("hello")
            self.capabilities = frozenset(self.hello["capabilities"])
        except BaseException:
            self.close()
            raise

    def call(self, method, **params):
        ident = self.next_id
        self.next_id += 1
        self.process.stdin.write(json.dumps(dict(version=0, id=ident, method=method, params=params)) + "\n")
        self.process.stdin.flush()
        line = self.process.stdout.readline()
        if not line:
            raise ProtocolError("transport", "host closed its output")
        response = json.loads(line)
        if response.get("id") != ident or response.get("version") != 0:
            raise ProtocolError("transport", "response mismatch")
        if "error" in response:
            error = response["error"]
            raise ProtocolError(error["code"], error["message"])
        return response["result"]

    def require(self, capability):
        if capability not in self.capabilities:
            raise ProtocolError("unsupported", capability)

    def functions(self):
        self.require("functions.list")
        return self.call("functions.list")

    def read(self, address, size):
        self.require("memory.read")
        return bytes(self.call("memory.read", address=hex(address), size=size))

    def close(self):
        try:
            self.process.stdin.close()
            self.process.wait(timeout=2)
        except subprocess.TimeoutExpired:
            self.process.kill()
            self.process.wait()
        finally:
            self.process.stdout.close()

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()

```

import argparse
import os
import runpy
import sys
from pathlib import Path
from . import Client
from .compat import install

def main():
    parser = argparse.ArgumentParser(description="Run a trusted script with a Fission compatibility profile")
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--fixture")
    mode.add_argument("--binary")
    parser.add_argument("--profile", choices=["ida", "binaryninja", "ghidra", "angr"], required=True)
    parser.add_argument("script")
    parser.add_argument("script_args", nargs=argparse.REMAINDER)
    args = parser.parse_args()
    root = Path(__file__).resolve().parents[2]
    default = root / "target" / "debug" / ("fission-plugin-host.exe" if os.name == "nt" else "fission-plugin-host")
    host = os.environ.get("FISSION_PLUGIN_HOST", str(default))
    command = [host, "--fixture", args.fixture] if args.fixture else [host, "--binary", args.binary]
    with Client(command) as client:
        globals_ = install(args.profile, client, args.binary)
        sys.argv = [args.script, *args.script_args]
        sys.path.insert(0, str(Path(args.script).resolve().parent))
        runpy.run_path(args.script, init_globals=globals_, run_name="__main__")

if __name__ == "__main__":
    main()

```py
"""Small, explicitly scoped source-compatibility profiles."""
import sys
from types import ModuleType, SimpleNamespace
from . import ProtocolError

class Function:
    def __init__(self, row):
        self.start = self.addr = int(row["address"], 16)
        self.name = row["name"]
        self.size = row["size"]

    def getName(self):
        return self.name

    def getEntryPoint(self):
        return Address(self.start)

class Address:
    def __init__(self, value):
        self.value = value

    def getOffset(self):
        return self.value

    def __str__(self):
        return format(self.value, "x")

def install(profile, client, binary=None):
    def functions():
        return sorted((Function(row) for row in client.functions()), key=lambda f: f.start)

    def at(addr):
        return next((f for f in functions() if f.start == addr), None)

    def module(name, **attrs):
        if name in sys.modules:
            raise ProtocolError("profile_conflict", name)
        obj = ModuleType(name)
        obj.__dict__.update(attrs)
        sys.modules[name] = obj
        return obj

    if profile == "ida":
        def addresses(start=None, end=None):
            return (f.start for f in functions()
                    if (start is None or f.start >= start) and (end is None or f.start < end))

        def name(addr):
            f = next((f for f in functions() if f.start <= addr < f.start + max(f.size, 1)), None)
            return f.name if f else ""

        def get_bytes(addr, size):
            try:
                return client.read(addr, size)
            except ProtocolError as exc:
                if exc.code == "unmapped":
                    return None
                raise

        module("idautils", Functions=addresses)
        module("idc", get_func_name=name)
        module("ida_bytes", get_bytes=get_bytes)
        module("idaapi", BADADDR=(1 << 64) - 1)
        return {}

    if profile == "binaryninja":
        class BinaryView:
            @property
            def functions(self):
                return functions()

            def get_function_at(self, addr):
                return at(addr)

            def read(self, addr, size):
                return client.read(addr, size)

        module("binaryninja", BinaryView=BinaryView)
        return {"bv": BinaryView()}

    if profile == "ghidra":
        class Manager:
            def getFunctionAt(self, addr):
                return at(addr.getOffset())

        def first():
            return next(iter(functions()), None)

        manager = Manager()
        program = SimpleNamespace(getFunctionManager=lambda: manager)
        return {
            "currentProgram": program,
            "toAddr": lambda value: Address(int(value, 0) if isinstance(value, str) else int(value)),
            "getFirstFunction": first,
            "getFunctionAt": manager.getFunctionAt,
        }

    if profile == "angr":
        from pathlib import Path
        class Project:
            def __init__(self, filename, **kwargs):
                if binary is None or Path(filename).resolve() != Path(binary).resolve():
                    raise ProtocolError("unsupported", "Project must refer to the active binary")
                if kwargs:
                    raise ProtocolError("unsupported", "Project options are not implemented")
                self.kb = SimpleNamespace(functions={f.addr: f for f in functions()})

        module("angr", Project=Project)
        return {}

    raise ProtocolError("unsupported", profile)

```

`tests/test_conformance.py`:

```py
import os
from pathlib import Path
import subprocess
import sys
import unittest
from fission_plugin_api import Client, ProtocolError

ROOT = Path(__file__).resolve().parents[1]
HOST = os.environ.get("FISSION_PLUGIN_HOST", str(ROOT / "target/debug" / ("fission-plugin-host.exe" if os.name == "nt" else "fission-plugin-host")))
FIXTURE = str(ROOT / "fixtures/minimal.json")

class Conformance(unittest.TestCase):
    def test_host(self):
        with Client([HOST, "--fixture", FIXTURE]) as client:
            self.assertEqual(client.hello["backend"], "fixture")
            self.assertEqual([f["name"] for f in client.functions()], ["main", "helper"])
            self.assertEqual(client.read(0x401000, 2), bytes([144, 195]))
            for method, params, code in [
                ("memory.read", {"address": "0x401003", "size": 2}, "unmapped"),
                ("memory.read", {"address": "0x401000", "size": 1048577}, "invalid_params"),
                ("analysis.fake", {}, "unsupported"),
            ]:
                with self.assertRaises(ProtocolError) as cm:
                    client.call(method, **params)
                self.assertEqual(cm.exception.code, code)

    def test_scripts(self):
        for profile, script, expected in [
            ("ida", "ida_functions.py", "0x401000 main"),
            ("binaryninja", "bn_functions.py", "0x401002 helper"),
            ("ghidra", "ghidra_functions.py", "401000 main"),
        ]:
            with self.subTest(profile=profile):
                env = dict(os.environ, PYTHONPATH=str(ROOT / "python"), FISSION_PLUGIN_HOST=HOST)
                out = subprocess.check_output([sys.executable, "-m", "fission_plugin_api",
                    "--fixture", FIXTURE, "--profile", profile, str(ROOT / "examples" / script)],
                    env=env, text=True)
                self.assertIn(expected, out)

if __name__ == "__main__":
    unittest.main()
