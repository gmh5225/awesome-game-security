## 虚假战眼的原理
- 在 Steam 与 Epic 上（微软商店似乎不行，需要手动 Rockstar 启动器关闭）使用 `-noBattleEye` 参数启动 GTA5（禁用 BE 内核驱动 + 用户态 DLL）
- 验证BE状态：`is_session_started ≠ 0 && character_index ≠ 0`
- 如BE未被禁用 → 报错 `Pls Ensure Game Running And Disabled BattlEye!`

###AOB扫描 —— `sub_000E09A0`
- 扫描15个游戏内部地址（`ReadProcessMemory` + 模式匹配）
- 定位：`world`、`script_global`、`session`、BE标志等

### 打开游戏进程 —— `sub_0006D610`
- `OpenProcess(PROCESS_ALL_ACCESS, FALSE, pid)`
- 验证游戏仍在运行

### BE运行时补丁 —— `sub_000E1DC0`
- `WriteProcessMemory(hGame, BE_Function+5, patch_5bytes, 5)`
- BE 检测函数 → 始终返回 `"No anomalies"`
- `ReadProcessMemory`：读取 Stats / Globals / Locals 位置
- `WriteProcessMemory`：修改 Stats / Globals / Locals 位置
- D3D11：GUI 渲染
- 重启监控：检测 `battleye_restart_reason`
- 回流补丁：如果BE被恢复或者被发现 → 重新执行 `ReadProcessMemory` 与 `WriteProcessMemory`

static void init_critical_section() {

            if (g_CSInitialized) {
                DeleteCriticalSection(&g_CriticalSection);
                g_CSInitialized = false;
            }
        });
    }
}

    init_global_state,     
    init_critical_section,

    }
}

#include "game.h"

GameInterface::GameInterface()
    : m_hProcess(nullptr), m_pid(0), m_attached(false),
      m_gameBase(0), m_gameSize(0) {
    InitializeCriticalSection(&m_cs);
}

GameInterface::~GameInterface() {
    detach();
    DeleteCriticalSection(&m_cs);
}

bool GameInterface::attach() {

    if (m_attached) {
        detach();
    }

    m_pid = get_process_id_by_name(L"GTA5_Enhanced.exe");
    if (m_pid == 0) {
        MessageBoxA(nullptr, "GTA5_Enhanced.exe not found!", "Error", MB_OK | MB_ICONERROR);
        return false;
    }

    m_hProcess = OpenProcess(PROCESS_ALL_ACCESS, FALSE, m_pid);
    if (m_hProcess == nullptr) {
        MessageBoxA(nullptr, "OpenProcess failed! Run as admin.", "Error", MB_OK | MB_ICONERROR);
        return false;
    }

    m_gameBase = get_module_base(m_pid, L"GTA5_Enhanced.exe");
    if (m_gameBase == 0) {
        CloseHandle(m_hProcess);
        m_hProcess = nullptr;
        MessageBoxA(nullptr, "Module base not found!", "Error", MB_OK | MB_ICONERROR);
        return false;
    }

    auto modules = enum_process_modules(m_pid);
    for (const auto& mod : modules) {
        if (_wcsicmp(mod.first.c_str(), L"GTA5_Enhanced.exe") == 0) {
            m_gameBase = mod.second;

            MODULEINFO modInfo = {};
            HANDLE hDummy = OpenProcess(PROCESS_QUERY_INFORMATION | PROCESS_VM_READ,
                                        FALSE, m_pid);
            if (hDummy) {
                HMODULE hMod = (HMODULE)mod.second;
                GetModuleInformation(hDummy, hMod, &modInfo, sizeof(modInfo));
                m_gameSize = (uint64_t)modInfo.SizeOfImage;
                CloseHandle(hDummy);
            } else {
                m_gameSize = 0x40000000; 
            }
            break;
        }
    }

    if (!scan_signatures()) {
        detach();
        MessageBoxA(nullptr, "Signature scan failed!", "Error", MB_OK | MB_ICONERROR);
        return false;
    }

    m_attached = true;
    return true;
}

void GameInterface::detach() {
    if (m_hProcess != nullptr) {
        CloseHandle(m_hProcess);
        m_hProcess = nullptr;
    }
    m_pid = 0;
    m_attached = false;
    m_gameBase = 0;
    m_gameSize = 0;
    m_addresses = GameAddresses(); 
    m_stat_name_cache.clear();
}

bool GameInterface::scan_signatures() {
    if (m_hProcess == nullptr || m_gameBase == 0) {
        return false;
    }

    uint64_t worldAddr = pattern_scan(m_hProcess, m_gameBase, m_gameSize,
                                      SIG_WORLD, MASK_WORLD);
    if (worldAddr != 0) {

        int32_t relOffset = memory_read_value<int32_t>(m_hProcess, worldAddr + 3);
        m_addresses.world_ptr = memory_read_value<uint64_t>(
            m_hProcess, worldAddr + 7 + relOffset);
    }

    uint64_t sgAddr = pattern_scan(m_hProcess, m_gameBase, m_gameSize,
                                    SIG_SCRIPT_GLOBAL, MASK_SCRIPT_GLOBAL);
    if (sgAddr != 0) {
        int32_t relOffset = memory_read_value<int32_t>(m_hProcess, sgAddr + 3);
        m_addresses.script_global_ptr = sgAddr + 7 + relOffset;
    }

    uint64_t riAddr = pattern_scan(m_hProcess, m_gameBase, m_gameSize,
                                    SIG_REPLAY_INTERFACE, MASK_REPLAY_INTERFACE);
    if (riAddr != 0) {

        m_addresses.replay_interface_ptr = riAddr;
    }

    uint64_t issAddr = pattern_scan(m_hProcess, m_gameBase, m_gameSize,
                                     SIG_IS_SESSION_STARTED, MASK_IS_SESSION_STARTED);
    if (issAddr != 0) {
        int32_t relOffset = memory_read_value<int32_t>(m_hProcess, issAddr + 2);
        m_addresses.is_session_started = issAddr + 6 + relOffset;
    }

    uint64_t beAddr = pattern_scan(m_hProcess, m_gameBase, m_gameSize,
                                    SIG_BATTLEYE_BYPASS, MASK_BATTLEYE_BYPASS);
    if (beAddr != 0) {
        int32_t relOffset = memory_read_value<int32_t>(m_hProcess, beAddr + 2);
        m_addresses.battleye_bypass_flag = beAddr + 6 + relOffset;
    }

    if (m_addresses.world_ptr != 0 || m_addresses.script_global_ptr != 0) {
        m_addresses.initialized = true;
        return true;
    }

    return false;
}

bool memory_read(HANDLE hProcess, uint64_t address, void* buffer, size_t size) {
    if (address == 0 || buffer == nullptr || size == 0) {
        return false;
    }
    SIZE_T bytesRead = 0;
    BOOL result = ReadProcessMemory(hProcess, (LPCVOID)address,
                                    buffer, size, &bytesRead);
    return (result != FALSE && bytesRead == size);
}

bool memory_write(HANDLE hProcess, uint64_t address, const void* buffer, size_t size) {
    if (address == 0 || buffer == nullptr || size == 0) {
        return false;
    }

    DWORD oldProtect = 0;
    VirtualProtectEx(hProcess, (LPVOID)address, size,
                     PAGE_EXECUTE_READWRITE, &oldProtect);

    SIZE_T bytesWritten = 0;
    BOOL result = WriteProcessMemory(hProcess, (LPVOID)address,
                                     buffer, size, &bytesWritten);

    VirtualProtectEx(hProcess, (LPVOID)address, size,
                     oldProtect, &oldProtect);

    return (result != FALSE && bytesWritten == size);
}

std::string memory_read_string(HANDLE hProcess, uint64_t address, size_t maxLen) {
    if (address == 0) return "";
    std::vector<char> buf(maxLen + 1);
    SIZE_T bytesRead = 0;
    if (!ReadProcessMemory(hProcess, (LPCVOID)address, buf.data(),
                           maxLen, &bytesRead)) {
        return "";
    }
    buf[bytesRead] = '\0';

    size_t actualLen = strnlen(buf.data(), maxLen);
    return std::string(buf.data(), actualLen);
}

DWORD get_process_id_by_name(const wchar_t* processName) {

    HANDLE hSnapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (hSnapshot == INVALID_HANDLE_VALUE) {
        return 0;
    }

    PROCESSENTRY32W pe32 = { sizeof(PROCESSENTRY32W) };
    DWORD pid = 0;

    if (Process32FirstW(hSnapshot, &pe32)) {
        do {

            if (_wcsicmp(pe32.szExeFile, processName) == 0) {
                pid = pe32.th32ProcessID;
                break;
            }
        } while (Process32NextW(hSnapshot, &pe32));
    }

    CloseHandle(hSnapshot);
    return pid;
}

uint64_t get_module_base(DWORD pid, const wchar_t* moduleName) {
    HANDLE hSnapshot = CreateToolhelp32Snapshot(
        TH32CS_SNAPMODULE | TH32CS_SNAPMODULE32, pid);
    if (hSnapshot == INVALID_HANDLE_VALUE) {
        return 0;
    }

    MODULEENTRY32W me32 = { sizeof(MODULEENTRY32W) };
    uint64_t base = 0;

    if (Module32FirstW(hSnapshot, &me32)) {
        do {
            if (_wcsicmp(me32.szModule, moduleName) == 0) {
                base = (uint64_t)me32.modBaseAddr;
                break;
            }
        } while (Module32NextW(hSnapshot, &me32));
    }

    CloseHandle(hSnapshot);
    return base;
}

std::vector<ThreadInfo> enum_process_threads(DWORD pid) {
    std::vector<ThreadInfo> threads;

    HANDLE hSnapshot = CreateToolhelp32Snapshot(TH32CS_SNAPTHREAD, 0);
    if (hSnapshot == INVALID_HANDLE_VALUE) {
        return threads;
    }

    THREADENTRY32 te32 = { sizeof(THREADENTRY32) };

    if (Thread32First(hSnapshot, &te32)) {
        do {
            if (te32.th32OwnerProcessID == pid) {
                threads.push_back({te32.th32ThreadID, te32.th32OwnerProcessID});
            }
        } while (Thread32Next(hSnapshot, &te32));
    }

    CloseHandle(hSnapshot);
    return threads;
}

uint64_t pattern_scan(HANDLE hProcess, uint64_t start, uint64_t size,
                      const uint8_t* pattern, const char* mask) {
    if (size < 0x1000) return 0;

    size_t patternLen = strlen(mask);
    std::vector<uint8_t> buffer(0x1000); 

    for (uint64_t offset = 0; offset <= size - patternLen; offset += (0x1000 - patternLen)) {
        SIZE_T bytesRead = 0;
        uint64_t readAddr = start + offset;
        size_t readSize = (std::min)((uint64_t)0x1000, size - offset);

        if (!ReadProcessMemory(hProcess, (LPCVOID)readAddr,
                               buffer.data(), readSize, &bytesRead)) {
            continue;
        }

        for (size_t i = 0; i <= bytesRead - patternLen; i++) {
            bool found = true;
            for (size_t j = 0; j < patternLen; j++) {
                if (mask[j] == 'x' && buffer[i + j] != pattern[j]) {
                    found = false;
                    break;
                }

            }
            if (found) {
                return readAddr + i;
            }
        }
    }

    return 0;
}

uint64_t pattern_scan_str(HANDLE hProcess, uint64_t start, uint64_t size,
                          const char* patternStr) {

    std::vector<uint8_t> pattern;
    std::string mask;

    const char* p = patternStr;
    while (*p) {

        while (*p == ' ') p++;
        if (!*p) break;

        if (*p == '?') {
            pattern.push_back(0);
            mask.push_back('?');
            p++;
        } else {
            char hex[3] = { p[0], p[1], '\0' };
            pattern.push_back((uint8_t)strtoul(hex, nullptr, 16));
            mask.push_back('x');
            p += 2;
        }
    }

    return pattern_scan(hProcess, start, size, pattern.data(), mask.c_str());
}

uint64_t dereference_pointer_chain(HANDLE hProcess, uint64_t base,
                                   const std::vector<uint64_t>& offsets) {
    uint64_t addr = base;
    for (size_t i = 0; i < offsets.size(); i++) {
        addr = memory_read_value<uint64_t>(hProcess, addr);
        if (addr == 0) return 0;
        addr += offsets[i];
    }
    return addr;
}

bool OverlayRenderer::create_render_target() {
    if (m_renderer.device == nullptr || m_renderer.swapChain == nullptr) {
        return false;
    }

    if (m_renderer.renderTargetView != nullptr) {
        m_renderer.renderTargetView->Release();
        m_renderer.renderTargetView = nullptr;
    }

    ID3D11Texture2D* backBuffer = nullptr;
    HRESULT hr = m_renderer.swapChain->GetBuffer(
        0, __uuidof(ID3D11Texture2D), (void**)&backBuffer);
    if (FAILED(hr) || backBuffer == nullptr) {
        return false;
    }

    hr = m_renderer.device->CreateRenderTargetView(
        backBuffer, nullptr, &m_renderer.renderTargetView);
    backBuffer->Release();

    return SUCCEEDED(hr);
}

    viewport.Width = (float)m_renderer.windowWidth;
    viewport.Height = (float)m_renderer.windowHeight;
    viewport.MinDepth = 0.0f;
    viewport.MaxDepth = 1.0f;

    m_renderer.swapChain->Present(0, 0);
}

LRESULT CALLBACK OverlayRenderer::wnd_proc(HWND hwnd, UINT msg,
                                            WPARAM wParam, LPARAM lParam) {

    OverlayRenderer* self = nullptr;
    if (msg == WM_NCCREATE) {
        CREATESTRUCTW* cs = (CREATESTRUCTW*)lParam;
        self = (OverlayRenderer*)cs->lpCreateParams;
        SetWindowLongPtrW(hwnd, GWLP_USERDATA, (LONG_PTR)self);
    } else {
        self = (OverlayRenderer*)GetWindowLongPtrW(hwnd, GWLP_USERDATA);
    }

    switch (msg) {
    case WM_DESTROY:
        PostQuitMessage(0);
        return 0;

    case WM_SIZE:
        if (self != nullptr && self->m_renderer.swapChain != nullptr) {
            self->m_renderer.windowWidth = LOWORD(lParam);
            self->m_renderer.windowHeight = HIWORD(lParam);

            self->m_renderer.context->OMSetRenderTargets(0, nullptr, nullptr);
            if (self->m_renderer.renderTargetView) {
                self->m_renderer.renderTargetView->Release();
                self->m_renderer.renderTargetView = nullptr;
            }
            self->m_renderer.swapChain->ResizeBuffers(
                1, self->m_renderer.windowWidth, self->m_renderer.windowHeight,
                DXGI_FORMAT_R8G8B8A8_UNORM, 0);
            self->create_render_target();
        }
        return 0;

    case WM_KEYDOWN:
        if (self != nullptr && wParam == VK_INSERT) {
            self->toggle_menu();
        }
        return 0;
    }

    return DefWindowProcW(hwnd, msg, wParam, lParam);
}

void OverlayRenderer::run_message_loop() {
    MSG msg = {};
    while (true) {

        while (PeekMessageW(&msg, nullptr, 0, 0, PM_REMOVE)) {
            TranslateMessage(&msg);
            DispatchMessageW(&msg);

            if (msg.message == WM_QUIT) {
                return;
            }
        }

        update_overlay_position();

        render_frame();
    }
}

void OverlayRenderer::update_overlay_position() {
    if (m_renderer.targetHwnd == nullptr || m_renderer.hwnd == nullptr) {
        return;
    }

    RECT targetRect;
    if (GetClientRect(m_renderer.targetHwnd, &targetRect)) {
        POINT pt = { targetRect.left, targetRect.top };
        ClientToScreen(m_renderer.targetHwnd, &pt);

        int width = targetRect.right - targetRect.left;
        int height = targetRect.bottom - targetRect.top;

        if (width != m_renderer.windowWidth || height != m_renderer.windowHeight) {
            SetWindowPos(m_renderer.hwnd, nullptr,
                         pt.x, pt.y, width, height,
                         SWP_NOZORDER);
            m_renderer.windowWidth = width;
            m_renderer.windowHeight = height;
        } else {
            SetWindowPos(m_renderer.hwnd, nullptr,
                         pt.x, pt.y, 0, 0,
                         SWP_NOSIZE | SWP_NOZORDER);
        }
    }
}

void OverlayRenderer::cleanup_d3d() {

    if (m_renderer.renderTargetView) {
        m_renderer.renderTargetView->Release();
        m_renderer.renderTargetView = nullptr;
    }
    if (m_renderer.swapChain) {
        m_renderer.swapChain->Release();
        m_renderer.swapChain = nullptr;
    }
    if (m_renderer.context) {
        m_renderer.context->Release();
        m_renderer.context = nullptr;
    }
    if (m_renderer.device) {
        m_renderer.device->Release();
        m_renderer.device = nullptr;
    }
    m_renderer.initialized = false;
}
