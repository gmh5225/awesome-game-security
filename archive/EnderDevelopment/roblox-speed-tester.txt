```md
# SpeedTester

SpeedTester is a Roblox script designed to test movement anti-cheat systems in published games. It provides a GUI for controlling movement speed and observing server rollback behavior, helping developers identify weaknesses in their anti-cheat mechanisms.

## Install

1. Download the SpeedTester script from the GitHub repository.
2. Open your Roblox game in Roblox Studio.
3. Create a new LocalScript in StarterPlayerScripts.
4. Copy and paste the contents of the downloaded script into the LocalScript.

## Use

1. Run your game in Roblox Studio or on a live server.
2. Open the GUI by pressing a key (e.g., F5).
3. Use the speed slider to set the desired speed.
4. Click the "Go" button to start moving at the set speed.
5. Observe the status and rollback indicators to test the anti-cheat system.
```

```lua
-- Movement Logic

local speed = 16

local function updateSpeed()
    speed = tonumber(speedSlider.Text) or 16
    speedLabel.Text = "Speed: " .. speed
end

local function startMovement()
    isMoving = true
    statusLabel.Text = "Status: Moving"
    while isMoving do
        local deltaTime = RunService.Heartbeat:Wait()
        local moveDirection = Vector3.new(0, 0, 0)
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then
            moveDirection = moveDirection + Vector3.new(0, 0, -1)
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then
            moveDirection = moveDirection + Vector3.new(-1, 0, 0)
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then
            moveDirection = moveDirection + Vector3.new(0, 0, 1)
        end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then
            moveDirection = moveDirection + Vector3.new(1, 0, 0)
        end
        if moveDirection.Magnitude > 0 then
            moveDirection = moveDirection.Unit
        end
        character:TranslateBy(moveDirection * speed * deltaTime)
    end
end

local function stopMovement()
    isMoving = false
    statusLabel.Text = "Status: Idle"
end

-- Event Connections
goButton.MouseButton1Click:Connect(function()
    if isMoving then
        stopMovement()
    else
        startMovement()
    end
end)

speedSlider.FocusLost:Connect(updateSpeed)

-- Initial Setup
updateSpeed()
```

```md
# Roblox Movement Anti-Cheat Tester

This project contains a Roblox Lua script with a GUI for testing movement anti-cheat in a published game. The GUI includes speed control, status indicators, and rollback observation. The script tests unauthorized movement without triggering server rollback.

## File Structure

- `src/main.lua`: The main script with the GUI and anti-cheat testing functionality.
- `src/README.md`: This file, providing instructions on how to use the script.

## How to Use

1. Place the `main.lua` script in the `StarterPlayerScripts` folder of your Roblox game.
2. Run the game and open the GUI to test movement anti-cheat.
3. Use the speed control to adjust the movement speed and observe the status indicators.
4. Monitor the rollback observation to ensure unauthorized movement does not trigger server rollback.
```

```lua
-- Roblox Movement Anti-Cheat Tester

local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")

local player = Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local humanoid = character:WaitForChild("Humanoid")

-- GUI Setup
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "MovementAntiCheatTester"
screenGui.Parent = player:WaitForChild("PlayerGui")

local frame = Instance.new("Frame")
frame.Size = UDim2.new(0, 300, 0, 200)
frame.Position = UDim2.new(0.5, -150, 0.5, -100)
frame.AnchorPoint = Vector2.new(0.5, 0.5)
frame.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
frame.BorderSizePixel = 0
frame.Parent = screenGui

-- Draggable Frame
local dragging
local dragInput
local dragStart
local startPos

local function update(input)
    local delta = input.Position - dragStart
    frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

frame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = frame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

frame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        update(input)
    end
end)

-- Speed Control
local speedLabel = Instance.new("TextLabel")
speedLabel.Size = UDim2.new(0, 100, 0, 30)
speedLabel.Position = UDim2.new(0, 10, 0, 10)
speedLabel.Text = "Speed: " .. humanoid.WalkSpeed
speedLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
speedLabel.BackgroundTransparency = 1
speedLabel.Parent = frame

local speedSlider = Instance.new("TextBox")
speedSlider.Size = UDim2.new(0, 100, 0, 30)
speedSlider.Position = UDim2.new(0, 120, 0, 10)
speedSlider.Text = tostring(humanoid.WalkSpeed)
speedSlider.TextColor3 = Color3.fromRGB(255, 255, 255)
speedSlider.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
speedSlider.Parent = frame

speedSlider.FocusLost:Connect(function()
    local newSpeed = tonumber(speedSlider.Text)
    if newSpeed then
        humanoid.WalkSpeed = newSpeed
        speedLabel.Text = "Speed: " .. newSpeed
    end
end)

-- Status Indicators
local statusLabel = Instance.new("TextLabel")
statusLabel.Size = UDim2.new(0, 200, 0, 30)
statusLabel.Position = UDim2.new(0, 50, 0, 50)
statusLabel.Text = "Status: Idle"
statusLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
statusLabel.BackgroundTransparency = 1
statusLabel.Parent = frame

-- Rollback Observation
local rollbackLabel = Instance.new("TextLabel")
rollbackLabel.Size = UDim2.new(0, 200, 0, 30)
rollbackLabel.Position = UDim2.new(0, 50, 0, 90)
rollbackLabel.Text = "Rollback: None"
rollbackLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
rollbackLabel.BackgroundTransparency = 1
rollbackLabel.Parent = frame

-- Test Unauthorized Movement
local function testUnauthorizedMovement()
    local originalSpeed = humanoid.WalkSpeed
    humanoid.WalkSpeed = originalSpeed * 2
    statusLabel.Text = "Status: Testing"
    
    wait(2)
    
    humanoid.WalkSpeed = originalSpeed
    statusLabel.Text = "Status: Idle"
    rollbackLabel.Text = "Rollback: None"
end

-- Go Button
local goButton = Instance.new("TextButton")
goButton.Size = UDim2.new(0, 100, 0, 30)
goButton.Position = UDim2.new(0, 100, 0, 130)
goButton.Text = "Go"
goButton.TextColor3 = Color3.fromRGB(255, 255, 255)
goButton.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
goButton.Parent = frame

goButton.MouseButton1Click:Connect(testUnauthorizedMovement)

-- Character Added/Removed Handling
player.CharacterAdded:Connect(function(newCharacter)
    character = newCharacter
    humanoid = character:WaitForChild("Humanoid")
    speedLabel.Text = "Speed: " .. humanoid.WalkSpeed
    speedSlider.Text = tostring(humanoid.WalkSpeed)
end)

player.CharacterRemoving:Connect(function()
    character = nil
    humanoid = nil
end)
```
