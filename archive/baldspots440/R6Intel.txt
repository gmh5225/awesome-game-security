```sql
-- R6Intel v4.4 durable persistence
create extension if not exists pgcrypto;
create table if not exists players (id uuid primary key default gen_random_uuid(),profile_id text unique not null,platform text not null,current_username text not null,first_seen timestamptz default now(),last_seen timestamptz default now(),ever_sanctioned boolean default false,first_sanction_seen timestamptz,latest_response jsonb,last_refreshed_at timestamptz);
create index if not exists players_username_idx on players(lower(current_username));
create index if not exists players_platform_idx on players(platform);
create table if not exists aliases (id bigint generated always as identity primary key,player_id uuid references players(id) on delete cascade,name text not null,first_seen timestamptz default now(),last_seen timestamptz default now());
create index if not exists aliases_player_idx on aliases(player_id);
create index if not exists aliases_name_idx on aliases(lower(name));
create table if not exists snapshots (id bigint generated always as identity primary key,player_id uuid references players(id) on delete cascade,scanned_at timestamptz default now(),kd numeric,hs numeric,wr numeric,kpm numeric,matches integer,wins integer,losses integer,kills integer,deaths integer,rank text,rp integer,peak_rp integer,overall_kd numeric,overall_hs numeric,overall_kpr numeric,banned boolean,statistical_score integer,final_score integer,model_version text);
create index if not exists snapshots_player_time_idx on snapshots(player_id,scanned_at desc);
alter table players add column if not exists latest_response jsonb;
alter table players add column if not exists last_refreshed_at timestamptz;
alter table snapshots add column if not exists overall_kd numeric;
```

`V4-CHANGES.md`:

```md
# R6Intel v4 Public Beta

Implemented in this build:
- Ubisoft / Xbox / PSN selector and platform-aware backend requests.
- Refined score bands and natural labels (Moderately Suspicious, Highly Suspicious, Extremely Suspicious).
- Current Performance: K/D 40%, HSR 30%, KPM 20%, Win Rate 10%.
- K/D thresholds: 1.70 / 1.90 / 2.10 / 2.40.
- HSR thresholds: 60 / 70 / 80 / 90.
- KPM thresholds: 6 / 7 / 8 / 9.
- Win-rate thresholds: 60 / 70 / 80 / 90.
- Compound anomaly bonuses at 10+ matches: +5 / +10 / +15 / +20 / +25 / +30 / +35, non-stacking.
- Perfect-stat anomaly: eligible 100% percentage statistic at 10+ matches = +25, non-stacking.
- Current Performance / Historical Anomalies combine 50/50 when both are available.
- Weighted last-five-snapshot historical baseline, including K/D, HSR, KPM and WR changes.
- RP velocity calculated from the latest stored snapshot.
- Confirmed current sanction = +50 final-score modifier, capped at 100; statistical score retained separately.
- Previously confirmed sanction is permanently retained in local history.
- Confidence: 1–9 Low, 10–24 Medium, 25–49 High, 50+ Very High; 0 = Insufficient.
- Meaningful-change-only snapshots, retained without the old 50-snapshot cap.
- Alias first/last-seen tracking keyed by platform + profile UUID when available.
- Permanent-profile API endpoint by profile ID and browser URL update after search.
- 24-hour profile-view deduplication using a one-way daily salted IP hash; raw IP is not written by this feature.
- R6Data retry delays 1s / 3s / 10s plus 12s request timeout and cached-profile fallback.
- FAQ, Terms, Privacy and Public Beta messaging.
- Model version stored with snapshots (v4.0-beta.1).
- SUPABASE_SCHEMA.sql included for the planned production database migration.

Important limitation:
- The current R6Data normalization still has no reliable HSR source, so HSR remains `null` / `—` until the provider exposes a usable field.
- Supabase credentials were not supplied, so this ZIP uses the existing local JSON persistence at runtime. The production schema is included, but the live Supabase adapter cannot be safely configured without the project URL/key.

```

```md
# R6Intel v4.2

- Adds R6Data `operatorStats` request in `ranked` mode with no seasonYear, which returns `seasonYear: all`.
- Aggregates all-ranked operator kills, deaths, headshots, rounds, assists, first bloods/deaths, and time.
- Calculates Overall Ranked K/D, HSR, Kills/Round, Round Win Rate, and other baseline metrics.
- Does NOT sum operator `matchesPlayed`, because a single match can contain multiple operators and would be double-counted.
- Keeps latest-season rank/RP/wins/losses/matches/KPM separate from all-ranked baseline data.
- Adds current-season K/D vs all-ranked K/D as a historical anomaly signal.
- UI clearly separates All-Ranked Baseline from Most Recent Ranked Season.
```

`V4.4.2-CHANGES.md`:

```md
# R6Intel v4.4.2 Search Pipeline Fix

Root cause addressed:
- v4.4 held global R6Data concurrency slots while requests were sleeping/retrying.
- With five endpoints per player, this could make the UI appear frozen and delay later calls before they ever reached R6Data.

Changes:
- Default R6 concurrency raised to 5 so one player can fetch its five required endpoints together.
- Queue slots now cover only the actual HTTP request, not retry/backoff sleep.
- Every R6Data endpoint logs START, HTTP status, timeout and duration in Render.
- Individual HTTP attempts time out after 8 seconds.
- Maximum two HTTP attempts per endpoint.
- Full five-endpoint player lookup has a 20-second hard deadline.
- Supabase username-cache lookup is capped at 1.5 seconds.
- Existing endpoint cache and in-flight deduplication remain enabled.

```

`V4.4.3-CHANGES.md`:

```md
# R6Intel v4.4.3 Rate-Limit Fix

Observed in Render:
- all five R6Data endpoints were launched together,
- all five returned HTTP 429,
- the search then failed because no essential endpoint succeeded.

Changes:
- R6Data requests are globally paced instead of burst-fired.
- Default concurrency is 1.
- Default minimum gap between R6Data request starts is 750 ms.
- 429 retry uses Retry-After when available; otherwise waits ~2.5 seconds.
- Full player lookup deadline increased to 30 seconds to allow paced requests.
- Existing endpoint caching and in-flight deduplication remain enabled.

Optional Render tuning:
R6_CONCURRENCY=1
R6_MIN_INTERVAL_MS=750

```

```md
# R6Intel v4.5 Model + History

- Suspicion Score is now statistical only. Confirmed sanctions remain a separate status/signal and no longer add +50 to the displayed model score.
- Uses season HSR when the provider exposes it; otherwise uses All-Ranked HSR in performance scoring instead of treating HSR as missing.
- Adds sample-size reliability scaling:
  <5 matches 25%, 5-9 45%, 10-19 70%, 20-29 90%, 30+ 100%.
- Historical snapshot comparison begins after one prior snapshot rather than requiring two.
- Historical HSR comparison falls back to Overall HSR when season-specific HSR is unavailable.
- Saves richer snapshots: season ID, overall kills/deaths/headshots/rounds, performance score, historical score, confidence, model version.
- Keeps the proven v4.4.3 R6Data pacing/caching path unchanged.
```

```js
function showLegal(route,push=true){
  const map={faq:FAQ,terms:TERMS,privacy:PRIVACY};
  const key=route.replace(/^\/+/,'').split('/')[0];
  if(!map[key])return showHome(push);
  
  $('#heroEmpty').classList.add('hidden');
  $('#playerShell').classList.add('hidden');
  $('#errorBox').classList.add('hidden');
  $('#legalPage').innerHTML=`<a class="back-link" href="/">← Back to R6Intel</a>${map[key]}`;
  $('#legalPage').classList.remove('hidden');
  window.scrollTo({top:0,behavior:'smooth'});
}
function bindRoutes(){
  // Use normal href navigation. Express serves index.html for /faq, /terms and /privacy.
  // This avoids click-handler/history collisions and makes links work with direct URLs.
}
function routeFromLocation(){
  const path=(location.pathname.replace(/\/+$/,'') || '/');
  if(['/faq','/terms','/privacy'].includes(path)){
    showLegal(path,false);
  } else if(path==='/'){
    $('#legalPage').classList.add('hidden');
  }
}
```

```js
function combine(perf,disc,n){
  const cats=[];
  if(perf?.score!=null)cats.push({name:'Performance',score:perf.score,weight:.50});
  if(disc?.score!=null)cats.push({name:'Historical Anomalies',score:disc.score,weight:.50});
  const tw=cats.reduce((a,x)=>a+x.weight,0);
  let statisticalScore=tw?Math.round(cats.reduce((a,x)=>a+x.score*(x.weight/tw),0)):null;
  const perfectStatBonus=Number.isFinite(n.matches)&&n.matches>=10&&[n.wr,n.hs].some(v=>Number.isFinite(v)&&Math.abs(v-100)<1e-9)?25:0;
  if(statisticalScore!=null)statisticalScore=Math.min(100,statisticalScore+perfectStatBonus);
  const sanctionModifier=n.banned===true?50:0;
  // v4.5: suspicionScore is statistical only. Sanctions are displayed separately and never inflate the model score.
  const score=statisticalScore;
  return{
    score,statisticalScore,perfectStatBonus,sanctionModifier,
    sanctionStatus:n.banned===true?'confirmed':(n.banned===false?'none':'unknown'),
    categories:tw?cats.map(x=>({...x,effectiveWeight:x.weight/tw,contribution:x.score*(x.weight/tw)})):[]
  };
}
```
