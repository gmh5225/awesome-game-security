import numpy as np
import pandas as pd

rng = np.random.default_rng(42)

n_players = 100_000

players = pd.DataFrame({
    "player_id": np.arange(n_players),
    "skill_proxy": rng.uniform(0, 1, n_players),
    "tenure_days": rng.integers(1, 1500, n_players),
    "region": rng.choice(
        ["NA", "EU", "KR", "BR"],
        size=n_players
    )
})

# Hidden synthetic ground truth
players["true_cheating"] = rng.binomial(
    1, 0.025, n_players
)

# Existing detector is more likely to flag
# cheating players, but misses some.
flag_probability = np.where(
    players["true_cheating"] == 1,
    0.70,
    0.035
)

players["old_model_flag"] = (
    rng.random(n_players) < flag_probability
).astype(int)

# Only flagged players receive investigation labels
players["reviewed"] = players["old_model_flag"]

players["observed_label"] = np.where(
    players["reviewed"] == 1,
    players["true_cheating"],
    np.nan
)

print("Total players:", len(players))
print("True cheating prevalence:",
      round(players["true_cheating"].mean(), 4))
print("Flagged players:",
      players["old_model_flag"].sum())
print("Reviewed players:",
      players["reviewed"].sum())
print("Unlabeled players:",
      players["observed_label"].isna().sum())

print("\nObserved label counts:")
print(players["observed_label"].value_counts(dropna=False))

# Counter-Strike 2 Anti-Cheat ML Research

**Independent applied machine learning research | Synthetic and illustrative data | Not a deployed anti-cheat system**

This repository explores how to design, evaluate, and operate anti-cheat detection in a multiplayer game setting. It examines individual and coordinated behavior, label-selection bias, concept drift, skill-segment fairness, human-review capacity, and failure handling. Counter-Strike 2 provides the *gameplay framing*; this is **not Valve telemetry, VAC integration, or a production service**.

## The player problem

An anti-cheat detector must surface suspicious behavior without treating exceptional legitimate skill as cheating. False positives can damage player trust, and a high offline AUC does not automatically produce a safe enforcement system. The proposed decision process therefore separates **risk scoring**, **investigative review**, and **any enforcement action**. Model scores are investigative signals, not proof.

## Proposed system architecture

See the [architecture diagram and operational decision path](docs/ARCHITECTURE.md).

At a high level: game-event telemetry → validated feature windows → candidate detectors → calibrated risk scoring → review queue → human adjudication and delayed labels → monitoring and controlled retraining. Missing telemetry or model failure should **abstain or fall back safely**, not trigger automatic punishment.

Several read files from absolute paths such as `/srv/synthetic_anticheat_telemetry.csv` or `/srv/Counter Strike cheating.csv`; those source files are not included in this repository. Other cells generate their own synthetic examples. Open a notebook in Jupyter or Colab, inspect its dependencies and required inputs, and run applicable cells in order. Do not assume every notebook will run from a clean checkout.

Typical Python packages used across the notebooks include `pandas`, `numpy`, `scikit-learn`, `matplotlib`, and `networkx`. No pinned environment or CI verification is currently provided.

## Responsible use and limitations

- All modeled labels and gameplay scenarios are synthetic, illustrative, or locally supplied; none should be interpreted as verified player cheating.
- Synthetic data can encode unrealistic separability, label bias, and temporal or match-identifier inconsistencies; high model scores may not transfer to real gameplay.
- Some analyses use assumed counts or hand-entered outcomes to illustrate trade-offs. These are identified above.
- A prototype score must **not** automatically ban players. Review capacity, appeal processes, evidence quality, segment fairness and privacy controls are central to any real deployment.
- Proposed streaming, serving, latency SLOs, canary rollout, rollback and production monitoring are architectural considerations, not validated deployments.

# Proposed Anti-Cheat ML Architecture

> **Design proposal, not a deployed system.** The notebooks simulate selected modeling and operational decisions; they do not implement the full diagram.

```mermaid
flowchart TD
    A[Gameplay events / match telemetry] --> B[Schema validation and privacy controls]
    B --> C[Event windows and feature engineering]
    C --> D[Behavioral classifier]
    C --> E[Anomaly detector]
    C --> F[Player-interaction graph features]
    D --> G[Score calibration and policy guardrails]
    E --> G
    F --> G
    G --> H{Sufficient telemetry and healthy models?}
    H -->|Yes| I[Rank investigative candidates]
    H -->|No| J[Abstain / safe fallback and log reason]
    I --> K[Capacity-limited human review]
    K --> L[Adjudication and delayed labels]
    L --> M[Offline evaluation, drift and fairness monitoring]
    J --> M
    M --> N[Retraining and controlled rollout proposal]
    N -.-> D
```

## Proposed data and inference path

1. **Ingest and validate:** match and player event streams require event-time ordering, schema checks, privacy minimization, deduplication, and late-event handling. These are design requirements, not features proven by this repository.
2. **Build features:** compute rolling reaction-time distributions, timing regularity, target accuracy and repeated-action signals; partition by match/time and avoid using post-decision information. Graph features may summarize co-play links, recency and clustering.
3. **Score:** use a supervised baseline alongside anomaly signals. Compare models using time-based holdouts, PR-AUC, calibration, and review-budget precision. A combined score requires independent validation before adoption.
4. **Decision policy:** prioritize cases for review under a capacity budget; use reason codes and evidence provenance. Do not equate a risk score with guilt.
5. **Fail safely:** on missing required features, return `insufficient_telemetry`; on unavailable models, return `model_unavailable` and use a validated fallback or abstain. Normal gameplay continues.
6. **Monitor:** check input drift, label delay/selection bias, calibration, false positives by skill segment, review backlog, inference errors, and patch-related shifts. Investigate before retraining.
7. **Roll out cautiously:** proposed offline backtesting → shadow scoring → limited canary → A/B or staged evaluation, with rollback triggers and documented ownership. None of these release stages are demonstrated as deployed here.

## Latency and throughput caveats

The skill-parsing notebook reports **simulated/local scoring measurements** (P99 10.059 ms), not network, serialization, feature-store, queue, or end-to-end production latency. The real-time notebook performs **illustrative throughput and backlog calculations** using assumed traffic and worker capacity; it does not load-test a deployed streaming service. Real deployment would need representative hardware, concurrent traffic, feature freshness measurements, P95/P99 service SLOs, load shedding, and disaster recovery tests.

## Threats to validity and player trust

High-skill legitimate players may resemble scripted behavior; cheats adapt after detection; review labels are missing not at random; graph connections can reflect innocent teammates; patch changes may shift features. A safe system requires careful segment evaluation, independent labels, human adjudication, privacy and appeals, not just higher detection rates.

"""Generate reproducible, fully synthetic CS2-inspired player telemetry.

"""
import argparse

           "median_reaction_ms", "timing_regularities", "target_accuracy",
           "repeated_action_ratio", "synthetic_pattern", "is_suspicious_simulated"]

def generate(n=12000, seed=42, days=60):
    if n < 100 or days < 10:
        raise ValueError("n must be >= 100 and days >= 10")
    rng = np.random.default_rng(seed)
    day = rng.integers(0, days, n)
    skill = rng.uniform(0, 1, n)
    suspicious = rng.binomial(1, 0.065, n)
    style = np.where(suspicious == 0, "none",
                     np.where(rng.random(n) < 0.5, "fast_reaction", "precision_pattern"))
    reaction = 290 - 72 * skill - 65 * (style == "fast_reaction") - 18 * (style == "precision_pattern") + rng.normal(0, 22, n)
    accuracy = np.clip(0.27 + 0.36 * skill + 0.23 * (style == "precision_pattern") + 0.12 * (style == "fast_reaction") + rng.normal(0, 0.07, n), 0, 1)
    regularity = np.clip(0.23 + 0.17 * skill + 0.38 * suspicious + rng.normal(0, 0.11, n), 0, 1)
    repeated = np.clip(0.13 + 0.09 * skill + 0.31 * suspicious + rng.normal(0, 0.09, n), 0, 1)
    apm = np.maximum(1, 90 + 90 * skill + 40 * suspicious + rng.normal(0, 25, n))
    return pd.DataFrame({
        "player_id": np.arange(100000, 100000 + n),
        "match_id": np.arange(200000, 200000 + n),
        "match_day": day,
        "skill_proxy": skill.round(5),
        "actions_per_min": apm.round(3),
        "median_reaction_ms": reaction.clip(50, 600).round(3),
        "timing_regularities": regularity.round(5),
        "target_accuracy": accuracy.round(5),
        "repeated_action_ratio": repeated.round(5),
        "synthetic_pattern": style,
        "is_suspicious_simulated": suspicious
    })[COLUMNS]

if __name__ == "__main__":
    p = argparse.ArgumentParser()
    p.add_argument("--rows", type=int, default=12000)
    p.add_argument("--seed", type=int, default=42)
    p.add_argument("--output", default="data/synthetic_anticheat_telemetry.csv")
    a = p.parse_args()
    out = Path(a.output)
    out.parent.mkdir(parents=True, exist_ok=True)
    df = generate(a.rows, a.seed)
    df.to_csv(out, index=False)
    print(f"Generated {len(df)} synthetic rows to {out}; suspicious rate={df.is_suspicious_simulated.mean():.3f}")

```py
"""Reproducible synthetic baseline: temporal split, models, fairness and calibration.

No network, external data, production deployment or enforcement.
"""
import argparse
import json
from pathlib import Path
import numpy as np
import pandas as pd
from sklearn.ensemble import IsolationForest
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import average_precision_score, brier_score_loss, precision_score, recall_score, roc_auc_score
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler

FEATURES = ["skill_proxy", "actions_per_min", "median_reaction_ms",
            "timing_regularities", "target_accuracy", "repeated_action_ratio"]
LABEL = "is_suspicious_simulated"

def run(data_path, output_dir, threshold=0.7):
    df = pd.read_csv(data_path)
    needed = set(FEATURES + [LABEL, "match_day", "player_id", "match_id"])
    missing = needed - set(df.columns)
    if missing:
        raise ValueError(f"Missing columns: {sorted(missing)}")
    if df[list(needed)].isna().any().any():
        raise ValueError("Missing values in required fields")
    days = np.sort(df.match_day.unique())
    if len(days) < 5:
        raise ValueError("Need at least five unique days")
    cutoff = days[int(len(days) * 0.8)]
    train, test = df[df.match_day < cutoff].copy(), df[df.match_day >= cutoff].copy()
    if train.empty or test.empty or train[LABEL].nunique() < 2 or test[LABEL].nunique() < 2:
        raise ValueError("Temporal train/test split must have both classes")
    if set(train.match_id) & set(test.match_id):
        raise ValueError("Match identifiers leak across time split")
    if set(train.player_id) & set(test.player_id):
        raise ValueError("Player identifiers overlap across time split; review evaluation design")
    xtrain, xtest, ytrain, ytest = train[FEATURES], test[FEATURES], train[LABEL], test[LABEL]
    lr = make_pipeline(StandardScaler(), LogisticRegression(max_iter=1000, random_state=42))
    lr.fit(xtrain, ytrain)
    prob = lr.predict_proba(xtest)[:, 1]
    iso = make_pipeline(StandardScaler(), IsolationForest(n_estimators=200, contamination=0.07, random_state=42))
    iso.fit(xtrain)
    anomaly = -iso.decision_function(xtest)
    prediction = (prob >= threshold).astype(int)
    metrics = {
        "dataset": "synthetic_generated", "train_rows": int(len(train)), "test_rows": int(len(test)),
        "cutoff_day": int(cutoff), "test_prevalence": float(ytest.mean()), "threshold": threshold,
        "logistic_regression": {
            "roc_auc": float(roc_auc_score(ytest, prob)),
            "average_precision": float(average_precision_score(ytest, prob)),
            "brier": float(brier_score_loss(ytest, prob)),
            "precision": float(precision_score(ytest, prediction, zero_division=0)),
            "recall": float(recall_score(ytest, prediction, zero_division=0))
        },
        "isolation_forest": {
            "roc_auc": float(roc_auc_score(ytest, anomaly)),
            "average_precision": float(average_precision_score(ytest, anomaly))
        }
    }
    out = Path(output_dir)
    out.mkdir(parents=True, exist_ok=True)
    evaluation = test[["player_id", "match_day", "skill_proxy", LABEL]].copy()
    evaluation["predicted_probability"] = prob
    evaluation["predicted_label"] = prediction
    evaluation["skill_group"] = pd.qcut(evaluation.skill_proxy, q=5, labels=["Very Low", "Low", "Medium", "High", "Very High"])
    groups = []
    for group, subset in evaluation.groupby("skill_group", observed=True):
        normal = subset[subset[LABEL] == 0]
        fp = int(normal.predicted_label.sum())
        groups.append({"skill_group": str(group), "observations": int(len(subset)),
                       "normal_observations": int(len(normal)), "false_positives": fp,
                       "false_positive_rate": float(fp / len(normal)) if len(normal) else None})
    metrics["skill_segments"] = groups
    evaluation.to_csv(out / "predictions.csv", index=False)
    with (out / "metrics.json").open("w") as f:
        json.dump(metrics, f, indent=2)
    print(json.dumps(metrics, indent=2))
    return metrics

if __name__ == "__main__":
    p = argparse.ArgumentParser()
    p.add_argument("--data", default="data/synthetic_anticheat_telemetry.csv")
    p.add_argument("--output-dir", default="outputs")
    p.add_argument("--threshold", type=float, default=0.7)
    args = p.parse_args()
    if not 0 <= args.threshold <= 1:
        p.error("threshold must be between 0 and 1")
    run(args.data, args.output_dir, args.threshold)

import tempfile
import unittest
from pathlib import Path
from generate_synthetic_data import generate
from run_baseline import run

class BaselineTests(unittest.TestCase):
    def test_deterministic_generation(self):
        a = generate(1200, 42)
        b = generate(1200, 42)
        self.assertTrue(a.equals(b))
        self.assertEqual(a.shape, (1200, 11))
        self.assertFalse(a.isna().any().any())

    def test_end_to_end(self):
        with tempfile.TemporaryDirectory() as tmp:
            path = Path(tmp) / "data.csv"
            generate(3000, 42).to_csv(path, index=False)
            result = run(path, Path(tmp) / "out")
            self.assertGreater(result["logistic_regression"]["roc_auc"], 0.5)
            self.assertEqual(len(result["skill_segments"]), 5)
            self.assertTrue((Path(tmp) / "out" / "metrics.json").exists())

if __name__ == "__main__":
    unittest.main()
