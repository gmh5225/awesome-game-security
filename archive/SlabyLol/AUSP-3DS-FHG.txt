#include "anti_usp.h"
#include "config.h"
#include "scanner.h"
#include "screen.h"
#include "splash.h"
#include <stdio.h>

static AuspActionFn g_action = NULL;

void anti_usp_set_action(AuspActionFn fn)
{
    g_action = fn;
}

AuspResult anti_usp_check(void)
{
    if (ausp_scan_process())
        return AUSP_PLUGIN_FOUND;
    return AUSP_CLEAN;
}

void anti_usp_show_screen(void)
{
    ausp_draw_screen();
}

void anti_usp_run(void)
{
    ausp_show_splash();

    if (anti_usp_check() != AUSP_PLUGIN_FOUND)
        return;

    if (AUSP_REACTION_MODE == AUSP_REACTION_SCREEN ||
        AUSP_REACTION_MODE == AUSP_REACTION_BOTH)
    {
        anti_usp_show_screen();
    }

    if (AUSP_REACTION_MODE == AUSP_REACTION_ACTION ||
        AUSP_REACTION_MODE == AUSP_REACTION_BOTH)
    {
        if (g_action)
            g_action();
    }

    if (AUSP_FORCE_EXIT)
        svcExitProcess();
}

const char *anti_usp_version(void)
{
    static char buf[16];
    snprintf(buf, sizeof(buf), "%d.%d.%d",
             AUSP_VERSION_MAJOR, AUSP_VERSION_MINOR, AUSP_VERSION_PATCH);
    return buf;
}

        gfxSwapBuffers();
        gspWaitForVBlank();

    gfxExit();
    return 0;

#include "scanner.h"
#include "signatures.h"
#include "config.h"
#include <string.h>
#include <stdbool.h>

static bool region_contains(const u8 *base, u32 size, const u8 *sig, u32 siglen)
{
    if (!base || size < siglen || !sig || siglen == 0)
        return false;

    const u8 *end = base + size - siglen;
    for (const u8 *p = base; p <= end; p++) {
        if (memcmp(p, sig, siglen) == 0)
            return true;
    }
    return false;
}

static int collect_regions(AuspMemRegion *out, int max)
{
    u32 addr = 0;
    int count = 0;
    MemInfo info;
    PageInfo page;

    while (count < max && addr < 0x40000000) {
        Result res = svcQueryMemory(&info, &page, addr);
        if (R_FAILED(res))
            break;
        if (info.size == 0)
            break;

        bool interesting = false;

        if (AUSP_SCAN_CODE && (info.perm & MEMPERM_EXECUTE))
            interesting = true;
        if (AUSP_SCAN_DATA && (info.perm & MEMPERM_READ) && !(info.perm & MEMPERM_EXECUTE))
            interesting = true;

        if (interesting &&
            info.size >= AUSP_MIN_REGION_SIZE &&
            (info.state == MEMSTATE_CODE || info.state == MEMSTATE_PRIVATE || info.state == MEMSTATE_SHARED))
        {
            out[count].base  = info.base_addr;
            out[count].size  = info.size;
            out[count].state = info.state;
            out[count].perm  = info.perm;
            count++;
        }

        u32 next = info.base_addr + info.size;
        if (next <= addr)
            break;
        addr = next;
    }

    return count;
}

int ausp_scan_process(void)
{
    AuspMemRegion regions[AUSP_MAX_REGIONS];
    int n = collect_regions(regions, AUSP_MAX_REGIONS);

    int found_cheat = 0;
    int found_nimbus = 0;

    for (int i = 0; i < n; i++) {
        const u8 *base = (const u8 *)(uintptr_t)regions[i].base;
        u32 size = regions[i].size;

        for (u32 s = 0; s < AUSP_SIGNATURE_COUNT; s++) {
            if (region_contains(base, size,
                                ausp_signatures[s].data,
                                ausp_signatures[s].length))
            {
                if (ausp_signatures[s].is_nimbus)
                    found_nimbus = 1;
                else
                    found_cheat = 1;
            }
        }
    }

    if (AUSP_IGNORE_NIMBUS && found_nimbus && !found_cheat)
        return 0;

    return found_cheat;
}

        hidScanInput();

        if (AUSP_EXIT_ON_KEY && kDown)

        gfxSwapBuffers();
        gspWaitForVBlank();

    gfxExit();

#include "config.h"

static void draw_logo(float cx, float cy, float scale)
{
    float s = scale;
    u32 accent = C2D_Color32(AUSP_SPLASH_ACCENT_R, AUSP_SPLASH_ACCENT_G, AUSP_SPLASH_ACCENT_B, AUSP_SPLASH_ACCENT_A);
    u32 bg     = C2D_Color32(AUSP_SPLASH_BG_R, AUSP_SPLASH_BG_G, AUSP_SPLASH_BG_B, AUSP_SPLASH_BG_A);

    C2D_DrawRectSolid(cx - 55*s, cy - 45*s, 0.4f, 110*s, 90*s, accent);
    C2D_DrawRectSolid(cx - 48*s, cy - 38*s, 0.5f, 96*s, 76*s, bg);

    C2D_DrawCircleSolid(cx, cy - 5*s, 0.6f, 28*s, accent);
    C2D_DrawCircleSolid(cx, cy - 5*s, 0.7f, 18*s, bg);

    C2D_DrawRectSolid(cx - 8*s, cy + 18*s, 0.6f, 16*s, 22*s, accent);
}

void ausp_show_splash(void)
{
    if (!AUSP_SPLASH_ENABLE)
        return;

    gfxInitDefault();
    C3D_Init(C3D_DEFAULT_CMDBUF_SIZE);
    C2D_Init(C2D_DEFAULT_MAX_OBJECTS);
    C2D_Prepare();

    C3D_RenderTarget *top = C2D_CreateScreenTarget(GFX_TOP, GFX_LEFT);

    C2D_TextBuf buf = C2D_TextBufNew(4096);
    C2D_Text txtTitle, txtSub, txtSupported;

    C2D_TextParse(&txtTitle, buf, "AUSP-3DS");
    C2D_TextOptimize(&txtTitle);

    C2D_TextParse(&txtSub, buf, "Anti-Cheat Protection");
    C2D_TextOptimize(&txtSub);

    C2D_TextParse(&txtSupported, buf, "Supported by AUSP-3DS");
    C2D_TextOptimize(&txtSupported);

    u32 colBg     = C2D_Color32(AUSP_SPLASH_BG_R, AUSP_SPLASH_BG_G, AUSP_SPLASH_BG_B, AUSP_SPLASH_BG_A);
    u32 colText   = C2D_Color32(AUSP_SPLASH_TEXT_R, AUSP_SPLASH_TEXT_G, AUSP_SPLASH_TEXT_B, AUSP_SPLASH_TEXT_A);
    u32 colSub    = C2D_Color32(AUSP_SPLASH_SUB_R, AUSP_SPLASH_SUB_G, AUSP_SPLASH_SUB_B, AUSP_SPLASH_SUB_A);
    u32 colAccent = C2D_Color32(AUSP_SPLASH_ACCENT_R, AUSP_SPLASH_ACCENT_G, AUSP_SPLASH_ACCENT_B, AUSP_SPLASH_ACCENT_A);

    int frames = 0;
    while (aptMainLoop() && frames < AUSP_SPLASH_DURATION)
    {
        hidScanInput();
        if (hidKeysDown() & KEY_START)
            break;

        C3D_FrameBegin(C3D_FRAME_SYNCDRAW);
        C2D_TargetClear(top, colBg);
        C2D_SceneBegin(top);

        draw_logo(200.0f, 95.0f, 1.15f);

        C2D_DrawText(&txtTitle, C2D_AlignCenter | C2D_WithColor,
                     200.0f, 155.0f, 0.5f, 0.85f, 0.85f, colText);

        C2D_DrawText(&txtSub, C2D_AlignCenter | C2D_WithColor,
                     200.0f, 185.0f, 0.5f, 0.55f, 0.55f, colSub);

        C2D_DrawText(&txtSupported, C2D_AlignCenter | C2D_WithColor,
                     200.0f, 215.0f, 0.5f, 0.50f, 0.50f, colAccent);

        C3D_FrameEnd(0);
        frames++;
    }

    C2D_TextBufDelete(buf);
    C2D_Fini();
    C3D_Fini();
    gfxExit();
}

`README.md`:

```md
# AUSP-3DS

**Anti-USP-3DS** – Advanced lightweight plugin detection for Nintendo 3DS homebrew.

Detects NTR, 3GX, CTRPluginFramework and related loaders **without any filesystem access**.

Always shows a clean **citro2d splash** with the AUSP logo and "Supported by AUSP-3DS" at the start of every homebrew that integrates it.

**Nimbus (Pretendo) is explicitly ignored** so legitimate online players are not blocked.

Fully configurable reaction after detection:
- Show anti-cheat screen
- Call a custom action / callback
- Or both

## Features

- Memory-only detection (no SD card / folder checks)
- Dynamic memory region discovery via `svcQueryMemory`
- Multiple known signatures
- **Ignores Nimbus / Pretendo plugin** (configurable)
- Configurable reaction: screen only / action only / both
- Automatic citro2d splash screen with logo on every start
- Fully customizable texts and colors
- Drop-in ready

## Installation (IMPORTANT)

1. Download this repository as ZIP
2. Extract it
3. Copy the entire **`AUSP`** folder into your project  
   - Either into `source/` **or** into the project root (both work)
4. Open your `Makefile` and add these lines in the **variables section** (before any rules):

```makefile
SOURCES  += AUSP
INCLUDES += AUSP
LIBS     += -lcitro2d -lcitro3d
```

   Do **not** use `include AUSP/Makefile.inc` if you get Makefile errors.

5. In your code (as early as possible):

```c
#include "anti_usp.h"

int main(int argc, char **argv)
{
    anti_usp_run();

    // rest of your game
}
```

### Common errors & fixes

| Error | Fix |
|-------|-----|
| `ausp_signatures` / `AUSP_SIGNATURE_COUNT` undeclared | Update to latest AUSP (fixed in v1.4.3 – signatures moved to .c) |
| `C2D_Color32 undeclared` | Update to latest AUSP |
| `*** missing separator` | Recipe lines need a real Tab, not spaces |
| `overriding recipe for target 'build'` | Remove any `include AUSP/Makefile.inc`, add the 3 lines manually |
| `anti_usp.h: No such file` | Make sure `INCLUDES += AUSP` is set and use `#include "anti_usp.h"` |
