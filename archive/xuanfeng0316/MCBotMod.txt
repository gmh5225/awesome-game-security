`README.md`:

```md
# MCBotMod

Through a Fabric mod, Minecraft client's native methods are exposed as a TCP + JSON interface. Any program capable of TCP communication can call this mod's interface to operate a real, complete game client.

**Not packet simulation, not a headless process — it calls the game's native methods through the interface and runs on a real client.**

## Core Features

**Real Client**
Graphics, rendering, and UI are fully preserved; program operations are visible in sync with the screen. Other Fabric mods work as usual, and both premium and offline login follow the vanilla process. Any flow that requires a graphical interface is unrestricted.

**Native Methods**
Exposes the game's own methods; the project's ceiling is the game's ceiling. TCP + JSON — any language supporting TCP communication can connect. The mod supports concurrent operations, allowing multiple actions to run simultaneously.

## Human–Program Coexistence

Player input and program output are additive, not mutually exclusive. When the program stops sending commands, the player takes over immediately; in emergencies, the player can intervene directly via keyboard. Switching between fully automatic, semi-automatic, and fully manual modes is seamless.

## Startup

 No additional programs need to be deployed. All you need is a Minecraft Java Edition Fabric client, this mod, and two dependencies (Fabric API, Fabric Language Kotlin) — and it's ready to run.

## AI Integration
AI can directly generate calls that match the interface's expected format, and the returned format natively fits AI parsing needs, making integration frictionless. Integration examples are included in the `TestScript/Ai/` directory.
```

`TestScript/Ai/ai.py`:

```py
# Copyright (C) 2026 xuanfeng0316
# SPDX-License-Identifier: GPL-3.0-or-later

import json
import socket
import sys
import time
import urllib.request
import urllib.error
import re

API_KEY = "sk-"  # API key
API_URL = "https://api.deepseek.com/chat/completions"  # Endpoint
MODEL = "deepseek-flash"  # Model

# Example
#API_KEY = "sk-xxxxxxxxxxxxxxxxxxxxxxxx"
#API_URL = "https://api.deepseek.com/chat/completions"
#MODEL = "deepseek-flash"

MOD_HOST = "127.0.0.1"  # MCBotMod host
MOD_PORT = 25566  # MCBotMod port

MAX_TOKENS = 20000  # Context limit
EST_CHARS_PER_TOKEN = 2.5  # Token estimation ratio

def est_tokens(text):
    return int(len(text) / EST_CHARS_PER_TOKEN)

def load_file(path):
    with open(path, "r", encoding="utf-8") as f:
        return f.read()

def load_tools(path):
    return json.loads(load_file(path))

class ModClient:
    def __init__(self, host, port):
        self.sock = socket.create_connection((host, port))
        self.f = self.sock.makefile("rw", encoding="utf-8")
        self.next_id = 1

    def call(self, method, params):
        req_id = self.next_id
        self.next_id += 1
        req = {"id": req_id, "method": method, "params": params}
        self.f.write(json.dumps(req, ensure_ascii=False) + "\n")
        self.f.flush()
        line = self.f.readline()
        if not line:
            return {"ok": False, "error": {"code": "DISCONNECTED", "message": "socket closed"}}
        return json.loads(line)

def call_llm(messages, tools):
    body = {
        "model": MODEL,
        "messages": messages,
        "tools": tools,
        "tool_choice": "auto",
    }
    data = json.dumps(body, ensure_ascii=False).encode("utf-8")
    req = urllib.request.Request(
        API_URL,
        data=data,
        headers={
            "Content-Type": "application/json",
            "Authorization": "Bearer " + API_KEY,
        },
        method="POST",
    )
    try:
        with urllib.request.urlopen(req, timeout=120) as resp:
            return json.loads(resp.read().decode("utf-8"))
    except urllib.error.HTTPError as e:
        print("[HTTP ERROR]", e.code)
        print(e.read().decode("utf-8", errors="replace"))
        sys.exit(1)

def truncate_history(messages):
    while True:
        total = 0
        for m in messages:
            content = m.get("content")
            if isinstance(content, str):
                total += est_tokens(content)
            elif isinstance(content, list):
                total += est_tokens(json.dumps(content, ensure_ascii=False))
            if "tool_calls" in m:
                total += est_tokens(json.dumps(m["tool_calls"], ensure_ascii=False))
        if total <= MAX_TOKENS:
            break

        if len(messages) <= 2:
            break

        i = 1
        if i < len(messages) and messages[i].get("role") == "user":
            j = i + 1
            while j < len(messages) and messages[j].get("role") != "user":
                j += 1
            del messages[i:j]
        else:
            messages.pop()

def connect_mod(host, port):
    first_failure = True
    while True:
        try:
            mod = ModClient(host, port)
            print(f"[Connected] MCBotMod is ready at {host}:{port}")
            return mod
        except OSError as exc:
            if first_failure:
                print(f"[Connection failed] {host}:{port} - {exc}. Retrying...")
                first_failure = False
            time.sleep(2)


def main():
    prompt = load_file("prompt.txt")
    tools = load_tools("Tool.json")

    mod = connect_mod(MOD_HOST, MOD_PORT)

    messages = [
        {"role": "system", "content": prompt},
    ]

    print("MCBotMod AI console. Type /bye or Ctrl+C to exit.\n")

    while True:
        try:
            user_input = input("You> ")
        except (EOFError, KeyboardInterrupt):
            print()
            break

        if user_input.strip() == "/bye":
            break
        if not user_input.strip():
            continue

        turn_user_message = {"role": "user", "content": user_input}
        messages.append(turn_user_message)

        try:
            while True:
                truncate_history(messages)
                resp = call_llm(messages, tools)
                choice = resp["choices"][0]
                msg = choice["message"]

                messages.append(msg)

                if msg.get("content"):
                    print("AI> " + msg["content"])

                tool_calls = msg.get("tool_calls")
                if not tool_calls:
                    break

                for tc in tool_calls:
                    fn = tc["function"]
                    args = json.loads(fn["arguments"])
                    method = args.get("method")
                    params = args.get("params", {})
                    print(f"[Call] {method} {json.dumps(params, ensure_ascii=False)}")
                    result = mod.call(method, params)
                    print(f"[Return] {json.dumps(result, ensure_ascii=False)}")
                    messages.append({
                        "role": "tool",
                        "tool_call_id": tc["id"],
                        "content": json.dumps(result, ensure_ascii=False),
                    })
        except KeyboardInterrupt:
            if messages and messages[-1] == turn_user_message:
                messages.pop()
            else:
                while len(messages) > 0 and messages[-1].get("role") not in {"user", "assistant", "tool"}:
                    messages.pop()
                while len(messages) > 0 and messages[-1].get("role") in {"assistant", "tool"}:
                    messages.pop()
                if len(messages) > 0 and messages[-1].get("role") == "user":
                    last_user = messages[-1]
                    if last_user.get("content") == user_input:
                        messages.pop()
            print("\n[Interrupted] Current turn was discarded and returned to the prompt.")
            continue

    print("Bye.")

if __name__ == "__main__":
    main()
```

```
#   Important for patching:
#
#   (2) This script targets any POSIX shell, so it avoids extensions provided
#       by Bash, Ksh, etc; in particular arrays are avoided.
#
#       The "traditional" practice of packing multiple parameters into a
#       space-separated string is a well documented source of bugs and security
#       problems, so this is (mostly) avoided, by progressively accumulating
#       options in "$@", and eventually passing that to Java.
#
#       Where the inherited environment variables (DEFAULT_JVM_OPTS, JAVA_OPTS,
#       and GRADLE_OPTS) rely on word-splitting, this is performed explicitly;
#       see the in-line comments for details.
#
#       There are tweaks for specific operating systems such as AIX, CygWin,

    APP_HOME=${app_path%"${app_path##*/}"}  # leaves a trailing /; empty if no leading path
    [ -h "$app_path" ]
do
    ls=$( ls -ld "$app_path" )
    link=${ls#*' -> '}
    case $link in             #(
      /*)   app_path=$link ;; #(
      *)    app_path=$APP_HOME$link ;;
    esac
done

    echo "$*"
} >&2

    echo
    echo "$*"
    echo

} >&2

msys=false
darwin=false
nonstop=false
case "$( uname )" in                #(
  CYGWIN* )         cygwin=true  ;; #(
  Darwin* )         darwin=true  ;; #(
  MSYS* | MINGW* )  msys=true    ;; #(
  NONSTOP* )        nonstop=true ;;
esac

if [ -n "$JAVA_HOME" ] ; then

        # IBM's JDK on AIX uses strange locations for the executables

      max*)

        # shellcheck disable=SC2039,SC3045

            warn "Could not query maximum file descriptor limit"
    esac
    case $MAX_FD in  #(
      '' | soft) :;; #(

        # shellcheck disable=SC2039,SC3045

#   * args from the command line
#   * the main class name
#   * -classpath
#   * -D...appname settings
#   * --module-path (only if needed)
#   * DEFAULT_JVM_OPTS, JAVA_OPTS, and GRADLE_OPTS environment variables.

    JAVACMD=$( cygpath --unix "$JAVACMD" )

    # Now convert the arguments - kludge to limit ourselves to /bin/sh
    for arg do
        if
            case $arg in                                #(
              -*)   false ;;                            # don't mess with options #(
              /?*)  t=${arg#/} t=/${t%%/*}              # looks like a POSIX filepath
                    [ -e "$t" ] ;;                      #(
              *)    false ;;
            esac
        then
            arg=$( cygpath --path --ignore --mixed "$arg" )
        fi
        # Roll the args list around exactly as many times as the number of
        # args, so each arg winds up back in the position where it started, but
        # possibly modified.
        #
        # NB: a `for` loop captures its iteration list before it begins, so
        # changing the positional parameters here affects neither the number of
        # iterations, nor the values presented in `arg`.
        shift                   # remove old arg
        set -- "$@" "$arg"      # push replacement arg
    done
fi

DEFAULT_JVM_OPTS='"-Xmx64m" "-Xms64m"'

# Collect all arguments for the java command:
#   * DEFAULT_JVM_OPTS, JAVA_OPTS, and optsEnvironmentVar are not allowed to contain shell fragments,
#     and any embedded shellness will be escaped.
#   * For example: A user cannot expect ${Hostname} to be expanded, as it is an environment variable and will be
#     treated as '${Hostname}' itself on the command line.

set -- \
        "-Dorg.gradle.appname=$APP_BASE_NAME" \
        -jar "$APP_HOME/gradle/wrapper/gradle-wrapper.jar" \
        "$@"

if ! command -v xargs >/dev/null 2>&1
then
    die "xargs is not available"
fi

# Use "xargs" to parse quoted args.
#
# With -n1 it outputs one arg per line, with the quotes and backslashes removed.
#
# In Bash we could simply go:
#
#   readarray ARGS < <( xargs -n1 <<<"$var" ) &&
#   set -- "${ARGS[@]}" "$@"
#
# but POSIX shell has neither arrays nor command substitution, so instead we
# post-process each arg (as a line of input to sed) to backslash-escape any
# character that might be a shell metacharacter, then use eval to reverse
# that process (while maintaining the separation between arguments), and wrap
# the whole thing up as a single "set" statement.
#
# This will of course break if any of these variables contains a newline or
# an unmatched quote.
#

eval "set -- $(
        printf '%s\n' "$DEFAULT_JVM_OPTS $JAVA_OPTS $GRADLE_OPTS" |
        xargs -n1 |
        sed ' s~[^-[:alnum:]+,./:=@_]~\\&~g; ' |
        tr '\n' ' '
    )" '"$@"'
```

```kt
	private val GSON = Gson()
	private const val PORT = 25566

	private class Request(
		val id: JsonElement?,
		val method: String,
		val params: JsonObject,
		val writer: BufferedWriter,
	)

	private val queue = ConcurrentLinkedQueue<Request>()
	private val chatSubscribers = ConcurrentHashMap<BufferedWriter, Boolean>()

	@Volatile private var autoRespawn = false

	override fun onInitializeClient() {
		LOGGER.info("MCBotModClient starting, listening on $PORT")

		ClientTickEvents.END_CLIENT_TICK.register {
			TaskManager.tick()
			drain()
			if (autoRespawn) {
				val mc = Minecraft.getInstance()
				if (mc.screen is DeathScreen) {
					mc.player?.respawn()
				}
			}
		}

		Thread(::runServer, "mcbotmod-net").apply {
			isDaemon = true
			start()
		}
	}

	private fun runServer() {
		try {
			ServerSocket(PORT, 1, InetAddress.getLoopbackAddress()).use { server ->
				LOGGER.info("Listening on $PORT")
				while (true) {
					val socket = server.accept()
					LOGGER.info("Client connected: ${socket.remoteSocketAddress}")
					Thread({ handleClient(socket) }, "mcbotmod-client").apply {
						isDaemon = true
						start()
					}
				}
			}
		} catch (e: Exception) {
			LOGGER.error("Server error", e)
		}
	}

	private fun handleClient(socket: Socket) {
		val reader: BufferedReader = socket.getInputStream().bufferedReader()
		val writer: BufferedWriter = socket.getOutputStream().bufferedWriter()

		try {
			reader.forEachLine { line ->
				if (line.isBlank()) return@forEachLine
				try {
					val json = JsonParser.parseString(line).asJsonObject
					val id = json.get("id")
					val method = json.get("method").asString
					val params = json.getAsJsonObject("params") ?: JsonObject()

					when (method) {
						"ping" -> {
							respond(writer, id, "pong")
							return@forEachLine
						}
						"protocol_version" -> {
							respond(writer, id, 1)
							return@forEachLine
						}
					}

					queue.offer(Request(id, method, params, writer))
				} catch (e: Exception) {
					LOGGER.warn("Bad request: $line", e)
				}
			}
		} finally {
			TaskManager.cancelAllFor(writer)
			chatSubscribers.remove(writer)
		}
	}

	private fun drain() {
		while (true) {
			val req = queue.poll() ?: break
			try {
				val result = handle(req.method, req.params, req.writer)
				respond(req.writer, req.id, result)
			} catch (e: MethodException) {
				respondError(req.writer, req.id, e.code, e.message ?: "")
			} catch (e: Exception) {
				LOGGER.error("Handle error: ${req.method}", e)
				respondError(req.writer, req.id, "INTERNAL", e.message ?: "")
			}
		}
	}

	private class MethodException(val code: String, message: String) : Exception(message)

	private fun handle(method: String, params: JsonObject, writer: BufferedWriter): Any {
		return when (method) {
			"set_move" -> {
				Movement.set(
					forward = params.get("forward")?.asBoolean,
					backward = params.get("backward")?.asBoolean,
					left = params.get("left")?.asBoolean,
					right = params.get("right")?.asBoolean,
					jump = params.get("jump")?.asBoolean,
					sneak = params.get("sneak")?.asBoolean,
				)
				JsonObject()
			}
			"stop_move" -> {
				Movement.stop()
				JsonObject()
			}
			"set_sprint" -> {
				val enabled = params.get("enabled")?.asBoolean ?: false
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")
				player.isSprinting = enabled
				JsonObject()
			}
			"look" -> {
				Rotation.set(
					yaw = params.get("yaw")?.asFloat,
					pitch = params.get("pitch")?.asFloat,
				)
				JsonObject()
			}
			"get_status" -> {
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")

				val obj = JsonObject()
				obj.addProperty("health", player.health)
				obj.addProperty("food", player.foodData.foodLevel)
				obj.addProperty("saturation", player.foodData.saturationLevel)
				obj
			}
			"get_position" -> {
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")

				val obj = JsonObject()
				obj.addProperty("x", player.x)
				obj.addProperty("y", player.y)
				obj.addProperty("z", player.z)
				obj.addProperty("yaw", player.yRot)
				obj.addProperty("pitch", player.xRot)
				obj
			}
			"get_nearby_blocks" -> {
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")
				val level = player.level()

				val radius = (params.get("radius")?.asInt ?: 4).coerceIn(0, 32)
				val center = player.blockPosition()
				val blocks = com.google.gson.JsonArray()

				for (pos in BlockPos.betweenClosed(
					center.offset(-radius, -radius, -radius),
					center.offset(radius, radius, radius),
				)) {
					val state = level.getBlockState(pos)
					if (state.isAir) continue

					val entry = com.google.gson.JsonArray()
					entry.add(net.minecraft.core.registries.BuiltInRegistries.BLOCK.getKey(state.block).toString())
					val coord = com.google.gson.JsonArray()
					coord.add(pos.x); coord.add(pos.y); coord.add(pos.z)
					entry.add(coord)
					val props = JsonObject()
					for (prop in state.properties) {
						props.addProperty(prop.name, state.getValue(prop).toString())
					}
					entry.add(props)
					blocks.add(entry)
				}

				val obj = JsonObject()
				obj.add("blocks", blocks)
				obj
			}
			"get_nearby_entities" -> {
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")
				val level = player.level()

				val radius = (params.get("radius")?.asDouble ?: 16.0).coerceIn(0.0, 64.0)
				val entities = com.google.gson.JsonArray()

				val box = player.boundingBox.inflate(radius)
				for (entity in level.getEntities(player, box)) {
					val entry = JsonObject()
					entry.addProperty("id", entity.id)
					entry.addProperty("type", net.minecraft.core.registries.BuiltInRegistries.ENTITY_TYPE.getKey(entity.type).toString())
					entry.addProperty("x", entity.x)
					entry.addProperty("y", entity.y)
					entry.addProperty("z", entity.z)
					entry.addProperty("distance", player.distanceTo(entity))
					entities.add(entry)
				}

				val obj = JsonObject()
				obj.add("entities", entities)
				obj
			}
			"interact_block" -> {
				val mc = Minecraft.getInstance()
				val player = mc.player ?: throw MethodException("NOT_IN_GAME", "player is null")
				val gameMode = mc.gameMode ?: throw MethodException("NOT_IN_GAME", "gameMode is null")

				val pos = BlockPos(
					params.get("x").asInt,
					params.get("y").asInt,
					params.get("z").asInt,
				)
				val face = parseFace(params.get("face")?.asString, player, pos)
				val hand = parseHand(params.get("hand")?.asString)

				val center = Vec3(pos.x + 0.5, pos.y + 0.5, pos.z + 0.5)
				val hit = BlockHitResult(center, face, pos, false)
				val result = gameMode.useItemOn(player, hand, hit)

				val obj = JsonObject()
				obj.addProperty("result", result.toString())
				obj
			}
			"break_block" -> {
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")
				val pos = BlockPos(
					params.get("x").asInt,
					params.get("y").asInt,
					params.get("z").asInt,
				)
				val face = parseFace(params.get("face")?.asString, player, pos)
				val timeout = params.get("timeout")?.asLong

				val taskId = TaskManager.nextId()
				TaskManager.submit(BreakBlockTask(taskId, writer, pos, face, timeout))

				val obj = JsonObject()
				obj.addProperty("task", taskId)
				obj
			}
			"cancel_task" -> {
				val taskId = params.get("task").asString
				val cancelled = TaskManager.cancel(taskId)
				val obj = JsonObject()
				obj.addProperty("cancelled", cancelled)
				obj
			}
			"use_item" -> {
				val mc = Minecraft.getInstance()
				val player = mc.player ?: throw MethodException("NOT_IN_GAME", "player is null")
				val gameMode = mc.gameMode ?: throw MethodException("NOT_IN_GAME", "gameMode is null")

				val hand = parseHand(params.get("hand")?.asString)
				val result = gameMode.useItem(player, hand)

				val obj = JsonObject()
				obj.addProperty("result", result.toString())
				obj
			}
			"interact_entity" -> {
				val mc = Minecraft.getInstance()
				val player = mc.player ?: throw MethodException("NOT_IN_GAME", "player is null")
				val level = mc.level ?: throw MethodException("NOT_IN_GAME", "level is null")
				val gameMode = mc.gameMode ?: throw MethodException("NOT_IN_GAME", "gameMode is null")

				val entityId = params.get("entityId").asInt
				val entity = level.getEntity(entityId)
					?: throw MethodException("NOT_FOUND", "entity $entityId not found")
				val hand = parseHand(params.get("hand")?.asString)

				val hit = EntityHitResult(entity, entity.position())
				val result = gameMode.interact(player, entity, hit, hand)

				val obj = JsonObject()
				obj.addProperty("result", result.toString())
				obj
			}
			"attack_entity" -> {
				val mc = Minecraft.getInstance()
				val player = mc.player ?: throw MethodException("NOT_IN_GAME", "player is null")
				val level = mc.level ?: throw MethodException("NOT_IN_GAME", "level is null")
				val gameMode = mc.gameMode ?: throw MethodException("NOT_IN_GAME", "gameMode is null")

				val entityId = params.get("entityId").asInt
				val entity = level.getEntity(entityId)
					?: throw MethodException("NOT_FOUND", "entity $entityId not found")

				gameMode.attack(player, entity)

				JsonObject()
			}
			"list_singleplayer_worlds" -> {
				val mc = Minecraft.getInstance()
				val levelSource = mc.levelSource

				val candidates = levelSource.findLevelCandidates()
				val summaries = levelSource.loadLevelSummaries(candidates).get()

				val arr = com.google.gson.JsonArray()
				for (s in summaries) {
					val entry = JsonObject()
					entry.addProperty("id", s.levelId)
					entry.addProperty("name", s.levelName)
					entry.addProperty("lastPlayed", s.lastPlayed)
					arr.add(entry)
				}

				val obj = JsonObject()
				obj.add("worlds", arr)
				obj
			}
			"connect_server" -> {
				val host = params.get("host").asString
				val port = params.get("port")?.asInt ?: 25565

				val address = ServerAddress.parseString("$host:$port")
				val serverData = ServerData(host, "$host:$port", ServerData.Type.OTHER)

				val taskId = TaskManager.nextId()
				TaskManager.submit(ConnectServerTask(taskId, writer, address, serverData))

				val obj = JsonObject()
				obj.addProperty("task", taskId)
				obj
			}
			"open_singleplayer_world" -> {
				val levelId = params.get("id").asString
				val taskId = TaskManager.nextId()
				TaskManager.submit(OpenWorldTask(taskId, writer, levelId))

				val obj = JsonObject()
				obj.addProperty("task", taskId)
				obj
			}
			"disconnect" -> {
				Minecraft.getInstance().execute {
					Minecraft.getInstance().disconnectFromWorld(Component.literal("disconnected by bot"))
				}
				JsonObject()
			}
			"shutdown" -> {
				Minecraft.getInstance().execute {
					Minecraft.getInstance().stop()
				}
				JsonObject()
			}
			"get_server_info" -> {
				val host = params.get("host").asString
				val port = params.get("port")?.asInt ?: 25565

				val serverData = ServerData(host, "$host:$port", ServerData.Type.OTHER)
				val taskId = TaskManager.nextId()
				TaskManager.submit(ServerInfoTask(taskId, writer, serverData))

				val obj = JsonObject()
				obj.addProperty("task", taskId)
				obj
			}
			"send_chat" -> {
				val message = params.get("message").asString
				val connection = Minecraft.getInstance().player?.connection
					?: throw MethodException("NOT_IN_GAME", "player is null")

				if (message.startsWith("/")) {
					connection.sendCommand(message.substring(1))
				} else {
					connection.sendChat(message)
				}
				JsonObject()
			}
			"subscribe_chat" -> {
				val history = params.get("history")?.asBoolean ?: false
				chatSubscribers[writer] = true

				if (history) {
					val mc = Minecraft.getInstance()
					val chat = mc.gui.chat
					val accessor = chat as com.xuanfeng.mcbotmod.client.mixin.ChatAccessor
					val messages = accessor.allMessages
					for (msg in messages.reversed()) {
						pushChatTo(writer, msg.content().string, msg.source().name)
					}
				}

				JsonObject()
			}
			"key" -> {
				val keyParam = params.get("key")
				val key = if (keyParam.isJsonPrimitive && keyParam.asJsonPrimitive.isNumber) {
					keyParam.asInt
				} else {
					keyCodeFromName(keyParam.asString)
				}
				val action = params.get("action")?.asString ?: "click"
				val mc = Minecraft.getInstance()
				val handler = mc.keyboardHandler as com.xuanfeng.mcbotmod.client.mixin.KeyboardHandlerInvoker
				val window = mc.window.handle()

				val event = net.minecraft.client.input.KeyEvent(key, 0, 0)

				when (action.lowercase()) {
					"press" -> handler.invokeKeyPress(window, 1, event)
					"release" -> handler.invokeKeyPress(window, 0, event)
					else -> {
						handler.invokeKeyPress(window, 1, event)
						handler.invokeKeyPress(window, 0, event)
					}
				}
				JsonObject()
			}
			"respawn" -> {
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")
				player.respawn()
				JsonObject()
			}
			"auto_respawn" -> {
				val enabled = params.get("enabled")?.asBoolean ?: false
				autoRespawn = enabled
				JsonObject()
			}
			"esc" -> {
				val open = params.get("open")?.asBoolean ?: true
				val mc = Minecraft.getInstance()
				mc.execute {
					if (open) {
						mc.pauseGame(false)
					} else {
						mc.setScreen(null)
					}
				}
				JsonObject()
			}
			"get_inventory" -> {
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")
				val inv = player.inventory
				val level = player.level()

				val obj = JsonObject()
				obj.addProperty("selected", inv.selectedSlot)

				val main = com.google.gson.JsonArray()
				for (i in 0 until 36) {
					val stack = inv.getItem(i)
					if (stack.isEmpty) continue
					main.add(stackToJson(i, stack, level))
				}
				obj.add("main", main)

				val armor = com.google.gson.JsonArray()
				for (i in 36..39) {
					val stack = inv.getItem(i)
					if (stack.isEmpty) continue
					armor.add(stackToJson(i, stack, level))
				}
				obj.add("armor", armor)

				val offhand = inv.getItem(40)
				if (!offhand.isEmpty) {
					obj.add("offhand", stackToJson(40, offhand, level))
				}

				val mainHand = inv.getItem(inv.selectedSlot)
				if (!mainHand.isEmpty) {
					obj.add("mainHand", stackToJson(inv.selectedSlot, mainHand, level))
				}

				obj
			}
			"get_hotbar" -> {
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")
				val inv = player.inventory
				val level = player.level()

				val obj = JsonObject()
				obj.addProperty("selected", inv.selectedSlot)

				val slots = com.google.gson.JsonArray()
				for (i in 0 until 9) {
					val stack = inv.getItem(i)
					if (stack.isEmpty) continue
					slots.add(stackToJson(i, stack, level))
				}
				obj.add("slots", slots)

				val offhand = inv.getItem(40)
				if (!offhand.isEmpty) {
					obj.add("offhand", stackToJson(40, offhand, level))
				}

				val mainHand = inv.getItem(inv.selectedSlot)
				if (!mainHand.isEmpty) {
					obj.add("mainHand", stackToJson(inv.selectedSlot, mainHand, level))
				}

				obj
			}
			"select_slot" -> {
				val slot = params.get("slot").asInt
				if (slot < 0 || slot > 8) {
					throw MethodException("BAD_REQUEST", "slot must be 0-8")
				}
				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")
				player.inventory.setSelectedSlot(slot)
				JsonObject()
			}
			"move_to_offhand" -> {
				val slot = params.get("slot").asInt
				if (slot < 0 || slot > 35) {
					throw MethodException("BAD_REQUEST", "slot must be 0-35")
				}
				val mc = Minecraft.getInstance()
				val player = mc.player ?: throw MethodException("NOT_IN_GAME", "player is null")
				val gameMode = mc.gameMode ?: throw MethodException("NOT_IN_GAME", "gameMode is null")

				val menuSlot = if (slot < 9) 36 + slot else slot
				gameMode.handleContainerInput(0, menuSlot, 40, net.minecraft.world.inventory.ContainerInput.SWAP, player)
				JsonObject()
			}

			"goto" -> {
				val stop = params.get("stop")?.asBoolean ?: false
				if (stop) {
					TaskManager.cancelAllOfType { it is GotoTask }
					return JsonObject()
				}

				val player = Minecraft.getInstance().player
					?: throw MethodException("NOT_IN_GAME", "player is null")

				val targetX = parseCoord(params.get("x"), player.blockX, false)!!
				val targetY = parseCoord(params.get("y"), player.blockY, true)
				val targetZ = parseCoord(params.get("z"), player.blockZ, false)!!

				val allowBreak = params.get("allow_break")?.asBoolean ?: false
				val allowPlace = params.get("allow_place")?.asBoolean ?: false

				val taskId = TaskManager.nextId()
				TaskManager.submit(GotoTask(taskId, writer, targetX, targetY, targetZ, allowBreak, allowPlace))

				val obj = JsonObject()
				obj.addProperty("task", taskId)
				obj
			}

			else -> throw MethodException("UNKNOWN_METHOD", method)
		}
	}

	private fun parseCoord(input: JsonElement?, current: Int, allowQuestion: Boolean): Int? {
		if (input == null || input.isJsonNull) {
			throw MethodException("BAD_REQUEST", "missing coord")
		}
		if (input.isJsonPrimitive) {
			val p = input.asJsonPrimitive
			if (p.isNumber) return p.asInt
			if (p.isString) {
				val s = p.asString
				if (s == "?") {
					if (!allowQuestion) throw MethodException("BAD_REQUEST", "? only allowed for y")
					return null
				}
				if (s == "~") return current
				if (s.startsWith("~")) return current + s.substring(1).toInt()
				return s.toInt()
			}
		}
		throw MethodException("BAD_REQUEST", "bad coord")
	}

	fun pushChat(message: String, source: String) {
		for (writer in chatSubscribers.keys) {
			pushChatTo(writer, message, source)
		}
	}

	private fun pushChatTo(writer: BufferedWriter, message: String, source: String) {
		try {
			val data = JsonObject()
			data.addProperty("message", message)
			data.addProperty("source", source)
			val event = JsonObject()
			event.addProperty("event", "chat")
			event.add("data", data)
			synchronized(writer) {
				writer.write(GSON.toJson(event))
				writer.newLine()
				writer.flush()
			}
		} catch (e: Exception) {
		}
	}

	private fun parseFace(name: String?, player: LocalPlayer, pos: BlockPos): Direction {
		if (name != null) {
			return when (name.lowercase()) {
				"up" -> Direction.UP
				"down" -> Direction.DOWN
				"north" -> Direction.NORTH
				"south" -> Direction.SOUTH
				"east" -> Direction.EAST
				"west" -> Direction.WEST
				else -> throw MethodException("BAD_REQUEST", "invalid face: $name")
			}
		}
		val eye = player.eyePosition
		val dx = pos.x - eye.x.toInt()
		val dy = pos.y - eye.y.toInt()
		val dz = pos.z - eye.z.toInt()
		return Direction.getNearest(dx, dy, dz, Direction.UP) ?: Direction.UP
	}

	private fun parseHand(name: String?): InteractionHand {
		return when (name?.lowercase()) {
			null, "main" -> InteractionHand.MAIN_HAND
			"off" -> InteractionHand.OFF_HAND
			else -> throw MethodException("BAD_REQUEST", "invalid hand: $name")
		}
	}

	private fun keyCodeFromName(name: String): Int {
		val n = name.uppercase()
		if (n.length == 1 && n[0] in 'A'..'Z') return n[0].code
		if (n.length == 1 && n[0] in '0'..'9') return n[0].code
		return when (n) {
			"SPACE" -> 32
			"ESC", "ESCAPE" -> 256
			"ENTER", "RETURN" -> 257
			"TAB" -> 258
			"BACKSPACE" -> 259
			"INSERT" -> 260
			"DELETE" -> 261
			"RIGHT" -> 262
			"LEFT" -> 263
			"DOWN" -> 264
			"UP" -> 265
			"PAGE_UP" -> 266
			"PAGE_DOWN" -> 267
			"HOME" -> 268
			"END" -> 269
			"F1" -> 290; "F2" -> 291; "F3" -> 292; "F4" -> 293
			"F5" -> 294; "F6" -> 295; "F7" -> 296; "F8" -> 297
			"F9" -> 298; "F10" -> 299; "F11" -> 300; "F12" -> 301
			"SHIFT" -> 340
			"CTRL", "CONTROL" -> 341
			"ALT" -> 342
			else -> throw MethodException("BAD_REQUEST", "unknown key: $name")
		}
	}

	private fun stackToJson(slot: Int, stack: ItemStack, level: net.minecraft.world.level.Level): JsonObject {
		val obj = JsonObject()
		obj.addProperty("slot", slot)
		obj.addProperty("item", net.minecraft.core.registries.BuiltInRegistries.ITEM.getKey(stack.item).toString())
		obj.addProperty("count", stack.count)

		try {
			val ops = net.minecraft.resources.RegistryOps.create(
				net.minecraft.nbt.NbtOps.INSTANCE,
				level.registryAccess()
			)
			val tag = ItemStack.CODEC.encodeStart(ops, stack).getOrThrow()
			obj.addProperty("nbt", tag.toString())
		} catch (e: Exception) {
			obj.addProperty("nbt", "")
		}

        return obj
	}

	private class BreakBlockTask(
		override val id: String,
		override val writer: BufferedWriter,
		private val pos: BlockPos,
		private val face: Direction,
		private val timeoutMs: Long?,
	) : Task {
		private val startTime = System.currentTimeMillis()
		private var done = false

		override fun tick(): Boolean {
			if (done) return true

			val mc = Minecraft.getInstance()

				return true
			}

				return true
			}

			gameMode.startDestroyBlock(pos, face)
```

```kt
object TaskManager {

    @JvmStatic

    }

    @JvmStatic
    fun cancel(id: String): Boolean {

        pushTaskDone(task.writer, id, false, "CANCELLED", null)

        for (task in toRemove) {
            tasks.remove(task.id)
            task.cancel()
        }
    }

    @JvmStatic
    fun tick() {
        val it = tasks.entries.iterator()
        while (it.hasNext()) {
            val entry = it.next()
            if (entry.value.tick()) {
                it.remove()
            }
        }
    }

    fun pushTaskDone(writer: BufferedWriter, taskId: String, ok: Boolean, code: String?, result: JsonObject?) {

            val data = JsonObject()
            data.addProperty("task", taskId)
            data.addProperty("ok", ok)

                val err = JsonObject()
                err.addProperty("code", code)
                data.add("error", err)
            }
            if (result != null) {
                data.add("result", result)
            }
            val event = JsonObject()
            event.addProperty("event", "task_done")
            event.add("data", data)
            synchronized(writer) {
                writer.write(GSON.toJson(event))
                writer.newLine()
                writer.flush()
            }
        } catch (e: Exception) {
        }
    }

    fun cancelAllOfType(predicate: (Task) -> Boolean) {
        val toRemove = tasks.values.filter(predicate)

            tasks.remove(task.id)
```

```kt
package com.xuanfeng.mcbotmod.client.function.automove

import kotlin.math.abs
import kotlin.math.sqrt

object AStar {

    private const val MAX_NODES = 12000
    private const val TIME_LIMIT_NS = 50_000_000L
    private const val MIN_PROGRESS = 2.0
    private const val EPS = 1e-9

    fun search(
        ctx: CalcContext,
        startX: Int, startY: Int, startZ: Int,
        goalX: Int, goalY: Int?, goalZ: Int,
    ): Path? {
        val startTime = System.nanoTime()

        val start = PathNode(startX, startY, startZ)
        start.cost = 0.0
        start.estimatedCostToGoal = heuristic(startX, startY, startZ, goalX, goalY, goalZ)
        start.combinedCost = start.estimatedCostToGoal

        val open = BinaryHeap()
        open.insert(start)
        val nodes = HashMap<Long, PathNode>()
        nodes[packBlockKey(startX, startY, startZ)] = start

        val result = MoveResult()
        var best = start
        var expanded = 0

        while (!open.isEmpty()) {
            if (expanded >= MAX_NODES) break
            if ((expanded and 255) == 0 && System.nanoTime() - startTime > TIME_LIMIT_NS) break
            expanded++

            val cur = open.removeLowest()
            if (isGoal(cur.x, cur.y, cur.z, goalX, goalY, goalZ)) {
                return Path.from(cur, true)
            }

            val h = cur.estimatedCostToGoal
            val bh = best.estimatedCostToGoal
            if (h < bh - EPS || (abs(h - bh) <= EPS && cur.cost < best.cost)) {
                best = cur
            }

            for (move in Moves.all) {
                result.reset()
                move.apply(ctx, cur.x, cur.y, cur.z, result)
                if (result.cost >= COST_INF) continue
                if (result.cost.isNaN() || result.cost <= 0.0) continue

                val key = packBlockKey(result.x, result.y, result.z)
                val neighbor = nodes.getOrPut(key) {
                    PathNode(result.x, result.y, result.z).also {
                        it.estimatedCostToGoal = heuristic(result.x, result.y, result.z, goalX, goalY, goalZ)
                    }
                }

                val tentative = cur.cost + result.cost
                if (tentative < neighbor.cost) {
                    neighbor.previous = cur
                    neighbor.kind = result.kind
                    neighbor.cost = tentative
                    neighbor.combinedCost = tentative + neighbor.estimatedCostToGoal
                    if (neighbor.isOpen()) {
                        open.update(neighbor)
                    } else {
                        open.insert(neighbor)
                    }
                }
            }
        }

        if (best === start) return null
        if (start.estimatedCostToGoal - best.estimatedCostToGoal < MIN_PROGRESS) return null
        return Path.from(best, false)
    }

    private fun isGoal(x: Int, y: Int, z: Int, gx: Int, gy: Int?, gz: Int): Boolean {
        if (x != gx || z != gz) return false
        if (gy != null && y != gy) return false
        return true
    }

    private fun heuristic(x: Int, y: Int, z: Int, gx: Int, gy: Int?, gz: Int): Double {
        val dx = (x - gx).toDouble()
        val dz = (z - gz).toDouble()
        var h = sqrt(dx * dx + dz * dz) * WALK_ONE
        if (gy != null) {
            val dy = gy - y
            h += if (dy > 0) dy * JUMP_ONE_BLOCK else -dy * FALL_PER_BLOCK
        }
        return h
    }
}

class BinaryHeap {
    private val arr = ArrayList<PathNode>()

    fun isEmpty(): Boolean = arr.isEmpty()
    fun size(): Int = arr.size

    fun insert(node: PathNode) {
        node.heapPosition = arr.size
        arr.add(node)
        siftUp(arr.size - 1)
    }

    fun removeLowest(): PathNode {
        val top = arr[0]
        val last = arr.removeAt(arr.size - 1)
        top.heapPosition = -1
        if (arr.isNotEmpty()) {
            arr[0] = last
            last.heapPosition = 0
            siftDown(0)
        }
        return top
    }

    fun update(node: PathNode) {
        siftUp(node.heapPosition)
    }

    private fun siftUp(i0: Int) {
        var i = i0
        while (i > 0) {
            val parent = (i - 1) / 2
            if (arr[parent].combinedCost <= arr[i].combinedCost) break
            swap(i, parent)
            i = parent
        }
    }

    private fun siftDown(i0: Int) {
        var i = i0
        while (true) {
            val l = i * 2 + 1
            val r = i * 2 + 2
            var smallest = i
            if (l < arr.size && arr[l].combinedCost < arr[smallest].combinedCost) smallest = l
            if (r < arr.size && arr[r].combinedCost < arr[smallest].combinedCost) smallest = r
            if (smallest == i) break
            swap(i, smallest)
            i = smallest
        }
    }

    private fun swap(a: Int, b: Int) {
        val t = arr[a]; arr[a] = arr[b]; arr[b] = t
        arr[a].heapPosition = a
        arr[b].heapPosition = b
    }
}
```

```kt
package com.xuanfeng.mcbotmod.client.function.automove

import net.minecraft.client.multiplayer.ClientLevel
import net.minecraft.core.BlockPos
import net.minecraft.core.Direction
import net.minecraft.tags.FluidTags
import net.minecraft.world.entity.player.Player
import net.minecraft.world.level.block.BaseFireBlock
import net.minecraft.world.level.block.Blocks
import net.minecraft.world.level.block.FallingBlock
import net.minecraft.world.level.block.state.BlockState

class CalcContext(
    private val level: ClientLevel,
    private val player: Player,
    private val originX: Int,
    private val originY: Int,
    private val originZ: Int,
    val radius: Int,
    val allowBreak: Boolean,
    val allowPlace: Boolean,
) {
    private val states = HashMap<Long, BlockState>()
    private val infos = HashMap<Long, Int>()
    private val airState = Blocks.AIR.defaultBlockState()
    private val radiusSq = radius * radius

    val canPlace: Boolean = allowPlace && Hotbar.findPlaceSlot(player) >= 0

    fun isInBounds(x: Int, y: Int, z: Int): Boolean {
        if (y < level.minY || y > level.maxY) return false
        if (kotlin.math.abs(y - originY) > radius) return false
        val dx = x - originX
        val dz = z - originZ
        return dx * dx + dz * dz <= radiusSq
    }

    fun getBlock(x: Int, y: Int, z: Int): BlockState {
        if (!isInBounds(x, y, z)) return airState
        val key = packBlockKey(x, y, z)
        val cached = states[key]
        if (cached != null) return cached
        val pos = BlockPos(x, y, z)
        val state = if (level.isLoaded(pos)) level.getBlockState(pos) else airState
        states[key] = state
        return state
    }

    private fun infoOf(x: Int, y: Int, z: Int): Int {
        if (!isInBounds(x, y, z)) return 0
        val key = packBlockKey(x, y, z)
        val cached = infos[key]
        if (cached != null) return cached
        val pos = BlockPos(x, y, z)
        val info = if (level.isLoaded(pos)) computeInfo(getBlock(x, y, z), pos) else 0
        infos[key] = info
        return info
    }

    private fun computeInfo(state: BlockState, pos: BlockPos): Int {
        val shape = state.getCollisionShape(level, pos)
        val solid = !shape.isEmpty
        val hazard = isHazard(state)
        var info = 0
        if (!solid && !hazard) info = info or PASS
        if (solid) {
            info = info or SOLID
            val top = shape.max(Direction.Axis.Y)
            if (!hazard && top > 0.8 && top <= 1.0) info = info or FLOOR
        }
        if (state.fluidState.`is`(FluidTags.WATER)) info = info or WATER
        return info
    }

    private fun isHazard(state: BlockState): Boolean {
        if (state.fluidState.`is`(FluidTags.LAVA)) return true
        val block = state.block
        if (block is BaseFireBlock) return true
        return block in HAZARDS
    }

    fun isPassable(x: Int, y: Int, z: Int): Boolean = (infoOf(x, y, z) and PASS) != 0

    fun isSolid(x: Int, y: Int, z: Int): Boolean = (infoOf(x, y, z) and SOLID) != 0

    fun isFloor(x: Int, y: Int, z: Int): Boolean = (infoOf(x, y, z) and FLOOR) != 0

    fun wadePenalty(x: Int, y: Int, z: Int): Double =
        if ((infoOf(x, y, z) and WATER) != 0) WADE_PENALTY else 0.0

    fun canStandAt(x: Int, y: Int, z: Int): Boolean {
        if (!isFloor(x, y - 1, z)) return false
        if (!isPassable(x, y, z)) return false
        if (!isPassable(x, y + 1, z)) return false
        return true
    }

    fun findLandingY(x: Int, startY: Int, z: Int, maxFall: Int): Int? {
        if (!isPassable(x, startY, z) || !isPassable(x, startY + 1, z)) return null
        val lowest = maxOf(level.minY + 1, startY - maxFall)
        var ny = startY - 1
        while (ny >= lowest) {
            if (!isPassable(x, ny, z)) return null
            if (isFloor(x, ny - 1, z)) return ny
            ny--
        }
        return null
    }

    fun breakCost(x: Int, y: Int, z: Int): Double {
        val info = infoOf(x, y, z)
        if ((info and PASS) != 0) return 0.0
        if ((info and SOLID) == 0) return COST_INF
        val state = getBlock(x, y, z)
        val hardness = state.getDestroySpeed(level, BlockPos(x, y, z))
        if (hardness < 0f) return COST_INF
        if (getBlock(x, y + 1, z).block is FallingBlock) return COST_INF
        for (d in FLUID_CHECK) {
            if (!getBlock(x + d[0], y + d[1], z + d[2]).fluidState.isEmpty) return COST_INF
        }
        val tool = Hotbar.bestTool(player, state)
        val divisor = if (state.requiresCorrectToolForDrops() && !tool.correct) 100.0 else 30.0
        val ticks = hardness * divisor / tool.speed
        if (ticks > MAX_BREAK_TICKS) return COST_INF
        return BREAK_BASE + ticks
    }

    companion object {
        private const val PASS = 1
        private const val SOLID = 2
        private const val FLOOR = 4
        private const val WATER = 8
        private const val MAX_BREAK_TICKS = 600.0

        private val FLUID_CHECK = arrayOf(
            intArrayOf(0, 1, 0),
            intArrayOf(1, 0, 0),
            intArrayOf(-1, 0, 0),
            intArrayOf(0, 0, 1),
            intArrayOf(0, 0, -1),
        )

        private val HAZARDS = setOf(
            Blocks.CACTUS,
```

```kt
    private val allowBreak: Boolean,
```

```kt
package com.xuanfeng.mcbotmod.client.function.automove

const val COST_INF = Double.MAX_VALUE

const val WALK_ONE = 4.63
const val JUMP_ONE_BLOCK = 2.0
const val FALL_PER_BLOCK = 0.5
const val BREAK_BASE = 6.0
const val PLACE_COST = 6.0
const val DIAGONAL_MULT = 1.4142135623730951
const val WADE_PENALTY = 3.0
const val MAX_FALL = 3

enum class MoveKind { WALK, DIAGONAL, ASCEND, DESCEND, PARKOUR, PILLAR, BREAK, PLACE, DIG_DOWN }

class MoveResult {
    var x = 0
    var y = 0
    var z = 0
    var cost = COST_INF
    var kind = MoveKind.WALK

    fun reset() { cost = COST_INF }

    fun set(nx: Int, ny: Int, nz: Int, c: Double, k: MoveKind) {
        x = nx; y = ny; z = nz
        cost = c
        kind = k
    }
}

interface Move {
    fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult)
}

object Moves {
    private val cardinals = listOf(0 to -1, 0 to 1, 1 to 0, -1 to 0)
    private val diagonals = listOf(1 to -1, 1 to 1, -1 to -1, -1 to 1)

    val all: List<Move> = buildList {
        for ((dx, dz) in cardinals) add(Traverse(dx, dz))
        for ((dx, dz) in diagonals) add(Diagonal(dx, dz))
        for ((dx, dz) in cardinals) add(Ascend(dx, dz))
        for ((dx, dz) in cardinals) add(Descend(dx, dz))
        for ((dx, dz) in cardinals) for (dist in 2..4) add(Parkour(dx, dz, dist))
        add(Pillar())
        for ((dx, dz) in cardinals) add(BreakThrough(dx, dz))
        for ((dx, dz) in cardinals) add(PlaceAhead(dx, dz))
        add(DigDown())
    }
}

private class Traverse(val dx: Int, val dz: Int) : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        val nx = x + dx
        val nz = z + dz
        if (!ctx.canStandAt(nx, y, nz)) return
        result.set(nx, y, nz, WALK_ONE + ctx.wadePenalty(nx, y, nz), MoveKind.WALK)
    }
}

private class Diagonal(val dx: Int, val dz: Int) : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        val nx = x + dx
        val nz = z + dz
        if (!ctx.canStandAt(nx, y, nz)) return
        if (!ctx.isPassable(nx, y, z) || !ctx.isPassable(nx, y + 1, z)) return
        if (!ctx.isPassable(x, y, nz) || !ctx.isPassable(x, y + 1, nz)) return
        result.set(nx, y, nz, WALK_ONE * DIAGONAL_MULT + ctx.wadePenalty(nx, y, nz), MoveKind.DIAGONAL)
    }
}

private class Ascend(val dx: Int, val dz: Int) : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        val nx = x + dx
        val nz = z + dz
        val ny = y + 1
        if (!ctx.canStandAt(nx, ny, nz)) return
        if (!ctx.isPassable(x, y + 2, z)) return
        result.set(nx, ny, nz, WALK_ONE + JUMP_ONE_BLOCK + ctx.wadePenalty(nx, ny, nz), MoveKind.ASCEND)
    }
}

private class Descend(val dx: Int, val dz: Int) : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        val nx = x + dx
        val nz = z + dz
        val landingY = ctx.findLandingY(nx, y, nz, MAX_FALL) ?: return
        val fall = y - landingY
        if (fall <= 0) return
        result.set(
            nx, landingY, nz,
            WALK_ONE + fall * FALL_PER_BLOCK + ctx.wadePenalty(nx, landingY, nz),
            MoveKind.DESCEND,
        )
    }
}

private class Parkour(val dx: Int, val dz: Int, val dist: Int) : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        val nx = x + dx * dist
        val nz = z + dz * dist
        if (!ctx.canStandAt(nx, y, nz)) return
        if (!ctx.isPassable(nx, y + 2, nz)) return
        if (!ctx.isPassable(x, y + 2, z)) return
        if (dist == 4 && !ctx.canStandAt(x - dx, y, z - dz)) return
        for (i in 1 until dist) {
            val mx = x + dx * i
            val mz = z + dz * i
            if (!ctx.isPassable(mx, y, mz)) return
            if (!ctx.isPassable(mx, y + 1, mz)) return
            if (!ctx.isPassable(mx, y + 2, mz)) return
        }
        result.set(nx, y, nz, WALK_ONE * dist + JUMP_ONE_BLOCK + 1.0, MoveKind.PARKOUR)
    }
}

private class Pillar : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        if (!ctx.canPlace) return
        if (!ctx.isPassable(x, y + 1, z)) return
        if (!ctx.isPassable(x, y + 2, z)) return
        result.set(x, y + 1, z, PLACE_COST + JUMP_ONE_BLOCK, MoveKind.PILLAR)
    }
}

private class BreakThrough(val dx: Int, val dz: Int) : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        if (!ctx.allowBreak) return
        val nx = x + dx
        val nz = z + dz
        if (!ctx.isFloor(nx, y - 1, nz)) return
        val feet = ctx.breakCost(nx, y, nz)
        val head = ctx.breakCost(nx, y + 1, nz)
        if (feet >= COST_INF || head >= COST_INF) return
        if (feet == 0.0 && head == 0.0) return
        result.set(nx, y, nz, WALK_ONE + feet + head, MoveKind.BREAK)
    }
}

private class PlaceAhead(val dx: Int, val dz: Int) : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        if (!ctx.canPlace) return
        val nx = x + dx
        val nz = z + dz
        if (!ctx.isFloor(x, y - 1, z)) return
        if (!ctx.isPassable(nx, y - 1, nz)) return
        if (!ctx.isPassable(nx, y, nz)) return
        if (!ctx.isPassable(nx, y + 1, nz)) return
        result.set(nx, y, nz, PLACE_COST + WALK_ONE, MoveKind.PLACE)
    }
}

private class DigDown : Move {
    override fun apply(ctx: CalcContext, x: Int, y: Int, z: Int, result: MoveResult) {
        if (!ctx.allowBreak) return
        if (!ctx.isFloor(x, y - 1, z)) return
        val breakCost = ctx.breakCost(x, y - 1, z)
        if (breakCost >= COST_INF || breakCost == 0.0) return
        val lowest = y - MAX_FALL
        var ny = y - 1
        while (!ctx.isFloor(x, ny - 1, z)) {
            ny--
            if (ny < lowest) return
            if (!ctx.isPassable(x, ny, z)) return
        }
        val fall = y - ny
        result.set(
            x, ny, z,
            breakCost + fall * FALL_PER_BLOCK + ctx.wadePenalty(x, ny, z),
            MoveKind.DIG_DOWN,
        )
    }
}
```

```kt
class Path(
    val nodes: List<PathNode>,
    val cost: Double,
    val complete: Boolean = true,
) {
    val length: Int get() = nodes.size

    fun first(): PathNode? = nodes.firstOrNull()
    fun last(): PathNode? = nodes.lastOrNull()
    operator fun get(index: Int): PathNode = nodes[index]

            val list = ArrayList<PathNode>()
            var cur: PathNode? = endNode

                list.add(cur)
                cur = cur.previous
            }
            list.reverse()
            return Path(list, endNode.cost, complete)
        }
```

```kt
/*

 *

package com.xuanfeng.mcbotmod.client.function.automove

import net.minecraft.client.Minecraft
import net.minecraft.client.multiplayer.ClientLevel
import net.minecraft.core.BlockPos
import net.minecraft.core.Direction
import net.minecraft.world.InteractionHand
import net.minecraft.world.entity.player.Player
import net.minecraft.world.phys.BlockHitResult
import net.minecraft.world.phys.Vec3
import kotlin.math.abs
import kotlin.math.atan2
import kotlin.math.sign
import kotlin.math.sqrt

class PathExecutor(
    private val mc: Minecraft,
    private val targetX: Int,
    private val targetY: Int?,
    private val targetZ: Int,
    private val allowBreak: Boolean,
    private val allowPlace: Boolean,
) {

    enum class Status { RUNNING, SUCCESS, FAILED, TIMEOUT, STUCK }

    private var path: Path? = null
    private var nodeIndex = 1
    private var ticksOnNode = 0
    private var ticksTotal = 0
    private var airTicks = 0
    private var replanStreak = 0
    private var placeCooldown = 0
    private var breaking = false

    @Volatile
    private var cancelled = false

    fun cancel() {
        cancelled = true
        mc.execute { stopAll() }
    }

    fun tick(): Status {
        if (cancelled) {
            stopAll()
            return Status.FAILED
        }
        val player = mc.player
        val level = mc.level
        if (player == null || level == null) {
            stopAll()
            return Status.FAILED
        }

        if (++ticksTotal > MAX_TOTAL_TICKS) {
            stopAll()
            return Status.TIMEOUT
        }
        if (placeCooldown > 0) placeCooldown--

        val grounded = player.onGround() || player.isInWater
        if (grounded && isAtGoal(player.blockX, player.blockY, player.blockZ)) {
            stopAll()
            return Status.SUCCESS
        }

        var p = path
        if (p == null || nodeIndex >= p.length) {
            if (!grounded) {
                releaseMovement()
                if (++airTicks > MAX_AIR_TICKS) {
                    stopAll()
                    return Status.STUCK
                }
                return Status.RUNNING
            }
            airTicks = 0
            val planned = plan(level, player)
            if (planned == null || planned.length < 2) {
                stopAll()
                return Status.FAILED
            }
            path = planned
            p = planned
            nodeIndex = 1
            ticksOnNode = 0
        }
        val route = p ?: return Status.FAILED

        val bx = player.blockX
        val by = player.blockY
        val bz = player.blockZ
        if (grounded) {
            val limit = minOf(nodeIndex + 3, route.length - 1)
            for (i in limit downTo nodeIndex) {
                val n = route[i]
                if (n.x == bx && n.y == by && n.z == bz) {
                    nodeIndex = i + 1
                    ticksOnNode = 0
                    replanStreak = 0
                    stopBreaking()
                    return Status.RUNNING
                }
            }
        }

        val node = route[nodeIndex]
        val prev = route[nodeIndex - 1]

        val dx = node.x + 0.5 - player.x
        val dz = node.z + 0.5 - player.z
        if (grounded && (dx * dx + dz * dz > FAR_SQ || abs(node.y - player.y) > FAR_Y)) {
            return invalidate()
        }
        if (!isNodeValid(level, node)) return invalidate()

        if (node.kind == MoveKind.DIG_DOWN && digDownStep(player, level, prev)) {
            return Status.RUNNING
        }

        if (node.kind == MoveKind.BREAK && breakStep(player, level, prev, node)) {
            return Status.RUNNING
        }
        stopBreaking()

        if (++ticksOnNode > MAX_TICKS_ON_NODE) return invalidate()

        when (node.kind) {
            MoveKind.PLACE -> {
                val floor = BlockPos(node.x, node.y - 1, node.z)
                if (!hasCollision(level, floor)) {
                    if (Hotbar.findPlaceSlot(player) < 0) return invalidate()
                    releaseMovement()
                    placeBridge(player, prev, node)
                    return Status.RUNNING
                }
            }
            MoveKind.PILLAR -> {
                if (Hotbar.findPlaceSlot(player) < 0) return invalidate()
                pillar(player, level, prev, grounded)
                return Status.RUNNING
            }
            else -> {}
        }

        if (node.kind == MoveKind.DIG_DOWN) {
            releaseMovement()
            return Status.RUNNING
        }

        walk(route, nodeIndex, player, grounded)
        return Status.RUNNING
    }

    private fun isAtGoal(x: Int, y: Int, z: Int): Boolean {
        if (x != targetX || z != targetZ) return false
        if (targetY != null && y != targetY) return false
        return true
    }

    private fun plan(level: ClientLevel, player: Player): Path? {
        val ctx = CalcContext(
            level, player,
            player.blockX, player.blockY, player.blockZ,
            RADIUS, allowBreak, allowPlace,
        )
        return AStar.search(ctx, player.blockX, player.blockY, player.blockZ, targetX, targetY, targetZ)
    }

    private fun invalidate(): Status {
        stopAll()
        path = null
        ticksOnNode = 0
        if (++replanStreak > MAX_REPLANS) return Status.STUCK
        return Status.RUNNING
    }

    private fun hasCollision(level: ClientLevel, pos: BlockPos): Boolean {
        return !level.getBlockState(pos).getCollisionShape(level, pos).isEmpty
    }

    private fun isNodeValid(level: ClientLevel, node: PathNode): Boolean {
        if (node.kind == MoveKind.BREAK || node.kind == MoveKind.DIG_DOWN) return true
        val feet = BlockPos(node.x, node.y, node.z)
        return !hasCollision(level, feet) && !hasCollision(level, feet.above())
    }

    private fun dirOf(dx: Int, dz: Int): Direction = when {
        dx > 0 -> Direction.EAST
        dx < 0 -> Direction.WEST
        dz > 0 -> Direction.SOUTH
        else -> Direction.NORTH
    }

    private fun lookAt(player: Player, target: Vec3) {
        val eye = player.eyePosition
        val dx = target.x - eye.x
        val dy = target.y - eye.y
        val dz = target.z - eye.z
        val yaw = Math.toDegrees(atan2(-dx, dz)).toFloat()
        val pitch = Math.toDegrees(-atan2(dy, sqrt(dx * dx + dz * dz))).toFloat()
        player.yRot = yaw
        player.xRot = pitch
    }

    private fun breakStep(player: Player, level: ClientLevel, prev: PathNode, node: PathNode): Boolean {
        val face = dirOf(node.x - prev.x, node.z - prev.z).opposite
        for (y in intArrayOf(node.y + 1, node.y)) {
            val pos = BlockPos(node.x, y, node.z)
            if (!hasCollision(level, pos)) continue
            releaseMovement()
            val state = level.getBlockState(pos)
            val tool = Hotbar.bestTool(player, state)
            Hotbar.select(mc, tool.slot)
            lookAt(player, Vec3.atCenterOf(pos))
            mc.gameMode?.continueDestroyBlock(pos, face)
            player.swing(InteractionHand.MAIN_HAND)
            breaking = true
            return true
        }
        return false
    }

    private fun digDownStep(player: Player, level: ClientLevel, prev: PathNode): Boolean {
        val pos = BlockPos(prev.x, prev.y - 1, prev.z)
        if (!hasCollision(level, pos)) return false

        val ox = prev.x + 0.5 - player.x
        val oz = prev.z + 0.5 - player.z
        if (ox * ox + oz * oz > 0.04) {
            if (ticksOnNode++ > MAX_TICKS_ON_NODE) return false
            val opts = mc.options
            player.yRot = Math.toDegrees(atan2(-ox, oz)).toFloat()
            player.xRot = 0f
            opts.keyUp.isDown = true
            opts.keySprint.isDown = false
            opts.keyJump.isDown = false
            return true
        }

        releaseMovement()
        val state = level.getBlockState(pos)
        val tool = Hotbar.bestTool(player, state)
        Hotbar.select(mc, tool.slot)
        lookAt(player, Vec3.atCenterOf(pos))
        mc.gameMode?.continueDestroyBlock(pos, Direction.UP)
        player.swing(InteractionHand.MAIN_HAND)
        breaking = true
        return true
    }

    private fun placeAgainst(player: Player, support: BlockPos, face: Direction, hit: Vec3) {
        if (placeCooldown > 0) return
        val slot = Hotbar.findPlaceSlot(player)
        if (slot < 0) return
        Hotbar.select(mc, slot)
        lookAt(player, hit)
        val lp = mc.player ?: return
        mc.gameMode?.useItemOn(lp, InteractionHand.MAIN_HAND, BlockHitResult(hit, face, support, false))
        player.swing(InteractionHand.MAIN_HAND)
        placeCooldown = PLACE_COOLDOWN_TICKS
    }

    private fun placeBridge(player: Player, prev: PathNode, node: PathNode) {
        val face = dirOf(node.x - prev.x, node.z - prev.z)
        val support = BlockPos(prev.x, prev.y - 1, prev.z)
        val hit = Vec3(
            support.x + 0.5 + face.stepX * 0.5,
            support.y + 0.5,
            support.z + 0.5 + face.stepZ * 0.5,
        )
        placeAgainst(player, support, face, hit)
    }

    private fun pillar(player: Player, level: ClientLevel, prev: PathNode, grounded: Boolean) {
        val opts = mc.options
        val cx = prev.x + 0.5
        val cz = prev.z + 0.5
        val cell = BlockPos(prev.x, prev.y, prev.z)

        opts.keySprint.isDown = false
        if (hasCollision(level, cell)) {
            opts.keyUp.isDown = false
            opts.keyJump.isDown = false
            return
        }

        val ox = cx - player.x
        val oz = cz - player.z
        if (grounded && ox * ox + oz * oz > 0.0625) {
            player.yRot = Math.toDegrees(atan2(-ox, oz)).toFloat()
            player.xRot = 0f
            opts.keyUp.isDown = true
            opts.keyJump.isDown = false
            return
        }

        opts.keyUp.isDown = false
        opts.keyJump.isDown = true
        val support = BlockPos(prev.x, prev.y - 1, prev.z)
        val hit = Vec3(cx, prev.y.toDouble(), cz)
        if (player.y > prev.y + 1.0) {
            placeAgainst(player, support, Direction.UP, hit)
        } else {
            lookAt(player, hit)
        }
    }

    private fun shouldSprint(p: Path, idx: Int): Boolean {
        val node = p[idx]
        if (node.kind == MoveKind.PARKOUR) return true
        if (idx + 1 >= p.length) return false
        val prev = p[idx - 1]
        val next = p[idx + 1]
        if (next.kind == MoveKind.PARKOUR) return true
        if (node.kind != MoveKind.WALK && node.kind != MoveKind.DIAGONAL) return false
        if (next.y != node.y) return false
        return next.x - node.x == node.x - prev.x && next.z - node.z == node.z - prev.z
    }

    private fun walk(p: Path, idx: Int, player: Player, grounded: Boolean) {
        val node = p[idx]
        val prev = p[idx - 1]
        val opts = mc.options

        val dx = node.x + 0.5 - player.x
        val dz = node.z + 0.5 - player.z
        player.yRot = Math.toDegrees(atan2(-dx, dz)).toFloat()
        player.xRot = 0f

        opts.keyUp.isDown = true
        opts.keySprint.isDown = shouldSprint(p, idx)

        var jump = when (node.kind) {
            MoveKind.ASCEND -> node.y > player.blockY
            MoveKind.PARKOUR -> {
                val sx = (node.x - prev.x).sign
                val sz = (node.z - prev.z).sign
                val progress = (player.x - (prev.x + 0.5)) * sx + (player.z - (prev.z + 0.5)) * sz
                grounded && progress >= PARKOUR_JUMP_PROGRESS
            }
            else -> false
        }
        if (player.isInWater && node.y >= player.blockY) jump = true
        opts.keyJump.isDown = jump
    }

    private fun stopBreaking() {
        if (breaking) {
            mc.gameMode?.stopDestroyBlock()
            breaking = false
        }
    }

    private fun releaseMovement() {
        val opts = mc.options
        opts.keyUp.isDown = false
        opts.keySprint.isDown = false
        opts.keyJump.isDown = false
    }

    private fun stopAll() {
        releaseMovement()
        stopBreaking()
    }

    companion object {
        const val RADIUS = 24
        const val MAX_TICKS_ON_NODE = 60
        const val MAX_TOTAL_TICKS = 20 * 60 * 5
        const val MAX_REPLANS = 6
        const val MAX_AIR_TICKS = 100
        const val FAR_SQ = 49.0
        const val FAR_Y = 6.0
        const val PLACE_COOLDOWN_TICKS = 3
        const val PARKOUR_JUMP_PROGRESS = 0.3
    }
}
```

```kt
fun packBlockKey(x: Int, y: Int, z: Int): Long {
    return (x.toLong() shl 40) or
            ((z.toLong() and 0xFFFFFF) shl 16) or
            (y.toLong() and 0xFFFF)

class PathNode(
    val x: Int,
    val y: Int,
    val z: Int,
) {
    var cost: Double = Double.MAX_VALUE
    var estimatedCostToGoal: Double = 0.0
    var combinedCost: Double = Double.MAX_VALUE
    var previous: PathNode? = null
    var heapPosition: Int = -1
    var kind: MoveKind = MoveKind.WALK

    fun isOpen(): Boolean = heapPosition != -1

    override fun hashCode(): Int {
        var h = x
        h = h * 31 + y
        h = h * 31 + z
        return h
    }

    override fun equals(other: Any?): Boolean {
        if (this === other) return true
        if (other !is PathNode) return false
        return x == other.x && y == other.y && z == other.z
    }
}
```
