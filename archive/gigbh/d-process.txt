`d-process`:

```
#!/bin/sh
set -eu

SOURCE="/usr/local/share/d-process/main.c"
if [ -n "${HOME:-}" ]; then
 BASE="$HOME/.local/share/d-process"
else
 BASE="/tmp/d-process-$(id -u)"
fi
BIN_DIR="$BASE/bin"
RUN_DIR="$BASE/run"

usage() {
 echo "Usage: d-process <name>"
 echo "       d-process killall"
}

if [ "$#" -ne 1 ]; then
 usage
 exit 1
fi

cmd="$1"

if [ -z "$cmd" ]; then
 usage
 exit 1
fi

if [ "$cmd" = "killall" ]; then
 mkdir -p "$RUN_DIR"
 for piddir in /proc/[0-9]*; do
  [ -d "$piddir" ] || continue
  pid="${piddir#/proc/}"
  [ "$pid" = "$$" ] && continue
  [ -r "$piddir/cmdline" ] || continue
  args=$(tr '\0' ' ' < "$piddir/cmdline" 2>/dev/null || true)
  case "$args" in
   "$BIN_DIR/"*)
    kill "$pid" 2>/dev/null || true
    ;;
  esac
 done
 rm -f "$RUN_DIR"/*.pid 2>/dev/null || true
 exit 0
fi

if ! command -v gcc >/dev/null 2>&1; then
 echo "gcc not found" >&2
 exit 1
fi

if [ ! -f "$SOURCE" ]; then
 echo "source not found" >&2
 exit 1
fi

name="${cmd##*/}"
safe=$(printf '%s' "$name" | tr -c 'A-Za-z0-9._-' '_')

if [ -z "$safe" ] || [ "$safe" = "." ] || [ "$safe" = ".." ]; then
 echo "invalid process name" >&2
 exit 1
fi

mkdir -p "$BIN_DIR" "$RUN_DIR"
binary="$BIN_DIR/$safe"
tmp="$BIN_DIR/.$safe.$$.tmp"
gcc "$SOURCE" -o "$tmp"
mv -f "$tmp" "$binary"
nohup "$binary" >/dev/null 2>&1 &
pid=$!
printf '%s\n%s\n%s\n' "$pid" "$binary" "$name" > "$RUN_DIR/$safe.$pid.pid"
echo "$pid"

```

`install.sh`:

```sh
#!/bin/sh
set -eu

if [ "$(id -u)" -ne 0 ]; then
 echo "Run as root" >&2
 exit 1
fi

if [ ! -f main.c ] || [ ! -f d-process ]; then
 echo "Project files not found" >&2
 exit 1
fi

mkdir -p /usr/local/share/d-process /usr/local/bin
cp main.c /usr/local/share/d-process/main.c
cp d-process /usr/local/bin/d-process
chmod 755 /usr/local/bin/d-process
chmod 644 /usr/local/share/d-process/main.c

echo "Decoy-Process installed"
echo ""
echo "Commands:"
echo "  d-process <name>    create and run fake process"
echo "  d-process killall   kill all fake processes"
echo ""
echo "Example:"
echo "  d-process Steam.exe"

```
