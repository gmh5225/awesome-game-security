ENV PYTHONUNBUFFERED=1
# Run as an unprivileged user that owns only the data directory

EXPOSE 20022

HEALTHCHECK --interval=30s --timeout=5s --start-period=20s CMD wget -qO- http://127.0.0.1:20022/healthz || exit 1

USER app

# Rate limits, presence and the session cache live in process memory, so run a single gevent worker
CMD ["gunicorn", "--worker-class", "gevent", "--workers", "1", "--worker-connections", "1000", "--graceful-timeout", "5", "--timeout", "60", "--bind", "0.0.0.0:20022", "app:app"]

A retro maze chase arcade game for the browser. Eat every dot, grab a power pill to turn the tables, and stay away from the Static Gang. It looks and sounds like an early 80s cabinet *(low resolution pixels, CRT glow, chiptune effects)* but the hero, the chasers, the mazes, the bonus items and every sound are original.

Every score on the leaderboard is verified: the browser records the input of every tick, and the server replays that log through the exact same simulation code before it stores anything.

## Features

- Eleven modes, each with its own all time and today leaderboard
- Every score is replayed on the server before it is stored, so a tampered run can't get on the board
- Fits any screen with no scrolling: desktop, laptop, tablet, and phones held either way
- Phones get a floating joystick that appears wherever your thumb lands, and coin, start, menu and sound buttons. The CRT shader starts off there *(it is slow on phone GPUs)* and can be turned back on
- MENU or Esc pauses a run, and pressing it again quits back to the title without submitting
- Installs to the home screen as an app
- Optional acid-arcade accounts: signed in players are ranked under their account name and their best scores show on the arcade. Guests still play and enter 3 initials

## The cast

Our hero is **BYTE**, a little cube critter with a box jaw and an antenna.

The Static Gang:

| Name  | Personality   | How it hunts                                                     |
|-------|---------------|------------------------------------------------------------------|
| RAZOR | the hunter    | heads straight for BYTE, and speeds up as the maze empties       |
| VEX   | the cutter    | aims four tiles ahead of where BYTE is going                     |
| MURK  | the pincer    | moves to the opposite side of BYTE from the nearest other chaser |
| FIZZ  | the prankster | wanders to random spots, then pounces when BYTE gets close       |

They take turns between scatter and chase phases, and a power pill makes them vulnerable for a while *(less time on later levels)*.

## Modes

Each mode has its own leaderboard, all time and today.

- **Classic**: one maze, the standard rules and difficulty curve.
- **Maze Pack**: four hand built mazes that rotate every level.
- **Shifting**: wall gates slide open and shut on a timer *(they flash before they move)*.
- **Procedural**: a new symmetric maze generated from the run seed every level, always fully connected and never a dead end.
- **Mayhem**: power-ups and new chaser types join the gang.
- **Darkness**: you only see a small light around BYTE, the gang glows faintly.
- **Escape**: a labyrinth several screens big that scrolls with BYTE and grows every level. Find the exit while the gang patrols and hunts you when you get close. Reaching the exit scores 1,000 per level plus 20 for every second left on the clock, and dots and pills are scattered along the way.
- **Survival**: the classic maze with endless dots that refill when you clear them. Another chaser from the whole cast joins every 30 seconds up to ten, and everyone speeds up a little each minute. Every second survived is worth 10 points.
- **Hunter**: the roles swap. You play one of the gang and four BYTEs eat the dots and run from you. Catch all four before they clear the maze *(200, 400, 800, 1,600, plus a bonus for speed)*. The BYTEs path find: two gather opposite halves of the maze, one runs for a power pill when you get close to the team, one is bold and feeds right under your nose, and all of them flee toward where you can't cut them off. A power pill turns them all red and they hunt you together, two straight at you and two cutting you off. If they catch you it costs a life and your eyes race back to the house while the BYTEs keep going; back out of the house you're safe from that pill. Each member has its own ability on START: RAZOR is the fastest, VEX dashes 4 tiles ahead, MURK drops a wall block behind it for 5 seconds and FIZZ teleports.
- **Daily**: one seed per day, the same for everyone, with a random mix of twists.
- **Custom**: stack the twists you want *(pack or procedural mazes, shifting, mayhem, darkness, escape)*. Escape replaces the maze and can't be combined with shifting.

Mayhem power-ups: speed boots, freeze, dot magnet, phase *(walk through walls)*, decoy, shield, bomb, double points and teleport.

Mayhem chasers: SHARD *(splits in two when eaten)*, WARP *(teleports near you)*, PHAZE *(walks through walls)*, ECHO *(copies your moves)*, DOZER *(sleeps until you walk past)*, LURK *(ambushes far ahead and sprints when close)* and MUNCH *(eats your dots and steals score)*.

Also in the game: tunnels, bonus items *(chip, battery, floppy, cassette, joystick, gem, rocket, key)*, an extra life at 10,000, faster levels, three intermissions, attract mode with the cast introduction, a demo and the high score tables, and arcade style 3 letter initials.

## Controls

| Action       | Keys                                                  |
|--------------|-------------------------------------------------------|
| Move         | arrows or WASD *(turns are buffered, so press early)* |
| Coin / start | C, Enter or Space *(the gang member's ability in Hunter)* |
| Pause        | P or Esc                                              |
| Quit to menu | Esc or MENU while paused                              |
| Sound        | M                                                     |
| CRT effect   | V *(curvature has its own button)*                    |

On a phone, put your thumb down anywhere and slide to steer: the joystick follows your thumb, so you can roll straight into a new direction. A quick flick works too, and a tap on the screen starts a game.

## How it works

- `static/core.js` is the whole game simulation. It runs on a fixed 60 Hz tick, uses integer fixed point math and a seeded PRNG, and never touches `Date` or `Math.random`, so the same seed and inputs always give the same game.
- The browser loads the core plus `font.js` *(hand drawn bitmap font)*, `render.js` *(224x288 canvas with a WebGL CRT pass)*, `audio.js` *(Web Audio synth)*, `input.js`, `page.js` *(fits the cabinet to the screen, arcade account button)* and `main.js`.
- `POST /api/run` returns a server chosen seed and a single use run id. When the game ends the browser posts the run length encoded input log to `/api/run/<id>/submit`. The server replays it with QuickJS *(`replay.js`, run as a subprocess with time and memory limits)* and stores the score the replay produced. Unknown ids, reused ids, oversized logs, logs that stop before game over, and logs longer than the time since the run started are all refused.
- Scores live in SQLite at `data/maze.db` *(`MAZE_DB`)*. Starting runs and submitting them are limited per address, and runs that never finish are pruned after 4 hours.

## Running with Docker

```sh
./setup.sh
```

This builds the `acidmaze` image *(Alpine with the `quickjs` package for the replay)* and starts a container bound to `127.0.0.1:20022` with the `acidmaze_data` volume mounted at `/data`.

| Variable          | Default        | What it does                                                                                                                                  |
|-------------------|----------------|-----------------------------------------------------------------------------------------------------------------------------------------------|
| `MAZE_BIND`       | `127.0.0.1`    | Address the port is published on                                                                                                              |
| `MAZE_PROXIES`    | `0`            | Number of reverse proxies in front of the app, so the per address limits see the real client. `setup.sh` sets `1` for nginx                   |
| `MAZE_PUBLIC_URL` |                | Main address of the site, e.g. `https://maze.acid.vegas`. Canonical links, share cards, robots.txt and the sitemap use it instead of the host |
| `MAZE_DB`         | `data/maze.db` | Path of the scores database *(`/data/maze.db` in the container)*                                                                              |
| `MAZE_QJS`        | `/usr/bin/qjs` | Path of the QuickJS binary used for the replay                                                                                                |

Don't set `MAZE_PROXIES` when the app is reached directly, since anyone could then fake their address with an `X-Forwarded-For` header.

## acid-arcade-db

Without `ARCADE_URL` the game has no accounts and every score is saved under 3 initials. Setting it connects the game to acid-arcade-db. Players signed in on any acid-arcade site are signed in here too *(the shared `arcade_sso` cookie)*, the page header gets a sign in button and an arcade link, and their runs go on the boards under their account name. Guest runs still go on the same boards with initials, shown in grey and tagged as guests. After each account run the game sends the player's best score in any mode *(the arcade leaderboard number)*, their best in each mode, games played and highest level to their arcade profile. Each account run is also reported as a `run` activity with the score the server replay produced, which earns casino chips *(score / 50 dollars, under the arcade's daily caps)* and counts toward the arcade's daily missions. Renames and deleted accounts are picked up every minute, and deleting an account removes its runs.

| Variable            | Default                   | What it does                                                                                       |
|---------------------|---------------------------|----------------------------------------------------------------------------------------------------|
| `ARCADE_URL`        |                           | Internal acid-arcade-db address, e.g. `http://acid-arcade-db:20021`                                |
| `ARCADE_PUBLIC_URL` | `https://arcade.{domain}` | Public site, for sign in, profile pictures and account settings *(`{domain}` is the player's parent domain)* |
| `ARCADE_GAME`       |                           | This game's slug in acid-arcade-db. `setup.sh` sets it to `maze`                                   |
| `ARCADE_KEY`        |                           | This game's key, needed when acid-arcade-db has `ARCADE_GAME_KEYS` set                             |

```sh
ARCADE_URL=http://acid-arcade-db:20021 ARCADE_KEY=... MAZE_PUBLIC_URL=https://maze.acid.vegas ./setup.sh
```

## Running without Docker

You need Python 3.12+ and QuickJS *(`qjs`, the Debian/Ubuntu and Alpine package is `quickjs`)*.

```sh
python3 -m venv .venv
.venv/bin/pip install -r requirements.txt
./run.sh
```

`run.sh` (re)starts gunicorn detached on `127.0.0.1:20004`, writing its log to `data/server.log`. Set `MAZE_HOST` and `MAZE_PORT` to bind somewhere else. For a quick local run, `.venv/bin/python app.py` serves on `127.0.0.1:20004`.

## Tests

```sh
qjs --std tests/test_core.js
.venv/bin/python -m unittest discover -s tests -v
```

The core test validates every maze *(symmetric, connected, no dead ends)*, 60 procedural mazes, and checks that every mode is deterministic. The API tests replay a run recorded in the browser and expect the same score, and check that tampered, reused, unknown, incomplete and oversized submissions are handled, plus account runs in arcade mode and the rate limit.

#!/usr/bin/env python3
# acid-maze - Developed by acidvegas in Python (https://github.com/acidvegas)
# app.py

import datetime
import hashlib
import json
import os
import re
import secrets
import sqlite3
import subprocess
import threading
import time

try:
	from flask import Flask, Response, abort, g, jsonify, render_template, request, send_from_directory
	from werkzeug.middleware.proxy_fix import ProxyFix
except ImportError:
	raise ImportError('missing flask library (pip install flask)')

import arcade


BASE_DIR      = os.path.dirname(os.path.abspath(__file__))
DB_PATH       = os.environ.get('MAZE_DB', os.path.join(BASE_DIR, 'data', 'maze.db'))
QJS_PATH      = os.environ.get('MAZE_QJS', '/usr/bin/qjs') # QuickJS binary used for the server side replay
PUBLIC_URL    = os.environ.get('MAZE_PUBLIC_URL', '').rstrip('/') # canonical site for crawlers and share cards, e.g. https://maze.acid.vegas
PROXIES       = int(os.environ.get('MAZE_PROXIES', '0')) # reverse proxies in front of the app, so the real client IP is used
REPLAY_JS     = os.path.join(BASE_DIR, 'replay.js')
CACHE_VERSION = int(time.time())
MODES         = ('classic', 'pack', 'shifting', 'procedural', 'mayhem', 'darkness', 'escape', 'survival', 'hunter', 'daily', 'custom')
MODE_NAMES    = {'classic': 'Classic', 'pack': 'Maze Pack', 'shifting': 'Shifting', 'procedural': 'Procedural', 'mayhem': 'Mayhem', 'darkness': 'Darkness', 'escape': 'Escape', 'survival': 'Survival', 'hunter': 'Hunter', 'daily': 'Daily', 'custom': 'Custom'}
GANG          = ('razor', 'vex', 'murk', 'fizz') # hunter mode: the gang member played
TWISTS        = ('pack', 'procedural', 'shifting', 'mayhem', 'darkness', 'escape')
MAX_BODY      = 512 * 1024
MAX_RUNS      = 60_000
MAX_TICKS     = 60 * 60 * 60
RUN_LIFETIME  = 4 * 3600
REPLAY_LIMIT  = 30
CLOCK_SLACK   = 600
INITIALS_RE   = re.compile(r'^[A-Z0-9 .]{3}$')
LIMITS        = {'run': (60, 600), 'submit': (40, 600)} # kind -> (requests, seconds) allowed per address
HITS          = {} # (kind, address) -> recent request times
HITS_LOCK     = threading.Lock()
ONLINE_SECS   = 60 # a signed in player counts as online for this long after their last request
SEEN          = {} # user id -> time of their last signed in request, for arcade presence
SESSIONS      = {} # token -> (acid-arcade-db session, expires at), so signed in requests don't all wait on acid-arcade-db
SESSIONS_LOCK = threading.Lock()
SESSIONS_MAX  = 10_000
SESSIONS_TTL  = 30

app = Flask(__name__)
app.config['MAX_CONTENT_LENGTH'] = MAX_BODY

if PROXIES:
	app.wsgi_app = ProxyFix(app.wsgi_app, x_for=PROXIES, x_proto=PROXIES) # not x_host, so a forged X-Forwarded-Host can't change the canonical and share URLs


@app.errorhandler(arcade.ArcadeDown)
def arcade_down(e: arcade.ArcadeDown) -> tuple:
	'''Answer 503 when acid-arcade-db can't be reached.'''

	return jsonify(error='account server is down'), 503


@app.errorhandler(arcade.ArcadeError)
def arcade_error(e: arcade.ArcadeError) -> tuple:
	'''Pass an acid-arcade-db refusal on to the player.'''

	return jsonify(error=str(e)), e.status


def arcade_poll():
	'''Follow renames and deletions from acid-arcade-db forever. Runs in a daemon thread in arcade mode.'''

	since = 0
	while True:
		try:
			data = arcade.changes(since)
			for user in data['changed']:
				mirror_user(user['user_id'], user['username'])
			for user_id in data['deleted']:
				delete_user(user_id)
			since = data['now']
		except Exception as e:
			print(f'arcade changes poll failed: {e}')
		time.sleep(60)


def arcade_presence():
	'''Report players seen in the last minute to acid-arcade-db every 30 seconds. Runs in a daemon thread in arcade mode.'''

	while True:
		try:
			cutoff = time.time() - ONLINE_SECS
			for user_id in [u for u, t in SEEN.items() if t < cutoff]:
				SEEN.pop(user_id, None)
			arcade.presence(list(SEEN))
		except Exception as e:
			print(f'arcade presence failed: {e}')
		time.sleep(30)


def arcade_session(token: str) -> dict:
	'''Return acid-arcade-db's session for a token, or None. Good sessions are cached for SESSIONS_TTL seconds. Raises arcade.ArcadeDown when it can't be reached.'''

	now = time.time()
	with SESSIONS_LOCK:
		cached = SESSIONS.get(token)
	if cached and cached[1] > now:
		return cached[0]
	session = arcade.session(token)
	if not session:
		return None
	mirror_user(session['user_id'], session['username'])
	with SESSIONS_LOCK:
		for k in [k for k, (_, exp) in SESSIONS.items() if exp < now] + list(SESSIONS)[:max(0, len(SESSIONS) - SESSIONS_MAX + 1)]:
			SESSIONS.pop(k, None)
		SESSIONS[token] = (session, now + SESSIONS_TTL)
	return session


def arcade_stats(user_id: int) -> tuple:
	'''
	Return (main, stats) for a player's arcade summary: the best score in any mode is the main stat.

	:param user_id: acid-arcade-db user id
	'''

	conn = connect()
	try:
		rows  = conn.execute('SELECT mode, MAX(score) AS best, COUNT(*) AS played, MAX(level) AS level FROM runs WHERE user_id = ? AND score IS NOT NULL GROUP BY mode', (user_id,)).fetchall()
		feats = [json.loads(r['feats']) for r in conn.execute('SELECT feats FROM runs WHERE user_id = ? AND feats IS NOT NULL', (user_id,))]
	finally:
		conn.close()
	bests  = {r['mode']: r['best'] for r in rows}
	levels = {r['mode']: r['level'] for r in rows}
	stats  = {'Classic best': bests.get('classic', 0), 'Games played': sum(r['played'] for r in rows), 'Highest level': max((v for m, v in levels.items() if m != 'survival'), default=0)} # survival's level is just minutes survived
	for mode in MODES[1:]:
		if mode in bests:
			stats[f'{MODE_NAMES[mode]} best'] = bests[mode]
	fast  = [f['fastClear'] for f in feats if f.get('fastClear')]
	extra = {
		'Escape best level'    : levels.get('escape', 0),
		'Escape most time left': max((f.get('escLeft', 0) for f in feats), default=0),
		'Survival longest secs': max((f.get('survSecs', 0) for f in feats), default=0),
		'Hunter catches'       : sum(f.get('catches', 0) for f in feats),
		'Hunter best level'    : levels.get('hunter', 0),
		'Hunter fastest clear' : min(fast, default=0),
		'Most gang on one pill': max((f.get('combo', 0) for f in feats), default=0),
		'Flawless levels'      : sum(f.get('flawless', 0) for f in feats)
	}
	stats.update({k: v for k, v in extra.items() if v})
	return max(bests.values(), default=0), stats


@app.post('/auth/logout')
def auth_logout():
	'''End the current arcade session here and on every site of the domain. Arcade mode only.'''

	if not arcade.enabled():
		abort(404)
	token = request.headers.get('X-Auth-Token') or ''
	with SESSIONS_LOCK:
		SESSIONS.pop(token, None)
	resp = jsonify(ok=True)
	if token:
		arcade.logout(token)
	arcade.clear_cookie(resp, request.host)
	return resp


@app.get('/auth/me')
def auth_me():
	'''Return the signed in player, if any. Arcade mode only.'''

	if not arcade.enabled():
		abort(404)
	token = request.headers.get('X-Auth-Token') or ''
	session = arcade_session(token) if token else None
	if not session:
		return jsonify(user=None)
	SEEN[session['user_id']] = time.time()
	resp = jsonify(user={'user_id': session['user_id'], 'username': session['username']})
	if not request.cookies.get(arcade.COOKIE): # share a sign-in made before shared sign-in existed
		arcade.set_cookie(resp, request.host, token)
	return resp


@app.get('/auth/sso')
def auth_sso():
	'''Swap the shared sign-in cookie set by any site on this domain for a session here. Arcade mode only.'''

	if not arcade.enabled():
		abort(404)
	reply = arcade.sso(request.cookies.get(arcade.COOKIE))
	if not reply:
		resp = jsonify(error='not signed in')
		arcade.clear_cookie(resp, request.host)
		return resp, 401
	mirror_user(reply['user_id'], reply['username'])
	return jsonify(token=reply['token'], user_id=reply['user_id'], username=reply['username'])


def connect() -> sqlite3.Connection:
	'''Open a connection to the scores database, creating or upgrading the schema.'''

	os.makedirs(os.path.dirname(DB_PATH), exist_ok=True)
	conn = sqlite3.connect(DB_PATH, timeout=10)
	conn.row_factory = sqlite3.Row
	conn.execute('PRAGMA journal_mode=WAL')
	conn.execute('CREATE TABLE IF NOT EXISTS runs (id TEXT PRIMARY KEY, mode TEXT NOT NULL, twists TEXT NOT NULL, seed INTEGER NOT NULL, created REAL NOT NULL, used INTEGER NOT NULL DEFAULT 0, score INTEGER, level INTEGER, ticks INTEGER, initials TEXT, finished REAL, day TEXT, user_id INTEGER)')
	if 'user_id' not in [r['name'] for r in conn.execute('PRAGMA table_info(runs)')]: # databases from before arcade accounts
		conn.execute('ALTER TABLE runs ADD COLUMN user_id INTEGER')
	if 'feats' not in [r['name'] for r in conn.execute('PRAGMA table_info(runs)')]: # per run feats from the replay, for arcade achievements
		conn.execute('ALTER TABLE runs ADD COLUMN feats TEXT')
	conn.execute('CREATE INDEX IF NOT EXISTS runs_board ON runs (mode, score DESC)')
	conn.execute('CREATE INDEX IF NOT EXISTS runs_user ON runs (user_id)')
	conn.execute('CREATE TABLE IF NOT EXISTS users (id INTEGER PRIMARY KEY, username TEXT NOT NULL)') # mirror of acid-arcade-db accounts
	conn.commit()
	return conn


def daily_seed(day: str) -> int:
	'''
	Seed shared by everyone for a given UTC day.

	:param day: the day as YYYY-MM-DD
	'''

	return int.from_bytes(hashlib.sha256(f'acid-maze-daily-{day}'.encode()).digest()[:4], 'big') & 0x7FFF_FFFF


def db() -> sqlite3.Connection:
	'''Return the request scoped SQLite connection.'''

	if 'db' not in g:
		g.db = connect()
	return g.db


def delete_user(user_id: int):
	'''
	Delete every run and the mirror row of an account deleted on acid-arcade-db.

	:param user_id: acid-arcade-db user id
	'''

	with SESSIONS_LOCK:
		for token in [t for t, (session, _) in SESSIONS.items() if session['user_id'] == user_id]:
			del SESSIONS[token]
	SEEN.pop(user_id, None)
	conn = connect()
	try:
		conn.execute('DELETE FROM runs WHERE user_id = ?', (user_id,))
		conn.execute('DELETE FROM users WHERE id = ?', (user_id,))
		conn.commit()
	finally:
		conn.close()

def limited(kind: str, addr: str) -> bool:
	'''
	Count a request against a per address limit and return True if it is over the limit.

	:param kind: a key of LIMITS
	:param addr: the client address
	'''

	limit, window = LIMITS[kind]
	now = time.time()
	with HITS_LOCK:
		if len(HITS) > 20_000:
			for key in [k for k, times in HITS.items() if not times or now - times[-1] > LIMITS[k[0]][1]]:
				del HITS[key]
		times = [t for t in HITS.get((kind, addr), []) if now - t < window]
		over  = len(times) >= limit
		if not over:
			times.append(now)
		HITS[(kind, addr)] = times
	return over

@app.post('/api/run')
def post_run():
	'''Start a run: the server picks the seed and returns a single use run id. Signed in players' runs are tied to their account.'''

	data   = request.get_json(silent=True) or {}
	mode   = data.get('mode', 'classic')
	twists = ''
	if mode not in MODES:
		return jsonify(error='unknown mode'), 400
	if mode == 'custom':
		parts = str(data.get('twists', '')).split('+')
		if any(p not in TWISTS for p in parts):
			return jsonify(error='unknown twist'), 400
		twists = '+'.join(t for t in TWISTS if t in parts)
	elif mode == 'hunter':
		twists = str(data.get('twists', ''))
		if twists not in GANG:
			return jsonify(error='unknown gang member'), 400
	if limited('run', request.remote_addr):
		return jsonify(error='too many runs, wait a few minutes'), 429
	user_id, username = None, None
	token = request.headers.get('X-Auth-Token') or ''
	if arcade.enabled() and token:
		try:
			session = arcade_session(token)
		except arcade.ArcadeDown:
			session = None # play on as a guest rather than not at all
		if session:
			user_id, username = session['user_id'], session['username']
			SEEN[user_id] = time.time()
	now    = time.time()
	run_id = secrets.token_hex(16)
	seed   = daily_seed(today()) if mode == 'daily' else secrets.randbelow(0x7FFF_FFFF)
	conn   = db()
	conn.execute('DELETE FROM runs WHERE score IS NULL AND created < ?', (now - RUN_LIFETIME,)) # unfinished, quit and failed runs
	conn.execute('INSERT INTO runs (id, mode, twists, seed, created, user_id) VALUES (?, ?, ?, ?, ?, ?)', (run_id, mode, twists, seed, now, user_id))
	conn.commit()
	return jsonify(run_id=run_id, seed=seed, mode=mode, twists=twists, account=username)


@app.post('/api/run/<run_id>/submit')
def post_submit(run_id: str):
	'''
	Submit the per tick input log of a finished run. The server replays it and
	stores the score the replay produced, ignoring anything the client claims.

	:param run_id: the run id returned by POST /api/run
	'''

	conn = db()
	data = request.get_json(silent=True)
	if not isinstance(data, dict) or not isinstance(data.get('inputs'), list):
		return jsonify(error='expected {"inputs": [[dir, count], ...]}'), 400
	inputs = data['inputs']
	if len(inputs) > MAX_RUNS:
		return jsonify(error='input log too large'), 413
	total = 0
	for item in inputs:
		if not (isinstance(item, list) and len(item) == 2 and all(type(v) is int for v in item) and (item[0] & 7) <= 4 and 0 <= item[0] <= 12 and 1 <= item[1] <= MAX_TICKS):
			return jsonify(error='malformed input log'), 400
		total += item[1]
	if total > MAX_TICKS:
		return jsonify(error='input log too large'), 413
	if limited('submit', request.remote_addr):
		return jsonify(error='too many submissions, wait a few minutes'), 429
	row = conn.execute('SELECT * FROM runs WHERE id = ?', (run_id,)).fetchone()
	if not row:
		return jsonify(error='unknown run'), 404
	now = time.time()
	if row['used'] or now - row['created'] > RUN_LIFETIME:
		return jsonify(error='run already submitted or expired'), 409
	cur = conn.execute('UPDATE runs SET used = 1 WHERE id = ? AND used = 0', (run_id,))
	conn.commit()
	if not cur.rowcount:
		return jsonify(error='run already submitted or expired'), 409
	try:
		result = replay(row['mode'], row['seed'], row['twists'], inputs)
	except (subprocess.SubprocessError, ValueError) as exc:
		return jsonify(error=f'replay failed: {exc}'), 422
	if not result.get('over'):
		return jsonify(error='input log ends before game over'), 422
	if result['ticks'] > (now - row['created']) * 60 + CLOCK_SLACK:
		return jsonify(error='run is longer than the time since it started'), 422
	day = today()
	conn.execute('UPDATE runs SET score = ?, level = ?, ticks = ?, finished = ?, day = ?, feats = ? WHERE id = ?', (result['score'], result['level'], result['ticks'], now, day, json.dumps(result.get('feats') or {}), run_id))
	conn.commit()
	account = None
	if row['user_id']:
		account = (conn.execute('SELECT username FROM users WHERE id = ?', (row['user_id'],)).fetchone() or {'username': None})['username']
		push_stats(row['user_id'])
		push_run(row['user_id'], run_id, row['mode'], result['score'], result['level'], request.remote_addr)
	rank_all   = conn.execute('SELECT COUNT(*) FROM runs WHERE mode = ? AND score > ?', (row['mode'], result['score'])).fetchone()[0] + 1
	rank_today = conn.execute('SELECT COUNT(*) FROM runs WHERE mode = ? AND day = ? AND score > ?', (row['mode'], day, result['score'])).fetchone()[0] + 1
	return jsonify(score=result['score'], level=result['level'], ticks=result['ticks'], rank_all=rank_all, rank_today=rank_today, account=account, qualifies=not row['user_id'] and result['score'] > 0 and min(rank_all, rank_today) <= 100)


def push_run(user_id: int, run_id: str, mode: str, score: int, level: int, ip: str):
	'''
	Report a finished account run to acid-arcade-db for casino rewards in the background. Does nothing in standalone mode.

	:param user_id: acid-arcade-db user id
	:param run_id: the run id, so the run is only counted once
	:param mode: game mode
	:param score: the score the server replay produced
	:param level: the level the server replay reached
	:param ip: the player's address
	'''

	if not arcade.enabled():
		return

	def push():
		try:
			arcade.activity(user_id, 'run', run_id, ip, {'mode': mode, 'score': score, 'level': level})
		except Exception as e:
			print(f'arcade activity failed for user {user_id}: {e}')

	threading.Thread(target=push, daemon=True).start()


def push_stats(user_id: int):
	'''
	Send a player's summary stats to acid-arcade-db in the background. Does nothing in standalone mode.

	:param user_id: acid-arcade-db user id
	'''

	if not arcade.enabled():
		return

	def push():
		try:
			main, stats = arcade_stats(user_id)
			arcade.push_stats(user_id, main, stats)
		except Exception as e:
			print(f'arcade stats push failed for user {user_id}: {e}')

	threading.Thread(target=push, daemon=True).start()


def replay(mode: str, seed: int, twists: str, inputs: list) -> dict:
	'''
	Replay an input log through the shared JS core with QuickJS.

	:param mode: game mode
	:param seed: server chosen seed
	:param twists: normalized twist list for custom mode
	:param inputs: run length encoded input log
	'''

	payload = json.dumps({'mode': mode, 'seed': seed, 'twists': twists, 'inputs': inputs})
	proc    = subprocess.run([QJS_PATH, '--std', '--memory-limit', str(256 * 1024 * 1024), '--stack-size', str(1024 * 1024), REPLAY_JS], input=payload, capture_output=True, text=True, timeout=REPLAY_LIMIT)
	if proc.returncode != 0:
		raise ValueError(proc.stderr.strip()[:200] or 'qjs exited with an error')
	return json.loads(proc.stdout.strip().splitlines()[-1][:4096])

def call(path: str, body: dict = None) -> dict:
	'''GET (no body) or POST JSON to the internal API and return the JSON reply.'''

	req = urllib.request.Request(ARCADE_URL + path, data=None if body is None else json.dumps(body).encode(), headers={'Content-Type': 'application/json', 'X-Arcade-Key': ARCADE_KEY})
	try:
		with urllib.request.urlopen(req, timeout=TIMEOUT) as resp:
			return json.load(resp)
	except urllib.error.HTTPError as e:
		try:
			message = json.load(e).get('error') or str(e)
		except ValueError:
			message = str(e)
		raise ArcadeError(message, e.code)
	except (urllib.error.URLError, OSError, ValueError, http.client.HTTPException) as e: # ValueError is a reply that is not JSON, HTTPException a broken one such as IncompleteRead
		raise ArcadeDown(str(e))

/* acid-maze - chiptune sound effects synthesised with Web Audio (all original) */

const Sfx = (function () {
	'use strict';

	let ac = null, master = null, muted = false, siren = null, sirenGain = null, lfo = null, lfoGain = null, sirenMode = 'off';
	let dotFlip = 0;

	try { muted = localStorage.getItem('acid-maze-mute') === '1'; } catch (e) { muted = false; }

	function ensure() {
		if (ac) return true;
		const AC = window.AudioContext || window.webkitAudioContext;
		if (!AC) return false;
		ac = new AC();
		master = ac.createGain();
		master.gain.value = muted ? 0 : 0.32;
		master.connect(ac.destination);
		return true;
	}

	function note(n) {
		return 440 * Math.pow(2, (n - 69) / 12);
	}

	function tone(freq, dur, type, vol, slideTo, delay) {
		if (!ac || muted) return;
		const t0 = ac.currentTime + (delay || 0);
		const o = ac.createOscillator(), g = ac.createGain();
		o.type = type || 'square';
		o.frequency.setValueAtTime(freq, t0);
		if (slideTo) o.frequency.exponentialRampToValueAtTime(slideTo, t0 + dur);
		g.gain.setValueAtTime(vol || 0.2, t0);
		g.gain.exponentialRampToValueAtTime(0.0008, t0 + dur);
		o.connect(g);
		g.connect(master);
		o.start(t0);
		o.stop(t0 + dur + 0.02);
	}

	function noise(dur, vol, delay, hp) {
		if (!ac || muted) return;
		const t0 = ac.currentTime + (delay || 0), len = Math.floor(ac.sampleRate * dur);
		const buf = ac.createBuffer(1, len, ac.sampleRate), d = buf.getChannelData(0);
		let v = 0;
		for (let i = 0; i < len; i++) { if (i % 6 === 0) v = Math.random() * 2 - 1; d[i] = v; }
		const s = ac.createBufferSource(), g = ac.createGain(), f = ac.createBiquadFilter();
		f.type = 'highpass';
		f.frequency.value = hp || 800;
		s.buffer = buf;
		g.gain.setValueAtTime(vol || 0.15, t0);
		g.gain.exponentialRampToValueAtTime(0.0008, t0 + dur);
		s.connect(f);
		f.connect(g);
		g.connect(master);
		s.start(t0);
	}

	function melody(notes, step, type, vol, offset) {
		notes.forEach(function (n, i) {
			if (n === null) return;
			const len = Array.isArray(n) ? n[1] : 1, key = Array.isArray(n) ? n[0] : n;
			tone(note(key + (offset || 0)), step * len * 0.95, type, vol, null, i * step);
		});
	}

	/* the eating sound: a two tone "bip-bup" that alternates pitch each dot */
	function dot() {
		dotFlip ^= 1;
		if (dotFlip) tone(330, 0.07, 'triangle', 0.35, 520);
		else tone(520, 0.07, 'triangle', 0.35, 300);
	}

	function startSiren() {
		if (!ac || siren) return;
		siren = ac.createOscillator();
		sirenGain = ac.createGain();
		lfo = ac.createOscillator();
		lfoGain = ac.createGain();
		siren.type = 'sawtooth';
		lfo.type = 'triangle';
		lfo.connect(lfoGain);
		lfoGain.connect(siren.frequency);
		const f = ac.createBiquadFilter();
		f.type = 'lowpass';
		f.frequency.value = 1400;
		siren.connect(f);
		f.connect(sirenGain);
		sirenGain.connect(master);
		sirenGain.gain.value = 0;
		siren.start();
		lfo.start();
	}

	/* mode: off | chase | fright | eyes; level 0..1 rises as the maze empties */
	function setSiren(mode, level) {
		if (!ac) return;
		startSiren();
		const t = ac.currentTime;
		if (mode === 'off' || muted) { sirenGain.gain.setTargetAtTime(0, t, 0.03); sirenMode = mode; return; }
		if (mode === 'chase') {
			siren.frequency.setTargetAtTime(180 + level * 160, t, 0.1);
			lfo.frequency.setTargetAtTime(3.2 + level * 3, t, 0.1);
			lfoGain.gain.setTargetAtTime(40 + level * 30, t, 0.1);
			sirenGain.gain.setTargetAtTime(0.05, t, 0.05);
		} else if (mode === 'fright') {
			siren.frequency.setTargetAtTime(110, t, 0.05);
			lfo.frequency.setTargetAtTime(9, t, 0.05);
			lfoGain.gain.setTargetAtTime(60, t, 0.05);
			sirenGain.gain.setTargetAtTime(0.07, t, 0.05);
		} else if (mode === 'eyes') {
			siren.frequency.setTargetAtTime(700, t, 0.05);
			lfo.frequency.setTargetAtTime(14, t, 0.05);
			lfoGain.gain.setTargetAtTime(220, t, 0.05);
			sirenGain.gain.setTargetAtTime(0.04, t, 0.05);
		}
		sirenMode = mode;
	}

	function play(ev) {
		if (!ensure() || muted) return;
		if (ac.state === 'suspended') ac.resume();
		switch (ev) {
			case 'dot': dot(); break;
			case 'pill': tone(120, 0.25, 'square', 0.25, 60); tone(240, 0.25, 'square', 0.12, 960); break;
			case 'eat': tone(300, 0.35, 'square', 0.22, 1800); tone(150, 0.35, 'triangle', 0.25, 900); break;
			case 'death':
				for (let i = 0; i < 9; i++) tone(note(76 - i * 2), 0.11, 'square', 0.18, note(72 - i * 2), i * 0.11);
				noise(0.5, 0.2, 1.0, 400);
				break;
			case 'extra': melody([84, 88, 91, 96, 91, 96], 0.07, 'square', 0.16); break;
			case 'fruit': melody([79, 83, 86, 91], 0.05, 'triangle', 0.3); break;
			case 'fruitspawn': tone(1200, 0.08, 'triangle', 0.12, 1600); break;
			case 'start':
				melody([60, 67, 63, 67, 72, 70, 67, 63, 65, 68, 72, 75, [74, 2], [79, 2]], 0.15, 'square', 0.12);
				melody([36, null, 43, null, 39, null, 43, null, 41, null, 44, null, 43, null, [43, 2]], 0.15, 'triangle', 0.3);
				break;
			case 'intermission':
				melody([72, 76, 79, 76, 77, 81, 84, 81, 79, 76, 72, 74, [76, 2], null, 67, 72, 76, [79, 3]], 0.13, 'square', 0.11);
				melody([48, null, 55, null, 53, null, 57, null, 55, null, 52, null, 48, null, 55, null, [48, 3]], 0.13, 'triangle', 0.28);
				break;
			case 'clear': melody([72, 79, 84, 79, 84, 91], 0.06, 'square', 0.1); break;
			case 'coin': tone(988, 0.06, 'square', 0.15); tone(1319, 0.25, 'square', 0.15, null, 0.06); break;
			case 'select': tone(660, 0.04, 'square', 0.1); break;
			case 'gate': tone(90, 0.3, 'sawtooth', 0.15, 45); noise(0.2, 0.08, 0, 200); break;
			case 'warp': tone(1800, 0.2, 'sine', 0.2, 200); break;
			case 'wake': tone(200, 0.3, 'sawtooth', 0.15, 600); break;
			case 'split': noise(0.15, 0.2, 0, 2000); tone(900, 0.2, 'triangle', 0.2, 1800); break;
			case 'steal': tone(160, 0.06, 'square', 0.12, 120); break;
			case 'shield': tone(700, 0.3, 'triangle', 0.25, 200); noise(0.2, 0.1); break;
			case 'gameover': melody([67, 63, 60, [55, 3]], 0.2, 'triangle', 0.3); break;
			default:
				if (ev.indexOf('power:') === 0) {
					if (ev === 'power:bomb') { noise(0.6, 0.35, 0, 100); tone(80, 0.5, 'square', 0.3, 30); }
					else melody([72, 76, 79, 84], 0.045, 'square', 0.14);
				}
		}
	}

	function setMuted(m) {
		muted = m;
		try { localStorage.setItem('acid-maze-mute', m ? '1' : '0'); } catch (e) { /* storage blocked */ }
		if (master) master.gain.value = m ? 0 : 0.32;
	}

	return {
		isMuted: function () { return muted; }, play: play, setMuted: setMuted, setSiren: setSiren, unlock: ensure,
		sirenMode: function () { return sirenMode; }
	};
})();

/* ---------- PRNG (mulberry32) ---------- */

function Rng(seed) {
	this.s = seed >>> 0;
}
Rng.prototype.next = function () {
	this.s = (this.s + 0x6D2B79F5) >>> 0;
	let t = this.s;
	t = Math.imul(t ^ (t >>> 15), t | 1);
	t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
	return (t ^ (t >>> 14)) >>> 0;
};
Rng.prototype.int = function (n) {
	return this.next() % n;
};


/* ---------- config ---------- */

function resolveConfig(mode, seed, twists) {
	const cfg = { mode: mode, maze: 'classic', shifting: false, mayhem: false, darkness: false, escape: false, survival: false, hunter: false, member: '' };
	if (mode === 'pack') cfg.maze = 'pack';
	else if (mode === 'shifting') { cfg.maze = 'pack'; cfg.shifting = true; }
	else if (mode === 'procedural') cfg.maze = 'procedural';
	else if (mode === 'mayhem') cfg.mayhem = true;
	else if (mode === 'darkness') { cfg.maze = 'pack'; cfg.darkness = true; }
	else if (mode === 'escape') { cfg.maze = 'escape'; cfg.escape = true; }
	else if (mode === 'survival') cfg.survival = true;
	else if (mode === 'hunter') { cfg.maze = 'pack'; cfg.hunter = true; cfg.member = GANG.indexOf(twists) >= 0 ? twists : 'razor'; }
	else if (mode === 'daily') {
		const r = new Rng(seed ^ 0x5EED);
		cfg.maze = ['classic', 'pack', 'procedural'][r.int(3)];
		cfg.shifting = r.int(2) === 1;
		cfg.mayhem = r.int(2) === 1;
		cfg.darkness = r.int(5) < 2;
		if (!cfg.shifting && !cfg.mayhem && !cfg.darkness) cfg.mayhem = true;
		if (cfg.maze === 'classic' && !(cfg.shifting && cfg.mayhem)) cfg.shifting = true;
		if (r.int(4) === 0) { cfg.maze = 'escape'; cfg.escape = true; cfg.shifting = false; }
	} else if (mode === 'custom') {
		const list = (twists || '').split('+');
		if (list.indexOf('pack') >= 0) cfg.maze = 'pack';
		if (list.indexOf('procedural') >= 0) cfg.maze = 'procedural';
		cfg.shifting = list.indexOf('shifting') >= 0;
		cfg.mayhem = list.indexOf('mayhem') >= 0;
		cfg.darkness = list.indexOf('darkness') >= 0;
		if (list.indexOf('escape') >= 0) { cfg.maze = 'escape'; cfg.escape = true; cfg.shifting = false; }
	}
	return cfg;
}

function normalizeTwists(twists) {
	const list = (twists || '').split('+');
	return TWISTS.filter(function (t) { return list.indexOf(t) >= 0; }).join('+');
}


/* ---------- maze building ---------- */

function emptyMaze(name, wall, w, h) {
	w = w || COLS;
	h = h || ROWS;
	return {
		name: name, wall: wall, w: w, h: h,
		cells: new Uint8Array(w * h),
		dots: new Uint8Array(w * h),
		tunnel: new Uint8Array(w * h),
		dotCount: 0
	};
}

function inHouseBox(x, y) {
	return y >= 12 && y <= 16 && x >= 10 && x <= 17;
}

function parseMaze(def) {
	const m = emptyMaze(def.name, def.wall);
	for (let y = 0; y < ROWS; y++) {
		const half = def.rows[y];
		const row = half + half.split('').reverse().join('');
		for (let x = 0; x < COLS; x++) {
			const ch = row[x], i = y * COLS + x;
			if (ch === '#') m.cells[i] = C_WALL;
			else if (ch === 'H') m.cells[i] = C_HOUSE;
			else if (ch === '-') m.cells[i] = C_DOOR;
			else {
				m.cells[i] = C_PATH;
				if (ch === '.') m.dots[i] = 1;
				else if (ch === 'o') m.dots[i] = 2;
				else if (ch === 't') m.tunnel[i] = 1;
			}
			if (inHouseBox(x, y) && m.cells[i] === C_WALL) m.cells[i] = C_HOUSE;
		}
	}
	finishMaze(m);
	return m;
}

function finishMaze(m) {
	m.dotCount = 0;
	for (let i = 0; i < m.dots.length; i++) if (m.dots[i]) m.dotCount++;
}

/* Procedural maze: a symmetric lattice graph with edges removed at random while
 * every node keeps degree >= 2 (no dead ends) and the graph stays connected. */
function proceduralMaze(seed, level) {
	const r = new Rng((seed ^ Math.imul(level, 0x9E3779B1)) >>> 0);
	const XS = [1, 5, 9, 18, 22, 26], YS = [1, 5, 8, 11, 14, 17, 20, 23, 26, 29];
	const nodes = [], index = {};
	for (let j = 0; j < YS.length; j++) for (let i = 0; i < XS.length; i++) {
		index[XS[i] + ',' + YS[j]] = nodes.length;
		nodes.push([XS[i], YS[j]]);
	}
	const edges = [];
	const key = function (a, b) { return a < b ? a + ':' + b : b + ':' + a; };
	const alive = {};
	for (let j = 0; j < YS.length; j++) for (let i = 0; i < XS.length; i++) {
		const a = index[XS[i] + ',' + YS[j]];
		if (i + 1 < XS.length && !(XS[i] === 9 && YS[j] === 14)) edges.push([a, index[XS[i + 1] + ',' + YS[j]]]);
		if (j + 1 < YS.length) edges.push([a, index[XS[i] + ',' + YS[j + 1]]]);
	}
	edges.forEach(function (e) { alive[key(e[0], e[1])] = true; });
	const mirrorNode = function (n) { return index[(27 - nodes[n][0]) + ',' + nodes[n][1]]; };
	const required = {};
	[[9, 11, 18, 11], [9, 17, 18, 17], [9, 23, 18, 23], [9, 11, 9, 14], [9, 14, 9, 17], [18, 11, 18, 14], [18, 14, 18, 17]].forEach(function (q) {
		required[key(index[q[0] + ',' + q[1]], index[q[2] + ',' + q[3]])] = true;
	});
	const adj = nodes.map(function () { return []; });
	edges.forEach(function (e) { adj[e[0]].push(e[1]); adj[e[1]].push(e[0]); });
	const degree = function (n) {
		let d = 0;
		adj[n].forEach(function (o) { if (alive[key(n, o)]) d++; });
		return d;
	};
	const connected = function () {
		const seen = new Uint8Array(nodes.length), stack = [0];
		seen[0] = 1;
		let count = 1;
		while (stack.length) {
			const n = stack.pop();
			adj[n].forEach(function (o) {
				if (!seen[o] && alive[key(n, o)]) { seen[o] = 1; count++; stack.push(o); }
			});
		}
		return count === nodes.length;
	};
	const order = edges.filter(function (e) { return nodes[e[0]][0] <= 13 || nodes[e[1]][0] <= 13; });
	for (let i = order.length - 1; i > 0; i--) {
		const k = r.int(i + 1), t = order[i];
		order[i] = order[k];
		order[k] = t;
	}
	const target = 14 + r.int(8);
	let removed = 0;
	for (let i = 0; i < order.length && removed < target; i++) {
		const e = order[i], k1 = key(e[0], e[1]), k2 = key(mirrorNode(e[0]), mirrorNode(e[1]));
		if (required[k1] || !alive[k1]) continue;
		alive[k1] = false;
		alive[k2] = false;
		const touched = [e[0], e[1], mirrorNode(e[0]), mirrorNode(e[1])];
		if (touched.some(function (n) { return degree(n) < 2; }) || !connected()) {
			alive[k1] = true;
			alive[k2] = true;
		} else removed++;
	}
	const m = emptyMaze('SECTOR ' + level, 4 + (level % 4));
	m.cells.fill(C_WALL);
	const carve = function (x, y) { m.cells[y * COLS + x] = C_PATH; m.dots[y * COLS + x] = 1; };
	edges.forEach(function (e) {
		if (!alive[key(e[0], e[1])]) return;
		const a = nodes[e[0]], b = nodes[e[1]];
		for (let x = Math.min(a[0], b[0]); x <= Math.max(a[0], b[0]); x++) for (let y = Math.min(a[1], b[1]); y <= Math.max(a[1], b[1]); y++) carve(x, y);
	});
	const ty = [8, 14, 20][r.int(3)];
	for (let x = 0; x < COLS; x++) {
		if (x <= 1 || x >= 26) {
			const i = ty * COLS + x;
			m.cells[i] = C_PATH;
			if (x === 0 || x === 27) { m.dots[i] = 0; m.tunnel[i] = 1; }
		}
	}
	for (let y = 10; y <= 18; y++) for (let x = 8; x <= 19; x++) {
		const i = y * COLS + x;
		if (y >= 11 && y <= 17 && x >= 9 && x <= 18) m.dots[i] = 0;
		if (inHouseBox(x, y)) { m.cells[i] = C_HOUSE; m.dots[i] = 0; }
	}
	m.cells[12 * COLS + 13] = C_DOOR;
	m.cells[12 * COLS + 14] = C_DOOR;
	[[1, 5], [26, 5], [1, 23], [26, 23]].forEach(function (p) { m.dots[p[1] * COLS + p[0]] = 2; });
	finishMaze(m);
	return m;
}

/* Escape maze: a labyrinth several screens big that grows every level, carved by a seeded depth first walk with
 * extra walls knocked out for loops. The exit is a random cell far (by path) from the start, the gang patrols
 * from spawn points away from the start, and dots and pills are sparse. */
function escapeMaze(seed, level) {
	const r = new Rng((seed ^ Math.imul(level, 0x85EBCA6B) ^ 0xE5CA9E) >>> 0);
	const cw = Math.min(55, 20 + 7 * level), ch = Math.min(61, 22 + 8 * level);
	const w = cw * 2 + 1, h = ch * 2 + 1;
	const m = emptyMaze('ESCAPE ' + level, level % 8, w, h);
	m.cells.fill(C_WALL);
	const seen = new Uint8Array(cw * ch), stack = [r.int(cw * ch)];
	seen[stack[0]] = 1;
	m.cells[(((stack[0] / cw) | 0) * 2 + 1) * w + (stack[0] % cw) * 2 + 1] = C_PATH;
	while (stack.length) {
		const c = stack[stack.length - 1], cx = c % cw, cy = (c / cw) | 0, next = [];
		for (let d = 1; d <= 4; d++) {
			const nx = cx + DX[d], ny = cy + DY[d];
			if (nx >= 0 && ny >= 0 && nx < cw && ny < ch && !seen[ny * cw + nx]) next.push(d);
		}
		if (!next.length) { stack.pop(); continue; }
		const d = next[r.int(next.length)], n = (cy + DY[d]) * cw + cx + DX[d];
		seen[n] = 1;
		m.cells[(cy * 2 + 1 + DY[d]) * w + cx * 2 + 1 + DX[d]] = C_PATH;
		m.cells[(cy * 2 + 1 + DY[d] * 2) * w + cx * 2 + 1 + DX[d] * 2] = C_PATH;
		stack.push(n);
	}
	for (let y = 1; y < h - 1; y++) for (let x = 1; x < w - 1; x++) {
		if ((x + y) % 2 === 0 || m.cells[y * w + x] !== C_WALL) continue;
		if (r.int(7) === 0) m.cells[y * w + x] = C_PATH;
	}
	const sx = r.int(cw) * 2 + 1, sy = r.int(ch) * 2 + 1;
	const dist = new Int32Array(w * h).fill(-1), q = [sy * w + sx];
	dist[q[0]] = 0;
	let far = 0;
	for (let qi = 0; qi < q.length; qi++) {
		const i = q[qi];
		far = Math.max(far, dist[i]);
		for (let d = 1; d <= 4; d++) {
			const j = i + DX[d] + DY[d] * w;
			if (m.cells[j] === C_PATH && dist[j] < 0) { dist[j] = dist[i] + 1; q.push(j); }
		}
	}
	const exits = [], spawns = [];
	for (let i = 0; i < q.length; i++) {
		const t = q[i], x = t % w, y = (t / w) | 0;
		if (x % 2 === 0 || y % 2 === 0) continue;
		if (dist[t] * 5 >= far * 4) exits.push(t);
		if (dist[t] >= 24 && dist2(x, y, sx, sy) >= 144) spawns.push(t);
	}
	const exit = exits[r.int(exits.length)];
	m.start = [sx, sy];
	m.exit = [exit % w, (exit / w) | 0];
	m.par = Math.floor(w * h / 20);
	for (let i = spawns.length - 1; i > 0; i--) {
		const k = r.int(i + 1), t = spawns[i];
		spawns[i] = spawns[k];
		spawns[k] = t;
	}
	m.spawns = spawns.filter(function (t) { return t !== exit; }).slice(0, Math.min(24, Math.floor(w * h / 400))).map(function (t) { return [t % w, (t / w) | 0]; });
	for (let i = 0; i < q.length; i++) if (q[i] !== q[0] && q[i] !== exit && r.int(10) === 0) m.dots[q[i]] = 1;
	for (let k = 0; k < Math.min(8, 3 + level); k++) {
		const t = q[1 + r.int(q.length - 1)];
		if (t !== exit) m.dots[t] = 2;
	}
	finishMaze(m);
	return m;
}

function mazeFor(cfg, seed, level) {
	if (cfg.maze === 'escape') return escapeMaze(seed, level);
	if (cfg.maze === 'procedural') return proceduralMaze(seed, level);
	if (cfg.maze === 'pack') return parseMaze(MAZES[PACK[(level - 1) % PACK.length]]);
	return parseMaze(MAZES.classic);
}

/* Shifting walls: single thickness interior wall tiles between two corridors
 * become gates in two alternating groups. Opening a gate only adds paths. */
function findGates(g) {
	const m = g.maze, list = [];
	for (let y = 2; y < ROWS - 2; y++) for (let x = 2; x <= 12; x++) {
		if (m.cells[y * COLS + x] !== C_WALL) continue;
		if (y >= 9 && y <= 19 && x >= 7) continue;
		const open = function (dx, dy) { return m.cells[(y + dy) * COLS + x + dx] === C_PATH; };
		const h = open(-1, 0) && open(1, 0) && !open(0, -1) && !open(0, 1);
		const v = open(0, -1) && open(0, 1) && !open(-1, 0) && !open(1, 0);
		if (h || v) list.push([x, y]);
	}
	for (let i = list.length - 1; i > 0; i--) {
		const k = g.rng.int(i + 1), t = list[i];
		list[i] = list[k];
		list[k] = t;
	}
	g.gates = [];
	list.slice(0, 8).forEach(function (p, n) {
		const group = n % 2;
		g.gates.push({ x: p[0], y: p[1], group: group, closed: true });
		g.gates.push({ x: 27 - p[0], y: p[1], group: group, closed: true });
	});
	g.gateGroup = 0;
	g.gateT = 420;
	applyGates(g, true);
}

function applyGates(g, force) {
	g.gates.forEach(function (gt) {
		const want = gt.group !== g.gateGroup;
		if (want === gt.closed) return;
		if (want && !force && occupied(g, gt.x, gt.y)) return;
		gt.closed = want;
		g.maze.cells[gt.y * COLS + gt.x] = want ? C_WALL : C_PATH;
	});
}

function occupied(g, tx, ty) {
	const cx = tx * TILE + HALF, cy = ty * TILE + HALF;
	const near = function (e) { return Math.abs(wrapDelta(g.maze, e.x - cx)) < TILE && Math.abs(e.y - cy) < TILE; };
	if (near(g.hero)) return true;
	return g.chasers.some(near);
}


/* ---------- geometry helpers ---------- */

function cellAt(m, tx, ty) {
	if (ty < 0 || ty >= m.h) return C_WALL;
	tx = ((tx % m.w) + m.w) % m.w;
	return m.cells[ty * m.w + tx];
}

function centerOff(v) {
	return (((v - HALF) % TILE) + TILE) % TILE;
}

function interiorWall(m, tx, ty) {
	if (tx < 1 || tx > m.w - 2 || ty < 1 || ty > m.h - 2) return false;
	return m.cells[ty * m.w + tx] === C_WALL;
}

function tileOf(v) {
	return Math.floor(v / TILE);
}

function wrapDelta(m, d) {
	const wrap = m.w * TILE;
	if (d > wrap / 2) return d - wrap;
	if (d < -wrap / 2) return d + wrap;
	return d;
}

function wrapX(m, x) {
	const wrap = m.w * TILE;
	return ((x % wrap) + wrap) % wrap;
}

function dist2(ax, ay, bx, by) {
	return (ax - bx) * (ax - bx) + (ay - by) * (ay - by);
}

function heroCan(g, tx, ty) {
	const c = cellAt(g.maze, tx, ty);
	if (c === C_PATH) return true;
	return g.fx.phase > 0 && interiorWall(g.maze, ((tx % g.maze.w) + g.maze.w) % g.maze.w, ty);
}

function chaserCan(g, c, tx, ty) {
	const cell = cellAt(g.maze, tx, ty);
	if (cell === C_PATH) return true;
	return c.kind === 'phaze' && (c.state === 'active' || c.state === 'eyes') && interiorWall(g.maze, ((tx % g.maze.w) + g.maze.w) % g.maze.w, ty);
}

function moveGrid(m, e, dist, decide) {
	let guard = 0;
	while (dist > 0 && guard++ < 8) {
		const ox = centerOff(e.x), oy = centerOff(e.y);
		if (ox === 0 && oy === 0) decide(e);
		if (e.dir === NONE) return;
		const horiz = e.dir === LEFT || e.dir === RIGHT;
		const off = horiz ? ox : oy;
		const forward = e.dir === RIGHT || e.dir === DOWN;
		const to = off === 0 ? TILE : (forward ? TILE - off : off);
		const s = Math.min(dist, to);
		e.x = wrapX(m, e.x + DX[e.dir] * s);
		e.y += DY[e.dir] * s;
		dist -= s;
	}
}

function randomOpenTile(g, minFrom, tries) {
	let best = null, bestD = -1;
	for (let i = 0; i < tries; i++) {
		const m = g.maze, tx = 1 + g.rng.int(m.w - 2), ty = 1 + g.rng.int(m.h - 2);
		if (m.cells[ty * m.w + tx] !== C_PATH || m.tunnel[ty * m.w + tx]) continue;
		if (!g.cfg.escape && ty >= 11 && ty <= 17 && tx >= 9 && tx <= 18) continue;
		let d = 1e9;
		minFrom.forEach(function (e) { d = Math.min(d, dist2(tx, ty, tileOf(e.x), tileOf(e.y))); });
		if (d > bestD) { bestD = d; best = [tx, ty]; }
	}
	return best;
}


/* ---------- game setup ---------- */

function create(opts) {
	const seed = opts.seed >>> 0;
	const g = {
		mode: opts.mode, twists: normalizeTwists(opts.twists), seed: seed,
		cfg: resolveConfig(opts.mode, seed, opts.twists),
		rng: new Rng(seed ^ 0xA5C1D),
		tick: 0, score: 0, lives: 3, level: 1, extra: false, over: false,
		phase: 'start', phaseT: 240, events: [], popups: [],
		maze: null, hero: null, chasers: [], gates: [], gateGroup: 0, gateT: 0,
		dotsLeft: 0, dotsEaten: 0, lifeDots: 0, frightT: 0, combo: 0, freezeT: 0, escT: 0, survT: 0, survAdded: 0, prey: [], blocks: [], abil: 0, abilityIn: false, huntT: 0, clean: true,
		feats: { escLeft: 0, survSecs: 0, catches: 0, fastClear: 0, combo: 0, flawless: 0 }, // for arcade achievements
		modeIdx: 0, modeT: 0, fruit: null, fruitsSpawned: 0,
		pickup: null, pickupT: 600, fx: null, decoy: null, interKind: 0, nextId: 1
	};
	loadLevel(g);
	return g;
}

function loadLevel(g) {
	g.maze = mazeFor(g.cfg, g.seed, g.level);
	g.dotsLeft = g.maze.dotCount;
	g.dotsEaten = 0;
	g.fruitsSpawned = 0;
	g.escT = 0;
	g.clean = true;
	g.gates = [];
	if (g.cfg.shifting) findGates(g);
	resetActors(g);
}

function modeTable(level) {
	if (level === 1) return [420, 1200, 420, 1200, 300, 1200, 300, -1];
	if (level < 5) return [420, 1200, 420, 1200, 300, -1];
	return [300, 1200, 300, 1200, 300, -1];
}

function newChaser(g, kind, slot, release) {
	const xs = [14 * TILE, 12 * TILE, 16 * TILE];
	const c = {
		id: g.nextId++, kind: kind, x: xs[slot], y: 14 * TILE + HALF, dir: UP,
		state: 'house', fright: false, release: release, temp: false, ttl: 0,
		wx: 1, wy: 1, timer: 0, awake: 0, warn: 0, scx: 0, scy: 0
	};
	const corners = { razor: [25, -3], vex: [2, -3], murk: [27, 31], fizz: [0, 31] };
	const k = corners[kind] || [[0, -3], [27, -3], [0, 31], [27, 31]][g.rng.int(4)];
	c.scx = k[0];
	c.scy = k[1];
	return c;
}

function resetActors(g) {
	if (g.cfg.hunter) { resetHunt(g); return; }
	g.hero = { x: 14 * TILE, y: 23 * TILE + HALF, dir: LEFT, want: NONE };
	if (g.cfg.escape) g.hero = { x: g.maze.start[0] * TILE + HALF, y: g.maze.start[1] * TILE + HALF, dir: NONE, want: NONE };
	g.fx = { boots: 0, freeze: 0, magnet: 0, phase: 0, double: 0, shield: false };
	g.decoy = null;
	g.frightT = 0;
	g.freezeT = 0;
	g.fruit = null;
	g.pickup = null;
	g.pickupT = 600;
	g.modeIdx = 0;
	g.modeT = modeTable(g.level)[0];
	g.lifeDots = 0;
	const f = Math.max(30, 100 - (g.level - 1) * 10);
	if (g.cfg.escape) {
		g.chasers = g.maze.spawns.map(function (p, i) {
			const c = newChaser(g, g.cfg.mayhem && i % 3 === 2 ? SPECIALS[g.rng.int(SPECIALS.length)] : GANG[i % GANG.length], 0, 0);
			c.state = 'active';
			c.x = p[0] * TILE + HALF;
			c.y = p[1] * TILE + HALF;
			c.dir = NONE;
			c.hx = c.wx = p[0];
			c.hy = c.wy = p[1];
			c.alert = 0;
			return c;
		});
		return;
	}
	g.chasers = [
		newChaser(g, 'razor', 0, 0),
		newChaser(g, 'vex', 0, (60 * f / 100) | 0),
		newChaser(g, 'murk', 1, (300 * f / 100) | 0),
		newChaser(g, 'fizz', 2, (540 * f / 100) | 0)
	];
	const r0 = g.chasers[0];
	r0.state = 'active';
	r0.y = 11 * TILE + HALF;
	r0.dir = LEFT;
	if (g.cfg.mayhem) {
		const pool = SPECIALS.slice(), count = Math.min(3, g.level);
		for (let i = 0; i < count; i++) {
			const kind = pool.splice(g.rng.int(pool.length), 1)[0];
			const c = newChaser(g, kind, i % 3, ((720 + i * 240) * f / 100) | 0);
			if (kind === 'dozer') {
				const t = randomOpenTile(g, [g.hero], 40);
				c.state = 'active';
				c.x = t[0] * TILE + HALF;
				c.y = t[1] * TILE + HALF;
				c.dir = NONE;
			}
			g.chasers.push(c);
		}
	}
	for (let i = 0; i < g.survAdded; i++) g.chasers.push(survivor(g, (240 + i * 120) * f / 100 | 0));
}

/* survival: one more chaser from the whole cast, released from the house */
function survivor(g, release) {
	const pool = GANG.concat(SPECIALS);
	return newChaser(g, pool[g.rng.int(pool.length)], g.rng.int(3), release);
}


/* ---------- speeds ---------- */

function speeds(level) {
	if (level === 1) return { hero: 256, heroF: 288, chaser: 240, fright: 160, tunnel: 128 };
	if (level < 5) return { hero: 288, heroF: 304, chaser: 272, fright: 176, tunnel: 144 };
	if (level < 21) return { hero: 320, heroF: 320, chaser: 304, fright: 192, tunnel: 160 };
	return { hero: 288, heroF: 288, chaser: 304, fright: 192, tunnel: 160 };
}

function chaserSpeed(g, c) {
	const sp = speeds(g.level);
	if (c.state === 'eyes') return 480;
	if (c.state === 'entering') return 384;
	if (c.state !== 'active') return 128;
	if (c.fright) return sp.fright;
	const i = tileOf(c.y) * g.maze.w + ((tileOf(c.x) % g.maze.w) + g.maze.w) % g.maze.w;
	if (g.maze.tunnel[i]) return sp.tunnel;
	let s = sp.chaser;
	if (c.kind === 'razor' && !g.cfg.escape) {
		const od = Math.min(80, 20 + 5 * g.level);
		if (g.dotsLeft <= od / 2) s += 32;
		else if (g.dotsLeft <= od) s += 16;
	}
	if (c.kind === 'mini') s += 48;
	if (c.kind === 'phaze') s = (s * 85 / 100) | 0;
	if (c.kind === 'dozer') s = c.awake > 0 ? (s * 115 / 100) | 0 : 0;
	if (c.kind === 'lurk' && dist2(tileOf(c.x), tileOf(c.y), tileOf(g.hero.x), tileOf(g.hero.y)) <= 36) s = (s * 125 / 100) | 0;
	if (g.cfg.escape && c.alert === 0) s = (s * 75 / 100) | 0;
	return s;
}


/* ---------- chaser AI ---------- */

function chaseTarget(g, c) {
	const h = g.hero;
	let hx = tileOf(h.x), hy = tileOf(h.y), hd = h.dir;
	if (g.decoy) { hx = g.decoy.tx; hy = g.decoy.ty; hd = NONE; }
	const cx = tileOf(c.x), cy = tileOf(c.y);
	switch (c.kind) {
		case 'vex': return [hx + 4 * DX[hd], hy + 4 * DY[hd]];
		case 'lurk': return [hx + 8 * DX[hd], hy + 8 * DY[hd]];
		case 'murk': {
			let best = null, bd = 1e9;
			g.chasers.forEach(function (o) {
				if (o === c || o.state !== 'active') return;
				const d = dist2(cx, cy, tileOf(o.x), tileOf(o.y));
				if (d < bd) { bd = d; best = o; }
			});
			if (!best) return [hx, hy];
			return [2 * hx - tileOf(best.x), 2 * hy - tileOf(best.y)];
		}
		case 'fizz':
			if (dist2(cx, cy, hx, hy) <= 36) return [hx, hy];
			return [c.wx, c.wy];
		case 'munch': {
			let best = null, bd = 1e9;
			const m = g.maze;
			for (let i = 0; i < m.dots.length; i++) {
				if (m.dots[i] !== 1) continue;
				const d = dist2(cx, cy, i % m.w, (i / m.w) | 0);
				if (d < bd) { bd = d; best = [i % m.w, (i / m.w) | 0]; }
			}
			return best || [hx, hy];
		}
		default: return [hx, hy];
	}
}

function chaserDecide(g, c) {
	const tx = tileOf(c.x), ty = tileOf(c.y);
	if (c.state === 'eyes' && g.cfg.escape && tx === c.hx && ty === c.hy) {
		c.state = 'active';
		c.fright = false;
		c.alert = 0;
	}
	if (c.state === 'eyes' && !g.cfg.escape && ty === 11 && (tx === 13 || tx === 14)) {
		c.state = 'entering';
		c.dir = NONE;
		return;
	}
	const opts = [];
	for (let d = 1; d <= 4; d++) if (d !== OPP[c.dir] && chaserCan(g, c, tx + DX[d], ty + DY[d])) opts.push(d);
	if (!opts.length) {
		const r = OPP[c.dir];
		c.dir = r && chaserCan(g, c, tx + DX[r], ty + DY[r]) ? r : NONE;
		return;
	}
	if (c.fright && c.state === 'active') {
		c.dir = opts[g.rng.int(opts.length)];
		return;
	}
	if (c.kind === 'echo' && c.state === 'active' && opts.indexOf(g.hero.dir) >= 0) {
		c.dir = g.hero.dir;
		return;
	}
	let t;
	if (g.cfg.escape) t = c.state === 'eyes' ? [c.hx, c.hy] : (c.alert > 0 ? chaseTarget(g, c) : [c.wx, c.wy]);
	else if (c.state === 'eyes') t = [13, 11];
	else if (g.modeIdx % 2 === 0 && c.kind !== 'munch' && c.kind !== 'dozer') t = [c.scx, c.scy];
	else t = chaseTarget(g, c);
	let best = opts[0], bd = Infinity;
	opts.forEach(function (d) {
		const dd = dist2(tx + DX[d], ty + DY[d], t[0], t[1]);
		if (dd < bd) { bd = dd; best = d; }
	});
	c.dir = best;
}

function moveDirect(c, tx, ty, s) {
	const dx = tx - c.x, dy = ty - c.y;
	if (dx !== 0) { c.x += Math.sign(dx) * Math.min(s, Math.abs(dx)); return false; }
	if (dy !== 0) { c.y += Math.sign(dy) * Math.min(s, Math.abs(dy)); return false; }
	return true;
}

function updateChaser(g, c) {
	const s = chaserSpeed(g, c);
	if (c.state === 'house') {
		const base = 14 * TILE + HALF;
		c.y += c.dir === UP ? -64 : 64;
		if (c.y <= base - 3 * PX) c.dir = DOWN;
		if (c.y >= base + 3 * PX) c.dir = UP;
		const thresholds = { murk: 30, fizz: 60 };
		if (c.release > 0) c.release--;
		if (c.release <= 0 || (thresholds[c.kind] && g.lifeDots >= thresholds[c.kind] && g.level > 1)) c.state = 'leaving';
		return;
	}
	if (c.state === 'leaving') {
		if (c.x !== 14 * TILE) { moveDirect(c, 14 * TILE, c.y, s); return; }
		if (moveDirect(c, 14 * TILE, 11 * TILE + HALF, s)) {
			c.state = 'active';
			c.dir = g.rng.int(2) ? LEFT : RIGHT;
			if (c.kind === 'dozer') c.awake = 240;
		}
		return;
	}
	if (c.state === 'entering') {
		if (moveDirect(c, 14 * TILE, 14 * TILE + HALF, s)) {
			if (c.temp) { c.dead = true; return; }
			c.state = 'leaving';
			c.fright = false;
			c.release = 0;
		}
		return;
	}
	if (g.cfg.escape && c.state === 'active') {
		// patrol around the spawn point, hunt for a while after spotting BYTE
		if (dist2(tileOf(c.x), tileOf(c.y), tileOf(g.hero.x), tileOf(g.hero.y)) <= SIGHT) c.alert = 180;
		else if (c.alert > 0) c.alert--;
		if (--c.timer <= 0) {
			c.timer = 240;
			c.wx = c.hx - 10 + g.rng.int(21);
			c.wy = c.hy - 10 + g.rng.int(21);
		}
	} else if (c.kind === 'fizz' && --c.timer <= 0) {
		c.timer = 180;
		c.wx = g.rng.int(COLS);
		c.wy = g.rng.int(ROWS);
	}
	if (c.state === 'active' && c.kind === 'dozer') {
		if (c.awake > 0) c.awake--;
		else if (dist2(tileOf(c.x), tileOf(c.y), tileOf(g.hero.x), tileOf(g.hero.y)) <= 25) { c.awake = 480; g.events.push('wake'); }
		if (c.awake === 0 && centerOff(c.x) === 0 && centerOff(c.y) === 0) { c.dir = NONE; return; }
	}
	if (c.state === 'active' && c.kind === 'warp' && !c.fright) {
		c.timer++;
		c.warn = c.timer >= 300 ? 1 : 0;
		if (c.timer >= 360) {
			c.timer = 0;
			const t = randomOpenTile(g, g.chasers.filter(function (o) { return o !== c; }), 12);
			const hx = tileOf(g.hero.x), hy = tileOf(g.hero.y);
			if (t && dist2(t[0], t[1], hx, hy) >= 36) {
				c.x = t[0] * TILE + HALF;
				c.y = t[1] * TILE + HALF;
				c.dir = NONE;
				g.events.push('warp');
			}
		}
	}
	if (c.temp && c.state === 'active' && --c.ttl <= 0) { c.dead = true; return; }
	moveGrid(g.maze, c, s, function (e) { chaserDecide(g, e); });
	if (c.kind === 'munch' && c.state === 'active' && !c.fright) {
		const i = tileOf(c.y) * COLS + ((tileOf(c.x) % COLS) + COLS) % COLS;
		if (g.maze.dots[i] === 1) {
			g.maze.dots[i] = 0;
			g.dotsLeft--;
			g.score = Math.max(0, g.score - 10);
			g.events.push('steal');
		}
	}
}

function reverseAll(g) {
	g.chasers.forEach(function (c) { if (c.state === 'active' && c.dir) c.dir = OPP[c.dir]; });
}


/* ---------- scoring and pickups ---------- */

function addScore(g, pts) {
	g.score += g.fx.double > 0 ? pts * 2 : pts;
	if (!g.extra && g.score >= 10000) {
		g.extra = true;
		g.lives++;
		g.events.push('extra');
	}
}

function popup(g, x, y, text) {
	g.popups.push({ x: x, y: y, text: String(text), t: 90 });
}

function eatAt(g, tx, ty) {
	if (ty < 0 || ty >= g.maze.h || tx < 0 || tx >= g.maze.w) return;
	const i = ty * g.maze.w + tx, d = g.maze.dots[i];
	if (!d) return;
	g.maze.dots[i] = 0;
	g.dotsLeft--;
	g.dotsEaten++;
	g.lifeDots++;
	if (d === 1) {
		addScore(g, 10);
		g.events.push('dot');
	} else {
		addScore(g, 50);
		g.events.push('pill');
		const ft = g.level < FRIGHT.length ? FRIGHT[g.level] : 0;
		g.combo = 0;
		reverseAll(g);
		if (ft > 0) {
			g.frightT = ft;
			g.chasers.forEach(function (c) { if (c.state !== 'eyes' && c.state !== 'entering') c.fright = true; });
		}
	}
	if (!g.cfg.escape && (g.dotsEaten === 70 || g.dotsEaten === 170)) {
		const b = BONUS[Math.min(g.level, BONUS.length) - 1];
		g.fruit = { kind: b[0], value: b[1], x: 14 * TILE, y: 17 * TILE + HALF, t: 570 };
		g.events.push('fruitspawn');
	}
}

function eatChaser(g, c) {
	const pts = 200 * Math.pow(2, Math.min(g.combo, 4));
	g.combo++;
	g.feats.combo = Math.max(g.feats.combo, g.combo);
	addScore(g, pts);
	popup(g, c.x, c.y, g.fx.double > 0 ? pts * 2 : pts);
	c.state = 'eyes';
	c.fright = false;
	g.freezeT = 45;
	g.events.push('eat');
	if (c.kind === 'shard') {
		for (let k = 0; k < 2; k++) {
			const m = newChaser(g, 'mini', 0, 0);
			m.state = 'active';
			m.temp = true;
			m.ttl = 900;
			m.x = c.x - centerOff(c.x);
			m.y = c.y - centerOff(c.y);
			m.dir = k ? LEFT : RIGHT;
			m.scx = k ? 0 : 27;
			if (g.cfg.escape) { m.hx = m.wx = c.hx; m.hy = m.wy = c.hy; m.alert = 180; }
			g.chasers.push(m);
		}
		g.events.push('split');
	}
}

function spawnPickup(g) {
	let t = null;
	if (g.cfg.escape) {
		// escape: somewhere on BYTE's screen, not on top of him
		const hx = tileOf(g.hero.x), hy = tileOf(g.hero.y), m = g.maze;
		for (let i = 0; i < 20 && !t; i++) {
			const tx = hx - 12 + g.rng.int(25), ty = hy - 14 + g.rng.int(29);
			if (tx > 0 && ty > 0 && tx < m.w - 1 && ty < m.h - 1 && m.cells[ty * m.w + tx] === C_PATH && dist2(tx, ty, hx, hy) >= 16) t = [tx, ty];
		}
	} else t = randomOpenTile(g, [g.hero].concat(g.chasers), 20);
	if (!t) return;
	g.pickup = { type: PICKUPS[g.rng.int(PICKUPS.length)], x: t[0] * TILE + HALF, y: t[1] * TILE + HALF, t: 600 };
}

function usePickup(g, type) {
	const fx = g.fx, h = g.hero;
	g.events.push('power:' + type);
	popup(g, h.x, h.y, type.toUpperCase());
	if (type === 'boots') fx.boots = 480;
	else if (type === 'freeze') fx.freeze = 240;
	else if (type === 'magnet') fx.magnet = 480;
	else if (type === 'phase') fx.phase = 300;
	else if (type === 'double') fx.double = 600;
	else if (type === 'shield') fx.shield = true;
	else if (type === 'decoy') g.decoy = { tx: tileOf(h.x), ty: tileOf(h.y), x: h.x, y: h.y, t: 360 };
	else if (type === 'bomb') {
		g.chasers.forEach(function (c) {
			if (c.state !== 'active') return;
			if (dist2(tileOf(c.x), tileOf(c.y), tileOf(h.x), tileOf(h.y)) <= 36) {
				c.state = 'eyes';
				c.fright = false;
				addScore(g, 200);
				popup(g, c.x, c.y, 200);
			}
		});
	} else if (type === 'teleport') {
		const t = randomOpenTile(g, g.chasers.filter(function (c) { return c.state === 'active'; }), 30);
		if (t) {
			h.x = t[0] * TILE + HALF;
			h.y = t[1] * TILE + HALF;
			h.dir = NONE;
		}
	}
}


/* ---------- hunter: the player is one of the gang, four AI BYTEs eat the dots ---------- */

const ABILITY_CD = { razor: 0, vex: 360, murk: 480, fizz: 600 };
const PREY_SPEED = 75, MUNCH = 8; // hunter balance: BYTE speed as % of the hunter's (+1 per level, up to 95), ticks a BYTE stops on each dot

function resetHunt(g) {
	g.blocks.forEach(function (b) { g.maze.cells[b.y * g.maze.w + b.x] = C_PATH; });
	g.blocks = [];
	g.hero = { x: 14 * TILE, y: 11 * TILE + HALF, dir: LEFT, want: NONE, eyes: false, immune: false };
	g.fx = { boots: 0, freeze: 0, magnet: 0, phase: 0, double: 0, shield: false };
	g.chasers = [];
	g.hdKey = '';
	g.homeD = null;
	g.halfKey = -1;
	g.frightT = 0;
	g.abil = 0;
	g.huntT = 0;
	g.prey = [];
	for (let i = 0; i < 4; i++) {
		const t = randomOpenTile(g, [g.hero].concat(g.prey), 30);
		g.prey.push({ id: i, x: t[0] * TILE + HALF, y: t[1] * TILE + HALF, dir: NONE, caught: false, munch: 0 });
	}
}

/* breadth first walk over the open tiles from (sx, sy): steps to every tile, -1 where it can't reach */
function pathDist(g, sx, sy) {
	const m = g.maze, w = m.w, dist = new Int16Array(w * m.h).fill(-1);
	walkStart(w * m.h);
	const q = WALK.queue;
	let len = 1;
	q[0] = sy * w + sx;
	dist[q[0]] = 0;
	for (let qi = 0; qi < len; qi++) {
		const i = q[qi], x = i % w, y = (i / w) | 0;
		for (let d = 1; d <= 4; d++) {
			const ny = y + DY[d];
			if (ny < 0 || ny >= m.h) continue;
			const j = ny * w + (x + DX[d] + w) % w;
			if (dist[j] < 0 && m.cells[j] === C_PATH) { dist[j] = dist[i] + 1; q[len++] = j; }
		}
	}
	return dist;
}

/* the hunter's distance map, rebuilt only when the hunter changes tile or a wall block comes or goes */
function hunterDist(g) {
	const h = g.hero, key = tileOf(h.y) * g.maze.w + ((tileOf(h.x) % g.maze.w) + g.maze.w) % g.maze.w + ':' + g.blocks.length + ':' + g.level;
	if (g.hdKey !== key + ':' + h.dir) {
		g.hd = pathDist(g, ((tileOf(h.x) % g.maze.w) + g.maze.w) % g.maze.w, tileOf(h.y));
		g.hdNext = null;
		g.hdKey = key + ':' + h.dir;
	}
	return g.hd;
}

/* the same from the hunter's next tile, to tell who it's heading for; only built when a BYTE nearby asks */
function hunterNextDist(g) {
	if (!g.hdNext) {
		const next = hunterAhead(g, 1);
		g.hdNext = pathDist(g, next % g.maze.w, (next / g.maze.w) | 0);
	}
	return g.hdNext;
}

/* scratch buffers for the walks below, reused so a replay doesn't spend its time allocating: a tile is visited
 * this walk when its stamp matches the walk's number */
const WALK = { stamp: new Int32Array(0), first: new Int8Array(0), depth: new Int16Array(0), queue: new Int32Array(0), n: 0 };

function walkStart(size) {
	if (WALK.stamp.length < size) {
		WALK.stamp = new Int32Array(size);
		WALK.first = new Int8Array(size);
		WALK.depth = new Int16Array(size);
		WALK.queue = new Int32Array(size);
		WALK.n = 0;
	}
	return ++WALK.n;
}

/* walk from (tx, ty) toward the first tile goal() accepts, only through tiles ok(tile, steps) allows; returns the
 * first step or 0 */
function seek(g, tx, ty, goal, ok) {
	const m = g.maze, w = m.w, n = walkStart(w * m.h), stamp = WALK.stamp, first = WALK.first, depth = WALK.depth, q = WALK.queue;
	let len = 1;
	q[0] = ty * w + tx;
	stamp[q[0]] = n;
	depth[q[0]] = 0;
	for (let qi = 0; qi < len; qi++) {
		const i = q[qi], x = i % w, y = (i / w) | 0;
		if (qi > 0 && goal(i)) return first[i];
		for (let d = 1; d <= 4; d++) {
			const nx = (x + DX[d] + w) % w, ny = y + DY[d];
			if (ny < 0 || ny >= m.h) continue;
			const j = ny * w + nx;
			if (stamp[j] === n || m.cells[j] !== C_PATH || !ok(j, depth[i] + 1)) continue;
			stamp[j] = n;
			depth[j] = depth[i] + 1;
			first[j] = qi === 0 ? d : first[i];
			q[len++] = j;
		}
	}
	return 0;
}

/* room to run: tiles reachable from (nx, ny), one step away from the BYTE at `from`, that the BYTE gets to before
 * the hunter, up to 16 steps out. A dead end the hunter can close off has almost none */
function room(g, from, nx, ny, hd) {
	const m = g.maze, w = m.w, n = walkStart(w * m.h), stamp = WALK.stamp, depth = WALK.depth, q = WALK.queue;
	let len = 1, count = 0;
	q[0] = ny * w + nx;
	stamp[q[0]] = n;
	stamp[from] = n;
	depth[q[0]] = 1;
	for (let qi = 0; qi < len; qi++) {
		const i = q[qi], x = i % w, y = (i / w) | 0;
		count++;
		if (depth[i] >= 16) continue;
		for (let d = 1; d <= 4; d++) {
			const jx = (x + DX[d] + w) % w, jy = y + DY[d];
			if (jy < 0 || jy >= m.h) continue;
			const j = jy * w + jx;
			if (stamp[j] === n || m.cells[j] !== C_PATH || (hd[j] >= 0 && hd[j] <= depth[i] + 1)) continue;
			stamp[j] = n;
			depth[j] = depth[i] + 1;
			q[len++] = j;
		}
	}
	return count;
}

/* dots left in the left (0) or right (1) half of the maze, counted once per BYTE meal rather than per decision */
function halfDots(g, side) {
	if (g.halfKey !== g.dotsLeft) {
		const m = g.maze;
		g.halves = [0, 0];
		for (let i = 0; i < m.dots.length; i++) if (m.dots[i] === 1) g.halves[i % m.w < m.w / 2 ? 0 : 1]++;
		g.halfKey = g.dotsLeft;
	}
	return g.halves[side];
}

/* the tile up to n steps ahead of the hunter along its heading */
function hunterAhead(g, n) {
	const h = g.hero, w = g.maze.w;
	let x = ((tileOf(h.x) % w) + w) % w, y = tileOf(h.y);
	for (let k = 0; k < n && h.dir; k++) {
		const nx = (x + DX[h.dir] + w) % w, ny = y + DY[h.dir];
		if (!heroCan(g, nx, ny)) break;
		x = nx;
		y = ny;
	}
	return y * w + x;
}

/* BYTE roles: 0 and 1 gather opposite halves of the maze, 2 runs for a power pill when the hunter gets close to
 * anyone, 3 is bold and gathers right under the hunter's nose. Everyone flees toward the tiles they can reach
 * before the hunter (so never into a dead end the hunter can close), and grabs a pill instead when one is closer
 * to them than to the hunter. Powered up, they hunt the hunter together: two straight at it, two cutting it off. */
function preyDecide(g, p) {
	const m = g.maze, w = m.w, tx = ((tileOf(p.x) % w) + w) % w, ty = tileOf(p.y), here = ty * w + tx, h = g.hero;
	const hd = hunterDist(g), threat = !h.eyes, pills = function (i) { return m.dots[i] === 2; };
	const powered = g.frightT > 0 && !h.immune;
	let dir = 0;
	if (powered && threat) {
		const target = p.id === 1 ? hunterAhead(g, 4) : (p.id === 2 ? hunterAhead(g, 8) : hd.indexOf(0));
		dir = seek(g, tx, ty, function (i) { return i === target; }, function () { return true; });
	} else if (threat) {
		const near = hd[here], danger = p.id === 3 ? 5 : 9;
		let closing = near >= 0 && near <= 3;
		// the hunter is coming for us when its next step along its heading is closer to us than where it is now
		if (!closing && near >= 0 && near <= danger && h.dir) closing = hunterNextDist(g)[here] >= 0 && hunterNextDist(g)[here] < near;
		const teamNear = g.prey.some(function (o) { return !o.caught && hd[tileOf(o.y) * w + ((tileOf(o.x) % w) + w) % w] <= 12; });
		const first = function (i) { return hd[i] < 0 || near < 0; };
		if (near >= 0 && near <= danger && closing) {
			// beat the hunter to a pill if one is closer to us, otherwise run where the hunter can't cut us off
			dir = seek(g, tx, ty, pills, function (j, steps) { return first(j) || steps < hd[j]; });
			if (!dir) {
				let best = -1;
				for (let d = 1; d <= 4; d++) {
					const nx = (tx + DX[d] + w) % w, ny = ty + DY[d];
					if (ny < 0 || ny >= m.h || !heroCan(g, nx, ny)) continue;
					const n = ny * w + nx;
					if (hd[n] >= 0 && hd[n] <= 1) continue;
					const score = room(g, here, nx, ny, hd) * 64 + hd[n];
					if (score > best) { best = score; dir = d; }
				}
			}
		} else if (p.id === 2 && teamNear) {
			dir = seek(g, tx, ty, pills, function (j) { return hd[j] < 0 || hd[j] > 3; });
		}
		if (!dir) {
			// gather: nearest dot (in our half for the gatherers) along a path that keeps clear of the hunter
			const keep = p.id === 3 ? 2 : 5, half = function (i) { return p.id > 1 || (p.id === 0) === (i % w < w / 2); };
			const ok = function (j) { return m.dots[j] !== 2 && (hd[j] < 0 || hd[j] > keep); }; // pills are saved for emergencies
			if (p.id <= 1 && halfDots(g, p.id)) dir = seek(g, tx, ty, function (i) { return m.dots[i] && half(i); }, ok);
			dir = dir || seek(g, tx, ty, function (i) { return m.dots[i]; }, ok) ||
				seek(g, tx, ty, function (i) { return m.dots[i]; }, function (j) { return hd[j] < 0 || hd[j] > keep; }) || // only pills left
				seek(g, tx, ty, function (i) { return m.dots[i]; }, function () { return true; }); // only dots by the hunter left: risk it
		}
	} else dir = seek(g, tx, ty, function (i) { return m.dots[i] === 1; }, function (j) { return m.dots[j] !== 2; }) || seek(g, tx, ty, function (i) { return m.dots[i]; }, function () { return true; });
	if (dir) { p.dir = dir; return; }
	// nothing to do: keep going, or turn wherever is open
	if (p.dir && heroCan(g, tx + DX[p.dir], ty + DY[p.dir])) return;
	p.dir = NONE;
	for (let d = 1; d <= 4; d++) if (heroCan(g, tx + DX[d], ty + DY[d])) { p.dir = d; break; }
}

/* the hunter's eyes take the shortest way home to the house door */
function eyesDecide(g, h) {
	const w = g.maze.w, tx = ((tileOf(h.x) % w) + w) % w, ty = tileOf(h.y);
	if (!g.homeD) g.homeD = pathDist(g, 14, 11);
	let best = h.dir, bd = 1e9;
	for (let d = 1; d <= 4; d++) {
		const nx = (tx + DX[d] + w) % w, ny = ty + DY[d], v = ny >= 0 && ny < g.maze.h ? g.homeD[ny * w + nx] : -1;
		if (v >= 0 && v < bd && heroCan(g, nx, ny)) { bd = v; best = d; }
	}
	h.dir = best;
}

function useAbility(g) {
	const h = g.hero, k = g.cfg.member, tx = tileOf(h.x), ty = tileOf(h.y);
	if (k === 'vex') {
		// dash up to 4 tiles straight ahead
		if (!h.dir) return;
		let n = 0;
		while (n < 4 && heroCan(g, tx + DX[h.dir] * (n + 1), ty + DY[h.dir] * (n + 1))) n++;
		if (!n) return;
		h.x = wrapX(g.maze, (tx + DX[h.dir] * n) * TILE + HALF);
		h.y = (ty + DY[h.dir] * n) * TILE + HALF;
	} else if (k === 'murk') {
		// drop a wall block on the tile behind for 5 seconds
		const bx = ((tx - DX[h.dir] + g.maze.w) % g.maze.w), by = ty - DY[h.dir];
		if (!h.dir || cellAt(g.maze, bx, by) !== C_PATH || g.maze.tunnel[by * g.maze.w + bx]) return;
		const busy = g.prey.some(function (p) { return !p.caught && Math.abs(wrapDelta(g.maze, p.x - (bx * TILE + HALF))) < TILE && Math.abs(p.y - (by * TILE + HALF)) < TILE; });
		if (busy) return;
		g.maze.cells[by * g.maze.w + bx] = C_WALL;
		g.blocks.push({ x: bx, y: by, t: 300 });
	} else if (k === 'fizz') {
		const t = randomOpenTile(g, [h], 12);
		if (!t) return;
		h.x = t[0] * TILE + HALF;
		h.y = t[1] * TILE + HALF;
		h.dir = NONE;
	} else return;
	g.abil = ABILITY_CD[k];
	g.events.push(k === 'fizz' ? 'warp' : 'power:' + k);
}

function catchPrey(g) {
	const h = g.hero;
	if (h.eyes) return false;
	for (let i = 0; i < g.prey.length; i++) {
		const p = g.prey[i];
		if (p.caught || Math.abs(wrapDelta(g.maze, p.x - h.x)) >= HIT || Math.abs(p.y - h.y) >= HIT) continue;
		if (g.frightT > 0 && !h.immune) {
			// caught by a powered BYTE: costs a life, and the hunter's eyes race home while the BYTEs carry on
			if (g.lives <= 1) {
				g.phase = 'dying';
				g.phaseT = 150;
				g.events.push('death');
				return true;
			}
			g.lives--;
			g.clean = false;
			h.eyes = true;
			h.want = NONE;
			g.homeD = null;
			popup(g, p.x, p.y, 'GOTCHA');
			g.events.push('eat');
			return false;
		}
		const n = g.prey.filter(function (q) { return q.caught; }).length, pts = 200 << n;
		p.caught = true;
		g.feats.catches++;
		addScore(g, pts);
		popup(g, p.x, p.y, pts);
		g.events.push('eat');
	}
	return false;
}

function hunterTick(g) {
	const h = g.hero, sp = speeds(g.level), preyS = (sp.hero * Math.min(95, PREY_SPEED + g.level) / 100) | 0;
	g.huntT++;
	if (g.frightT > 0 && --g.frightT === 0) h.immune = false;
	if (g.abil > 0) g.abil--;
	g.blocks = g.blocks.filter(function (b) {
		if (--b.t > 0) return true;
		g.maze.cells[b.y * g.maze.w + b.x] = C_PATH;
		return false;
	});
	if (h.eyes) {
		moveGrid(g.maze, h, 480, function (e) { eyesDecide(g, e); });
		if (tileOf(h.y) === 11 && (tileOf(h.x) === 13 || tileOf(h.x) === 14)) {
			h.x = 14 * TILE;
			h.y = 11 * TILE + HALF;
			h.dir = LEFT;
			h.eyes = false;
			h.immune = g.frightT > 0; // back from the house the hunter is a threat again, like an eaten chaser
		}
	} else {
		if (g.abilityIn && g.abil === 0 && g.cfg.member !== 'razor') useAbility(g);
		if (h.want && h.want === OPP[h.dir]) h.dir = h.want;
		lateTurn(g, h);
		moveGrid(g.maze, h, g.cfg.member === 'razor' ? (sp.hero * 115 / 100) | 0 : sp.hero, function (e) { heroDecide(g, e); });
	}
	if (catchPrey(g)) return;
	for (let i = 0; i < g.prey.length; i++) {
		const p = g.prey[i];
		if (p.caught) continue;
		if (p.munch > 0) { p.munch--; continue; }
		const ti = tileOf(p.y) * g.maze.w + ((tileOf(p.x) % g.maze.w) + g.maze.w) % g.maze.w;
		moveGrid(g.maze, p, g.maze.tunnel[ti] ? sp.tunnel : preyS, function (e) { preyDecide(g, e); });
		const j = tileOf(p.y) * g.maze.w + ((tileOf(p.x) % g.maze.w) + g.maze.w) % g.maze.w, d = g.maze.dots[j];
		if (!d) continue;
		g.maze.dots[j] = 0;
		g.dotsLeft--;
		p.munch = MUNCH;
		if (d === 2) {
			g.frightT = Math.max(120, 360 - 30 * (g.level - 1));
			h.immune = false;
			g.events.push('pill');
		}
	}
	if (catchPrey(g)) return;
	if (g.prey.every(function (p) { return p.caught; })) {
		const secs = Math.floor(g.huntT / TPS), pts = 1000 + Math.max(0, 90 - secs) * 10;
		g.feats.fastClear = g.feats.fastClear ? Math.min(g.feats.fastClear, Math.max(1, secs)) : Math.max(1, secs);
		addScore(g, pts);
		popup(g, h.x, h.y, pts);
		g.phase = 'clear';
		g.phaseT = 150;
		g.events.push('clear');
	} else if (g.dotsLeft <= 0) {
		g.phase = 'dying';
		g.phaseT = 150;
		g.events.push('death');
	}
}


/* ---------- per tick ---------- */

function heroDecide(g, h) {
	const tx = tileOf(h.x), ty = tileOf(h.y);
	if (h.want && heroCan(g, tx + DX[h.want], ty + DY[h.want])) h.dir = h.want;
	else if (h.dir && !heroCan(g, tx + DX[h.dir], ty + DY[h.dir])) h.dir = NONE;
}

/* a turn pressed up to 3 pixels after passing a tile center snaps back and takes it */
function lateTurn(g, h) {
	if (!h.want || !h.dir || h.want === h.dir || h.want === OPP[h.dir]) return;
	const horiz = h.dir === LEFT || h.dir === RIGHT, off = centerOff(horiz ? h.x : h.y);
	const past = h.dir === RIGHT || h.dir === DOWN ? off : (off ? TILE - off : 0);
	if (past === 0 || past > 3 * PX) return;
	const tx = tileOf(h.x), ty = tileOf(h.y);
	if (!heroCan(g, tx + DX[h.want], ty + DY[h.want])) return;
	if (horiz) h.x = tx * TILE + HALF;
	else h.y = ty * TILE + HALF;
	h.dir = h.want;
}

function collide(g) {
	const h = g.hero;
	for (let i = 0; i < g.chasers.length; i++) {
		const c = g.chasers[i];
		if (c.state !== 'active') continue;
		if (Math.abs(wrapDelta(g.maze, c.x - h.x)) >= HIT || Math.abs(c.y - h.y) >= HIT) continue;
		if (c.fright) { eatChaser(g, c); continue; }
		if (g.fx.freeze > 0 || (c.kind === 'dozer' && c.awake === 0)) continue;
		if (g.fx.shield) {
			g.fx.shield = false;
			c.state = 'eyes';
			g.events.push('shield');
			continue;
		}
		g.phase = 'dying';
		g.phaseT = 150;
		g.events.push('death');
		return true;
	}
	return false;
}

function playTick(g) {
	const h = g.hero, fx = g.fx;
	if (g.freezeT > 0) {
		g.freezeT--;
		g.chasers.forEach(function (c) { if (c.state === 'eyes' || c.state === 'entering') updateChaser(g, c); });
		return;
	}
	if (g.frightT > 0) {
		if (--g.frightT === 0) g.chasers.forEach(function (c) { c.fright = false; });
	} else if (!g.cfg.escape) {
		const tbl = modeTable(g.level);
		if (g.modeT > 0 && --g.modeT === 0 && g.modeIdx + 1 < tbl.length) {
			g.modeIdx++;
			g.modeT = tbl[g.modeIdx];
			reverseAll(g);
		}
	}
	if (g.fruit && --g.fruit.t <= 0) g.fruit = null;
	if (fx.boots > 0) fx.boots--;
	if (fx.freeze > 0) fx.freeze--;
	if (fx.magnet > 0) fx.magnet--;
	if (fx.double > 0) fx.double--;
	if (fx.phase > 0) {
		fx.phase--;
		if (fx.phase === 0 && cellAt(g.maze, tileOf(h.x), tileOf(h.y)) !== C_PATH) fx.phase = 1;
	}
	if (g.decoy && --g.decoy.t <= 0) g.decoy = null;
	if (g.cfg.mayhem) {
		if (g.pickup) { if (--g.pickup.t <= 0) g.pickup = null; }
		else if (--g.pickupT <= 0) { g.pickupT = 600; spawnPickup(g); }
	}
	if (g.cfg.shifting && g.gates.length) {
		if (--g.gateT <= 0) {
			g.gateT = 420;
			g.gateGroup ^= 1;
			g.events.push('gate');
		}
		applyGates(g, false);
	}
	// hero
	if (h.want && h.want === OPP[h.dir]) h.dir = h.want;
	lateTurn(g, h);
	const sp = speeds(g.level);
	let hs = g.frightT > 0 ? sp.heroF : sp.hero;
	if (fx.boots > 0) hs += (hs * 3 / 10) | 0;
	moveGrid(g.maze, h, hs, function (e) { heroDecide(g, e); });
	const htx = ((tileOf(h.x) % g.maze.w) + g.maze.w) % g.maze.w, hty = tileOf(h.y);
	eatAt(g, htx, hty);
	if (fx.magnet > 0) for (let dy = -3; dy <= 3; dy++) for (let dx = -3; dx <= 3; dx++) eatAt(g, htx + dx, hty + dy);
	if (g.fruit && Math.abs(g.fruit.x - h.x) < HIT && Math.abs(g.fruit.y - h.y) < HIT) {
		addScore(g, g.fruit.value);
		popup(g, g.fruit.x, g.fruit.y, g.fx.double > 0 ? g.fruit.value * 2 : g.fruit.value);
		g.events.push('fruit');
		g.fruit = null;
	}
	if (g.pickup && Math.abs(g.pickup.x - h.x) < HIT && Math.abs(g.pickup.y - h.y) < HIT) {
		const t = g.pickup.type;
		g.pickup = null;
		usePickup(g, t);
	}
	if (collide(g)) return;
	// chasers
	const frozen = fx.freeze > 0;
	for (let i = 0; i < g.chasers.length; i++) {
		const c = g.chasers[i];
		if (frozen && c.state === 'active') continue;
		updateChaser(g, c);
	}
	if (g.chasers.some(function (c) { return c.dead; })) g.chasers = g.chasers.filter(function (c) { return !c.dead; });
	if (collide(g)) return;
	if (g.cfg.escape) {
		g.escT++;
		if (htx === g.maze.exit[0] && hty === g.maze.exit[1]) {
			g.feats.escLeft = Math.max(g.feats.escLeft, escapeTime(g));
			const pts = 1000 * g.level + escapeBonus(g);
			addScore(g, pts);
			popup(g, h.x, h.y, pts);
			g.phase = 'clear';
			g.phaseT = 150;
			g.events.push('clear');
		}
	} else if (g.cfg.survival) {
		// survival: 10 points a second, one more chaser every 30s up to 10, a little faster every minute, dots refill
		g.survT++;
		if (g.survT % TPS === 0) { addScore(g, 10); g.feats.survSecs = g.survT / TPS; }
		if (g.survT % (TPS * 30) === 0 && 4 + g.survAdded < 10) {
			g.survAdded++;
			g.chasers.push(survivor(g, 0));
			g.events.push('wake');
		}
		if (g.survT % (TPS * 60) === 0 && g.level < 20) g.level++;
		if (g.dotsLeft <= 0) {
			const fresh = mazeFor(g.cfg, g.seed, 1);
			g.maze.dots.set(fresh.dots);
			g.dotsLeft = fresh.dotCount;
			g.dotsEaten = 0;
			g.events.push('clear');
		}
	} else if (g.dotsLeft <= 0) {
		g.phase = 'clear';
		g.phaseT = 150;
		g.events.push('clear');
	}
}

function step(g, input) {
	g.events = [];
	if (g.over) return;
	g.tick++;
	if ((input & 7) >= 1 && (input & 7) <= 4) g.hero.want = input & 7;
	g.abilityIn = (input & 8) !== 0;
	if (g.popups.length) g.popups = g.popups.filter(function (p) { return --p.t > 0; });
	if (g.tick >= MAX_TICKS) {
		g.over = true;
		g.phase = 'over';
		return;
	}
	if (g.phase === 'play') { if (g.cfg.hunter) hunterTick(g); else playTick(g); return; }
	if (--g.phaseT > 0) return;
	if (g.phase === 'start' || g.phase === 'ready') {
		g.phase = 'play';
	} else if (g.phase === 'dying') {
		g.clean = false;
		g.lives--;
		if (g.lives <= 0) {
			g.over = true;
			g.phase = 'over';
			g.events.push('gameover');
		} else {
			if (g.cfg.hunter) loadLevel(g); // the BYTEs start the level over with every dot back
			else resetActors(g);
			g.phase = 'ready';
			g.phaseT = 120;
		}
	} else if (g.phase === 'clear') {
		if (g.level >= 5 && g.clean) g.feats.flawless = 1;
		if (INTERMISSIONS[g.level]) {
			g.phase = 'intermission';
			g.interKind = INTERMISSIONS[g.level];
			g.phaseT = 480;
			g.events.push('intermission');
		} else nextLevel(g);
	} else if (g.phase === 'intermission') {
		nextLevel(g);
	}
}

function nextLevel(g) {
	g.level++;
	loadLevel(g);
	g.phase = 'ready';
	g.phaseT = 120;
	g.events.push('level');
}


/* ---------- replay and hashing ---------- */

function replay(opts, runs) {
	const g = create(opts);
	for (let i = 0; i < runs.length && !g.over; i++) {
		const dir = runs[i][0] | 0;
		let n = runs[i][1] | 0;
		while (n-- > 0 && !g.over) step(g, dir);
	}
	return { score: g.score, level: g.level, ticks: g.tick, over: g.over, feats: g.feats };
}

function hash(g) {
	let h = 2166136261 >>> 0;
	const mix = function (v) { h = Math.imul(h ^ (v | 0), 16777619) >>> 0; };
	mix(g.tick); mix(g.score); mix(g.lives); mix(g.level); mix(g.dotsLeft); mix(g.rng.s);
	mix(g.hero.x); mix(g.hero.y); mix(g.hero.dir);
	g.chasers.forEach(function (c) { mix(c.x); mix(c.y); mix(c.dir); mix(c.fright ? 1 : 0); });
	g.prey.forEach(function (p) { mix(p.x); mix(p.y); mix(p.caught ? 1 : 0); });
	return h >>> 0;
}

/* escape: seconds left on the level clock, worth 20 points each when BYTE reaches the exit */
function escapeTime(g) {
	return Math.max(0, g.maze.par - Math.floor(g.escT / TPS));
}

function escapeBonus(g) {
	return escapeTime(g) * 20;
}

function bonusFor(level) {
	return BONUS[Math.min(level, BONUS.length) - 1];
}

root.MazeCore = {
	TILE: TILE, HALF: HALF, PX: PX, COLS: COLS, ROWS: ROWS, TPS: TPS, MAX_TICKS: MAX_TICKS,
	NONE: NONE, UP: UP, LEFT: LEFT, DOWN: DOWN, RIGHT: RIGHT, DX: DX, DY: DY, OPP: OPP,
	C_PATH: C_PATH, C_WALL: C_WALL, C_HOUSE: C_HOUSE, C_DOOR: C_DOOR,
	ABILITY_CD: ABILITY_CD, MODES: MODES, TWISTS: TWISTS, GANG: GANG, SPECIALS: SPECIALS, PICKUPS: PICKUPS, MAZES: MAZES, PACK: PACK,
	Rng: Rng, bonusFor: bonusFor, cellAt: cellAt, create: create, escapeMaze: escapeMaze, escapeTime: escapeTime, hash: hash, heroCan: heroCan,
	mazeFor: mazeFor, normalizeTwists: normalizeTwists, parseMaze: parseMaze, proceduralMaze: proceduralMaze,
	replay: replay, resolveConfig: resolveConfig, step: step, tileOf: tileOf
};

})(typeof globalThis !== 'undefined' ? globalThis : this);

	const atlases = {};

	function atlas(color) {
		if (atlases[color]) return atlases[color];
		const c = document.createElement('canvas');
		c.width = ORDER.length * 8;
		c.height = 8;
		const x = c.getContext('2d');
		x.fillStyle = color;
		ORDER.forEach(function (ch, n) {
			G[ch].split('|').forEach(function (row, y) {
				for (let i = 0; i < 5; i++) if (row[i] === '#') x.fillRect(n * 8 + i + 1, y, 1, 1);
			});
		});
		atlases[color] = c;
		return c;
	}

	function draw(ctx, text, x, y, color, scale) {
		const a = atlas(color || '#fff'), s = scale || 1;
		text = String(text).toUpperCase();
		for (let i = 0; i < text.length; i++) {
			const n = ORDER.indexOf(text[i]);
			if (n >= 0) ctx.drawImage(a, n * 8, 0, 8, 8, Math.round(x + i * 8 * s), Math.round(y), 8 * s, 8 * s);
		}
	}

	function center(ctx, text, y, color, scale, width) {
		const s = scale || 1;
		draw(ctx, text, Math.round(((width || 224) - String(text).length * 8 * s) / 2), y, color, s);
	}

	function glyph(ch) {
		return G[ch] ? G[ch].split('|') : null;
	}

	return { center: center, draw: draw, glyph: glyph };

/* acid-maze - keyboard and floating touch joystick input with a sticky direction buffer */

const Input = (function () {
	'use strict';

	const KEYS = {
		ArrowUp: 1, KeyW: 1, ArrowLeft: 2, KeyA: 2, ArrowDown: 3, KeyS: 3, ArrowRight: 4, KeyD: 4
	};
	let dir = 0;
	const handlers = {};
	const queue = [];

	function emit(name, arg) {
		(handlers[name] || []).forEach(function (fn) { fn(arg); });
	}

	function on(name, fn) {
		(handlers[name] = handlers[name] || []).push(fn);
	}

	function press(d) {
		dir = d;
		queue.push(d);
		emit('dir', d);
	}

	/* floating joystick: wherever a thumb lands becomes the stick's center, sliding past a small dead zone steers,
	   and the center trails the thumb so it can roll straight into a new direction. A quick flick works as a swipe,
	   and a tap on the screen starts the game */
	function initStick(area, screen, ring) {
		const DEAD = 10, REACH = 34;
		let id = null, cx = 0, cy = 0, held = 0, moved = false, onScreen = false;
		const place = function (kx, ky) {
			ring.style.transform = 'translate(' + cx + 'px,' + cy + 'px)';
			ring.firstElementChild.style.transform = 'translate(' + kx + 'px,' + ky + 'px)';
		};
		area.addEventListener('touchstart', function (e) {
			if (id !== null || e.target.closest('button, a, input, select, .table-wrap, .boards, .account-menu')) return;
			const t = e.changedTouches[0];
			id = t.identifier;
			cx = t.clientX;
			cy = t.clientY;
			held = 0;
			moved = false;
			onScreen = screen.contains(e.target);
			place(0, 0);
			ring.hidden = false;
		}, { passive: true });
		area.addEventListener('touchmove', function (e) {
			const t = Array.prototype.find.call(e.changedTouches, function (c) { return c.identifier === id; });
			if (!t) return;
			e.preventDefault();
			let dx = t.clientX - cx, dy = t.clientY - cy;
			const len = Math.hypot(dx, dy);
			if (len > REACH) { cx += dx * (1 - REACH / len); cy += dy * (1 - REACH / len); dx = t.clientX - cx; dy = t.clientY - cy; }
			place(dx, dy);
			if (len < DEAD) return;
			moved = true;
			const ax = Math.abs(dx), ay = Math.abs(dy);
			if (held && Math.max(ax, ay) < Math.min(ax, ay) * 1.25) return; // near a diagonal keep the current direction
			const d = ax > ay ? (dx < 0 ? 2 : 4) : (dy < 0 ? 1 : 3);
			if (d !== held) { held = d; press(d); }
		}, { passive: false });
		const end = function (e) {
			if (!Array.prototype.some.call(e.changedTouches, function (c) { return c.identifier === id; })) return;
			id = null;
			ring.hidden = true;
			if (e.type === 'touchend' && onScreen && !moved) { e.preventDefault(); emit('tap'); }
		};
		area.addEventListener('touchend', end);
		area.addEventListener('touchcancel', end);
		area.addEventListener('contextmenu', function (e) { e.preventDefault(); });
		document.addEventListener('touchmove', function (e) { // no page pan or bounce, the score list still scrolls
			if (!e.target.closest('.table-wrap, .account-menu')) e.preventDefault();
		}, { passive: false });
	}

	function init(screen) {
		window.addEventListener('keydown', function (e) {
			if (e.target && (e.target.tagName === 'INPUT' || e.target.tagName === 'SELECT')) return;
			if (KEYS[e.code]) {
				e.preventDefault();
				if (!e.repeat) press(KEYS[e.code]);
				return;
			}
			if (e.code === 'Enter' || e.code === 'Space') { e.preventDefault(); emit('start'); }
			else if (e.code === 'KeyC' || e.code === 'Digit5') emit('coin');
			else if (e.code === 'KeyM') emit('mute');
			else if (e.code === 'KeyV') emit('crt');
			else if (e.code === 'KeyP') emit('pause');
			else if (e.code === 'Escape') emit('menu');
			else if (e.code === 'Backspace') { e.preventDefault(); emit('erase'); }
			else if (/^Key[A-Z]$|^Digit[0-9]$/.test(e.code)) emit('char', e.code.slice(-1));
		});
		initStick(document.querySelector('.room'), screen, document.getElementById('thumbstick'));
		document.querySelectorAll('[data-act]').forEach(function (b) {
			b.addEventListener('click', function (e) { e.preventDefault(); b.blur(); emit(b.dataset.act); });
		});
	}

	return {
		current: function () { return dir; },
		drain: function () { return queue.splice(0, queue.length); },
		init: init, on: on, press: press,
		reset: function () { dir = 0; queue.length = 0; }
	};
})();

				Sfx.play('select');
			} else if (currentMode() === 'custom') {
				app.customSel = (app.customSel + (d === 2 ? -1 : 1) + M.TWISTS.length + 1) % (M.TWISTS.length + 1);
				Sfx.play('select');
			}
		} else if (app.state === 'initials') {
			const i = app.initialsIdx, c = CHARS.indexOf(app.initials[i]);
			if (d === 1 || d === 3) {
				app.initials[i] = CHARS[(c + (d === 1 ? 1 : -1) + CHARS.length) % CHARS.length];
				Sfx.play('select');
			} else if (d === 4) advanceInitial();
			else if (d === 2 && i > 0) app.initialsIdx--;
		}
	}

	function onChar(ch) {
		if (app.state !== 'initials' || CHARS.indexOf(ch) < 0) return;
		app.initials[app.initialsIdx] = ch;
		advanceInitial();
	}

	/* ---------- update ---------- */

	function playEvents(evs) {
		evs.forEach(function (e) { Sfx.play(e); });
	}

	function updateSiren(g) {
		if (g.phase !== 'play') { Sfx.setSiren('off'); return; }
		let mode = 'chase';
		if (g.chasers.some(function (c) { return c.state === 'eyes' || c.state === 'entering'; }) || (g.cfg.hunter && g.hero.eyes)) mode = 'eyes';
		else if (g.frightT > 0) mode = 'fright';
		let level = 1 - g.dotsLeft / Math.max(1, g.maze.dotCount);
		if (g.cfg.escape && mode === 'chase') {
			// escape: the siren only sounds while someone hunts BYTE, rising as the nearest hunter closes in
			let near = 1e9;
			g.chasers.forEach(function (c) {
				if (c.state === 'active' && c.alert > 0) near = Math.min(near, Math.hypot(M.tileOf(c.x) - M.tileOf(g.hero.x), M.tileOf(c.y) - M.tileOf(g.hero.y)));
			});
			if (near > 1e8) mode = 'off';
			level = Math.max(0, 1 - near / 12);
		}
		if (mode !== Sfx.sirenMode() || app.t % 30 === 0) Sfx.setSiren(mode, level);
	}

	function newDemo() {
		const modes = ['classic', 'mayhem', 'pack', 'procedural', 'shifting', 'escape'];
		const mode = modes[app.demoMode++ % modes.length];
		app.demo = M.create({ mode: mode, seed: 1000 + app.demoMode * 7, twists: '' });
		app.demo.phase = 'ready';
		app.demo.phaseT = 60;
	}

	function update() {
		app.t++;
		Input.drain().forEach(onDir);
		if (app.state === 'attract') {
			app.at++;
			if (app.at === 1360) newDemo();
			if (app.at > 1360 && app.demo) {
				M.step(app.demo, botDir(app.demo));
				if (app.demo.phase === 'dying' && app.demo.phaseT === 1) app.at = Math.max(app.at, 3000);
				if (app.demo.over || app.demo.phase === 'clear') app.at = Math.max(app.at, 3000);
			}
			if (app.at === 3000) { app.boardMode = M.MODES[(app.demoMode + 5) % M.MODES.length]; loadBoard(app.boardMode, 'all'); }
			if (app.at >= 3420) app.at = 0;
		} else if (app.state === 'play') {
			if (app.paused) return;
			const g = app.game;
			const d = BOT ? botDir(g, BOT_DIE > 0 && g.tick > BOT_DIE * 60) : Input.current();
			const v = d | (app.ability ? 8 : 0); // bit 8: hunter ability pressed this tick
			app.ability = false;
			logInput(v);
			M.step(g, v);
			playEvents(g.events);
			updateSiren(g);
			if (g.over) finishRun();
		} else if (app.state === 'preview') {
			if (app.game.phase === 'intermission') M.step(app.game, 0);
		} else if (app.state === 'verify') {
			app.at++;
			if (app.result && app.at > 240) afterVerify();
		} else if (app.state === 'initials') {
			if (++app.initialsT > 60 * 30) { app.initialsIdx = 3; submitInitials(); }
		} else if (app.state === 'board') {
			if (++app.boardT > 600) toAttract();
		}
	}

	function syncToggles() {
		try {
			localStorage.setItem('acid-maze-crt', app.crt ? '1' : '0');
			localStorage.setItem('acid-maze-curve', app.curve ? '1' : '0');
		} catch (e) { /* storage blocked */ }
		R.setCrt(app.crt, app.curve);
		const set = function (key, on, label) { document.querySelectorAll('[data-toggle="' + key + '"]').forEach(function (el) { el.classList.toggle('on', on); el.textContent = label + (on ? ' ON' : ' OFF'); }); };
		set('mute', !Sfx.isMuted(), 'SOUND');
		set('crt', app.crt, 'CRT');
		set('curve', app.curve, 'CURVE');
	}

	/* ---------- marquee: the title drawn in our own bitmap font as SVG pixels ---------- */

	function buildMarquee() {
		const svg = document.getElementById('marquee-svg');
		if (!svg) return;
		const ns = 'http://www.w3.org/2000/svg', text = 'ACID MAZE', px = 7;
		const ox = (400 - (text.length * 6 - 1) * px) / 2, oy = 12;
		let rects = '';
		for (let i = 0; i < text.length; i++) {
			const rows = Font.glyph(text[i]);
			if (!rows) continue;
			rows.forEach(function (row, y) {
				for (let x = 0; x < 5; x++) if (row[x] === '#') rects += '<rect x="' + (ox + (i * 6 + x) * px) + '" y="' + (oy + y * px) + '" width="' + (px - 1) + '" height="' + (px - 1) + '"/>';
			});
		}
		svg.innerHTML = '<defs><linearGradient id="mq" x1="0" y1="0" x2="1" y2="1"><stop offset="0" stop-color="#ff2fd6"/><stop offset="0.5" stop-color="#ffd34a"/><stop offset="1" stop-color="#2fe6ff"/></linearGradient>' +
			'<filter id="neon" x="-20%" y="-40%" width="140%" height="180%"><feGaussianBlur stdDeviation="3" result="b"/><feMerge><feMergeNode in="b"/><feMergeNode in="b"/><feMergeNode in="SourceGraphic"/></feMerge></filter></defs>' +
			'<g fill="#3a0a38" transform="translate(3 3)">' + rects + '</g><g fill="url(#mq)" filter="url(#neon)">' + rects + '</g>';
		svg.setAttribute('xmlns', ns);
	}

	/* ---------- boot ---------- */

	function boot() {
		const screen = document.getElementById('screen');
		R.init(screen);
		Page.init();
		buildMarquee();
		app.crt = !Page.touch(); // the CRT pass is slow on phones, so it starts off there
		try {
			const crt = localStorage.getItem('acid-maze-crt'), curve = localStorage.getItem('acid-maze-curve');
			if (crt !== null) app.crt = crt !== '0';
			if (curve !== null) app.curve = curve !== '0';
		} catch (e) { /* storage blocked */ }
		Input.init(document.getElementById('screen-wrap'));
		Input.on('coin', onCoin);
		Input.on('start', onStart);
		Input.on('tap', onStart);
		Input.on('menu', onMenu);
		Input.on('char', onChar);
		Input.on('erase', function () { if (app.state === 'initials') { if (app.initialsIdx > 0) app.initialsIdx--; } else onMenu(); });
		Input.on('mute', function () { Sfx.setMuted(!Sfx.isMuted()); syncToggles(); });
		Input.on('crt', function () { app.crt = !app.crt; syncToggles(); });
		Input.on('curve', function () { app.curve = !app.curve; syncToggles(); });
		Input.on('pause', function () { setPaused(!app.paused); });
		document.addEventListener('visibilitychange', function () { if (document.visibilityState === 'hidden') setPaused(true); });
		syncToggles();
		initPanel();
		M.MODES.forEach(function (m) { loadBoard(m, 'all'); });
		api('/api/daily').then(function (j) { app.daily = j; }).catch(function () { return null; });
		const scene = params.get('scene');
		if (scene === 'cast') app.at = 1100;
		else if (scene === 'scores') { app.at = 3001; app.boardMode = params.get('mode') || 'classic'; }
		else if (scene === 'demo') app.at = 1355;
		else if (scene === 'select') app.state = 'select';
		else if (scene === 'intermission') {
			app.state = 'preview';
			app.run = { mode: 'classic' };
			app.game = M.create({ mode: 'classic', seed: 1 });
			app.game.phase = 'intermission';
			app.game.interKind = +params.get('k') || 1;
			app.game.phaseT = 480 - (+params.get('at') || 120);
		}
		const auto = params.get('mode');
		if (params.get('autostart') === '1' && auto && M.MODES.indexOf(auto) >= 0) {
			app.modeIdx = M.MODES.indexOf(auto);
			if (auto === 'custom') {
				const tw = (params.get('twists') || '').split('+');
				M.TWISTS.forEach(function (k) { app.custom[k] = tw.indexOf(k) >= 0; });
			}
			app.credits = 1;
			startRun();
		}
		let last = performance.now(), acc = 0;
		const frame = function (now) {
			acc += Math.min(250, now - last);
			last = now;
			let n = 0;
			while (acc >= STEP && n < 12) { update(); acc -= STEP; n++; }
			if (n >= 12) acc = 0;
			draw();
			requestAnimationFrame(frame);
		};

			if (value === null) localStorage.removeItem(key);
			else localStorage.setItem(key, value);
		} catch (e) { /* storage blocked */ }

		let w = Math.floor(Math.min(availW, availH * 7 / 9));

			if (over <= 0) break;
			w = Math.floor(w - over - 1);

		const k = w / 224;
		if (k >= 2 && Math.floor(k) / k >= 0.9) w = Math.floor(k) * 224; // whole pixel scaling when it costs little
		wrap.style.width = Math.max(120, w) + 'px';
		Render.resize();
	}

		return fetch(path, { method: post ? 'POST' : 'GET', headers: { 'X-Auth-Token': token } }).then(function (r) {
			return r.json().catch(function () { return {}; }).then(function (data) { return { ok: r.ok, status: r.status, data: data }; });

	function buildMaze(g) {
		const m = g.maze, mw = m.w * 8, mh = m.h * 8;
		const c = canvas(mw, mh), x = c.getContext('2d'), img = x.createImageData(mw, mh);

			if (ty < 0 || ty >= m.h) return true;
			if (tx < 0 || tx >= m.w) return M.cellAt(m, tx, ty) === M.C_WALL;
			return m.cells[ty * m.w + tx] === M.C_WALL;
		};
		const solid = new Uint8Array(mw * mh), INSET = 3;
		for (let ty = 0; ty < m.h; ty++) for (let tx = 0; tx < m.w; tx++) {
			if (!isWall(tx, ty)) continue;
			for (let py = 0; py < 8; py++) for (let px = 0; px < 8; px++) {
				let ok = true;
				for (let ny = -1; ny <= 1 && ok; ny++) for (let nx = -1; nx <= 1 && ok; nx++) {
					if ((nx || ny) && !isWall(tx + nx, ty + ny)) {
						const ddx = nx < 0 ? px + 1 : (nx > 0 ? 8 - px : 99), ddy = ny < 0 ? py + 1 : (ny > 0 ? 8 - py : 99);
						const d = nx && ny ? ddx + ddy - 1 : Math.min(ddx, ddy);
						if (d < INSET + (nx && ny ? 1 : 0)) ok = false;
					}
				}
				if (ok) solid[(ty * 8 + py) * mw + tx * 8 + px] = 1;
			}
		}
		for (let y = 0; y < mh; y++) for (let x2 = 0; x2 < mw; x2++) {
			const i = y * mw + x2;
			if (!solid[i]) continue;
			const edge = x2 === 0 || y === 0 || x2 === mw - 1 || y === mh - 1 ? false :
				!solid[i - 1] || !solid[i + 1] || !solid[i - mw] || !solid[i + mw];
			const edgeX = (x2 === 0 || x2 === mw - 1) && !solid[i - mw] !== !solid[i + mw];
			const col = edge || edgeX ? line : fill;
			img.data[i * 4] = col[0];
			img.data[i * 4 + 1] = col[1];
			img.data[i * 4 + 2] = col[2];
			img.data[i * 4 + 3] = 255;
		}
		x.putImageData(img, 0, 0);
		if (g.cfg.escape) return c;
		// house box with its door
		x.strokeStyle = pal[0];
		x.lineWidth = 1;
		x.strokeRect(10 * 8 + 3.5, 12 * 8 + 3.5, 8 * 8 - 7, 5 * 8 - 7);
		x.strokeRect(10 * 8 + 5.5, 12 * 8 + 5.5, 8 * 8 - 11, 5 * 8 - 11);
		x.clearRect(13 * 8, 12 * 8 + 2, 16, 5);
		x.fillStyle = '#ffb02f';
		x.fillRect(13 * 8, 12 * 8 + 4, 16, 2);
		return c;
	}

	function hexRgb(col) {
		if (col[0] === '#') { const n = parseInt(col.slice(1), 16); return [(n >> 16) & 255, (n >> 8) & 255, n & 255]; }
		return col.match(/\d+/g).map(Number);
	}

	function drawMaze(g, t, flash) {
		const key = g.level + ':' + g.maze.name + ':' + g.seed + ':' + g.gates.map(function (q) { return q.closed ? 1 : 0; }).join('') + (flash ? 'f' : '');
		if (key !== mazeKey || !mazeCache) {
			mazeCache = buildMaze(g);
			mazeKey = key;
		}
		if (flash && Math.floor(t / 12) % 2) {
			lx.globalCompositeOperation = 'source-over';
			lx.filter = 'grayscale(1) brightness(2.2)';
			lx.drawImage(mazeCache, 0, OY);
			lx.filter = 'none';
		} else lx.drawImage(mazeCache, 0, OY);
		// gates
		g.gates.forEach(function (q) {
			const px = q.x * 8, py = OY + q.y * 8, warn = g.gateT < 60 && Math.floor(t / 6) % 2;
			lx.fillStyle = q.closed ? (warn ? '#ffffff' : '#ff4fd8') : 'rgba(255,79,216,0.25)';
			if (q.closed) lx.fillRect(px + 1, py + 1, 6, 6);
			else { lx.fillRect(px + 1, py + 1, 1, 1); lx.fillRect(px + 6, py + 6, 1, 1); lx.fillRect(px + 6, py + 1, 1, 1); lx.fillRect(px + 1, py + 6, 1, 1); }
			if (q.closed) { lx.fillStyle = '#2a0630'; lx.fillRect(px + 2, py + 2 + (Math.floor(t / 8) % 4), 4, 1); }
		});
		// dots
		const dots = g.maze.dots;
		lx.fillStyle = '#ffe3c2';
		for (let i = 0; i < dots.length; i++) {
			if (dots[i] !== 1) continue;
			lx.fillRect((i % g.maze.w) * 8 + 3, OY + ((i / g.maze.w) | 0) * 8 + 3, 2, 2);
		}
		if (Math.floor(t / 10) % 2 === 0 || g.phase !== 'play') {
			lx.fillStyle = '#fff1dc';
			for (let i = 0; i < dots.length; i++) {
				if (dots[i] !== 2) continue;
				const px = (i % g.maze.w) * 8, py = OY + ((i / g.maze.w) | 0) * 8;
				lx.fillRect(px + 1, py + 2, 6, 4);
				lx.fillRect(px + 2, py + 1, 4, 6);
			}
		}
		if (g.cfg.escape) {
			// the exit: a pulsing portal
			const px = g.maze.exit[0] * 8, py = OY + g.maze.exit[1] * 8, k = Math.floor(t / 6) % 4;
			lx.fillStyle = '#2fe6ff';
			lx.fillRect(px - 1, py - 1, 10, 10);
			lx.fillStyle = '#000';
			lx.fillRect(px, py, 8, 8);
			lx.fillStyle = ['#ff2fd6', '#ffffff', '#2fe6ff', '#ffffff'][k];
			lx.fillRect(px + 3 - (k & 1), py + 3 - (k & 1), 2 + 2 * (k & 1), 2 + 2 * (k & 1));
		}
	}

	/* ---------- actors ---------- */

	function sx(v) { return Math.round(v / M.PX); }

	function drawSprite(img, x, y, flip, alpha) {
		lx.save();
		if (alpha !== undefined) lx.globalAlpha = alpha;
		const w = img.logical || img.width, o = w >> 1;
		y -= (w - 13) >> 1;
		if (flip) {
			lx.translate(x + o + 1, y);
			lx.scale(-1, 1);
			lx.drawImage(img, 0, 0, w, w);
		} else lx.drawImage(img, x - o, y, w, w);
		lx.restore();
	}

	function drawHero(g, t, x, y, dir, alpha) {
		const h = g.hero;
		const moving = g.phase === 'play' && h.dir !== M.NONE;
		const frame = moving ? Math.floor(t / 4) % 4 : 1;
		if (dir === M.LEFT || dir === M.RIGHT) h.face = dir;
		const img = heroSprite(frame, dir);
		drawSprite(img, x, y, (h.face || M.LEFT) === M.LEFT, alpha);
	}

	function drawDeath(g, x, y) {
		const k = 150 - g.phaseT;
		if (k < 30) { drawHero(g, 0, x, y - 6, g.hero.face || M.LEFT); return; }
		const p = (k - 30) / 90;
		if (p >= 1) return;
		const img = heroSprite(0, M.RIGHT);
		const S = img.width, ix = img.getContext('2d').getImageData(0, 0, S, S).data, q = (img.logical || S) / S, o = (img.logical || S) >> 1;
		for (let py = 0; py < S; py++) for (let px = 0; px < S; px++) {
			const a = ix[(py * S + px) * 4 + 3];
			if (!a) continue;
			const n = ((px * 73 + py * 151 + 7) % 97) / 97;
			if (n < p) continue;
			const fall = Math.round(p * p * 18 * (0.5 + n));
			const i = (py * S + px) * 4;
			lx.fillStyle = n < p + 0.12 ? '#ffffff' : 'rgb(' + ix[i] + ',' + ix[i + 1] + ',' + ix[i + 2] + ')';
			lx.fillRect(x - o + px * q + (px * q - o) * p * 1.5, y - o + py * q + fall, q, q);
		}
	}

	function chaserMode(g, c) {
		if (c.state === 'eyes' || c.state === 'entering') return 'eyes';
		if (c.fright) return g.frightT < 120 && Math.floor(g.frightT / 12) % 2 ? 'flash' : 'fright';
		if (c.kind === 'dozer' && c.state === 'active' && c.awake === 0) return 'sleep';
		return 'normal';
	}

	function drawChaser(g, c, t, alphaOverride) {
		const info = CAST[c.kind], mode = chaserMode(g, c);
		const look = c.dir || M.DOWN;
		let alpha = info.alpha !== undefined && mode === 'normal' ? info.alpha : 1;
		if (alphaOverride !== undefined) alpha = alphaOverride;
		if (c.kind === 'warp' && c.warn && Math.floor(t / 3) % 2) alpha *= 0.3;
		if (c.kind === 'mini' && c.ttl < 120 && Math.floor(t / 4) % 2) alpha *= 0.3;
		const img = chaserSprite(c.kind, Math.floor(t / 8) % 2, look, mode);
		const x = sx(c.x), y = OY + sx(c.y) - 6;
		drawSprite(img, x, y, false, alpha);
		if (g.fx && g.fx.freeze > 0 && c.state === 'active') {
			lx.fillStyle = 'rgba(143,232,255,0.45)';
			lx.fillRect(x - 6, y, 13, 13);
		}
		if (mode === 'sleep' && Math.floor(t / 30) % 2) Font.draw(lx, 'Z', x + 3, y - 7, '#d0965a');
	}

	/* hunter: the player is a gang member, the BYTEs flee (and turn red and chase after a pill) */
	function drawHunt(g, t) {
		g.blocks.forEach(function (b) {
			const px = b.x * 8, py = OY + b.y * 8;
			lx.fillStyle = b.t < 60 && Math.floor(t / 4) % 2 ? '#ffffff' : '#b06bff';
			lx.fillRect(px, py, 8, 8);
			lx.fillStyle = '#1a0a2e';
			lx.fillRect(px + 2, py + 2, 4, 4);
		});
		const angry = g.frightT > 0 && !g.hero.immune && !(g.frightT < 120 && Math.floor(g.frightT / 12) % 2);
		g.prey.forEach(function (p) {
			if (p.caught) return;
			if (p.dir === M.LEFT || p.dir === M.RIGHT) p.face = p.dir;
			const frame = p.munch > 0 ? Math.floor(t / 3) % 4 : (p.dir ? Math.floor(t / 4) % 4 : 1);
			drawSprite(heroSprite(frame, p.dir, angry ? '#ff3355' : undefined), sx(p.x), OY + sx(p.y) - 6, (p.face || M.LEFT) === M.LEFT);
		});
		const h = g.hero, look = h.dir || M.LEFT;
		let mode = 'normal';
		if (g.phase === 'dying' || h.eyes) mode = 'eyes';
		else if (g.frightT > 0 && !h.immune) mode = g.frightT < 120 && Math.floor(g.frightT / 12) % 2 ? 'flash' : 'fright';
		drawSprite(chaserSprite(g.cfg.member, Math.floor(t / 8) % 2, look, mode), sx(h.x), OY + sx(h.y) - 6, false);
	}

	function drawActors(g, t) {
		if (g.cfg.hunter) { drawHunt(g, t); return; }
		const h = g.hero, hx = sx(h.x), hy = OY + sx(h.y);
		if (g.fruit) lx.drawImage(itemSprite(g.fruit.kind), sx(g.fruit.x) - 5, OY + sx(g.fruit.y) - 5);
		if (g.pickup) {
			const p = PICK[g.pickup.type], px = sx(g.pickup.x), py = OY + sx(g.pickup.y);
			const blink = g.pickup.t < 120 && Math.floor(t / 5) % 2;
			if (!blink) {
				lx.fillStyle = p[1];
				lx.fillRect(px - 5, py - 4, 11, 9);
				lx.fillRect(px - 4, py - 5, 9, 11);
				lx.fillStyle = '#000';
				lx.fillRect(px - 4, py - 3, 9, 7);
				Font.draw(lx, p[0], px - 4, py - 3, p[1]);
			}
		}
		if (g.decoy && Math.floor(t / 3) % 3) drawSprite(heroSprite(1, M.RIGHT, '#9fb0ff'), sx(g.decoy.x), OY + sx(g.decoy.y) - 6, false, 0.7);
		if (g.phase === 'dying') drawDeath(g, hx, hy);
		else {
			drawHero(g, t, hx, hy - 6, h.dir || h.face || M.LEFT, g.fx.phase > 0 ? 0.55 : undefined);
			if (g.fx.shield) {
				lx.strokeStyle = Math.floor(t / 4) % 2 ? '#3fa8ff' : '#bfe4ff';
				lx.beginPath();
				lx.arc(hx + 0.5, hy + 0.5, 9, 0, Math.PI * 2);
				lx.stroke();
			}
		}
		if (g.phase !== 'dying' || 150 - g.phaseT < 30) {
			if (g.phase !== 'clear') g.chasers.forEach(function (c) { drawChaser(g, c, t, g.cfg.darkness ? 0 : undefined); });
		}
	}

	function drawDarkness(g, t) {
		const h = g.hero, hx = sx(h.x), hy = OY + sx(h.y);
		const r = g.phase === 'dying' ? 70 : 44 + (g.fx.boots > 0 ? 6 : 0);
		const grad = lx.createRadialGradient(hx, hy, r * 0.35, hx, hy, r);
		grad.addColorStop(0, 'rgba(0,0,0,0)');
		grad.addColorStop(1, 'rgba(0,0,0,0.97)');
		lx.fillStyle = grad;
		lx.fillRect(camX, OY + camY, W, M.ROWS * 8);
		if (g.phase === 'clear' || (g.phase === 'dying' && 150 - g.phaseT >= 30)) return;
		g.chasers.forEach(function (c) {
			const x = sx(c.x), y = OY + sx(c.y);
			const near = Math.max(0, 1 - Math.hypot(x - hx, y - hy) / r);
			const glow = lx.createRadialGradient(x, y, 0, x, y, 10);
			glow.addColorStop(0, chaserMode(g, c) === 'normal' ? CAST[c.kind].color : '#8070ff');
			glow.addColorStop(1, 'rgba(0,0,0,0)');
			lx.globalAlpha = 0.28 + 0.1 * Math.sin(t / 9 + c.id);
			lx.fillStyle = glow;
			lx.fillRect(x - 10, y - 10, 20, 20);
			lx.globalAlpha = 1;
			drawChaser(g, c, t, 0.25 + near * 0.75);
		});
	}

	function drawPopups(g) {
		g.popups.forEach(function (p) {
			const txt = p.text, x = sx(p.x) - txt.length * 4, y = OY + sx(p.y) - 4 - Math.round((90 - p.t) / 10);
			Font.draw(lx, txt, Math.max(camX, Math.min(camX + W - txt.length * 8, x)), y, /^\d+$/.test(txt) ? '#3ff0ff' : '#ffd34a');
		});
	}

	/* ---------- HUD ---------- */

	function drawHud(g, hi, t, label) {
		if (Math.floor(t / 20) % 2 === 0 || !g || g.over) Font.draw(lx, '1UP', 24, 0, '#ff4fd8');
		Font.draw(lx, 'HIGH SCORE', 72, 0, '#ff4fd8');
		const score = g ? g.score : 0;
		Font.draw(lx, String(score || '00'), 56 - String(score || '00').length * 8, 9, '#ffffff');
		const his = String(Math.max(hi || 0, score) || '00');
		Font.draw(lx, his, 136 - his.length * 8, 9, '#ffffff');
		if (label) Font.draw(lx, label, W - 4 - label.length * 8, 9, '#3ff0ff');
		if (!g) return;
		for (let i = 0; i < Math.min(g.lives - 1, 5); i++) {
			if (g.cfg.hunter) lx.drawImage(chaserSprite(g.cfg.member, 0, M.RIGHT, 'normal'), 9 + i * 17, H - 15);
			else lx.drawImage(heroSprite(1, M.RIGHT), 8 + i * 17, H - 16, 16, 16);
		}
		let x = W - 14;
		for (let l = g.level; l >= Math.max(1, g.level - 6); l--) {
			lx.drawImage(itemSprite(M.bonusFor(l)[0]), x, H - 13);
			x -= 12;
		}
		if (g.cfg.escape && g.maze.exit) {
			const left = String(M.escapeTime(g));
			Font.draw(lx, 'T', 88, H - 12, '#ff4fd8');
			Font.draw(lx, left, 98, H - 12, M.escapeTime(g) < 20 && Math.floor(t / 15) % 2 ? '#ff3355' : '#ffd34a');
		}
		if (g.cfg.hunter && g.prey) {
			// ability: name when ready, seconds of cooldown otherwise. RAZOR's speed is always on
			const name = { razor: 'FAST', vex: 'DASH', murk: 'WALL', fizz: 'WARP' }[g.cfg.member];
			if (g.abil > 0) Font.draw(lx, String(Math.ceil(g.abil / M.TPS)), 88, H - 12, '#55577a');
			else Font.draw(lx, name, 88, H - 12, CAST[g.cfg.member].color);
		}
		if (g.cfg.survival) {
			const s = Math.floor(g.survT / M.TPS);
			Font.draw(lx, 'T', 88, H - 12, '#ff4fd8');
			Font.draw(lx, Math.floor(s / 60) + ':' + String(s % 60).padStart(2, '0'), 98, H - 12, '#ffd34a');
		}
		if (g.cfg.mayhem && g.fx) {
			let fx = 88;
			['boots', 'freeze', 'magnet', 'phase', 'double'].forEach(function (k) {
				if (g.fx[k] > 0) { Font.draw(lx, PICK[k][0], fx, H - 12, PICK[k][1]); fx += 9; }
			});
			if (g.fx.shield) Font.draw(lx, 'S', fx, H - 12, PICK.shield[1]);
		}
	}

	function drawPhaseText(g, t) {
		if (g.phase === 'start') {
			Font.center(lx, 'PLAYER ONE', OY + 11 * 8, '#3ff0ff');
			Font.center(lx, 'GET SET!', OY + 17 * 8, '#ffd34a');
		} else if (g.phase === 'ready') Font.center(lx, 'GET SET!', OY + 17 * 8, '#ffd34a');
		if (g.phase === 'start' || (g.phase === 'ready' && g.phaseT > 90)) {
			const name = g.maze.name;
			Font.center(lx, name, OY + 23 * 8 + 1, '#ffffff');
		}
	}

	/* ---------- intermissions ---------- */

	function drawIntermission(g, t) {
		const k = 480 - g.phaseT, mid = OY + 15 * 8;
		const titles = { 1: 'THE HEIST', 2: 'TRAFFIC JAM', 3: 'NIGHT SHIFT' };
		Font.center(lx, 'INTERMISSION', OY + 4 * 8, '#ff4fd8');
		Font.center(lx, titles[g.interKind], OY + 6 * 8, '#3ff0ff');
		lx.fillStyle = '#2fe6ff';
		lx.fillRect(0, mid + 10, W, 1);
		const f = Math.floor(t / 4) % 4;
		if (g.interKind === 1) {
			// FIZZ grabs the bonus item and runs, BYTE gives chase, FIZZ trips and flees in a panic
			const stop = 60;
			if (k < 130) {
				const fx = W + 10 - k * 1.33;
				drawSprite(chaserSprite('fizz', f % 2, M.LEFT, 'normal'), Math.round(fx), mid - 6);
				lx.drawImage(itemSprite('gem'), Math.round(fx) - 18, mid - 4);
				drawSprite(heroSprite(f, M.LEFT), Math.round(fx + 44), mid - 6, true);
			} else if (k < 230) {
				const hop = k < 160 ? Math.abs(((k - 130) % 16) - 8) : 0;
				drawSprite(chaserSprite('fizz', 0, M.UP, 'normal'), stop, mid - 6 - hop);
				lx.drawImage(itemSprite('gem'), Math.round(stop - 30 - Math.min(20, (k - 130) * 0.5)), mid - 4);
				drawSprite(heroSprite(f, M.LEFT), Math.round(stop + 44 - (k - 130) * 0.25), mid - 6, true);
				if (k > 170) Font.draw(lx, '!', stop - 3, mid - 20, '#ffffff');
			} else {
				const fx = stop + (k - 230) * 1.6;
				drawSprite(chaserSprite('fizz', f % 2, M.RIGHT, 'fright'), Math.round(fx), mid - 6);
				drawSprite(heroSprite(f, M.RIGHT), Math.round(fx - 26), mid - 6);
				Font.center(lx, '+2000', mid - 26, '#3ff0ff');
			}
		} else if (g.interKind === 2) {
			// the gang marches in a conga line, VEX stalls, everybody piles up, BYTE strolls past
			const order = ['razor', 'vex', 'murk', 'fizz'];
			const stop = 112;
			order.forEach(function (kind, i) {
				let x = -10 + k * 0.9 - i * 18;
				if (k > 140) x = Math.min(x, stop - i * (k > 200 ? 9 : 18));
				drawSprite(chaserSprite(kind, f % 2, M.RIGHT, k > 220 && i > 0 ? 'fright' : 'normal'), Math.round(x), mid - 6);
			});
			if (k > 200) Font.draw(lx, '?!', stop - 8, mid - 20, '#ffffff');
			if (k > 260) drawSprite(heroSprite(f, M.LEFT), Math.round(W + 10 - (k - 260) * 1.1), mid - 22, true);
		} else {
			// lights out: only eyes are visible, then the lights snap back on
			const lit = k > 300;
			['razor', 'vex', 'murk', 'fizz'].forEach(function (kind, i) {
				const x = 30 + i * 46 + Math.round(Math.sin((k + i * 40) / 25) * 12), y = mid - 6 + Math.round(Math.cos((k + i * 30) / 30) * 8);
				drawSprite(chaserSprite(kind, f % 2, i % 2 ? M.LEFT : M.RIGHT, lit ? 'fright' : 'eyes'), x, y);
			});
			if (lit) {
				drawSprite(heroSprite(f, M.RIGHT), Math.round(-10 + (k - 300) * 1.6), mid + 14);
				Font.center(lx, 'BOO!', mid - 30, '#ffd34a');
			}
		}
	}

	/* ---------- public scenes ---------- */

	function game(g, t, hi, label) {
		clear();
		if (g.phase === 'intermission') {
			drawIntermission(g, t);
		} else {
			if (g.cfg.escape) {
				// the camera stays centered on BYTE, stopping at the edges of the maze
				const vh = M.ROWS * 8;
				camX = Math.max(0, Math.min(g.maze.w * 8 - W, sx(g.hero.x) - (W >> 1)));
				camY = Math.max(0, Math.min(g.maze.h * 8 - vh, sx(g.hero.y) - (vh >> 1)));
				lx.save();
				lx.beginPath();
				lx.rect(0, OY, W, vh);
				lx.clip();
				lx.translate(-camX, -camY);
			} else camX = camY = 0;
			drawMaze(g, t, g.phase === 'clear');
			drawActors(g, t);
			if (g.cfg.darkness) drawDarkness(g, t);
			drawPopups(g);
			if (g.cfg.escape) lx.restore();
			drawPhaseText(g, t);
		}
		drawHud(g, hi, t, label);
	}

	function clear() {
		lx.globalAlpha = 1;
		lx.fillStyle = '#000';
		lx.fillRect(0, 0, W, H);
	}

	function present() {
		if (gl) {
			const w = out.width, h = out.height;
			gl.viewport(0, 0, w, h);
			gl.bindTexture(gl.TEXTURE_2D, glTex);
			gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, low);
			gl.uniform2f(gl.getUniformLocation(glProg, 'res'), W, H);
			gl.uniform2f(gl.getUniformLocation(glProg, 'outRes'), w, h);
			gl.uniform1f(gl.getUniformLocation(glProg, 'crt'), crt ? 1 : 0);
			gl.uniform1f(gl.getUniformLocation(glProg, 'curve'), crt && curve ? 1 : 0);
			gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
		} else {
			ctx2d.imageSmoothingEnabled = false;
			ctx2d.drawImage(low, 0, 0, out.width, out.height);
			if (crt) {
				ctx2d.globalCompositeOperation = 'lighter';
				ctx2d.globalAlpha = 0.35;
				ctx2d.filter = 'blur(' + Math.round(out.width / 160) + 'px)';
				ctx2d.drawImage(low, 0, 0, out.width, out.height);
				ctx2d.filter = 'none';
				ctx2d.globalAlpha = 1;
				ctx2d.globalCompositeOperation = 'source-over';
			}
		}
		out.parentNode.classList.toggle('crt', crt);
	}

	function resize() {
		const r = out.getBoundingClientRect(), dpr = window.devicePixelRatio || 1;
		const k = Math.min(4, Math.max(1, r.width * dpr / W));
		const w = Math.round(W * k), h = Math.round(H * k);
		if (out.width !== w || out.height !== h) { out.width = w; out.height = h; }
	}

	function initGL() {
		try {
			gl = out.getContext('webgl', { antialias: false, preserveDrawingBuffer: true });
		} catch (e) { gl = null; }
		if (!gl) return false;
		const vs = 'attribute vec2 p; varying vec2 uv; void main() { uv = p * 0.5 + 0.5; uv.y = 1.0 - uv.y; gl_Position = vec4(p, 0.0, 1.0); }';
		const fs = [
			'precision mediump float;',
			'uniform sampler2D tex; uniform vec2 res; uniform vec2 outRes; uniform float crt; uniform float curve; varying vec2 uv;',
			'vec2 warp(vec2 q) { vec2 c = q * 2.0 - 1.0; c *= 1.0 + curve * vec2(c.y * c.y * 0.045, c.x * c.x * 0.06); return c * 0.5 + 0.5; }',
			'vec3 tap(vec2 q) { return texture2D(tex, q).rgb; }',
			'void main() {',
			'  vec2 q = warp(uv);',
			'  if (q.x < 0.0 || q.y < 0.0 || q.x > 1.0 || q.y > 1.0) { gl_FragColor = vec4(0.0, 0.0, 0.0, 1.0); return; }',
			'  vec3 col = tap(q);',
			'  if (crt > 0.5) {',
			'    vec2 px = 1.0 / res; vec3 glow = vec3(0.0);',
			'    for (int i = -1; i <= 1; i++) for (int j = -1; j <= 1; j++) { glow += tap(q + vec2(float(i), float(j)) * px * 1.6); }',
			'    glow /= 9.0;',
			'    float line = 0.62 + 0.38 * pow(abs(sin(q.y * res.y * 3.14159)), 0.6);',
			'    float mask = 0.9 + 0.1 * step(0.5, fract(uv.x * outRes.x / 3.0));',
			'    col = col * line * mask * 1.15 + glow * 0.65;',
			'    vec2 v = q * (1.0 - q.yx); float vig = pow(v.x * v.y * 18.0, 0.22);',
			'    col *= clamp(vig, 0.0, 1.0);',
			'  }',
			'  gl_FragColor = vec4(col, 1.0);',
			'}'
		].join('\n');
		const sh = function (type, src) {
			const s = gl.createShader(type);
			gl.shaderSource(s, src);
			gl.compileShader(s);
			if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) throw new Error(gl.getShaderInfoLog(s));
			return s;
		};
		try {
			glProg = gl.createProgram();
			gl.attachShader(glProg, sh(gl.VERTEX_SHADER, vs));
			gl.attachShader(glProg, sh(gl.FRAGMENT_SHADER, fs));
			gl.linkProgram(glProg);
			gl.useProgram(glProg);
		} catch (e) { gl = null; return false; }
		const buf = gl.createBuffer();
		gl.bindBuffer(gl.ARRAY_BUFFER, buf);
		gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]), gl.STATIC_DRAW);
		const loc = gl.getAttribLocation(glProg, 'p');
		gl.enableVertexAttribArray(loc);
		gl.vertexAttribPointer(loc, 2, gl.FLOAT, false, 0, 0);
		glTex = gl.createTexture();
		gl.bindTexture(gl.TEXTURE_2D, glTex);
		gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.NEAREST);
		gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.NEAREST);
		gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
		gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
		return true;
	}

	function init(display) {
		out = display;
		low = canvas(W * SS, H * SS);
		lx = low.getContext('2d');
		lx.scale(SS, SS);
		lx.imageSmoothingEnabled = false;
		if (!initGL()) ctx2d = out.getContext('2d');
		resize();
		window.addEventListener('resize', resize);
	}

	function setCrt(on, curved) {
		crt = on;
		curve = curved;
	}

	return {
		W: W, H: H, OY: OY, CAST: CAST, MODE_NAMES: MODE_NAMES, PICK: PICK,
		chaserSprite: chaserSprite, clear: clear, ctx: function () { return lx; }, drawHud: drawHud, drawSprite: drawSprite,
		game: game, heroSprite: heroSprite, init: init, itemSprite: itemSprite, present: present, resize: resize, setCrt: setCrt,
		usingGL: function () { return !!gl; }
	};
})();

const M = globalThis.MazeCore;
let failures = 0;

function check(name, ok, info) {
	if (!ok) failures++;
	print((ok ? 'ok   ' : 'FAIL ') + name + (ok || !info ? '' : '  ' + info));
}

function validateMaze(m) {
	const problems = [];
	const open = function (x, y) { return M.cellAt(m, x, y) === M.C_PATH; };
	let start = -1, total = 0;
	for (let y = 0; y < M.ROWS; y++) for (let x = 0; x < M.COLS; x++) {
		if (m.cells[y * M.COLS + x] !== m.cells[y * M.COLS + (M.COLS - 1 - x)]) problems.push('asym ' + x + ',' + y);
		if (!open(x, y)) continue;
		total++;
		if (start < 0) start = y * M.COLS + x;
		let n = 0;
		for (let d = 1; d <= 4; d++) if (open(x + M.DX[d], y + M.DY[d])) n++;
		if (n < 2) problems.push('deadend ' + x + ',' + y);
		if (x + 1 < M.COLS && y + 1 < M.ROWS && open(x + 1, y) && open(x, y + 1) && open(x + 1, y + 1)) problems.push('2x2 ' + x + ',' + y);
	}
	const seen = new Uint8Array(M.COLS * M.ROWS), stack = [start];
	seen[start] = 1;
	let count = 1;
	while (stack.length) {
		const i = stack.pop(), x = i % M.COLS, y = (i / M.COLS) | 0;
		for (let d = 1; d <= 4; d++) {
			const nx = ((x + M.DX[d]) + M.COLS) % M.COLS, ny = y + M.DY[d];
			if (!open(nx, ny) || seen[ny * M.COLS + nx]) continue;
			seen[ny * M.COLS + nx] = 1;
			count++;
			stack.push(ny * M.COLS + nx);
		}
	}
	if (count !== total) problems.push('disconnected ' + count + '/' + total);
	[[13, 11], [14, 11], [13, 23], [14, 23], [9, 14], [18, 14]].forEach(function (p) { if (!open(p[0], p[1])) problems.push('required ' + p); });
	if (M.cellAt(m, 13, 12) !== M.C_DOOR) problems.push('door');
	return problems;
}

function run(mode, seed, ticks, inputSeed, twists) {
	const g = M.create({ mode: mode, seed: seed, twists: twists });
	const r = new M.Rng(inputSeed);
	let dir = 2;
	for (let t = 0; t < ticks && !g.over; t++) {
		if (r.int(40) === 0) dir = 1 + r.int(4);
		M.step(g, dir);
	}
	return g;
}

function escapeProblems(m) {
	const problems = [], dist = new Int32Array(m.w * m.h).fill(-1), q = [m.start[1] * m.w + m.start[0]];
	dist[q[0]] = 0;
	for (let qi = 0; qi < q.length; qi++) for (let d = 1; d <= 4; d++) {
		const j = q[qi] + M.DX[d] + M.DY[d] * m.w;
		if (m.cells[j] === M.C_PATH && dist[j] < 0) { dist[j] = dist[q[qi]] + 1; q.push(j); }
