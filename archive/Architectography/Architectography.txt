# Architectography v1.0.0

Современный инструмент для быстрой и удобной проверки ПК на стороннее ПО.

## Возможности
- **Быстрый запуск утилит:** Everything, ShellBags, BrowserDownloadsView, ExecutedProgramsList, USBDeview, WinPrefetchView, BrowsingHistoryView, JumpListsView, UserAssistView, LastActivityView, Process Hacker, JournalTrace, System Informer.
- **Мониторинг процессов:** Живой поиск по имени и полному пути с фильтрацией .exe и .dll.
- **Сведения о системе:** Детальная сводка по железу, аптайму, сборке и дате установки Windows с копированием в один клик.
- **Проверка истории и сайтов:** Быстрый доступ к популярным ресурсам и платёжным шлюзам.
- **Кастомизация:** Смена акцентного цвета интерфейса, мягкие UI-звуки и атмосферный снегопад на фоне.

## Запуск из исходников
```bash
pip install -r requirements.txt
python main.py
```

## Скачать готовую сборку
Готовый скомпилированный файл доступен во вкладке **[Releases](../../releases)**.

## Поддержать разработку

import os

import requests


ADMIN_TOOLS = {"journaltrace", "systeminformer"}

TOOL_SOURCES = {

    "processhacker":          {"url": "https://github.com/processhacker/processhacker/releases/download/v2.39/ProcessHacker-2.39-bin.zip"},
    "journaltrace":           {"url": None,  "github": "processhacker/processhacker", "asset_contains": "JournalTrace"},
    "systeminformer":         {"url": None,  "github": "winsiderss/systeminformer",   "asset_contains": "systeminformer"},
}

CHECK_SITES = [
    {"name": "VirusTotal",      "desc": "Проверка файлов и URL на вирусы",     "url": "https://www.virustotal.com"},
    {"name": "Any.Run",         "desc": "Интерактивная песочница вредоносных программ", "url": "https://any.run"},
    {"name": "Hybrid Analysis", "desc": "Анализ поведения файлов",             "url": "https://www.hybrid-analysis.com"},

class API:
    """Python side of the PyWebView js_api exposed to JavaScript."""

    def __init__(self, window=None):
        self._window = window
        self._tools_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), "tools")
        os.makedirs(self._tools_dir, exist_ok=True)

    # ------------------------------------------------------------------
    # Window control
    # ------------------------------------------------------------------
    def minimize(self):
        if self._window:
            self._window.minimize()
        return "minimized"

    def close(self):
        if self._window:
            self._window.destroy()
        return "closed"

    def open_folder(self):
        """Open the project folder in Explorer."""
        try:
            project_dir = os.path.dirname(os.path.abspath(__file__))
            os.startfile(project_dir)
            return json.dumps({"status": "ok"})
        except Exception as e:
            return json.dumps({"status": "error", "error": str(e)})

    def open_website(self, url: str):
        """Open a URL in the default browser."""
        try:
            import webbrowser
            webbrowser.open(url)
            return json.dumps({"status": "ok"})
        except Exception as e:
            return json.dumps({"status": "error", "error": str(e)})

    # ------------------------------------------------------------------
    # System info
    # ------------------------------------------------------------------
    def get_system_info(self):
        """Return a dict of system information."""
        try:
            uname = platform.uname()
            vm = psutil.virtual_memory()
            disk = psutil.disk_usage('/')
            info = {
                "ОС":               f"{uname.system} {uname.release} ({uname.version})",
                "Имя компьютера":   uname.node,
                "Процессор":        uname.processor or platform.processor(),
                "Ядер (логич.)":    psutil.cpu_count(logical=True),
                "Ядер (физич.)":    psutil.cpu_count(logical=False),
                "ОЗУ (всего)":      f"{vm.total // (1024**3)} ГБ",
                "ОЗУ (свободно)":   f"{vm.available // (1024**3)} ГБ",
                "Диск (всего)":     f"{disk.total // (1024**3)} ГБ",
                "Диск (свободно)":  f"{disk.free // (1024**3)} ГБ",
                "Платформа":        platform.architecture()[0],
                "Python":           platform.python_version(),
            }
            return json.dumps(info, ensure_ascii=False)
        except Exception as e:
            return json.dumps({"error": str(e)})

    # ------------------------------------------------------------------
    # Processes
    # ------------------------------------------------------------------
    def get_processes(self):
        """Return a JSON array of running processes {name, pid, type, path}."""
        try:
            procs = []
            for p in psutil.process_iter(['pid', 'name', 'exe']):
                try:
                    info = p.info
                    name = info['name'] or ''
                    exe  = info['exe'] or ''
                    ext  = os.path.splitext(name)[1].lower() if name else ''
                    ptype = ext if ext else 'other'
                    procs.append({
                        "name": name,
                        "pid":  info['pid'],
                        "type": ptype,
                        "path": exe,
                    })
                except (psutil.NoSuchProcess, psutil.AccessDenied):
                    pass
            return json.dumps(procs, ensure_ascii=False)
        except Exception as e:
            return json.dumps({"error": str(e)})

    # ------------------------------------------------------------------
    # Tools / Library
    # ------------------------------------------------------------------
    def list_tools(self):
        """Return array of {name, installed} for all known tools."""
        result = []
        for key in TOOL_SOURCES:
            exe = self._find_tool_exe(key)
            result.append({"name": key, "installed": exe is not None})
        return json.dumps(result, ensure_ascii=False)

    def _resolve_github_url(self, repo: str, asset_contains: str) -> str:
        """Fetch latest release from GitHub API and return browser_download_url for matching asset."""
        api_url = f"https://api.github.com/repos/{repo}/releases/latest"
        headers = {"User-Agent": "Mozilla/5.0", "Accept": "application/vnd.github+json"}
        resp = requests.get(api_url, headers=headers, timeout=15)
        resp.raise_for_status()
        data = resp.json()
        needle = asset_contains.lower()
        for asset in data.get("assets", []):
            name = asset.get("name", "").lower()
            if needle in name and (name.endswith(".zip") or name.endswith(".exe")):
                return asset["browser_download_url"]
        raise RuntimeError(
            f"GitHub release для {repo} не содержит ассета с '{asset_contains}'. "
            f"Доступные файлы: {[a['name'] for a in data.get('assets', [])]}"
        )

    def launch_or_download_tool(self, tool_name: str):
        """Launch if present; otherwise download (via GitHub API or direct URL), then launch."""
        try:
            key = tool_name.lower()
            exe_path = self._find_tool_exe(key)
            if exe_path:
                return self._do_launch(key, exe_path, downloaded=False)

            source = TOOL_SOURCES.get(key)
            if not source:
                return json.dumps({"status": "not_found",
                                   "error": f"Утилита '{tool_name}' не найдена в словаре."})

            # --- Resolve download URL ---
            if source.get("github"):
                try:
                    url = self._resolve_github_url(source["github"], source["asset_contains"])
                except Exception as gh_err:
                    return json.dumps({"status": "error",
                                       "error": f"GitHub API: {gh_err}"})
            else:
                url = source["url"]

            # --- Determine target path ---
            is_exe = url.lower().endswith(".exe")
            if is_exe:
                # Save .exe directly into tools/<key>/
                dest_dir = os.path.join(self._tools_dir, key)
                os.makedirs(dest_dir, exist_ok=True)
                exe_name = url.split("/")[-1]
                local_path = os.path.join(dest_dir, exe_name)
            else:
                tmp_zip = os.path.join(tempfile.gettempdir(), f"{key}.zip")

            # --- Download ---
            headers = {"User-Agent": "Mozilla/5.0"}
            with requests.get(url, headers=headers, stream=True, timeout=90) as r:
                r.raise_for_status()
                target = local_path if is_exe else tmp_zip
                with open(target, "wb") as f:
                    for chunk in r.iter_content(chunk_size=16384):
                        if chunk:
                            f.write(chunk)

            # --- Validate file size ---
            if os.path.getsize(target) < 10_000:
                os.remove(target)
                return json.dumps({"status": "error",
                                   "error": f"Скачанный файл слишком мал — возможно, ссылка недействительна."})

            # --- Unzip if needed ---
            if not is_exe:
                dest_dir = os.path.join(self._tools_dir, key)
                os.makedirs(dest_dir, exist_ok=True)
                with zipfile.ZipFile(tmp_zip, "r") as z:
                    z.extractall(dest_dir)
                os.remove(tmp_zip)

            # --- Find and launch ---
            exe_path = self._find_tool_exe(key)
            if exe_path:
                return self._do_launch(key, exe_path, downloaded=True)
            return json.dumps({"status": "downloaded_but_not_found",
                               "error": "Файл скачан и распакован, но .exe не найден в папке tools."})

        except requests.HTTPError as e:
            return json.dumps({"status": "error", "error": f"HTTP {e.response.status_code}: {e}"})
        except requests.ConnectionError as e:
            return json.dumps({"status": "error", "error": f"Нет соединения: {e}"})
        except requests.Timeout:
            return json.dumps({"status": "error", "error": "Превышено время ожидания загрузки (90 с)."})
        except zipfile.BadZipFile as e:
            return json.dumps({"status": "error", "error": f"Архив повреждён: {e}"})
        except Exception as e:
            return json.dumps({"status": "error", "error": str(e)})

    def _do_launch(self, key: str, exe_path: str, downloaded: bool):
        prefix = "downloaded_and_" if downloaded else ""
        if key in ADMIN_TOOLS:
            ctypes.windll.shell32.ShellExecuteW(
                None, "runas", exe_path, None, os.path.dirname(exe_path), 1
            )
            return json.dumps({"status": f"{prefix}launched_admin", "path": exe_path})
        else:
            os.startfile(exe_path)
            return json.dumps({"status": f"{prefix}launched", "path": exe_path})

    def _find_tool_exe(self, tool_name: str):
        keyword = tool_name.lower()
        # Search ./tools/ recursively
        for root, _, files in os.walk(self._tools_dir):
            for f in files:
                if f.lower().endswith('.exe') and keyword in f.lower():
                    return os.path.join(root, f)
        # Fallback: known install locations
        pf = os.environ.get("ProgramFiles", r"C:\Program Files")
        pf86 = os.environ.get("ProgramFiles(x86)", r"C:\Program Files (x86)")
        fallbacks = {
            "everything":     [os.path.join(pf, "Everything", "Everything.exe")],
            "processhacker":  [os.path.join(pf, "Process Hacker 2", "ProcessHacker.exe"),
                               os.path.join(pf86, "Process Hacker 2", "ProcessHacker.exe")],
            "systeminformer": [os.path.join(pf, "SystemInformer", "SystemInformer.exe")],
        }
        for fb_key, paths in fallbacks.items():
            if keyword in fb_key or fb_key in keyword:
                for path in paths:
                    if os.path.isfile(path):
                        return path
        return None

    # ------------------------------------------------------------------
    # Check tab – site list
    # ------------------------------------------------------------------
    def get_check_sites(self):
        return json.dumps(CHECK_SITES, ensure_ascii=False)

    # ------------------------------------------------------------------
    # System item launcher (Services, NVIDIA, etc.)
    # ------------------------------------------------------------------
    def launch_system_item(self, item_name: str):
        try:
            cmds = {
                "nvidia_control_panel": ["explorer.exe",
                    r"shell:AppsFolder\NVIDIACorp.NVIDIAControlPanel_56jybvy8sckqj!NVIDIACorp.NVIDIAControlPanel"],
                "services":             ["services.msc"],
                "data_usage":           ["ms-settings:datausage"],

            return json.dumps({"status": "error", "error": str(e)})

/* script.js – Full rewrite, clean, no duplicates */
'use strict';

// ============================================================
// 1. SNOW ANIMATION (requestAnimationFrame, no setInterval)
// ============================================================
const snowCanvas = document.getElementById('snowCanvas');
const snowCtx    = snowCanvas.getContext('2d');
let   snowFlakes = [];
let   snowRAF    = null;
// snowActive declared below with localStorage
const MAX_FLAKES = 45;

function resizeCanvas() {
    snowCanvas.width  = window.innerWidth;
    snowCanvas.height = window.innerHeight;
}

function mkFlake() {
    const r = Math.random() * 2.5 + 1.5;
    return {
        x:  Math.random() * snowCanvas.width,
        y:  Math.random() * -snowCanvas.height,
        r,
        vy: r * 0.4 + Math.random() * 0.6,
        vx: (Math.random() - 0.5) * 0.4,
    };
}

function populateFlakes() {
    while (snowFlakes.length < MAX_FLAKES) snowFlakes.push(mkFlake());
}

function snowStep() {
    snowCtx.clearRect(0, 0, snowCanvas.width, snowCanvas.height);
    snowCtx.fillStyle = 'rgba(255,255,255,0.35)';
    for (const f of snowFlakes) {
        f.x += f.vx;
        f.y += f.vy;
        if (f.x > snowCanvas.width)  f.x = 0;
        if (f.x < 0)                 f.x = snowCanvas.width;
        if (f.y > snowCanvas.height) { f.y = -f.r; f.x = Math.random() * snowCanvas.width; }
        snowCtx.beginPath();
        snowCtx.arc(f.x, f.y, f.r, 0, Math.PI * 2);
        snowCtx.fill();
    }
    if (snowActive) snowRAF = requestAnimationFrame(snowStep);
}

function startSnow() {
    if (snowRAF) return;
    resizeCanvas();
    populateFlakes();
    snowActive = true;
    snowRAF = requestAnimationFrame(snowStep);
}

function stopSnow() {
    snowActive = false;
    if (snowRAF) { cancelAnimationFrame(snowRAF); snowRAF = null; }
    snowCtx.clearRect(0, 0, snowCanvas.width, snowCanvas.height);
}

window.addEventListener('resize', () => { if (snowActive) resizeCanvas(); });

function ensureAudioCtx() {
    if (!_audioCtx) _audioCtx = new (window.AudioContext || window.webkitAudioContext)();
}

function playClick() {
    if (!_soundEnabled) return;
    ensureAudioCtx();
    const ctx  = _audioCtx;
    const osc  = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(150, ctx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(55, ctx.currentTime + 0.035);
    gain.gain.setValueAtTime(0.03, ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.04);
    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start();
    osc.stop(ctx.currentTime + 0.045);
}

// 3. PYWEBVIEW API WRAPPER (safe, polling fallback)
// ============================================================
function callApi(method, ...args) {
    return new Promise((resolve, reject) => {
        const attempt = () => {
            if (window.pywebview && window.pywebview.api &&
                typeof window.pywebview.api[method] === 'function') {
                try {
                    const result = window.pywebview.api[method](...args);
                    if (result && typeof result.then === 'function') {
                        result.then(resolve).catch(reject);
                    } else {
                        resolve(result);
                    }
                } catch(e) { reject(e); }
            } else {
                setTimeout(attempt, 150);
            }
        };
        attempt();
    });
}

let currentTab  = 'info';
const contentEl = document.getElementById('content');

function setActiveTab(tab) {
    currentTab = tab;
    document.querySelectorAll('.menu-item').forEach(btn =>
        btn.classList.toggle('active', btn.dataset.tab === tab)
    );
    // Re-trigger CSS animation
    contentEl.style.animation = 'none';
    // force reflow
    void contentEl.offsetWidth;
    contentEl.style.animation = '';
    renderTabContent(tab);
}

function initTabs() {
    document.querySelectorAll('.menu-item').forEach(btn => {
        btn.addEventListener('click', () => {
            playClick();
            setActiveTab(btn.dataset.tab);
        });
    });
    setActiveTab(currentTab);
}

function renderProcesses() {
    contentEl.innerHTML = `
        <div style="display:flex;align-items:center;gap:12px;margin-bottom:14px;flex-wrap:wrap;">
            <h2 style="margin:0;">Процессы</h2>
            <input type="text" id="procSearch" placeholder="Поиск по имени / пути…" style="flex:1;min-width:180px;"/>
            <label style="cursor:pointer;"><input type="radio" name="procType" value="all" checked> Все</label>
            <label style="cursor:pointer;"><input type="radio" name="procType" value=".exe"> .exe</label>
            <label style="cursor:pointer;"><input type="radio" name="procType" value=".dll"> .dll</label>
            <button id="procRefresh" style="padding:4px 10px;background:var(--accent);border:none;border-radius:4px;color:#fff;cursor:pointer;">↻ Обновить</button>
        </div>
        <div id="procWrap" style="overflow-x:auto;">
            <div id="procLoading" style="padding:20px;color:var(--text-muted);">Загрузка процессов…</div>
        </div>`;

    function loadProcs() {
        const wrap = document.getElementById('procWrap');
        const loadEl = document.getElementById('procLoading');
        if (loadEl) loadEl.textContent = 'Загрузка процессов…';

        callApi('get_processes').then(res => {
            let data;
            try { data = typeof res === 'string' ? JSON.parse(res) : res; }
            catch(e) { document.getElementById('procWrap').innerHTML = '<p style="color:red;">Ошибка парсинга данных</p>'; return; }

            if (!Array.isArray(data)) {
                document.getElementById('procWrap').innerHTML = `<p style="color:red;">${JSON.stringify(data)}</p>`;
                return;
            }

            const table = document.createElement('table');
            table.id = 'procTable';
            table.style.cssText = 'width:100%;border-collapse:collapse;font-size:13px;';
            table.innerHTML = `
                <thead>
                    <tr style="background:#232a36;position:sticky;top:0;">
                        <th style="padding:7px 8px;text-align:left;border-bottom:1px solid #2b3240;">Имя</th>
                        <th style="padding:7px 8px;text-align:left;border-bottom:1px solid #2b3240;">PID</th>
                        <th style="padding:7px 8px;text-align:left;border-bottom:1px solid #2b3240;">Тип</th>
                        <th style="padding:7px 8px;text-align:left;border-bottom:1px solid #2b3240;">Путь</th>
                    </tr>
                </thead>
                <tbody></tbody>`;
            const tbody = table.querySelector('tbody');
            data.forEach(p => {
                const tr = document.createElement('tr');
                tr.dataset.name = (p.name || '').toLowerCase();
                tr.dataset.path = (p.path || '').toLowerCase();
                tr.dataset.type = (p.type || '').toLowerCase();
                tr.innerHTML = `
                    <td style="padding:5px 8px;border-bottom:1px solid #1e2430;">${p.name}</td>
                    <td style="padding:5px 8px;border-bottom:1px solid #1e2430;">${p.pid}</td>
                    <td style="padding:5px 8px;border-bottom:1px solid #1e2430;">${p.type || ''}</td>
                    <td style="padding:5px 8px;border-bottom:1px solid #1e2430;word-break:break-all;">${p.path}</td>`;
                tbody.appendChild(tr);
            });

            const procWrap = document.getElementById('procWrap');
            procWrap.innerHTML = '';
            procWrap.appendChild(table);

            function applyFilter() {
                const term    = (document.getElementById('procSearch')?.value || '').toLowerCase();
                const typeVal = document.querySelector('input[name="procType"]:checked')?.value || 'all';
                table.querySelectorAll('tbody tr').forEach(row => {
                    const matchName = row.dataset.name.includes(term) || row.dataset.path.includes(term);
                    const matchType = typeVal === 'all' || row.dataset.type === typeVal;
                    row.style.display = (matchName && matchType) ? '' : 'none';
                });
            }

            document.getElementById('procSearch')?.addEventListener('input', applyFilter);
            document.querySelectorAll('input[name="procType"]').forEach(r => r.addEventListener('change', applyFilter));
        }).catch(err => {
            const w = document.getElementById('procWrap');
            if (w) w.innerHTML = `<p style="color:red;">Ошибка: ${err}</p>`;
        });
    }

    document.getElementById('procRefresh')?.addEventListener('click', () => { playClick(); loadProcs(); });
    loadProcs();
}

// 8. TOOL LAUNCHER HELPER
// ============================================================
function launchTool(name) {
    callApi('launch_or_download_tool', name.toLowerCase())
        .then(res => {
            let r;
            try { r = typeof res === 'string' ? JSON.parse(res) : res; }
            catch(e) { r = { status: String(res) }; }
            if (!r.status || r.status.includes('error') || r.status === 'not_found') {
                showToast(`Не удалось запустить ${name}: ${r.status}`, true);
            } else {
                showToast(`${name}: ${r.status}`);
            }
        })
        .catch(err => showToast(`Ошибка: ${err}`, true));
}

// 9. TOAST NOTIFICATIONS
// ============================================================
function showToast(msg, isError = false) {
    const t = document.createElement('div');
    t.textContent = msg;
    t.style.cssText = `
        position:fixed;bottom:24px;right:24px;padding:10px 16px;
        background:${isError ? '#7f1d1d' : '#1e3a5f'};
        color:#fff;border-radius:6px;font-size:13px;
        z-index:9999;opacity:1;transition:opacity 0.4s;
        border:1px solid ${isError ? '#ef4444' : '#3b82f6'};
        max-width:340px;word-break:break-word;`;
    document.body.appendChild(t);
    setTimeout(() => { t.style.opacity = '0'; setTimeout(() => t.remove(), 400); }, 2800);
}

// 10. INIT — wait for pywebviewready, then start everything

    window.pywebviewReady = true;
    init();
});

// Fallback polling: in case pywebviewready already fired or never fires
(function waitForInit() {
    if (window.pywebviewReady) return;
    // Also covers opening index.html directly in a browser
    if (document.readyState === 'complete' || document.readyState === 'interactive') {
        // If pywebview API not needed (browser test), still init UI
        if (!window.pywebview) {
            init();
            return;
        }
    }
    if (window.pywebview && window.pywebview.api) {
        init();
        window.pywebviewReady = true;
        return;
    }
    setTimeout(waitForInit, 150);
})();
