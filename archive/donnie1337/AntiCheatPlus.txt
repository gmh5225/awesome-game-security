```java
    public String color(String text) {
        if (text == null) return "";

        Matcher matcher = HEX_COLOR_PATTERN.matcher(text);
        StringBuffer buffer = new StringBuffer();
        while (matcher.find()) {
            String hex = matcher.group(1);
            StringBuilder legacy = new StringBuilder("§x");
            for (char c : hex.toCharArray()) {
                legacy.append('§').append(c);
            }
            matcher.appendReplacement(buffer, Matcher.quoteReplacement(legacy.toString()));
        }
        matcher.appendTail(buffer);

        return ChatColor.translateAlternateColorCodes('&', buffer.toString());
    }

    public String message(String path, String fallback) {
        return color(getConfig().getString(path, fallback));
    }
```

```java
package com.redeedge.anticheatplus.anticheat;

import org.bukkit.GameMode;
import org.bukkit.Location;
import org.bukkit.Material;
import org.bukkit.block.Block;
import org.bukkit.entity.Player;

public final class CheckUtil {

    private CheckUtil() {
    }

    public static boolean exemptMovement(Player player) {
        if (player == null) return true;
        GameMode mode = player.getGameMode();
        return mode == GameMode.CREATIVE
                || mode == GameMode.SPECTATOR
                || player.getAllowFlight()
                || player.isFlying()
                || player.isGliding()
                || player.isSwimming()
                || player.isRiptiding()
                || player.isInsideVehicle()
                || player.isDead();
    }

    public static boolean sameWorld(Location a, Location b) {
        return a != null
                && b != null
                && a.getWorld() != null
                && a.getWorld().equals(b.getWorld());
    }

    public static double horizontalDistance(Location a, Location b) {
        if (!sameWorld(a, b)) return 0.0D;
        double dx = b.getX() - a.getX();
        double dz = b.getZ() - a.getZ();
        return Math.sqrt(dx * dx + dz * dz);
    }

    public static boolean isLiquidAround(Location location) {
        if (location == null || location.getWorld() == null) return false;
        String feet = location.getBlock().getType().name();
        String below = location.clone().subtract(0.0D, 0.2D, 0.0D).getBlock().getType().name();
        return feet.contains("WATER") || feet.contains("LAVA")
                || below.contains("WATER") || below.contains("LAVA");
    }

    public static boolean isClimbableOrSpecial(Location location) {
        if (location == null) return false;
        String feet = location.getBlock().getType().name();
        String below = location.clone().subtract(0.0D, 1.0D, 0.0D).getBlock().getType().name();

        return feet.contains("LADDER")
                || feet.contains("VINE")
                || feet.contains("SCAFFOLD")
                || feet.contains("WEB")
                || below.contains("SLIME")
                || below.contains("HONEY")
                || below.contains("ICE")
                || below.contains("SOUL_SAND")
                || below.contains("PISTON");
    }

    public static boolean hasSolidSupport(Location location) {
        if (location == null || location.getWorld() == null) return false;
        Block below = location.clone().subtract(0.0D, 0.12D, 0.0D).getBlock();
        Material type = below.getType();
        return type.isSolid();
    }

    public static boolean insideSolid(Location location) {
        if (location == null || location.getWorld() == null) return false;
        Block feet = location.getBlock();
        Block head = location.clone().add(0.0D, 1.0D, 0.0D).getBlock();
        return isBlocking(feet.getType()) && isBlocking(head.getType());
    }

    public static boolean isBlocking(Material material) {
        if (material == null || !material.isSolid()) return false;
        String name = material.name();
        return !name.contains("DOOR")
                && !name.contains("TRAPDOOR")
                && !name.contains("FENCE_GATE")
                && !name.contains("SIGN")
                && !name.contains("BANNER");
    }
}
```

```java
package com.redeedge.anticheatplus.anticheat;

import org.bukkit.attribute.Attribute;
import org.bukkit.attribute.AttributeInstance;
import org.bukkit.entity.Player;
import org.bukkit.plugin.Plugin;

public final class CombatCompatibility {

    private CombatCompatibility() {
    }

    public static boolean isCombatePlusEnabled(Player player) {
        if (player == null || player.getServer() == null) return false;
        Plugin plugin = player.getServer().getPluginManager().getPlugin("CombatePlus");
        return plugin != null && plugin.isEnabled();
    }

    public static boolean isLegacyCombat(Player player) {
        if (player == null) return false;

        AttributeInstance attackSpeed = player.getAttribute(Attribute.ATTACK_SPEED);
        double speed = attackSpeed == null ? 4.0D : attackSpeed.getValue();

        // CombatePlus configura 100.0 para neutralizar o cooldown 1.9+.
        return isCombatePlusEnabled(player) && speed >= 20.0D;
    }

    public static double entityReach(Player player) {
        if (player == null) return 3.0D;

        AttributeInstance range = player.getAttribute(Attribute.ENTITY_INTERACTION_RANGE);
        if (range == null) return 3.0D;

        return Math.max(0.0D, range.getValue());
    }

    public static int hardCpsLimit(Player player) {
        // PvP 1.8 não possui attack cooldown. Um limite bruto muito baixo causaria
        // falsos positivos em jitter/butterfly legítimos. Regularidade continua sendo analisada.
        return isLegacyCombat(player) ? 30 : 20;
    }

    public static int regularitySampleFloor(Player player) {
        return isLegacyCombat(player) ? 18 : 14;
    }

    public static double velocityTolerance(Player player) {
        // CombatePlus encerra sprint após um hit bem-sucedido, reproduzindo o reset
        // de sprint do PvP antigo. Mantemos margem ligeiramente maior no check.
        return isLegacyCombat(player) ? 0.09D : 0.12D;
    }
}
```

```java
package com.redeedge.anticheatplus.anticheat;

import com.redeedge.anticheatplus.AntiCheatPlusPlugin;
import org.bukkit.Bukkit;
import org.bukkit.Location;
import org.bukkit.entity.ArmorStand;
import org.bukkit.entity.Entity;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.EventPriority;
import org.bukkit.event.Listener;
import org.bukkit.event.entity.EntityDamageByEntityEvent;
import org.bukkit.event.player.PlayerInteractAtEntityEvent;
import org.bukkit.util.Vector;

import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public final class KillAuraDecoyManager implements Listener {

    private final AntiCheatPlusPlugin plugin;
    private final ViolationManager violations;
    private final Map<UUID, UUID> owners = new ConcurrentHashMap<>();
    private final Map<UUID, Long> lastSpawn = new ConcurrentHashMap<>();

    public KillAuraDecoyManager(AntiCheatPlusPlugin plugin, ViolationManager violations) {
        this.plugin = plugin;
        this.violations = violations;
    }

    public void probe(Player player) {
        if (player == null || !player.isOnline()) return;

        long now = System.currentTimeMillis();
        long last = lastSpawn.getOrDefault(player.getUniqueId(), 0L);
        if (now - last < 8_000L) return;
        lastSpawn.put(player.getUniqueId(), now);

        Location base = player.getLocation().clone();
        Vector backward = base.getDirection().setY(0.0D);
        if (backward.lengthSquared() < 0.0001D) {
            backward = new Vector(0.0D, 0.0D, 1.0D);
        } else {
            backward.normalize();
        }
        Location spawn = base.clone()
                .subtract(backward.multiply(1.65D))
                .add(0.0D, 0.15D, 0.0D);

        ArmorStand decoy = player.getWorld().spawn(spawn, ArmorStand.class, stand -> {
            stand.setVisible(false);
            stand.setGravity(false);
            stand.setSilent(true);
            stand.setSmall(false);
            stand.setArms(true);
            stand.setBasePlate(false);
            stand.setCollidable(false);
            stand.setCustomNameVisible(false);
            stand.setPersistent(false);
        });

        owners.put(decoy.getUniqueId(), player.getUniqueId());

        for (Player online : Bukkit.getOnlinePlayers()) {
            if (!online.getUniqueId().equals(player.getUniqueId())) {
                online.hideEntity(plugin, decoy);
            }
        }

        Bukkit.getScheduler().runTaskLater(plugin, () -> remove(decoy), 12L);
    }

    public boolean isDecoy(Entity entity) {
        return entity != null && owners.containsKey(entity.getUniqueId());
    }

    private void remove(Entity entity) {
        if (entity == null) return;
        owners.remove(entity.getUniqueId());
        if (entity.isValid()) entity.remove();
    }

    @EventHandler(priority = EventPriority.HIGHEST)
    public void onDamage(EntityDamageByEntityEvent event) {
        if (!(event.getDamager() instanceof Player attacker)) return;
        UUID expected = owners.get(event.getEntity().getUniqueId());
        if (expected == null) return;

        event.setCancelled(true);
        if (expected.equals(attacker.getUniqueId())) {
            violations.flag(attacker, "KillAura");
        }
        remove(event.getEntity());
    }

    @EventHandler(priority = EventPriority.HIGHEST)
    public void onInteract(PlayerInteractAtEntityEvent event) {
        UUID expected = owners.get(event.getRightClicked().getUniqueId());
        if (expected == null) return;

        event.setCancelled(true);
        if (expected.equals(event.getPlayer().getUniqueId())) {
            violations.flag(event.getPlayer(), "KillAura");
        }
        remove(event.getRightClicked());
    }

    public void clear(UUID playerId) {
        lastSpawn.remove(playerId);
        owners.entrySet().removeIf(entry -> entry.getValue().equals(playerId));
    }
}
```

```java
package com.redeedge.anticheatplus.anticheat;

import com.redeedge.anticheatplus.AntiCheatPlusPlugin;
import org.bukkit.Bukkit;
import org.bukkit.entity.Player;

import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public final class ViolationManager {

    private final AntiCheatPlusPlugin plugin;
    private final Map<ViolationKey, ViolationState> states = new ConcurrentHashMap<>();

    public ViolationManager(AntiCheatPlusPlugin plugin) {
        this.plugin = plugin;
    }

    public int flag(Player player, String checkName) {
        if (player == null || !player.isOnline() || checkName == null || checkName.isBlank()) {
            return 0;
        }

        String checkKey = checkName.toLowerCase(java.util.Locale.ROOT);
        if (!plugin.getConfig().getBoolean("anticheat.checks." + checkKey + ".enabled", true)) {
            return 0;
        }

        long now = System.currentTimeMillis();
        long resetAfter = Math.max(30_000L,
                plugin.getConfig().getLong("anticheat.alerts.reset-after-ms", 120_000L));
        long minimumGap = Math.max(500L,
                plugin.getConfig().getLong("anticheat.alerts.minimum-gap-ms", 5_000L));
        long cycleResetAfter = Math.max(minimumGap,
                plugin.getConfig().getLong("anticheat.alerts.cycle-reset-after-ms", 60_000L));

        ViolationKey key = new ViolationKey(player.getUniqueId(), checkKey);
        ViolationState state = states.computeIfAbsent(key, ignored -> new ViolationState());

        synchronized (state) {
            if (state.lastDetection > 0L && now - state.lastDetection > resetAfter) {
                state.stage = 0;
                state.stageThreeSince = 0L;
                state.lastAlert = 0L;
            }

            state.lastDetection = now;

            if (state.stage >= 3
                    && state.stageThreeSince > 0L
                    && now - state.stageThreeSince >= cycleResetAfter) {
                state.stage = 0;
                state.stageThreeSince = 0L;
                state.lastAlert = 0L;
            }

            if (state.lastAlert > 0L && now - state.lastAlert < minimumGap) {
                return state.stage;
            }

            int stage;
            if (state.stage < 3) {
                state.stage++;
                stage = state.stage;
                if (stage == 3) {
                    state.stageThreeSince = now;
                }
            } else {
                stage = 3;
            }

            state.lastAlert = now;
            if (Bukkit.isPrimaryThread()) {
                broadcast(player, checkName, stage);
            } else {
                Bukkit.getScheduler().runTask(plugin, () -> {
                    if (player.isOnline()) {
                        broadcast(player, checkName, stage);
                    }
                });
            }
            return stage;
        }
    }

    public int getStage(UUID playerId, String checkName) {
        if (playerId == null || checkName == null) return 0;
        ViolationState state = states.get(new ViolationKey(playerId, checkName.toLowerCase(java.util.Locale.ROOT)));
        return state == null ? 0 : state.stage;
    }

    public void clear(UUID playerId) {
        if (playerId == null) return;
        states.keySet().removeIf(key -> key.playerId.equals(playerId));
    }

    private void broadcast(Player suspect, String checkName, int stage) {
        String message = switch (stage) {
            case 1 -> "&c&lGUARDIÃO &f" + suspect.getName()
                    + " possivelmente está usando &c" + checkName + "&f.";
            case 2 -> "&c&lGUARDIÃO &f" + suspect.getName()
                    + " está suspeito de uso de &c" + checkName + "&f.";
            default -> "&c&lGUARDIÃO &f" + suspect.getName()
                    + " apresenta forte indício de uso de &c" + checkName + "&f.";
        };

        String colored = plugin.color(message);
        for (Player staff : Bukkit.getOnlinePlayers()) {
            if (staff.hasPermission("anticheatplus.alertas")) {
                staff.sendMessage(colored);
            }
        }

        plugin.getLogger().warning("[GUARDIAO] " + suspect.getName()
                + " falhou " + checkName + " (nível " + stage + "/3)");
    }

    private record ViolationKey(UUID playerId, String checkName) {
    }

    private static final class ViolationState {
        private int stage;
        private long lastDetection;
        private long lastAlert;
        private long stageThreeSince;
    }
}
```

```java
package com.redeedge.anticheatplus.anticheat.listeners;

import com.redeedge.anticheatplus.anticheat.ViolationManager;
import org.bukkit.Location;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.EventPriority;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerMoveEvent;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public final class BehaviorCheckListener implements Listener {

    private final ViolationManager violations;
    private final Map<UUID, BehaviorState> states = new ConcurrentHashMap<>();

    public BehaviorCheckListener(ViolationManager violations) {
        this.violations = violations;
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onMove(PlayerMoveEvent event) {
        Location to = event.getTo();
        if (to == null) return;

        Player player = event.getPlayer();
        BehaviorState state = states.computeIfAbsent(
                player.getUniqueId(),
                ignored -> new BehaviorState()
        );

        long now = System.currentTimeMillis();
        if (now - state.lastSampleAt < 500L) return;
        state.lastSampleAt = now;

        MotionSignature signature = MotionSignature.from(to);
        state.samples.addLast(signature);
        while (state.samples.size() > 72) {
            state.samples.removeFirst();
        }

        if (hasRepeatedRoute(state.samples, 8, 3)
                || hasRepeatedRoute(state.samples, 12, 3)) {
            violations.flag(player, "AutoBot");
            state.samples.clear();
        }
    }

    private boolean hasRepeatedRoute(
            Deque<MotionSignature> samples,
            int patternLength,
            int repetitions
    ) {
        int required = patternLength * repetitions;
        if (samples.size() < required) return false;

        MotionSignature[] array = samples.toArray(MotionSignature[]::new);
        int start = array.length - required;

        for (int repetition = 1; repetition < repetitions; repetition++) {
            int offset = start + repetition * patternLength;
            for (int i = 0; i < patternLength; i++) {
                if (!array[start + i].approximatelyEquals(array[offset + i])) {
                    return false;
                }
            }
        }

        int unique = 0;
        MotionSignature previous = null;
        for (int i = 0; i < patternLength; i++) {
            MotionSignature current = array[start + i];
            if (previous == null || !current.approximatelyEquals(previous)) unique++;
            previous = current;
        }

        return unique >= 4;
    }

    public void clear(UUID playerId) {
        states.remove(playerId);
    }

    private record MotionSignature(int x, int y, int z, int yawSector, int pitchSector) {

        private static MotionSignature from(Location location) {
            return new MotionSignature(
                    location.getBlockX(),
                    location.getBlockY(),
                    location.getBlockZ(),
                    Math.floorMod(Math.round(location.getYaw() / 15.0F), 24),
                    Math.round(location.getPitch() / 15.0F)
            );
        }

        private boolean approximatelyEquals(MotionSignature other) {
            if (other == null) return false;
            return Math.abs(x - other.x) <= 1
                    && Math.abs(y - other.y) <= 1
                    && Math.abs(z - other.z) <= 1
                    && circularSectorDistance(yawSector, other.yawSector, 24) <= 1
                    && Math.abs(pitchSector - other.pitchSector) <= 1;
        }

        private int circularSectorDistance(int a, int b, int modulo) {
            int diff = Math.abs(a - b);
            return Math.min(diff, modulo - diff);
        }
    }

    private static final class BehaviorState {
        private final Deque<MotionSignature> samples = new ArrayDeque<>();
        private long lastSampleAt;
    }
}
```

```java
        BlockState state = state(player);
        long now = System.currentTimeMillis();

        state.placeTimes.addLast(now);

        // blocos no mesmo tick. Agrupamos eventos em uma janela curta de tick.

    private void checkImpossiblePlacement(Player player, BlockPlaceEvent event) {
        Block placed = event.getBlockPlaced();
        Location eye = player.getEyeLocation();
        Location center = placed.getLocation().add(0.5D, 0.5D, 0.5D);

        if (CheckUtil.sameWorld(eye, center) && eye.distance(center) > 6.25D) {
            violations.flag(player, "BadPackets");
        }

        if (!placed.getType().isBlock() || !placed.getType().isSolid()) return;

        boolean hasSupport = false;
        for (BlockFace face : new BlockFace[]{
                BlockFace.DOWN, BlockFace.UP, BlockFace.NORTH,
                BlockFace.SOUTH, BlockFace.EAST, BlockFace.WEST
        }) {
            if (!placed.getRelative(face).getType().isAir()) {
                hasSupport = true;
                break;
            }
        }

        if (!hasSupport && event.getBlockAgainst().getType().isAir()) {
            violations.flag(player, "BadPackets");
        }
    }

    private void checkScaffold(Player player, Block placed, BlockState state, long now) {
        Location playerLocation = player.getLocation();
        Location blockCenter = placed.getLocation().add(0.5D, 0.5D, 0.5D);

        double horizontal = CheckUtil.horizontalDistance(playerLocation, blockCenter);
        boolean underFeet = placed.getY() <= playerLocation.getBlockY()
                && playerLocation.getY() - placed.getY() <= 2.2D
                && horizontal <= 1.75D;

        double velocityHorizontal = Math.sqrt(
                player.getVelocity().getX() * player.getVelocity().getX()
                        + player.getVelocity().getZ() * player.getVelocity().getZ()
        );

        boolean notLookingDown = playerLocation.getPitch() < 35.0F;
        boolean moving = velocityHorizontal > 0.10D || player.isSprinting();

        // Aoba Scaffold procura posições vazias em um raio configurável
        // (4 blocos por padrão) e pode colocar múltiplos blocos no mesmo tick.
        boolean bridgeLevel = placed.getY() <= playerLocation.getBlockY()
                && playerLocation.getY() - placed.getY() <= 2.4D
                && horizontal > 1.75D
                && horizontal <= 4.5D;

        if (bridgeLevel && moving && state.sameTickPlaces >= 2) {
            state.aobaRadiusScaffoldBuffer += 2;
            if (state.aobaRadiusScaffoldBuffer >= 4) {
                violations.flag(player, "Scaffold");
                state.aobaRadiusScaffoldBuffer = 0;
            }
        } else {
            state.aobaRadiusScaffoldBuffer = Math.max(0, state.aobaRadiusScaffoldBuffer - 1);
        }

        if (underFeet && notLookingDown && moving && !player.isSneaking()) {
            state.scaffoldBuffer += 2;
        } else {
            state.scaffoldBuffer = Math.max(0, state.scaffoldBuffer - 1);
        }

        // Meteor Fast Tower usa 0.5 Y por padrão enquanto coloca o bloco
        // diretamente abaixo do jogador.
        double velocityY = player.getVelocity().getY();
        if (underFeet
                && velocityY >= 0.46D
                && velocityY <= 0.56D
                && !player.isSneaking()) {
            state.meteorTowerBuffer++;
            if (state.meteorTowerBuffer >= 3) {
                violations.flag(player, "Scaffold");
                state.meteorTowerBuffer = 0;
            }
        } else {
            state.meteorTowerBuffer = Math.max(0, state.meteorTowerBuffer - 1);
        }

        // ScaffoldWalk do Wurst coloca automaticamente sob os pés e trabalha
        // em cadência curta, mesmo quando a rotação enviada parece legítima.
        if (underFeet && moving && !player.isSneaking() && state.lastPlaceAt > 0L) {
            long cadence = now - state.lastPlaceAt;
            if (cadence >= 70L && cadence <= 280L) {
                state.underFeetCadenceBuffer++;
                if (state.underFeetCadenceBuffer >= 6) {
                    violations.flag(player, "Scaffold");
                    state.underFeetCadenceBuffer = 0;
                }
            } else {
                state.underFeetCadenceBuffer = Math.max(0, state.underFeetCadenceBuffer - 1);
            }
        } else {
            state.underFeetCadenceBuffer = Math.max(0, state.underFeetCadenceBuffer - 1);
        }

        if (state.lastPlacedLocation != null && state.lastPlaceAt > 0L) {
            long interval = now - state.lastPlaceAt;
            int dx = Math.abs(placed.getX() - state.lastPlacedLocation.getBlockX());
            int dy = placed.getY() - state.lastPlacedLocation.getBlockY();
            int dz = Math.abs(placed.getZ() - state.lastPlacedLocation.getBlockZ());

            boolean diagonalBridge = dx == 1 && dz == 1 && Math.abs(dy) <= 1;
            boolean tower = dx == 0 && dz == 0 && dy == 1;

            if (interval < 115L && diagonalBridge && moving) {
                state.scaffoldBuffer += 2;
            }

            if (interval < 145L && tower && !player.isSneaking()) {
                state.towerBuffer++;
                if (state.towerBuffer >= 3) {
                    violations.flag(player, "Scaffold");
                    state.towerBuffer = 0;
                }
            } else {
                state.towerBuffer = Math.max(0, state.towerBuffer - 1);
            }
        }

        state.lastPlacedLocation = placed.getLocation().clone();
        state.lastPlaceAt = now;

        if (state.scaffoldBuffer >= 5) {
            violations.flag(player, "Scaffold");
            state.scaffoldBuffer = 0;
        }
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onBreak(BlockBreakEvent event) {
        Player player = event.getPlayer();
        if (player.getGameMode() == GameMode.CREATIVE) return;

        BlockState state = state(player);
        long now = System.currentTimeMillis();
        Block block = event.getBlock();

        ItemStack tool = player.getInventory().getItemInMainHand();
        long minimum = estimatedMinimumBreakMillis(block.getType(), tool, player);
        boolean legitimateInstantBreak = minimum > 0L && minimum <= 25L;

        if (checkNuker(player, block, state, now)
                || checkMultiBreak(player, block, state, now)) {
            event.setCancelled(true);
            state.damageStart.remove(BlockKey.of(block));
            return;
        }

        Long start = state.damageStart.remove(BlockKey.of(block));
        if (start != null) {
            long elapsed = now - start;

            if (minimum > 0L && elapsed + 35L < minimum) {
                violations.flag(player, "FastBreak");
            }

            // O Aoba permite multiplicadores discretos (1.25x, 1.5x...)
            // que podem ficar acima do limite duro. Só acumulamos suspeita
            // após várias quebras e ignoramos Haste legítimo.
            //
            // Ferramentas legítimas capazes de instant-break (por exemplo,
            // picaretas customizadas com Eficiência X da loja) não podem
            // alimentar este buffer apenas por quebrarem dentro do mesmo tick.
            boolean hasHaste = player.getActivePotionEffects().stream()
                    .anyMatch(effect -> effect.getType().getKey().getKey().contains("haste"));
            long softMinimum = Math.max(minimum, Math.round(minimum / 0.42D * 0.70D));
            if (!legitimateInstantBreak
                    && !hasHaste
                    && softMinimum > 80L
                    && elapsed + 25L < softMinimum) {
                state.softFastBreakBuffer++;
                if (state.softFastBreakBuffer >= 5) {
                    violations.flag(player, "FastBreak");
                    state.softFastBreakBuffer = 0;
                }
            } else {
                state.softFastBreakBuffer = Math.max(0, state.softFastBreakBuffer - 1);
            }
        } else if (!legitimateInstantBreak && block.getType().getHardness() > 1.0F) {
            // Ausência de BlockDamageEvent é apenas suspeita quando a ferramenta
            // não consegue quebrar legitimamente o bloco de forma instantânea.
            violations.flag(player, "FastBreak");
        }

        // Em instant-break legítimo, o cliente/servidor não garante uma
        // animação de swing observável para cada BlockBreakEvent. Portanto,
        // ausência de swing só é evidência quando o bloco não pode ser
        // quebrado legitimamente de forma instantânea com a ferramenta atual.
        if (!legitimateInstantBreak
                && now - state.lastSwingAt > 450L
                && block.getType().getHardness() > 0.0F) {
            violations.flag(player, "FastBreak");
        }

        if (state.lastBreakAt > 0L) {
            long gap = now - state.lastBreakAt;
            if (!legitimateInstantBreak
                    && gap <= 35L
                    && block.getType().getHardness() > 0.2F) {
                state.zeroDelayBreakBuffer++;
                if (state.zeroDelayBreakBuffer >= 5) {
                    violations.flag(player, "FastBreak");
                    state.zeroDelayBreakBuffer = 0;
                }
            } else {
                state.zeroDelayBreakBuffer = Math.max(0, state.zeroDelayBreakBuffer - 1);
            }
        }
        state.lastBreakAt = now;

        state.breakTimes.addLast(now);
        trimOneSecond(state.breakTimes, now);
        if (!legitimateInstantBreak && state.breakTimes.size() > 14) {
            violations.flag(player, "FastBreak");
        }

        registerBotAction(player, state.breakBotTimes, now);
    }

    private boolean checkMultiBreak(Player player, Block block, BlockState state, long now) {
        Location center = block.getLocation().add(0.5D, 0.5D, 0.5D);

        // Janela equivalente a aproximadamente um tick. O objetivo não é
        // reconhecer um formato específico (3x3, 5x5, túnel, cubo etc.),
        // mas impedir que um client efetive várias quebras distintas de uma vez.
        if (state.multiBreakWindowStarted == 0L
                || now - state.multiBreakWindowStarted > 65L) {
            state.multiBreakWindowStarted = now;
            state.multiBreakCount = 1;
            state.multiBreakBlocks.clear();
            state.multiBreakBlocks.add(BlockKey.of(block));
            return false;
        }

        BlockKey key = BlockKey.of(block);
        if (!state.multiBreakBlocks.add(key)) {
            return false;
        }

        state.multiBreakCount++;

        // Em Survival/Adventure, uma única ação legítima não deve produzir
        // uma rajada de blocos distintos no mesmo tick. Hacks de AreaMine,
        // Tunnel, Excavator, PacketMine e Nuker editável fazem exatamente isso.
        if (state.multiBreakCount >= 2) {
            state.multiBreakBuffer += state.multiBreakCount >= 4 ? 2 : 1;

            if (state.multiBreakBuffer >= 3) {
                violations.flag(player, "Nuker");
                state.multiBreakBuffer = 0;
            }

            // O primeiro bloco da janela pode ser a quebra legítima que iniciou
            // a ação. Todos os alvos extras são cancelados antes de alterarem
            // o mundo, independentemente do formato da área configurada.
            return true;
        }

        state.multiBreakBuffer = Math.max(0, state.multiBreakBuffer - 1);
        return false;
    }

    private boolean checkNuker(Player player, Block block, BlockState state, long now) {
        Location eye = player.getEyeLocation();
        Location center = block.getLocation().add(0.5D, 0.5D, 0.5D);

        if (!CheckUtil.sameWorld(eye, center)) {
            state.nukerBuffer = 0;
            state.lastBrokenLocation = center.clone();
            state.lastNukerBreakAt = now;
            return false;
        }

        double distance = eye.distance(center);

        // Fora do alcance plausível: bloqueamos antes do Bukkit efetivar a quebra.
        if (distance > 6.25D) {
            violations.flag(player, "Nuker");
            state.nukerBuffer = 0;
            return true;
        }

        // Uma quebra legítima precisa corresponder ao primeiro bloco atingido
        // pelo raio da câmera. O Nuker do Aoba envia START/STOP_DESTROY_BLOCK
        // diretamente para alvos do raio sem precisar que o jogador esteja
        // olhando para eles.
        RayTraceResult ray = player.getWorld().rayTraceBlocks(
                eye,
                eye.getDirection(),
                6.25D,
                FluidCollisionMode.NEVER,
                true
        );

        Block aimed = ray == null ? null : ray.getHitBlock();
        boolean aimedAtTarget = aimed != null
                && aimed.getWorld().getUID().equals(block.getWorld().getUID())
                && aimed.getX() == block.getX()
                && aimed.getY() == block.getY()
                && aimed.getZ() == block.getZ();

        if (!aimedAtTarget) {
            state.nukerBuffer += 2;

            // O bloco é preservado imediatamente. Só elevamos o alerta após
            // repetição, para um evento isolado não acusar o jogador.
            if (state.nukerBuffer >= 4) {
                violations.flag(player, "Nuker");
                state.nukerBuffer = 0;
            }

            return true;
        }

        long gap = state.lastNukerBreakAt <= 0L
                ? Long.MAX_VALUE
                : now - state.lastNukerBreakAt;

        boolean rapidBreak = gap <= 300L;
        boolean spatialJump = false;

        if (state.lastBrokenLocation != null
                && CheckUtil.sameWorld(state.lastBrokenLocation, center)
                && rapidBreak) {
            double jump = state.lastBrokenLocation.distance(center);
            spatialJump = jump > 2.35D;
        }

        // Se a câmera acompanha corretamente o bloco, não bloqueamos apenas
        // por velocidade: Eficiência X e Super Quebrador podem ser legítimos.
        // Saltos espaciais rápidos ainda alimentam evidência para o Guardião.
        if (rapidBreak && spatialJump) {
            state.nukerBuffer++;
            if (state.nukerBuffer >= 5) {
                violations.flag(player, "Nuker");
                state.nukerBuffer = 0;
            }
        } else {
            state.nukerBuffer = Math.max(0, state.nukerBuffer - 1);
        }

        state.lastBrokenLocation = center.clone();
        state.lastNukerBreakAt = now;
        return false;
    }

    private long estimatedMinimumBreakMillis(Material material, ItemStack tool, Player player) {
        float hardness = Math.max(0.0F, material.getHardness());
        if (hardness <= 0.0F) return 0L;

        String block = material.name();
        String held = tool == null ? "" : tool.getType().name();

        double speed = 1.0D;
        if (held.contains("NETHERITE")) speed = 9.0D;
        else if (held.contains("DIAMOND")) speed = 8.0D;
        else if (held.contains("IRON")) speed = 6.0D;
        else if (held.contains("STONE")) speed = 4.0D;
        else if (held.contains("WOODEN") || held.contains("GOLDEN")) speed = 2.0D;

        boolean likelyCorrectTool =
                (isPickaxeBlock(block) && held.contains("PICKAXE"))
                        || (isAxeBlock(block) && held.contains("AXE"))
                        || (isShovelBlock(block) && held.contains("SHOVEL"));

        if (!likelyCorrectTool) speed = Math.min(speed, 1.5D);

        Enchantment efficiency = Enchantment.getByName("EFFICIENCY");
        if (efficiency == null) efficiency = Enchantment.getByName("DIG_SPEED");
        int level = efficiency == null || tool == null ? 0 : tool.getEnchantmentLevel(efficiency);
        if (level > 0) speed += (level * level) + 1.0D;

        // HabilidadesPlus/Super Quebrador aplica um modificador transitório
        // server-side em MINING_EFFICIENCY. Esse bônus é legítimo e precisa
        // entrar no mesmo cálculo físico do FastBreak; não desativamos o check.
        if (player != null && likelyCorrectTool) {
            AttributeInstance miningEfficiency = player.getAttribute(Attribute.MINING_EFFICIENCY);
            if (miningEfficiency != null) {
                speed += Math.max(0.0D, miningEfficiency.getValue());
            }
        }

        double ticks = Math.max(1.0D, (hardness * 30.0D) / Math.max(1.0D, speed));
        long vanillaLike = Math.round(ticks * 50.0D);

        // Conservador: só acusa quando a quebra fica muito abaixo do tempo físico estimado.
        return Math.max(25L, Math.round(vanillaLike * 0.42D));
    }

    private boolean isPickaxeBlock(String name) {
        return name.contains("STONE")
                || name.contains("ORE")
                || name.contains("DEEPSLATE")
                || name.contains("OBSIDIAN")
                || name.contains("ANCIENT_DEBRIS")
                || name.contains("METAL")
                || name.contains("IRON")
                || name.contains("GOLD")
                || name.contains("COPPER");
    }

    private boolean isAxeBlock(String name) {
        return name.contains("LOG")
                || name.contains("WOOD")
                || name.contains("PLANK")
                || name.contains("STEM")
                || name.contains("HYPHAE");
    }

    private boolean isShovelBlock(String name) {
        return name.contains("DIRT")
                || name.contains("SAND")
                || name.contains("GRAVEL")
                || name.contains("CLAY")
                || name.contains("SNOW");
    }

    private void registerBotAction(Player player, Deque<Long> times, long now) {
        times.addLast(now);
        while (times.size() > 36) {
            times.removeFirst();
        }

        if (times.size() < 24) return;

        Long[] values = times.toArray(Long[]::new);
        double sum = 0.0D;
        for (int i = 1; i < values.length; i++) {
            sum += values[i] - values[i - 1];
        }

        double mean = sum / (values.length - 1);
        if (mean < 180.0D || mean > 5_000.0D) return;

        double variance = 0.0D;
        for (int i = 1; i < values.length; i++) {
            double diff = (values[i] - values[i - 1]) - mean;
            variance += diff * diff;
        }

        double deviation = Math.sqrt(variance / (values.length - 1));
        if (deviation < 7.0D) {
            violations.flag(player, "AutoBot");
            times.clear();
        }
    }

    private boolean hasHighlyRegularIntervals(Deque<Long> times, int minimumSamples, double maxDeviation) {
        if (times.size() < minimumSamples) return false;

        Long[] array = times.toArray(Long[]::new);
        double sum = 0.0D;
        int count = 0;
        for (int i = 1; i < array.length; i++) {
            long delta = array[i] - array[i - 1];
            if (delta <= 0L) continue;
            sum += delta;
            count++;
        }
        if (count < minimumSamples - 1) return false;

        double mean = sum / count;
        double variance = 0.0D;
        for (int i = 1; i < array.length; i++) {
            long delta = array[i] - array[i - 1];
            if (delta <= 0L) continue;
            double diff = delta - mean;
            variance += diff * diff;
        }

        return Math.sqrt(variance / count) <= maxDeviation;
    }

    private void trimOneSecond(Deque<Long> times, long now) {
        while (!times.isEmpty() && now - times.peekFirst() > 1_000L) {
            times.removeFirst();
        }
    }

    public void clear(UUID playerId) {
        states.remove(playerId);
    }

    private BlockState state(Player player) {
        return states.computeIfAbsent(player.getUniqueId(), ignored -> new BlockState());
    }

    private record BlockKey(UUID world, int x, int y, int z) {
        private static BlockKey of(Block block) {
            return new BlockKey(
                    block.getWorld().getUID(),
                    block.getX(),
                    block.getY(),
                    block.getZ()
            );
        }
    }

        private final Deque<Long> breakTimes = new ArrayDeque<>();
        private final Deque<Long> placeBotTimes = new ArrayDeque<>();
        private final Deque<Long> breakBotTimes = new ArrayDeque<>();
        private final Map<BlockKey, Long> damageStart = new HashMap<>();
        private int scaffoldBuffer;
        private int towerBuffer;
        private int underFeetCadenceBuffer;
        private int aobaRadiusScaffoldBuffer;
        private int meteorTowerBuffer;
        private int zeroDelayBreakBuffer;
        private int softFastBreakBuffer;
        private int nukerBuffer;
        private int multiBreakBuffer;
        private int multiBreakCount;
        private int sameTickPlaces;
        private long placeTickWindowStarted;
        private long lastSwingAt;
        private long lastPlaceAt;
        private long lastBreakAt;
        private long lastNukerBreakAt;
        private long multiBreakWindowStarted;
        private Location lastPlacedLocation;

        private final java.util.Set<BlockKey> multiBreakBlocks = new java.util.HashSet<>();
    }
```

```java
@SuppressWarnings("deprecation")
public final class ChatCheckListener implements Listener {

    private final ViolationManager violations;
    private final Map<UUID, ChatState> states = new ConcurrentHashMap<>();

    public ChatCheckListener(ViolationManager violations) {
        this.violations = violations;
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onChat(AsyncPlayerChatEvent event) {
        Player player = event.getPlayer();
        long now = System.currentTimeMillis();
        String normalized = event.getMessage().trim().replaceAll("\\s+", " ").toLowerCase();

        ChatState state = states.computeIfAbsent(player.getUniqueId(), ignored -> new ChatState());

        boolean spam = false;

        if (state.lastMessageAt > 0L && now - state.lastMessageAt < 650L) {
            violations.flag(player, "Spam");
            spam = true;
        }

        if (state.lastMessage != null
                && state.lastMessage.equals(normalized)
                && now - state.lastMessageAt < 3_000L) {
            violations.flag(player, "Spam");
            spam = true;
        }

        if (spam) {
            event.setCancelled(true);
        }

        state.lastMessage = normalized;
        state.lastMessageAt = now;
    }

    public void clear(UUID playerId) {
        states.remove(playerId);
    }

    private static final class ChatState {
        private String lastMessage;
        private long lastMessageAt;
    }
}
```

```java
package com.redeedge.anticheatplus.anticheat.listeners;

import com.redeedge.anticheatplus.AntiCheatPlusPlugin;
import com.redeedge.anticheatplus.anticheat.CheckUtil;
import com.redeedge.anticheatplus.anticheat.CombatCompatibility;
import com.redeedge.anticheatplus.anticheat.KillAuraDecoyManager;
import com.redeedge.anticheatplus.anticheat.ViolationManager;
import org.bukkit.Location;
import org.bukkit.entity.Entity;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.EventPriority;
import org.bukkit.event.Listener;
import org.bukkit.event.entity.EntityDamageByEntityEvent;
import org.bukkit.event.inventory.InventoryType;
import org.bukkit.event.player.PlayerAnimationEvent;

public final class CombatCheckListener implements Listener {

    private final AntiCheatPlusPlugin plugin;
    private final ViolationManager violations;
    private final KillAuraDecoyManager decoys;
    private final Map<UUID, CombatState> states = new ConcurrentHashMap<>();
    private final Map<UUID, Deque<TargetSnapshot>> targetHistory = new ConcurrentHashMap<>();

    public CombatCheckListener(
            AntiCheatPlusPlugin plugin,
            ViolationManager violations,
            KillAuraDecoyManager decoys
    ) {
        this.plugin = plugin;
        this.violations = violations;
        this.decoys = decoys;
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onMove(PlayerMoveEvent event) {
        Location to = event.getTo();
        if (to == null) return;

        Player player = event.getPlayer();
        CombatState state = state(player);

        float yawDelta = angleDifference(to.getYaw(), event.getFrom().getYaw());
        float pitchDelta = Math.abs(to.getPitch() - event.getFrom().getPitch());
        float signedYawDelta = signedAngleDifference(to.getYaw(), event.getFrom().getYaw());
        long now = System.currentTimeMillis();

        // Aoba KBDisplacer faz um snap de yaw (180° por padrão), envia a
        // rotação e retorna imediatamente. Só marcamos após repetição em combate.
        if (Math.abs(signedYawDelta) >= 120.0F
                && Math.abs(state.lastSignedYawDelta) >= 120.0F
                && Math.signum(signedYawDelta) != Math.signum(state.lastSignedYawDelta)
                && now - state.lastRotationSampleAt <= 140L
                && now - state.lastCombatAt <= 350L) {
            state.spoofTurnBuffer++;
            if (state.spoofTurnBuffer >= 3) {
                violations.flag(player, "KillAura");
                state.spoofTurnBuffer = 0;
            }
        } else if (now - state.lastCombatAt > 500L) {
            state.spoofTurnBuffer = Math.max(0, state.spoofTurnBuffer - 1);
        }

        state.lastSignedYawDelta = signedYawDelta;
        state.lastRotationSampleAt = now;

        state.previousYawDelta = state.lastYawDelta;
        state.previousPitchDelta = state.lastPitchDelta;
        state.lastYawDelta = yawDelta;
        state.lastPitchDelta = pitchDelta;

        if (yawDelta > 0.01F || pitchDelta > 0.01F) {
            state.rotationSamples.addLast(new RotationSample(
                    System.currentTimeMillis(),
                    yawDelta,
                    pitchDelta
            ));
            while (state.rotationSamples.size() > 40) state.rotationSamples.removeFirst();
        }

        rememberTargetPosition(player, to);
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onAttack(EntityDamageByEntityEvent event) {
        if (!(event.getDamager() instanceof Player attacker)) return;
        if (decoys.isDecoy(event.getEntity())) return;

        Entity target = event.getEntity();
        CombatState state = state(attacker);
        long now = System.currentTimeMillis();

        state.lastCombatAt = now;

        boolean auraSuspicious = checkKillAura(attacker, target, state, now);
        checkHitbox(attacker, target);
        checkRotationPattern(attacker, state);

        state.attackCount++;
        if (auraSuspicious
                || violations.getStage(attacker.getUniqueId(), "KillAura") > 0
                || state.attackCount % 4 == 0) {
            decoys.probe(attacker);
        }

        if (target instanceof Player victim) {
            state(victim).lastReceivedCombatHitAt = now;
        }
    }

    private boolean checkKillAura(Player attacker, Entity target, CombatState state, long now) {
        boolean suspicious = false;
        Location eye = attacker.getEyeLocation();
        Location center = target.getLocation().clone()
                .add(0.0D, Math.max(0.2D, target.getHeight() * 0.5D), 0.0D);

        if (CheckUtil.sameWorld(eye, center)) {
            Vector toward = center.toVector().subtract(eye.toVector());
            if (toward.lengthSquared() > 0.0001D) {
                toward.normalize();
                double facing = eye.getDirection().normalize().dot(toward);
                if (facing < 0.10D) {
                    violations.flag(attacker, "KillAura");
                    suspicious = true;
                }
            }
        }

        UUID targetId = target.getUniqueId();
        if (state.lastTarget != null
                && !state.lastTarget.equals(targetId)
                && now - state.lastTargetAt < 170L) {
            violations.flag(attacker, "KillAura");
            suspicious = true;
        }

        state.lastTarget = targetId;
        state.lastTargetAt = now;

        InventoryType topType = attacker.getOpenInventory().getTopInventory().getType();
        if (topType != InventoryType.CRAFTING && topType != InventoryType.CREATIVE) {
            violations.flag(attacker, "KillAura");
            suspicious = true;
        }

        return suspicious;
    }

    private void checkRotationPattern(Player attacker, CombatState state) {
        float yaw = state.lastYawDelta;
        float pitch = state.lastPitchDelta;

        if (yaw <= 0.0F && pitch <= 0.0F) return;

        double acceleration = Math.abs(yaw - state.previousYawDelta)
                + Math.abs(pitch - state.previousPitchDelta);

        if (yaw > 25.0F && acceleration < 0.015D) {
            state.roboticRotationBuffer++;
        } else if (nearlyRepeatedRotation(state.rotationSamples)) {
            state.roboticRotationBuffer++;
        } else {
            state.roboticRotationBuffer = Math.max(0, state.roboticRotationBuffer - 1);
        }

        if (state.roboticRotationBuffer >= 4) {
            violations.flag(attacker, "KillAura");
            state.roboticRotationBuffer = 0;
        }
    }

    private boolean nearlyRepeatedRotation(Deque<RotationSample> samples) {
        if (samples.size() < 8) return false;

        RotationSample[] array = samples.toArray(RotationSample[]::new);
        int start = Math.max(1, array.length - 8);
        double sumDiff = 0.0D;
        int compared = 0;

        for (int i = start; i < array.length; i++) {
            RotationSample previous = array[i - 1];
            RotationSample current = array[i];
            sumDiff += Math.abs(current.yawDelta - previous.yawDelta);
            sumDiff += Math.abs(current.pitchDelta - previous.pitchDelta);
            compared++;
        }

        return compared >= 6 && (sumDiff / compared) < 0.025D;
    }

    private void checkHitbox(Player attacker, Entity target) {
        Location eye = attacker.getEyeLocation();
        if (!CheckUtil.sameWorld(eye, target.getLocation())) return;

        double configuredReach = CombatCompatibility.entityReach(attacker);
        double maxReach = Math.max(3.0D, configuredReach) + 0.55D;
        Vector start = eye.toVector();
        Vector direction = eye.getDirection().normalize();

        boolean rayHit = intersectsEntityBox(start, direction, maxReach, target.getLocation(),
                target.getWidth(), target.getHeight(), 0.10D);

        if (!rayHit) {
            Deque<TargetSnapshot> history = targetHistory.get(target.getUniqueId());
            if (history != null) {
                long now = System.currentTimeMillis();
                for (TargetSnapshot snapshot : history) {
                    if (now - snapshot.time > 250L) continue;
                    if (intersectsEntityBox(start, direction, maxReach, snapshot.location,
                            snapshot.width, snapshot.height, 0.14D)) {
                        rayHit = true;
                        break;
                    }
                }
            }
        }

        Location center = target.getLocation().clone()
                .add(0.0D, Math.max(0.2D, target.getHeight() * 0.5D), 0.0D);
        double centerDistance = eye.distance(center);
        double centerLimit = Math.max(3.0D, configuredReach) + 0.85D;

        if (!rayHit || centerDistance > centerLimit || !attacker.hasLineOfSight(target)) {
            violations.flag(attacker, "Hitbox");
        }
    }

    private boolean intersectsEntityBox(
            Vector origin,
            Vector direction,
            double maxDistance,
            Location location,
            double width,
            double height,
            double expansion
    ) {
        double half = Math.max(0.15D, width * 0.5D) + expansion;
        double minX = location.getX() - half;
        double maxX = location.getX() + half;
        double minY = location.getY() - expansion;
        double maxY = location.getY() + Math.max(0.4D, height) + expansion;
        double minZ = location.getZ() - half;
        double maxZ = location.getZ() + half;

        double tMin = 0.0D;
        double tMax = maxDistance;

        double[] o = {origin.getX(), origin.getY(), origin.getZ()};
        double[] d = {direction.getX(), direction.getY(), direction.getZ()};
        double[] min = {minX, minY, minZ};
        double[] max = {maxX, maxY, maxZ};

        for (int axis = 0; axis < 3; axis++) {
            if (Math.abs(d[axis]) < 1.0E-9D) {
                if (o[axis] < min[axis] || o[axis] > max[axis]) return false;
                continue;
            }

            double inv = 1.0D / d[axis];
            double t1 = (min[axis] - o[axis]) * inv;
            double t2 = (max[axis] - o[axis]) * inv;
            if (t1 > t2) {
                double swap = t1;
                t1 = t2;
                t2 = swap;
            }

            tMin = Math.max(tMin, t1);
            tMax = Math.min(tMax, t2);
            if (tMax < tMin) return false;
        }

        return tMin <= maxDistance && tMax >= 0.0D;
    }

    private void rememberTargetPosition(Player player, Location location) {
        Deque<TargetSnapshot> history = targetHistory.computeIfAbsent(
                player.getUniqueId(),
                ignored -> new ArrayDeque<>()
        );
        long now = System.currentTimeMillis();
        history.addFirst(new TargetSnapshot(
                now,
                location.clone(),
                player.getWidth(),
                player.getHeight()
        ));

        while (history.size() > 8) history.removeLast();
        while (!history.isEmpty() && now - history.peekLast().time > 400L) {
            history.removeLast();
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onSwing(PlayerAnimationEvent event) {
        Player player = event.getPlayer();
        CombatState state = state(player);
        long now = System.currentTimeMillis();

        if (now - state.lastCombatAt > 2_000L) {
            state.clicks.clear();
            return;
        }

        Deque<Long> clicks = state.clicks;
        clicks.addLast(now);
        while (!clicks.isEmpty() && now - clicks.peekFirst() > 1_000L) {
            clicks.removeFirst();
        }

        int hardCps = CombatCompatibility.hardCpsLimit(player);
        int regularityFloor = CombatCompatibility.regularitySampleFloor(player);

        if (clicks.size() >= hardCps
                || (clicks.size() >= regularityFloor
                && hasUnnaturallyRegularIntervals(clicks, regularityFloor))) {
            violations.flag(player, "AutoClicker");
        }
    }

    private boolean hasUnnaturallyRegularIntervals(Deque<Long> clicks, int minimumSamples) {
        if (clicks.size() < minimumSamples) return false;

        Long[] values = clicks.toArray(Long[]::new);
        double sum = 0.0D;
        int count = 0;
        for (int i = 1; i < values.length; i++) {
            long delta = values[i] - values[i - 1];
            if (delta <= 0L || delta > 250L) continue;
            sum += delta;
            count++;
        }

        if (count < Math.max(10, minimumSamples - 4)) return false;
        double mean = sum / count;

        double variance = 0.0D;
        int measured = 0;
        for (int i = 1; i < values.length; i++) {
            long delta = values[i] - values[i - 1];
            if (delta <= 0L || delta > 250L) continue;
            double diff = delta - mean;
            variance += diff * diff;
            measured++;
        }

        if (measured == 0) return false;
        double deviation = Math.sqrt(variance / measured);

        int repeated = 0;
        Long previous = null;
        for (int i = 1; i < values.length; i++) {
            long delta = values[i] - values[i - 1];
            if (previous != null && Math.abs(delta - previous) <= 1L) repeated++;
            previous = delta;
        }

        return (mean < 110.0D && deviation < 3.5D) || repeated >= 8;
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onVelocity(PlayerVelocityEvent event) {
        Player player = event.getPlayer();
        CombatState state = state(player);
        long now = System.currentTimeMillis();

        if (now - state.lastReceivedCombatHitAt > 350L) return;

        Vector expected = event.getVelocity().clone();
        double expectedHorizontal = Math.sqrt(
                expected.getX() * expected.getX() + expected.getZ() * expected.getZ()
        );
        if (expectedHorizontal < 0.16D && Math.abs(expected.getY()) < 0.12D) return;

        Location start = player.getLocation().clone();

        plugin.getServer().getScheduler().runTaskLater(plugin, () -> {
            if (!player.isOnline()) return;
            Location end = player.getLocation();
            if (!CheckUtil.sameWorld(start, end)) return;
            if (CheckUtil.isLiquidAround(end) || CheckUtil.isClimbableOrSpecial(end)) return;
            if (CheckUtil.insideSolid(end)) return;

            double movedHorizontal = CheckUtil.horizontalDistance(start, end);
            double movedVertical = Math.abs(end.getY() - start.getY());

            double horizontalTolerance = CombatCompatibility.velocityTolerance(player);
            boolean ignoredHorizontal = expectedHorizontal >= 0.16D
                    && movedHorizontal < Math.max(0.025D, expectedHorizontal * horizontalTolerance);
            boolean ignoredVertical = expected.getY() > 0.18D
                    && movedVertical < Math.max(0.02D, expected.getY() * 0.08D);

            double horizontalRatio = expectedHorizontal <= 0.0001D
                    ? 1.0D
                    : movedHorizontal / expectedHorizontal;
            double verticalRatio = expected.getY() <= 0.0001D
                    ? 1.0D
                    : movedVertical / Math.abs(expected.getY());

            if (ignoredHorizontal || ignoredVertical) {
                violations.flag(player, "Velocity");
                state.velocitySuppressionBuffer = 0;
                return;
            }

            // AntiKnockback configurável pode reduzir parcialmente a resposta em vez de zerá-la.
            boolean stronglyReduced = (expectedHorizontal >= 0.22D && horizontalRatio < 0.38D)
                    || (expected.getY() > 0.22D && verticalRatio < 0.32D);

            if (stronglyReduced) {
                state.velocitySuppressionBuffer++;
                if (state.velocitySuppressionBuffer >= 3) {
                    violations.flag(player, "Velocity");
                    state.velocitySuppressionBuffer = 0;
                }
            } else {
                state.velocitySuppressionBuffer = Math.max(0, state.velocitySuppressionBuffer - 1);
            }
        }, 4L);
    }

    public void clear(UUID playerId) {
        states.remove(playerId);
        targetHistory.remove(playerId);
    }

    private CombatState state(Player player) {
        return states.computeIfAbsent(player.getUniqueId(), ignored -> new CombatState());
    }

    private float angleDifference(float a, float b) {
        return Math.abs(signedAngleDifference(a, b));
    }

    private float signedAngleDifference(float a, float b) {
        float diff = (a - b) % 360.0F;
        if (diff > 180.0F) diff -= 360.0F;
        if (diff < -180.0F) diff += 360.0F;
        return diff;
    }

    private record RotationSample(long time, float yawDelta, float pitchDelta) {
    }

    private record TargetSnapshot(long time, Location location, double width, double height) {
    }

    private static final class CombatState {
        private final Deque<Long> clicks = new ArrayDeque<>();
        private final Deque<RotationSample> rotationSamples = new ArrayDeque<>();
        private UUID lastTarget;
        private long lastTargetAt;
        private long lastCombatAt;
        private long lastReceivedCombatHitAt;
        private float lastYawDelta;
        private float lastPitchDelta;
        private float previousYawDelta;
        private float previousPitchDelta;
        private int roboticRotationBuffer;
        private int velocitySuppressionBuffer;
        private int spoofTurnBuffer;
        private int attackCount;
        private float lastSignedYawDelta;
        private long lastRotationSampleAt;
    }
}
```

```java
package com.redeedge.anticheatplus.anticheat.listeners;

import com.redeedge.anticheatplus.anticheat.CheckUtil;
import com.redeedge.anticheatplus.anticheat.ViolationManager;
import org.bukkit.Location;
import org.bukkit.Material;
import org.bukkit.entity.Player;
import org.bukkit.entity.Vehicle;
import org.bukkit.event.EventHandler;
import org.bukkit.event.EventPriority;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerInteractEvent;
import org.bukkit.event.player.PlayerMoveEvent;
import org.bukkit.event.vehicle.VehicleMoveEvent;
import org.bukkit.inventory.ItemStack;

import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public final class FlightVehicleCheckListener implements Listener {

    private final ViolationManager violations;
    private final Map<UUID, ElytraState> elytraStates = new ConcurrentHashMap<>();
    private final Map<UUID, VehicleState> vehicleStates = new ConcurrentHashMap<>();

    public FlightVehicleCheckListener(ViolationManager violations) {
        this.violations = violations;
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onUseRocket(PlayerInteractEvent event) {
        Player player = event.getPlayer();
        if (!player.isGliding()) return;

        ItemStack item = event.getItem();
        if (item == null || item.getType() != Material.FIREWORK_ROCKET) return;

        switch (event.getAction()) {
            case RIGHT_CLICK_AIR, RIGHT_CLICK_BLOCK ->
                    elytraState(player).lastRocketAt = System.currentTimeMillis();
            default -> {
            }
        }
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onElytraMove(PlayerMoveEvent event) {
        Player player = event.getPlayer();
        if (!player.isGliding()) return;

        Location to = event.getTo();
        Location from = event.getFrom();
        if (to == null || !CheckUtil.sameWorld(from, to)) return;

        ElytraState state = elytraState(player);
        long now = System.currentTimeMillis();

        double dy = to.getY() - from.getY();
        double horizontal = CheckUtil.horizontalDistance(from, to);
        float yawDelta = angleDifference(to.getYaw(), from.getYaw());

        if (state.lastSafe == null || CheckUtil.hasSolidSupport(from)) {
            state.lastSafe = from.clone();
        }

        boolean boosted = now - state.lastRocketAt < 1_900L;
        if (boosted) {
            state.upwardTicks = 0;
            state.levelTicks = 0;
        } else {
            if (dy > 0.28D) {
                state.upwardTicks += 2;
            } else if (dy > 0.035D) {
                // ExtraElytra do Wurst permite ganho contínuo de altura em passos pequenos.
                state.upwardTicks++;
            } else {
                state.upwardTicks = Math.max(0, state.upwardTicks - 1);
            }

            if (state.upwardTicks >= 8) {
                handleElytraViolation(player, state);
                state.upwardTicks = 0;
            }

            if (Math.abs(dy) < 0.018D && horizontal > 0.65D) {
                state.levelTicks++;
                if (state.levelTicks >= 30) {
                    handleElytraViolation(player, state);
                    state.levelTicks = 0;
                }
            } else {
                state.levelTicks = Math.max(0, state.levelTicks - 1);
            }
        }

        if (horizontal > 3.25D) {
            handleElytraViolation(player, state);
        }

        if (!boosted && horizontal > 1.25D && yawDelta > 115.0F) {
            state.impossibleTurnBuffer++;
            if (state.impossibleTurnBuffer >= 2) {
                handleElytraViolation(player, state);
                state.impossibleTurnBuffer = 0;
            }
        } else {
            state.impossibleTurnBuffer = Math.max(0, state.impossibleTurnBuffer - 1);
        }

        if (state.lastHorizontal > 1.4D
                && horizontal < 0.08D
                && !CheckUtil.hasSolidSupport(to)
                && !CheckUtil.isLiquidAround(to)) {
            handleElytraViolation(player, state);
        }

        if (!boosted && horizontal > state.lastHorizontal + 0.035D && horizontal > 0.45D) {
            state.accelerationBuffer++;
            if (state.accelerationBuffer >= 10) {
                handleElytraViolation(player, state);
                state.accelerationBuffer = 0;
            }
        } else {
            state.accelerationBuffer = Math.max(0, state.accelerationBuffer - 1);
        }

        state.lastHorizontal = horizontal;
    }

    private void handleElytraViolation(Player player, ElytraState state) {
        int stage = violations.flag(player, "FlyElytra");

        if (stage >= 2 && state.lastSafe != null && player.isOnline()) {
            player.teleport(state.lastSafe.clone());
        }

        if (stage >= 3) {
            player.setGliding(false);
            ItemStack chest = player.getInventory().getChestplate();
            if (chest != null && chest.getType() == Material.ELYTRA) {
                player.getInventory().setChestplate(null);
                Map<Integer, ItemStack> leftover = player.getInventory().addItem(chest);
                for (ItemStack item : leftover.values()) {
                    player.getWorld().dropItemNaturally(player.getLocation(), item);
                }
            }
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onVehicleMove(VehicleMoveEvent event) {
        Vehicle vehicle = event.getVehicle();
        if (vehicle.getPassengers().isEmpty()) return;

        Player player = null;
        for (org.bukkit.entity.Entity passenger : vehicle.getPassengers()) {
            if (passenger instanceof Player found) {
                player = found;
                break;
            }
        }
        if (player == null) return;

        Location from = event.getFrom();
        Location to = event.getTo();
        if (!CheckUtil.sameWorld(from, to)) return;

        VehicleState state = vehicleStates.computeIfAbsent(
                player.getUniqueId(),
                ignored -> new VehicleState()
        );

        double horizontal = CheckUtil.horizontalDistance(from, to);
        double dy = to.getY() - from.getY();
        String type = vehicle.getType().name();
        String below = to.clone().subtract(0.0D, 0.5D, 0.0D).getBlock().getType().name();
        boolean supported = to.clone().subtract(0.0D, 0.5D, 0.0D).getBlock().getType().isSolid();
        boolean liquid = below.contains("WATER") || below.contains("LAVA");

        double maxSpeed = vehicleSpeedLimit(type);
        if (horizontal > maxSpeed) {
            handleVehicleViolation(player, vehicle);
        }

        if (type.contains("BOAT")
                && !liquid
                && !below.contains("ICE")
                && horizontal > 0.90D) {
            handleVehicleViolation(player, vehicle);
        }

        if (!supported && !liquid && dy > 0.55D) {
            state.flyBuffer++;
            if (state.flyBuffer >= 3) {
                handleVehicleViolation(player, vehicle);
                state.flyBuffer = 0;
            }
        } else if (!supported && !liquid && Math.abs(dy) < 0.015D && horizontal > 0.15D) {
            state.hoverBuffer++;
            if (state.hoverBuffer >= 8) {
                handleVehicleViolation(player, vehicle);
                state.hoverBuffer = 0;
            }
        } else {
            state.flyBuffer = 0;
            state.hoverBuffer = 0;
        }

        if (type.contains("HORSE") && dy > 1.25D && horizontal > 0.65D) {
            handleVehicleViolation(player, vehicle);
        }
    }

    private void handleVehicleViolation(Player player, Vehicle vehicle) {
        int stage = violations.flag(player, "Vehicle");
        if (stage >= 3 && vehicle.isValid()) {
            vehicle.eject();
        }
    }

    private double vehicleSpeedLimit(String type) {
        if (type.contains("MINECART")) return 2.20D;
        if (type.contains("BOAT")) return 2.80D;
        if (type.contains("HORSE")) return 1.80D;
        if (type.contains("PIG") || type.contains("STRIDER")) return 1.50D;
        return 2.40D;
    }

    private ElytraState elytraState(Player player) {
        return elytraStates.computeIfAbsent(
                player.getUniqueId(),
                ignored -> new ElytraState()
        );
    }

    private float angleDifference(float a, float b) {
        float diff = (a - b) % 360.0F;
        if (diff > 180.0F) diff -= 360.0F;
        if (diff < -180.0F) diff += 360.0F;
        return Math.abs(diff);
    }

    public void clear(UUID playerId) {
        elytraStates.remove(playerId);
        vehicleStates.remove(playerId);
    }

    private static final class ElytraState {
        private long lastRocketAt;
        private int upwardTicks;
        private int levelTicks;
        private int impossibleTurnBuffer;
        private int accelerationBuffer;
        private double lastHorizontal;
        private Location lastSafe;
    }

    private static final class VehicleState {
        private int flyBuffer;
        private int hoverBuffer;
    }
}
```

```java
    public InteractionCheckListener(ViolationManager violations) {

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onInteract(PlayerInteractEvent event) {
        Player player = event.getPlayer();

        // Interact valida apenas interações reais com clique direito.
        // LEFT_CLICK_BLOCK é mineração e pertence aos checks de Block/FastBreak.
        switch (event.getAction()) {
            case RIGHT_CLICK_BLOCK -> {
                if (event.getClickedBlock() != null) {
                    checkBlockInteraction(player, event.getClickedBlock());
                }
            }
            default -> {
            }
        }

        ItemStack item = event.getItem();
        if (item == null) return;

        switch (event.getAction()) {
            case RIGHT_CLICK_AIR, RIGHT_CLICK_BLOCK -> {
                Material type = item.getType();
                if (isTimedUse(type)) {
                    useStates.put(
                            player.getUniqueId(),
                            new UseState(type, System.currentTimeMillis())
                    );
                }
            }
            default -> {
            }
        }
    }

    private void checkBlockInteraction(Player player, Block clicked) {
        Location eye = player.getEyeLocation();
        Location center = clicked.getLocation().add(0.5D, 0.5D, 0.5D);
        if (!CheckUtil.sameWorld(eye, center)) return;

        double distance = eye.distance(center);
        if (distance > 6.25D) {
            violations.flag(player, "Interact");
            return;
        }

        Vector toward = center.toVector().subtract(eye.toVector());
        if (toward.lengthSquared() < 0.0001D) return;

        double facing = eye.getDirection().normalize().dot(toward.normalize());
        if (distance > 2.0D && facing < 0.30D) {
            violations.flag(player, "Interact");
        }

        Block target = player.getTargetBlockExact(6);
        if (target != null
                && !target.equals(clicked)
                && target.getType().isSolid()
                && target.getLocation().distanceSquared(clicked.getLocation()) > 0.25D) {
            violations.flag(player, "Interact");
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onInteractEntity(PlayerInteractEntityEvent event) {
        Player player = event.getPlayer();
        Entity target = event.getRightClicked();

        Location eye = player.getEyeLocation();
        Location targetLocation = target.getLocation().clone()
                .add(0.0D, Math.max(0.2D, target.getHeight() * 0.5D), 0.0D);

        if (!CheckUtil.sameWorld(eye, targetLocation)) return;

        Vector toward = targetLocation.toVector().subtract(eye.toVector());
        double distance = toward.length();

        if (distance > 4.75D || !player.hasLineOfSight(target)) {
            violations.flag(player, "Interact");
            return;
        }

        if (distance > 1.5D && toward.lengthSquared() > 0.0001D) {
            double facing = eye.getDirection().normalize().dot(toward.normalize());
            if (facing < 0.20D) {
                violations.flag(player, "Interact");
            }
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onConsume(PlayerItemConsumeEvent event) {
        Player player = event.getPlayer();
        UseState state = useStates.remove(player.getUniqueId());
        if (state == null) return;

        long elapsed = System.currentTimeMillis() - state.startedAt;
        String type = state.material.name();

        long minimum = type.contains("POTION") ? 1_250L : 900L;
        if (elapsed < minimum) {
            violations.flag(player, "Delay");
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onBowShot(EntityShootBowEvent event) {
        if (!(event.getEntity() instanceof Player player)) return;
        if (event.getBow() == null) return;

        UseState state = useStates.remove(player.getUniqueId());
        if (state == null) return;

        String type = state.material.name();
        if (!type.contains("BOW")) return;

        long elapsed = System.currentTimeMillis() - state.startedAt;
        if (event.getForce() >= 0.75F && elapsed < 550L) {
            violations.flag(player, "Delay");
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onInventoryClick(InventoryClickEvent event) {
        if (!(event.getWhoClicked() instanceof Player player)) return;

        long now = System.currentTimeMillis();
        InventoryState state = inventoryStates.computeIfAbsent(
                player.getUniqueId(),
                ignored -> new InventoryState()
        );

        state.clicks.addLast(now);
        while (!state.clicks.isEmpty() && now - state.clicks.peekFirst() > 750L) {
            state.clicks.removeFirst();
        }

        if (state.clicks.size() >= 8 && inventoryPatternImpossible(state.clicks)) {
            violations.flag(player, "Delay");
            state.clicks.clear();
        }

        TotemState totem = totemStates.get(player.getUniqueId());
        if (totem != null
                && now - totem.dangerAt <= 500L
                && (isTotem(event.getCurrentItem()) || isTotem(event.getCursor()))) {
            totem.lastEmergencySwapAt = now;
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onSwapHands(PlayerSwapHandItemsEvent event) {
        Player player = event.getPlayer();
        TotemState state = totemStates.get(player.getUniqueId());
        if (state == null) return;

        long now = System.currentTimeMillis();
        if (now - state.dangerAt <= 500L
                && (isTotem(event.getMainHandItem()) || isTotem(event.getOffHandItem()))) {
            state.lastEmergencySwapAt = now;
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onDamage(EntityDamageEvent event) {
        if (!(event.getEntity() instanceof Player player)) return;

        double postHealth = player.getHealth() - event.getFinalDamage();
        if (postHealth > 5.0D) return;

        if (player.getInventory().getItemInOffHand().getType() != Material.TOTEM_OF_UNDYING) {
            TotemState state = totemStates.computeIfAbsent(
                    player.getUniqueId(),
                    ignored -> new TotemState()
            );
            state.dangerAt = System.currentTimeMillis();
            state.lastEmergencySwapAt = 0L;
        }
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onResurrect(EntityResurrectEvent event) {
        if (!(event.getEntity() instanceof Player player)) return;

        TotemState state = totemStates.get(player.getUniqueId());
        if (state == null || state.lastEmergencySwapAt <= 0L) return;

        long reaction = state.lastEmergencySwapAt - state.dangerAt;
        if (reaction >= 0L && reaction < 65L) {
            violations.flag(player, "Delay");
        }

        state.dangerAt = 0L;
        state.lastEmergencySwapAt = 0L;
    }

    private boolean inventoryPatternImpossible(Deque<Long> clicks) {
        Long[] times = clicks.toArray(Long[]::new);
        if (times.length < 8) return false;

        int instant = 0;
        double sum = 0.0D;
        int count = 0;

        for (int i = 1; i < times.length; i++) {
            long delta = times[i] - times[i - 1];
            if (delta <= 30L) instant++;
            if (delta > 0L) {
                sum += delta;
                count++;
            }
        }

        if (instant >= 5) return true;
        if (count == 0) return false;

        double mean = sum / count;
        double variance = 0.0D;
        for (int i = 1; i < times.length; i++) {
            long delta = times[i] - times[i - 1];
            if (delta <= 0L) continue;
            double diff = delta - mean;
            variance += diff * diff;
        }

        double deviation = Math.sqrt(variance / count);
        return mean < 55.0D && deviation < 4.0D;
    }

    private boolean isTimedUse(Material material) {
        String name = material.name();
        return material.isEdible()
                || name.contains("POTION")
                || name.contains("BOW")
                || name.contains("CROSSBOW")
                || name.contains("TRIDENT");
    }

    private boolean isTotem(ItemStack item) {
        return item != null && item.getType() == Material.TOTEM_OF_UNDYING;
    }

    public void clear(UUID playerId) {
        useStates.remove(playerId);
        inventoryStates.remove(playerId);
        totemStates.remove(playerId);
    }

    private record UseState(Material material, long startedAt) {
    }

    private static final class InventoryState {
        private final Deque<Long> clicks = new ArrayDeque<>();
    }

    private static final class TotemState {
        private long dangerAt;
        private long lastEmergencySwapAt;
    }
}
```

```java
package com.redeedge.anticheatplus.anticheat.listeners;

import com.redeedge.anticheatplus.AntiCheatPlusPlugin;
import com.redeedge.anticheatplus.anticheat.CheckUtil;
import com.redeedge.anticheatplus.anticheat.ViolationManager;
import org.bukkit.Location;
import org.bukkit.Material;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.EventPriority;
import org.bukkit.event.Listener;
import org.bukkit.event.entity.EntityDamageEvent;
import org.bukkit.event.player.PlayerInteractEvent;
import org.bukkit.event.player.PlayerItemConsumeEvent;
import org.bukkit.event.player.PlayerMoveEvent;
import org.bukkit.event.player.PlayerTeleportEvent;
import org.bukkit.event.player.PlayerVelocityEvent;
import org.bukkit.inventory.ItemStack;
import org.bukkit.potion.PotionEffectType;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public final class MovementCheckListener implements Listener {

    private final AntiCheatPlusPlugin plugin;
    private final ViolationManager violations;
    private final Map<UUID, MovementState> states = new ConcurrentHashMap<>();
    private final Map<UUID, Long> usingSlowItemUntil = new ConcurrentHashMap<>();

    public MovementCheckListener(AntiCheatPlusPlugin plugin, ViolationManager violations) {
        this.plugin = plugin;
        this.violations = violations;
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onMove(PlayerMoveEvent event) {
        Location from = event.getFrom();
        Location to = event.getTo();
        if (to == null || !CheckUtil.sameWorld(from, to)) return;

        Player player = event.getPlayer();
        MovementState state = state(player);
        long now = System.currentTimeMillis();

        updateLegitimateFlightState(player, to, state, now);

        if (now < state.freezeUntil && state.lastSafe != null) {
            event.setTo(state.lastSafe.clone());
            return;
        }

        boolean positionChanged = from.getX() != to.getX()
                || from.getY() != to.getY()
                || from.getZ() != to.getZ();

        if (!positionChanged) {
            checkRotationPackets(player, to, state, now);
            return;
        }

        double horizontal = CheckUtil.horizontalDistance(from, to);
        double dy = to.getY() - from.getY();
        double totalSquared = from.distanceSquared(to);

        checkBadPackets(player, from, to, totalSquared, state, now);
        checkPhase(player, from, to, horizontal, totalSquared, state);

        if (now < state.graceUntil) {
            rememberEnvironment(to, state);
            updateFallState(player, to, state);
            updateSafeLocation(to, state);
            return;
        }

        boolean liquid = CheckUtil.isLiquidAround(to);
        boolean special = CheckUtil.isClimbableOrSpecial(to);

        checkTransitionBoosts(player, to, horizontal, dy, state);
        checkNoSlow(player, horizontal, now);

        boolean legitimateFlight = isLegitimateFlightActive(player);
        boolean flightLandingGrace = isFlightLandingGrace(player, to, state, now);

        if (!legitimateFlight && !flightLandingGrace) {
            checkNoFall(player, from, to, dy, state);
        } else {
            resetFlightFallTracking(state);
        }

        if (!CheckUtil.exemptMovement(player)) {
            if (!liquid && !special) {
                checkSpeed(player, to, horizontal, dy, state);
                if (!legitimateFlight && !flightLandingGrace) {
                    checkFly(player, to, horizontal, dy, state);
                } else {
                    resetFlyTracking(state);
                }
            }
            checkJesus(player, to, horizontal, dy, state);
        } else {
            resetTransient(state);
        }

        updateFallState(player, to, state);
        rememberEnvironment(to, state);
        updateSafeLocation(to, state);
    }

    private void updateLegitimateFlightState(
            Player player,
            Location to,
            MovementState state,
            long now
    ) {
        boolean active = isLegitimateFlightActive(player);

        if (state.wasLegitimateFlying && !active
                && !CheckUtil.hasSolidSupport(to)
                && !CheckUtil.isLiquidAround(to)) {
            state.flightLandingGrace = true;
            state.flightLandingGraceUntil = now + 30_000L;
            resetFlightFallTracking(state);
            resetFlyTracking(state);
        }

        state.wasLegitimateFlying = active;

        if (active) {
            state.flightLandingGrace = false;
            state.flightLandingGraceUntil = 0L;
            resetFlightFallTracking(state);
            resetFlyTracking(state);
        }
    }

    private boolean isLegitimateFlightActive(Player player) {
        return player.getAllowFlight() && player.isFlying();
    }

    private boolean isFlightLandingGrace(
            Player player,
            Location to,
            MovementState state,
            long now
    ) {
        if (!state.flightLandingGrace) return false;

        boolean landed = CheckUtil.hasSolidSupport(to)
                || CheckUtil.isLiquidAround(to)
                || player.isOnGround();

        if (landed || now > state.flightLandingGraceUntil) {
            state.flightLandingGrace = false;
            state.flightLandingGraceUntil = 0L;
            resetFlightFallTracking(state);
            resetFlyTracking(state);
            return false;
        }

        // O EssentialsPlus usa Slow Falling ao desligar /fly no ar.
        // Mesmo se o efeito acabar perto do chão, mantemos a tolerância até pousar.
        return player.hasPotionEffect(PotionEffectType.SLOW_FALLING)
                || now <= state.flightLandingGraceUntil;
    }

    private void resetFlyTracking(MovementState state) {
        state.hoverTicks = 0;
        state.upwardAirTicks = 0;
        state.flightAntiKickBuffer = 0;
        state.aobaAntiKickBuffer = 0;
        state.meteorFlightPulseBuffer = 0;
        state.lastMeteorFlightPulseAt = 0L;
        state.flyAirTicks = 0;
        state.previousDy = 0.0D;
    }

    private void resetFlightFallTracking(MovementState state) {
        state.midairFallResetBuffer = 0;
        state.maxFallDistance = 0.0F;
        state.serverMeasuredFall = 0.0D;
        state.lastFallDistance = 0.0F;
    }

    private void checkSpeed(Player player, Location to, double horizontal, double dy, MovementState state) {
        boolean supported = CheckUtil.hasSolidSupport(to);
        double walkMultiplier = Math.max(0.5D, player.getWalkSpeed() / 0.2D);
        double threshold = (supported ? 0.72D : 0.84D) * walkMultiplier;

        if (player.isSprinting()) threshold += 0.10D;
        if (Math.abs(dy) > 0.35D) threshold += 0.10D;

        // Wurst SpeedHack aplica mini-jumps de ~0.1 e limita o horizontal a ~0.66.
        // Isso fica abaixo de limites ingênuos de speed, então analisamos a assinatura repetida.
        boolean miniHopSignature = horizontal >= 0.52D
                && horizontal <= 0.72D
                && dy > 0.055D
                && dy < 0.16D;

        if (miniHopSignature) {
            state.miniHopSpeedBuffer++;
            if (state.miniHopSpeedBuffer >= 3) {
                violations.flag(player, "Speed");
                state.miniHopSpeedBuffer = 0;
            }
        } else {
            state.miniHopSpeedBuffer = Math.max(0, state.miniHopSpeedBuffer - 1);
        }

        // Meteor Speed/Strafe sustenta ~0.43-0.44 blocos/tick durante o ciclo
        // de strafe NCP. Sprint-jump vanilla normalmente fica abaixo disso.
        String belowName = to.clone().subtract(0.0D, 0.15D, 0.0D).getBlock().getType().name();
        boolean slipperySurface = belowName.contains("ICE")
                || belowName.contains("SLIME")
                || belowName.contains("HONEY");
        boolean meteorStrafeSignature = horizontal >= 0.405D
                && horizontal <= 0.455D
                && !player.hasPotionEffect(PotionEffectType.SPEED)
                && !slipperySurface;

        if (meteorStrafeSignature) {
            state.meteorStrafeBuffer++;
            if (state.meteorStrafeBuffer >= 6) {
                violations.flag(player, "Speed");
                state.meteorStrafeBuffer = 0;
            }
        } else {
            state.meteorStrafeBuffer = Math.max(0, state.meteorStrafeBuffer - 1);
        }

        if (horizontal > threshold) {
            state.speedBuffer++;
            if (state.speedBuffer >= 2) {
                violations.flag(player, "Speed");
                state.speedBuffer = 0;
                state.miniHopSpeedBuffer = 0;
                state.flightAntiKickBuffer = 0;
                state.aobaAntiKickBuffer = 0;
            }
        } else {
            state.speedBuffer = Math.max(0, state.speedBuffer - 1);
        }
    }

    private void checkFly(Player player, Location to, double horizontal, double dy, MovementState state) {
        if (CheckUtil.hasSolidSupport(to)) {
            state.hoverTicks = 0;
            state.upwardAirTicks = 0;
            state.flightAntiKickBuffer = 0;
            state.aobaAntiKickBuffer = 0;
            state.meteorFlightPulseBuffer = 0;
            state.lastMeteorFlightPulseAt = 0L;
            state.flyAirTicks = 0;
            state.previousDy = 0.0D;
            return;
        }

        state.flyAirTicks++;

        if (Math.abs(dy) <= 0.025D && horizontal > 0.025D) {
            state.hoverTicks++;
            if (state.hoverTicks >= 8) {
                int stage = violations.flag(player, "Fly");
                state.hoverTicks = 0;
                if (stage >= 3) setback(player, state);
            }
        } else {
            state.hoverTicks = Math.max(0, state.hoverTicks - 2);
        }

        // Wurst Flight Anti-Kick alterna pequenos movimentos verticais (~0.035).
        boolean antiKickStep = Math.abs(dy) >= 0.015D && Math.abs(dy) <= 0.065D;
        if (antiKickStep && horizontal >= 0.0D) {
            if (state.previousDy != 0.0D
                    && Math.signum(dy) != Math.signum(state.previousDy)
                    && Math.abs(Math.abs(dy) - Math.abs(state.previousDy)) < 0.025D) {
                state.flightAntiKickBuffer++;
            } else {
                state.flightAntiKickBuffer = Math.max(0, state.flightAntiKickBuffer - 1);
            }

            if (state.flightAntiKickBuffer >= 4) {
                int stage = violations.flag(player, "Fly");
                state.flightAntiKickBuffer = 0;
                if (stage >= 3) setback(player, state);
            }
        } else {
            state.flightAntiKickBuffer = Math.max(0, state.flightAntiKickBuffer - 1);
        }

        // Aoba Fly (Synthetic + AntiKick) soma -0.08 Y a cada tick.
        // Uma descida praticamente constante nesse valor, repetida no ar,
        // não corresponde à curva de gravidade vanilla.
        if (dy <= -0.060D && dy >= -0.100D
                && Math.abs(dy - state.previousDy) <= 0.012D
                && !player.hasPotionEffect(PotionEffectType.SLOW_FALLING)) {
            state.aobaAntiKickBuffer++;
            if (state.aobaAntiKickBuffer >= 4) {
                int stage = violations.flag(player, "Fly");
                state.aobaAntiKickBuffer = 0;
                if (stage >= 3) setback(player, state);
            }
        } else {
            state.aobaAntiKickBuffer = Math.max(0, state.aobaAntiKickBuffer - 1);
        }

        // Meteor Flight em modo Packet injeta uma queda de aproximadamente
        // -0.03130 Y em intervalos regulares para zerar floating ticks.
        // O pulso do Meteor só é relevante durante voo contínuo. Em um pulo
        // vanilla o jogador pode cruzar naturalmente essa mesma faixa de dy uma
        // vez perto do ápice. Exigimos tempo contínuo no ar e movimento vertical
        // previamente quase neutro para não somar evidências entre sprint-jumps.
        boolean meteorPacketPulse = state.flyAirTicks >= 10
                && Math.abs(state.previousDy) <= 0.015D
                && dy <= -0.0285D
                && dy >= -0.0345D
                && !player.hasPotionEffect(PotionEffectType.SLOW_FALLING);
        if (meteorPacketPulse) {
            long now = System.currentTimeMillis();
            if (state.lastMeteorFlightPulseAt > 0L) {
                long gap = now - state.lastMeteorFlightPulseAt;
                if (gap >= 650L && gap <= 1_650L) {
                    state.meteorFlightPulseBuffer++;
                    if (state.meteorFlightPulseBuffer >= 3) {
                        int stage = violations.flag(player, "Fly");
                        state.meteorFlightPulseBuffer = 0;
                        if (stage >= 3) setback(player, state);
                    }
                } else if (gap > 2_000L) {
                    state.meteorFlightPulseBuffer = 0;
                }
            }
            state.lastMeteorFlightPulseAt = now;
        }

        if (dy > 0.42D && state.previousDy > 0.0D) {
            state.upwardAirTicks++;
            if (state.upwardAirTicks >= 3) {
                int stage = violations.flag(player, "Fly");
                state.upwardAirTicks = 0;
                if (stage >= 3) setback(player, state);
            }
        } else if (dy <= 0.0D) {
            state.upwardAirTicks = 0;
        }

        state.previousDy = dy;
    }

    private void checkJesus(Player player, Location to, double horizontal, double dy, MovementState state) {
        Material feet = to.getBlock().getType();
        Material below = to.clone().subtract(0.0D, 0.15D, 0.0D).getBlock().getType();

        boolean feetLiquid = feet.name().contains("WATER") || feet.name().contains("LAVA");
        boolean belowLiquid = below.name().contains("WATER") || below.name().contains("LAVA");
        boolean liquid = feetLiquid || belowLiquid;

        if (liquid
                && !player.isSwimming()
                && !player.isRiptiding()
                && Math.abs(dy) <= 0.065D
                && horizontal > 0.06D) {
            state.waterWalkTicks++;
            if (state.waterWalkTicks >= 6) {
                violations.flag(player, "Jesus");
                state.waterWalkTicks = 0;
            }
        } else {
            state.waterWalkTicks = 0;
        }

        // Meteor Jesus/NCP altera Y em cerca de -0.02 enquanto o jogador
        // permanece sobre a superfície do líquido. Em vanilla não é possível
        // sustentar esse padrão sem entrar efetivamente no fluido.
        boolean meteorSurfacePulse = belowLiquid
                && !feetLiquid
                && !player.isInsideVehicle()
                && !player.isSwimming()
                && dy <= -0.0175D
                && dy >= -0.0245D;

        if (meteorSurfacePulse) {
            state.meteorJesusBuffer++;
            if (state.meteorJesusBuffer >= 5) {
                violations.flag(player, "Jesus");
                state.meteorJesusBuffer = 0;
            }
        } else {
            state.meteorJesusBuffer = Math.max(0, state.meteorJesusBuffer - 1);
        }
    }

    private void checkNoSlow(Player player, double horizontal, long now) {
        Long until = usingSlowItemUntil.get(player.getUniqueId());

        if (until != null && now > until) {
            usingSlowItemUntil.remove(player.getUniqueId());
            until = null;
        }

        // Um simples RIGHT_CLICK não significa que o jogador permaneceu usando
        // o item. O estado temporário acima serve apenas como confirmação de que
        // um slow-use começou; para acusar NoSlow exigimos que o cliente esteja
        // realmente com a mão levantada/uso contínuo naquele momento.
        boolean activelyUsingSlowItem = until != null
                && player.isHandRaised();

        if (activelyUsingSlowItem
                && player.isSprinting()
                && horizontal > 0.30D) {
            violations.flag(player, "NoSlow");
        }
    }

    private void checkPhase(
            Player player,
            Location from,
            Location to,
            double horizontal,
            double totalSquared,
            MovementState state
    ) {
        boolean destinationInside = CheckUtil.insideSolid(to);
        boolean crossedSolid = totalSquared > 1.0D && segmentCrossesSolid(from, to);

        if ((horizontal < 0.06D || !destinationInside) && !crossedSolid) {
            state.phaseTicks = Math.max(0, state.phaseTicks - 1);
            return;
        }

        boolean fromSafe = !CheckUtil.insideSolid(from);
        state.phaseTicks += crossedSolid ? 3 : (fromSafe ? 2 : 1);

        if (state.phaseTicks >= 3) {
            int stage = violations.flag(player, "Phase");
            state.phaseTicks = 0;

            if (state.lastSafe != null) {
                state.freezeUntil = System.currentTimeMillis() + 750L;
                player.teleport(state.lastSafe.clone());
            }

            if (stage >= 3) {
                state.freezeUntil = System.currentTimeMillis() + 1_500L;
            }
        }
    }

    private void checkBadPackets(
            Player player,
            Location from,
            Location to,
            double totalSquared,
            MovementState state,
            long now
    ) {
        boolean invalid = !Double.isFinite(to.getX())
                || !Double.isFinite(to.getY())
                || !Double.isFinite(to.getZ())
                || !Float.isFinite(to.getYaw())
                || !Float.isFinite(to.getPitch())
                || Math.abs(to.getPitch()) > 90.01F
                || !Double.isFinite(totalSquared);

        if (invalid) {
            violations.flag(player, "BadPackets");
            return;
        }

        if (totalSquared > 64.0D) {
            violations.flag(player, "BadPackets");
        }

        state.movePackets.addLast(now);
        while (!state.movePackets.isEmpty() && now - state.movePackets.peekFirst() > 1_000L) {
            state.movePackets.removeFirst();
        }

        if (state.movePackets.size() > 75) {
            violations.flag(player, "BadPackets");
            state.movePackets.clear();
        }

        double dy = to.getY() - from.getY();
        if (Math.abs(dy) > 8.0D && now >= state.graceUntil) {
            violations.flag(player, "BadPackets");
        }

        checkRotationPackets(player, to, state, now);
    }

    private void checkRotationPackets(Player player, Location to, MovementState state, long now) {
        if (!Float.isFinite(to.getYaw())
                || !Float.isFinite(to.getPitch())
                || Math.abs(to.getPitch()) > 90.01F) {
            violations.flag(player, "BadPackets");
        }

        state.lastRotationAt = now;
    }

    private void checkTransitionBoosts(
            Player player,
            Location to,
            double horizontal,
            double dy,
            MovementState state
    ) {
        boolean liquidNow = CheckUtil.isLiquidAround(to);
        boolean climbNow = isClimbable(to);

        if (state.wasLiquid && !liquidNow && horizontal > 1.05D) {
            violations.flag(player, "Speed");
        }

        if (state.wasClimbable && !climbNow && (horizontal > 0.95D || dy > 0.75D)) {
            violations.flag(player, "Speed");
        }
    }

    private void checkNoFall(Player player, Location from, Location to, double dy, MovementState state) {
        boolean wasSupported = CheckUtil.hasSolidSupport(from);
        boolean supported = CheckUtil.hasSolidSupport(to);

        if (!supported && !CheckUtil.isLiquidAround(to) && !CheckUtil.isClimbableOrSpecial(to)) {
            if (dy < 0.0D) {
                state.serverMeasuredFall += -dy;
            } else if (dy > 0.20D) {
                state.serverMeasuredFall = 0.0D;
            }
        }

        float currentFall = player.getFallDistance();
        if (!supported
                && state.lastFallDistance > 3.5F
                && currentFall + 1.0F < state.lastFallDistance
                && !CheckUtil.isLiquidAround(to)
                && !CheckUtil.isClimbableOrSpecial(to)) {
            state.midairFallResetBuffer++;
            if (state.midairFallResetBuffer >= 2) {
                violations.flag(player, "NoFall");
                state.midairFallResetBuffer = 0;
            }
        } else {
            state.midairFallResetBuffer = Math.max(0, state.midairFallResetBuffer - 1);
        }

        state.lastFallDistance = currentFall;

        if (!wasSupported && supported
                && Math.max(state.maxFallDistance, state.serverMeasuredFall) >= 4.5D) {
            float capturedFall = (float)Math.max(state.maxFallDistance, state.serverMeasuredFall);
            state.serverMeasuredFall = 0.0D;
            long landingAt = System.currentTimeMillis();

            plugin.getServer().getScheduler().runTaskLater(plugin, () -> {
                MovementState current = states.get(player.getUniqueId());
                if (current == null || !player.isOnline()) return;
                if (capturedFall < 4.5F) return;
                if (current.lastFallDamageAt < landingAt - 250L
                        || current.lastFallDamageAt > landingAt + 700L) {
                    violations.flag(player, "NoFall");
                }
            }, 2L);
        }
    }

    private void updateFallState(Player player, Location to, MovementState state) {
        if (CheckUtil.hasSolidSupport(to) || CheckUtil.isLiquidAround(to)) {
            state.maxFallDistance = 0.0F;
            if (CheckUtil.isLiquidAround(to)) state.serverMeasuredFall = 0.0D;
        } else {
            state.maxFallDistance = Math.max(state.maxFallDistance, player.getFallDistance());
        }
    }

    private boolean segmentCrossesSolid(Location from, Location to) {
        if (!CheckUtil.sameWorld(from, to)) return false;
        double distance = from.distance(to);
        if (distance < 0.5D) return false;

        int samples = Math.min(48, Math.max(3, (int)Math.ceil(distance / 0.25D)));
        for (int i = 1; i < samples; i++) {
            double t = i / (double)samples;
            Location probe = from.clone().add(
                    (to.getX() - from.getX()) * t,
                    (to.getY() - from.getY()) * t,
                    (to.getZ() - from.getZ()) * t
            );
            if (CheckUtil.insideSolid(probe)) return true;
        }
        return false;
    }

    private void rememberEnvironment(Location to, MovementState state) {
        state.wasLiquid = CheckUtil.isLiquidAround(to);
        state.wasClimbable = isClimbable(to);
    }

    private boolean isClimbable(Location location) {
        String feet = location.getBlock().getType().name();
        return feet.contains("LADDER")
                || feet.contains("VINE")
                || feet.contains("SCAFFOLD");
    }

    private void updateSafeLocation(Location to, MovementState state) {
        if (!CheckUtil.insideSolid(to)
                && CheckUtil.hasSolidSupport(to)
                && !CheckUtil.isLiquidAround(to)) {
            state.lastSafe = to.clone();
        }
    }

    private void setback(Player player, MovementState state) {
        if (state.lastSafe == null || !player.isOnline()) return;
        state.graceUntil = System.currentTimeMillis() + 1_000L;
        player.teleport(state.lastSafe.clone());
    }

    private void resetTransient(MovementState state) {
        state.hoverTicks = 0;
        state.waterWalkTicks = 0;

    }
```

```java
package com.redeedge.anticheatplus.anticheat.listeners;

import com.redeedge.anticheatplus.anticheat.ViolationManager;
import org.bukkit.GameMode;
import org.bukkit.Location;
import org.bukkit.Material;
import org.bukkit.block.Block;
import org.bukkit.block.BlockFace;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.EventPriority;
import org.bukkit.event.Listener;
import org.bukkit.event.block.BlockBreakEvent;
import org.bukkit.event.block.BlockPlaceEvent;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.HashMap;
import java.util.Iterator;
import java.util.Locale;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Detecção comportamental de Xray.
 *
 * O servidor não consegue observar diretamente um X-Ray puramente visual.
 * Por isso o check correlaciona descobertas de minérios naturais, exposição,
 * frequência de veins e eficiência do caminho até minérios ocultos.
 */
public final class XrayCheckListener implements Listener {

    private static final long SHORT_WINDOW_MS = 120_000L;
    private static final long LONG_WINDOW_MS = 300_000L;
    private static final long VEIN_CONTINUATION_MS = 20_000L;
    private static final long PLACED_ORE_TTL_MS = 30L * 60L * 1000L;
    private static final double VEIN_RADIUS_SQUARED = 25.0D;

    private final ViolationManager violations;
    private final Map<UUID, MiningState> states = new ConcurrentHashMap<>();

    public XrayCheckListener(ViolationManager violations) {
        this.violations = violations;
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onPlace(BlockPlaceEvent event) {
        if (!isTrackedOre(event.getBlockPlaced().getType())) return;

        MiningState state = state(event.getPlayer());
        long now = System.currentTimeMillis();
        cleanupPlaced(state, now);
        state.placedOres.put(BlockKey.of(event.getBlockPlaced()), now);
    }

    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onBreak(BlockBreakEvent event) {
        Player player = event.getPlayer();
        if (player.getGameMode() == GameMode.CREATIVE
                || player.getGameMode() == GameMode.SPECTATOR) {
            return;
        }

        MiningState state = state(player);
        long now = System.currentTimeMillis();
        cleanupPlaced(state, now);

        Block block = event.getBlock();
        OreProfile ore = OreProfile.of(block.getType());

        state.allBreaks.addLast(now);
        state.blocksSinceRareVein++;
        trim(state, now);

        if (ore == null) return;

        BlockKey key = BlockKey.of(block);
        if (state.placedOres.remove(key) != null) {
            return;
        }

        Location location = block.getLocation().add(0.5D, 0.5D, 0.5D);
        boolean exposed = isExposed(block);
        VeinState vein = findVein(state, ore.family, location, now);
        boolean newVein = vein == null;

        if (newVein) {
            vein = new VeinState(
                    ore.family,
                    location.clone(),
                    now,
                    !exposed
            );
            state.veins.addLast(vein);
        } else {
            vein.lastLocation = location.clone();
            vein.lastAt = now;
        }

        boolean hiddenVein = vein.initiallyHidden;

        if (newVein && ore.rare) {
            registerRareVeinTargeting(state, location, hiddenVein, now);
        }

        state.samples.addLast(new OreSample(
                now,
                ore.family,
                ore.weight,
                ore.rare,
                newVein,
                hiddenVein
        ));

        trim(state, now);
        evaluate(player, state);
    }

    private void registerRareVeinTargeting(
            MiningState state,
            Location location,
            boolean hidden,
            long now
    ) {
        if (!hidden) {
            state.blocksSinceRareVein = 0;
            state.lastRareVeinLocation = location.clone();
            state.lastRareVeinAt = now;
            return;
        }

        if (state.lastRareVeinLocation != null
                && state.lastRareVeinAt > 0L
                && now - state.lastRareVeinAt <= LONG_WINDOW_MS
                && state.lastRareVeinLocation.getWorld() != null
                && location.getWorld() != null
                && state.lastRareVeinLocation.getWorld().equals(location.getWorld())) {

            double distance = state.lastRareVeinLocation.distance(location);
            int ordinaryBreaks = Math.max(0, state.blocksSinceRareVein - 1);

            // Um trajeto muito eficiente entre duas veins raras totalmente
            // ocultas é um sinal forte do padrão produzido por X-Ray visual.
            double minimumNaturalDig = Math.max(3.0D, distance * 0.55D);
            if (distance >= 7.0D && ordinaryBreaks < minimumNaturalDig) {
                state.hiddenTargetingScore += 2;
            } else if (distance >= 5.0D && ordinaryBreaks < distance * 0.85D) {
                state.hiddenTargetingScore++;
            } else {
                state.hiddenTargetingScore = Math.max(0, state.hiddenTargetingScore - 1);
            }
        }

        state.blocksSinceRareVein = 0;
        state.lastRareVeinLocation = location.clone();
        state.lastRareVeinAt = now;
    }

    private void evaluate(Player player, MiningState state) {
        WindowStats shortStats = stats(state, SHORT_WINDOW_MS);
        WindowStats longStats = stats(state, LONG_WINDOW_MS);

        boolean diamondBurst = shortStats.diamondBlocks >= 10
                && shortStats.diamondVeins >= 2;

        boolean rareBurst = shortStats.rareBlocks >= 12
                && shortStats.rareVeins >= 3;

        boolean oreBurst = shortStats.weightedScore >= 22.0D
                && shortStats.totalVeins >= 5
                && shortStats.totalBlocks >= 18;

        boolean sustainedTargeting = longStats.weightedScore >= 40.0D
                && longStats.rareVeins >= 5
                && longStats.totalVeins >= 8;

        boolean extremeDiamondRun = longStats.diamondBlocks >= 20
                && longStats.diamondVeins >= 4;

        // Assinatura especialmente relevante ao X-Ray padrão do Wurst:
        // várias veins raras que estavam completamente fechadas por blocos.
        boolean hiddenRareBurst = shortStats.hiddenRareVeins >= 3
                && shortStats.hiddenRareBlocks >= 8;

        boolean hiddenDiamondRun = longStats.hiddenDiamondVeins >= 3
                && longStats.hiddenDiamondBlocks >= 12;

        // "Only show exposed" do Wurst remove esse sinal. Nesse modo,
        // continuamos dependendo das métricas gerais de frequência/veins.
        boolean directHiddenTargeting = state.hiddenTargetingScore >= 5
                && longStats.hiddenRareVeins >= 3;

        // Evita acusar alguém que simplesmente esteja quebrando milhares de
        // blocos rapidamente: a proporção de minérios precisa continuar alta.
        double rareVeinPerBreak = state.allBreaks.isEmpty()
                ? 0.0D
                : longStats.rareVeins / (double) state.allBreaks.size();

        boolean improbableYield = longStats.rareVeins >= 5
                && state.allBreaks.size() >= 20
                && rareVeinPerBreak >= 0.10D;

        if (diamondBurst
                || rareBurst
                || oreBurst
                || sustainedTargeting
                || extremeDiamondRun
                || hiddenRareBurst
                || hiddenDiamondRun
                || directHiddenTargeting
                || improbableYield) {
            violations.flag(player, "Xray");
        }
    }

    private WindowStats stats(MiningState state, long windowMs) {
        long cutoff = System.currentTimeMillis() - windowMs;
        WindowStats result = new WindowStats();

        for (OreSample sample : state.samples) {
            if (sample.time < cutoff) continue;

            result.totalBlocks++;

            double exposureMultiplier = sample.hidden ? 1.35D : 1.0D;
            result.weightedScore += sample.weight * exposureMultiplier;

            if (sample.newVein) result.totalVeins++;

            if (sample.family == OreFamily.DIAMOND) {
                result.diamondBlocks++;
                if (sample.newVein) result.diamondVeins++;

                if (sample.hidden) {
                    result.hiddenDiamondBlocks++;
                    if (sample.newVein) result.hiddenDiamondVeins++;
                }
            }

            if (sample.rare) {
                result.rareBlocks++;
                if (sample.newVein) result.rareVeins++;

                if (sample.hidden) {
                    result.hiddenRareBlocks++;
                    if (sample.newVein) result.hiddenRareVeins++;
                }
            }
        }

        return result;
    }

    private boolean isExposed(Block block) {
        for (BlockFace face : new BlockFace[]{
                BlockFace.DOWN,
                BlockFace.UP,
                BlockFace.NORTH,
                BlockFace.SOUTH,
                BlockFace.EAST,
                BlockFace.WEST
        }) {
            Material neighbor = block.getRelative(face).getType();

            if (neighbor.isAir()
                    || !neighbor.isOccluding()
                    || neighbor.name().contains("WATER")
                    || neighbor.name().contains("LAVA")) {
                return true;
            }
        }
        return false;
    }

    private VeinState findVein(
            MiningState state,
            OreFamily family,
            Location location,
            long now
    ) {
        Iterator<VeinState> iterator = state.veins.descendingIterator();
        while (iterator.hasNext()) {
            VeinState vein = iterator.next();
            if (now - vein.lastAt > VEIN_CONTINUATION_MS) continue;
            if (vein.family != family) continue;
            if (vein.lastLocation.getWorld() == null
                    || location.getWorld() == null
                    || !vein.lastLocation.getWorld().equals(location.getWorld())) {
                continue;
            }

            if (vein.lastLocation.distanceSquared(location) <= VEIN_RADIUS_SQUARED) {
                return vein;
            }
        }
        return null;
    }

    private void trim(MiningState state, long now) {
        while (!state.samples.isEmpty()
                && now - state.samples.peekFirst().time > LONG_WINDOW_MS) {
            state.samples.removeFirst();
        }

        while (!state.veins.isEmpty()
                && now - state.veins.peekFirst().lastAt > LONG_WINDOW_MS) {
            state.veins.removeFirst();
        }

        while (!state.allBreaks.isEmpty()
                && now - state.allBreaks.peekFirst() > LONG_WINDOW_MS) {
            state.allBreaks.removeFirst();
        }
    }

    private void cleanupPlaced(MiningState state, long now) {
```

```java
        this.plugin = plugin;
        this.punishments = punishments;
    }

    @Override
    public boolean onCommand(CommandSender sender, Command command, String label, String[] args) {
        if (!sender.hasPermission("anticheatplus.command.ban")) {
            sender.sendMessage(plugin.message("messages.no-permission", "&cVocê não tem permissão."));
            return true;
        }

        if (args.length < 1) {
            sender.sendMessage(plugin.color("&cUse: /ban <jogador> [motivo]"));
            return true;
        }

        ResolvedTarget target = TargetResolver.resolve(args[0]);
        if (target == null) {
            target = punishments.resolveKnownTarget(args[0]);
        }
        if (target == null) {
            sender.sendMessage(plugin.message("messages.player-not-found", "&cEsse jogador nunca entrou no servidor."));
            return true;
        }

        if (punishments.isSelf(sender, target)) {
            sender.sendMessage(plugin.color("&cVocê não pode banir a si mesmo."));
            return true;
        }

        if (punishments.isExempt(target, "ban")
                && !sender.hasPermission("anticheatplus.bypass.override")) {
            sender.sendMessage(plugin.color("&cEsse jogador está protegido contra banimentos."));
            return true;
        }

        String reason = CommandUtil.joinReason(args, 1);
        int maxReasonLength = punishments.getMaxReasonLength();
        if (reason.length() > maxReasonLength) {
            sender.sendMessage(plugin.color("&c[Erro] O motivo pode ter no máximo &f" + maxReasonLength + " &ccaracteres."));
            return true;
        }

        PunishmentRecord activeBan = punishments.getActiveBan(target.uuid());
        if (activeBan != null) {
            sender.sendMessage(plugin.color(
                    "&c[Ban] " + target.name() + " já está banido. "
                            + "&7ID: &f#" + activeBan.id()
                            + " &8• &7Staff: &f" + activeBan.actorName()
                            + " &8• &7Motivo: &f" + activeBan.reason()
            ));
            return true;
        }

        PunishmentRecord record = punishments.ban(target, sender, reason);
        if (record == null) {
            sender.sendMessage(plugin.color("&cNão foi possível aplicar o banimento."));
            return true;
        }

        sender.sendMessage(plugin.color(
                "&a[Ban] &f" + target.name() + " &afoi banido permanentemente. "
                        + "&8(#" + record.id() + ")"
        ));
        return true;
    }

    @Override
    public List<String> onTabComplete(CommandSender sender, Command command, String alias, String[] args) {
        if (args.length != 1) return List.of();
        return punishments.getKnownPlayerNames(args[0]).stream()
                .filter(name -> !(sender instanceof org.bukkit.entity.Player player)
                        || !player.getName().equalsIgnoreCase(name))
                .toList();
    }
}
```

```java
        Player online = Bukkit.getPlayerExact(input);

            targetName = online.getName();

            if (target != null) {
                ip = punishments.getKnownIp(target.uuid());
                targetName = target.name();
            }
            if (ip == null && looksLikeIp(input)) {
                ip = input;
            }
        }

        if (ip == null || ip.isBlank()) {
            sender.sendMessage(plugin.color("&cNão foi possível descobrir o IP desse jogador."));
            return true;
        }

        if (punishments.isIpBanned(ip)) {
            sender.sendMessage(plugin.color("&cEsse IP já está banido."));
            return true;
        }

        IpBanRecord record = punishments.banIp(ip, targetName, sender, CommandUtil.joinReason(args, 1));
        if (record == null) {
            sender.sendMessage(plugin.color("&cNão foi possível banir esse IP."));
            return true;
        }

        sender.sendMessage(plugin.color("&aIP banido com sucesso. &8(#" + record.id() + ")"));
        return true;
    }

    private boolean looksLikeIp(String value) {
        return value != null && (value.matches("[0-9.]+") || value.matches("[0-9a-fA-F:]+"));
    }
```

```java
package com.redeedge.anticheatplus.commands;

import org.bukkit.Bukkit;
import org.bukkit.command.CommandSender;
import org.bukkit.entity.Player;

import java.util.Arrays;
import java.util.List;
import java.util.Locale;

final class CommandUtil {

    private CommandUtil() {
    }

    static String joinReason(String[] args, int start) {
        if (args.length <= start) {
            return "";
        }
        return String.join(" ", Arrays.copyOfRange(args, start, args.length)).trim();
    }

    static List<String> completeOnlinePlayers(String prefix, CommandSender sender) {
        String normalized = prefix == null ? "" : prefix.toLowerCase(Locale.ROOT);
        return Bukkit.getOnlinePlayers().stream()
                .filter(player -> !player.equals(sender))
                .map(Player::getName)
                .filter(name -> name.toLowerCase(Locale.ROOT).startsWith(normalized))
                .sorted(String.CASE_INSENSITIVE_ORDER)
                .toList();
    }
}
```

```java
    private static final Set<String> BLOCKED = Set.of(
            "banlist",
            "minecraft:banlist",
            "bukkit:banlist",
            "kill",
            "minecraft:kill",
            "bukkit:kill",
            "advancement",
            "minecraft:advancement",
            "bukkit:advancement",
            "attribute",
            "minecraft:attribute",
            "bukkit:attribute",
            "bossbar",
            "minecraft:bossbar",
            "bukkit:bossbar"
    );

    private final AntiCheatPlusPlugin plugin;

    public DisabledVanillaCommandsListener(AntiCheatPlusPlugin plugin) {
        this.plugin = plugin;
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onPlayerCommand(PlayerCommandPreprocessEvent event) {
        String root = extractRoot(event.getMessage());
        if (!BLOCKED.contains(root)) {
            return;
        }

        event.setCancelled(true);
        deny(event.getPlayer());
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onServerCommand(ServerCommandEvent event) {
        String root = extractRoot(event.getCommand());
        if (!BLOCKED.contains(root)) {
            return;
        }

        event.setCancelled(true);
        deny(event.getSender());
    }

    private void deny(CommandSender sender) {
        sender.sendMessage(plugin.color("&cEsse comando está desativado neste servidor."));
    }

    private String extractRoot(String raw) {
        if (raw == null) {
            return "";
        }

        String normalized = raw.trim();
        while (normalized.startsWith("/")) {
            normalized = normalized.substring(1);
        }

        int space = normalized.indexOf(' ');
        if (space >= 0) {
            normalized = normalized.substring(0, space);
        }

        return normalized.toLowerCase(Locale.ROOT);
    }
}
```

```java
            event.disallow(

                    punishments.buildBanScreen(ban)
            );
```

```java
    private final AntiCheatPlusPlugin plugin;

    private final Map<String, IpBanRecord> activeIpBans = new ConcurrentHashMap<>();

    public PunishmentManager(AntiCheatPlusPlugin plugin) {
        this.plugin = plugin;
        this.dataFile = new File(plugin.getDataFolder(), "punishments.yml");

    public synchronized void load() {
        history.clear();
        ipHistory.clear();
        activeBans.clear();
        activeIpBans.clear();
        knownAddresses.clear();
        nextId.set(1L);

        if (!dataFile.exists()) return;

        YamlConfiguration data = YamlConfiguration.loadConfiguration(dataFile);
        long maxId = 0L;

        ConfigurationSection records = data.getConfigurationSection("records");
        if (records != null) {
            for (String rawId : records.getKeys(false)) {
                try {
                    long id = Long.parseLong(rawId);
                    String base = "records." + rawId + ".";
                    PunishmentType type = PunishmentType.valueOf(data.getString(base + "type", "WARN"));
                    UUID targetUuid = UUID.fromString(data.getString(base + "target-uuid", ""));
                    String targetName = data.getString(base + "target-name", "Desconhecido");
                    String actorUuidRaw = data.getString(base + "actor-uuid", "");
                    UUID actorUuid = actorUuidRaw == null || actorUuidRaw.isBlank() ? null : UUID.fromString(actorUuidRaw);
                    PunishmentRecord record = new PunishmentRecord(
                            id,
                            type,
                            targetUuid,
                            targetName,
                            actorUuid,
                            data.getString(base + "actor-name", "CONSOLE"),
                            data.getString(base + "reason", "Sem motivo informado."),
                            data.getLong(base + "created-at", 0L),
                            data.getBoolean(base + "active", false),
                            data.getBoolean(base + "delivered", true)
                    );
                    history.add(record);
                    if (type == PunishmentType.BAN && record.active()) activeBans.put(targetUuid, record);
                    maxId = Math.max(maxId, id);
                } catch (IllegalArgumentException exception) {
                    plugin.getLogger().warning("Registro inválido ignorado em punishments.yml: " + rawId);
                }
            }
        }

        ConfigurationSection ipBans = data.getConfigurationSection("ip-bans");
        if (ipBans != null) {
            for (String rawId : ipBans.getKeys(false)) {
                try {
                    long id = Long.parseLong(rawId);
                    String base = "ip-bans." + rawId + ".";
                    String actorUuidRaw = data.getString(base + "actor-uuid", "");
                    UUID actorUuid = actorUuidRaw == null || actorUuidRaw.isBlank() ? null : UUID.fromString(actorUuidRaw);
                    IpBanRecord record = new IpBanRecord(
                            id,
                            normalizeIp(data.getString(base + "ip", "")),
                            data.getString(base + "target-name", data.getString(base + "ip", "")),
                            actorUuid,
                            data.getString(base + "actor-name", "CONSOLE"),
                            data.getString(base + "reason", "Sem motivo informado."),
                            data.getLong(base + "created-at", 0L),
                            data.getBoolean(base + "active", false)
                    );
                    if (!record.ip().isBlank()) {
                        ipHistory.add(record);
                        if (record.active()) activeIpBans.put(record.ip(), record);
                        maxId = Math.max(maxId, id);
                    }
                } catch (IllegalArgumentException exception) {
                    plugin.getLogger().warning("Banimento de IP inválido ignorado: " + rawId);
                }
            }
        }

        ConfigurationSection known = data.getConfigurationSection("known-addresses");
        if (known != null) {
            for (String rawUuid : known.getKeys(false)) {
                try {
                    UUID uuid = UUID.fromString(rawUuid);
                    String base = "known-addresses." + rawUuid + ".";
                    String ip = normalizeIp(data.getString(base + "ip", ""));
                    if (!ip.isBlank()) knownAddresses.put(uuid, new KnownAddress(data.getString(base + "name", "Desconhecido"), ip));
                } catch (IllegalArgumentException ignored) {
                    plugin.getLogger().warning("Endereço conhecido inválido ignorado: " + rawUuid);
                }
            }
        }

        history.sort(Comparator.comparingLong(PunishmentRecord::id));
        ipHistory.sort(Comparator.comparingLong(IpBanRecord::id));
        nextId.set(maxId + 1L);
    }

    public synchronized void save() {
        YamlConfiguration data = new YamlConfiguration();

        for (PunishmentRecord record : history) {
            String base = "records." + record.id() + ".";
            data.set(base + "type", record.type().name());
            data.set(base + "target-uuid", record.targetUuid().toString());
            data.set(base + "target-name", record.targetName());
            data.set(base + "actor-uuid", record.actorUuid() == null ? "" : record.actorUuid().toString());
            data.set(base + "actor-name", record.actorName());
            data.set(base + "reason", record.reason());
            data.set(base + "created-at", record.createdAt());
            data.set(base + "active", record.active());
            data.set(base + "delivered", record.delivered());
        }

        for (IpBanRecord record : ipHistory) {
            String base = "ip-bans." + record.id() + ".";
            data.set(base + "ip", record.ip());
            data.set(base + "target-name", record.targetName());
            data.set(base + "actor-uuid", record.actorUuid() == null ? "" : record.actorUuid().toString());
            data.set(base + "actor-name", record.actorName());
            data.set(base + "reason", record.reason());
            data.set(base + "created-at", record.createdAt());
            data.set(base + "active", record.active());
        }

        for (Map.Entry<UUID, KnownAddress> entry : knownAddresses.entrySet()) {
            String base = "known-addresses." + entry.getKey() + ".";
            data.set(base + "name", entry.getValue().name());
            data.set(base + "ip", entry.getValue().ip());
        }

        try {
            File parent = dataFile.getParentFile();
            if (parent != null && !parent.exists()) parent.mkdirs();
            data.save(dataFile);
        } catch (IOException exception) {

    }

    public ResolvedTarget resolveKnownTarget(String input) {
        if (input == null || input.isBlank()) return null;
        String query = input.trim();

        try {
            UUID uuid = UUID.fromString(query);
            Player online = Bukkit.getPlayer(uuid);
            if (online != null) {
                return new ResolvedTarget(uuid, online.getName(), online, online);
            }

            KnownAddress known = knownAddresses.get(uuid);
            if (known != null) {
                return new ResolvedTarget(uuid, known.name(), null, null);
            }

            for (int i = history.size() - 1; i >= 0; i--) {
                PunishmentRecord record = history.get(i);
                if (record.targetUuid().equals(uuid)) {
                    return new ResolvedTarget(uuid, record.targetName(), null, null);
                }
            }
        } catch (IllegalArgumentException ignored) {
        }

        for (Map.Entry<UUID, KnownAddress> entry : knownAddresses.entrySet()) {
            KnownAddress known = entry.getValue();
            if (known.name().equalsIgnoreCase(query)) {
                Player online = Bukkit.getPlayer(entry.getKey());
                return new ResolvedTarget(
                        entry.getKey(),
                        known.name(),
                        online,
                        online
                );
            }
        }

        for (int i = history.size() - 1; i >= 0; i--) {
            PunishmentRecord record = history.get(i);
            if (record.targetName().equalsIgnoreCase(query)) {
                Player online = Bukkit.getPlayer(record.targetUuid());
                return new ResolvedTarget(
                        record.targetUuid(),
                        record.targetName(),
                        online,
                        online
                );
            }
        }

        return null;
    }

    public synchronized List<String> getKnownPlayerNames(String prefix) {
        String normalized = prefix == null ? "" : prefix.trim().toLowerCase(java.util.Locale.ROOT);
        Set<String> names = new LinkedHashSet<>();

        for (Player player : Bukkit.getOnlinePlayers()) {
            if (player.getName().toLowerCase(java.util.Locale.ROOT).startsWith(normalized)) {
                names.add(player.getName());
            }
        }

        for (KnownAddress known : knownAddresses.values()) {
            if (known.name() != null
                    && !known.name().isBlank()
                    && known.name().toLowerCase(java.util.Locale.ROOT).startsWith(normalized)) {
                names.add(known.name());
            }
        }

        for (int i = history.size() - 1; i >= 0; i--) {
            String name = history.get(i).targetName();
            if (name != null
                    && !name.isBlank()
                    && name.toLowerCase(java.util.Locale.ROOT).startsWith(normalized)) {
                names.add(name);
            }
        }

        return names.stream()
                .sorted(String.CASE_INSENSITIVE_ORDER)
                .toList();
    }

    public synchronized IpBanRecord banIp(String ip, String targetName, CommandSender actor, String reason) {
        String normalized = normalizeIp(ip);
        if (normalized.isBlank() || activeIpBans.containsKey(normalized)) return null;

        UUID actorUuid = actor instanceof Player player ? player.getUniqueId() : null;
        IpBanRecord record = new IpBanRecord(
                nextId.getAndIncrement(),
                normalized,
                targetName == null || targetName.isBlank() ? normalized : targetName,
                actorUuid,
                actorName(actor),
                normalizeReason(reason),
                System.currentTimeMillis(),
                true
        );

        ipHistory.add(record);
        activeIpBans.put(normalized, record);
        save();

        for (Player online : Bukkit.getOnlinePlayers()) {
            if (online.getAddress() != null
                    && online.getAddress().getAddress() != null
                    && normalizeIp(online.getAddress().getAddress().getHostAddress()).equals(normalized)) {
                online.kickPlayer(buildIpBanScreen(record));
            }
        }
        return record;
    }

        IpBanRecord record = activeIpBans.remove(normalized);

        save();
        return record;
    }

    public synchronized PunishmentRecord kick(Player target, CommandSender actor, String reason) {
        if (target == null || !target.isOnline()) return null;
        ResolvedTarget resolved = new ResolvedTarget(target.getUniqueId(), target.getName(), target, target);
        PunishmentRecord record = createRecord(PunishmentType.KICK, resolved, actor, normalizeReason(reason), false, true);
        save();
        target.kickPlayer(buildKickScreen(record));
        return record;
    }

    public synchronized List<PunishmentRecord> getUnreadWarnings(UUID uuid) {
        List<PunishmentRecord> result = new ArrayList<>();
        for (PunishmentRecord record : history) {
            if (record.type() == PunishmentType.WARN && record.targetUuid().equals(uuid) && !record.delivered()) result.add(record);
        }
        return result;
    }

    public synchronized void markWarningsDelivered(List<PunishmentRecord> records) {
        boolean changed = false;
        for (PunishmentRecord record : records) {
            if (record != null && !record.delivered()) {
                record.setDelivered(true);
                changed = true;
            }
        }
        if (changed) save();
    }

    public void sendWarning(Player player, PunishmentRecord record) {
        player.sendMessage(plugin.color("&c&lADVERTÊNCIA"));
        player.sendMessage(plugin.color("&7Motivo: &f" + record.reason()));
        player.sendMessage(plugin.color("&7Aplicada por: &f" + record.actorName()));
    }

    public int getHistorySize() {
        synchronized (this) {
            return history.size() + ipHistory.size();
        }
    }

    public boolean isExempt(ResolvedTarget target, String action) {
        return target != null
                && target.onlinePlayer() != null
                && target.onlinePlayer().hasPermission("anticheatplus.bypass." + action);
    }

    public boolean isSelf(CommandSender actor, ResolvedTarget target) {
        return actor instanceof Player player && target != null && player.getUniqueId().equals(target.uuid());
    }

    public String actorName(CommandSender sender) {
        return sender == null ? "CONSOLE" : sender.getName();
    }

    public static String normalizeIp(String ip) {
        if (ip == null) return "";
        String normalized = ip.trim().toLowerCase();
        int zone = normalized.indexOf('%');
        if (zone >= 0) normalized = normalized.substring(0, zone);
        return normalized;
    }

    private PunishmentRecord createRecord(
            PunishmentType type,
            ResolvedTarget target,
            CommandSender actor,
            String reason,
            boolean active,
            boolean delivered
    ) {
        UUID actorUuid = actor instanceof Player player ? player.getUniqueId() : null;
        PunishmentRecord record = new PunishmentRecord(
                nextId.getAndIncrement(), type, target.uuid(), target.name(), actorUuid,
                actorName(actor), reason, System.currentTimeMillis(), active, delivered
        );
        history.add(record);
        return record;
    }

    private String normalizeReason(String reason) {
        if (reason == null || reason.isBlank()) {
            return "Sem motivo.";
        }
        String trimmed = reason.trim();
        int max = getMaxReasonLength();
        return trimmed.length() > max ? trimmed.substring(0, max) : trimmed;
    }
```

```java
package com.redeedge.anticheatplus.punishment;

import org.bukkit.Bukkit;
import org.bukkit.OfflinePlayer;
import org.bukkit.entity.Player;

import java.util.UUID;

public final class TargetResolver {

    private TargetResolver() {
    }

    public static ResolvedTarget resolve(String input) {
        if (input == null || input.isBlank()) {
            return null;
        }

        Player exactOnline = Bukkit.getPlayerExact(input);
        if (exactOnline != null) {
            return new ResolvedTarget(
                    exactOnline.getUniqueId(),
                    exactOnline.getName(),
                    exactOnline,
                    exactOnline
            );
        }

        try {
            UUID uuid = UUID.fromString(input);
            OfflinePlayer offline = Bukkit.getOfflinePlayer(uuid);
            String name = offline.getName();
            if (name == null || name.isBlank()) {
                name = uuid.toString();
            }
            return new ResolvedTarget(uuid, name, offline, offline.getPlayer());
        } catch (IllegalArgumentException ignored) {
        }

        OfflinePlayer offline = Bukkit.getOfflinePlayer(input);
        if (!offline.hasPlayedBefore() && !offline.isOnline()) {
            return null;
        }

        String name = offline.getName() == null ? input : offline.getName();
        return new ResolvedTarget(offline.getUniqueId(), name, offline, offline.getPlayer());
    }
}
```
