static std::filesystem::path getSingleMacroFilePath() {
    auto saveDir = Mod::get()->getSaveDir();
    std::filesystem::path stdDir(saveDir.string());
    std::error_code ec;
    if (!std::filesystem::exists(stdDir, ec)) {
        std::filesystem::create_directories(stdDir, ec);
    }
    return stdDir / "macros.lolbot";
}

static int getLevelKey(GJGameLevel* level) {
    if (!level) return 0;
    if (level->m_levelID.value() > 0) {
        return level->m_levelID.value();
    }
    std::string name = level->m_levelName;
    int hash = 0;
    for (unsigned char c : name) {
        hash = hash * 131 + c;
    }
    return hash == 0 ? 1 : -std::abs(hash);
}

static void saveMacroToSingleFile(int levelKey) {
    if (levelKey == 0) return;
    auto filePath = getSingleMacroFilePath();

    struct StoredLevel {
        int key;
        std::vector<BotAction> actions;
    };
    std::vector<StoredLevel> allLevels;

    if (std::filesystem::exists(filePath)) {
        std::ifstream in(filePath, std::ios::binary);
        char magic[4] = {};
        int levelCount = 0;
        if (in.read(magic, 4) && std::memcmp(magic, "LB01", 4) == 0 &&
            in.read(reinterpret_cast<char*>(&levelCount), sizeof(levelCount))) {
            for (int i = 0; i < levelCount; ++i) {
                int k = 0;
                uint32_t count = 0;
                if (!in.read(reinterpret_cast<char*>(&k), sizeof(k)) ||
                    !in.read(reinterpret_cast<char*>(&count), sizeof(count))) {
                    break;
                }
                std::vector<BotAction> acts(count);
                if (count > 0) {
                    in.read(reinterpret_cast<char*>(acts.data()), count * sizeof(BotAction));
                }
                if (k != levelKey) {
                    allLevels.push_back({ k, std::move(acts) });
                }
            }
        }
    }

    if (!g_botState.replayActions.empty()) {
        allLevels.push_back({ levelKey, g_botState.replayActions });
    }

    std::ofstream out(filePath, std::ios::binary | std::ios::trunc);
    if (!out) return;

    const char magic[4] = { 'L', 'B', '0', '1' };
    int levelCount = static_cast<int>(allLevels.size());
    out.write(magic, 4);
    out.write(reinterpret_cast<const char*>(&levelCount), sizeof(levelCount));

    for (const auto& item : allLevels) {
        uint32_t count = static_cast<uint32_t>(item.actions.size());
        out.write(reinterpret_cast<const char*>(&item.key), sizeof(item.key));
        out.write(reinterpret_cast<const char*>(&count), sizeof(count));
        if (count > 0) {
            out.write(reinterpret_cast<const char*>(item.actions.data()), count * sizeof(BotAction));
        }
    }
}

static bool loadMacroFromSingleFile(int levelKey) {
    g_botState.replayActions.clear();
    if (levelKey == 0) return false;

    auto filePath = getSingleMacroFilePath();
    if (!std::filesystem::exists(filePath)) return false;

    std::ifstream in(filePath, std::ios::binary);
    char magic[4] = {};
    int levelCount = 0;
    if (!in.read(magic, 4) || std::memcmp(magic, "LB01", 4) != 0 ||
        !in.read(reinterpret_cast<char*>(&levelCount), sizeof(levelCount))) {
        return false;
    }

    for (int i = 0; i < levelCount; ++i) {
        int k = 0;
        uint32_t count = 0;
        if (!in.read(reinterpret_cast<char*>(&k), sizeof(k)) ||
            !in.read(reinterpret_cast<char*>(&count), sizeof(count))) {
            return false;
        }
        if (k == levelKey) {
            g_botState.replayActions.resize(count);
            if (count > 0) {
                in.read(reinterpret_cast<char*>(g_botState.replayActions.data()), count * sizeof(BotAction));
            }
            return true;
        }
        else if (count > 0) {
            in.seekg(static_cast<std::streamoff>(count * sizeof(BotAction)), std::ios::cur);
        }
    }
    return false;
}

static float getEffectiveSpeed() {
    if (!g_botState.speedhackEnabled) return 1.0f;
    return g_botState.speedhack;
}

static void syncFmodAudioSpeed() {
    auto fmod = FMODAudioEngine::sharedEngine();
    if (!fmod) return;

    float targetPitch = 1.0f;
    float effSpeed = getEffectiveSpeed();
    if (PlayLayer::get() && g_botState.speedhackMusic && std::abs(effSpeed - 1.0f) > 0.001f) {
        targetPitch = effSpeed;
    }

    if (fmod->m_globalChannel) {
        fmod->m_globalChannel->setPitch(targetPitch);
    }
    if (fmod->m_backgroundMusicChannel) {
        fmod->m_backgroundMusicChannel->setPitch(targetPitch);
    }
    if (fmod->m_system) {
        FMOD::ChannelGroup* masterGroup = nullptr;
        if (fmod->m_system->getMasterChannelGroup(&masterGroup) == FMOD_OK && masterGroup) {
            masterGroup->setPitch(targetPitch);
        }
    }
}

    Mod::get()->setSavedValue<float>("speedhack-value", g_botState.speedhack);
    Mod::get()->setSavedValue<bool>("speedhack-enabled", g_botState.speedhackEnabled);

}

static void applyFpsTps() {
    auto director = CCDirector::sharedDirector();
    if (!director) return;

    if (g_botState.fpsTpsEnabled && g_botState.targetFpsTps >= 30) {
        g_botState.targetFpsTps = std::clamp(g_botState.targetFpsTps, static_cast<int64_t>(30), MAX_FPS_TPS);
        director->setAnimationInterval(1.0 / 10000.0);
        if (auto gm = GameManager::sharedState()) {
            int clampedInt = static_cast<int>(std::min<int64_t>(g_botState.targetFpsTps, 2147483647LL));
            gm->m_customFPSTarget = clampedInt;
            gm->setGameVariable("0116", true);
            gm->setGameVariable("0030", false);
            gm->updateCustomFPS();
        }
    }
    else {
        if (auto gm = GameManager::sharedState()) {
            gm->setGameVariable("0116", false);
            gm->updateCustomFPS();
        }
    }
    Mod::get()->setSavedValue<bool>("fps-tps-enabled", g_botState.fpsTpsEnabled);
    Mod::get()->setSavedValue<int64_t>("fps-tps-value", g_botState.targetFpsTps);
}

static void cycleBotMode() {
    if (g_botState.mode == BotMode::Off) {
        g_botState.mode = BotMode::Record;
        g_botState.replayActions.clear();
        g_botState.replayActions.reserve(65536);
        g_botState.checkpoints.clear();
    }
    else if (g_botState.mode == BotMode::Record) {
        g_botState.mode = BotMode::Play;
        g_botState.playbackCursor = 0;
    }
    else {
        g_botState.mode = BotMode::Off;
    }
}

static std::string getKeyName(enumKeyCodes key) {
    if (key == KEY_None || key == KEY_Unknown) return "None";
    if (key == KEY_Alt || key == KEY_LeftMenu || key == KEY_RightMenu) return "Alt";
    if (key == KEY_Shift || key == KEY_LeftShift || key == KEY_RightShift) return "Shift";
    if (key == KEY_Control || key == KEY_LeftControl || key == KEY_RightContol) return "Ctrl";
    if (key == KEY_Tab) return "Tab";
    if (key == KEY_GraveAccent) return "~";
    if (key == KEY_Insert) return "Insert";
    if (key == KEY_Delete) return "Delete";
    if (key == KEY_Home) return "Home";
    if (key == KEY_End) return "End";
    if (key >= KEY_F1 && key <= KEY_F12) {
        return fmt::format("F{}", static_cast<int>(key - KEY_F1) + 1);
    }
    if (key >= KEY_A && key <= KEY_Z) {
        char ch = static_cast<char>('A' + (key - KEY_A));
        return std::string(1, ch);
    }
    if (key >= KEY_Zero && key <= KEY_Nine) {
        char ch = static_cast<char>('0' + (key - KEY_Zero));
        return std::string(1, ch);
    }
    return fmt::format("Key({})", static_cast<int>(key));
}

static enumKeyCodes normalizeKey(enumKeyCodes key) {
    if (key == KEY_LeftMenu || key == KEY_RightMenu) return KEY_Alt;
    if (key == KEY_LeftShift || key == KEY_RightShift) return KEY_Shift;
    if (key == KEY_LeftControl || key == KEY_RightContol) return KEY_Control;
    return key;
}

class MegaHackWindowNode : public CCNode {
public:
    CCLayerColor* m_bodyBg = nullptr;
    CCLayerColor* m_headerBg = nullptr;
    CCLabelBMFont* m_headerTitle = nullptr;
    CCMenu* m_menu = nullptr;
    float m_width = 176.f;
    float m_height = 220.f;
    float m_headerHeight = 20.f;

    bool initWindow(const char* title, float width, float height) {
        if (!CCNode::init()) return false;
        m_width = width;
        m_height = height;
        this->setContentSize({ width, height });
        this->setAnchorPoint({ 0.5f, 0.5f });

        m_bodyBg = CCLayerColor::create({ 16, 18, 28, 215 }, width, height);
        m_bodyBg->setPosition({ 0.f, 0.f });
        this->addChild(m_bodyBg, 1);

        m_headerBg = CCLayerColor::create({ 45, 95, 215, 245 }, width, m_headerHeight);
        m_headerBg->setPosition({ 0.f, height - m_headerHeight });
        this->addChild(m_headerBg, 2);

        m_headerTitle = CCLabelBMFont::create(title, "bigFont.fnt");
        m_headerTitle->setScale(0.42f);
        m_headerTitle->setPosition({ width / 2.f, height - m_headerHeight / 2.f + 1.f });
        this->addChild(m_headerTitle, 3);

        m_menu = CCMenu::create();
        m_menu->setPosition({ 0.f, 0.f });
        this->addChild(m_menu, 4);

        return true;
    }

    void applyTheme(MenuTheme theme) {
        if (!m_bodyBg || !m_headerBg) return;
        switch (theme) {
        case MenuTheme::MegaHack:
            m_bodyBg->setColor({ 16, 18, 28 });
            m_bodyBg->setOpacity(215);
            m_headerBg->setColor({ 45, 95, 215 });
            m_headerBg->setOpacity(245);
            break;
        case MenuTheme::GeometryDash:
            m_bodyBg->setColor({ 20, 35, 75 });
            m_bodyBg->setOpacity(175);
            m_headerBg->setColor({ 40, 175, 65 });
            m_headerBg->setOpacity(235);
            break;
        case MenuTheme::Dark:
            m_bodyBg->setColor({ 12, 12, 14 });
            m_bodyBg->setOpacity(225);
            m_headerBg->setColor({ 38, 38, 44 });
            m_headerBg->setOpacity(245);
            break;
        case MenuTheme::Neon:
            m_bodyBg->setColor({ 8, 22, 28 });
            m_bodyBg->setOpacity(210);
            m_headerBg->setColor({ 0, 185, 175 });
            m_headerBg->setOpacity(240);
            break;
        }
    }

    bool isHeaderTouched(CCPoint worldPos) {
        CCPoint local = this->convertToNodeSpace(worldPos);
        return local.x >= 0.f && local.x <= m_width &&
            local.y >= (m_height - m_headerHeight) && local.y <= m_height;
    }
};

    // Перетаскивание окон мышкой за верхнюю шапку
    bool ccTouchBegan(CCTouch* touch, CCEvent* event) override {
        CCPoint pos = touch->getLocation();
        if (m_globalWin && m_globalWin->isHeaderTouched(pos)) {
            m_draggedWin = m_globalWin;
            m_dragOffset = ccpSub(m_globalWin->getPosition(), pos);
            return true;
        }
        if (m_playerWin && m_playerWin->isHeaderTouched(pos)) {
            m_draggedWin = m_playerWin;
            m_dragOffset = ccpSub(m_playerWin->getPosition(), pos);
            return true;
        }
        return FLAlertLayer::ccTouchBegan(touch, event);
    }

    void ccTouchMoved(CCTouch* touch, CCEvent* event) override {
        if (m_draggedWin) {
            CCPoint newPos = ccpAdd(touch->getLocation(), m_dragOffset);
            m_draggedWin->setPosition(newPos);
            if (m_draggedWin == m_globalWin) {
                g_botState.globalWinPos = newPos;
            }
            else if (m_draggedWin == m_playerWin) {
                g_botState.playerWinPos = newPos;
            }
            return;
        }
        FLAlertLayer::ccTouchMoved(touch, event);
    }

    void ccTouchEnded(CCTouch* touch, CCEvent* event) override {
        if (m_draggedWin) {
            m_draggedWin = nullptr;
            return;
        }
        FLAlertLayer::ccTouchEnded(touch, event);
    }

    void keyBackClicked() override {
        closeMenu();
    }

    void refreshUI() {
        if (m_globalWin) m_globalWin->applyTheme(g_botState.theme);
        if (m_playerWin) m_playerWin->applyTheme(g_botState.theme);

        // Global items
        if (m_lblBotMode) {

            m_lblBotMode->setString(s.c_str());
            styleToggleLabel(m_lblBotMode, g_botState.mode != BotMode::Off);
        }
        if (m_lblFps) {
            std::string s = g_botState.fpsTpsEnabled
                ? fmt::format("FPS: {}", g_botState.targetFpsTps)
                : "FPS Bypass";

        if (m_lblSpeed) {

        styleToggleLabel(m_lblClicks, g_botState.showClickCounter);
        styleToggleLabel(m_lbl1stClick, g_botState.countStartClick);
        styleToggleLabel(m_lblUnlockIcons, g_botState.unlockAllIcons);
        if (m_lblTheme) {
            std::string s = fmt::format("Theme: {}", getThemeName());
            m_lblTheme->setString(s.c_str());
        }
        if (m_lblKeybinds) {
            styleToggleLabel(m_lblKeybinds, g_botState.activePopup == PopupType::KeybindsSettings);

        styleToggleLabel(m_lblNoSpikes, g_botState.noSpikes);
        styleToggleLabel(m_lblNoMirror, g_botState.noMirror);

            }

    void onPopupMinus(CCObject*) {
        if (g_botState.activePopup == PopupType::FpsSettings) {
            g_botState.fpsTpsEnabled = true;
            g_botState.targetFpsTps = std::max<int64_t>(60, g_botState.targetFpsTps - 60);
            applyFpsTps();
        }
        else if (g_botState.activePopup == PopupType::SpeedSettings) {
            g_botState.speedhackEnabled = true;
            applySpeedhack(std::max(0.10f, g_botState.speedhack - 0.25f));
        }
        refreshUI();
    }

    void onPopupPlus(CCObject*) {
        if (g_botState.activePopup == PopupType::FpsSettings) {
            g_botState.fpsTpsEnabled = true;
            g_botState.targetFpsTps = std::min<int64_t>(MAX_FPS_TPS, g_botState.targetFpsTps + 60);
            applyFpsTps();
        }
        else if (g_botState.activePopup == PopupType::SpeedSettings) {
            g_botState.speedhackEnabled = true;
            applySpeedhack(std::min(3.00f, g_botState.speedhack + 0.25f));
        }
        refreshUI();
    }

    void onPopupMainAction(CCObject*) {
        if (g_botState.activePopup == PopupType::BotSettings) {
            cycleBotMode();
        }
        else if (g_botState.activePopup == PopupType::FpsSettings) {
            g_botState.fpsTpsEnabled = !g_botState.fpsTpsEnabled;
            applyFpsTps();
        }
        else if (g_botState.activePopup == PopupType::SpeedSettings) {
            g_botState.speedhackEnabled = !g_botState.speedhackEnabled;
            applySpeedhack(g_botState.speedhack);
        }
        else if (g_botState.activePopup == PopupType::ThemeSettings) {
            onCycleTheme(nullptr);
            return;
        }
        refreshUI();
    }

    void onPopupExtraAction(CCObject*) {
        if (g_botState.activePopup == PopupType::BotSettings) {
            saveMacroToSingleFile(g_botState.currentLevelKey);
        }
        else if (g_botState.activePopup == PopupType::FpsSettings) {
            g_botState.fpsTpsEnabled = true;
            g_botState.targetFpsTps = std::min<int64_t>(MAX_FPS_TPS, g_botState.targetFpsTps * 2);
            applyFpsTps();
        }
        else if (g_botState.activePopup == PopupType::SpeedSettings) {
            g_botState.speedhackMusic = !g_botState.speedhackMusic;
            Mod::get()->setSavedValue<bool>("speedhack-music", g_botState.speedhackMusic);
            syncFmodAudioSpeed();
        }
        else if (g_botState.activePopup == PopupType::ThemeSettings) {
            togglePopup(PopupType::KeybindsSettings);
            return;
        }
        refreshUI();
    }

    void onAutoComplete(CCObject*) {
        if (auto pl = PlayLayer::get()) {
            closeMenu();
            Loader::get()->queueInMainThread([pl]() {
                if (PlayLayer::get() == pl && !pl->m_hasCompletedLevel) {
                    if (pl->m_isPaused) {
                        pl->resume();
                    }
                    bool prevNoclip = g_botState.noclip;
                    g_botState.noclip = true;
                    pl->levelComplete();
                    g_botState.noclip = prevNoclip;
                }
                });
        }
    }

    static LolollBotMenuLayer* create() {
        auto ret = new LolollBotMenuLayer();
        if (ret && ret->init()) {
            ret->autorelease();
            return ret;
        }
        CC_SAFE_DELETE(ret);
        return nullptr;
    }

static LolollBotMenuLayer* getOpenMenu() {

                }

        }

    }

class $modify(LolollKeyboardDispatcher, CCKeyboardDispatcher) {
    bool dispatchKeyboardMSG(enumKeyCodes key, bool isKeyDown, bool isKeyRepeat, double timestamp) {
        if (isKeyDown && !isKeyRepeat) {
            enumKeyCodes normKey = normalizeKey(key);

            if (g_botState.waitingForBind != KeybindTarget::None) {
                if (normKey != KEY_Unknown && normKey != KEY_None) {
                    enumKeyCodes assigned = (normKey == KEY_Escape) ? KEY_None : normKey;
                    switch (g_botState.waitingForBind) {
                    case KeybindTarget::ToggleMenu:
                        if (assigned != KEY_None) g_botState.toggleKey = assigned;
                        break;
                    case KeybindTarget::CycleBot:
                        g_botState.botKey = assigned;
                        break;
                    case KeybindTarget::ToggleNoclip:
                        g_botState.noclipKey = assigned;
                        break;
                    case KeybindTarget::ToggleSpeedhack:
                        g_botState.speedKey = assigned;
                        break;
                    default:
                        break;
                    }
                    saveAllKeybinds();
                    g_botState.waitingForBind = KeybindTarget::None;
                    if (auto menu = getOpenMenu()) {
                        menu->refreshUI();
                    }
                }
                return true;
            }

            if (g_botState.toggleKey != KEY_None && normKey == normalizeKey(g_botState.toggleKey)) {
                toggleLolollBotMenu();
                return true;
            }
            if (g_botState.botKey != KEY_None && normKey == normalizeKey(g_botState.botKey)) {
                cycleBotMode();
                if (auto menu = getOpenMenu()) menu->refreshUI();
                return true;
            }
            if (g_botState.noclipKey != KEY_None && normKey == normalizeKey(g_botState.noclipKey)) {
                g_botState.noclip = !g_botState.noclip;
                if (auto menu = getOpenMenu()) menu->refreshUI();
                return true;
            }
            if (g_botState.speedKey != KEY_None && normKey == normalizeKey(g_botState.speedKey)) {
                g_botState.speedhackEnabled = !g_botState.speedhackEnabled;
                applySpeedhack(g_botState.speedhack);
                if (auto menu = getOpenMenu()) menu->refreshUI();
                return true;
            }
        }
        return CCKeyboardDispatcher::dispatchKeyboardMSG(key, isKeyDown, isKeyRepeat, timestamp);
    }
};

            g_botState.currentFrame = cp.frame;

                    masterGroup->setPitch(1.0f);
                }
            }
        }
    }

        PlayLayer::postUpdate(dt);

            if (g_botState.forceTrail) this->m_player1->activateStreak();
            if (g_botState.noTrail) this->m_player1->deactivateStreak(true);
        }

        updateClickCounterUI();

        if (m_hasCompletedLevel || m_isPaused) {
            return;
        }

        g_botState.currentFrame++;

        if (g_botState.mode == BotMode::Play) {
            while (
                g_botState.playbackCursor < g_botState.replayActions.size() &&
                g_botState.replayActions[g_botState.playbackCursor].frame <= g_botState.currentFrame

                const auto& action = g_botState.replayActions[g_botState.playbackCursor];

                        if (action.isPress) {
                            targetPlayer->pushButton(PlayerButton::Jump);
                        }
                        else {
                            targetPlayer->releaseButton(PlayerButton::Jump);
                        }
                    }
                }
                g_botState.playbackCursor++;
            }
        }
    }
};

static void saveMacroToDisk(const std::filesystem::path& path) {
    if (path.empty() || g_botState.replayActions.empty()) {
        return;
    }
    std::ofstream out(path, std::ios::binary | std::ios::trunc);
    if (!out.is_open()) {

    }

    out.write(reinterpret_cast<const char*>(&count), sizeof(count));
    out.write(reinterpret_cast<const char*>(g_botState.replayActions.data()), count * sizeof(BotAction));
    out.close();

static void loadMacroFromDisk(const std::filesystem::path& path) {
    g_botState.replayActions.clear();
    std::error_code ec;
    if (path.empty() || !std::filesystem::exists(path, ec)) {
        return;
    }

    std::ifstream in(path, std::ios::binary);
    if (!in.is_open()) {
        return;
    }

    size_t count = 0;
    in.read(reinterpret_cast<char*>(&count), sizeof(count));
    if (count > 0 && count < 5000000) {
        g_botState.replayActions.resize(count);
        in.read(reinterpret_cast<char*>(g_botState.replayActions.data()), count * sizeof(BotAction));
    }
    in.close();

    if (auto existingPopup = typeinfo_cast<LolollBotPopup*>(scene->getChildByID("lolollbot-main-popup"_spr))) {
        existingPopup->removeFromParentAndCleanup(true);
        return;
    }

    if (auto popup = LolollBotPopup::create()) {
        popup->show();
    }
}

class $modify(LolollPlayLayer, PlayLayer) {
    bool init(GJGameLevel* level, bool useReplay, bool dontCreateObjects) {
        if (!PlayLayer::init(level, useReplay, dontCreateObjects)) {
            return false;
        }

        auto macroPath = getMacroPathForLevel(level);
        g_botState.currentMacroFile = macroPath.string();
        loadMacroFromDisk(macroPath);

        g_botState.currentFrame = 0;
        g_botState.playbackCursor = 0;
        g_botState.checkpoints.clear();
        applySpeedhack(g_botState.speedhack);

        return true;
    }

    void destroyPlayer(PlayerObject* player, GameObject* object) {
        if (g_botState.noclip && object != m_anticheatSpike) {
            return;
        }
        PlayLayer::destroyPlayer(player, object);
    }

    void storeCheckpoint(CheckpointObject* checkpoint) {
        PlayLayer::storeCheckpoint(checkpoint);
        if (g_botState.practiceFix && this->m_isPracticeMode) {
            g_botState.checkpoints.push_back({
                g_botState.currentFrame,
                g_botState.replayActions.size()
            });
        }
    }

    void removeCheckpoint(bool first) {
        PlayLayer::removeCheckpoint(first);
        if (g_botState.practiceFix && !g_botState.checkpoints.empty()) {
            g_botState.checkpoints.pop_back();
        }
    }

    void resetLevel() {
        PlayLayer::resetLevel();

        if (g_botState.practiceFix && this->m_isPracticeMode && !g_botState.checkpoints.empty()) {
            const auto& cp = g_botState.checkpoints.back();
            g_botState.currentFrame = cp.frame;

            if (g_botState.mode == BotMode::Record && cp.recordedCount <= g_botState.replayActions.size()) {
                g_botState.replayActions.resize(cp.recordedCount);
                g_botState.replayActions.push_back({ cp.frame, false, false });
                g_botState.replayActions.push_back({ cp.frame, false, true });
            }

            g_botState.playbackCursor = 0;
            while (
                g_botState.playbackCursor < g_botState.replayActions.size() &&
                g_botState.replayActions[g_botState.playbackCursor].frame < g_botState.currentFrame
            ) {
                g_botState.playbackCursor++;
            }
            return;
        }

        g_botState.currentFrame = 0;
        g_botState.playbackCursor = 0;
        g_botState.checkpoints.clear();

        if (g_botState.mode == BotMode::Record) {
            g_botState.replayActions.clear();
        }
    }

    void levelComplete() {
        PlayLayer::levelComplete();
        if (g_botState.mode == BotMode::Record && !g_botState.replayActions.empty()) {
            if (!g_botState.currentMacroFile.empty()) {
                saveMacroToDisk(g_botState.currentMacroFile);
            }
            g_botState.mode = BotMode::Play;
        }
    }

    void onQuit() {
        applySpeedhack(1.0f);
        PlayLayer::onQuit();
    }

    void postUpdate(float dt) {
        PlayLayer::postUpdate(dt);

        if (m_hasCompletedLevel || m_isPaused) {
            return;
        }

        g_botState.currentFrame++;

        if (g_botState.mode == BotMode::Play) {
            while (
                g_botState.playbackCursor < g_botState.replayActions.size() &&
                g_botState.replayActions[g_botState.playbackCursor].frame <= g_botState.currentFrame
            ) {
                const auto& action = g_botState.replayActions[g_botState.playbackCursor];
                if (action.frame == g_botState.currentFrame) {
                    if (action.isPress) {
                        this->pushButton(1, !action.isPlayer2);
                    } else {
                        this->releaseButton(1, !action.isPlayer2);
                    }
                }
                g_botState.playbackCursor++;
            }
        }
    }
};

class $modify(LolollPlayerObject, PlayerObject) {
    bool pushButton(PlayerButton button) {
        if (g_botState.mode == BotMode::Record) {
            g_botState.replayActions.push_back({
                g_botState.currentFrame,
                true,
                this->m_isSecondPlayer
            });
        }
        return PlayerObject::pushButton(button);
    }

    bool releaseButton(PlayerButton button) {
        if (g_botState.mode == BotMode::Record) {
            g_botState.replayActions.push_back({
                g_botState.currentFrame,
                false,
                this->m_isSecondPlayer
            });
        }
        return PlayerObject::releaseButton(button);
    }
};
