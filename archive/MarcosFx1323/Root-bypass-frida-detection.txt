# Universal Android Root & Frida Detection Bypass

Este script de Frida (`.js`) está diseñado para evadir las técnicas modernas de detección de root y análisis de integridad del sistema en Android. Utiliza un enfoque híbrido (hooks nativos a nivel de `libc` y hooks en el entorno de ejecución de Java/ART) para camuflar la presencia de frameworks como Magisk, KernelSU, APatch, y herramientas de instrumentación dinámica.

## 🚀 Características y Elementos Ocultos

El script divide su funcionamiento en varias capas para cubrir la mayoría de los vectores de detección comunes:

### Capa Nativa (C/C++)
*   **Propiedades del Sistema:** Hookea `__system_property_get` y `__system_property_read_callback` para falsear propiedades críticas como `ro.debuggable`, `ro.build.tags`, `ro.secure`, y `ro.build.selinux`.
*   **Archivos y Directorios:** Intercepta llamadas a `fopen`, `stat`, `access`, `open`, `openat`, entre otras. Redirige consultas a rutas prohibidas (ej. `/system/bin/su`, `/sbin/su`, `/data/local/magisk`) hacia un directorio inexistente (`/nonexistent`), simulando un error `ENOENT` natural.
*   **Puntos de Montaje:** Hookea la lectura de `/proc/mounts`, `/proc/self/mounts` y variantes. Reemplaza en tiempo real los tokens sospechosos (como `magisk`, `supersu`, `busybox`) por caracteres inofensivos (`x`) para ocultar la sobreescritura del sistema de archivos.

### Capa Java (ART)
*   **Clase Build:** Sobrescribe en memoria los campos `TAGS`, `TYPE` y `FINGERPRINT` de `android.os.Build` para simular un dispositivo de producción (`user`, `release-keys`).
*   **I/O y Sistema de Archivos:** Hookea `java.io.File` (`exists`, `canRead`, etc.) y `libcore.io.BlockOs/Posix` para bloquear comprobaciones de rutas de root con excepciones `FileNotFoundException`.
*   **Ejecución de Comandos:** Bloquea el uso de `Runtime.exec()` y `ProcessBuilder` si detectan invocaciones a `su` o `which su`. También intercepta y modifica comandos `getprop` para que devuelvan los valores falseados.
*   **PackageManager y Binder (Ocultación de Apps):** 
    *   Filtra las listas de `getInstalledPackages` y `getInstalledApplications`.
    *   Hookea `IPackageManager$Stub$Proxy` a nivel Binder (más profundo que la API pública).
    *   **Oculta aplicaciones de root** conocidas (Magisk, KernelSU, SuperSU, LSPosed, etc.).
    *   Hace que las apps de root parezcan "no lanzables" devolviendo `null` en `getLaunchIntentForPackage` y `LauncherApps.getActivityList`.

---

## 📋 Requisitos

*   **Dispositivo Android** con acceso root (Magisk, KernelSU, APatch) o un entorno de pruebas vulnerable.
*   **Frida Server** ejecutándose en el dispositivo (o usar Frida Gadget inyectado en el APK).
*   **Frida CLI** instalado en tu PC (`pip install frida-tools`).
*   El proceso objetivo debe poder ser "spawneado" (iniciado por Frida) para que los hooks nativos actúen antes de que la app ejecute sus comprobaciones.

---


```js
/*
 * Universal Android Root & Frida Detection Bypass
 * 
 * Note: This script operates in user-space. It cannot bypass SELinux 
 * policy probes executed by the isolated `app_zygote` process, as those 
 * happen in a separate PID where Frida is not injected. 
 * Those specific detections require kernel-level bypasses (e.g., KernelSU/APatch).
 */

// ============================================================================
// NATIVE HOOKS: Hide Frida from /proc/self/maps
// ============================================================================

function hideFridaFromMaps() {
    const libc = Process.getModuleByName("libc.so");
    const trackedFds = new Set();

    function isMapsPath(pathPtr) {
        try {
            const path = Memory.readCString(pathPtr);
            return path && /^\/proc\/(\d+|self|thread-self)\/(task\/\d+\/)?maps$/.test(path);
        } catch (e) {
            return false;
        }
    }

    function hookOpen(symbol, pathArgIndex) {
        const ptr = libc.findExportByName(symbol);
        if (!ptr) return;

        Interceptor.attach(ptr, {
            onEnter(args) {
                this.isMaps = isMapsPath(args[pathArgIndex]);
            },
            onLeave(retval) {
                const fd = retval.toInt32();
                if (this.isMaps && fd >= 0) {
                    trackedFds.add(fd);
                }
            }
        });
    }

    ['open', '__open_2'].forEach(sym => hookOpen(sym, 0));
    ['openat', '__openat_2'].forEach(sym => hookOpen(sym, 1));

    const readPtr = libc.findExportByName("read");
    if (readPtr) {
        Interceptor.attach(readPtr, {
            onEnter(args) {
                this.fd = args[0].toInt32();
                this.buf = args[1];
                this.isTracked = trackedFds.has(this.fd);
            },
            onLeave(retval) {
                const n = retval.toInt32();
                if (!this.isTracked || n <= 0) return;

                try {
                    const data = Memory.readUtf8String(this.buf, n);
                    if (!data || data.length !== n) return;

                    const lines = data.split('\n');
                    let modified = false;
                    
                    const suspiciousTokens = [
                        "frida", "linjector", "gum-js-loop", "gum-realloc",
                        "frida-agent", "frida-gadget", "pool-frida"
                    ];

                    for (let i = 0; i < lines.length; i++) {
                        let line = lines[i];
                        let shouldFilter = false;

                        for (const token of suspiciousTokens) {
                            if (line.includes(token)) {
                                shouldFilter = true;
                                break;
                            }
                        }

                        if (!shouldFilter) {
                            const parts = line.trim().split(/\s+/);
                            if (parts.length === 5 && parts[1].includes('x')) {
                                shouldFilter = true;
                            }
                        }

                        if (shouldFilter) {
                            lines[i] = " ".repeat(line.length);
                            modified = true;
                        }
                    }

                    if (modified) {
                        const cleaned = lines.join('\n');
                        const bytes = new Uint8Array(n);
                        for (let i = 0; i < n; i++) {
                            bytes[i] = cleaned.charCodeAt(i) & 0xff;
                        }
                        Memory.writeByteArray(this.buf, bytes.buffer);
                        console.log("[+] Filtered anonymous executable pages from /proc/maps");
                    }
                } catch (e) {}
            }
        });
    }
}

hideFridaFromMaps();

// ============================================================================
// JAVA HOOKS: Root Detection Bypass
// ============================================================================

Java.perform(function () {
    // 1. File.exists() Bypass
    const File = Java.use("java.io.File");
    const rootPaths = [
        "/system/app/Superuser.apk",
        "/sbin/su", "/system/bin/su", "/system/xbin/su",
        "/data/local/xbin/su", "/data/local/bin/su",
        "/system/sd/xbin/su", "/system/bin/failsafe/su",
        "/data/local/su", "/su/bin/su",
        "/system/xbin/busybox"
    ];

    File.exists.implementation = function () {
        const path = this.getAbsolutePath();
        if (rootPaths.includes(path)) {
            console.log(`[-] Blocked File.exists() check: ${path}`);
            return false;
        }
        return this.exists();
    };

    // 2. Runtime.exec() Bypass
    const Runtime = Java.use("java.lang.Runtime");
    const isSuCommand = (cmd) => {
        if (!cmd) return false;
        return /(^|[\s\/])su([\s\/]|$)/.test(cmd);
    };

    Runtime.exec.overload("java.lang.String").implementation = function (cmd) {
        if (isSuCommand(cmd)) {
            console.log(`[-] Blocked Runtime.exec(): ${cmd}`);
            throw Java.use("java.io.IOException").$new("Command not found");
        }
        return this.exec(cmd);
    };

    Runtime.exec.overload("[Ljava.lang.String;").implementation = function (arr) {
        const joined = arr.join(" ");
        if (isSuCommand(joined)) {
            console.log(`[-] Blocked Runtime.exec([]): ${joined}`);
            throw Java.use("java.io.IOException").$new("Command not found");
        }
        return this.exec(arr);
    };

    // 3. Build.TAGS Bypass
    try {
        Java.use("android.os.Build").TAGS.value = "release-keys";
        console.log("[+] Spoofed Build.TAGS to release-keys");
    } catch (e) {}

    // 4. PackageManager Bypass (Robust version)
    const PackageManager = Java.use("android.app.ApplicationPackageManager");
    const rootPackages = [
        "com.topjohnwu.magisk", "eu.chainfire.supersu",
        "com.noshufou.android.su", "com.koushikdutta.superuser",
        "com.zachspong.temprootremovejb", "com.ramdroid.appquarantine",
        "me.weishu.kernelsu", "com.dergoogler.mmrl",
        "com.github.kr328.kernelsu.webui", "io.github.a13e300.ksuwebui"
    ];

    const isRootPkg = (pkg) => {
        if (!pkg) return false;
        const p = String(pkg).toLowerCase();
        if (rootPackages.includes(p)) return true;
        if (p.includes("magisk") || p.includes("kernelsu") || p.includes("ksuwebui")) return true;
        return false;
    };

    // Safely extract package name regardless of Java Bridge return type
    const extractPkgName = (item) => {
        try {
            if (!item) return "";
            const pn = item.packageName;
            if (!pn) return "";
            if (typeof pn === 'string') return pn;
            return pn.value || pn.toString() || "";
        } catch (e) {
            return "";
        }
    };

    // Prevent detectors from parsing the package name from the exception message
    const throwPkgNotFound = () => {
        throw Java.use("android.content.pm.PackageManager$NameNotFoundException").$new("Package not found");
    };

    // 4a. Block getPackageInfo and getApplicationInfo (ALL overloads)
    PackageManager.getPackageInfo.overloads.forEach(function(ov) {
        ov.implementation = function() {
            const pkg = arguments[0];
            if (isRootPkg(pkg)) {
                console.log(`[-] Blocked getPackageInfo: ${pkg}`);
                throwPkgNotFound();
            }
            return ov.apply(this, arguments);
        };
    });

    try {
        PackageManager.getPackageInfoAsUser.overloads.forEach(function(ov) {
            ov.implementation = function() {
                const pkg = arguments[0];
                if (isRootPkg(pkg)) {
                    console.log(`[-] Blocked getPackageInfoAsUser: ${pkg}`);
                    throwPkgNotFound();
                }
                return ov.apply(this, arguments);
            };
        });
    } catch(e) {}

    PackageManager.getApplicationInfo.overloads.forEach(function(ov) {
        ov.implementation = function() {
            const pkg = arguments[0];
            if (isRootPkg(pkg)) {
                console.log(`[-] Blocked getApplicationInfo: ${pkg}`);
                throwPkgNotFound();
            }
            return ov.apply(this, arguments);
        };
    });

    try {
        PackageManager.getApplicationInfoAsUser.overloads.forEach(function(ov) {
            ov.implementation = function() {
                const pkg = arguments[0];
                if (isRootPkg(pkg)) {
                    console.log(`[-] Blocked getApplicationInfoAsUser: ${pkg}`);
                    throwPkgNotFound();
                }
                return ov.apply(this, arguments);
            };
        });
    } catch(e) {}

    // 4b. Filter Installed Packages/Applications Lists (ALL overloads)
    const filterList = (result, label) => {
        if (!result) return result;
        
        const ArrayList = Java.use("java.util.ArrayList");
        const filtered = ArrayList.$new();
        const it = result.iterator();
        
        while (it.hasNext()) {
            const item = it.next();
            const pkg = extractPkgName(item);
            if (isRootPkg(pkg)) {
                console.log(`[-] Hidden from ${label}: ${pkg}`);
            } else {
                filtered.add(item);
            }
        }
        return filtered;
    };

    PackageManager.getInstalledPackages.overloads.forEach(function(ov) {
        ov.implementation = function() {
            return filterList(ov.apply(this, arguments), "getInstalledPackages");
        };
    });

    try {
        PackageManager.getInstalledPackagesAsUser.overloads.forEach(function(ov) {
            ov.implementation = function() {
                return filterList(ov.apply(this, arguments), "getInstalledPackagesAsUser");
            };
        });
    } catch(e) {}

    PackageManager.getInstalledApplications.overloads.forEach(function(ov) {
        ov.implementation = function() {
            return filterList(ov.apply(this, arguments), "getInstalledApplications");
        };
    });

    try {
        PackageManager.getInstalledApplicationsAsUser.overloads.forEach(function(ov) {
            ov.implementation = function() {
                return filterList(ov.apply(this, arguments), "getInstalledApplicationsAsUser");
            };
        });
    } catch(e) {}

    // 4c. getLaunchIntentForPackage
    try {
        PackageManager.getLaunchIntentForPackage.implementation = function (pkg) {
            if (isRootPkg(pkg)) {
                console.log(`[-] Blocked getLaunchIntentForPackage: ${pkg}`);
                return null;
            }
            return this.getLaunchIntentForPackage(pkg);
        };
    } catch (e) {}

    // 4d. queryIntentActivities
    PackageManager.queryIntentActivities.overloads.forEach(function (ov) {
        ov.implementation = function () {
            const result = ov.apply(this, arguments);
            if (!result) return result;

            const ArrayList = Java.use("java.util.ArrayList");
            const filtered = ArrayList.$new();
            const it = result.iterator();
            
            while (it.hasNext()) {
                const ri = it.next();
                let pkg = "";
                if (ri.activityInfo.value) pkg = extractPkgName(ri.activityInfo.value);
                else if (ri.serviceInfo.value) pkg = extractPkgName(ri.serviceInfo.value);
                else if (ri.providerInfo.value) pkg = extractPkgName(ri.providerInfo.value);
                
                if (isRootPkg(pkg)) {
                    console.log(`[-] Hidden intent result: ${pkg}`);
                } else {
                    filtered.add(ri);
                }
            }
            return filtered;
        };
    });

    // 4e. resolveActivity
    PackageManager.resolveActivity.overloads.forEach(function (ov) {
        ov.implementation = function () {
            const r = ov.apply(this, arguments);
            if (r && r.activityInfo.value) {
                const pkg = extractPkgName(r.activityInfo.value);
                if (isRootPkg(pkg)) {
                    console.log(`[-] Hidden resolved activity: ${pkg}`);
                    return null;
                }
            }
            return r;
        };
    });

    console.log("[+] Root bypass hooks successfully injected.");
});
```
