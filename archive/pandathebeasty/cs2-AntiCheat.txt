using CounterStrikeSharp.API.Core.Attributes;
using CounterStrikeSharp.API.Core;
using CS2AntiCheat.Handlers;
using CS2AntiCheat.Integration;
using Microsoft.Extensions.Logging;

namespace CS2AntiCheat.Core
{
    [MinimumApiVersion(374)]
    public class ACCore : BasePlugin
    {
        public override string ModuleName => "CS2-AntiCheat";
        public override string ModuleVersion => "1.0.0";
        public override string ModuleAuthor => "panda";
        public override string ModuleDescription => "Anti-Cheat for CS2";

        public override void Unload(bool hotReload)
        {
            foreach (var module in Globals.Modules)
            {
                try { module.OnUnload(); }
                catch (Exception e) { Globals.Log($"Exception in {module.Name}.OnUnload -> {e.Message}"); }
            }
        }

        public override void Load(bool hotReload)
        {
            // Always-on load logging (goes to the CounterStrikeSharp log file in addons/.../logs/,
            // independent of the General.json "Debug" switch which only gates verbose runtime logs).
            Logger.LogInformation("Loading {Name} v{Version} (hotReload: {HotReload})", ModuleName, ModuleVersion, hotReload);

            Globals.PreInit(this, Logger);
            Globals.Log($"Loading (hotReload: {hotReload})");

            if (hotReload == true)
            {
                Globals.Initialize(hotReload);
            }

            CommandHandler.Initialize(this);

            EventListeners.Initialize(this);
            EventHandlers.Initialize(this, hotReload);

            UserMessagesHandler.Initialize(this);

            GeneralConfig.Initialize();
            Globals.RefreshServerInfo();
            DiscordIntegration.InitializeWebhook();

            // Always-on: written to the CSSharp log file regardless of the Debug switch. The per-module
            // summary is logged from Globals.Initialize (which runs on map start / hot reload, once the
            // modules actually exist).
            Logger.LogInformation("Loaded {Name} v{Version}", ModuleName, ModuleVersion);
        }
    }
}

using System.Text.Json;

namespace CS2AntiCheat.Core
{
    internal class BaseConfig
    {
        internal static readonly JsonSerializerOptions JsonSerializerOptions = new JsonSerializerOptions()
        {
            WriteIndented = true
        };
    }

    internal class BaseConfig<T> where T : new()
    {
        internal T Config { get; private set; }
        protected readonly string configPath = string.Empty;

        internal BaseConfig(string path)
        {
            string folderPath = $"{Globals.GetModuleDirectory()}/Configs/";
            if (Directory.Exists(folderPath) == false)
            {
                Directory.CreateDirectory(folderPath);
            }

            configPath = $"{folderPath}{path}.json";
            Config = new T();

            Load();
        }

        internal bool Save()
        {
            string json = JsonSerializer.Serialize(Config, BaseConfig.JsonSerializerOptions);
            File.WriteAllText(configPath, json);

            return true;
        }

        internal bool Load()
        {
            if (File.Exists(configPath) == false)
            {
                Save();
                return false;
            }

            string json = File.ReadAllText(configPath);
            T? possibleConfig = JsonSerializer.Deserialize<T>(json, BaseConfig.JsonSerializerOptions);

            if (possibleConfig == null)
            {
                return false;
            }

            Config = possibleConfig;
            return true;
        }
    }
}

using CounterStrikeSharp.API;
using CounterStrikeSharp.API.ValveConstants.Protobuf;
using CS2AntiCheat.Detections;
using CS2AntiCheat.Integration;

namespace CS2AntiCheat.Core
{
    internal struct DetectionMetadata
    {
        internal BaseModule module;
        internal PlayerData player;
        internal DateTime time;
        internal string reason;
    }

    internal static class DetectionHandler
    {
        internal static void OnPlayerDetected(DetectionMetadata metadata)
        {
            string punishment;

            switch (metadata.module.ActionType)
            {
                case ActionType.None:
                {
                    return;
                }

                case ActionType.Log:
                {
                    Globals.Log($"{metadata.player.Controller.PlayerName} is suspected of using {metadata.module.Name} ({metadata.reason})");
                    punishment = "No punishment (log only).";
                    break;
                }

                case ActionType.Kick:
                {
                    Globals.Log($"{metadata.player.Controller.PlayerName} was kicked for using {metadata.module.Name} ({metadata.reason})");
                    metadata.player.Disconnect(NetworkDisconnectionReason.NETWORK_DISCONNECT_DISCONNECT_BY_SERVER);
                    punishment = "Player was kicked.";
                    break;
                }

                case ActionType.Ban:
                {
                    Globals.Log($"{metadata.player.Controller.PlayerName} was banned for using {metadata.module.Name} ({metadata.reason})");
                    BanPlayerWithCommands(metadata);
                    punishment = "Player was banned.";
                    break;
                }

                default:
                {
                    return;
                }
            }

            // Discord notification, gated by the global DiscordAlerts switch (General.json).
            if (GeneralConfig.GetGeneralConfig().Config.DiscordAlerts == true)
            {
                NotifyDiscord(metadata, punishment);
            }
        }

        /// <summary>
        /// Runs the configurable ban + kick commands (in that order) on the server console. Both
        /// come from General.json and support placeholders; an empty template skips that step.
        /// </summary>
        private static void BanPlayerWithCommands(DetectionMetadata metadata)
        {
            var cfg = GeneralConfig.GetGeneralConfig().Config;
            var banCommand = BuildCommand(cfg.BanCommand, metadata);
            var kickCommand = BuildCommand(cfg.KickCommand, metadata);

            // Execute on the main thread/next frame (detections can fire off the main thread).
            Server.NextFrame(() =>
            {
                if (string.IsNullOrWhiteSpace(banCommand) == false)
                {
                    Globals.Log($"Executing ban command: {banCommand}");
                    Server.ExecuteCommand(banCommand);
                }

                if (string.IsNullOrWhiteSpace(kickCommand) == false)
                {
                    Globals.Log($"Executing kick command: {kickCommand}");
                    Server.ExecuteCommand(kickCommand);
                }
            });
        }

        /// <summary>Fills the configurable command template with this detection's player data.</summary>
        private static string BuildCommand(string template, DetectionMetadata metadata)
        {
            if (string.IsNullOrWhiteSpace(template))
            {
                return string.Empty;
            }

            var controller = metadata.player.Controller;
            var userId = controller.UserId?.ToString() ?? "0";
            var steamId64 = controller.AuthorizedSteamID?.SteamId64.ToString() ?? "0";
            var steamId2 = controller.AuthorizedSteamID?.SteamId2 ?? metadata.player.SteamID;

            return template
                .Replace("{id}", userId)
                .Replace("{steamid}", steamId64)
                .Replace("{steamid2}", steamId2)
                .Replace("{name}", controller.PlayerName)
                .Replace("{reason}", metadata.reason)
                .Replace("{module}", metadata.module.Name);
        }

        private static void NotifyDiscord(DetectionMetadata metadata, string punishment)
        {
            var controller = metadata.player.Controller;
            var steamId64 = controller.AuthorizedSteamID?.SteamId64.ToString() ?? "0";

            var report = new DetectionReport
            {
                PlayerName = controller.PlayerName,
                SteamId64 = steamId64,
                ProfileUrl = $"https://steamcommunity.com/profiles/{steamId64}",
                Detection = metadata.module.Name,
                Evidence = metadata.reason,
                Punishment = punishment,
                Map = Server.MapName,
                Address = Globals.ServerAddress,
                ServerName = Globals.ServerName
            };

            DiscordIntegration.SendDetection(report);
        }
    }
}

using CounterStrikeSharp.API.Core;
using CounterStrikeSharp.API.Modules.Admin;
using CounterStrikeSharp.API.Modules.Commands;
using CS2AntiCheat.Handlers;

namespace CS2AntiCheat.Core
{
    public class GeneralConfigData
    {
        // Master on/off switch for the whole anti-cheat. When false, no detection runs (player
        // tracking still works, so toggling with tbac_reload takes effect immediately).
        public bool Enabled { get; set; } = true;

        // When true, the plugin prints all console log lines (init messages, detection logs, debug
        // info). When false, the console stays silent - Discord alerts still work independently.
        public bool Debug { get; set; } = false;

        // Prefix prepended to every console log line (e.g. "[CS2AC] Loaded ..."). Change it to
        // whatever tag you want to grep for in the server console. A trailing space is added
        // automatically, so leave it as just the tag. Leave empty for no prefix.
        public string LogPrefix { get; set; } = "[CS2AC]";

        // Permission flag required to run the plugin's admin commands (tbac_reload, tbac_*_action,
        // tbac_aimbot_angle, ...). The server console always bypasses this.
        public string CommandPermission { get; set; } = "@css/root";

        // Commands run (on the server console) when a module's action is Ban. They run in order:
        // BanCommand first, then KickCommand. Both are optional - leave empty to skip that step.
        // Placeholders: {id} = userid (use #{id} to target by userid), {steamid} = SteamID64,
        //               {steamid2} = STEAM_0:..., {name} = player name, {reason} = detection reason,
        //               {module} = detection module name.
        public string BanCommand { get; set; } = "css_ban #{id} 0 \"ANTICHEAT: {reason}\"";
        public string KickCommand { get; set; } = "css_kick #{id}";

        // Global switch for Discord detection alerts. When false, no detection sends to Discord
        // regardless of module. (A webhook URL is still required for alerts to work when true.)
        public bool DiscordAlerts { get; set; } = true;

        // Full Discord webhook URL, e.g. https://discord.com/api/webhooks/<id>/<token>
        // Leave empty to disable Discord notifications.
        public string DiscordWebhookUrl { get; set; } = string.Empty;

        // Who to ping on a detection message. Options:
        //   ""           = no ping
        //   "everyone"   = pings @everyone
        //   "here"       = pings @here
        //   "<role id>"  = pings that role (numeric Discord role id, e.g. "123456789012345678")
        public string DiscordMentionRoleId { get; set; } = string.Empty;
    }

    internal static class GeneralConfig
    {
        private static BaseConfig<GeneralConfigData> config = null!;

        public static void Initialize()
        {
            LoadConfig();

            CommandHandler.RegisterCommand("tbac_reload", "Reloads the config", OnReloadCommand);
        }

        public static BaseConfig<GeneralConfigData> GetGeneralConfig()
        {
            return config;
        }

        private static void LoadConfig()
        {
            config = new BaseConfig<GeneralConfigData>("General");

            // Mirror the switches into Globals so the hot paths can read them cheaply.
            Globals.Enabled = config.Config.Enabled;
            Globals.DebugLogging = config.Config.Debug;
            Globals.LogPrefix = config.Config.LogPrefix;

            Globals.Log($"Loaded general config (anti-cheat {(Globals.Enabled ? "ENABLED" : "DISABLED")})");
        }

        // ----- Commands ----- \\

        private static void OnReloadCommand(CCSPlayerController? player, CommandInfo command)
        {
            if (CommandHandler.HasPermission(player) == false)
            {
                return;
            }

            LoadConfig();
        }
    }
}

using CounterStrikeSharp.API;
using CounterStrikeSharp.API.Core;
using CounterStrikeSharp.API.Modules.Cvars;
using CounterStrikeSharp.API.ValveConstants.Protobuf;
using Microsoft.Extensions.Logging;
using CS2AntiCheat.Detections;
using CS2AntiCheat.Detections.Modules;

namespace CS2AntiCheat.Core
{
    internal class PlayerData
    {
        internal required CCSPlayerController Controller;
        internal required CCSPlayerPawn Pawn;

        internal required int Index;
        internal required bool IsBot;

        internal string PlayerName => Controller.PlayerName;
        internal string SteamID => Controller.AuthorizedSteamID?.SteamId2 ?? "Invalid SteamID";

        internal CCSWeaponBaseGun GetWeapon()
        {
            CPlayer_WeaponServices? weaponServices = Pawn.WeaponServices;
            if (weaponServices == null)
            {
                return null!;
            }

            CBasePlayerWeapon? weaponBase = weaponServices.ActiveWeapon.Value;
            if (weaponBase == null)
            {
                return null!;
            }

            return new CCSWeaponBaseGun(weaponBase.Handle);
        }

        internal void Disconnect(NetworkDisconnectionReason reason)
        {
            Controller.Disconnect(reason);
        }
    }

    internal static class Globals
    {
        private static bool initializedOnce = false;

        private static ACCore? pluginCore = null;
        private static ILogger? logger = null;

        // Master on/off switch, driven by General.json "Enabled". Checked by the detection dispatch
        // so the whole anti-cheat can be turned off from config (reload with tbac_reload).
        internal static bool Enabled = true;

        // Gates all console logging. Driven by General.json "Debug". Off until the config loads, so
        // nothing prints before we know the setting.
        internal static bool DebugLogging = false;

        // Prefix prepended to every console log line. Driven by General.json "LogPrefix". Defaults to
        // "[CS2AC]" so early init lines (before the config loads) are still tagged consistently.
        internal static string LogPrefix = "[CS2AC]";

        // Server info for Discord embeds, cached on the main thread (ConVar reads are main-thread only).
        internal static string ServerName = "CS2 Server";
        internal static string ServerAddress = "unknown";

        /// <summary>Refreshes the cached hostname + ip:port. Call on the main thread (map start/load).</summary>
        internal static void RefreshServerInfo()
        {
            try
            {
                var hostname = ConVar.Find("hostname")?.StringValue;
                if (string.IsNullOrWhiteSpace(hostname) == false)
                {
                    ServerName = hostname;
                }

                var ip = ConVar.Find("ip")?.StringValue;
                var port = ConVar.Find("hostport")?.GetPrimitiveValue<int>();
                if (string.IsNullOrWhiteSpace(ip) == false && ip != "0.0.0.0" && port is > 0)
                {
                    ServerAddress = $"{ip}:{port}";
                }
            }
            catch
            {
                // keep previous values on failure
            }
        }

        internal static PlayerData[] Players = [];
        internal static BaseModule[] Modules = [];

        internal static void PreInit(ACCore core, ILogger log)
        {
            pluginCore = core;
            logger = log;
        }

        internal static void Initialize(bool forceReinitialize)
        {
            Players = new PlayerData[Server.MaxPlayers];
            if (initializedOnce == true && forceReinitialize == false)
            {
                return;
            }

            Log($"Globals Initializing (forced: {forceReinitialize})");

            Modules =
            [
                new Aimbot(),
                new Backtrack(),
                new BunnyHop(),
                new RapidFire(),
                new UntrustedAngles(),
                new ImpossibleYaw(),
                new DuckSpam(),
                new UserCmdRate(),
                new NoRecoil(),
                new AntiDLLModule()
            ];

            initializedOnce = true;

            // Always-on: record which detection modules were loaded (written to the CSSharp log file).
            LogInfo($"Initialized {Modules.Length} detection module(s): {string.Join(", ", Modules.Select(m => m.Name))}");
        }

        internal static string GetModuleDirectory()
        {
            if (pluginCore == null)
            {
                return string.Empty;
            }

            return pluginCore.ModuleDirectory;
        }

        internal static void Log(string message)
        {
            if (DebugLogging == false)
            {
                return;
            }

            var line = string.IsNullOrEmpty(LogPrefix) ? message : $"{LogPrefix} {message}";
            logger?.Log(LogLevel.Information, line);
        }

        /// <summary>
        /// Always-on logging: writes to the CounterStrikeSharp log file (addons/.../logs/) regardless
        /// of the "Debug" switch. Use for load/init milestones that should always be recorded.
        /// </summary>
        internal static void LogInfo(string message)
        {
            var line = string.IsNullOrEmpty(LogPrefix) ? message : $"{LogPrefix} {message}";
            logger?.Log(LogLevel.Information, line);
        }
    }
}

        {
            if (player.IsBot == true)
            {
                return;
            }

            playerData[player.Index] = new PlayerAimbotData()
            {
                eyeAngleHistory = new Vector3[aimbotMaxHistory],
                historyIndex = 0,

                detections = 0
            };
        }

        internal override void OnPlayerDead(PlayerData victim, PlayerData shooter)
        {
            if (config.Config.DetectionEnabled == false)
            {
                return;
            }

            if (shooter.IsBot == true)
            {
                return;
            }

            PlayerAimbotData aimbotData = playerData[shooter.Index];

            int historyIndex = aimbotData.historyIndex;
            Vector3 lastAngle = aimbotData.eyeAngleHistory[historyIndex];

            float maxAngle = config.Config.MaxAimbotAngle;

            historyIndex++;
            for (int i = historyIndex; i < aimbotMaxHistory; i++)
            {
                if (historyIndex >= aimbotMaxHistory)
                {
                    historyIndex = 0;
                }

                Vector3 currentAngle = aimbotData.eyeAngleHistory[historyIndex];
                float angleDiff = Distance(lastAngle, currentAngle);

                //Normalize the angle so we can use it for our aimbot detection logic
                if (angleDiff > 180f)
                {
                    angleDiff = MathF.Abs(angleDiff - 360);
                }

                //Server.PrintToChatAll($"{i}: {shooter.Controller.PlayerName} -> {angleDiff}");

                if (angleDiff > maxAngle)
                {
                    OnAimbotDetected(shooter, aimbotData, angleDiff);
                    break;
                }

                lastAngle = currentAngle;
                historyIndex++;
            }
        }

            {
                return;

            }
        }

            {

                {
                    continue;
                }

                if (player.IsBot == true)

                    continue;
                }

            }
        }

                return;
            }

            OnPlayerDetected(player, reason);
        }

    /*
     * Module: AntiDLL (self-contained port of KillStr3aK/CS2-AntiDLL)
     * Purpose: Detect cheat DLLs by the game-event listeners they register. Cheats subscribe to
     *          engine events that legitimate clients never listen to; when a client registers a
     *          listener for any event in the Blacklist, that's the detection.
     *
     * HOW: hooks two native functions via CounterStrikeSharp dynamic functions -
     *        CGameEventManager_Init                                 (grab the event manager)
     *        CSource1LegacyGameEventGameSystem::ListenBitsReceived  (a client sent its listen bits)
     *      When the manager is captured, each blacklisted event NAME is resolved to its listen-bit
     *      index (CGameEventManager::GetEventDescriptor) and OR-ed into a 512-bit blacklist mask.
     *      At runtime we read the client's listen-bits straight from the CLCMsg_ListenEvents message
     *      (CUtlVector<uint32> at CLCMsg_ListenEvents::EventArray) and AND them with the mask - any
     *      set bit is a blacklisted event the client subscribed to. No proxy-listener walk, no
     *      per-event FindListener call.
     *
     * The native signatures/offsets live in gamedata/CS2-AntiCheat.AntiDLL.json (editable, no recompile).
     * The event blacklist lives in Configs/AntiDLL.json (editable, no recompile).
     */
    internal class AntiDLLModule : BaseModule
    {
        internal override string Name => "AntiDLL";
        internal override ActionType ActionType => config.Config.DetectionAction;

        private readonly BaseConfig<AntiDLLSaveData> config;
        private IGameEventManager2? gameEventManager;
        private bool hooked;

        // Precomputed blacklist as listen-bit words (16 x uint32 = 512 bits). Built once from the
        // event names when the event manager is captured: for each blacklisted name we resolve its
        // bit index and set that bit. At runtime we just AND the client's listen-bits with this mask.
        private readonly uint[] blacklistMask = new uint[16];
        private bool maskBuilt;

        // Maps a bit index back to the event name(s) it was built from, for readable detection logs.
        private readonly Dictionary<int, string> bitToName = new();

        internal AntiDLLModule() : base()
        {
            config = new BaseConfig<AntiDLLSaveData>("AntiDLL");

            CommandHandler.RegisterCommand("tbac_antidll_action", "Which action to take on the player. 0 = none | 1 = log | 2 = kick | 3 = ban", OnActionCommand);

            TryHook();

            Globals.Log($"AntiDLL Initialized (blacklist: {config.Config.Blacklist.Count} events)");
        }

        /// <summary>
        /// Installs the native hooks. Wrapped in try/catch: if the gamedata signatures are missing or
        /// stale for the current CS2 build, the module stays idle instead of crashing the plugin.
        /// </summary>
        private void TryHook()
        {
            if (hooked == true || config.Config.DetectionEnabled == false)
            {
                return;
            }

            // Bit-parsing path is verified for both Windows (server.dll) and Linux (libserver.so):
            // the message layout (CLCMsg_ListenEvents::EventCount/EventData) and the name->bit resolver
            // (CGameEventManager::GetEventDescriptor) have per-platform signatures/offsets in gamedata.
            try
            {
                IGameEventManager2.Init.Hook(OnGameEventManagerInit, HookMode.Pre);
                CSource1LegacyGameEventGameSystem.ListenBitsReceived.Hook(OnListenBitsReceived, HookMode.Pre);
                hooked = true;
            }
            catch (Exception e)
            {
                Globals.Log($"AntiDLL could not install hooks (bad/missing gamedata signatures?): {e.Message}");
            }
        }

        private HookResult OnGameEventManagerInit(DynamicHook hook)
        {
            gameEventManager = hook.GetParam<IGameEventManager2>(0);
            BuildBlacklistMask();
            return HookResult.Continue;
        }

        /// <summary>
        /// Translates every blacklisted event name into its listen-bit index (via the event manager)
        /// and sets that bit in blacklistMask. Runs once, right after the manager is available. If an
        /// event name is unknown to this build, it is skipped (logged), not fatal.
        /// </summary>
        private void BuildBlacklistMask()
        {
            if (maskBuilt == true || gameEventManager == null)
            {
                return;
            }

            Array.Clear(blacklistMask);
            bitToName.Clear();

            int resolved = 0;
            foreach (var name in config.Config.Blacklist)
            {
                int bit = gameEventManager.GetEventBitIndex(name);
                if (bit < 0 || bit >= 512)
                {
                    continue; // unknown event on this build - skip
                }

                blacklistMask[bit >> 5] |= 1u << (bit & 31);
                bitToName[bit] = name;
                resolved++;
            }

            maskBuilt = true;
            Globals.LogInfo($"AntiDLL blacklist mask built ({resolved}/{config.Config.Blacklist.Count} events resolved to bits)");
        }

        private HookResult OnListenBitsReceived(DynamicHook hook)
        {
            if (config.Config.DetectionEnabled == false)
            {
                return HookResult.Continue;
            }

            // Make sure the mask is ready (manager may have been captured but mask not yet built).
            if (maskBuilt == false)
            {
                BuildBlacklistMask();
                if (maskBuilt == false)
                {
                    return HookResult.Continue;
                }
            }

            CLCMsg_ListenEvents msg = hook.GetParam<CLCMsg_ListenEvents>(1);
            CPlayerSlot slot = msg.GetPlayerSlot();

            CCSPlayerController? controller = Utilities.GetPlayerFromSlot(slot);
            if (controller == null || controller.IsValid == false || controller.IsBot == true)
            {
                return HookResult.Continue;
            }

            // Read the client's listen-bits straight from the message and AND against the blacklist.
            uint[] bits = msg.GetEventBits();
            if (bits.Length == 0)
            {
                return HookResult.Continue;
            }

            var hits = new List<string>();
            int words = Math.Min(bits.Length, blacklistMask.Length);
            for (int w = 0; w < words; w++)
            {
                uint matched = bits[w] & blacklistMask[w];
                while (matched != 0)
                {
                    int bitInWord = System.Numerics.BitOperations.TrailingZeroCount(matched);
                    matched &= matched - 1; // clear lowest set bit
                    int globalBit = (w << 5) + bitInWord;
                    hits.Add(bitToName.TryGetValue(globalBit, out var n) ? n : $"bit#{globalBit}");
                }
            }

            if (hits.Count == 0)
            {
                return HookResult.Continue;
            }

            int index = controller.Slot;
            if (index < 0 || index >= Globals.Players.Length)
            {
                return HookResult.Continue;
            }

            PlayerData? player = Globals.Players[index];
            if (player == null)
            {
                return HookResult.Continue;
            }

            OnPlayerDetected(player, $"AntiDLL -> blacklisted event listener(s): {string.Join(", ", hits)}");
            return HookResult.Continue;
        }

        internal override void OnUnload()
        {
            Shutdown();
        }

        internal void Shutdown()
        {
            if (hooked == false)
            {
                return;
            }

            try
            {
                CSource1LegacyGameEventGameSystem.ListenBitsReceived.Unhook(OnListenBitsReceived, HookMode.Pre);
                IGameEventManager2.Init.Unhook(OnGameEventManagerInit, HookMode.Pre);
            }
            catch { /* ignore */ }

            hooked = false;
        }

        // ----- Commands ----- \\

        private void OnActionCommand(CCSPlayerController? player, CommandInfo command)
        {
            if (CommandHandler.HasPermission(player) == false) return;
            if (command.ArgCount != 2) return;
            if (int.TryParse(command.ArgByIndex(1), out int action) == false) return;
            config.Config.DetectionAction = (ActionType)action;
            config.Save();
        }
    }
}

namespace CS2AntiCheat.AntiDLLSdk
{
    using CounterStrikeSharp.API;
    using CounterStrikeSharp.API.Core;

    using System.Runtime.InteropServices;

    /// <summary>
    /// Wrapper over the parsed CNetMessagePB&lt;CMsgSource1LegacyListenEvents&gt; a client sends to say
    /// which game events it wants to listen to.
    ///
    /// Layout (verified from CSource1LegacyGameEventGameSystem::ListenBitsReceived on both platforms):
    ///   [msg + 0x58]  player slot (int)
    ///   [msg + 0x48]  event-bit element count (int)
    ///   [msg + 0x50]  pointer to the uint32 word array holding the listen bits (up to 16 words = 512 bits)
    ///                 Bit N set == the client listens to the event with descriptor index N.
    /// On Windows these two offsets are the count+data members of a CUtlVector&lt;uint32&gt; that starts at
    /// 0x48 (count at 0x48, data ptr at 0x50). On Linux they are separate fields at the same offsets.
    /// Both offsets come from gamedata so they can be updated without recompiling.
    /// </summary>
    public class CLCMsg_ListenEvents : NativeObject
    {
        private static readonly int EventCountOffset = GameData.GetOffset("CLCMsg_ListenEvents::EventCount");
        private static readonly int EventDataOffset = GameData.GetOffset("CLCMsg_ListenEvents::EventData");

        private unsafe CPlayerSlot m_Slot => Marshal.PtrToStructure<CPlayerSlot>(base.Handle + 80);

        public CLCMsg_ListenEvents(nint ptr) : base(ptr)
            { }

        public CPlayerSlot GetPlayerSlot() => this.m_Slot;

        /// <summary>
        /// Reads the listen-bits vector from the message and returns it as up to 16 uint32 words
        /// (512 bits). A missing/empty vector returns an empty array. Never throws on bad data.
        /// </summary>
        public uint[] GetEventBits()
        {
            try
            {
                int count = Marshal.ReadInt32(base.Handle + EventCountOffset);
                if (count <= 0 || count > 16)
                {
                    return [];
                }

                nint data = (nint)Marshal.ReadInt64(base.Handle + EventDataOffset);
                if (data == nint.Zero)
                {
                    return [];
                }

                var words = new uint[count];
                for (int i = 0; i < count; i++)
                {
                    words[i] = unchecked((uint)Marshal.ReadInt32(data + (i * 4)));
                }
                return words;
            }
            catch
            {
                return [];
            }
        }
    }
}

    using CounterStrikeSharp.API;
    using CounterStrikeSharp.API.Core;
    using CounterStrikeSharp.API.Modules.Memory.DynamicFunctions;

    /// <summary>
    /// The game system that receives a client's "listen bits" message. We only hook
    /// ListenBitsReceived to intercept the CLCMsg_ListenEvents; detection reads the event-bit vector
    /// straight from that message and matches it against a precomputed blacklist mask, so the old
    /// Listeners offset / proxy-listener walk is no longer needed.
    /// </summary>

    using CounterStrikeSharp.API.Modules.Memory;
    using CounterStrikeSharp.API.Modules.Memory.DynamicFunctions;

    using System.Runtime.InteropServices;

    public class IGameEventManager2 : NativeObject
    {
        public static readonly MemoryFunctionVoid<IGameEventManager2> Init = new MemoryFunctionVoid<IGameEventManager2>(GameData.GetSignature("CGameEventManager_Init"));

        // bool GetEventDescriptor(CGameEventManager* this, const char* name, int* outBitIndex)
        // Returns a descriptor pointer (non-null = found) and writes the event's bit index into
        // *outBitIndex (the engine ORs in bit 31 as a flag, so mask with 0x7FFFFFFF).
        private static readonly MemoryFunctionWithReturn<nint, string, nint, nint> GetEventDescriptorFn =
            new MemoryFunctionWithReturn<nint, string, nint, nint>(GameData.GetSignature("CGameEventManager::GetEventDescriptor"));

        public IGameEventManager2(nint ptr) : base(ptr)
            { }

        /// <summary>
        /// Resolves an event name to its listen-bit index. Returns -1 if the event does not exist
        /// (or on any failure). The returned index is the bit position in the client's listen-bits
        /// vector (see CLCMsg_ListenEvents.GetEventBits).
        /// </summary>
        public int GetEventBitIndex(string eventName)
        {
            try
            {
                nint outIndex = Marshal.AllocHGlobal(sizeof(int));
                try
                {
                    Marshal.WriteInt32(outIndex, -1);
                    nint descriptor = GetEventDescriptorFn.Invoke(base.Handle, eventName, outIndex);
                    if (descriptor == nint.Zero)
                    {
                        return -1;
                    }

                    int raw = Marshal.ReadInt32(outIndex);
                    if (raw == -1)
                    {
                        return -1;
                    }

                    return raw & 0x7FFFFFFF; // strip the bit-31 flag the engine sets
                }
                finally
                {
                    Marshal.FreeHGlobal(outIndex);
                }
            }
            catch
            {
                return -1;
            }
        }
    }

        public int PingThresholdMs { get; set; } = 150;

        // kill from a player on a genuinely bad connection).
        public int SuspiciousKillsBeforeLog { get; set; } = 4;

     * Purpose: Flag kills made with abnormally high latency, which is the condition backtrack abuse
     *          depends on. This is NOT a reliable backtrack detector - server-side CS2 has no access
     *          to the lag-compensation hitbox history needed to prove backtracking. Treat hits as
     *          leads for manual demo review only.
     */

    {

        internal override ActionType ActionType => config.Config.DetectionAction;

        private readonly BacktrackData[] playerData;

            CommandHandler.RegisterCommand("tbac_backtrack_action", "Which action to take on the player. 0 = none | 1 = log | 2 = kick | 3 = ban", OnActionCommand);

            Globals.Log($"Backtrack Initialized (log-only investigation aid)");
        }

            {
                return;
            }

            playerData[player.Index] = new BacktrackData { suspiciousKills = 0 };

        internal override void OnPlayerDead(PlayerData victim, PlayerData shooter)
        {
            if (config.Config.DetectionEnabled == false || shooter.IsBot == true)
            {
                return;
            }

            // Self-kills / world deaths are not relevant.
            if (shooter.Index == victim.Index)
            {
                return;
            }

            if (data == null)
            {
                return;

                data.suspiciousKills = 0;
            }
        }

        {
            if (CommandHandler.HasPermission(player) == false) return;
            if (command.ArgCount != 2) return;
            if (int.TryParse(command.ArgByIndex(1), out int action) == false) return;
            config.Config.DetectionAction = (ActionType)action;
            config.Save();
        }

                if (data.detections > 0)
                {
                    data.detections--;
                }
            }

using CounterStrikeSharp.API;
using CounterStrikeSharp.API.Core;
using CounterStrikeSharp.API.Modules.Admin;
using CounterStrikeSharp.API.Modules.Commands;
using CS2AntiCheat.Core;
using CS2AntiCheat.Handlers;

namespace CS2AntiCheat.Detections.Modules
{
    public class NoRecoilSaveData
    {
        public bool DetectionEnabled { get; set; } = true;

        // Per your request this defaults to Ban. NOTE: server-side no-recoil detection is inherently
        // less certain than the physical-impossibility modules; consider running it on Log (1) first
        // and watching for false positives before leaving it on Ban (3).
        public ActionType DetectionAction { get; set; } = ActionType.Ban;

        // Minimum consecutive full-auto shots in one burst before the burst is evaluated. Long
        // bursts are where recoil compensation is unmistakable, so a high value avoids tap/burst FPs.
        public int MinBurstShots { get; set; } = 12;

        // Aim-punch magnitude (degrees) that must have accumulated over the burst for it to count as
        // "recoil was clearly applied". Below this the engine barely kicked, so we skip (no verdict).
        public float MinAccumulatedPunch { get; set; } = 4.0f;

        // Maximum total vertical view movement (degrees) across the burst for it to be considered
        // "not compensating". A legit player pulls down noticeably against spray; a no-recoil cheat
        // leaves the view almost still. Conservative so only near-zero compensation trips it.
        public float MaxPitchCompensation { get; set; } = 1.0f;

        public int MaxDetectionsBeforeAction { get; set; } = 2;
    }

    internal class NoRecoilData
    {
        internal int burstShots;
        internal float burstStartPitch;
        internal float lastPunch;
        internal float accumulatedPunch;
        internal int lastShotTick;
        internal int detections;
    }

    /*
     * Module: No Recoil (conservative)
     * Purpose: Flag sustained full-auto bursts where clear recoil (aim punch) was applied but the
     *          player's view barely moved to compensate - the signature of recoil removal.
     * CAUTION: This is the least certain module (unlike the pure-physics ones). Thresholds are
     *          deliberately conservative; prefer Log first on a live server.
     */
    internal class NoRecoil : BaseModule
    {
        internal override string Name => "NoRecoil";
        internal override ActionType ActionType => config.Config.DetectionAction;

        private const int BurstGapTicks = 16; // >~0.25s between shots ends the burst

        private readonly BaseConfig<NoRecoilSaveData> config;
        private readonly NoRecoilData[] playerData;

        internal NoRecoil() : base()
        {
            config = new BaseConfig<NoRecoilSaveData>("NoRecoil");
            playerData = new NoRecoilData[Server.MaxPlayers];

            CommandHandler.RegisterCommand("tbac_norecoil_action", "Which action to take on the player. 0 = none | 1 = log | 2 = kick | 3 = ban", OnActionCommand);

            Globals.Log($"NoRecoil Initialized");
        }

        internal override void OnPlayerJoin(PlayerData player)
        {
            if (player.IsBot == true)
            {
                return;
            }

            playerData[player.Index] = new NoRecoilData();
        }

        internal override void OnPlayerShoot(PlayerData player)
        {
            if (config.Config.DetectionEnabled == false || player.IsBot == true)
            {
                return;
            }

            NoRecoilData data = playerData[player.Index];
            if (data == null)
            {
                return;
            }

            int now = Server.TickCount;
            float pitch = player.Pawn.EyeAngles.X;
            float punch = GetPunchMagnitude(player);

            // A gap between shots ends the previous burst; start a fresh one.
            if (data.burstShots == 0 || now - data.lastShotTick > BurstGapTicks)
            {
                StartBurst(data, now, pitch, punch);
                return;
            }

            data.burstShots++;
            data.lastShotTick = now;

            // Accumulate how much recoil (aim punch) the engine applied during the burst.
            float punchDelta = MathF.Abs(punch - data.lastPunch);
            data.accumulatedPunch += punchDelta;
            data.lastPunch = punch;

            // Only evaluate once the burst is long enough to be unambiguous.
            if (data.burstShots < config.Config.MinBurstShots)
            {
                return;
            }

            // How much the player actually moved their view vertically across the burst.
            float pitchCompensation = MathF.Abs(pitch - data.burstStartPitch);

            bool recoilClearlyApplied = data.accumulatedPunch >= config.Config.MinAccumulatedPunch;
            bool noCompensation = pitchCompensation <= config.Config.MaxPitchCompensation;

            if (recoilClearlyApplied && noCompensation)
            {
                data.detections++;
                if (data.detections >= config.Config.MaxDetectionsBeforeAction)
                {
                    OnPlayerDetected(player,
                        $"NoRecoil -> burst {data.burstShots} shots, punch {data.accumulatedPunch:F2} deg, view moved {pitchCompensation:F2} deg");
                    data.detections = 0;
                }

                // Reset the burst so one long spray is counted once.
                StartBurst(data, now, pitch, punch);
            }
        }

        private static void StartBurst(NoRecoilData data, int tick, float pitch, float punch)
        {
            data.burstShots = 1;
            data.burstStartPitch = pitch;
            data.lastPunch = punch;
            data.accumulatedPunch = 0f;
            data.lastShotTick = tick;
        }

        /// <summary>Magnitude of the applied aim punch (recoil kick) in degrees, 0 if unavailable.</summary>
        private static float GetPunchMagnitude(PlayerData player)
        {
            var punchServices = player.Pawn.AimPunchServices;
            if (punchServices == null)
            {
                return 0f;
            }

            var angle = punchServices.PredictableBaseAngle;
            return MathF.Sqrt((angle.X * angle.X) + (angle.Y * angle.Y));
        }

        private void OnActionCommand(CCSPlayerController? player, CommandInfo command)
        {
            if (CommandHandler.HasPermission(player) == false) return;
            if (command.ArgCount != 2) return;
            if (int.TryParse(command.ArgByIndex(1), out int action) == false) return;
            config.Config.DetectionAction = (ActionType)action;
            config.Save();
        }
    }
}

            Globals.Log($"RapidFire Initialized");
        }

        internal override void OnPlayerJoin(PlayerData player)
        {
            if (player.IsBot == true)
            {
                return;
            }

            playerData[player.Index] = new RapidFireData()
            {
                lastBulletShotTick = 0,
                lastNextAttackTick = 0,
                rapidDetections = 0
            };
        }

        internal override void OnPlayerShoot(PlayerData player)
        {
            if (config.Config.DetectionEnabled == false)
            {
                return;
            }

            if (player.IsBot == true)
            {
                return;
            }

            CCSWeaponBaseGun weapon = player.GetWeapon();
            string weaponName = weapon.DesignerName;

            if (weaponName == "weapon_healthshot")
            {
                return;
            }

            int nextAttack;
            switch (weapon.WeaponMode)
            {
                case CSWeaponMode.Primary_Mode: nextAttack = weapon.NextPrimaryAttackTick; break;
                case CSWeaponMode.Secondary_Mode: nextAttack = weapon.NextSecondaryAttackTick; break;
                default: Globals.Log($"Unsupported WeaponMode: {(int)weapon.WeaponMode} on {weaponName}"); return;
            }

            int serverTickCount = Server.TickCount;
            RapidFireData data = playerData[player.Index];

            // Authoritative check: the engine sets NextPrimary/SecondaryAttackTick to the earliest
            // tick the weapon is ALLOWED to fire again. A shot landing strictly before that tick is
            // physically impossible without manipulating the fire rate. This is far more reliable
            // than the previous "two shots one tick apart" heuristic (which flagged legit fast guns).
            //
            // nextAttack was captured AFTER this shot, so it already points at the next allowed tick.
            // We compare the gap since the previous shot to how long the previous shot's cooldown
            // should have been. A conservative 1-tick tolerance absorbs rounding / subtick timing.
            int tickDiff = serverTickCount - data.lastBulletShotTick;

            // Reset the burst tracking after a long pause between shots.
            if (tickDiff > 64)
            {
                data.lastBulletShotTick = serverTickCount;
                data.lastNextAttackTick = nextAttack;
                data.rapidDetections = 0;
                return;
            }

            // If this shot occurred before the previously-promised next-allowed tick (with a 1-tick
            // tolerance), the weapon fired faster than the engine permits.
            bool firedTooEarly = data.lastNextAttackTick > 0 && serverTickCount < data.lastNextAttackTick - 1;

            if (firedTooEarly)
            {
                data.rapidDetections++;
                Globals.Log($"RapidFire suspect -> {player.Controller.PlayerName} | {weaponName} | shotTick {serverTickCount} < allowed {data.lastNextAttackTick} | gap {tickDiff}");
            }
            else if (data.rapidDetections > 0)
            {
                // Decay so an occasional timing blip never accumulates into a ban.
                data.rapidDetections--;
            }

            if (data.rapidDetections >= config.Config.MaxDetectionsBeforeAction)
            {
                OnPlayerDetected(player, $"RapidFire -> {weaponName} fired faster than engine cooldown ({data.rapidDetections} times)");
                data.rapidDetections = 0;
            }

            data.lastBulletShotTick = serverTickCount;
            data.lastNextAttackTick = nextAttack;
        }

        // ----- Commands ----- \\

        private void OnActionCommand(CCSPlayerController? player, CommandInfo command)
        {
            if (CommandHandler.HasPermission(player) == false)
            {
                return;
            }

            if (command.ArgCount != 2)
            {
                return;
            }

            string arg = command.ArgByIndex(1);
            if (int.TryParse(arg, out int action) == false)
            {
                return;
            }

            ActionType actionType = (ActionType)action;
            if (config.Config.DetectionAction.HasFlag(actionType) == false)
            {
                return;
            }

            config.Config.DetectionAction = actionType;
            config.Save();
        }
    }

```cs
using CounterStrikeSharp.API;
using CounterStrikeSharp.API.Core;
using CounterStrikeSharp.API.Modules.Admin;
using CounterStrikeSharp.API.Modules.Commands;
using CS2AntiCheat.Core;
using CS2AntiCheat.Handlers;

namespace CS2AntiCheat.Detections.Modules
{
    public class UserCmdRateSaveData
    {
        public bool DetectionEnabled { get; set; } = true;
        public ActionType DetectionAction { get; set; } = ActionType.Kick;

        // Max user commands a client may process per server tick on average over the window. The
        // engine normally processes ~1 usercmd per tick; sustained values well above that indicate
        // command flooding / choke exploits. 2.0 leaves generous headroom for legit jitter.
        public float MaxCommandsPerTick { get; set; } = 2.0f;

        // Sampling window in server ticks before evaluating the average.
        public int WindowTicks { get; set; } = 64;

        public int MaxDetectionsBeforeAction { get; set; } = 3;
    }

    internal class UserCmdRateData
    {
        internal uint lastCommandNumber;
        internal bool hasLast;
        internal int windowStartTick;
        internal uint windowStartCommand;
        internal int detections;
    }

    /*
     * Module: UserCmd Rate
     * Purpose: Flag clients processing far more user commands than ticks (command flooding).
     * Safe-ish: uses LastCommandNumberProcessed vs ticks elapsed; conservative threshold.
     */
    internal class UserCmdRate : BaseModule
    {
        internal override string Name => "UserCmdRate";
        internal override ActionType ActionType => config.Config.DetectionAction;

        private readonly BaseConfig<UserCmdRateSaveData> config;
        private readonly UserCmdRateData[] playerData;

        internal UserCmdRate() : base()
        {
            config = new BaseConfig<UserCmdRateSaveData>("UserCmdRate");
            playerData = new UserCmdRateData[Server.MaxPlayers];

            CommandHandler.RegisterCommand("tbac_usercmdrate_action", "Which action to take on the player. 0 = none | 1 = log | 2 = kick | 3 = ban", OnActionCommand);

            Globals.Log($"UserCmdRate Initialized");
        }

        internal override void OnPlayerJoin(PlayerData player)
        {
            if (player.IsBot == true)
            {
                return;
            }

            playerData[player.Index] = new UserCmdRateData { hasLast = false, detections = 0 };
        }

        internal override void OnPlayerTick(PlayerData player)
        {
            if (config.Config.DetectionEnabled == false || player.IsBot == true)
            {
                return;
            }

            UserCmdRateData data = playerData[player.Index];
            if (data == null)
            {
                return;
            }

            var movement = player.Pawn.MovementServices;
            if (movement == null)
            {
                return;
            }

            uint cmd = movement.LastCommandNumberProcessed;
            int now = Server.TickCount;

            if (data.hasLast == false)
            {
                data.hasLast = true;
                data.lastCommandNumber = cmd;
                data.windowStartTick = now;
                data.windowStartCommand = cmd;
                return;
            }

            data.lastCommandNumber = cmd;

            int ticksElapsed = now - data.windowStartTick;
            if (ticksElapsed < config.Config.WindowTicks)
            {
                return;
            }

            // Commands processed across the window vs ticks that passed.
            long commandsProcessed = (long)cmd - data.windowStartCommand;
            data.windowStartTick = now;
            data.windowStartCommand = cmd;

            // Ignore nonsensical windows (counter reset on respawn/reconnect).
            if (commandsProcessed <= 0 || ticksElapsed <= 0)
            {
                return;
            }

            float perTick = (float)commandsProcessed / ticksElapsed;
            if (perTick > config.Config.MaxCommandsPerTick)
            {
                data.detections++;
                if (data.detections >= config.Config.MaxDetectionsBeforeAction)
                {
                    OnPlayerDetected(player, $"UserCmdRate -> {perTick:F2} cmds/tick over {ticksElapsed} ticks (limit {config.Config.MaxCommandsPerTick})");
                    data.detections = 0;
                }
            }
            else if (data.detections > 0)
            {
                data.detections--;
            }
        }

        private void OnActionCommand(CCSPlayerController? player, CommandInfo command)
        {
            if (CommandHandler.HasPermission(player) == false) return;
            if (command.ArgCount != 2) return;
            if (int.TryParse(command.ArgByIndex(1), out int action) == false) return;
            config.Config.DetectionAction = (ActionType)action;
            config.Save();
        }
    }
}

using CounterStrikeSharp.API.Core;

namespace CS2AntiCheat.Handlers
{
    internal static class CommandHandler
    {
        private static BasePlugin? plugin;

        /// <summary>
        /// True when the caller may run an admin command. The server console (null) always passes;
        /// an in-game player must hold the configurable permission (General.json CommandPermission,
        /// default @css/root). Replies are left to the caller.

                return false;
            }

            string flag = GeneralConfig.GetGeneralConfig().Config.CommandPermission;
            if (string.IsNullOrWhiteSpace(flag))
            {
                return true; // no restriction configured
            }

            return AdminManager.PlayerHasPermissions(player, flag);
        }

        internal static void Initialize(BasePlugin basePlugin)
        {
            plugin = basePlugin;

            Globals.Log($"CommandHandler Initialized");
        }

        internal static bool RegisterCommand(string command, string description, CommandInfo.CommandCallback handler)
        {
            if (plugin == null)
            {
                return false;
            }

            if (string.IsNullOrEmpty(command) == true)
            {
                return false;
            }

            if (handler == null)
            {
                return false;
            }

            plugin.AddCommand(command, description, handler);
            return true;
        }
    }

                {
                    CCSPlayerController? controller = Utilities.GetPlayerFromSlot(i);
                    if (controller == null || controller.IsValid == false)
                    {

                    }

                    if (controller.Connected != PlayerConnectedState.Connected)

                    }

                }
            }

        private static HookResult OnPlayerActivate(EventPlayerActivate activateEvent, GameEventInfo _)
        {
            CCSPlayerController? controller = activateEvent.Userid;
            if (controller == null || controller.IsValid == false)
            {
                return HookResult.Continue;
            }

            // Normal players are already getting handled correctly. No need to do that here
            if (controller.IsBot == false)
            {
                return HookResult.Continue;
            }

            // Skip out on SourceTV since we don't want to track a "player" like that
            if (controller.IsHLTV == true)
            {
                return HookResult.Continue;
            }

            //Globals.Log($"Bot activated -> {controller.Slot} | {controller.PlayerName}");
            OnPlayerJoined(controller);

            return HookResult.Continue;
        }

        private static HookResult OnPlayerConnectFull(EventPlayerConnectFull connectEvent, GameEventInfo _)
        {
            OnPlayerJoined(connectEvent.Userid);

            return HookResult.Continue;
        }

        private static HookResult OnPlayerDisconnect(EventPlayerDisconnect disconnectEvent, GameEventInfo _)
        {
            OnPlayerLeft(disconnectEvent.Userid);

            return HookResult.Continue;
        }

        private static HookResult OnPlayerJump(EventPlayerJump jumpEvent, GameEventInfo _)
        {
            CCSPlayerController? controller = jumpEvent.Userid;
            if (controller == null || controller.IsValid == false)
            {
                return HookResult.Continue;
            }

            PlayerData player = Globals.Players[controller.Slot];
            BaseCaller.OnPlayerJump(player);

            return HookResult.Continue;
        }

        private static HookResult OnPlayerHurt(EventPlayerHurt hurtEvent, GameEventInfo _)
        {
            CCSPlayerController? victimController = hurtEvent.Userid;
            CCSPlayerController? shooterController = hurtEvent.Attacker;

            if (victimController == null || victimController.IsValid == false ||
                shooterController == null || shooterController.IsValid == false)
            {
                return HookResult.Continue;
            }

            PlayerData victim = Globals.Players[victimController.Slot];
            PlayerData shooter = Globals.Players[shooterController.Slot];

            BaseCaller.OnPlayerHurt(victim, shooter, (HitGroup_t)hurtEvent.Hitgroup);

            return HookResult.Continue;
        }

        private static HookResult OnPlayerDeath(EventPlayerDeath deathEvent, GameEventInfo _)
        {

            CCSPlayerController? shooterController = deathEvent.Attacker;

                shooterController == null || shooterController.IsValid == false)
            {
                return HookResult.Continue;
            }

            PlayerData victim = Globals.Players[victimController.Slot];
            PlayerData shooter = Globals.Players[shooterController.Slot];

            if (victim == null || shooter == null)
            {
                return HookResult.Continue;
            }

            BaseCaller.OnPlayerDead(victim, shooter);

            return HookResult.Continue;
        }

            BaseCaller.OnPlayerShoot(shooter);

            return HookResult.Continue;
        }

        {
            BaseCaller.OnRoundStart();

            return HookResult.Continue;
        }

        private static HookResult OnRoundEnd(EventRoundEnd roundEndEvent, GameEventInfo _)
        {
            BaseCaller.OnRoundEnd();

            return HookResult.Continue;
        }

        private static void OnPlayerJoined(CCSPlayerController? controller)
        {
            if (controller == null || controller.IsValid == false)
            {
                Globals.Log($"WARNING: Controller is invalid when player joined");
                return;
            }

            CCSPlayerPawn? pawn = controller.PlayerPawn.Value;
            if (pawn == null)
            {
                Globals.Log($"WARNING: Pawn is invalid when player ({controller.PlayerName}) joined");
                return;
            }

            int playerIndex = controller.Slot;
            PlayerData player = new PlayerData()
            {
                Controller = controller,
                Pawn = pawn,

                Index = playerIndex,
                IsBot = controller.IsBot
            };

            Globals.Players[playerIndex] = player;
            BaseCaller.OnPlayerJoin(player);

            //Globals.Log($"Player joined -> {playerIndex} | {controller.PlayerName}");
        }

        private static void OnPlayerLeft(CCSPlayerController? controller)
        {
            if (controller == null || controller.IsValid == false)
            {
                Globals.Log($"WARNING: Controller is invalid when player left");
                return;
            }

            int playerIndex = controller.Slot;
            if (Globals.Players[playerIndex] == null)
            {
                return;
            }

            CCSPlayerPawn? pawn = controller.PlayerPawn.Value;
            if (pawn == null)
            {
                Globals.Log($"WARNING: Pawn is invalid when player ({controller.PlayerName}) left");
            }

            PlayerData player = Globals.Players[playerIndex];

            BaseCaller.OnPlayerLeave(player);
            Globals.Players[playerIndex] = null!;

            //Globals.Log($"Player left -> {playerIndex} | {controller.PlayerName}");
        }
    }

using CS2AntiCheat.Core;
using System.Text.Json;
using System.Text.Json.Serialization;

namespace CS2AntiCheat.Integration
{
    /// <summary>All data needed to render one detection embed. Filled on the main thread.</summary>
    internal struct DetectionReport
    {
        internal string PlayerName;
        internal string SteamId64;
        internal string ProfileUrl;
        internal string Detection;   // module name
        internal string Evidence;    // reason
        internal string Punishment;  // human text for the action taken
        internal string Map;
        internal string Address;     // ip:port

    internal class DiscordIntegration
    {
        // ----- Webhook payload shapes (snake_case to match the Discord API) -----
#pragma warning disable IDE1006 // Naming Styles
        private class WebhookData
        {
            public string username { get; set; } = "CS2AC";
            public string? content { get; set; }
            public AllowedMentions allowed_mentions { get; set; } = new();
            public List<Embed>? embeds { get; set; }
        }

        private class AllowedMentions
        {
            public List<string> parse { get; set; } = new();
            public List<string> roles { get; set; } = new();
        }

        private class Embed
        {
            public EmbedAuthor? author { get; set; }
            public int color { get; set; }
            public List<EmbedField> fields { get; set; } = new();
            public EmbedThumbnail? thumbnail { get; set; }
            public EmbedFooter? footer { get; set; }
            public string? timestamp { get; set; }
        }

        private class EmbedAuthor
        {
            public string name { get; set; } = string.Empty;
        }

        private class EmbedField
        {
            public string name { get; set; } = string.Empty;
            public string value { get; set; } = string.Empty;
            [JsonPropertyName("inline")] public bool inlineField { get; set; }
        }

        private class EmbedThumbnail
        {
            public string url { get; set; } = string.Empty;
        }

        private class EmbedFooter
        {
            public string text { get; set; } = string.Empty;
        }
#pragma warning restore IDE1006 // Naming Styles

        private const int EmbedColorRed = 0xE02424; // red bar on the left of the embed

        private static HttpClient webClient = null!;
        private static string webClientAddress = string.Empty;

        private static string mentionPrefix = string.Empty;
        private static AllowedMentions allowedMentions = new();

        internal static bool InitializeWebhook()
        {
            string url = GeneralConfig.GetGeneralConfig().Config.DiscordWebhookUrl;

            if (string.IsNullOrWhiteSpace(url) == true)
            {
                Globals.Log("Unable to initialize Webhook ('DiscordWebhookUrl' not set)");
                return false;
            }

            webClient ??= new HttpClient();
            webClientAddress = url.Trim();

            BuildMention();
            return true;
        }

        /// <summary>
        /// Builds the ping prefix + allowed_mentions from DiscordMentionRoleId:
        ///   "" -> no ping; "everyone" -> @everyone; "here" -> @here; "&lt;role id&gt;" -> that role.
        /// </summary>
        private static void BuildMention()
        {
            mentionPrefix = string.Empty;
            allowedMentions = new AllowedMentions();

            var value = GeneralConfig.GetGeneralConfig().Config.DiscordMentionRoleId?.Trim() ?? string.Empty;
            if (string.IsNullOrEmpty(value))
            {
                return;
            }

            if (value.Equals("everyone", StringComparison.OrdinalIgnoreCase))
            {
                mentionPrefix = "@everyone ";
                allowedMentions.parse.Add("everyone");
            }
            else if (value.Equals("here", StringComparison.OrdinalIgnoreCase))
            {
                mentionPrefix = "@here ";
                allowedMentions.parse.Add("everyone");
            }
            else
            {
                var roleId = value.Trim('<', '>', '@', '&', ' ');
                mentionPrefix = $"<@&{roleId}> ";
                allowedMentions.roles.Add(roleId);
            }
        }

        /// <summary>
        /// Sends a detection as a rich embed (resolves the Steam avatar first for the thumbnail).
        /// </summary>
        internal static async void SendDetection(DetectionReport report)
        {
            if (webClient == null)
            {
                return; // webhook not configured
            }

            string avatarUrl = await GetSteamAvatarAsync(report.SteamId64);

            var embed = new Embed
            {
                author = new EmbedAuthor { name = report.ServerName },
                color = EmbedColorRed,
                timestamp = DateTime.UtcNow.ToString("o"),
                footer = new EmbedFooter { text = string.Empty }, // Discord shows date/time next to it
                fields =
                [
                    new EmbedField { name = "Player", value = $"[{report.PlayerName}]({report.ProfileUrl})", inlineField = true },
                    new EmbedField { name = "SteamID64", value = $"`{report.SteamId64}`", inlineField = true },
                    new EmbedField { name = "Detection", value = $"`{report.Detection}`", inlineField = false },
                    new EmbedField { name = "Evidence", value = $"```{Truncate(report.Evidence, 1000)}```", inlineField = false },
                    new EmbedField { name = "Punishment", value = report.Punishment, inlineField = false },
                    new EmbedField { name = "Map", value = $"`{report.Map}`", inlineField = true },
                    new EmbedField { name = "Address", value = $"`{report.Address}`", inlineField = true }
                ]
            };

            if (string.IsNullOrEmpty(avatarUrl) == false)
            {
                embed.thumbnail = new EmbedThumbnail { url = avatarUrl };
            }

            var payload = new WebhookData
            {
                username = "CS2AC",
                content = string.IsNullOrEmpty(mentionPrefix) ? null : mentionPrefix.TrimEnd(),
                allowed_mentions = allowedMentions,
                embeds = [embed]
            };

            await PostAsync(payload);
        }

        private static async Task PostAsync(WebhookData payload)
        {
            string json;
            try
            {
                json = JsonSerializer.Serialize(payload, new JsonSerializerOptions
                {
                    DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull
                });
            }
            catch (Exception e)
            {
                Globals.Log($"Unable to send Webhook (Serialize Exception | {e.Message})");
                return;
            }

            try
            {
                var content = new StringContent(json, System.Text.Encoding.UTF8, "application/json");
                var response = await webClient.PostAsync(webClientAddress, content);
                if (response.StatusCode != System.Net.HttpStatusCode.OK &&
                    response.StatusCode != System.Net.HttpStatusCode.NoContent)
                {
                    Globals.Log($"Discord webhook returned {response.StatusCode}");
                }
            }
            catch (Exception e)
            {
                Globals.Log($"Unable to send Webhook (Request Exception | {e.Message})");
            }
        }

        /// <summary>
        /// Resolves a player's full Steam avatar from their public profile XML. No API key needed.
        /// Returns "" on any failure (the embed then shows no thumbnail).
        /// </summary>
        private static async Task<string> GetSteamAvatarAsync(string steamId64)
        {
            if (string.IsNullOrEmpty(steamId64) || steamId64 == "0")
            {
                return string.Empty;
            }

            try
            {
                var xml = await webClient.GetStringAsync($"https://steamcommunity.com/profiles/{steamId64}/?xml=1");
                const string open = "<avatarFull><![CDATA[";
                const string close = "]]></avatarFull>";
                var start = xml.IndexOf(open, StringComparison.Ordinal);
                if (start < 0) return string.Empty;
                start += open.Length;
                var end = xml.IndexOf(close, start, StringComparison.Ordinal);
                return end < 0 ? string.Empty : xml.Substring(start, end - start);
            }
            catch
            {
                return string.Empty;
            }
        }

        private static string Truncate(string value, int max)
        {
            if (string.IsNullOrEmpty(value)) return string.Empty;
            return value.Length <= max ? value : value[..max];
        }
    }

`README.md`:

```md
# CS2-AntiCheat

A modular, server-side anti-cheat for Counter-Strike 2 built on
[CounterStrikeSharp](https://github.com/roflmuffin/CounterStrikeSharp). Inspired by
[SourceMod Anti-Cheat (SMAC)](https://github.com/Silenci0/SMAC).

Each detection is a self-contained **module** with its own config file and its own punishment action
(none / log / kick / ban), so you can tune, disable, or escalate each check independently. Violations
can be mirrored to Discord and to the CounterStrikeSharp log file.

> Author: **panda** · Version: **1.0.0** · Target: **.NET 10 / CounterStrikeSharp API ≥ 374**

---

## Detection modules

| Module            | What it looks for                                                        | Config file                  |
|-------------------|--------------------------------------------------------------------------|------------------------------|
| **Aimbot**        | Suspicious aim snaps / angle changes toward targets                      | `Aimbot.json`                |
| **Backtrack**     | Hitting players at positions they no longer occupy                       | `Backtrack.json`             |
| **BunnyHop**      | Automated / scripted bunnyhopping patterns                               | `BunnyHop.json`              |
| **RapidFire**     | Firing a semi-auto weapon faster than humanly possible                   | `RapidFire.json`             |
| **UntrustedAngles** | View angles outside the range the engine should ever produce           | `UntrustedAngles.json`       |
| **ImpossibleYaw** | Yaw deltas that are physically impossible between ticks                  | `ImpossibleYaw.json`         |
| **DuckSpam**      | Crouch-spam (duck toggling) beyond human rates                           | `DuckSpam.json`              |
| **UserCmdRate**   | Abnormal user-command rates (command injection / timing cheats)          | `UserCmdRate.json`           |
| **NoRecoil**      | Recoil compensation that no human input could achieve                    | `NoRecoil.json`              |
| **AntiDLL**       | Cheat DLLs that subscribe to blacklisted engine game events              | `AntiDLL.json` (+ gamedata)  |

All modules are enabled by default. Each writes its own `Configs/plugins/CS2-AntiCheat/<Module>.json`
on first run.

---

## Installation

1. Install [CounterStrikeSharp](https://github.com/roflmuffin/CounterStrikeSharp) (Metamod + CSSharp)
   and make sure it loads.
2. Copy the **CS2-AntiCheat** folder into:
   ```
   game/csgo/addons/counterstrikesharp/plugins/
   ```
   so you end up with `.../plugins/CS2-AntiCheat/CS2-AntiCheat.dll`.
3. Start the server (or `css_plugins load CS2-AntiCheat`). Config files are generated on first load.
4. Check `addons/counterstrikesharp/logs/` — you should see the load lines (see **Logging** below).

**Platforms:** Windows and Linux are both supported. The AntiDLL module uses per-platform
signatures/offsets shipped in `gamedata/CS2-AntiCheat.AntiDLL.json`.

---

## Configuration

### General (`Configs/plugins/CS2-AntiCheat/General.json`)

| Key                   | Default                                             | Description                                                                 |
|-----------------------|-----------------------------------------------------|-----------------------------------------------------------------------------|
| `Enabled`             | `true`                                              | Master on/off switch for all detections.                                    |
| `Debug`               | `false`                                             | Verbose runtime logging to the console/log. Load milestones log regardless. |
| `LogPrefix`           | `[CS2AC]`                                            | Prefix prepended to every console log line. Empty = no prefix.              |
| `CommandPermission`   | `@css/root`                                          | Permission flag required to run admin commands. Server console always passes. |
| `BanCommand`          | `css_ban #{id} 0 "ANTICHEAT: {reason}"`             | Command run (server console) when a module's action is **Ban**.             |
| `KickCommand`         | `css_kick #{id}`                                    | Command run after the ban command (also used for **Kick**).                 |
| `DiscordAlerts`       | `true`                                              | Master switch for Discord detection alerts.                                 |
| `DiscordWebhookUrl`   | `""`                                                | Discord webhook URL. Empty = Discord disabled.                              |
| `DiscordMentionRoleId`| `""`                                                | `""` = no ping · `everyone` · `here` · a numeric role id.                   |

Command placeholders: `{id}` (userid — use `#{id}` to target by userid), `{steamid}` (SteamID64),
`{steamid2}` (STEAM_0:…), `{name}`, `{reason}`, `{module}`.

### Per-module (`Configs/plugins/CS2-AntiCheat/<Module>.json`)

Every module config has at least:

| Key                | Values                                  | Description                                            |
|--------------------|-----------------------------------------|--------------------------------------------------------|
| `DetectionEnabled` | `true` / `false`                        | Turn this single module on or off.                     |
| `DetectionAction`  | `0` none · `1` log · `2` kick · `3` ban | What to do when this module detects a player.          |

Individual modules add their own thresholds/tunables on top of those two keys.

### AntiDLL (`Configs/plugins/CS2-AntiCheat/AntiDLL.json`)

AntiDLL detects cheat DLLs by the engine game-event listeners they register — cheats subscribe to
events a normal client never listens to.

| Key                | Default        | Description                                                              |
|--------------------|----------------|--------------------------------------------------------------------------|
| `DetectionEnabled` | `true`         | Enable AntiDLL.                                                          |
| `DetectionAction`  | `3` (Ban)      | Action on detection. Listening to a blacklisted event is unambiguous.    |
| `Blacklist`        | ~160 events    | Game-event names that flag a client. Edit freely — no recompile needed.  |

The native **signatures/offsets** live in `gamedata/CS2-AntiCheat.AntiDLL.json` (editable, no
recompile). If a game update breaks them, update that file. See **How AntiDLL works** below.

## Logging

Load milestones are **always** written to `addons/counterstrikesharp/logs/` (independent of `Debug`):

```
[CS2AC] Loaded CS2-AntiCheat v1.0.0
[CS2AC] Initialized 10 detection module(s): Aimbot, Backtrack, BunnyHop, RapidFire, UntrustedAngles, ImpossibleYaw, DuckSpam, UserCmdRate, NoRecoil, AntiDLL
[CS2AC] AntiDLL blacklist mask built (152/160 events resolved to bits)
```

The AntiDLL "mask built" line is the key health check: `X/Y resolved` with `X > 0` means AntiDLL
successfully translated blacklisted event names into listen-bit indices for your game build. `0`
means the gamedata signatures are stale for your current CS2 version.

Verbose runtime logs (per-player events, ticks, detections) are gated by `"Debug": true` in
`General.json` so the log file stays clean in production. Change it and run `tbac_reload`.

## How AntiDLL works

When a client connects it tells the server which game events it wants to listen to (a
`CLCMsg_ListenEvents` message carrying a bitmask of event indices). Legitimate clients only ever
subscribe to a small, fixed set; many cheat DLLs subscribe to extra engine events to drive their
features. AntiDLL flags any client that listens to an event in the `Blacklist`.

Implementation (fully self-contained, ported from
[KillStr3aK/CS2-AntiDLL](https://github.com/KillStr3aK/CS2-AntiDLL)):

1. On the event manager's init, each blacklisted event **name** is resolved to its **listen-bit
   index** via `CGameEventManager::GetEventDescriptor`, building a 512-bit blacklist mask once.
2. `CSource1LegacyGameEventGameSystem::ListenBitsReceived` is hooked. On each message, the client's
   listen-bits are read straight from the message and `AND`-ed with the mask — any set bit is a
   blacklisted event the client subscribed to.

No per-event lookups and no proxy-listener walk at runtime. Signatures/offsets are in
`gamedata/CS2-AntiCheat.AntiDLL.json` with separate Windows and Linux entries:

| Key                                               | Purpose                                                         |
|---------------------------------------------------|-----------------------------------------------------------------|
| `CSource1LegacyGameEventGameSystem::ListenBitsReceived` | Function hooked to intercept a client's listen-bits.      |
| `CLCMsg_ListenEvents::EventCount` / `EventData`   | Offsets of the event-bit count + data pointer in the message.   |
| `CGameEventManager::GetEventDescriptor`           | Resolves an event name to its listen-bit index (at init).       |

All native reads are wrapped in `try/catch`; if a signature is stale the module stays idle and logs
it instead of crashing the server.
