`README.md`:

```md
# Ghidra Teak DSP

A WIP Ghidra plugin for the 3DS/DSi XpertTeak. Most of the Teak instruction set has been implemented, enough to fully disassemble both the 3DS Home Menu's and Sound App's firmwares. Accesses to unimplemented context registers such as ar/arp are represented through pcodeops whenever possible. Includes a CDC/DSP1 binary loader.

## Known issues
* Many instructions not present within the above firmwares are still missing from the specification
* Does not properly handle middle-endianness for 32-bit loads/stores. As far as I can tell, Ghidra has no way to fully support this
* Cannot properly detect push sv -> ret sequences used for function pointers (might be possible to handle with a script?)
* stepi0/stepj0 stepping not implemented (didn't find any sequences to test on...)
* No sat/sata-based saturation (lim is implemented as a pcodeop)
* No modulo/address bit-reverse
* ArStep/ArpStep configs are hardcoded
* Broken call analysis in certain functions, causing the decompiler to display stack writes
* Decompilation can be rather ugly due to lots of sign-extension/zero-extension IR ops

## AI disclosure
AI was used in the making of this project for the following things:
* Creating the initial script for CDC loading
* Suggesting ideas for implementing certain features
* Reviewing and finding bugs in the specification

Everything else, including writing the documentation and the SLEIGH specification, was done by a human.

```

`data/README.txt`:

```txt
The "data" directory is intended to hold data files that will be used by this module and will
not end up in the .jar file, but will be present in the zip or tar file.  Typically, data
files are placed here rather than in the resources directory if the user may need to edit them.

An optional data/languages directory can exist for the purpose of containing various Sleigh language
specification files and importer opinion files.  

The data/buildLanguage.xml is used for building the contents of the data/languages directory.

The skel language definition has been commented-out within the skel.ldefs file so that the 
skeleton language does not show-up within Ghidra.

See the Sleigh language documentation (docs/languages/index.html) for details Sleigh language 
specification syntax.
 
```

```sinc
@define ALBOP_SET   0
@define ALBOP_RST   1
@define ALBOP_CHNG  2
@define ALBOP_ADDV  3
@define ALBOP_TST0  4
@define ALBOP_TST1  5
@define ALBOP_CMPV  6
@define ALBOP_SUBV  7

macro AlbSet(result, a, b)
{
	result = a | b;
	fz = result == 0;
	fm = result s< 0;
}

macro AlbRst(result, a, b)
{
	result = ~a & b;
	fz = result == 0;
	fm = result s< 0;
}

macro AlbChng(result, a, b)
{
	result = a ^ b;
	fz = result == 0;
	fm = result s< 0;
}

macro AlbAddv(result, a, b)
{
	result = a + b;
	fz = result == 0;
	fc0 = carry(a, b);
	
	# TODO: is there a way to simplify the logic here?
	local a32:4 = sext(a);
	local b32:4 = sext(b);
	fm = (a32 + b32) s< 0;
}

macro AlbTst0(a, b)
{
	fz = (a & b) == 0;
}

macro AlbTst1(a, b)
{
	fz = (a & ~b) == 0;
}

macro AlbSubv(result, a, b)
{
	result = b - a;
	fz = result == 0;
	fc0 = a > b;
	
	# TODO: is there a way to simplify the logic here?
	local a32:4 = sext(a);
	local b32:4 = sext(b);
	fm = (b32 - a32) s< 0;
}

macro AlbCmpv(a, b)
{
	local tmp:2;
	AlbSubv(tmp, a, b);
}
```

```sinc
@define ALM_OR    0
@define ALM_AND   1
@define ALM_XOR   2
@define ALM_ADD   3
@define ALM_TST0  4
@define ALM_TST1  5
@define ALM_CMP   6
@define ALM_SUB   7
@define ALM_MSU   8
@define ALM_ADDH  9
@define ALM_ADDL  10
@define ALM_SUBH  11
@define ALM_SUBL  12
@define ALM_SQR   13
@define ALM_SQRA  14
@define ALM_CMPU  15

@define ALU_OR    0
@define ALU_AND   1
@define ALU_XOR   2
@define ALU_ADD   3
@define ALU_CMP   6
@define ALU_SUB   7

macro AlmOr(acc, op)
{
	local tmp:8 = zext(op);
	local result:8 = acc | tmp;
	SetAccAndFlag(acc, result);
}

macro AlmAnd(acc, op)
{
	local tmp:8 = zext(op);
	local result:8 = acc & tmp;
	SetAccAndFlag(acc, result);
}

macro AlmXor(acc, op)
{
	local tmp:8 = zext(op);
	local result:8 = acc ^ tmp;
	SetAccAndFlag(acc, result);
}

macro AlmAdd(acc, op)
{
	local tmp:8 = sext(op);
	local result:8;
	Add40(result, acc, tmp);
	SatAndSetAccAndFlag(acc, result);
}

macro AlmSub(acc, op)
{
	local tmp:8 = sext(op);
	local result:8;
	Sub40(result, acc, tmp);
	SatAndSetAccAndFlag(acc, result);
}

macro AlmMsu(acc, op, prod)
{
	local tmp:2 = op;
	local result:8;
	Add40(result, acc, prod);
	SatAndSetAccAndFlag(acc, result);
	x0 = tmp;
	HwMultiplySS(p0, x0, y0);
}

macro AlmTst0(acc, op)
{
	fz = (acc:2 & op) == 0;
}

macro AlmTst1(acc, op)
{
	fz = (acc:2 & ~op) == 0;
}

macro AlmCmp(acc, op)
{
	local tmp:8 = sext(op);
	local result:8;
	Sub40(result, acc, tmp);
	SetAccFlag(result);
}

macro AlmAddh(acc, op)
{
	local tmp_lo:4 = zext(op);
	tmp_lo = tmp_lo << 16;
	local tmp:8 = sext(tmp_lo);
	local result:8;
	Add40(result, acc, tmp);
	SatAndSetAccAndFlag(acc, result);
}

macro AlmAddl(acc, op)
{
	local tmp:8 = zext(op);
	local result:8;
	Add40(result, acc, tmp);
	SatAndSetAccAndFlag(acc, result);
}

macro AlmSubh(acc, op)
{
	local tmp_lo:4 = zext(op);
	tmp_lo = tmp_lo << 16;
	local tmp:8 = sext(tmp_lo);
	local result:8;
	Sub40(result, acc, tmp);
	SatAndSetAccAndFlag(acc, result);
}

macro AlmSubl(acc, op)
{
	local tmp:8 = zext(op);
	local result:8;
	Sub40(result, acc, tmp);
	SatAndSetAccAndFlag(acc, result);
}

macro AlmSqr(op)
{
	local tmp:2 = op;
	x0 = tmp; y0 = tmp;
	HwMultiplySS(p0, x0, y0);
}

macro AlmCmpu(acc, op)
{
	local tmp:8 = zext(op);
	local result:8;
	Sub40(result, acc, tmp);
	SetAccFlag(result);
}
```

```slaspec
# Macros
macro SetAccFlag(value)
{
	local tmp:8 = zext(value);
	fz = tmp == 0;
	fm = tmp s< 0;
	local tmp32:4 = trunc(tmp);
	local tmp32_s:8 = sext(tmp32);
	fe = tmp != tmp32_s;
	
	# TODO: fn
}

macro PushWord(x)

	local mysave:2 = x;

	*[data]:2 sp = mysave;

macro Add40(result, a, b)
{
	result = a + b;
	fc0 = carry(a, b);
	fv = scarry(a, b);
	fvl = fvl | fv;
}

macro Sub40(result, a, b)
{
	result = a - b;
	fc0 = a < b;
	fv = sborrow(a, b);
	fvl = fvl | fv;
}

# TODO: Shift operations do saturation but in a weird way compared to everything else
# TODO: Arithmetic variations (s == 0)
macro ShiftLeft40(acc_dst, val, shift)
{
	local result:8 = zext(val) << shift;
	fc0 = ((result >> 40) & 0x1) != 0;
	SetAccAndFlag(acc_dst, result:8);
}

macro ShiftRight40(acc_dst, val, shift)
{
	local result:8 = zext(val);
	fc0 = ((result >> (shift - 1)) & 0x1) != 0;
	result = result >> shift;
	SetAccAndFlag(acc_dst, result:8);
}

# Used when the shift value is not an immediate
macro ShiftGen40(acc_dst, val, shift)
{
	if (shift s< 0) goto <shift_right>;
	ShiftLeft40(acc_dst, val, shift);
	goto <done>;
<shift_right>
	ShiftRight40(acc_dst, val, -shift);
<done>
}

macro ProductToBus40(out, prod, shift_ctrl)
{
	local tmp:8 = prod;
	
	if (shift_ctrl == 1) goto <shift_right>;
	if (shift_ctrl < 2) goto <done>;
	tmp = tmp << (shift_ctrl - 1);
	goto <done>;
<shift_right>
	tmp = tmp s>> 1;
<done>
    out = tmp;
}

	local tmp2:8 = s1;
	SatAndSetAccAndFlag(d0, tmp);
	SatAndSetAccAndFlag(d1, tmp2);
}
```

```java
package teak;

import java.io.IOException;
import java.io.InputStream;
import java.util.ArrayList;
import java.util.Collection;
import java.util.List;

import ghidra.app.util.MemoryBlockUtils;
import ghidra.app.util.bin.BinaryReader;
import ghidra.app.util.bin.ByteProvider;
import ghidra.app.util.importer.MessageLog;
import ghidra.app.util.opinion.AbstractProgramWrapperLoader;
import ghidra.app.util.opinion.LoadSpec;
import ghidra.program.model.address.Address;
import ghidra.program.model.address.AddressOutOfBoundsException;
import ghidra.program.model.address.AddressOverflowException;
import ghidra.program.model.address.AddressSpace;
import ghidra.program.model.lang.LanguageCompilerSpecPair;
import ghidra.program.model.listing.Program;
import ghidra.program.model.mem.MemoryBlock;
import ghidra.util.exception.CancelledException;
import ghidra.util.task.TaskMonitor;

public class CdcLoader extends AbstractProgramWrapperLoader {

	private static final String LANGUAGE_ID = "Teak:LE:16:default";
	private static final String COMPILER_ID = "default";

	private static final int HEADER_SIZE = 0x300;
	private static final int MAGIC_OFFSET = 0x100;
	private static final int MAX_SEGMENTS = 10;
	private static final int SEGMENT_RECORD_SIZE = 0x30;

	private static final int TYPE_PROGRAM_A = 0;
	private static final int TYPE_PROGRAM_B = 1;
	private static final int TYPE_DATA = 2;

	private static final int FLAG_RECV_DATA = 1;

	@Override
	public String getName() {
		return "CDC (DSP1)";
	}

	@Override
	public Collection<LoadSpec> findSupportedLoadSpecs(ByteProvider provider) throws IOException {
		List<LoadSpec> loadSpecs = new ArrayList<>();
		if (isCdc(provider)) {
			loadSpecs.add(new LoadSpec(this, 0,
				new LanguageCompilerSpecPair(LANGUAGE_ID, COMPILER_ID), true));
		}
		return loadSpecs;
	}

	@Override
	protected void load(Program program, ImporterSettings settings)
			throws CancelledException, IOException {

		ByteProvider provider = settings.provider();
		MessageLog log = settings.log();
		TaskMonitor monitor = settings.monitor();

		AddressSpace codeSpace = program.getAddressFactory().getAddressSpace("code");
		AddressSpace dataSpace = program.getAddressFactory().getAddressSpace("data");
		if (codeSpace == null || dataSpace == null) {
			log.appendMsg("Teak language is missing the 'code' or 'data' address space");
			return;
		}

		BinaryReader reader = new BinaryReader(provider, true);
		reader.setPointerIndex(MAGIC_OFFSET);
		reader.readNextByteArray(4); // magic
		long binarySize = reader.readNextUnsignedInt();
		int enabledRegions = reader.readNextUnsignedShort();
		reader.readNextUnsignedShort(); // padding
		reader.readNextUnsignedByte(); // unk
		reader.readNextUnsignedByte(); // specialType
		int numSegments = reader.readNextUnsignedByte();
		int flags = reader.readNextUnsignedByte();
		reader.readNextUnsignedInt(); // specialStart
		reader.readNextUnsignedInt(); // specialSize
		reader.readNextLong(); // unk2

		if (binarySize != provider.length()) {
			log.appendMsg(String.format("Header size 0x%X does not match file length 0x%X",
				binarySize, provider.length()));
		}
		log.appendMsg(String.format("Enabled regions: program=0x%02X data=0x%02X",
			enabledRegions & 0xFF, (enabledRegions >> 8) & 0xFF));
		if ((flags & FLAG_RECV_DATA) != 0) {
			log.appendMsg("Flag bit0 set: component expects DSP:RecvData handshake");
		}

		List<Address> codeStarts = new ArrayList<>();
		for (int i = 0; i < numSegments; i++) {
			monitor.checkCancelled();
			long record = 0x120L + (long) i * SEGMENT_RECORD_SIZE;
			reader.setPointerIndex(record);

			long offset = reader.readNextUnsignedInt();
			long start = reader.readNextUnsignedInt();
			long size = reader.readNextUnsignedInt();
			reader.readNextByteArray(3);
			int type = reader.readNextUnsignedByte();
			reader.readNextByteArray(0x20); // sha256 hash

			if (size == 0) {
				log.appendMsg(String.format("Segment %d is empty", i));
				continue;
			}
			if (offset < HEADER_SIZE || offset > provider.length() || size > provider.length() - offset) {
				log.appendMsg(String.format(
					"Segment %d file range 0x%X+0x%X is outside the file", i, offset, size));
				continue;
			}
			if (type != TYPE_PROGRAM_A && type != TYPE_PROGRAM_B && type != TYPE_DATA) {
				log.appendMsg(String.format("Segment %d has unknown memory type %d", i, type));
				continue;
			}

			boolean data = type == TYPE_DATA;
			AddressSpace space = data ? dataSpace : codeSpace;
			String name = (data ? "data_" : "code_") + i;
			String comment = String.format("CDC segment %d type %d file+0x%X",
				i, type, offset);

			try (InputStream segmentBytes = provider.getInputStream(offset)) {
				MemoryBlock block = MemoryBlockUtils.createInitializedBlock(program, false, name,
					wordAddress(space, start), segmentBytes, size, comment, "CDC", true, true,
					!data, log, monitor);
				if (block != null && !data) {
					codeStarts.add(block.getStart());
				}
			}
			catch (AddressOutOfBoundsException | AddressOverflowException e) {
				log.appendMsg(String.format("Segment %d address 0x%X is not in the %s space: %s",
					i, start, space.getName(), e.getMessage()));
			}
		}
	}

	private static Address wordAddress(AddressSpace space, long wordOffset) {
		return space.getAddress(wordOffset, true);
	}

	private static boolean isCdc(ByteProvider provider) throws IOException {
		if (provider.length() < HEADER_SIZE) {
			return false;
		}
		byte[] magic = provider.readBytes(MAGIC_OFFSET, 4);
		if (magic.length != 4 || magic[0] != 'D' || magic[1] != 'S' || magic[2] != 'P' ||
			magic[3] != '1') {
			return false;
		}
		int numSegments = provider.readByte(0x10E);
		return numSegments >= 1 && numSegments <= MAX_SEGMENTS;
	}
}
```
