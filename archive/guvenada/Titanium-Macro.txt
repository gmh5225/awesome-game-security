## Overview

Titanium Macro Engine is a high-performance, precision-focused automation framework designed for sophisticated input orchestration. Operating at the kernel level via an interception driver wrapper, it bypasses standard user-mode API restrictions (such as those bypassed by anti-cheat systems or complex game engines), ensuring flawless execution of complex macro scripts with zero latency.

> **Note**: This repository contains the core execution engine and user interface. Proprietary administrative tools and stealth mechanisms are strictly omitted from this public release.

---

## Key Features

- **Kernel-Mode Injection**: Utilizes a highly optimized wrapper for the Interception driver, injecting events directly into the HID stack before they reach the OS user-space.
- **Microsecond Precision Timing**: Employs performance counters to guarantee sub-millisecond accuracy during event playback, essential for frame-perfect operations.
- **Hardware-Level Event Capture**: Records absolute mouse trajectories and raw keyboard scan codes, rather than relying on unreliable virtual key events.
- **Asynchronous Execution Architecture**: The UI, input capturing, and playback engines run on isolated daemon threads, preventing any GUI locking during intense automation sequences.
- **DPI-Aware Rendering**: The PyQt6 interface dynamically scales across multiple high-resolution displays without layout distortion.

## Technical Architecture

The codebase is strictly organized following solid engineering principles, separating concerns between UI logic, state management, and low-level hardware communication.

- `src/core/inputs/`: Manages driver-level I/O.
- `src/core/controller.py`: The central nervous system orchestrating UI events and underlying hardware routines.
- `src/core/playback/`: The highly tuned engine responsible for replaying `MacroEvent` sequences.

## Installation & Deployment

Due to the kernel-level nature of the engine, administrator privileges are strictly enforced.

```bash
# Clone the repository
git clone https://github.com/guvenada/Titanium-Macro.git

# Navigate to the directory
cd Titanium-Macro

# Install Python dependencies (requires Python 3.10+)
pip install -r requirements.txt

# Install Tesseract OCR (required for Vision features)
# Download from: https://github.com/UB-Mannheim/tesseract/wiki
# Place the installation at: src/resources/bin/tesseract/

# Execute the core engine (Must be run as Administrator)
python main.py

import json
import os
from pathlib import Path
from typing import Any, Dict, Optional
from threading import Lock

class SettingsManager:
    """
    Singleton class to manage application settings.
    Loads from and saves to a JSON file.
    """
    _instance: Optional['SettingsManager'] = None
    _lock: Lock = Lock()
    
    DEFAULT_SETTINGS: Dict[str, Any] = {
        "app_name": "Titanium Macro",
        "version": "0.1.0",
        "theme": "dark",
        "language": "en",
        "window_size": [1280, 720],
        "max_fps": 60
    }

    CONFIG_DIR = Path("config")
    CONFIG_FILE = CONFIG_DIR / "settings.json"

    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super(SettingsManager, cls).__new__(cls)
                    cls._instance._initialized = False
        return cls._instance

    def __init__(self):
        if self._initialized:
            return
            
        self._settings: Dict[str, Any] = self.DEFAULT_SETTINGS.copy()
        self._load_settings()
        self._initialized = True

    def _load_settings(self) -> None:
        """Loads settings from the JSON file if it exists."""
        if not self.CONFIG_FILE.exists():
            self._save_settings()  # Create default file
            return

        try:
            with open(self.CONFIG_FILE, 'r', encoding='utf-8') as f:
                loaded_settings = json.load(f)
                # Update defaults with loaded values (preserves new defaults in future updates)
                self._settings.update(loaded_settings)
        except (json.JSONDecodeError, OSError) as e:
            # Fallback to defaults is already handled by initialization
            print(f"Error loading settings: {e}")

    def _save_settings(self) -> None:
        """Saves current settings to the JSON file."""
        self.CONFIG_DIR.mkdir(parents=True, exist_ok=True)
        try:
            with open(self.CONFIG_FILE, 'w', encoding='utf-8') as f:
                json.dump(self._settings, f, indent=4)
        except OSError as e:
            print(f"Error saving settings: {e}")

    def get(self, key: str, default: Any = None) -> Any:
        """Retrieves a setting value."""
        return self._settings.get(key, default)

    def set(self, key: str, value: Any) -> None:
        """Sets a setting value and saves to file."""
        self._settings[key] = value
        self._save_settings()

    def get_all(self) -> Dict[str, Any]:
        """Returns a copy of all settings."""
        return self._settings.copy()

# Global instance
settings = SettingsManager()

                return

                for pid, pdata in profiles_data.items():
                    try:
                        self._profiles[pid] = Profile.from_dict(pdata)

                        logger.error(f"Failed to load profile {pid}: {e}")
                
                # Ensure active profile exists
                if self._active_profile_id not in self._profiles:
                    logger.warning(f"Active profile '{self._active_profile_id}' not found. resetting to default.")
                    if "default" in self._profiles:
                        self._active_profile_id = "default"
                    else:
                        # Serious corruption? Recreate default
                        self._create_default_config()

            except Exception as e:
                logger.error(f"Failed to load config: {e}. Recreating default.")
                self._create_default_config()

        data = DEFAULT_CONFIG
        self._active_profile_id = data["active_profile_id"]

        }

                "active_profile_id": self._active_profile_id,
                "language": getattr(self, "language", "TR"),
                "profiles": {
                    pid: p.to_dict() for pid, p in self._profiles.items()
                }
            }

                temp_path = self.config_path.with_suffix(".tmp")
                with open(temp_path, 'w', encoding='utf-8') as f:
                    json.dump(data, f, indent=4)

            except Exception as e:
                logger.error(f"Failed to save config: {e}")

    def start_recording(self):
        with self._lock:
            if self.is_playing:

                    pass

import math
import random
from typing import Tuple, List

def calculate_distance(p1: Tuple[int, int], p2: Tuple[int, int]) -> float:
    """Calculates Euclidean distance between two points."""
    return math.sqrt((p2[0] - p1[0])**2 + (p2[1] - p1[1])**2)

def bezier_point(t: float, control_points: List[Tuple[int, int]]) -> Tuple[float, float]:
    """
    Calculates a point on a Bezier curve at parameter t (0.0 to 1.0).
    Supports arbitrary number of control points (n-degree).
    For 3 points (Quadratic): B(t) = (1-t)^2 P0 + 2(1-t)t P1 + t^2 P2
    For 4 points (Cubic): B(t) = (1-t)^3 P0 + 3(1-t)^2 t P1 + 3(1-t)t^2 P2 + t^3 P3
    """
    n = len(control_points) - 1
    x = 0
    y = 0
    
    for i, point in enumerate(control_points):
        # Binomial coefficient
        binom = math.factorial(n) / (math.factorial(i) * math.factorial(n - i))
        # Bernstein polynomial
        bernstein = binom * ((1 - t) ** (n - i)) * (t ** i)
        
        x += point[0] * bernstein
        y += point[1] * bernstein
        
    return x, y

def generate_control_points(
    start: Tuple[int, int], 
    end: Tuple[int, int], 
    strength: float = 0.5
) -> List[Tuple[int, int]]:
    """
    Generates randomized control points for a Cubic Bezier curve (Start, C1, C2, End).
    Strength determines how far control points deviate from the straight line.
    """
    dist = calculate_distance(start, end)
    if dist < 5:
        return [start, end] # Too short for curve

    # Midpoint
    mx = (start[0] + end[0]) / 2
    my = (start[1] + end[1]) / 2
    
    # Perpendicular vector (simple rotation)
    dx = end[0] - start[0]
    dy = end[1] - start[1]
    
    # Normalize perpendicular
    if dist == 0: return [start, end]
    px = -dy / dist
    py = dx / dist
    
    # Random deviation amount
    # Max deviation is roughly half distance * strength
    deviation = (dist * 0.5) * strength
    
    # Create two control points pulled in somewhat random directions
    # but generally ensuring flow.
    
    # Offset 1
    noise1 = random.uniform(-0.5, 1.5) * deviation
    c1x = start[0] + (dx * 0.33) + (px * noise1)
    c1y = start[1] + (dy * 0.33) + (py * noise1)

    # Offset 2
    noise2 = random.uniform(-1.5, 0.5) * deviation
    c2x = start[0] + (dx * 0.66) + (px * noise2)
    c2y = start[1] + (dy * 0.66) + (py * noise2)
    
    return [start, (int(c1x), int(c1y)), (int(c2x), int(c2y)), end]

import time
import threading
from pynput import mouse
from src.core.config.humanizer_settings import HumanizerSettings
from src.core.humanizer.trajectory import TrajectoryGenerator

class HumanMouseActor:
    """
    Wraps the system mouse controller to execute human-like movements.
    """
    def __init__(self, settings: HumanizerSettings = HumanizerSettings()):
        self.mouse = mouse.Controller()
        self.generator = TrajectoryGenerator(settings)
        self.settings = settings

    def move_to(self, x: int, y: int, duration: float = 0.0):
        """
        Blocking call to move mouse to x,y in a human-like fashion.
        """
        start_pos = self.mouse.position
        path = self.generator.generate_path(start_pos, (x, y), duration)
        
        for px, py, delay in path:
            # We can check for stop flags here if we integrated it, 
            # but this is a low-level actor. Uppler layers handle stop logic 
            # by not calling this, or we rely on small steps returning fast.
            
            self.mouse.position = (px, py)
            
            # Precision sleep
            if delay > 0:
               target = time.perf_counter() + delay
               while time.perf_counter() < target:
                   pass

    def click(self, x: int = -1, y: int = -1, button=mouse.Button.left):
        """
        clicks at current position or moves then clicks.
        Adds micro-delays between press and release.
        """

import math
import random
from typing import List, Tuple
from src.core.config.humanizer_settings import HumanizerSettings
from src.core.humanizer.math_utils import calculate_distance, generate_control_points, bezier_point

class TrajectoryGenerator:
    """
    Generates human-like mouse paths.
    """
    def __init__(self, settings: HumanizerSettings = HumanizerSettings()):
        self.settings = settings

    def calculate_only(self, start: Tuple[int, int], end: Tuple[int, int]) -> List[Tuple[int, int]]:
        """
        Returns just the coordinates (x, y) for visualization.
        """
        path_data = self.generate_path(start, end)
        return [(p[0], p[1]) for p in path_data]

    def generate_path(
        self, 
        start: Tuple[int, int], 
        end: Tuple[int, int], 
        duration: float = 0.0
    ) -> List[Tuple[int, int, float]]:
        """
        Generates a list of (x, y, delay) tuples.
        If duration is 0, it is calculated based on distance and average speed.
        """
        distance = calculate_distance(start, end)
        
        # 1. Determine Duration if not provided
        if duration <= 0:
            # Fitts's Law approximation or simple speed
            # Heuristic: base time + distance / speed
            # Add randomness to speed
            speed = self.settings.average_speed * random.uniform(
                1.0 - self.settings.speed_variance, 
                1.0 + self.settings.speed_variance
            )
            # Minimum duration for very short moves (e.g. 0.1s)
            duration = max(0.1, distance / speed)

        # 2. Generate Control Points (Path Shape)
        control_points = generate_control_points(start, end, self.settings.curvature_strength)

        # 3. generate points with velocity profile
        # Steps count depends on duration. We want high resolution.
        # e.g. 60-120 steps per second? or 1 step per pixel?
        # Let's say we update every ~10ms for smooth play, but actor can handle faster.
        # Let's aim for ~100 steps per second of duration.
        steps = int(max(duration * 100, distance / 2, 10)) 
        
        path = []
        for i in range(steps + 1):
            t = i / steps
            
            # Apply Easing (Velocity Profile) to 't'
            # Sigmoid or Cubic Ease In/Out: t_warped
            # Simple Ease-In-Out: 3t^2 - 2t^3 (Smoothstep)
            t_warped = t * t * (3 - 2 * t)
            
            # Add randomness to 't' (Micro-pauses) if desired, but risky for smooth flow.
            # We stick to geometric warping.

            x, y = bezier_point(t_warped, control_points)
            
            # Add Noise/Tremor
            if self.settings.tremor_strength > 0:
                noise_x = random.uniform(-1, 1) * self.settings.tremor_strength * 2
                noise_y = random.uniform(-1, 1) * self.settings.tremor_strength * 2
                # Don't noise start/end too much or accuracy fails
                # Fade noise at edges
                edge_factor = 1.0 - (2 * abs(0.5 - t)) # 0 at ends, 1 at center
                x += noise_x * edge_factor
                y += noise_y * edge_factor

            # Delay for this segment
            # Uniform delay per step = duration / steps
            # But since steps represent time (t), the delay is constant.
            step_delay = duration / steps
            
            path.append((int(x), int(y), step_delay))
            
        return path

from src.utils.screen_reader import screen_reader

class Condition(ABC):
    @abstractmethod
    def evaluate(self, context: Dict[str, Any]) -> bool:
        pass
    
    @abstractmethod
    def to_dict(self) -> Dict[str, Any]:
        pass

    @staticmethod
    def from_dict(data: Dict[str, Any]) -> 'Condition':
        c_type = data.get("type")
        if c_type == "PixelColorCondition":
            return PixelColorCondition.from_dict(data)
        elif c_type == "VariableCondition":
            return VariableCondition.from_dict(data)
        # Add TimeCondition etc.
        raise ValueError(f"Unknown condition type: {c_type}")

@dataclass
class PixelColorCondition(Condition):
    x: int
    y: int
    target_color: Tuple[int, int, int] # RGB
    tolerance: int = 10

    def evaluate(self, context: Dict[str, Any]) -> bool:
        return screen_reader.pixel_matches_color(
            self.x, self.y, self.target_color, self.tolerance
        )

    def to_dict(self) -> Dict[str, Any]:
        return {
            "type": "PixelColorCondition",
            "x": self.x,
            "y": self.y,
            "target_color": self.target_color,
            "tolerance": self.tolerance
        }

    @staticmethod
    def from_dict(data: Dict[str, Any]) -> 'PixelColorCondition':
        return PixelColorCondition(
            x=data["x"],
            y=data["y"],
            target_color=tuple(data["target_color"]),
            tolerance=data.get("tolerance", 10)
        )

@dataclass
class VariableCondition(Condition):
    """
    Checks if a variable in context meets a criteria.
    Operator: '==', '!=', '>', '<', '>=', '<='
    """
    variable_name: str
    operator: str
    value: Any

    def evaluate(self, context: Dict[str, Any]) -> bool:
        current_val = context.get(self.variable_name)
        if current_val is None:
            return False
            
        if self.operator == "==": return current_val == self.value
        if self.operator == "!=": return current_val != self.value
        if self.operator == ">": return current_val > self.value
        if self.operator == "<": return current_val < self.value
        if self.operator == ">=": return current_val >= self.value
        if self.operator == "<=": return current_val <= self.value
        return False

    def to_dict(self) -> Dict[str, Any]:
        return {
            "type": "VariableCondition",
            "variable_name": self.variable_name,
            "operator": self.operator,
            "value": self.value
        }

    @staticmethod
    def from_dict(data: Dict[str, Any]) -> 'VariableCondition':
        return VariableCondition(
            variable_name=data["variable_name"],
            operator=data["operator"],
            value=data["value"]
        )

from typing import Optional, Tuple
from src.core.models.coordinates import Point, CoordinateMode, Rect
from src.core.platform.window_manager import window_manager
from src.utils.monitor_info import get_primary_monitor_resolution
from src.utils.logger import logger

class CoordinateMapper:
    """
    Transforms logical points into physical screen coordinates.
    Handles relative positioning and window targeting.
    """
    
    def transform(self, point: Point, target_window: Optional[str] = None) -> Tuple[int, int]:
        """
        Returns (x, y) in absolute physical screen pixels.
        """
        
        if point.mode == CoordinateMode.ABSOLUTE:
            # Already absolute. 
            # CAUTION: If user recorded on 4K and plays on 1080p, this will fail.
            # We assume ABSOLUTE means "Raw Screen Coordinates".
            return int(point.x), int(point.y)

        elif point.mode == CoordinateMode.RELATIVE:
            # % of Primary Monitor
            # TODO: Support multiple monitors?
            mw, mh = get_primary_monitor_resolution()
            # If point.x is 0.5, result is mw * 0.5
            # We treat the screen as the reference rect (0, 0, mw, mh)
            screen_rect = Rect(0, 0, mw, mh)
            abs_point = point.to_absolute(screen_rect)
            return int(abs_point.x), int(abs_point.y)

        elif point.mode == CoordinateMode.WINDOW_RELATIVE:
            if not target_window:
                logger.warning("Window Relative mode requires a target_window name.")
                # Fallback: Treat as Absolute? Or fail?
                # Let's fail safe -> return 0,0 or raise error
                return 0, 0
            
            w_rect = window_manager.get_window_rect(target_window)
            if not w_rect:
                logger.error(f"Target window '{target_window}' not found! Cannot map coordinates.")
                return 0, 0
                
            abs_point = point.to_absolute(w_rect)
            return int(abs_point.x), int(abs_point.y)
            
        return int(point.x), int(point.y)

import time
import random
import threading
from typing import List, Dict, Any, Optional

from src.core.models.blocks import Block, ActionBlock, LoopBlock, ConditionBlock, DelayBlock
from src.core.playback.engine import SystemPlayer
from src.core.playback.manager import playback_manager
from src.core.playback.safety import FailSafeMonitor
from src.utils.logger import logger

class ScriptInterpreter:
    """
    Executes a tree of Blocks.
    Replaces the linear playback loop in SystemPlayer (eventually).
    Currently acts as a higher-level orchestrator that might use SystemPlayer for actions.
    """
    def __init__(self, player: SystemPlayer):
        self.player = player
        self.context: Dict[str, Any] = {}
        self._stop_requested = False
        self._pause_event = threading.Event()
        self._pause_event.set()

    def execute_script(self, blocks: List[Block]):
        self._stop_requested = False
        self._pause_event.set()
        self.context = {} # Reset context
        
        try:
            for block in blocks:
                if self._stop_requested: break
                self._execute_block(block)
        except Exception as e:
            logger.error(f"Interpreter Error: {e}")

    def stop(self):
        self._stop_requested = True
        self._pause_event.set() # Unblock

    def pause(self):
        self._pause_event.clear()

    def resume(self):
        self._pause_event.set()

    def _check_control_flow(self):
        if self._stop_requested:
            raise InterruptedError("Script stopped")
        
        if not self._pause_event.is_set():
            self._pause_event.wait()
            if self._stop_requested:
                 raise InterruptedError("Script stopped during pause")

    def _execute_block(self, block: Block):
        self._check_control_flow()

        if isinstance(block, ActionBlock):
            # Delegate to SystemPlayer's lower level execution
            # But SystemPlayer was designed for linear list.
            # We can use its internal `_execute_event` method if we expose it or make it public.
            # Or we can just reuse the logic here.
            # Ideally, SystemPlayer should just obtain the next event.
            # For now, let's call the private execution method or move it out.
            # Accessing protected member is okay within the core package.
            self.player._execute_event(block.event)
            # Add implicit delay if needed (from event timestamp delta) 
            # Logic Layer usually handles timing differently (Delays are explicit blocks mostly).
            # But converting linear to blocks: ActionBlock typically has no delay field here.
            # If we want timing, we need to respect the event's timestamp relative to previous?
            # Or we assume Blocks execute as fast as possible unless Delays?
            # "Requirements: ... transform from simple Event List to Script Tree."
            # Existing macros have timestamps.
            # If we strictly valid new Logic Engine: we rely on DelayBlocks.
            # If we are playing legacy linear macros: we need to handle timing.
            pass

        elif isinstance(block, LoopBlock):
            count = block.loop_count
            i = 0
            while True:
                if count != -1 and i >= count:
                    break
                
                self.context["loop_index"] = i
                for child in block.children:
                    self._execute_block(child)
                    if self._stop_requested: return
                
                i += 1

        elif isinstance(block, ConditionBlock):
            if block.condition.evaluate(self.context):
                for child in block.then_branch:
                    self._execute_block(child)
            else:
                for child in block.else_branch:
                    self._execute_block(child)

        elif isinstance(block, DelayBlock):
            ms = random.randint(block.min_ms, block.max_ms)
            time.sleep(ms / 1000.0)

from typing import List
from src.core.models.event import MacroEvent, EventType
from src.core.models.blocks import Block, ActionBlock, PathBlock, HotkeyBlock, DelayBlock

class InputMapper:
    """
    Translates raw MacroEvents into high-level UI Blocks.
    Aggregates mouse movements and hotkeys.
    """
    
    def to_blocks(self, events: List[MacroEvent]) -> List[Block]:
        blocks: List[Block] = []
        if not events:
            return blocks

        current_path_events: List[MacroEvent] = []
        
        # Simple aggregation for now
        # TODO: Advanced Hotkey detection (Ctrl Down -> Key Down -> Key Up -> Ctrl Up)
        
        for i, event in enumerate(events):
            # 1. Handle Mouse Moves (Path Aggregation)
            if event.event_type == EventType.MOVE:
                current_path_events.append(event)
                continue
            
            # If we were building a path, finalize it before handling this new non-move event
            if current_path_events:
                blocks.append(PathBlock(events=list(current_path_events)))
                current_path_events = []
            
            # 2. Handle simple events
            # For this iteration, we treat every non-move event as an ActionBlock.
            # Real hotkey detection requires lookahead or state machine.
            # Let's keep it 1:1 for keys for now, to ensure correctness first.
            blocks.append(ActionBlock(event=event))
        
        # Finalize trailing path
        if current_path_events:
            blocks.append(PathBlock(events=list(current_path_events)))

        return blocks

    def to_events(self, blocks: List[Block]) -> List[MacroEvent]:
        events: List[MacroEvent] = []
        
        for block in blocks:
            if isinstance(block, ActionBlock):
                events.append(block.event)
            elif isinstance(block, PathBlock):
                events.extend(block.events)
            elif isinstance(block, HotkeyBlock):
                events.extend(block.events)
            # Other blocks (Loop, Condition) are handled by Interpreter, 
            # not flat event list playback.
            # Ideally, the Player should execute Blocks, not Events.
            # But our SystemPlayer uses Events. 
            # SystemPlayer needs to be updated or we flatten here.
            
        return events

from enum import Enum
from dataclasses import dataclass
from typing import Tuple, NamedTuple

class CoordinateMode(Enum):
    ABSOLUTE = "ABSOLUTE"               # Exact pixel (x=100, y=200)
    RELATIVE = "RELATIVE"               # Percentage (x=0.5, y=0.5)
    WINDOW_RELATIVE = "WINDOW_RELATIVE" # Offset from window top-left (x=10, y=10)

class Rect(NamedTuple):
    x: int
    y: int
    width: int
    height: int

@dataclass
class Point:
    x: float
    y: float
    mode: CoordinateMode = CoordinateMode.ABSOLUTE

    def to_absolute(self, reference_rect: Rect) -> 'Point':
        """
        Converts this point to an ABSOLUTE point based on the reference rect.
        reference_rect should be the screen rect (for RELATIVE) or window rect (for WINDOW_RELATIVE).
        """
        if self.mode == CoordinateMode.ABSOLUTE:
            return Point(self.x, self.y, CoordinateMode.ABSOLUTE)
            
        elif self.mode == CoordinateMode.RELATIVE:
            # reference_rect is usually the screen or viewport
            abs_x = reference_rect.x + (self.x * reference_rect.width)
            abs_y = reference_rect.y + (self.y * reference_rect.height)
            return Point(int(abs_x), int(abs_y), CoordinateMode.ABSOLUTE)
            
        elif self.mode == CoordinateMode.WINDOW_RELATIVE:
            # reference_rect is the target window
            abs_x = reference_rect.x + self.x
            abs_y = reference_rect.y + self.y
            return Point(int(abs_x), int(abs_y), CoordinateMode.ABSOLUTE)
            
        return self

from enum import Enum, auto
from typing import Optional

from src.utils.logger import logger

class AppState(Enum):
    ACTIVE = auto()             # Window focused, UI updating at 60 FPS
    BACKGROUND_PLAYING = auto() # Macro running, UI updates suppressed
    IDLE_MINIMIZED = auto()     # Window minimized, low refresh rate (1 FPS)

class EcoManager:
    """
    Manages the application's power state.
    Adjusts polling rates and UI refresh intervals based on activity.
    """
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super(EcoManager, cls).__new__(cls)
            cls._instance._initialized = False
        return cls._instance

    def __init__(self):
        if self._initialized: return
        self._initialized = True
        
        self._state = AppState.ACTIVE
        self._refresh_rate = 60 # Hz
        
    @property
    def current_state(self) -> AppState:
        return self._state
        
    def set_state(self, state: AppState):
        if self._state == state:
            return
            
        logger.info(f"EcoManager: Transitioning to {state.name}")
        self._state = state
        
        if state == AppState.ACTIVE:
            self._refresh_rate = 60
            self._set_process_priority(high=False) # Normal
            
        elif state == AppState.BACKGROUND_PLAYING:
            self._refresh_rate = 1 # Update UI rarely
            self._set_process_priority(high=True) # Boost for macro execution
            
        elif state == AppState.IDLE_MINIMIZED:
            self._refresh_rate = 1 # 1 Hz or less
            self._set_process_priority(high=False) # Yield CPU
            
            # Trigger aggressive cleanup when minimized
            import gc
            gc.collect()

    def get_ui_interval(self) -> int:
        """Returns the recommended UI update interval in milliseconds."""
        if self._refresh_rate <= 0: return 1000
        return int(1000 / self._refresh_rate)

    def _set_process_priority(self, high: bool):
        try:
            import psutil
            import os
            p = psutil.Process(os.getpid())
            if high:
                if psutil.WINDOWS:
                    p.nice(psutil.HIGH_PRIORITY_CLASS)
                else:
                    p.nice(-10) # High priority on Unix
            else:
                if psutil.WINDOWS:
                    p.nice(psutil.NORMAL_PRIORITY_CLASS)
                else:
                    p.nice(0)
        except Exception as e:
            logger.warning(f"Failed to set process priority: {e}")

# Global Instance
eco_manager = EcoManager()

from src.utils.logger import logger

class PerformanceMonitor:
    """
    Monitors system resource usage and optimizes memory.
    """
    def __init__(self, interval: int = 60):
        self._interval = interval
        self._running = False
        self._thread = None

    def start(self):
        if self._running: return
        self._running = True
        self._thread = threading.Thread(target=self._monitor_loop, daemon=True, name="PerfMonitor")
        self._thread.start()
        logger.info("Performance Monitor started.")

    def stop(self):
        self._running = False
        if self._thread:
            self._thread.join(timeout=1.0)

    def _monitor_loop(self):
        process = psutil.Process()
        while self._running:
            try:
                # Log usage
                cpu = process.cpu_percent()
                mem = process.memory_info().rss / 1024 / 1024 # MB
                
                if mem > 500: # If using > 500MB
                    logger.warning(f"High Memory Usage: {mem:.2f} MB. Triggering GC.")
                    self.trigger_gc()
                
                # logger.debug(f"Resource Usage - CPU: {cpu:.1f}%, RAM: {mem:.2f} MB")
                
                time.sleep(self._interval)
            except Exception as e:
                logger.error(f"PerfMonitor Error: {e}")

    @staticmethod
    def trigger_gc():
        """Manually triggers Python Garbage Collection."""
        collected = gc.collect()
        logger.debug(f"GC Collected {collected} objects.")

from src.utils.logger import logger

class MemoryGuard:
    """
    Controls Python's Garbage Collection to prevent stuttering during critical operations.
    Usage:
        with MemoryGuard(disable_gc=True):
             # Critical high-performance code
    """
    def __init__(self, disable_gc: bool = True):
        self.disable_gc = disable_gc
        self.was_enabled = False

    def __enter__(self):
        if self.disable_gc:
            self.was_enabled = gc.isenabled()
            if self.was_enabled:
                gc.disable()
                # logger.debug("MemoryGuard: GC Disabled for critical section.")

    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.disable_gc and self.was_enabled:
            gc.enable()
            # Perform a manual collection to clean up what accumulated
            collected = gc.collect()
            # logger.debug(f"MemoryGuard: GC Enabled. Collected {collected} objects.")

    @staticmethod
    def force_clean():
        """Manually triggers a full garbage collection."""
        n = gc.collect()
        logger.debug(f"Manual GC Collection: {n} objects.")

import threading
import time
import psutil
from src.utils.logger import logger

class PerformanceMonitor:
    """
    Monitors system resource usage and optimizes app behavior.
    Prevents memory leaks and heavy CPU usage during idle.
    """
    def __init__(self):
        self._running = False
        self._thread = None
        self._process = psutil.Process()
        self.check_interval = 5.0 # Seconds
        self.max_memory_mb = 500 # Threshold

    def start(self):
        if self._running: return
        self._running = True
        self._thread = threading.Thread(target=self._monitor_loop, daemon=True, name="PerfMonitor")
        self._thread.start()
        logger.debug("Performance Monitor started.")

    def stop(self):
        self._running = False
        if self._thread:
            self._thread.join(timeout=1.0)
        logger.debug("Performance Monitor stopped.")

    def _monitor_loop(self):
        while self._running:
            try:
                # 1. Check Memory Usage
                mem_info = self._process.memory_info()
                rss_mb = mem_info.rss / 1024 / 1024
                
                if rss_mb > self.max_memory_mb:
                    logger.warning(f"High Memory Usage Detected: {rss_mb:.2f} MB. Requesting GC.")
                    import gc
                    gc.collect()
                
                # 2. Log stats occasionally (every 60s)
                # For now just silent unless issue
                
            except Exception as e:
                logger.error(f"PerfMonitor Error: {e}")
            
            time.sleep(self.check_interval)

# Global Instance
perf_monitor = PerformanceMonitor()

import platform
from abc import ABC, abstractmethod
from typing import Optional, Tuple
from src.core.models.coordinates import Rect
from src.utils.logger import logger

if platform.system() == "Windows":
    import win32gui
    import win32con
    import ctypes

class IWindowManager(ABC):
    @abstractmethod
    def get_window_rect(self, window_title: str) -> Optional[Rect]:
        pass

    @abstractmethod
    def bring_to_front(self, window_title: str) -> bool:
        pass

    @abstractmethod
    def is_window_active(self, window_title: str) -> bool:
        pass

class WindowsWindowManager(IWindowManager):
    def _find_window(self, title_substr: str) -> int:
        """Finds a window handle (HWND) by partial title match."""
        # This is a simple implementation. For production, we might want exact match or regex.
        hwnd_list = []
        
        def enum_cb(hwnd, results):
            if win32gui.IsWindowVisible(hwnd):
                txt = win32gui.GetWindowText(hwnd)
                if title_substr.lower() in txt.lower():
                    results.append(hwnd)
        
        win32gui.EnumWindows(enum_cb, hwnd_list)
        return hwnd_list[0] if hwnd_list else 0

    def get_window_rect(self, window_title: str) -> Optional[Rect]:
        hwnd = self._find_window(window_title)
        if not hwnd:
            logger.warning(f"Window not found: {window_title}")
            return None
            
        try:
            # GetWindowRect returns (left, top, right, bottom)
            # DwmGetWindowAttribute is better for Windows 10/11 invisible borders, 
            # but standard GetWindowRect is often "good enough" if DPI aware.
            # Let's stick to standard GetWindowRect for now, assuming DPI awareness is set.
            rect = win32gui.GetWindowRect(hwnd)
            x, y, r, b = rect
            w = r - x
            h = b - y
            return Rect(x, y, w, h)
        except Exception as e:
            logger.error(f"Failed to get window rect: {e}")
            return None

    def bring_to_front(self, window_title: str) -> bool:
        hwnd = self._find_window(window_title)
        if not hwnd: return False
        
        try:
            win32gui.ShowWindow(hwnd, win32con.SW_RESTORE)
            win32gui.SetForegroundWindow(hwnd)
            return True
        except Exception:
            return False

    def is_window_active(self, window_title: str) -> bool:
        fg_hwnd = win32gui.GetForegroundWindow()
        fg_title = win32gui.GetWindowText(fg_hwnd)
        return window_title.lower() in fg_title.lower()

# Factory
def get_window_manager() -> IWindowManager:

import time
import threading
from typing import Optional
from pynput import mouse, keyboard

from src.core.playback.interface import IPlayer
from src.core.models.macro import MacroScript
from src.core.models.event import MacroEvent, EventType
from src.utils.logger import logger
from src.core.optimization.memory_guard import MemoryGuard

class SystemPlayer(IPlayer):
    def __init__(self):
        self._running = False
        self._paused = False
        self._thread: Optional[threading.Thread] = None
        self._current_macro: Optional[MacroScript] = None
        self.progress = 0.0
        
        # Input Controllers
        self._mouse_ctl = mouse.Controller()
        self._keyboard_ctl = keyboard.Controller()

    from typing import Optional, Callable
    
    def play(self, macro: MacroScript, loops: int = 1, speed: float = 1.0, on_finished: Optional[Callable[[], None]] = None) -> None:
        if self._running:
            return
            
        self._current_macro = macro
        self._loops = loops
        self._speed = speed
        self._on_finished = on_finished
        self._running = True
        self._paused = False
        self._thread = threading.Thread(target=self._play_loop, daemon=True)
        self._thread.start()

    def stop(self) -> None:
        if not self._running: return
        
        self._running = False
        self._paused = False
        if self._thread and self._thread.is_alive():
            self._thread.join(timeout=1.0)
            
        self.progress = 0.0

    def pause(self) -> None:
        self._paused = True

    def resume(self) -> None:
        self._paused = False

    @property
    def is_playing(self) -> bool:
        return self._running

    @property
    def current_progress(self) -> float:
        return self.progress

    def _play_loop(self):
        # Suppress GC for smooth playback
        with MemoryGuard(disable_gc=True):
            if not self._current_macro:
                self._running = False
                return

            events = self._current_macro.events
            total_events = len(events)
            if total_events == 0:
                self._running = False
                return

            original_loops = getattr(self, '_loops', 1) # Set in play()
            current_loop = 0
            
            while self._running:
                if original_loops > 0 and current_loop >= original_loops:
                    break
                    
                start_time = time.perf_counter()
                
                # Simple Playback Logic
                for i, event in enumerate(events):
                    if not self._running: break
                    
                    while self._paused:
                        time.sleep(0.1)
                        if not self._running: break

                    # Calculate delay with speed multiplier
                    # original: target_offset = event.timestamp
                    # speed > 1.0 means shorter delay -> offset / speed
                    target_offset = event.timestamp / self._speed
                    
                    elapsed = time.perf_counter() - start_time
                    
                    wait = target_offset - elapsed
                    if wait > 0:
                        # Precision wait
                        if wait > 0.002: # Sleep for bulk
                            time.sleep(wait - 0.002)
                        while time.perf_counter() - start_time < target_offset:
                             pass # Spin lock for last ms

                    # Execute
                    try:
                        self._execute_event(event)
                    except Exception as e:
                        logger.error(f"Playback Error at event {i}: {e}")
                    
                    # Update Progress (approximate for loops)
                    self.progress = (current_loop + (i + 1) / total_events) / (original_loops if original_loops > 0 else 1)

                current_loop += 1
                logger.info(f"Playback Loop {current_loop} finished.")

        self._running = False
        logger.info("Playback Finished.")
        if self._on_finished:
            try:
                self._on_finished()
            except Exception as e:
                logger.error(f"Error in playback finished callback: {e}")

    def _execute_event(self, event: MacroEvent):
        if event.event_type == EventType.MOVE:
            self._mouse_ctl.position = (event.x, event.y)
            
        elif event.event_type == EventType.CLICK:
            # Format: "Button.left:pressed" or "Button.right:released"
            try:
                if not event.key_code: return
                parts = str(event.key_code).split(':')
                btn_str = parts[0] # Button.left
                action = parts[1] if len(parts) > 1 else 'click' # pressed/released
                
                # Parse Button
                btn = mouse.Button.left
                if 'right' in btn_str: btn = mouse.Button.right
                elif 'middle' in btn_str: btn = mouse.Button.middle
                elif 'x1' in btn_str: btn = mouse.Button.x1
                elif 'x2' in btn_str: btn = mouse.Button.x2
                
                if action == 'pressed':
                    self._mouse_ctl.press(btn)
                elif action == 'released':
                    self._mouse_ctl.release(btn)
                else:
                    self._mouse_ctl.click(btn, 1)
            except Exception as e:
                logger.warning(f"Failed to execute click: {e}")

        elif event.event_type == EventType.SCROLL:
            # dy is usually 1 or -1
            self._mouse_ctl.scroll(0, event.dy)
            
        elif event.event_type == EventType.KEY_PRESS or event.event_type == EventType.KEY_RELEASE:
            try:
                key_val = event.key_code
                if not key_val: return
                
                # Parse Key
                k = None
                s_val = str(key_val)
                
                if len(s_val) == 1:
                    k = s_val # Char
                elif s_val.startswith('Key.'):
                    # Key.space -> space
                    attr = s_val.split('.')[1]
                    k = getattr(keyboard.Key, attr, None)
                
                if k:
                    if event.event_type == EventType.KEY_PRESS:
                        self._keyboard_ctl.press(k)
                    else:
                        self._keyboard_ctl.release(k)
                else:
                    logger.warning(f"Unknown key: {key_val}")
                    
            except Exception as e:
                logger.warning(f"Failed to execute key: {e}")

import time
import threading
from typing import List, Optional
from queue import Queue, Empty
from pynput import mouse, keyboard

from src.core.interfaces import IRecorder
from src.core.models.event import MacroEvent, EventType
from src.utils.logger import logger

class SystemRecorder(IRecorder):
    """
    Implementation of IRecorder using pynput.
    Captures global mouse and keyboard events.
    Fixed: Implements all abstract methods (get_events, clear_events).
    """
    def __init__(self):
        self._events: Queue[MacroEvent] = Queue()
        self._mouse_listener: Optional[mouse.Listener] = None
        self._keyboard_listener: Optional[keyboard.Listener] = None
        self._is_recording = False
        self._stop_key = keyboard.Key.esc

    def start_recording(self) -> None:
        if self._is_recording:
            return

        logger.info("Starting System Recorder...")
        self._is_recording = True
        self.clear_events()

        # Start non-blocking listeners
        self._mouse_listener = mouse.Listener(
            on_move=self._on_move,
            on_click=self._on_click,
            on_scroll=self._on_scroll
        )
        self._keyboard_listener = keyboard.Listener(
            on_press=self._on_press,
            on_release=self._on_release
        )

        self._mouse_listener.start()
        self._keyboard_listener.start()
        logger.debug("Listeners started.")

    def stop_recording(self) -> List[MacroEvent]:
        if not self._is_recording:
            return []
            
        logger.info("Stopping System Recorder...")
        self._is_recording = False
        
        # Stop listeners
        if self._mouse_listener:
            self._mouse_listener.stop()
            self._mouse_listener = None
        
        if self._keyboard_listener:
            self._keyboard_listener.stop()
            self._keyboard_listener = None
            
        logger.debug("Listeners stopped.")
        
        events = self.get_events()
        
        # Filter out the Stop Key (F9) from the end
        # We look for Key.f9 press/release at the very end
        if events:
            # Check last few events
            while events and events[-1].event_type in (EventType.KEY_PRESS, EventType.KEY_RELEASE):
                k = str(events[-1].key_code)
                if 'f9' in k.lower():
                    events.pop()
                else:
                    break
                    
        return events

    def get_events(self) -> List[MacroEvent]:
        """
        Returns all events currently in the queue and clears it? 
        The interface implies 'get' might be destructive or snapshot.
        Usually 'get_events' in this context pulls them out to save.
        Accessing queue elements without removing is not thread-safe easily.
        We will drain the queue here.
        """
        results = []
        while not self._events.empty():
            try:
                results.append(self._events.get_nowait())
            except Empty:
                break
        return results

    def clear_events(self) -> None:
        with self._events.mutex:
            self._events.queue.clear()

    # --- Callbacks ---

    def _on_move(self, x, y):
        # logger.info(f"DEBUG: Move {x},{y}") 
        if not self._is_recording: return
        self._events.put(MacroEvent(
            timestamp=time.perf_counter(),
            event_type=EventType.MOVE,
            x=x,
            y=y
        ))

    def _on_click(self, x, y, button, pressed):
        logger.info(f"DEBUG: Click {button} {'pressed' if pressed else 'released'} at {x},{y}")
        if not self._is_recording: return
        
        btn_code = f"{button}"
        etype = EventType.CLICK 
        btn_code = f"{button}:{'pressed' if pressed else 'released'}"

        self._events.put(MacroEvent(
            timestamp=time.perf_counter(),
            event_type=etype,
            key_code=btn_code,
            x=x,
            y=y
        ))

    def _on_scroll(self, x, y, dx, dy):
        logger.info(f"DEBUG: Scroll {dx},{dy}")
        if not self._is_recording: return
        self._events.put(MacroEvent(
            timestamp=time.perf_counter(),
            event_type=EventType.SCROLL,
            x=x,
            y=y,
            dy=dy
        ))

    def _on_press(self, key):
        try:
            k = key.char
        except AttributeError:
            k = str(key)
        logger.info(f"DEBUG: Press {k}")
        
        if not self._is_recording: return
        
        if key == self._stop_key:
            pass 

        self._events.put(MacroEvent(
            timestamp=time.perf_counter(),
            event_type=EventType.KEY_PRESS,
            key_code=k
        ))

    def _on_release(self, key):
        try:
            k = key.char
        except AttributeError:
            k = str(key)
        logger.info(f"DEBUG: Release {k}")
        
        if not self._is_recording: return
        
        self._events.put(MacroEvent(
            timestamp=time.perf_counter(),
            event_type=EventType.KEY_RELEASE,
            key_code=k
        ))

import threading
import time
from typing import List, Optional
from src.core.interfaces import IRecorder
from src.core.recorder.engine import SystemRecorder
from src.core.models.event import MacroEvent, EventType
from src.utils.logger import logger

class RecorderManager:
    """
    Kayıt İşlerinin Tek Patronu. (Singleton Manager)
    
    Arayüz (Tüketici/GUI) ile Kayıt Motoru (Üretici/Engine) arasındaki
    köprüyü bu arkadaş kurar.
    """
    _instance: Optional['RecorderManager'] = None
    _lock: threading.RLock = threading.RLock()

    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super(RecorderManager, cls).__new__(cls)
                    cls._instance._initialized = False
        return cls._instance

    def __init__(self):
        if self._initialized:
            return
            
        self._recorder: IRecorder = SystemRecorder()
        self._is_recording = False
        self._start_time = 0.0
        self._recorded_events: List[MacroEvent] = []
        self._initialized = True

    @property
    def is_recording(self) -> bool:
        return self._is_recording

    def start(self) -> None:
        if self._is_recording:
            logger.warning("RecorderManager: Zaten kayıt alıyoruz şef, sakin.")
            return

        logger.info("RecorderManager: Kayıt seansı başlıyor...")
        self._start_time = time.perf_counter()
        self._recorder.start_recording()
        self._is_recording = True

    def stop(self) -> List[MacroEvent]:
        if not self._is_recording:
            logger.warning("RecorderManager: Ortada kayıt yok ki durduralım?")
            return []

        logger.info("RecorderManager: Kaydı bitiriyoruz...")
        raw_events = self._recorder.stop_recording()
        self._is_recording = False
        
        # Ham verileri al
        # raw_events = self._recorder.get_events() # stop_recording zaten listeyi döndü
        logger.info(f"RecorderManager: {len(raw_events)} ham olay (event) yakalandı.")
        
        # Temizle ve Paketi Hazırla
        self._recorded_events = self._normalize_events(raw_events)
        
        # Akıllı Kesim (Smart Trim):
        # Son 10-20 harekete bak, eğer bir tıklama (CLICK) varsa,
        # bu muhtemelen kullanıcının "Durdur" butonuna basışıdır.
        # Onu ve sonrasını uçur.
        if self._recorded_events:
            cut_index = -1
            # Son 20 olaya bakıyoruz (seri tıklamalara karşı önlem)
            params = self._recorded_events[-20:] if len(self._recorded_events) > 20 else self._recorded_events
            
            # Sondan başa doğru tarıyoruz
            start_search = len(self._recorded_events) - 1
            max_search = max(-1, len(self._recorded_events) - 21)
            
            for i in range(start_search, max_search, -1):
                e = self._recorded_events[i]
                if e.event_type == EventType.CLICK:
                    # Aha bir tık bulduk. Bu "Stop Button" adayıdır.
                    cut_index = i
                    # Genelde Stop tek tık olur. 
                    # Ama basılı tutup (Press) sonra Release yaptıysa?
                    # Neyse, bulduğumuz son tıklamadan itibaren keselim.
                    break
            
            if cut_index != -1:
                # Eğer Release ise, hemen önceki Press'i de kontrol et
                if cut_index > 0:
                    prev = self._recorded_events[cut_index - 1]
                    if prev.event_type == EventType.CLICK:
                         cut_index -= 1 # Press'i de dahil et kessin.
                
                logger.info(f"RecorderManager: Son tıklamayı (Stop Click) temizliyoruz. (Index: {cut_index})")
                self._recorded_events = self._recorded_events[:cut_index]
            
            # Sondaki boş mouse hareketlerini de temizle (Cleanup Trailing Moves)
            while self._recorded_events and self._recorded_events[-1].event_type == EventType.MOVE:
                self._recorded_events.pop()
        
        return self._recorded_events

    def get_last_recording(self) -> List[MacroEvent]:
        return self._recorded_events

    def _normalize_events(self, events: List[MacroEvent]) -> List[MacroEvent]:
        """
        Zaman damgalarını (Timestamp) hizaya sokuyoruz.
        Seçenek A: İlk olay 0.0 saniyede başlar.
        Seçenek B: Kayıt başladığı andan (start_time) itibaren sayarız.
        
        Biz Seçenek B'yi kullanıyoruz: Kayıt tuşuna basıldığı an 0.0 kabul edilir.
        Böylece kullanıcı kayda basıp 2 saniye beklediyse, o 2 saniye de makroya dahil olur.
        """
        if not events:
            return []

        normalized = []
        start_t = self._start_time
        
        for e in events:
            # t_event - t_start_recording
            # Negatif çıkarsa (olmaz ama) 0.0 yap.
            relative_time = max(0.0, e.timestamp - start_t)
            
            e.timestamp = relative_time
            normalized.append(e)
            
        return normalized

import cv2
import numpy as np
from dataclasses import dataclass
from typing import Optional, Tuple, Any

from src.utils.logger import logger
from src.utils.image_utils import load_image
from src.core.vision.screen import screen_grabber

@dataclass
class MatchResult:
    found: bool
    x: int = 0
    y: int = 0
    confidence: float = 0.0
    center_x: int = 0
    center_y: int = 0

class TemplateMatcher:
    """
    Locates images on the screen using OpenCV Template Matching.
    """
    
    def find_image(
        self, 
        template_path: str, 
        region: Optional[Tuple[int, int, int, int]] = None, 
        grayscale: bool = True,
        confidence_threshold: float = 0.9
    ) -> MatchResult:
        """
        Searches for the template image within the specified screen region.
        
        Args:
            template_path: Path to the image file.
            region: (left, top, width, height) tuple. None = Full Screen.
            grayscale: Use grayscale matching (Faster).
            confidence_threshold: Minimum match quality (0.0 - 1.0).
            
        Returns:
            MatchResult object.
        """
        
        # 1. Load Template
        template = load_image(template_path, grayscale=grayscale)
        if template is None:
            return MatchResult(False)
            
        # 2. Capture Screen
        # screen_grabber returns BGR
        screen_img = screen_grabber.capture(region)
        
        if grayscale:
            # Convert BGR to Gray
            screen_img = cv2.cvtColor(screen_img, cv2.COLOR_BGR2GRAY)
        
        # 3. Match Template
        try:
            # TM_CCOEFF_NORMED is robust to lighting changes and returns 0-1 score
            res = cv2.matchTemplate(screen_img, template, cv2.TM_CCOEFF_NORMED)
            
            # 4. Get Best Match
            min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(res)
            
            # max_val is the confidence score
            if max_val >= confidence_threshold:
                # Calculate absolute coordinates
                # max_loc is relative to the captured region
                relative_x, relative_y = max_loc
                
                # Get template dimensions
                h, w = template.shape[:2]
                
                # Region offset
                offset_x, offset_y = (region[0], region[1]) if region else (0, 0)
                
                final_x = offset_x + relative_x
                final_y = offset_y + relative_y
                
                center_x = final_x + (w // 2)
                center_y = final_y + (h // 2)
                
                return MatchResult(
                    found=True,
                    x=final_x,
                    y=final_y,
                    confidence=float(max_val),
                    center_x=center_x,
                    center_y=center_y
                )
            else:
                return MatchResult(found=False, confidence=float(max_val))

        except Exception as e:
            logger.error(f"Template matching error: {e}")
            return MatchResult(False)

import cv2
import numpy as np
import pytesseract
import os
import sys
from typing import Optional
from src.core.models.coordinates import Rect
from src.core.vision.screen import screen_grabber
from src.utils.logger import logger

class TextReader:
    """
    OCR Engine Wrapper.
    """
    def __init__(self):
        self._setup_tesseract()

    def _setup_tesseract(self):
        """
        Locates Tesseract binary. 
        Prioritizes local 'resources/bin' for portability.
        """
        # 1. Check local portable path
        # Frozen (Exe) vs Dev (Source)
        if getattr(sys, 'frozen', False):
            base_path = sys._MEIPASS
        else:
            base_path = os.getcwd()
            
        local_path = os.path.join(base_path, 'resources', 'bin', 'tesseract', 'tesseract.exe')
        
        if os.path.exists(local_path):
            pytesseract.pytesseract.tesseract_cmd = local_path
            logger.info(f"Using Portable Tesseract: {local_path}")
        else:
            logger.warning("Portable Tesseract not found. Falling back to System PATH.")

    def preprocess_image(self, image: np.ndarray) -> np.ndarray:
        """
        Optimizes image for text recognition:
        Grayscale -> Resize (Upscale) -> Thresholding
        """
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        
        # Upscale for better recognition of small text
        scale = 2
        width = int(gray.shape[1] * scale)
        height = int(gray.shape[0] * scale)
        dim = (width, height)
        resized = cv2.resize(gray, dim, interpolation=cv2.INTER_CUBIC)
        
        # Thresholding (Otsu's binarization is usually good for text)
        _, thresh = cv2.threshold(resized, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
        
        return thresh

    def read_text(self, region: Rect) -> str:
        """
        Captures the screen region and returns text.
        """
        try:
            # 1. Capture
            original_img = screen_grabber.grab_region(region)
            if original_img is None:
                return ""
                
            # 2. Preprocess
            processed_img = self.preprocess_image(original_img)
            
            # 3. OCR
            # config: --psm 6 (Assume a single uniform block of text) or 7 (Single line)
            text = pytesseract.image_to_string(processed_img, config='--psm 6')
            
            return text.strip()
            
        except Exception as e:
            logger.error(f"OCR Error: {e}")
            return ""

import mss
import numpy as np
import cv2
from typing import Tuple, Optional
from src.utils.logger import logger

class ScreenGrabber:
    """
    Captures screen content for computer vision.
    Optimized for high-performance ROI capturing.
    """
    def __init__(self):
        self.sct = mss.mss()

    def capture(self, region: Optional[Tuple[int, int, int, int]] = None) -> np.ndarray:
        """
        Captures a screenshot.
        
        Args:
            region: Tuple (left, top, width, height). If None, captures primary monitor.
            
        Returns:
            numpy.ndarray: Image in BGR format (ready for OpenCV).
        """
        try:
            if region:
                monitor = {
                    "left": int(region[0]),
                    "top": int(region[1]),
                    "width": int(region[2]),
                    "height": int(region[3])
                }
            else:
                # Capture primary monitor (monitor 1)
                monitor = self.sct.monitors[1]

            # Grab returns BGRA
            sct_img = self.sct.grab(monitor)
            
            # Convert to numpy array
            img = np.array(sct_img)
            
            # Drop Alpha channel (BGRA -> BGR) to be standard OpenCV format
            # MSS pixels are [B, G, R, A]
            img = img[:, :, :3]
            
            # Note: MSS returns BGRA, OpenCV expects BGR or Gray.
            # If we want grayscale immediately, we can convert here, 
            # but usually the Matcher handles conversion to keep this generic.
            
            return img

        except Exception as e:
            logger.error(f"Screen capture failed: {e}")
            # Return empty black image on failure to prevent crashes
            return np.zeros((100, 100, 3), dtype=np.uint8)

from PyQt6.QtWidgets import QStackedWidget, QWidget, QGraphicsOpacityEffect
from PyQt6.QtCore import Qt, QPropertyAnimation, QEasingCurve, QPoint, QParallelAnimationGroup

class FadingStackedWidget(QStackedWidget):
    """
    A QStackedWidget that cross-fades between pages.
    """
    def __init__(self, parent=None):
        super().__init__(parent)
        self.fade_anim = None

    def setCurrentIndex(self, index):
        if index == self.currentIndex():
            return
            
        # Get current and next widget
        current_widget = self.currentWidget()
        next_widget = self.widget(index)
        
        if not current_widget:
            super().setCurrentIndex(index)
            return

        # Prepare next widget
        # We need to make sure next widget is visible and sized correctly
        # The standard QStackedWidget only shows one at a time.
        # To animate, we might need a temporary overlay or just standard transition logic 
        # that QStackedWidget doesn't easily support out of the box without changing visibility.
        
        # Simpler Approach: 
        # 1. Fade OUT current
        # 2. Switch
        # 3. Fade IN next
        
        # Even Better (Cross-fade simulation):
        # We can implement transition logic here.
        # But QStackedWidget forces only one visible.
        
        # Let's use a fade-out-then-in approach for reliability with QStackedWidget
        
        self.stop_animation()
        
        # Setup Opacity Effect for Current
        self.effect_out = QGraphicsOpacityEffect(current_widget)
        current_widget.setGraphicsEffect(self.effect_out)
        
        self.anim_out = QPropertyAnimation(self.effect_out, b"opacity")
        self.anim_out.setDuration(200) # Fast fade out
        self.anim_out.setStartValue(1.0)
        self.anim_out.setEndValue(0.0)
        self.anim_out.setEasingCurve(QEasingCurve.Type.OutQuad)
        
        # When fade out done, switch and fade in
        self.anim_out.finished.connect(lambda: self._switch_and_fade_in(index, next_widget))
        self.anim_out.start()

    def _switch_and_fade_in(self, index, next_widget):
        # Reset Opacity of previous widget so it's ready for next time
        if self.currentWidget():
            self.currentWidget().setGraphicsEffect(None)
            
        super().setCurrentIndex(index)
        
        # Setup Opacity for Next
        self.effect_in = QGraphicsOpacityEffect(next_widget)
        next_widget.setGraphicsEffect(self.effect_in)
        next_widget.setVisible(True) # Ensure visible
        
        self.anim_in = QPropertyAnimation(self.effect_in, b"opacity")
        self.anim_in.setDuration(300)
        self.anim_in.setStartValue(0.0)
        self.anim_in.setEndValue(1.0)
        self.anim_in.setEasingCurve(QEasingCurve.Type.InQuad)
        
        # Cleanup effect after animation
        self.anim_in.finished.connect(lambda: next_widget.setGraphicsEffect(None))
        self.anim_in.start()

    def stop_animation(self):
        # Stop any running animations if user clicks fast

    def mousePressEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton:
            # Only drag if not clicking buttons (handled by layout but safe check)
            self.start_pos = event.globalPosition().toPoint() - self.parent_window.frameGeometry().topLeft()
            event.accept()

    def mouseMoveEvent(self, event):
        if self.start_pos:
            self.parent_window.move(event.globalPosition().toPoint() - self.start_pos)

from PyQt6.QtCore import QAbstractListModel, Qt, QModelIndex, QMimeData
from typing import List, Any
import pickle

from src.core.models.blocks import Block
from src.core.mapper.input_mapper import input_mapper
from src.core.models.macro import MacroScript

class MacroActionModel(QAbstractListModel):
    """
    Qt List Model for Macro Blocks.
    Sources data from a list of Block objects.
    """
    MIME_TYPE = "application/vnd.titanium.block"

    def __init__(self, parent=None):
        super().__init__(parent)
        self._blocks: List[Block] = []

    def set_events(self, events: List[Any]):
        """Loads events into the model."""
        self.beginResetModel()
        if events:
            # Convert raw events to high-level blocks
            self._blocks = input_mapper.to_blocks(events)
        else:
            self._blocks = []
        self.endResetModel()

    def get_blocks(self) -> List[Block]:
        return self._blocks

    def rowCount(self, parent=QModelIndex()) -> int:
        return len(self._blocks)

    def data(self, index: QModelIndex, role: int = Qt.ItemDataRole.DisplayRole) -> Any:
        if not index.isValid() or not (0 <= index.row() < len(self._blocks)):
            return None
        
        block = self._blocks[index.row()]
        
        if role == Qt.ItemDataRole.DisplayRole:
            return block # Return the whole block object to the delegate
            # The delegate will decide how to render it.
            
        return None

    def flags(self, index: QModelIndex) -> Qt.ItemFlag:
        default_flags = Qt.ItemFlag.ItemIsEnabled | Qt.ItemFlag.ItemIsSelectable
        if index.isValid():
            return default_flags | Qt.ItemFlag.ItemIsDragEnabled
        else:
            return default_flags | Qt.ItemFlag.ItemIsDropEnabled

    # --- Drag & Drop Support ---
    def supportedDropActions(self) -> Qt.DropAction:
        return Qt.DropAction.MoveAction

    def mimeTypes(self) -> List[str]:
        return [self.MIME_TYPE]

    def mimeData(self, indexes: List[QModelIndex]) -> QMimeData:
        mime_data = QMimeData()
        # Serialize selected blocks
        # Only support single selection move for simplicity first
        if not indexes:
            return mime_data
        
        row = indexes[0].row()
        block_data = pickle.dumps(row) # Just send the index for internal move
        mime_data.setData(self.MIME_TYPE, block_data)
        return mime_data

    def dropMimeData(self, data: QMimeData, action: Qt.DropAction, row: int, column: int, parent: QModelIndex) -> bool:
        if action == Qt.DropAction.IgnoreAction:
            return True
        if not data.hasFormat(self.MIME_TYPE):
            return False

        if row == -1:
            row = self.rowCount()

        # Retrieve source index
        src_row = pickle.loads(data.data(self.MIME_TYPE))
        
        if src_row == row: return False # No Move

        self.beginMoveRows(QModelIndex(), src_row, src_row, QModelIndex(), row)
        block = self._blocks.pop(src_row)
        # Adjust destination row if we removed from above
        dst_row = row if src_row > row else row - 1
        self._blocks.insert(dst_row, block)
        self.endMoveRows()
        
        return True

    def removeRow(self, row: int, parent=QModelIndex()) -> bool:
        if 0 <= row < len(self._blocks):
            self.beginRemoveRows(parent, row, row)
            self._blocks.pop(row)
            self.endRemoveRows()
            return True
        return False

from PyQt6.QtGui import QPainter, QColor, QPen, QPainterPath, QPolygonF
from PyQt6.QtCore import QPoint, QPointF, Qt
from typing import List, Tuple

class GhostCursorRender:
    """
    Handles drawing the Predictive Ghost Cursor.
    """
    def __init__(self):
        self.current_pos = QPoint(0, 0)
        self.target_pos = QPoint(0, 0)
        self.path_points: List[QPointF] = []
        self.visible = False

    def update_target(self, current: QPoint, target: QPoint, path_preview: List[Tuple[int, int]] = None):
        self.current_pos = current
        self.target_pos = target
        
        if path_preview:
            self.path_points = [QPointF(float(p[0]), float(p[1])) for p in path_preview]
        else:
            self.path_points = []
            
        self.visible = True

    def paint(self, painter: QPainter):
        if not self.visible:
            return

        painter.save()
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)

        # 1. Draw Connection Line (Bezier Polyline)
        pen = QPen(QColor(0, 122, 204, 150)) # Blueish, semi-transparent
        pen.setWidth(2)
        pen.setStyle(Qt.PenStyle.DashLine)
        painter.setPen(pen)
        
        if self.path_points:
            painter.drawPolyline(self.path_points)
        else:
            # Fallback
            painter.drawLine(self.current_pos, self.target_pos)

        # 2. Draw Target Circle
        # Inner Circle
        painter.setBrush(QColor(0, 122, 204, 100))
        painter.setPen(Qt.PenStyle.NoPen)
        radius = 10
        painter.drawEllipse(self.target_pos, radius, radius)
        
        # Outer Ring
        pen = QPen(QColor(0, 122, 204, 200))
        pen.setWidth(2)
        pen.setStyle(Qt.PenStyle.SolidLine)
        painter.setPen(pen)
        painter.setBrush(Qt.BrushStyle.NoBrush)
        painter.drawEllipse(self.target_pos, radius + 5, radius + 5)

        painter.restore()

from PyQt6.QtWidgets import QWidget

class FadeAnimation:
    """
    Helper to fade a widget in or out.
    """
    @staticmethod
    def create(target: QWidget, start_opacity: float, end_opacity: float, duration: int = 300) -> QPropertyAnimation:
        anim = QPropertyAnimation(target, b"windowOpacity")
        anim.setStartValue(start_opacity)
        anim.setEndValue(end_opacity)
        anim.setDuration(duration)
        anim.setEasingCurve(QEasingCurve.Type.InOutQuad)
        return anim

    @staticmethod
    def fade_in(target: QWidget, duration: int = 300):
        anim = FadeAnimation.create(target, 0.0, 1.0, duration)
        target.show()
        anim.start(QPropertyAnimation.DeletionPolicy.DeleteWhenStopped)

    @staticmethod
    def fade_out(target: QWidget, duration: int = 300):
        anim = FadeAnimation.create(target, 1.0, 0.0, duration)
        anim.finished.connect(target.hide)
        anim.start(QPropertyAnimation.DeletionPolicy.DeleteWhenStopped)

        import copy
        new_blocks = []

        for row in rows:
            if row < len(blocks):
                original = blocks[row]
                # Deepcopy ensures we don't reference the same object
                clone = copy.deepcopy(original)
                new_blocks.append(clone)
            
        # Insert them after the last selected row
        insert_row = rows[-1] + 1
        
        self.model.beginInsertRows(self.model.index(insert_row, 0).parent(), insert_row, insert_row + len(new_blocks) - 1)

        self.model.endInsertRows()

class StorageError(Exception):
    """Base exception for storage related errors."""

class IStorageProvider(ABC):
    """
    Interface for saving and loading MacroScripts.
    Allows swapping between JSON, XML, Binary, or Database backends.
    """

    @abstractmethod
    def save(self, macro: MacroScript, filepath: str) -> bool:
        """
        Saves the macro script to the specified filepath.
        Returns True if successful, raises StorageError otherwise.
        """
        pass

    @abstractmethod
    def load(self, filepath: str) -> MacroScript:
        """
        Loads a macro script from the specified filepath.
        Returns the MacroScript object, raises StorageError otherwise.
        """
        pass

import json
import os
import tempfile
import shutil
from pathlib import Path
from src.services.storage.interface import IStorageProvider, StorageError
from src.core.models.macro import MacroScript
from src.utils.logger import logger

class JsonStorageProvider(IStorageProvider):
    """
    Implementation of IStorageProvider using JSON.
    Features:
    - Atomic writes (write to temp, then move).
    - Data integrity checks.
    """

    def save(self, macro: MacroScript, filepath: str) -> bool:
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        
        try:
            # Serialize
            data = macro.to_dict()
            
            # Direct write (Simplest approach for Windows/Permission issues)
            with open(path, 'w', encoding='utf-8') as f:
                json.dump(data, f, indent=4)
                
            logger.info(f"Saved macro to: {path}")
            return True
            
        except Exception as e:
            logger.error(f"Failed to save macro: {e}")
            raise StorageError(f"Failed to save file: {e}")

    def load(self, filepath: str) -> MacroScript:
        path = Path(filepath)
        
        if not path.exists():
            raise StorageError(f"File not found: {path}")
            
        try:
            with open(path, 'r', encoding='utf-8') as f:
                data = json.load(f)
            
            # Basic Integrity Check
            if "version" not in data or "events" not in data:
                 raise ValueError("Missing required fields (version, events)")

            # Backward compatibility checks could go here
            
            # Deserialize
            macro = MacroScript.from_dict(data)
            logger.info(f"Loaded macro: {macro.name} ({len(macro.events)} events)")
            return macro

        except json.JSONDecodeError as e:
            logger.error(f"Corrupted JSON file: {e}")
            raise StorageError(f"File is corrupted or not a valid JSON: {e}")
        except Exception as e:
            logger.error(f"Failed to load macro: {e}")
            raise StorageError(f"Failed to load file: {e}")

def mutate_file(filepath: str):
    """
    Appends random bytes to the end of a file to change its hash.
    Does not affect executable logic if appended.
    """
    if not os.path.exists(filepath):
        print(f"Error: File {filepath} not found.")
        return

    # Generate random payload (16 to 64 bytes)
    size = random.randint(16, 64)
    payload = os.urandom(size)
    
    try:
        with open(filepath, "ab") as f:
            f.write(payload)
        print(f"Successfully mutated {filepath}. Added {size} bytes.")
    except Exception as e:
        print(f"Mutation failed: {e}")

if __name__ == "__main__":

def load_image(path: str, grayscale: bool = True) -> Optional[np.ndarray]:
    """
    Loads an image from disk with caching.
    
    Args:
        path: Absolute path to the image file.
        grayscale: If True, converts to grayscale immediately.

def set_dpi_awareness():
    """
    Sets the process to be DPI aware so that we get real pixel coordinates associated with the screen.
    Without this, Windows scales the coordinates (e.g. 1920x1080 becomes 1536x864 at 125% scale).
    """
    if platform.system() == "Windows":
        try:
            # PROCESS_PER_MONITOR_DPI_AWARE_V2 = 2
            ctypes.windll.shcore.SetProcessDpiAwareness(2)
        except Exception:
            try:
                # Fallback for older Windows
                ctypes.windll.user32.SetProcessDPIAware()
            except Exception:
                pass

def get_primary_monitor_resolution() -> Tuple[int, int]:
    """
    Returns the (width, height) of the primary monitor in real physical pixels.
    """
    if platform.system() == "Windows":
        user32 = ctypes.windll.user32
        return user32.GetSystemMetrics(0), user32.GetSystemMetrics(1)
    return (1920, 1080) # Fallback

import threading
import time
import psutil
import os
from src.utils.logger import logger
from src.core.optimization.memory_guard import MemoryGuard

class SelfProfiler:
    """
    Monitors process memory usage and triggers cleanup if threshold exceeded.
    """
    def __init__(self, threshold_mb: int = 300, interval: int = 10):
        self.threshold_mb = threshold_mb
        self.interval = interval
        self._running = False
        self._thread = None
        self._process = psutil.Process(os.getpid())

    def start(self):
        if self._running: return
        self._running = True
        self._thread = threading.Thread(target=self._monitor_loop, daemon=True, name="SelfProfiler")
        self._thread.start()
        logger.info("Self Profiler started.")

    def stop(self):
        self._running = False
        if self._thread:
            self._thread.join(timeout=1.0)

    def _monitor_loop(self):
        while self._running:
            try:
                mem_bytes = self._process.memory_info().rss
                mem_mb = mem_bytes / (1024 * 1024)
                
                if mem_mb > self.threshold_mb:
                    logger.warning(f"High RAM Usage detected: {mem_mb:.1f} MB. Triggering Emergency Cleanup.")
                    # 1. Clear caches (Future: ImageCache.clear())
                    # 2. Force GC
                    MemoryGuard.force_clean()
                    
                time.sleep(self.interval)
            except Exception as e:
                logger.error(f"Profiler Error: {e}")

class ScreenReader:
    """
    Fast screen reading utility using MSS.
    Optimized for single pixel retrieval.
    """
    def __init__(self):
        self.sct = mss.mss()

    def get_pixel(self, x: int, y: int) -> Tuple[int, int, int]:
        """
        Returns RGB tuple for the pixel at (x, y).
        MSS returns BGRA, so we convert it.
        """
        # mss.grab() expects a bounding box dictionary
        monitor = {"top": y, "left": x, "width": 1, "height": 1}
        
        # Grab the pixel
        # returns logic: [B, G, R, A] (on Windows usually)
        sct_img = self.sct.grab(monitor)
        
        # Access raw pixels
        # pixel(0, 0) returns (R, G, B) tuple in MSS newer versions or strictly ordered?
        # MSS generic return is BGRA.
        # Let's check sct_img.pixel(0,0) -> (r, g, b)
        
        return sct_img.pixel(0, 0)

    def pixel_matches_color(self, x: int, y: int, target_rgb: Tuple[int, int, int], tolerance: int = 10) -> bool:
        """
        Checks if pixel at x,y matches target_rgb within tolerance.
        """
        r, g, b = self.get_pixel(x, y)
        tr, tg, tb = target_rgb
        
        return (abs(r - tr) <= tolerance and
                abs(g - tg) <= tolerance and
                abs(b - tb) <= tolerance)

import re
from difflib import SequenceMatcher
from typing import Optional, Tuple

class StringParser:
    """
    Helper utilities for extracting data from OCR text.
    """
    
    @staticmethod
    def extract_number(text: str) -> Optional[float]:
        """
        Extracts the first number found in the string.
        Handles commas (1,000) and decimals (50.5).
        """
        # Remove commas
        clean_text = text.replace(',', '')
        
        # Regex for integer or float
        match = re.search(r'[-+]?\d*\.\d+|\d+', clean_text)
        if match:
            try:
                return float(match.group())
            except ValueError:
                return None
        return None

    @staticmethod
    def fuzzy_match(text: str, target: str, threshold: float = 0.8) -> bool:
        """
        Returns True if 'text' is similar enough to 'target'.
        Case-insensitive.
        """
        if not text or not target:
            return False
            
        text = text.lower().strip()
        target = target.lower().strip()
        
        # Direct match check first
        if target in text:
            return True
            
        # Levenshtein / Sequence Matcher
        ratio = SequenceMatcher(None, text, target).ratio()
        return ratio >= threshold

    @staticmethod
    def extract_variable(text: str, pattern: str) -> Optional[str]:
        """
        Extracts a value based on a Regex pattern with a capture group.
        Example: Pattern "HP: (\d+)" works on "HP: 100/100" -> returns "100"
        """
        try:
            match = re.search(pattern, text)
            if match and match.groups():
                return match.group(1)
        except Exception:
            pass
        return None

string_parser = StringParser()

import sys

from pathlib import Path

# Add project root to python path

from src.core.config.humanizer_settings import HumanizerSettings
from src.core.humanizer.trajectory import TrajectoryGenerator
from src.core.humanizer.math_utils import calculate_distance

def verify_humanizer():
    print("[TEST] Initializing Humanizer Verification...")
    
    settings = HumanizerSettings(humanity_level=80)
    generator = TrajectoryGenerator(settings)
    
    start = (100, 100)
    end = (800, 600)  # Diagonal move ~860px
    
    print(f"[INFO] Generating path from {start} to {end}...")
    
    t0 = time.perf_counter()
    path = generator.generate_path(start, end)
    t1 = time.perf_counter()
    
    print(f"[PASS] Path generated in {(t1-t0)*1000:.2f} ms.")
    print(f"[INFO] Path length: {len(path)} steps.")
    
    if not path:
        print("[FAIL] Path is empty!")
        return

    # 1. Verify Start and End
    p_start = (path[0][0], path[0][1])
    p_end = (path[-1][0], path[-1][1])
    
    # Bezier might start slightly off due to quantization or noise, but usually start is exact
    # End might have noise.
    print(f"[INFO] Path Start: {p_start}, End: {p_end}")
    
    # 2. Check Linearity (Curvature)
    # A straight line midpoint would be ((100+800)/2, (100+600)/2) = (450, 350)
    mid_idx = len(path) // 2
    mid_point = (path[mid_idx][0], path[mid_idx][1])
    
    straight_mid = ((start[0]+end[0])/2, (start[1]+end[1])/2)
    deviation = calculate_distance(mid_point, straight_mid)
    
    print(f"[INFO] Midpoint deviation from straight line: {deviation:.2f} pixels.")
    if deviation > 10:
        print("[PASS] Path is curved.")
    else:
        print("[WARN] Path is suspiciously straight (low curvature setting?).")

    # 3. Check Duration
    total_delay = sum(p[2] for p in path)
    print(f"[INFO] Total calculated duration: {total_delay:.2f}s")
    
    if 0.1 <= total_delay <= 5.0:
        print("[PASS] Duration is reasonable.")
    else:
        print("[FAIL] Duration is out of bounds.")

if __name__ == "__main__":
    verify_humanizer()

from src.core.playback.engine import SystemPlayer

def verify_logic():
    print("[TEST] Initializing Logic Layer Verification...")
    
    # 1. Setup Mock Player
    # We use the real SystemPlayer but we intercept execution to avoid real clicks
    player = SystemPlayer()
    original_execute = player._execute_event
    
    execution_log = []
    
    def mock_execute(event):
        execution_log.append(f"Action: {event.event_type.name}")
        print(f"[MOCK] Executing Action: {event.event_type.name}")

    player._execute_event = mock_execute
    
    interpreter = ScriptInterpreter(player)
    
    # 2. Build Script Tree
    # Loop 3 times
    #   If loop_index < 2:
    #       Click
    #   Else:
    #       Delay
    
    # Action Block
    click_event = MacroEvent(0, EventType.CLICK)
    action_block = ActionBlock(click_event)
    
    # Delay Block
    delay_block = DelayBlock(100, 200) # 100-200ms
    
    # Condition Block
    condition = VariableCondition("loop_index", "<", 2)
    condition_block = ConditionBlock(
        condition=condition,
        then_branch=[action_block],
        else_branch=[delay_block]
    )
    
    # Loop Block
    loop_block = LoopBlock(
        children=[condition_block],
        loop_count=3
    )
    
    print("[TEST] Script Tree Constructed.")
    
    # 3. Execute
    print("[TEST] Executing Script...")
    t0 = time.time()
    interpreter.execute_script([loop_block])
    t1 = time.time()
    
    print(f"[INFO] Execution completed in {t1-t0:.2f}s")
    
    # 4. Verify Log
    print(f"[INFO] Execution Log: {execution_log}")
    
    # Expected: Click, Click, Delay (No action logged for delay, but time passes)
    # Loop 0: index=0 (<2) -> Click
    # Loop 1: index=1 (<2) -> Click
    # Loop 2: index=2 (>=2) -> Delay
    
    expected_actions = ["Action: CLICK", "Action: CLICK"]
    
    if execution_log == expected_actions:
        print("[PASS] Logic executed correctly.")
    else:
        print(f"[FAIL] execution_log {execution_log} != {expected_actions}")

    # Restore
    player._execute_event = original_execute

import os

# Output file
OUTPUT_FILE = "master_setup.py"

# Filters
EXCLUDE_DIRS = {
    "__pycache__", "dist", ".git", ".gemini", ".vscode", "logs", "screenshots", 
    "build", "installer", "tools", # Don't pack the build artifacts or tools themselves except the ones we want?
    # Actually we WANT build/compile_nuitka.bat and installer/setup.iss
}
# We will whitelist specific tools if needed, but let's exclude the 'tools' dir to prevent packing this script itself loopingly if we aren't careful.
# But we DO want 'tools/updater_stub.py'.

INCLUDE_EXTENSIONS = {".py", ".md", ".txt", ".bat", ".iss", ".ini", ".qss", ".json"}

def collect_files():
    project_files = {}
    
    for root, dirs, files in os.walk('.'):
        # Exclude hidden or build dirs
        dirs[:] = [d for d in dirs if d not in EXCLUDE_DIRS and not d.startswith('.')]
        
        for file in files:
            if file == OUTPUT_FILE or file == "packager.py": continue
            if not any(file.endswith(ext) for ext in INCLUDE_EXTENSIONS): continue
            
            full_path = os.path.join(root, file)
            rel_path = os.path.relpath(full_path, '.')
            
            # Read content
            try:
                with open(full_path, 'r', encoding='utf-8') as f:
                    content = f.read()
                project_files[rel_path] = content
            except Exception as e:
                print(f"Skipping {rel_path}: {e}")

    # Explicitly include tools/updater_stub.py if it was skipped
    if "tools/updater_stub.py" not in project_files and os.path.exists("tools/updater_stub.py"):
         with open("tools/updater_stub.py", 'r', encoding='utf-8') as f:
             project_files["tools/updater_stub.py"] = f.read()

    return project_files

def generate_script(files):
    # Header
    script = [
        "import os",
        "import sys",
        "",
        "def create_project():",
        "    print('Initializing Titanium Macro Project Structure...')",
        "",
        "    # 1. Directory Structure (Explicit)",
        "    dirs = [",
        "        'src/core', 'src/gui', 'src/services', 'src/utils', 'src/resources',",
        "        'resources/bin/tesseract', 'logs', 'dist', 'build', 'installer'",
        "    ]",
        "    for d in dirs:",
        "        os.makedirs(d, exist_ok=True)",
        "",
        "    # 2. File Contents",
        "    files = {"
    ]
    
    # Body
    for path, content in files.items():
        # Sanitize content for python string literal
        # We use repr() but it might quote with ' or ".
        # Multiline strings are cleaner.
        safe_content = content.replace("\\", "\\\\").replace("'''", "\\'\\'\\'")
        # Path must use forward slashes for cross-platform compatibility in the dictionary key
        clean_path = path.replace("\\", "/")
        
        script.append(f"        '{clean_path}': '''{safe_content}''',")
        
    # Footer
    script.extend([
        "    }",
        "",
        "    # 3. Write Files",
        "    for path, content in files.items():",
        "        # Handle subdirectories automatically",
        "        dir_name = os.path.dirname(path)",
        "        if dir_name and not os.path.exists(dir_name):",
        "            os.makedirs(dir_name, exist_ok=True)",
        "",
        "        print(f'Writing {path}...')",
        "        with open(path, 'w', encoding='utf-8') as f:",
        "            f.write(content)",
        "",
        "    print('\\n[SUCCESS] Master Setup Complete.')",
        "    print('Run: pip install -r requirements.txt')",
        "    print('Run: python main.py')",
        "",
        "if __name__ == '__main__':",
        "    create_project()"
    ])
    
    return "\n".join(script)
