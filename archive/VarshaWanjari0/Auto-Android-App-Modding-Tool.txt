def get_device_abi():
    """Detects the current device's primary ABI."""
    try:
        # standard android property
        res = subprocess.run(["getprop", "ro.product.cpu.abi"], capture_output=True, text=True)
        abi = res.stdout.strip()
        if abi: return abi
    except: pass
    
    # Fallback to python platform check
    machine = platform.machine().lower()
    if "aarch64" in machine: return "arm64-v8a"
    if "armv7" in machine: return "armeabi-v7a"
    if "x86_64" in machine: return "x86_64"
    if "i686" in machine or "x86" in machine: return "x86"
    return "unknown"

    if not os.path.exists(bin_dir): os.makedirs(bin_dir, exist_ok=True)

            data = json.loads(res.read().decode())
            tag_name = data.get("tag_name", "Unknown")
            assets = data.get("assets", [])
            log(f"Detected latest release: [bold]{tag_name}[/]")

        log_err(f"GitHub API failed: {e}")

    tools = ["smali", "baksmali"]
    for tool in tools:
        log(f"Searching for [bold]{tool}[/] asset...")
        
        # Filter assets for this specific tool
        candidates = []
        for asset in assets:
            name = asset['name'].lower()
            if not name.endswith(".jar"): continue
            if "javadoc" in name or "sources" in name: continue
            
            # Ensure we don't grab 'baksmali' when looking for 'smali'
            if tool == "smali" and "baksmali" in name: continue
            if tool == "baksmali" and "baksmali" not in name: continue

        selected_asset = None

        if fat_candidates:

def file_selector(start_path=HOME_DIR, title="Select a File"):
    if not os.path.exists(start_path):
        start_path = HOME_DIR

    current_path = os.path.abspath(start_path)
    selected_index = 0
    items = []

    def get_items():
        nonlocal items
        try:
            all_items = sorted(os.listdir(current_path))
            dirs = [d for d in all_items if os.path.isdir(os.path.join(current_path, d))]
            files = [f for f in all_items if os.path.isfile(os.path.join(current_path, f))]
            items = [".."] + dirs + files
        except PermissionError:
            items = [".."]
        except OSError:
            items = [".."]

    get_items()

    bindings = KeyBindings()

    @bindings.add('up')
    def _(event):
        nonlocal selected_index
        selected_index = (selected_index - 1) % len(items)

    @bindings.add('down')
    def _(event):
        nonlocal selected_index
        selected_index = (selected_index + 1) % len(items)

    @bindings.add('enter')
    def _(event):
        nonlocal current_path, selected_index
        selection = items[selected_index]
        
        if selection == "..":
            parent = os.path.dirname(current_path)
            if parent == current_path: return 
            current_path = parent
            selected_index = 0
            get_items()
            return

        full_path = os.path.join(current_path, selection)
        
        if os.path.isdir(full_path):
            current_path = full_path
            selected_index = 0
            get_items()
        else:
            event.app.exit(result=full_path)

    @bindings.add('c-c')
    def _(event):
        event.app.exit(result=None)

    def get_content():
        lines = []
        lines.append(("class:title", f" {title} \n"))
        lines.append(("class:path", f" Path: {current_path} \n\n"))
        
        max_rows = 15
        start_idx = 0
        if selected_index > max_rows // 2:
            start_idx = selected_index - (max_rows // 2)
        
        end_idx = start_idx + max_rows
        display_items = items[start_idx:end_idx]

        for i, item in enumerate(display_items):
            actual_idx = start_idx + i
            full_p = os.path.join(current_path, item)
            is_dir = os.path.isdir(full_p) or item == ".."
            
            prefix = "> " if actual_idx == selected_index else "  "
            
            if actual_idx == selected_index: style = "class:selected"
            elif is_dir: style = "class:dir"
            else:
                style = "class:file"
                if item.endswith(".so"): style = "class:file-so"
                elif item.endswith(".js"): style = "class:file-js"
                elif item.endswith(".apk") or item.endswith(".xapk"): style = "class:file-apk"
            
            icon = "📁" if is_dir else "📄"
            if item == "..": icon = "⬆️ "
            
            lines.append((style, f"{prefix}{icon} {item}\n"))
        
        lines.append(("", "\n [ENTER] Open/Select  [Up/Down] Navigate  [Ctrl+C] Cancel"))
        return lines

    style = PTStyle.from_dict({
        'title': '#00ffff underline bold',
        'path': '#ffff00 bold',
        'selected': 'reverse #00ff00',
        'dir': '#ffffff bold',
        'file': '#dddddd',
        'file-so': '#00ff00 bold',
        'file-js': '#ff0000 bold',
        'file-apk': '#00afff bold',
    })

    layout = Layout(HSplit([
        Window(FormattedTextControl(get_content), height=None)
    ]))

    app = Application(layout=layout, key_bindings=bindings, full_screen=True, style=style)
    return app.run()

def hybrid_menu(menu_title, options, show_warning=False):
    bindings = KeyBindings()
    selected_index = 0

    banner_obj = get_banner_text()
    temp_console = Console(width=60, file=open(os.devnull, "w"), force_terminal=True, theme=custom_theme)

    def get_formatted_text():
        result = []
        result.append(("class:title", f" {menu_title} \n\n"))
        
        for i, option in enumerate(options):
            prefix = " > " if i == selected_index else "   "
            style = "class:selected" if i == selected_index else "class:text"
            result.append((style, f"{prefix}{i+1}. {option} \n"))
        
        if show_warning:
            result.append(("", "\n"))
            warn_msg = "Please run 'Install and Update' if running for the first time."
            result.append(("class:warning", warn_msg))
            
        return result

    @bindings.add('up')
    def _(event):
        nonlocal selected_index
        selected_index = (selected_index - 1) % len(options)

    @bindings.add('down')
    def _(event):
        nonlocal selected_index
        selected_index = (selected_index + 1) % len(options)

    @bindings.add('enter')
    def _(event):
        event.app.exit(result=selected_index)

    @bindings.add('c-c')
    def _(event):
        event.app.exit(result=None)

        Window(FormattedTextControl(ANSI(banner_ansi_string)), height=12),

        'selected': 'reverse #00ff00',
        'text': '#ffffff',
        'warning': '#ff0000 bold'

    layout = Layout(body)
    app = Application(layout=layout, key_bindings=bindings, full_screen=True, style=style)
    return app.run()

def select_architectures():
    options = ["arm64-v8a", "armeabi-v7a", "x86_64", "x86"]
    device_abi = get_device_abi()
    
    print_banner()
    console.print("[warning]Select Target Architectures for Injection:[/]")

        console.print(f"  [bold cyan]{i+1}.[/] {opt}{marker}")
        
    console.print("\n[dim]Enter numbers separated by comma (e.g. 1,2)[/]")
    console.print(f"[dim]Default (Enter) = {device_abi} & armeabi-v7a[/]")
    
    selection = console.input("[input]Selection: [/]").strip()
    
    selected_archs = []
    if not selection:
        # Default smart selection: Device ABI + 32-bit fallback
        selected_archs.append(device_abi)
        if device_abi == "arm64-v8a": selected_archs.append("armeabi-v7a")
        # Ensure we don't have duplicates if device_abi was already v7a
        return list(set(selected_archs))
        
    indices = [int(x.strip()) - 1 for x in selection.split(",") if x.strip().isdigit()]
    for idx in indices:
        if 0 <= idx < len(options):
            selected_archs.append(options[idx])
            
    return selected_archs

def get_major_library(apk_path):
    # 1. Extract Real App Label using aapt2
    aapt2 = get_aapt2_bin()
    app_label = ""
    
    try:
        # Dump badging to get the label
        res = subprocess.run([aapt2, "dump", "badging", apk_path], capture_output=True, text=True)
        # Regex to find application-label:'Something'
        match = re.search(r"application-label:'([^']+)'", res.stdout)
        if match:
            app_label = match.group(1)
    except Exception as e:
        log_warn(f"Could not extract app label: {e}")

    candidate_libs = []
    
    # 2. Generate Candidates if label exists
    if app_label:
        # Sanitize: "My Game!" -> "MyGame" (Alphanumeric only)
        clean_label = "".join(x for x in app_label if x.isalnum())
        
        if clean_label:
            # Case 1: Exact CamelCase (libExample.so)
            candidate_libs.append(f"lib{clean_label}.so")
            # Case 2: All Lowercase (libexample.so)
            candidate_libs.append(f"lib{clean_label.lower()}.so")

    # 3. Standard Major Libs (Fallback)
    major_libs = [
        "libil2cpp.so", "libunity.so", "libmain.so", "libmonosgen-2.0.so",
        "libflutter.so", "libapp.so", "libcocos2dcpp.so", "libcocos2djs.so",
        "libgUE4.so", "libue4.so", "libgodot_android.so", "libreactnativejni.so", 
        "libgame.so", "libnative-lib.so", "libgl2jni.so", "libgdx.so"
    ]
    
    try:
        with zipfile.ZipFile(apk_path, 'r') as z:
            files = z.namelist()
            
            # A. Check for Label-based libs (Priority)
            if candidate_libs:
                log(f"App Label: [cyan]{app_label}[/] -> Searching for: [dim]{', '.join(candidate_libs)}[/]")
                for f in files:
                    base = os.path.basename(f)
                    # Check in lib folders
                    if (f.startswith("lib/") and f.endswith(".so")):
                        if base in candidate_libs:
                            log(f"Found app-specific match: [bold green]{base}[/]")
                            return base

            # B. Fallback to Major Game Engines
            log("App-specific lib not found. Checking major game engines...")
            for f in files:
                base = os.path.basename(f)
                if (f.startswith("lib/") and f.endswith(".so")):
                    if base in major_libs:
                        return base
            
    except Exception as e:
        log_err(f"Library scan failed: {e}")
        
    return None

def determine_strategy(apk_path):
    options = ["Auto (Recommended)", "Patchelf (Native - Fast)", "Smali Patch (Dex - Reliable,slow)"]
    choice = hybrid_menu("Select Injection Method:", options)

    if choice == 0: 
        log("Auto-detecting best method...")
        target_so = get_major_library(apk_path)
        if target_so:
            log(f"Major library detected: [input]{target_so}[/]. Using Patchelf.")
            return "NATIVE", target_so
        else:
            log("No major library found. Using Smali Patch.")
            return "SMALI", None
    elif choice == 1: 
        target_so = get_major_library(apk_path)
        if not target_so:
            log_warn("No native libraries found. Falling back to Smali.")
            time.sleep(2)
            return "SMALI", None
        return "NATIVE", target_so
    else:
        return "SMALI", None

def force_inject_libs(apk_path, payload_info, selected_archs):
    log(f"Forcing library injection into: {selected_archs}")

    with zipfile.ZipFile(apk_path, 'a') as zf:
        for arch in selected_archs:
            src_file = payload_info['path']
            
            if "libfrida" in payload_info['name']:
                arch_specific_path = os.path.join(UAMT_ROOT, f"libfrida-{arch}.so")
                if os.path.exists(arch_specific_path):
                    src_file = arch_specific_path
                    log(f"  Using specific gadget: libfrida-{arch}.so")
            
            dest_path = f"lib/{arch}/{payload_info['name']}"
            log(f"    Adding [dim]{dest_path}[/]")
            try:
                zf.write(src_file, dest_path, compress_type=zipfile.ZIP_STORED)
            except Exception as e:
                log_warn(f"Failed to add lib to {arch}: {e}")
            
            for extra in payload_info.get('extra_files', []):
                extra_dest = f"lib/{arch}/{extra['name']}"
                try:
                    zf.write(extra["src"], extra_dest, compress_type=zipfile.ZIP_STORED)
                except: pass

def inject_native(apk_path, target_so, payload_info, selected_archs):
    log(f"Strategy: Native Injection (Target: [input]{target_so}[/])")
    
    work_dir = os.path.join(UAMT_ROOT, "native_work")
    if os.path.exists(work_dir): shutil.rmtree(work_dir)
    os.makedirs(work_dir)
    
    safe_output_name = "patched_" + os.path.basename(apk_path)
    temp_output_apk = os.path.join(UAMT_ROOT, safe_output_name)
    final_output_apk = os.path.join("/sdcard/UAMT", safe_output_name)
    if os.path.exists(temp_output_apk): os.remove(temp_output_apk)

    log("Locating target libraries...")
    targets_map = {} 
    found_archs = set()
    
    with zipfile.ZipFile(apk_path, 'r') as zin:
        for item in zin.infolist():
            arch_match = False
            for arch in selected_archs:
                if item.filename.startswith(f"lib/{arch}/") and item.filename.endswith(f"/{target_so}"):
                    arch_match = True
                    # FIX: correctly extract architecture from path (lib/ARCH/file.so)
                    current_arch = item.filename.split("/")[1]
                    found_archs.add(current_arch)
                    break
            
            if arch_match:
                temp_so_path = os.path.join(work_dir, item.filename.replace("/", "_"))
                with open(temp_so_path, 'wb') as f: f.write(zin.read(item.filename))
                
                log(f"Patching [input]{item.filename}[/]")
                try:
                    subprocess.run(["patchelf", "--add-needed", payload_info['name'], temp_so_path], check=True, stderr=subprocess.DEVNULL)
                    targets_map[item.filename] = temp_so_path
                except: log_warn(f"Failed to patch {item.filename}")
    
    # Notify user if selected arch was not found
    for arch in selected_archs:
        if arch not in found_archs:
            log_warn(f"Target library ({target_so}) NOT found in [bold]{arch}[/]. Skipping patch for this arch.")

    if not targets_map:
        log_err("Target library not found in any selected architectures.")
        return False

    log("Streaming APK & Injecting...")
    with zipfile.ZipFile(apk_path, 'r') as zin, zipfile.ZipFile(temp_output_apk, 'w') as zout:
        for item in zin.infolist():
            if item.filename.startswith("META-INF/"): continue
            if item.filename in targets_map:
                zout.write(targets_map[item.filename], item.filename, compress_type=zipfile.ZIP_STORED)
            else:
                compression = zipfile.ZIP_DEFLATED
                if item.filename == "resources.arsc" or item.filename.endswith(".so"):
                    compression = zipfile.ZIP_STORED
                zout.writestr(item.filename, zin.read(item.filename), compress_type=compression)

    force_inject_libs(temp_output_apk, payload_info, selected_archs)
    finalize_apk(temp_output_apk, final_output_apk)
    return True

def inject_smali(apk_path, payload_info, selected_archs):
    log("Strategy: Smali Injection (Reliable/Slow)")
    
    # Check for required tools
    for tool in ["baksmali", "smali", "aapt2", "zip", "unzip"]:
        if not shutil.which(tool):
            log_err(f"Tool '{tool}' not found! Run 'Install and Update'.")
            return False

    aapt2_exe = get_aapt2_bin()
    
    # 1. Get Main Activity using aapt2
    res = subprocess.run([aapt2_exe, "dump", "badging", apk_path], capture_output=True, text=True)
    match = re.search(r"launchable-activity: name='([^']+)'", res.stdout)
    if not match: 
        log_err("Could not find launchable activity.")
        return False
    
    activity_name = match.group(1)
    # Convert 'com.example.Game' to 'Lcom/example/Game;' for binary search
    activity_descriptor = "L" + activity_name.replace(".", "/") + ";"
    log(f"Target Activity: [input]{activity_name}[/]")
    
    # 2. Setup Temporary Workspace
    work_dir = os.path.join(UAMT_ROOT, "smali_work")
    if os.path.exists(work_dir): shutil.rmtree(work_dir)
    os.makedirs(work_dir)

    # 3. Extract Dex files
    log("Extracting Dex files...")
    dex_files = []
    try:
        with zipfile.ZipFile(apk_path, 'r') as z:
            for name in z.namelist():
                if name.startswith("classes") and name.endswith(".dex"):
                    z.extract(name, work_dir)
                    dex_files.append(name)
    except Exception as e:
        log_err(f"Extraction failed: {e}")
        return False

    # 4. Smart Scan: Find which Dex contains the Activity
    target_dex = None
    log("Scanning Dex files for target class...")
    
    for dex in dex_files:
        dex_path = os.path.join(work_dir, dex)
        try:
            with open(dex_path, 'rb') as f:
                content = f.read()
                # Search for the class descriptor byte signature
                if activity_descriptor.encode('utf-8') in content:
                    target_dex = dex
                    log(f"Found target class in: [bold green]{dex}[/]")
                    break
        except: pass
    
    if not target_dex:
        # Fallback: If binary scan fails (rare), default to classes.dex or error
        log_warn("Could not identify specific Dex. Defaulting to classes.dex")
        if "classes.dex" in dex_files: target_dex = "classes.dex"
        else: log_err("Analysis failed."); return False

    # 5. Decompile ONLY the target Dex
    log(f"Decompiling {target_dex}...")
    smali_dir = os.path.join(work_dir, "smali_out")
    dex_path = os.path.join(work_dir, target_dex)
    
    cmd = ["baksmali", "d", dex_path, "-o", smali_dir]
    if not run_with_progress(cmd, f"Processing {target_dex}"): return False

    # 6. Inject Code
    smali_rel = activity_name.replace(".", "/") + ".smali"
    target_smali_path = None
    
    for root, dirs, files in os.walk(smali_dir):
        if smali_rel.split("/")[-1] in files:
            potential = os.path.join(root, smali_rel.split("/")[-1])
            if activity_name.replace(".", os.sep) in potential:
                target_smali_path = potential
                break
    
    if target_smali_path:
        log(f"Injecting into [input]{os.path.basename(target_smali_path)}[/]...")
        lib_name = payload_info['name'].replace("lib", "").replace(".so", "")
        
        try:
            with open(target_smali_path, 'r') as f: lines = f.readlines()
            new_lines = []
            in_on_create = False
            injected = False
            
            payload = [
                f'\n    const-string v0, "{lib_name}"',
                '\n    invoke-static {v0}, Ljava/lang/System;->loadLibrary(Ljava/lang/String;)V\n'
            ]

            for line in lines:
                stripped = line.strip()
                if ".method" in stripped and "onCreate" in stripped and "Landroid/os/Bundle;" in stripped:
                    in_on_create = True
                    new_lines.append(line)
                    continue
                
                if in_on_create and ".end method" in stripped:
                    in_on_create = False

                if in_on_create and not injected:
                    if stripped.startswith(".locals"):
                        parts = stripped.split()
                        if len(parts) > 1 and int(parts[1]) == 0:
                            new_lines.append("    .locals 1\n")
                        else:
                            new_lines.append(line)
                        new_lines.extend(payload)
                        injected = True
                        continue
                    elif stripped.startswith(".registers"):
                        new_lines.append(line)
                        new_lines.extend(payload)
                        injected = True
                        continue

                new_lines.append(line)
            
            with open(target_smali_path, 'w') as f: f.writelines(new_lines)

        except Exception as e:
            log_err(f"Injection error: {e}")
            return False
    else:
        log_err("Activity Smali file not found in decompiled output.")
        return False

    # 7. Recompile ONLY the target Dex
    log(f"Recompiling {target_dex}...")
    cmd = ["smali", "a", smali_dir, "-o", dex_path]
    if not run_with_progress(cmd, "Building Dex"): return False

    # 8. Update APK
    log("Updating Target APK...")
    safe_output_name = "patched_" + os.path.basename(apk_path)
    temp_output_apk = os.path.join(UAMT_ROOT, safe_output_name)
    final_output_apk = os.path.join("/sdcard/UAMT", safe_output_name)
    
    if os.path.exists(temp_output_apk): os.remove(temp_output_apk)
    shutil.copy(apk_path, temp_output_apk)
    
    try:
        # Update only the single modified dex file
        cwd = os.getcwd()
        os.chdir(work_dir)
        subprocess.run(["zip", "-u", "-j", temp_output_apk, target_dex], check=True, stdout=subprocess.DEVNULL)
        os.chdir(cwd)
    except Exception as e:
        log_err(f"Failed to update APK zip: {e}")
        return False

    force_inject_libs(temp_output_apk, payload_info, selected_archs)
    finalize_apk(temp_output_apk, final_output_apk)
    return True

def finalize_apk(temp_path, final_path):
    log("Zipaligning...")
    aligned = temp_path.replace(".apk", "_aligned.apk")
    subprocess.run(["zipalign", "-p", "-f", "4", temp_path, aligned], capture_output=True)
    shutil.move(aligned, temp_path)

    if not os.path.exists(KEYSTORE_PATH):
        subprocess.run(["keytool", "-genkey", "-v", "-keystore", KEYSTORE_PATH, "-storepass", "android", "-alias", "androiddebugkey", "-keypass", "android", "-keyalg", "RSA", "-keysize", "2048", "-validity", "10000", "-dname", "CN=Android Debug,O=Android,C=US"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

    log("Signing...")
    cmd = ["apksigner", "sign", "--ks", KEYSTORE_PATH, "--ks-pass", "pass:android", temp_path]
    subprocess.run(cmd, capture_output=True)
    
    try:
        if not os.path.exists(os.path.dirname(final_path)): os.makedirs(os.path.dirname(final_path))
        shutil.copy(temp_path, final_path)
        log(f"Done! Saved to: [input]{final_path}[/]")
    except:
        log(f"Done! Saved internally: {temp_path}")
