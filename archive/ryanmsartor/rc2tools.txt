# rc2tools
A collection of tools for hacking Rolan's Curse 2 for Game Boy

## Extract character stats

Requires Python 3.10 or later and Bash. From the repository root, run:

```sh
./rc2t extract-json
```

This reads `input/Rolan's Curse II (USA).gb` and writes
`output/<rom-sha256>.json`, using the lowercase SHA-256 hash of the ROM bytes
as the filename. A different ROM revision therefore gets a different output
file.
Alternatively, supply a ROM or a complete 16-byte-per-row `.hex` dump produced
by `xxd` or PowerShell `Format-Hex`:

```sh
./rc2t extract-json "input/Rolan's Curse II (USA).gb.hex"
```

The command reads the mapped bytes from the supplied input, so modified stat
values are reflected in the output. It checks the ROM size and header, not a
fixed hash or expected stat values. Arrays contain decimal byte values for
levels 1 through 16; the mappings and their unresolved interpretations are
documented in [the ROM map](docs/rom-map.md). The output file is overwritten
if a file with that ROM hash already exists.

## Render graphics as assembled sprites

Rendering requires Pillow 10.0 or later. Install it with:

```sh
python3 -m pip install 'Pillow>=10.0'
```

To render the entire default ROM (including non-graphics data) to
`output/graphics.png`, run:

```sh
./rc2t render-graphics
```

To render from a start offset through the end of the ROM, run:

```sh
./rc2t render-graphics 0x8050
```

This also writes to `output/graphics.png`.

To render an inclusive-start, exclusive-end ROM byte range as a sheet of
assembled sprites, pass the range and output path:

```sh
./rc2t render-graphics 0x9106 0xB566 output/hero-sprites.png
```

Offsets are physical ROM offsets. Each 16-byte tile is decoded as 8×8 pixels;
every four consecutive tiles are assembled into one 16×16 sprite in
top-left, bottom-left, top-right, bottom-right input order. This column-first
layout was observed in the Hero sprite and is applied uniformly to the entire
range as an experiment. Any final group with fewer than four tiles is rendered
with missing quadrants left blank. A one-pixel gutter separates sprites.
The selected range may end at any byte, but must contain at least one byte. If
it ends in the middle of a pixel row, the missing byte is zero-padded; missing
rows in the final 16-byte tile are left blank.
The rendered sheet labels each column along the top with the address where its
first cel begins, and each row along the left with the address where its first
cel begins. Each cel uses 64 bytes; each row's left-axis address advances by
one full row (`columns * 64` bytes).

The default sheet uses 16 sprites per row. Its grayscale palette displays
transparent color index 0 as black, while the inverted non-transparent
palette displays index 3 as white.
The output extension can be `.png`, `.jpg`, or `.jpeg`. Options include
`--input` for a different ROM or full hex dump, `--columns` for sprites per
row, and `--scale` for nearest-neighbor enlargement. The specified output
file is overwritten if it exists. You can also invoke
`python3 bin/render_graphics.py` directly with the same range, output path,
and options. When run without a range, `./rc2t render-graphics` applies this
grouping to the entire ROM, including non-graphics data.

#!/usr/bin/env python3
"""Extract documented character tables from a ROM or its hex dump."""

import argparse
import hashlib
import json
import re
import sys
from pathlib import Path

ROM_SIZE = 0x20000
LEVELS = 16
CHARACTERS = ("hero", "pit", "lane", "kyle", "jude", "lee", "lora", "zold")

# Offsets and record fields are documented in docs/rom-map.md.
ABILITIES = {
    "hero": (
        ("a", 0x15449, ("damage",)),
        ("b", 0x15459, ("cost", "range", "damage")),
    ),
    "pit": (
        ("dagger", 0x154AD, ("range", "damage")),
        ("boomerang", 0x154DF, ("range", "damage")),
        ("bomb", 0x15511, ("cost", "damage", "unknown")),
    ),
    "lane": (
        ("a", 0x15709, ("damage",)),
        ("shield", 0x15719, ("cost", "unknown")),
        ("healing", 0x15799, ("cost", "unknown_1", "unknown_2")),
    ),
    "kyle": (
        ("a", 0x157C9, ("damage",)),
        ("play_dead", 0x157D9, ("cost", "unknown")),
        ("rod", 0x15898, ("cost", "damage")),
    ),
    "jude": (
        ("weak_a", 0x159B2, ("range", "damage")),
        ("strong_a", 0x159E4, ("range", "damage")),
        ("quake", 0x15A16, ("cost", "damage")),
    ),
    "lee": (
        ("a", 0x15B97, ("damage",)),
        ("rapidfire", 0x15BA7, ("cost", "damage")),
        ("firework", 0x15BC7, ("cost", "damage", "range")),
    ),
    "lora": (
        ("a", 0x15C34, ("range", "damage")),
        ("heal", 0x15C66, ("cost", "unknown_1", "unknown_2")),
        ("wing", 0x15C96, ("cost",)),
    ),
    "zold": (
        ("a", 0x15CB6, ("range", "damage")),
        ("b", 0x15CE8, ("cost", "damage")),
    ),
}

ADDITIONAL_TABLES = {
    "zold_a_damage": (0x15CA6, LEVELS, 1),
    "lookup_0x17165": (0x17165, 8, 2),
    "lookup_0x17175": (0x17175, 8, 2),
}

XXD_ROW = re.compile(
    r"^([0-9a-fA-F]{8}): ((?:[0-9a-fA-F]{4} ){7}[0-9a-fA-F]{4})  "
)
FORMAT_HEX_ROW = re.compile(
    r"^([0-9a-fA-F]{8})[ \t]+((?:[0-9a-fA-F]{2}[ \t]+){15}"
    r"[0-9a-fA-F]{2})(?:[ \t]|$)"
)
ROW_PREFIX = re.compile(r"^[0-9a-fA-F]{8}(?:[ \t]|:)")


def parse_hex_dump(raw: bytes) -> bytes:
    """Parse contiguous, full-width xxd or PowerShell Format-Hex rows."""
    encoding = "utf-16" if raw.startswith((b"\xff\xfe", b"\xfe\xff")) else "utf-8-sig"
    text = raw.decode(encoding)
    data = bytearray()
    for line_number, line in enumerate(text.split("\n"), 1):
        match = XXD_ROW.match(line) or FORMAT_HEX_ROW.match(line)
        if match:
            offset = int(match.group(1), 16)
            if offset != len(data):
                raise ValueError(
                    f"hex dump line {line_number}: expected offset "
                    f"0x{len(data):08X}, found 0x{offset:08X}"
                )
            data.extend(bytes.fromhex(match.group(2)))
        elif ROW_PREFIX.match(line):
            raise ValueError(f"invalid hex dump row at line {line_number}")
    if not data:
        raise ValueError("hex dump contains no recognizable rows")
    return bytes(data)


def read_rom(path: Path) -> bytes:
    raw = path.read_bytes()
    rom = parse_hex_dump(raw) if path.suffix.lower() == ".hex" else raw
    if len(rom) != ROM_SIZE:
        raise ValueError(f"expected a {ROM_SIZE}-byte ROM, found {len(rom)} bytes")
    if rom[0x134:0x143] != b"ROLAN'S CURSE 2" or rom[0x147:0x149] != b"\x06\x02":
        raise ValueError("ROM header does not match Rolan's Curse II (USA), MBC2, 128 KiB")
    return rom


def extract_json(rom: bytes) -> dict:
    """Read each mapped record directly from ROM, in level order."""
    characters = {}
    for index, name in enumerate(CHARACTERS):
        hp_start = 0x17065 + index * LEVELS
        mp_start = 0x170E5 + index * LEVELS
        character = {
            "hp": list(rom[hp_start:hp_start + LEVELS]),
            "mp": list(rom[mp_start:mp_start + LEVELS]),
            "abilities": {},
        }
        for ability, offset, fields in ABILITIES[name]:
            width = len(fields)
            character["abilities"][ability] = {
                field: [rom[offset + level * width + column] for level in range(LEVELS)]
                for column, field in enumerate(fields)
            }
        characters[name] = character
    additional_tables = {}
    for name, (offset, entries, width) in ADDITIONAL_TABLES.items():
        if width == 1:
            additional_tables[name] = list(rom[offset:offset + entries])
        else:
            additional_tables[name] = [
                list(rom[offset + index * width:offset + (index + 1) * width])
                for index in range(entries)
            ]
    return {"characters": characters, "additional_tables": additional_tables}


def output_path(root: Path, rom: bytes) -> Path:
    """Return the output path named by the source ROM's SHA-256."""
    digest = hashlib.sha256(rom).hexdigest()
    return root / "output" / f"{digest}.json"


def main() -> int:
    root = Path(__file__).resolve().parent.parent
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "input",
        nargs="?",
        type=Path,
        default=root / "input" / "Rolan's Curse II (USA).gb",
        help="ROM (.gb) or full xxd/Format-Hex dump (.hex)",
    )
    args = parser.parse_args()
    try:
        rom = read_rom(args.input)
        game = extract_json(rom)
        output = output_path(root, rom)
        output.parent.mkdir(parents=True, exist_ok=True)
        output.write_text(json.dumps(game, indent=2) + "\n", encoding="utf-8", newline="\n")
    except (OSError, UnicodeError, ValueError) as error:
        print(f"Error: {error}", file=sys.stderr)
        return 1
    print(f"Wrote {output}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env python3
"""Render a ROM byte range as assembled 16x16 Game Boy sprites."""

import argparse
import math
import sys
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

from extract_json import read_rom

TILE_BYTES = 16
TILE_WIDTH = 8
SPRITE_TILE_POSITIONS = ((0, 0), (0, 1), (1, 0), (1, 1))
SPRITE_WIDTH = TILE_WIDTH * 2
SPRITE_GUTTER = 1
GUTTER_COLOR = (192, 192, 192)
DEFAULT_ROM = Path(__file__).resolve().parent.parent / "input" / "Rolan's Curse II (USA).gb"
DEFAULT_PALETTE = (
    (0, 0, 0),
    (85, 85, 85),
    (170, 170, 170),
    (255, 255, 255),
)


def parse_offset(value: str) -> int:
    """Parse an offset written in decimal or with a 0x prefix."""
    try:
        return int(value, 0)
    except ValueError as error:
        raise argparse.ArgumentTypeError(f"invalid offset: {value}") from error


def decode_tiles(data: bytes) -> list[list[int]]:
    """Decode Game Boy 2bpp tiles, allowing a final tile to end at any byte."""
    if not data:
        raise ValueError("range must contain at least one byte")

    pixels = []
    for tile_start in range(0, len(data), TILE_BYTES):
        tile_data = data[tile_start:tile_start + TILE_BYTES].ljust(TILE_BYTES, b"\x00")
        tile = []
        for row in range(TILE_WIDTH):
            low, high = tile_data[row * 2:row * 2 + 2]
            tile.append([
                ((low >> bit) & 1) | (((high >> bit) & 1) << 1)
                for bit in range(7, -1, -1)
            ])
        pixels.append(tile)
    return pixels


def add_address_axes(
    image: Image.Image,
    sprite_count: int,
    columns: int,
    start_address: int,
    scale: int,
) -> Image.Image:
    """Add ROM address labels for the first cel in each column and row."""
    rows = math.ceil(sprite_count / columns)
    column_addresses = [
        f"0x{start_address + column * 4 * TILE_BYTES:04X}"
        for column in range(min(columns, sprite_count))
    ]
    row_addresses = [
        f"0x{start_address + row * columns * 4 * TILE_BYTES:04X}"
        for row in range(rows)
    ]
    font = ImageFont.load_default()
    column_bounds = [font.getbbox(address) for address in column_addresses]
    row_bounds = [font.getbbox(address) for address in row_addresses]
    column_label_width = max(right - left for left, _, right, _ in column_bounds)
    row_label_width = max(right - left for left, _, right, _ in row_bounds)
    label_height = max(bottom - top for _, top, _, bottom in column_bounds + row_bounds)

    cell_size = SPRITE_WIDTH * scale
    source_pitch = (SPRITE_WIDTH + SPRITE_GUTTER) * scale
    column_pitch = max(source_pitch, column_label_width + 8)
    row_pitch = max(source_pitch, label_height + 8)
    top_margin = label_height + 8
    left_margin = row_label_width + 8
    graphic_width = columns * cell_size + (columns - 1) * (column_pitch - cell_size)
    graphic_height = rows * cell_size + (rows - 1) * (row_pitch - cell_size)
    output = Image.new(
        "RGB",
        (left_margin + graphic_width, top_margin + graphic_height),
        (255, 255, 255),
    )

    for sprite_index in range(sprite_count):
        source_x = sprite_index % columns * source_pitch
        source_y = sprite_index // columns * source_pitch
        sprite = image.crop((source_x, source_y, source_x + cell_size, source_y + cell_size))
        target_x = left_margin + sprite_index % columns * column_pitch
        target_y = top_margin + sprite_index // columns * row_pitch
        output.paste(sprite, (target_x, target_y))

    draw = ImageDraw.Draw(output)
    for column, address in enumerate(column_addresses):
        left, top, right, bottom = font.getbbox(address)
        center_x = left_margin + column * column_pitch + cell_size / 2
        draw.text(
            (center_x - (right - left) / 2 - left, (top_margin - (bottom - top)) / 2 - top),
            address,
            fill=(0, 0, 0),
            font=font,
        )
    for row, address in enumerate(row_addresses):
        left, top, right, bottom = font.getbbox(address)
        center_y = top_margin + row * row_pitch + cell_size / 2
        draw.text(
            ((left_margin - (right - left)) / 2 - left, center_y - (bottom - top) / 2 - top),
            address,
            fill=(0, 0, 0),
            font=font,
        )

    for column in range(1, columns):
        x = left_margin + column * column_pitch - SPRITE_GUTTER * scale
        draw.line((x, top_margin, x, top_margin + graphic_height - 1), fill=GUTTER_COLOR)
    for row in range(1, rows):
        y = top_margin + row * row_pitch - SPRITE_GUTTER * scale
        draw.line((left_margin, y, left_margin + graphic_width - 1, y), fill=GUTTER_COLOR)
    return output


def render_sprites(
    data: bytes,
    columns: int = 16,
    scale: int = 4,
    palette: tuple[tuple[int, int, int], ...] = DEFAULT_PALETTE,
    start_address: int | None = None,
) -> Image.Image:
    """Render groups of four column-first tiles as 16x16 sprites."""
    if columns < 1:
        raise ValueError("columns must be at least 1")
    if scale < 1:
        raise ValueError("scale must be at least 1")
    if len(palette) != 4:
        raise ValueError("palette must contain exactly four colors")

    tiles = decode_tiles(data)
    sprite_count = math.ceil(len(tiles) / len(SPRITE_TILE_POSITIONS))
    rows = math.ceil(sprite_count / columns)
    width = columns * SPRITE_WIDTH + (columns - 1) * SPRITE_GUTTER
    height = rows * SPRITE_WIDTH + (rows - 1) * SPRITE_GUTTER
    image = Image.new("RGB", (width, height), palette[0])
    draw = ImageDraw.Draw(image)
    for column in range(1, columns):
        x = column * (SPRITE_WIDTH + SPRITE_GUTTER) - SPRITE_GUTTER
        draw.line((x, 0, x, height - 1), fill=GUTTER_COLOR)
    for row in range(1, rows):
        y = row * (SPRITE_WIDTH + SPRITE_GUTTER) - SPRITE_GUTTER
        draw.line((0, y, width - 1, y), fill=GUTTER_COLOR)
    pixels = image.load()
    for tile_index, tile in enumerate(tiles):
        sprite_index, tile_position = divmod(tile_index, len(SPRITE_TILE_POSITIONS))
        sprite_column, sprite_row = sprite_index % columns, sprite_index // columns
        tile_column, tile_row = SPRITE_TILE_POSITIONS[tile_position]
        tile_x = sprite_column * (SPRITE_WIDTH + SPRITE_GUTTER) + tile_column * TILE_WIDTH
        tile_y = sprite_row * (SPRITE_WIDTH + SPRITE_GUTTER) + tile_row * TILE_WIDTH
        for y, row in enumerate(tile):
            for x, color in enumerate(row):
                pixels[tile_x + x, tile_y + y] = palette[color]

    if scale > 1:
        image = image.resize((image.width * scale, image.height * scale), Image.Resampling.NEAREST)
    if start_address is not None:
        image = add_address_axes(image, sprite_count, columns, start_address, scale)
    return image


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("start", type=parse_offset, help="start ROM offset (inclusive)")
    parser.add_argument("end", type=parse_offset, help="end ROM offset (exclusive)")
    parser.add_argument("output", type=Path, help="output .png, .jpg, or .jpeg path")
    parser.add_argument(
        "--input",
        type=Path,
        default=DEFAULT_ROM,
        help="ROM or full xxd/Format-Hex dump (default: %(default)s)",
    )
    parser.add_argument("--columns", type=int, default=16, help="sprites per row (default: 16)")
    parser.add_argument("--scale", type=int, default=4, help="integer nearest-neighbor scale (default: 4)")
    args = parser.parse_args()

    try:
        if args.output.suffix.lower() not in (".png", ".jpg", ".jpeg"):
            raise ValueError("output extension must be .png, .jpg, or .jpeg")
        if args.start < 0 or args.end <= args.start:
            raise ValueError("range must have a nonnegative start and an end greater than the start")

        rom = read_rom(args.input)
        if args.end > len(rom):
            raise ValueError(f"range end 0x{args.end:X} exceeds ROM size 0x{len(rom):X}")
        data = rom[args.start:args.end]
        image = render_sprites(data, args.columns, args.scale, start_address=args.start)
        args.output.parent.mkdir(parents=True, exist_ok=True)
        image.save(args.output)
    except (OSError, UnicodeError, ValueError) as error:
        print(f"Error: {error}", file=sys.stderr)
        return 1

    print(f"Wrote {args.output}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

# Rolan's Curse II (USA) ROM map

This is an initial, evidence-based map of the ROM. ROM file offsets and CPU addresses are distinct, especially in the banked window. The separate [save file analysis](./save-file-analysis.md) documents observations from a hex-dumped `.sav` file; its labels are not ROM or CPU addresses.

## ROM and cartridge layout

The source examined here is `input/Rolan's Curse II (USA).gb` (131,072 bytes). The header bytes are at their standard Game Boy header offsets:

| ROM offset | Bytes | Meaning |
|---|---|---|
| `0x0100` | `00 C3 50 01` | Entry sequence: `NOP`, then jump to CPU address `0x0150`. |
| `0x0104-0x0133` | Nintendo logo data | Header logo field. |
| `0x0134-0x0143` | `ROLAN'S CURSE 2` and padding | Cartridge title. |
| `0x0143` | `00` | Non-CGB header flag. |
| `0x0147` | `06` | Cartridge type: MBC2 with battery. |
| `0x0148` | `02` | ROM size: 128 KiB (8 banks of 16 KiB). |
| `0x0149` | `00` | Header RAM-size code. MBC2 has its own small built-in RAM; see below. |

With MBC2, the CPU sees a fixed ROM window at `0x0000-0x3FFF` and a banked ROM window at `0x4000-0x7FFF`. The cartridge's RAM is the small MBC2 nibble RAM at `0xA000-0xA1FF` (512 four-bit cells); it is not a general 8 KiB SRAM. MBC2 bank selection is controlled by writes in the `0x2000-0x3FFF` range. Keep a bank number alongside any CPU address in `0x4000-0x7FFF` when documenting code or data there.

## CPU address spaces

These are hardware address windows, not interpretations of game variables:

| CPU address range | Hardware use |
|---|---|
| `0x0000-0x7FFF` | Cartridge ROM and MBC control writes. |
| `0x8000-0x9FFF` | Video RAM (tile/map data while in use). |
| `0xA000-0xBFFF` | Cartridge RAM window; for this MBC2 cartridge, only `0xA000-0xA1FF` is the MBC2 RAM area. |
| `0xC000-0xDFFF` | Work RAM. `0xE000-0xFDFF` echoes this region. |
| `0xFE00-0xFE9F` | Object attribute memory (sprite attributes). |
| `0xFF00-0xFF7F` | Hardware I/O registers. |
| `0xFF80-0xFFFE` | High RAM. |
| `0xFFFF` | Interrupt enable register. |

## Confirmed code landmarks

Offsets below are ROM file offsets that are also CPU addresses in the fixed bank:

| ROM offset / CPU address | Observation |
|---|---|
| `0x0000-0x003F` | Repeated short instruction sequences (`LD HL,imm16; ADD HL,DE; RET`) with immediate values 0 through 7. Their callers and purpose are not identified. |
| `0x0040` | VBlank vector saves AF/BC/DE/HL and jumps to `0x039C`. |
| `0x0048` | LCD STAT vector jumps to `0x0594`. |
| `0x0050`, `0x0058`, `0x0060` | Timer, serial, and joypad vectors contain `RETI` (`0xD9`). |
| `0x0100` | Reset entry jumps to `0x0150`. |
| `0x0150` | Startup begins. It calls `0x0C95`, initializes SP to `0xCBFF`, clears `0xFF80-0xFFFE`, and clears work RAM `0xC000-0xDFFF`. It also initializes other locations and calls routines at `0x0C7D`, `0x0CA2`, `0x0CA7`, and `0x0CB6`. |
| `0x01A4-0x01AD` | Startup writes `1` to `0x3FFF` (selecting ROM bank 1), then calls CPU address `0x5586`. In this call context that is physical ROM offset `0x05586`. |

These are landmarks, not a complete disassembly. In particular, routine names and responsibilities have not yet been established. Other calls in startup include `0x047C`, `0x04F6`, and `0x0511`; these are candidates for the next control-flow tracing pass.

## Bank inventory

The ROM has eight physical banks. The character tables described below are in bank 5. A graphics/tile-data region and related descriptors have been tentatively identified in bank 2; details and confidence are documented below. Other banks are not yet reliably classified by subsystem.

| Physical bank | ROM file offsets |
|---:|---|
| 0 | `0x00000-0x03FFF` |
| 1 | `0x04000-0x07FFF` |
| 2 | `0x08000-0x0BFFF` |
| 3 | `0x0C000-0x0FFFF` |
| 4 | `0x10000-0x13FFF` |
| 5 | `0x14000-0x17FFF` |
| 6 | `0x18000-0x1BFFF` |
| 7 | `0x1C000-0x1FFFF` |

The title is the only clear long ASCII string found in the initial string scan. Most game text, graphics, maps, and audio are likely stored as encoded or packed binary, so a plain-text scan cannot locate them. Some early bank bytes look table-like (for example, bank 2 starts with little-endian values `0x404E`, `0x408E`, `0x40CE`), but those are not yet proven pointers or assigned to any asset.

The tile match confirms the format and the listed sprite tiles; the surrounding descriptor tables look like graphics transfer lists, but their consuming code has not been traced. Treat their field meanings as tentative.

All source ranges below are physical ROM offsets in **bank 2**. For a CPU source pointer in the banked window, add `0x4000` to obtain the physical offset; conversely, subtract `0x4000` from the physical offset to obtain the CPU address with bank 2 selected. ## Character stat and ability tables

The right-margin annotations in [rc2.hex](./rc2.hex) identify these tables by matching values to known character stats at each level. All 8,192 rows of the dump were compared with the local ROM and their bytes matched. These are annotated identifications, not yet verified by tracing the game's table-reading code. All ranges below are inclusive physical ROM offsets in **bank 5**; subtract `0x10000` for the CPU address when bank 5 is selected.

Each listed table has 16 entries, ordered as levels 1 through 16 according to the annotations and the repeating layout. Values are raw hexadecimal bytes, not decimal: e.g. `1E` is decimal 30. For multi-byte records, a slash separates fields within one level; spaces separate successive levels. The byte order and field labels follow the margin notes; uncertain fields are retained as such.

The `./rc2t extract-json` command reads these ranges from the input ROM and writes `output/<rom-sha256>.json`, where the lowercase SHA-256 is computed from the ROM bytes (also when the input is a hex dump). It also accepts a full ROM hex dump as an argument. Its `characters` object groups HP, MP, and ability field arrays by character; array index 0 corresponds to level 1. Each ability field is a separate array. `additional_tables` contains the standalone Zold A damage table and two paired-byte lookup tables described below. The repeated bytes in Lane's healing and Lora's heal are named `unknown_1` and `unknown_2` because their roles are uncertain; other `unknown` fields are likewise unidentified. The tentative table boundaries and field interpretations below also apply to the JSON.

### Additional tables

The Zold A damage values also appear as a standalone 16-byte array immediately before the range/damage records. The ROM has code references to this array at CPU address `0x5CA6`; its values match the damage byte of each Zold A record:

Two more arrays follow the character MP tables. Each has eight two-byte entries. These are stored in the JSON as arrays of two-byte arrays under `additional_tables`; values below are shown as `first/second` bytes. Their exact purpose and second-byte meaning are unknown.

| Table | ROM range | Entries (hex) |
|---|---|---|
| Lookup at `0x17165` | `0x17165-0x17174` | `00/01 02/04 05/06 08/09 0B/0D 0E/10 11/12 14/16` |
| Lookup at `0x17175` | `0x17175-0x17184` | `00/01 03/04 05/07 08/0A 0C/0D 0E/0F 11/13 15/16` |

Fixed-bank code at `0x009DE-0x009EB` loads one of these table bases (`0x7165` or `0x7175` in the banked CPU window), indexes with the value previously read from `0xC3F8` using a two-byte stride, reads the first byte of that pair, and stores it at `0xC340`. The branch selecting the base depends on flags from a preceding helper call operating on work RAM at `0xC405`; the meaning of that state and what the lookup result represents have not been identified. The banked CPU addresses correspond to the listed physical offsets when ROM bank 5 is selected.

**Boundary and interpretation caveats:**

- Lane's shield annotation ends with `0F 31`, but the 16 cost/unknown pairs starting at `0x15719` end with `28 BE` at `0x15738`; `0F 31` occurs at `0x1573D-0x1573E`. The range above follows the 16-entry layout, not the annotated end marker.
- The Lane healing annotation ends `32 32 14`, and the Kyle A annotation begins `[ 32 ...`. With 16 three-byte healing records, Lane ends in `14 32 32` at `0x157C6-0x157C8`, and 16 Kyle A values then start with `33` at `0x157C9`. The annotated `32` would overlap Lane's final record. The boundary needs code or gameplay verification.
- Kyle's play-dead annotation identifies cost but does not explain the second byte. Lane and Lora healing notes use “damage” for the repeated values; whether those fields mean healing, damage, or another quantity is not established.
- Range, cost, damage, and level assignments are the notes' interpretations. No consuming routine has been traced yet, and the unknown bytes and any scaling or special values remain to be verified.

## Player name encoding

The notes identify `B000-B007` as the player name, using two bytes per character. The example's bytes at save offsets `0x0B0-0x0B7` match the `RY__` example below.

| Bytes | Character | Bytes | Character | Bytes | Character | Bytes | Character |
|---|---|---|---|---|---|---|---|
| `FA F0` | A | `FB F0` | B | `FC F0` | C | `FD F0` | D |
| `FE F0` | E | `FF F0` | F | `F0 F1` | G | `F1 F1` | H |
| `F2 F1` | I | `F3 F1` | J | `F4 F1` | K | `F5 F1` | L |
| `F6 F1` | M | `F7 F1` | N | `F8 F1` | O | `F9 F1` | P |

| `F2 F2` | Y | `F3 F2` | Z | `F4 F2` | `'` | `F7 F2` | `.` |
| `F9 F2` | `?` | `FA F2` | `!` | `FB F2` | `-` | `FE FF` | Null or space |

Examples from the notes:

| Name | Bytes |
|---|---|
| `RY__` | `FB F1 F2 F2 FE FF FE FF` |
| `TYAN` | `FD F1 F2 F2 FA F0 F7 F1` |
| `?!'.` | `F9 F2 FA F2 F4 F2 F7 F2` |

The first example uses underscores to show the two blank positions. Whether `FE FF` is a space, a null, or both is unresolved.

#!/usr/bin/env bash
set -euo pipefail

ROM_NAME="Rolan's Curse II (USA).gb"

rc2t_print_help_message() {
    echo ""
    echo "Usage: $(basename "$0") <command>"
    echo ""
    echo "Commands:"
    echo "  dump-hex        Dump the contents in hexadecimal"
    echo "  check-hash      Check the hash"
    echo "  extract-json    [rom-or-hex]  Extract character stats into output/<rom-sha256>.json"
    echo "  render-graphics [start [end output]] [options]  Render ROM tiles (default: entire ROM)"
}

_check_if_rom_exists() {
    if [ ! -f input/"$ROM_NAME" ]; then
        echo "Error: ROM file '$ROM_NAME' not found."
        exit 1
    fi
}

_get_hash() {
    sha256sum "input/$ROM_NAME" | awk '{print $1}'
}

##### MAIN #####

cd "$(dirname "$0")"

if [ $# -eq 0 ] || [ -z "$1" ]; then
    rc2t_print_help_message
    exit 1
fi

case "$1" in
    check-hash)
        _check_if_rom_exists
        _get_hash
        ;;

    dump-hex)
        _check_if_rom_exists
        xxd "input/$ROM_NAME" > "output/$ROM_NAME.hex"
        ;;

    extract-json)
        if [ "$#" -gt 2 ]; then
            echo "Error: extract-json accepts at most one input path." >&2
            exit 1
        fi
        python3 bin/extract_json.py "${@:2}"
        ;;

    render-graphics)
        if [ "$#" -eq 1 ]; then
            python3 bin/render_graphics.py 0 0x20000 output/graphics.png
        elif [ "$#" -eq 2 ]; then
            python3 bin/render_graphics.py "$2" 0x20000 output/graphics.png
        elif [[ "$3" == --* ]]; then
            python3 bin/render_graphics.py "$2" 0x20000 output/graphics.png "${@:3}"
        else
            python3 bin/render_graphics.py "${@:2}"
        fi
        ;;

    *)
        rc2t_print_help_message
        ;;
esac

def hex_dump(data, powershell=False):
    rows = []
    for offset in range(0, len(data), 16):
        chunk = data[offset:offset + 16]
        if powershell:
            rows.append(f"{offset:08X}   {chunk.hex(' ').upper()}  ................")
        else:
            groups = " ".join(chunk[i:i + 2].hex() for i in range(0, 16, 2))
            rows.append(f"{offset:08x}: {groups}  ................")
    text = "\n".join(rows) + "\n"
    return text.encode("utf-16" if powershell else "utf-8")


class ExtractGameTests(unittest.TestCase):
    def setUp(self):
        self.rom = bytearray((i * 37 + i // 256) % 256 for i in range(0x20000))

    def test_hex_dump_formats_and_invalid_rows(self):
        for powershell in (False, True):
